tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnChangeFaction = function(self)
    ChangeFaction:ApplyChangeFaction(self.nCurFactionId, nil, self.nSex)
    self:Close()
  end
}
tbWnd.tbOnToggleClick = {}

function tbWnd:OnCreate()
  self.tbButton = Ui:NewButtonGroup(self, tbButtonGroup, true, function(self, nIdx)
    if self.nCurFactionId == nIdx then
      return
    end
    self.nCurFactionId = nIdx
    self:SelectFaction()
  end)
  for nFaction, szPath in pairs(tbButtonGroup) do
    self.pPanel:SetActive(szPath, Faction:IsFactionOpen(nFaction))
  end
end

function tbWnd:OnClose()
  self.nCurFactionId = nil
end

function tbWnd:OnOpen()
  self.tbButton:Select(me.nFaction)
end

function tbWnd:SelectFaction()
  local bOpenMale = Faction:IsOpenSex(self.nCurFactionId, Npc.SexDef.MALE)
  local bOpenFemale = Faction:IsOpenSex(self.nCurFactionId, Npc.SexDef.FEMALE)
  if bOpenMale and bOpenFemale then
    bOpenMale = 0 == me.nSex
    bOpenFemale = 1 == me.nSex
  end
  self.pPanel:Label_SetText(tbControls.txtFactionDesc, Faction:GetDesc(self.nCurFactionId))
  self.pPanel:Sprite_SetSprite(tbControls.imgFactionInfo, Faction:GetRadarChartPath(self.nCurFactionId))
  self.nSex = 0
  if bOpenFemale then
    self.nSex = 1
  end
  local szName = Faction:GetName(self.nCurFactionId)
  local nSeries = Faction:GetSeries(self.nCurFactionId)
  for nKey, tb in pairs(tbSerisControls) do
    if nKey == nSeries then
      self.pPanel:Label_SetText(tb[2], szName)
      self.pPanel:SetActive(tb[1], true)
    else
      self.pPanel:SetActive(tb[1], false)
    end
  end
end
