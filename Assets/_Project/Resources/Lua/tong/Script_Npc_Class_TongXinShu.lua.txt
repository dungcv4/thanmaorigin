Require("Setting/Npc/Ai/SeventhEvening/TongXinShuCfg.lua")
local tbNpcClient = Npc.TongXinShu
local tbCfg = Npc.TongXinShu.Cfg

function tbNpcClient:OpenShopWindow()
  local function fnBuy()
    local tbWare = Shop:GetAWare(tbCfg.nSpecialWaterWareId)
    
    local nRemainCount = Shop:GetShopRemainCount(tbWare, me)
    if 0 == nRemainCount then
      me.CenterMsg(i18n.Get(Shop.Terms.InsufficientRemainInventory))
      return
    end
    Ui:OpenWindow("UIShopBuy", tbWare, true)
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\130\168\232\191\152\230\178\161\230\156\137\231\155\184\230\128\157\230\176\180\239\188\140\230\152\175\229\144\166\229\137\141\229\190\128\229\149\134\229\186\151\232\180\173\228\185\176\239\188\159"), {
    {fnBuy},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  }, nil, nil, nil)
end
