local tbWnd = Ui:CreateClass("UIMultiFactionFightResult", "UIPanel")
local tbControls = {
  txtNewKingTitle = "txtNewKingTitle",
  TemplateDefault = "ScrollView/List/TemplateDefault"
}
tbWnd.tbControls = tbControls
local ITEM_GRID_COUNT = 4

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.TemplateDefault)
end

function tbWnd:OnOpen()
  self.pPanel:Label_SetText(tbControls.txtNewKingTitle, "")
  MultiFactionFight:OnRequestNewInfomationInfo()
end

function tbWnd:UpdateSV(nWinnerFaction, tbData)
  self.nWinnerFaction = nWinnerFaction
  self.tbData = tbData
  local szFactionName = Faction:GetName(nWinnerFaction)
  self.pPanel:Label_SetText(tbControls.txtNewKingTitle, i18n.EasyGet("\229\164\169\228\184\139\231\172\172\228\184\128\233\151\168\230\180\190\226\128\148\226\128\148{0}", szFactionName))
  local nRow = math.ceil(#tbData / ITEM_GRID_COUNT)
  self.tbScrollView:Reset()
  for i = 1, nRow do
    self.tbScrollView:SetElement(i, function(tbElement)
      self:_fnSetPlayer(tbElement, i)
    end)
  end
end

function tbWnd:_fnSetPlayer(pItem, nIdx)
  for i = 1, ITEM_GRID_COUNT do
    local nDataIdx = (nIdx - 1) * ITEM_GRID_COUNT + i
    local szItemKey = string.format("FactionOtherObj/FactionInfo%d", i)
    pItem.pPanel:SetActive(szItemKey, true)
    if self.tbData[nDataIdx] then
      local szFaction = Faction:GetName(self.nWinnerFaction)
      local spFaction = Faction:GetIcon(self.nWinnerFaction)
      local spRoleHead = PlayerPortrait:GetSprite(self.tbData[nDataIdx].nPortrait)
      pItem.pPanel:Label_SetText(szItemKey .. "/Faction/txtFaction", szFaction)
      pItem.pPanel:Sprite_SetSprite(szItemKey .. "/PortraitObj/imgFactionIcon", spFaction)
      pItem.pPanel:Label_SetText(szItemKey .. "/txtFightPowerValue", self.tbData[nDataIdx].nFightPower)
      pItem.pPanel:Label_SetText(szItemKey .. "/txtPlayerName", self.tbData[nDataIdx].szName)
      pItem.pPanel:Label_SetText(szItemKey .. "/txtServer", i18n.EasyGet("\229\174\182\230\151\143"))
      pItem.pPanel:Label_SetText(szItemKey .. "/txtServerName", self.tbData[nDataIdx].szKinName)
      pItem.pPanel:Label_SetText(szItemKey .. "/PortraitObj/Level/Text", self.tbData[nDataIdx].nLevel)
      pItem.pPanel:Label_SetText(szItemKey .. "/PortraitObj/Level/TextMaster", self.tbData[nDataIdx].nLevel - GameSetting.MAX_PLAYER_LEVEL)
      pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Level/Text", self.tbData[nDataIdx].nLevel < GameSetting.MAX_PLAYER_LEVEL)
      pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Level/TextMaster", self.tbData[nDataIdx].nLevel >= GameSetting.MAX_PLAYER_LEVEL)
      pItem.pPanel:Sprite_SetSprite(szItemKey .. "/PortraitObj/Portrait/Image", spRoleHead)
      pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Portrait/Image", true)
    else
      pItem.pPanel:SetActive(szItemKey, false)
    end
  end
end

function tbWnd:OnClose()
end
