tbWnd.tbMemberSortFunc = {
  Default = function(tbMemberA, tbMemberB)
    if tbMemberA.nMemberId == me.dwID then
      return true
    end
    if tbMemberB.nMemberId == me.dwID then
      return false
    end
    if tbMemberA.nIsOnLine == tbMemberB.nIsOnLine then
      return tbMemberA.nCareer < tbMemberB.nCareer
    else
      return 1 == tbMemberA.nIsOnLine
    end
  end,
  Faction = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nFaction ~= tbMemberB.nFaction then
        return tbMemberA.nFaction > tbMemberB.nFaction
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nFaction ~= tbMemberB.nFaction then
        return tbMemberA.nFaction < tbMemberB.nFaction
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Rank = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nHonorStarLevel ~= tbMemberB.nHonorStarLevel then
        return tbMemberA.nHonorStarLevel > tbMemberB.nHonorStarLevel
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nHonorStarLevel ~= tbMemberB.nHonorStarLevel then
        return tbMemberA.nHonorStarLevel < tbMemberB.nHonorStarLevel
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Name = function(tbMemberA, tbMemberB)
    local aString = tbMemberA.szName or ""
    local bString = tbMemberB.szName or ""
    if tbWnd.bIsDown == true then
      if aString ~= bString then
        return aString > bString
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if aString ~= bString then
        return aString < bString
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Level = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nTotalLevel ~= tbMemberB.nTotalLevel then
        return tbMemberA.nTotalLevel > tbMemberB.nTotalLevel
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nTotalLevel ~= tbMemberB.nTotalLevel then
        return tbMemberA.nTotalLevel < tbMemberB.nTotalLevel
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Career = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nCareer ~= tbMemberB.nCareer then
        return tbMemberA.nCareer > tbMemberB.nCareer
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nCareer ~= tbMemberB.nCareer then
        return tbMemberA.nCareer < tbMemberB.nCareer
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Stock = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nStock ~= tbMemberB.nStock then
        return tbMemberA.nStock > tbMemberB.nStock
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nStock ~= tbMemberB.nStock then
        return tbMemberA.nStock < tbMemberB.nStock
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Active = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nWeekActive ~= tbMemberB.nWeekActive then
        return tbMemberA.nWeekActive > tbMemberB.nWeekActive
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nWeekActive ~= tbMemberB.nWeekActive then
        return tbMemberA.nWeekActive < tbMemberB.nWeekActive
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  ChuanGong = function(tbMemberA, tbMemberB)
    local function _IsCanChuangong(tbMemberData)
      local tbChuanGong = tbMemberData.tbChuanGong
      
      if tbChuanGong and 1 == tbMemberData.nIsOnLine then
        local tbStatus = ChuanGong:CheckStatusFromOtherData(tbMemberData.nTotalLevel, tbMemberData.nVipLevel, tbChuanGong.nRecieveTimes > 0, tbChuanGong.nSendTimes > 0, tbChuanGong.nLastSendTime)
        return ChuanGong:StatusCanSendOrGet(tbStatus)
      end
      return false
    end
    
    local function _IsCanChuangongCdOK(tbMemberData)
      local tbChuanGong = tbMemberData.tbChuanGong
      if tbChuanGong and 1 == tbMemberData.nIsOnLine then
        local tbStatus = ChuanGong:CheckStatusFromOtherData(tbMemberData.nTotalLevel, tbMemberData.nVipLevel, tbChuanGong.nRecieveTimes > 0, tbChuanGong.nSendTimes > 0, tbChuanGong.nLastSendTime)
        return ChuanGong:IsCdSuffice(tbStatus, ChuanGong.tbRelationType.Recieve) or ChuanGong:IsCdSuffice(tbStatus, ChuanGong.tbRelationType.Send)
      end
      return false
    end
    
    local aCanChuanGong = _IsCanChuangong(tbMemberA) and tbMemberA.nMemberId ~= me.dwID and 1 or 0
    local bCanChuanGong = _IsCanChuangong(tbMemberB) and tbMemberB.nMemberId ~= me.dwID and 1 or 0
    if aCanChuanGong ~= bCanChuanGong then
      return aCanChuanGong > bCanChuanGong
    end
    local aCDOK = _IsCanChuangongCdOK(tbMemberA) and tbMemberA.nMemberId ~= me.dwID and 1 or 0
    local bCDOK = _IsCanChuangongCdOK(tbMemberB) and tbMemberB.nMemberId ~= me.dwID and 1 or 0
    if aCDOK ~= bCDOK then
      return aCDOK > bCDOK
    end
    if tbWnd.bIsDown == true then
      if tbMemberA.nTotalLevel ~= tbMemberB.nTotalLevel then
        return tbMemberA.nTotalLevel > tbMemberB.nTotalLevel
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nTotalLevel ~= tbMemberB.nTotalLevel then
        return tbMemberA.nTotalLevel < tbMemberB.nTotalLevel
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end,
  Offline = function(tbMemberA, tbMemberB)
    if tbWnd.bIsDown == true then
      if tbMemberA.nIsOnLine ~= tbMemberB.nIsOnLine then
        return tbMemberA.nIsOnLine > tbMemberB.nIsOnLine
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    else
      if tbMemberA.nIsOnLine ~= tbMemberB.nIsOnLine then
        return tbMemberA.nIsOnLine < tbMemberB.nIsOnLine
      end
      return tbMemberA.nMemberId < tbMemberB.nMemberId
    end
  end
}

function tbWnd:OnSwitchPage(nPage)
  self.nCurPage = nPage
  local tbPagePanel = {
    [self.PAGE_DEF.Info] = self.panelTongInfo,
    [self.PAGE_DEF.Member] = self.panelTongMember,
    [self.PAGE_DEF.Domain] = self.panelTongActivity,
    [self.PAGE_DEF.RedBag] = self.panelTongRedBag,
    [self.PAGE_DEF.Resource] = self.panelTongResource
  }
  for i = 1, #tbPagePanel do
    if i == nPage then
      self.pPanel:SetActive(tbPagePanel[i], true)
    else
      self.pPanel:SetActive(tbPagePanel[i], false)
    end
  end
  self.tbTongResourcePanel:OnDeactivate()
  if nPage == self.PAGE_DEF.Info then
    self:OnUpdateBaseTongInfo()
    self:OnUpdateSimpleMemberList()
    Tong:UpdateBaseInfo()
    Tong:UpdateMemberList()
    local pAsyncRole = KPlayer.GetAsyncData(me.dwID)
    if not pAsyncRole then
      RemoteServer.ViewPlayerInfo(me.dwID, nil)
    end
    self.tbWndDomainStat:Hide()
    self.tbTongRedBagPanel:OnDeactivate()
  elseif nPage == self.PAGE_DEF.Member then
    Tong:UpdateBaseInfo()
    self:OnUpdateMemberList()
    Tong:UpdateMemberList()
    self.tbWndDomainStat:Hide()
    self.tbTongRedBagPanel:OnDeactivate()
  elseif nPage == self.PAGE_DEF.Domain then
    DomainBattle:QueryBaseReq()
    self:OnUpdateDomainInfo()
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.DomainBattleOpen)
    self.tbTongRedBagPanel:OnDeactivate()
  elseif nPage == self.PAGE_DEF.RedBag then
    self.tbWndDomainStat:Hide()
    self.tbTongRedBagPanel:OnActivate()
  elseif nPage == self.PAGE_DEF.Resource then
    self.tbWndDomainStat:Hide()
    self.tbTongResourcePanel:OnActivate(self.panelResourceDetails)
  end
end

function tbWnd:OnUpdateMemberList(szSortType)
  szSortType = szSortType or "Default"
  if szSortType then
    if self.szSortType == szSortType then
      self.bIsDown = not self.bIsDown
    else
      self.szSortType = szSortType
      self.bIsDown = true
    end
  end
  local tbMemberList = Tong:GetMemberList() or {}
  local tbOnlineMembers = Tong:GetMemberState() or {}
  table.sort(tbMemberList, self.tbMemberSortFunc[szSortType])
  local tbTongBaseInfo = Tong:GetBaseInfo() or {}
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local tbMember = tbMemberList[nIndex]
    local tbElem = tbLoopScroll:GetElement(nIndex)
    tbElem.tbMember = tbMember
    local pPanel = tbElem.pPanel
    local bOnLine = tbOnlineMembers[tbMember.nMemberId]
    self:BindPopMenuEvent(tbMember, pPanel)
    pPanel:Sprite_SetSprite("imgFaction", Faction:GetIcon(tbMember.nFaction))
    pPanel:Label_SetText("txtName", tbMember.szName)
    local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(tbMember.nHonorStarLevel, false)
    local obj = pPanel:GetObject("HonorTitle")
    local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
    pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
    Ui:SetVipLevelEffect(pPanel, tbMember.nVipLevel, "txtName")
    pPanel:Label_SetText("txtLevel", Lib:GetLevelShortText(tbMember.nTotalLevel))
    local szCareer = i18n.Get(Tong.Def.CareerName[tbMember.nCareer])
    if tbMember.nCareer == Tong.Def.Career_Leader and tbTongBaseInfo.nMasterId == tbMember.nMemberId then
      szCareer = szCareer .. "/" .. i18n.Get(Tong.Def.CareerName[Tong.Def.Career_Master])
    end
    pPanel:Label_SetText("txtJob", szCareer)
    pPanel:Label_SetText("txtStock", tbMember.nStock)
    pPanel:Label_SetText("txtActivity", tbMember.nWeekActive or 0)
    if bOnLine then
      pPanel:Label_SetText("txtOffline", i18n.Get(Tong.Cfg.Msg.Online))
    else
      pPanel:Label_SetText("txtOffline", string.format("<color=#352620>%s</color>", i18n.Get(Tong.Cfg.Msg.Offline)))
    end
    local tbChuanGong = tbMember.tbChuanGong
    local bChuanGongButtonActive = false
    if tbChuanGong and 1 == tbMember.nIsOnLine and me.dwID ~= tbMember.nMemberId then
      local tbStatus, szGetErr, szSendErr = ChuanGong:CheckStatusFromOtherData(tbMember.nTotalLevel, tbMember.nVipLevel, tbChuanGong.nRecieveTimes > 0, tbChuanGong.nSendTimes > 0, tbChuanGong.nLastSendTime)
      
      local function fnChuanGong(emRelation)
        if ChuanGong:IsCdSuffice(tbStatus, emRelation) then
          ChuanGong:RequestChuanGongC_Kin(tbMember.nMemberId, tbMember.nTotalLevel, tbMember.nVipLevel, {"UITong"}, emRelation)
        else
          local nNeedChuanGongAgainTime = ChuanGong:GetNeedJudgeCdLastSendTime(tbStatus, emRelation) + ChuanGong:GetSendCD(tbMember.nVipLevel) - GetTime()
          me.CenterMsg(i18n.Get(ChuanGong.tbRelationTypeToText.tbNeedAfter_0_MinTips[emRelation], math.ceil(nNeedChuanGongAgainTime / 60)))
        end
      end
      
      if tbStatus.bLevelAndTimesSufficeGet then
        Ui:Button_BindEvent(pPanel, "btnChuanGong", fnChuanGong, ChuanGong.tbRelationType.Recieve)
        if ChuanGong:IsCdSuffice(tbStatus, ChuanGong.tbRelationType.Recieve) then
          pPanel:Sprite_SetSprite("btnChuanGong", ChuanGong.Cfg.szRecieveIconPath)
        else
          pPanel:Sprite_SetSprite("btnChuanGong", ChuanGong.Cfg.szDisableRecieveIconPath)
        end
        bChuanGongButtonActive = true
      else
        if tbStatus.bLevelAndTimesSufficeSend then
          Ui:Button_BindEvent(pPanel, "btnChuanGong", fnChuanGong, ChuanGong.tbRelationType.Send)
          if ChuanGong:IsCdSuffice(tbStatus, ChuanGong.tbRelationType.Send) then
            pPanel:Sprite_SetSprite("btnChuanGong", ChuanGong.Cfg.szSendIconPath)
          else
            pPanel:Sprite_SetSprite("btnChuanGong", ChuanGong.Cfg.szDisableSendIconPath)
          end
          bChuanGongButtonActive = true
        else
        end
      end
    end
    pPanel:SetActive("btnChuanGong", bChuanGongButtonActive)
  end
  
  local nCount = Lib:CountTB(tbMemberList)
  self.tbTongMemberScrollView:SetElement(nCount, fnSetItem)
end

function tbWnd:OnSliderList(vec2)
  if vec2.y <= 0 or vec2.y >= 1 then
    return
  end
  if self.bButtonChange then
    self.bButtonChange = false
    self.nSliderLimit = nDefaultSlideLimit
    return
  end
  local y = math.floor(vec2.y * 10) / 10
  local nDiff = math.abs(y - self.nCurY)
  if 0.9 == y or 0 == y then
    self.pPanel:SetActive(self.btnGoTop, false)
    self.pPanel:SetActive(self.btnGoBottom, false)
    return
  end
  if nDiff < self.nSliderLimit then
    return
  end
  if y < self.nCurY then
    self.pPanel:SetActive(self.btnGoTop, false)
    self.pPanel:SetActive(self.btnGoBottom, true)
  elseif y > self.nCurY then
    self.pPanel:SetActive(self.btnGoBottom, false)
    self.pPanel:SetActive(self.btnGoTop, true)
  end
  self.nCurY = y
  self.nSliderLimit = nSwitchBtnLimit
end
