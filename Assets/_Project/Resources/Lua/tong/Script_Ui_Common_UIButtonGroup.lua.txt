local tbWnd = {}
Ui:SetClass("UIButtonGroup", tbWnd)

function Ui:NewButtonGroup(tbOwner, ...)
  local tbClass = Ui:NewClass(Ui:GetClass("UIButtonGroup"))
  tbClass.tbOwner = tbOwner
  tbClass:OnCreate(...)
  return tbClass
end

function tbWnd:OnCreate(tbBtnKeyList, bHighlight, func)
  if self.tbBtnKeyList then
    for k, v in pairs(self.tbBtnKeyList) do
      Ui:Button_UnBindEvent(self.tbOwner.pPanel, v)
    end
    self.tbBtnKeyList = nil
  end
  self.tbBtnKeyList = tbBtnKeyList
  self.bHighlight = bHighlight
  self.funcCallback = func
  for k, v in pairs(self.tbBtnKeyList) do
    Ui:Button_BindEvent(self.tbOwner.pPanel, v, self.OnClickElement, self, k)
  end
  if not self.tbOwner._tbBtnGroup then
    self.tbOwner._tbBtnGroup = {}
  end
  self.tbOwner._tbBtnGroup[self] = 1
end

function tbWnd:OnDestroy()
  if self.tbBtnKeyList then
    for k, v in pairs(self.tbBtnKeyList) do
      Ui:Button_UnBindEvent(self.tbOwner.pPanel, v)
    end
    self.tbBtnKeyList = nil
  end
end

function tbWnd:UpdateBtnKeyList(tbBtnKeyList)
  for k, v in pairs(self.tbBtnKeyList) do
    Ui:Button_UnBindEvent(self.tbOwner.pPanel, v)
  end
  self.tbBtnKeyList = tbBtnKeyList
  for k, v in pairs(self.tbBtnKeyList) do
    Ui:Button_BindEvent(self.tbOwner.pPanel, v, self.OnClickElement, self, k)
  end
end

function tbWnd:Select(key)
  self.OnClickElement(self, key)
end

function tbWnd:OnClickElement(key)
  self.key = key
  if self.bHighlight then
    for k, v in pairs(self.tbBtnKeyList) do
      self.tbOwner.pPanel:Button_SetEnable(v, k ~= key)
      local btn = self.tbOwner.pPanel:GetObject(v)
      local trans = btn:GetComponent(Ui.BGroupTrans)
      if trans then
        if k == key then
          trans:OnSelected()
        else
          trans:OnUnSelected()
        end
      end
    end
  end
  if self.funcCallback then
    self.funcCallback(self.tbOwner, key)
  end
end

function tbWnd:SetActiveAll(bActive)
  for k, v in pairs(self.tbBtnKeyList) do
    self.tbOwner.pPanel:SetActive(v, bActive)
  end
end

function tbWnd:SetActive(key, bActive)
  self.tbOwner.pPanel:SetActive(self.tbBtnKeyList[key], bActive)
end

function tbWnd:SetButtonText(key, szText)
  self.tbOwner.pPanel:Button_SetText(self.tbBtnKeyList[key], szText, "Text")
end

function tbWnd:LightRedPoint(key)
  local szBtnKey = self.tbBtnKeyList[key]
  RedPointMgr:ForceLightRedPoint(self.tbOwner.pPanel, szBtnKey)
end

function tbWnd:FadeRedPoint(key)
  local szBtnKey = self.tbBtnKeyList[key]
  RedPointMgr:ForceFadeRedPoint(self.tbOwner.pPanel, szBtnKey)
end

function tbWnd:GetKey()
  return self.key
end

function tbWnd:GetElementNum()
  return #self.tbBtnKeyList
end

function tbWnd:GetButtonKey(nIndex)
  return self.tbBtnKeyList[nIndex]
end

function tbWnd:ClearSelected()
  if not self:GetKey() then
    return
  end
  if self.key and self.tbBtnKeyList[self.key] and not self.bNoHighlight then
    self.tbOwner.pPanel:Button_SetEnable(self.tbBtnKeyList[self.key], true)
    local btn = self.tbOwner.pPanel:GetObject(self.tbBtnKeyList[self.key])
    local trans = btn:GetComponent(Ui.BGroupTrans)
    if trans then
      trans:OnUnSelected()
    end
  end
  self.key = nil
end
