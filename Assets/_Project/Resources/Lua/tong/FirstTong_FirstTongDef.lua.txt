function FirstTong:LoadFinalMatchSetting()
  self.tbFinalMatchSetting = Lib:EasyLoadTabFile("Setting/FirstTong/FirstTongMatch.tab", {
    "nDate",
    "nTodayIndex"
  })
  for nIndex, tbSetting in pairs(self.tbFinalMatchSetting) do
    for nMatchIndex, tbIndexSetting in ipairs(tbSetting) do
      local tbData = tbIndexSetting.tbszMatchList
      if tbData then
        for nIndex = #tbData, 1, -1 do
          if "" ~= tbData[nIndex] then
            tbData[nIndex] = Lib:SplitStr2Num(tbData[nIndex])
          else
            table.remove(tbData, nIndex)
          end
        end
      end
    end
  end
end

function FirstTong:LoadAwardSetting()
  self.tbTongRankAward = Lib:EasyLoadTabFile("Setting/FirstTong/TongRankAward.tab")
end

FirstTong:LoadFinalMatchSetting()
FirstTong:LoadAwardSetting()
