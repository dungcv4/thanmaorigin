local tbWnd = Ui:CreateWindow("UITongSetAnnounce")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSaveChange = "node/imgBG/Content/btnSaveChange",
  inputTongNotice = "node/imgBG/Content/TongNotice/inputTongNotice",
  txtTitle = "node/imgBG/txtTitle",
  PlaceholderText = "node/imgBG/Content/TongNotice/inputTongNotice/Placeholder"
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_TONG_DECLARATION,
      self.UpdateDeclaration
    },
    {
      EventNotify.emNOTIFY_SYNC_TONG_QUIT,
      self.Close
    }
  }
end

tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSaveChange = function(self)
    local szMsg = self.pPanel:Input_GetText(self.inputTongNotice)
    if self.bDeclaration then
      Tong:ModifyDeclaration(szMsg)
    else
      Tong:ModifyNotice(szMsg)
    end
    self:Close()
  end
}

function tbWnd:OnOpen(szTitle, bDeclaration)
  self.bDeclaration = bDeclaration
  self.pPanel:Label_SetText(self.txtTitle, szTitle)
  self.pPanel:Label_SetText(self.PlaceholderText, i18n.EasyGet("\230\172\162\232\191\142\229\138\160\229\133\165\230\136\145\228\187\172\231\154\132\229\184\174\228\188\154\239\188\129({0}\229\173\151\228\187\165\229\134\133)", Tong.Cfg.nMaxNoticeLength))
  if self.bDeclaration then
    self:UpdateDeclaration()
    Tong:UpdateDeclaration()
  else
    self.pPanel:Input_SetText(self.inputTongNotice, "")
  end
end

function tbWnd:UpdateDeclaration()
  if not self.bDeclaration then
    return
  end
  local szDeclaration = Tong:GetDeclaration()
  self.pPanel:Input_SetText(self.inputTongNotice, szDeclaration)
end
