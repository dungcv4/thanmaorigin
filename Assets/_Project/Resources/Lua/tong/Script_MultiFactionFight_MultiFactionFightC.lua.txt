function MultiFactionFight:TryGetRankData(szKey, nPage, bRequest)
  local nNow = GetTime()
  self.nLastRequestTime = self.nLastRequestTime or 0
  local bNeedRequest = false
  if nNow - self.nLastRequestTime > self.nSynRankDataCD then
    bNeedRequest = true
    self.nLastRequestTime = nNow
  end
  if not self.tbRankDataCache[szKey] or not self.tbRankDataCache[szKey][nPage] then
    bNeedRequest = true
    self.tbRankDataCache[szKey] = self.tbRankDataCache[szKey] or {}
    self.tbRankDataCache[szKey][nPage] = {}
  end
  bNeedRequest = bNeedRequest and bRequest
  if bNeedRequest then
    self:RequestRankData(szKey, nPage)
  end
  return self.tbRankDataCache[szKey][nPage], self.tbMyData[szKey]
end

function MultiFactionFight:RequestRankData(szKey, nPage)
  if self.nRequestTimer then
    Timer:Close(self.nRequestTimer)
    self.nRequestTimer = nil
  end
  self.nRequestTimer = Timer:Register(Env.GAME_FPS / 2, self.RequestRankData_Helper, self, szKey, nPage)
end

function MultiFactionFight:RequestRankData_Helper(szKey, nPage)
  if Ui.bInZone then
    RemoteServer.RequestZMultiFactionFight("OnC2ZGetRankData", szKey, nPage)
  else
    RemoteServer.RequestMultiFactionFight("OnC2WGetRankData", szKey, nPage)
  end
end

function MultiFactionFight:OnSynRankData(szKey, nPage, tbRankData, tbMyData)
  local tbTemp = {}
  tbTemp.nMaxNum = tbRankData.nMaxNum
  if "FactionRank" == szKey then
    for nIndex, tbInfo in ipairs(tbRankData) do
      tbTemp[nIndex] = {}
      tbTemp[nIndex].nScore = tbInfo.nSortKey1
      tbTemp[nIndex].nFaction = tbInfo.nSortKey2
    end
  else
    for nIndex, tbInfo in ipairs(tbRankData) do
      tbTemp[nIndex] = {}
      tbTemp[nIndex].nScore = tbInfo.nSortKey1
      tbTemp[nIndex].nKillNums = tbInfo.nSortKey2
      tbTemp[nIndex].dwPlayerId = tbInfo.nSortKey3
      tbTemp[nIndex].nFaction = tbInfo.nFaction
      tbTemp[nIndex].szName = string.format("S%d-%s", tbInfo.nServerId, tbInfo.szName)
    end
  end
  self.tbRankDataCache[szKey] = self.tbRankDataCache[szKey] or {}
  self.tbRankDataCache[szKey][nPage] = tbTemp
  self.tbMyData[szKey] = tbMyData
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_WARSITUATION_UPDATE, szKey, nPage)
end

function MultiFactionFight:OnLogout()
  MultiFactionFight.tbRankDataCache = {}
  MultiFactionFight.tbMyData = {}
end

function RaceMap:ShowRaceMapUI(nGameState, szState, nTime, nMyScore, nRankInFaction, nFactionScore, nFactionRank, tbBuffPos)
  local tbTime = {}
  local tbContent = {}
  if nGameState == MultiFactionFight.GAME_STATE.QUALIFIER then
    tbTime, tbContent = self:GetQualifierUI(szState, nTime, nMyScore, nRankInFaction)
  end
  if nGameState == MultiFactionFight.GAME_STATE.DECIDER then
    tbTime, tbContent = self:GetDeciderUI(szState, nTime, nMyScore, nFactionScore, nFactionRank)
  end
  FubenUIController:ShowDefaultInfo({tbTime}, tbContent)
  if tbBuffPos then
    self:OnGetBuffPos(tbBuffPos)
  end
end
