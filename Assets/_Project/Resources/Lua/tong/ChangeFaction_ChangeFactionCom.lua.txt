ChangeFaction.IS_SYS_OPEN = true
ChangeFaction.ERR_INVALID_SECTION = 1
ChangeFaction.ERR_DUPLICATE_SECTION = 2
ChangeFaction.ERR_NO_MONEY = 3
ChangeFaction.ERR_COOLING_DOWN = 4
ChangeFaction.ERR_INVALID_PARAMETER = 5
ChangeFaction.ERR_INVALID_FACTION = 6
ChangeFaction.ERR_INVALID_SEX = 7
ChangeFaction.ERR_REPEAT_SECTION = 8
ChangeFaction.ERR_NOTIN_XISUIDAO = 9
ChangeFaction.ERR_MASK_SKILL_STATUS = 10
ChangeFaction.SAVE_GROUP = GetGroupKey("ChangeFaction")
ChangeFaction.SAVE_KEY_CD = 1

function ChangeFaction:CanChangeFactionSect(pPlayer, nFactionSect)
  if pPlayer.IsMaskPlayerAttr() or pPlayer.IsMaskPlayerFactSkill() then
    return false, self.ERR_MASK_SKILL_STATUS
  end
  if not nFactionSect then
    return false, self.ERR_INVALID_SECTION
  end
  if not Faction:IsFactionSectOpen(pPlayer.nFaction, nFactionSect) then
    return false, self.ERR_INVALID_SECTION
  end
  if pPlayer.nFactionSect == nFactionSect then
    return false, self.ERR_DUPLICATE_SECTION
  end
  local nLeftTime = pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_KEY_CD) - GetTime()
  if nLeftTime > 0 then
    return false, self.ERR_COOLING_DOWN, {nLeftTime}
  end
  if pPlayer.nLevel > self.DEFINE.FREE_LEVEL_LIMIT and pPlayer.GetMoney(self.DEFINE.COST_TYPE) < self.DEFINE.COST_NUM then
    return false, self.ERR_NO_MONEY
  end
  return true
end

function ChangeFaction:CanChangeFaction(pPlayer, nFaction, nFactionSect, nSex)
  if not pPlayer then
    LogErr("ChangeFaction Param Error!")
    return false, self.ERR_INVALID_PARAMETER
  end
  if pPlayer.IsMaskPlayerAttr() or pPlayer.IsMaskPlayerFactSkill() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\230\151\160\230\179\149\229\136\135\230\141\162")
  end
  if not nFaction or not nSex then
    LogErr("ChangeFaction Param Error!", pPlayer.dwID)
    return false, self.ERR_INVALID_PARAMETER
  end
  if not Faction:IsFactionSectOpen(nFaction, nFactionSect) then
    return false, self.ERR_INVALID_FACTION
  end
  if not Faction:IsOpenSex(nFaction, nSex) then
    return false, self.ERR_INVALID_SEX
  end
  if nFaction == pPlayer.nFaction and nFactionSect == pPlayer.nFactionSect then
    return false, self.ERR_REPEAT_SECTION
  end
  if pPlayer.nMapTemplateId ~= XiSuiDao.Def.nMapId then
    return false, self.ERR_NOTIN_XISUIDAO
  end
  return true
end
