function FirstTong:OnSyncQualification(tbQualification, tbPlayerScore)
  self.tbQualification = tbQualification
  self.tbPlayerScore = tbPlayerScore
  local tbMemberList = Tong:GetMemberList() or {}
  local tbOnlineMembers = Tong:GetMemberState() or {}
  local tbTongBaseInfo = Tong:GetBaseInfo() or {}
  local tbUIRankDataList = {}
  for nRank, tbMember in ipairs(tbMemberList) do
    local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(tbMember.nHonorStarLevel, false)
    local szCareer = i18n.Get(Tong.Def.CareerName[tbMember.nCareer])
    if tbMember.nCareer == Tong.Def.Career_Leader and tbTongBaseInfo.nMasterId == tbMember.nMemberId then
      szCareer = szCareer .. "/" .. i18n.Get(Tong.Def.CareerName[Tong.Def.Career_Master])
    end
    local bOnLine = tbOnlineMembers[tbMember.nMemberId]
    local szOnLine = ""
    if bOnLine then
      szOnLine = i18n.Get(Tong.Cfg.Msg.Online)
    else
      szOnLine = string.format("<color=#352620>%s</color>", i18n.Get(Tong.Cfg.Msg.Offline))
    end
    table.insert(tbUIRankDataList, {
      [1] = {
        nPlayerId = tbMember.nMemberId,
        szPlayerName = tbMember.szName,
        tbTitle = {
          szHonorName = i18n.Parse(szHonorName),
          szHonorColor = szHonorColor
        },
        nVipLevel = tbMember.nVipLevel
      },
      [2] = szCareer,
      [3] = tbPlayerScore[tbMember.nMemberId] or 0,
      [4] = szOnLine,
      [5] = tbQualification[tbMember.nMemberId] or false
    })
  end
  table.sort(tbUIRankDataList, function(a, b)
    return a[3] > b[3]
  end)
  Ui:OpenWindow("UIGangRanking", {
    FirstTong.RequestApplyQualification,
    FirstTong
  }, nil, {
    [1] = i18n.EasyGet("\229\144\141\229\173\151"),
    [2] = i18n.EasyGet("\232\129\140\228\189\141"),
    [3] = i18n.EasyGet("\231\167\175\229\136\134"),
    [4] = i18n.EasyGet("\231\166\187\231\186\191"),
    [5] = nil
  }, tbUIRankDataList, FirstTong.FinalCfg.nPlayerNum)
end

function FirstTong:RequestGuessMatch(nGuessMatch, nGuessTongID, nGuessNum)
  if not self.tbFinalsData[nGuessMatch] then
    me.CenterMsg(i18n.EasyGet("\231\171\158\231\140\156\229\156\186\230\172\161\228\184\141\229\173\152\229\156\168"))
    return
  end
  if -1 ~= self.tbFinalsData[nGuessMatch].nWinner then
    me.CenterMsg(i18n.EasyGet("\230\173\164\229\156\186\230\175\148\232\181\155\229\183\178\231\187\147\230\157\159"))
    return
  end
  if self.tbMyMatchGuessData[nGuessMatch] then
    me.CenterMsg(i18n.EasyGet("\228\184\139\230\179\168\229\144\142\228\184\141\229\143\175\229\134\141\232\191\155\232\161\140\228\191\174\230\148\185"))
    return
  end
  if nGuessNum > self.GuessCfg.nMatchGuessMax then
    me.CenterMsg(i18n.EasyGet("\230\175\143\229\156\186\230\175\148\232\181\155\230\156\128\229\164\154\228\184\139\230\179\168{0}\228\184\170", self.GuessCfg.nMatchGuessMax))
    return
  end
  local nItemID = self.GuessCfg.nMatchGuessID
  local szItemName = Item:GetName(nItemID, me)
  local nNum = me.GetItemCountByTypeInBag(nItemID)
  if nGuessNum > nNum then
    me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
    return
  end
  
  local function fnAgree()
    RemoteServer.RequestFirstTong("ApplyGuessMatch", nGuessMatch, nGuessTongID, nGuessNum)
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\232\175\165\229\156\186\228\189\160\229\143\170\232\131\189\231\171\158\231\140\156\228\184\128\230\172\161\232\142\183\232\131\156\232\128\133\239\188\140\231\161\174\232\174\164\229\144\142\229\176\134\228\184\141\232\131\189\230\155\180\230\148\185\239\188\140\228\189\160\230\152\175\229\144\166\231\161\174\232\174\164\239\188\159"), {
    {fnAgree},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function FirstTong:IsNeedFubenBtn()
  for _, tbInfo in pairs(FirstTong.MAP_FIGHT) do
    if tbInfo.nMapId == me.nMapTemplateId then
      if self.tbPillarState then
        local tbBtnList = {}
        
        local function fnGoto(nIndex)
          if self.tbPillarState[nIndex] then
            if self.tbPillarState[nIndex].nTongId == me.dwTongId or FirstTong:CheckGmWarch(me) then
              RemoteServer.RequestFirstTong("Transmit", nIndex)
            else
              me.CenterMsg(i18n.EasyGet("\229\141\160\233\162\134\233\190\153\230\159\177\229\144\142\230\150\185\229\143\175\228\188\160\233\128\129"))
            end
          end
        end
        
        for nIndex, tbData in ipairs(self.tbPillarState) do
          if tbData.nTongId == me.dwTongId or FirstTong:CheckGmWarch(me) then
            table.insert(tbBtnList, {
              szName = i18n.Parse(tbData.szName),
              fnOnClick = function()
                fnGoto(nIndex)
              end
            })
          end
        end
        if self.szActiveState == "Perparing" then
          me.CenterMsg(i18n.EasyGet("\230\175\148\232\181\155\229\176\154\230\156\170\229\188\128\229\167\139"))
        end
        if self.szActiveState == "Fighting" and #tbBtnList <= 0 then
          tbBtnList = nil
          me.CenterMsg(i18n.EasyGet("\229\141\160\233\162\134\233\190\153\230\159\177\229\144\142\230\150\185\229\143\175\228\188\160\233\128\129"))
        end
        return true, nil, tbBtnList
      end
      break
    end
  end
  return false
end
