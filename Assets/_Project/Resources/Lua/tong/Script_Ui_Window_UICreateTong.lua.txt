local tbWnd = Ui:CreateWindow("UICreateTong")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_PORTRAIT_FRAME_CHANGE,
      self.UpdateMember
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnOK = "node/imgBG/Content/btnOK",
  txtCostNum = "node/imgBG/Content/Cost/txtCostNum",
  inputTongNotice = "node/imgBG/Content/TongNotice/inputTongNotice",
  inputTongName = "node/imgBG/Content/TongName/inputTongName",
  MemberTemplate = "node/imgBG/Content/TongTeam/MemberList/Member"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    self:OnClickOK()
  end
}

function tbWnd:OnCreate()
  self.tbMemberScrollView = Ui:NewScrollView(self, self.MemberTemplate)
end

function tbWnd:SetMemberElement(tbMember, tbElement, bCaptain)
  local pPanel = tbElement.pPanel
  pPanel:Label_SetText("txtName", tbMember.szName)
  pPanel:Label_SetText("txtKinName", tbMember.szKinName)
  pPanel:Label_SetText("PortraitObj/Level/Text", tbMember.nLevel)
  pPanel:Sprite_SetSprite("PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbMember.nPortrait))
  pPanel:Sprite_SetSprite("PortraitObj/imgFactionIcon", Faction:GetIcon(tbMember.nFaction))
  pPanel:SetActive("imgLeaderFlag", bCaptain)
  PlayerPortraitFrame:SetPortraitFrame(pPanel, "PortraitObj/PortraitFrame", tbMember.nPortraitFrameID)
end

function tbWnd:OnOpen()
  self.pPanel:Input_SetText(self.inputTongName, "")
  self.pPanel:Input_SetText(self.inputTongNotice, "")
  self.pPanel:Label_SetText(self.txtCostNum, Tong.Cfg.nCreateTongCost)
  self:UpdateMember()
end

function tbWnd:UpdateMember()
  self.tbMemberScrollView:Clear()
  self.tbMemberScrollView:SetElement(1, function(tbElement)
    self:SetMemberElement(TeamMgr:GetMyTeamMemberData(), tbElement, true)
  end)
  local tbMembers = TeamMgr:GetMembers()
  for i = 1, #tbMembers do
    self.tbMemberScrollView:SetElement(i + 1, function(tbElement)
      self:SetMemberElement(tbMembers[i], tbElement, false)
    end)
  end
end

function tbWnd:OnClickOK()
  local szTongName = self.pPanel:Input_GetText(self.inputTongName)
  local szTongNotice = self.pPanel:Input_GetText(self.inputTongNotice)
  local nRet = Tong:ApplyCreate(szTongName, szTongNotice)
  if nRet then
    self:Close()
  end
end
