function Tong:GetErrorMsg(nErrorCode)
  if type(nErrorCode) == "table" then
    local szMsg = ErrorMsg[nErrorCode[1]]
    if szMsg then
      return i18n.Get(szMsg, table.unpack(nErrorCode, 2))
    end
  else
    local szMsg = ErrorMsg[nErrorCode]
    if szMsg then
      if type(szMsg) == "table" then
        return i18n.Get(table.unpack(szMsg))
      else
        return i18n.Get(szMsg)
      end
    end
  end
end

function Tong:CheckMyAuthority(authority)
  local nMyCareer = self:GetMyCareer()
  if not nMyCareer then
    return false
  end
  local tbBaseInfo = self:GetBaseInfo()
  if nMyCareer == Tong.Def.Career_Leader and Tong.Def.Authority[nMyCareer][authority] then
    return true
  end
  if tbBaseInfo.nMasterId == me.dwID then
    return Tong.Def.Authority[Tong.Def.Career_Master][authority]
  else
    local nKinCareer = Kin:GetMyCareer()
    if not nKinCareer then
      return false
    end
    local nCareer = Tong.Def.KinCareer2TongCareer[nKinCareer]
    if not nCareer then
      return false
    end
    return Tong.Def.Authority[nCareer][authority]
  end
end

function Tong:OnSyncMemberList(tbMemberList)
  local tbCacheMember, nVersion, tbCacheMap = Tong:GetMemberList()
  local tbOnlineMembers = self:GetMemberState() or {}
  tbCacheMember = tbCacheMember or {}
  nVersion = nVersion or 0
  tbCacheMap = tbCacheMap or {}
  for _, tbMember in pairs(tbMemberList) do
    if not tbMember.bDelete then
      self:InsertClientMember(tbCacheMember, tbCacheMap, tbMember)
      tbOnlineMembers[tbMember.nMemberId] = 1 == tbMember.nIsOnLine
    else
      self:DeleteClientMember(tbCacheMember, tbCacheMap, tbMember)
      tbOnlineMembers[tbMember.nMemberId] = nil
    end
  end
  Tong:CacheData("MemberList", tbCacheMember, nVersion)
  Tong:CacheData("MemberMap", tbCacheMap, 0)
  Tong:SetMemberState(tbOnlineMembers)
end

function Tong:OnSyncMemberListEnd(nSyncVersion)
  local tbCacheMember, _ = Tong:GetMemberList()
  Tong:CacheData("MemberList", tbCacheMember, nSyncVersion)
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_TONG_MEMBERLIST)
end

function Tong:InsertClientMember(tbCacheMember, tbCacheMap, tbMember)
  if tbCacheMap[tbMember.nMemberId] then
    for k, v in pairs(tbCacheMember) do
      if v.nMemberId == tbMember.nMemberId then
        table.remove(tbCacheMember, k)
        break
      end
    end
  end
  table.insert(tbCacheMember, tbMember)
  tbCacheMap[tbMember.nMemberId] = tbMember
end

function Tong:DeleteClientMember(tbCacheMember, tbCacheMap, tbMember)
  if tbCacheMap[tbMember.nMemberId] then
    for k, v in pairs(tbCacheMember) do
      if v.nMemberId == tbMember.nMemberId then
        table.remove(tbCacheMember, k)
        break
      end
    end
  end
  tbCacheMap[tbMember.nMemberId] = nil
end

function Tong:ChangeName(szTongName, bFree)
  if not Tong:HasTong() then
    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.HasNotKin))
    return false
  end
  if not self:CheckMyAuthority(Tong.Def.Authority_ChangeName) then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.NoAuthority))
    return false
  end
  if not szTongName or "" == szTongName then
    return false
  end
  szTongName = ChatMgr:Filter4CharString(szTongName)
  if string.find(szTongName, "%s") then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.CantContainSpace))
    return false
  end
  if not CheckNameAvailable(szTongName) then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.IllegalWorld))
    return false
  end
  local nNameLen = Lib:StrLen(szTongName)
  if nNameLen > Tong.Cfg.nMaxTongNameLength or nNameLen < Tong.Cfg.nMinTongNameLength then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.TongNameLengthLimit, Tong.Cfg.nMinTongNameLength, Tong.Cfg.nMaxTongNameLength))
    return false
  end
  if not bFree and me.GetMoney("YuanBao") < Tong.Cfg.nChangNameCost then
    me.CenterMsg(i18n.EasyGet("\229\133\131\229\174\157\228\184\141\232\182\179{0}", Tong.Cfg.nChangNameCost))
    return false
  end
  RemoteServer.OnTongRequest("ChangeName", szTongName)
  return true
end

function Tong:Approve(nId)
  local tbApprovals = Tong:GetData("Approvals")
  if not tbApprovals then
    return
  end
  for i = 1, #tbApprovals do
    if tbApprovals[i].nId == nId then
      table.remove(tbApprovals, i)
      break
    end
  end
  Tong:CacheData("Approvals", tbApprovals, nil)
  Tong:SetManagerRedPoint(#tbApprovals > 0)
  RemoteServer.OnTongRequest("Approve", nId)
  Tong:SyncApprovals()
end

function Tong:Reject(nId)
  local tbApprovals = Tong:GetData("Approvals")
  if not tbApprovals then
    return
  end
  for i = 1, #tbApprovals do
    if tbApprovals[i].nId == nId then
      table.remove(tbApprovals, i)
      break
    end
  end
  Tong:CacheData("Approvals", tbApprovals, nil)
  Tong:SetManagerRedPoint(#tbApprovals > 0)
  RemoteServer.OnTongRequest("Reject", nId)
end

function Tong:UpdateAppraiseRedPoint()
  local curLoginTime = GetTime()
  local preloginTime = LocalData:GetRoleNumber("TongAppraiseRedPoint", -1)
  local timeOffset = 14400
  local isDiffWeek = Lib:IsDiffWeek(timeOffset, curLoginTime, preloginTime)
  if true == isDiffWeek and Tong:GetAppraiseOpen() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.TongAppraise)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.TongAppraise)
  end
end

function Tong:CheckCanChatForbid(nMemberId)
  if nMemberId == me.dwID then
    return false
  end
  if not self:CheckMyAuthority(Tong.Def.Authority_ChatForbid) then
    return false
  end
  local tbData = self:GetMemberData(nMemberId)
  local nCareer = tbData and tbData.nCareer
  local nMyCareer = self:GetMyCareer()
  if not (nCareer and nMyCareer) or nCareer <= nMyCareer then
    return false
  end
  local tbChatForbid = self:GetChatForbid()
  if tbChatForbid and tbChatForbid[nMemberId] then
    return false
  end
  return true
end

function Tong:ChatForbid(nMemberId)
  if not self:CheckCanChatForbid(nMemberId) then
    return
  end
  RemoteServer.OnTongRequest("ChatForbid", nMemberId)
end

function Tong:CheckCanChatUnforbid(nMemberId)
  if nMemberId == me.dwID then
    return false
  end
  if not self:CheckMyAuthority(Tong.Def.Authority_ChatForbid) then
    return false
  end
  local tbData = self:GetMemberData(nMemberId)
  local nCareer = tbData and tbData.nCareer
  local nMyCareer = self:GetMyCareer()
  if not (nCareer and nMyCareer) or nCareer <= nMyCareer then
    return false
  end
  local tbChatForbid = self:GetChatForbid()
  if not tbChatForbid or not tbChatForbid[nMemberId] then
    return false
  end
  return true
end

function Tong:ChatUnforbid(nMemberId)
  if not self:CheckCanChatUnforbid(nMemberId) then
    return
  end
  RemoteServer.OnTongRequest("ChatUnforbid", nMemberId)
end
