function Tong:LoadActivitySetting()
  local tbSetting = Lib:LoadTabFileEx("Setting/Tong/TongActivity.tab", "dsdd", nil, {
    "nID",
    
    "szJudgement",
    "nWeekAvg",
    "nDismiss"
  }, nil, nil, false)
  table.sort(tbSetting or {}, function(tbA, tbB)
    return tbA.nWeekAvg < tbB.nWeekAvg
  end)
  tbSetting = SetReadOnly(tbSetting)
  self.tbActivitySettings = tbSetting
end

Tong:LoadActivitySetting()
