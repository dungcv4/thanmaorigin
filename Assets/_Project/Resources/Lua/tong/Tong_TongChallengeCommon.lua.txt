function TongChallenge:CanEnterTongChallenge(pPlayer)
  if pPlayer.nLevel < self.ChallengeCfg.nEnterLevel then
    return false, i18n.Get(self.ChallengeCfg.szLevelTip)
  end
  return ActivityCheck:CheckCanEnterActivityEx(pPlayer, nil, nil, true, false, false, nil, nil, false)
end

function TongChallenge:GetRandomNearBossPos()
  local nPosIdx = math.random(1, #TongChallenge.tbNearBossPos)
  return TongChallenge.tbNearBossPos[nPosIdx]
end
