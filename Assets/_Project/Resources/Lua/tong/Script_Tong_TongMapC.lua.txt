Require("Script/Fuben/FubenUIController.lua")

function TongMap:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    Tong.Def.nMainMapTemplateId
  }, self)
end

function TongMap:IsNeedExitFubenButton()
  return false
end

function TongMap:OnLeave()
  if Ui:WindowVisible("UIKinFlagDisplay") then
    Ui:CloseWindow("UIKinFlagDisplay")
  end
  if Ui:WindowVisible("UIQuickUseItemOB") then
    Ui:CloseWindow("UIQuickUseItemOB")
  end
end

function TongMap:IsInFubenMap()
  if TongChallenge.bInChallengeArea and TongChallenge.nActivityStatus and TongChallenge.nActivityStatus ~= TongChallenge.tbActivityStatus.nEnding then
    return true
  end
  return false
end

function TongMap:OnFubenMapLoaded(pPlayer)
  if TongChallenge:IsActivityOpen() then
    FubenUIController:OpenDefaultFuBenUI()
    if TongChallenge.nMultiple > 1 then
      TongChallenge:OpenKinBossLootScoreWnd()
    end
  end
end

function TongMap:IsShowBtnScore()
  local bRet, pPunc = TongChallenge:IsShowBtnScore()
  if bRet then
    return bRet, pPunc
  end
end

function TongMap:GetFubenBtnAImagePathAndText()
  if TongChallenge:IsCanPlunder() then
    return TongChallenge:GetFubenBtnAImagePathAndText()
  end
end

function TongMap:IsShowRightCenterTimer()
  if TongChallenge:IsInTongChallengePlaying() then
    return true
  end
  return false
end

TongMap:RegisterFubenUIChecker()

function TongMap:IsNeedFubenBtn()
  return TongChallenge:IsCanPlunder(), {
    TongChallenge.ShowKinBossLoot,
    TongChallenge
  }
end
