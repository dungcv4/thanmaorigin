local tbWnd = Ui:CreateWindow("UITongMail")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_TONG_MAIL_COUNT,
      self.Update,
      self
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  inputMail = "node/imgBG/Content/KinMail/inputMail",
  btnSend = "node/imgBG/Content/btnSend",
  txtTimesLeftCount = "node/imgBG/Content/KinMail/txtTimesLeftCount"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSend = function(self)
    self:Send()
  end
}

function tbWnd:OnOpen()
  self.pPanel:Input_SetText(self.inputMail, "")
  self:Update()
  Tong:UpdateMailInfo()
end

function tbWnd:Update(szType)
  local nMailLeftCount = Tong:GetLeftMailCount() or 0
  self.pPanel:Label_SetText(self.txtTimesLeftCount, string.format("%d/%d", nMailLeftCount, Tong.Cfg.nMaxMailCount))
end

function tbWnd:Send()
  local szContent = self.pPanel:Input_GetText(self.inputMail)
  if Tong:SendMail(szContent) then
    self:Close()
  end
end
