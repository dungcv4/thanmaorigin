local tbWnd = Ui:CreateWindow("UITongApplyList")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_TONG_APPLYLIST,
      self.GenerateApplyList
    },
    {
      EventNotify.emNOTIFY_SYNC_TONG_QUIT,
      self.Close
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  Template = "node/imgBG/Content/imgList/List/Template"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbApplyListScrollView = Ui:NewScrollView(self, self.Template)
end

function tbWnd:OnOpen()
  self:GenerateApplyList()
  Tong:UpdateApplyInfo()
end

function tbWnd:GenerateApplyList()
  self.tbApplyListScrollView:Clear()
  local tbApplyerList = Tong:GetApplyerList()
  for i = 1, #tbApplyerList do
    self.tbApplyListScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtKinName", tbApplyerList[i].szName)
      pPanel:Label_SetText("txtKinNum", tbApplyerList[i].nCount)
      pPanel:Label_SetText("txtFightPowerNum", tbApplyerList[i].nFightPower)
      pPanel:Label_SetText("txtActivityNum", tbApplyerList[i].nWeekActive)
      Ui:Button_BindEvent(pPanel, "btnOK", self.AgreeApply, self, tbApplyerList[i].nKinId)
      Ui:Button_BindEvent(pPanel, "btnCancel", self.IgnoreApply, self, tbApplyerList[i].nKinId)
    end)
  end
end

function tbWnd:AgreeApply(nKinId)
  if not Tong:CheckMyAuthority(Tong.Def.Authority_Recruit) then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.NoAuthority))
    return
  end
  Tong:C2SAgreeApply(nKinId)
end

function tbWnd:IgnoreApply(nKinId)
  if not Tong:CheckMyAuthority(Tong.Def.Authority_Recruit) then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.NoAuthority))
    return
  end
  Tong:C2SDisAgreeApply(nKinId)
end

function tbWnd:OnTongNewApplyer()
end
