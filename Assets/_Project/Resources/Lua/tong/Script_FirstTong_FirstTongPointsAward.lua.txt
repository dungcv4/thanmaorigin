FirstTong.PointsAward = FirstTong.PointsAward or {}
local PointsAward = FirstTong.PointsAward

function PointsAward:RequestHandInData()
  RemoteServer.RequestFirstTong("RequestHandInData")
end

function PointsAward:UpdataHandInData(nTotalCount, nMyCount)
  local tbCacheData = {nTotalCount = nTotalCount, nMyCount = nMyCount}
  self.tbCacheData = tbCacheData
  EventNotify.OnNotify(EventNotify.emNOTIFY_UPDATE_EXTRA_AWARD)
end

function PointsAward:GetTotalCount()
  if not self.tbCacheData or not self.tbCacheData.nTotalCount then
    return 0
  end
  return self.tbCacheData.nTotalCount
end

function PointsAward:GetMyCount()
  if not self.tbCacheData or not self.tbCacheData.nMyCount then
    return 0
  end
  return self.tbCacheData.nMyCount
end

function PointsAward:UpdateLevelList(tbWnd)
  local nTotalCount = self:GetTotalCount()
  local tbLevel = FirstTong.JiJieAwardSetting
  tbWnd.tbLevelList:Clear()
  for nIndex, tbInfo in ipairs(tbLevel) do
    tbWnd.tbLevelList:SetElement(nIndex, function(tbElement)
      local tbAward = Award:GetAward(tbInfo.nAwardId, tbInfo.nAwardKey)
      local tbShowAwardScrollView = Ui:NewItemGridScrollView(tbElement, tbWnd.tbTemplateElement.reward, 1, 1)
      tbShowAwardScrollView:UpdateGenericItem(tbAward, true)
      tbElement.pPanel:Label_SetText(tbWnd.tbTemplateElement.txtNumber, i18n.EasyGet("\229\184\174\228\188\154\231\180\175\232\174\161{0}\231\167\175\229\136\134", tbInfo.nPoint))
      local bCompleted = nTotalCount >= tbInfo.nPoint
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.Prg, not bCompleted)
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.imgBar, not bCompleted)
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.txtPrg, not bCompleted)
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.imgComplete, bCompleted)
      tbElement.pPanel:Label_SetText(tbWnd.tbTemplateElement.CompleteText, i18n.EasyGet("\229\183\178\232\190\190\230\136\144"))
      if not bCompleted then
        tbElement.pPanel:Label_SetText(tbWnd.tbTemplateElement.txtPrg, string.format("%d/%d", nTotalCount, tbInfo.nPoint))
        tbElement.pPanel:Sprite_SetFillAmount(tbWnd.tbTemplateElement.imgBar, 0, nTotalCount / tbInfo.nPoint, 1, true)
      end
    end)
  end
end

function PointsAward:UpdateOtherInfo(tbWnd)
  local nMyCount = self:GetMyCount()
  local nTotalCount = self:GetTotalCount()
  tbWnd.pPanel:Label_SetText(tbWnd.szMineNumber, i18n.EasyGet("\230\136\145\231\180\175\232\174\161\229\162\158\229\138\160\231\167\175\229\136\134\239\188\154{0}", nMyCount))
  tbWnd.pPanel:Label_SetText(tbWnd.szTongNumber, i18n.EasyGet("\229\184\174\228\188\154\230\128\187\231\167\175\229\136\134\239\188\154{0}", nTotalCount))
  tbWnd.pPanel:Label_SetText(tbWnd.txtTitleNumber, i18n.EasyGet("\230\143\144\228\186\164\230\149\176\233\135\143"))
  tbWnd.pPanel:SetActive(tbWnd.Template, true)
  tbWnd.pPanel:SetActive(tbWnd.btnAdd, false)
  local nItemTemplateId = FirstTong.JiJieCfg.nTokenPayId
  local tbGrid = Ui:NewExistItemGrid(tbWnd, tbWnd.szItemGrid)
  tbGrid:SetGenericItem({"Item", nItemTemplateId})
  tbGrid:SetClickEnable(true)
  tbWnd.pPanel:Label_SetText(tbWnd.szItemName, Item:GetName(nItemTemplateId, me))
  local nHave = me.GetItemCountByTypeInBag(nItemTemplateId)
  tbWnd.pPanel:Label_SetText(tbWnd.szItemNums, i18n.EasyGet("\230\139\165\230\156\137\239\188\154{0}", nHave))
  tbWnd.pPanel:SetActive(tbWnd.btnHelp, false)
  tbWnd.pPanel:Label_SetText(tbWnd.txtInfo, i18n.EasyGet("\230\143\144\228\186\164\232\191\135\231\190\164\233\155\132\228\187\164\231\154\132\229\164\167\228\190\160\239\188\140\232\190\190\230\136\144\232\191\155\229\186\166\229\141\179\229\143\175\232\142\183\229\190\151\233\130\174\228\187\182\229\165\150\229\138\177"))
  tbWnd.pPanel:Button_SetText(tbWnd.btnGo, i18n.EasyGet("\229\137\141\229\190\128\228\184\138\228\186\164"), "Text")
end

function PointsAward:OnClickAdd(tbWnd)
  Ui:OpenWindow("UIAccess_Tip", nil, FirstTong.JiJieCfg.nTokenPayId)
end

function PointsAward:OnClickbtnGo(tbWnd)
  ActivityMgr:GotoNpc(FirstTong.JiJieCfg.nNpcId, FirstTong.JiJieCfg.nNpcMapId)
  tbWnd:Close()
end

function PointsAward:OnUiEvent(tbWnd, szEvent, ...)
  if tbWnd.ClassName ~= "UIReward_Newyear" then
    return
  end
  if "OnOpen" == szEvent then
    self:RequestHandInData()
    self:UpdateLevelList(tbWnd)
    self:UpdateOtherInfo(tbWnd)
  end
  if "OnClick" == szEvent then
    local tbParam = {
      ...
    }
    if "btnAdd" == tbParam[1] then
      self:OnClickAdd(tbWnd)
    elseif "btnGo" == tbParam[1] then
      self:OnClickbtnGo(tbWnd)
    end
  end
  if "UpdateReward" == szEvent then
    self:UpdateLevelList(tbWnd)
    self:UpdateOtherInfo(tbWnd)
  end
end
