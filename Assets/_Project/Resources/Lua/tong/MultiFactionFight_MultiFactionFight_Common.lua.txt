MultiFactionFight.GAME_STATE = {
  SIGNUP = 1,
  QUALIFIER = 2,
  DECIDER = 3,
  CLOSE = 4
}
MultiFactionFight.CLASS = {
  BASE = "Base",
  QUALIFIER = "Qualifier",
  DECIDER = "Decider"
}
MultiFactionFight.STATE_2_CLASS = {
  [MultiFactionFight.GAME_STATE.QUALIFIER] = MultiFactionFight.CLASS.QUALIFIER,
  [MultiFactionFight.GAME_STATE.DECIDER] = MultiFactionFight.CLASS.DECIDER
}
MultiFactionFight.nPerPagDataNums = 5
MultiFactionFight.tbClass = MultiFactionFight.tbClass or {}

function MultiFactionFight:CreateClass(szNewClass, szBaseClass)
  if szBaseClass and self.tbClass[szBaseClass] then
    self.tbClass[szNewClass] = Lib:NewClass(self.tbClass[szBaseClass])
  else
    self.tbClass[szNewClass] = {}
  end
  local tbClassInst = self.tbClass[szNewClass]
  return tbClassInst
end

function MultiFactionFight:GetClass(szClassName)
  if self.tbClass[szClassName] then
    return self.tbClass[szClassName]
  else
    LogErr("[MultiFactionFight]tbClass Not Exsit!", szClassName)
  end
end
