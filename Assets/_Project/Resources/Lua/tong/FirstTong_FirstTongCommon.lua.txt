function FirstTong:IsSysOpen()
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local nNowTime = GetTime()
  if nNowTime < Lib:GetConfigDate2Time(self.TimeCfg.nJiJieStart) or nNowTime > Lib:GetConfigDate2Time(self.TimeCfg.nFinalsEnd) then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  return true
end

function FirstTong:CheckInPreTime()
  if not self.IS_SYS_OPEN then
    return false
  end
  local nTime = GetTime()
  if nTime < Lib:ParseDateTime(tostring(self.TimeCfg.nJiJieStart)) or nTime > Lib:ParseDateTime(tostring(self.TimeCfg.nJiJieEnd)) then
    return false
  end
  return true
end

function FirstTong:CheckInFinalsTime()
  if not self.IS_SYS_OPEN then
    return false
  end
  local nTime = GetTime()
  if nTime < Lib:ParseDateTime(tostring(self.TimeCfg.nFinalsStart)) or nTime > Lib:ParseDateTime(tostring(self.TimeCfg.nFinalsEnd)) then
    return false
  end
  return true
end

function FirstTong:IsGameMap(nMapTid)
  if nMapTid == self.Cfg.nReadyMapId then
    return true
  end
  for _, tbData in pairs(self.MAP_FIGHT) do
    if tbData.nMapId == nMapTid then
      return true
    end
  end
  return false
end

function FirstTong:CheckGmWarch(pPlayer)
  local nGmCard = pPlayer.GetItemCountByTypeInBag(FirstTong.Cfg.nGmCardId)
  return nGmCard > 0
end

function FirstTong:GetStageGuessCfg(nPhase)
  if not nPhase then
    return
  end
  for nIndex, tbCfg in ipairs(FirstTong.GuessTimeCfg) do
    if nPhase == tbCfg.nPhase then
      tbCfg.nIndex = nIndex
      return tbCfg
    end
  end
end

function FirstTong:GetNowFinalsSatge(tbFinalsData)
  if not tbFinalsData then
    return
  end
  for nIndex = #FirstTong.GuessTimeCfg, 1, -1 do
    local tbCfg = FirstTong.GuessTimeCfg[nIndex]
    if tbCfg.nPhase and tbFinalsData[tbCfg.nPhase] then
      return tbCfg.nPhase
    end
  end
end
