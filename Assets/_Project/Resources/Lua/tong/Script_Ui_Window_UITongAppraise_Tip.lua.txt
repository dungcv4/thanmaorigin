local tbWnd = Ui:CreateWindow("UITongAppraise_Tip")
tbWnd.tbControls = {
  Template = "node/wndQiHunInfo/imgBg/Scroll View/Viewport/template/AttrTemp",
  Empty4Raycast = "node/wndQiHunInfo/Empty4Raycast",
  txtFamily = "node/wndQiHunInfo/imgBg/txtFamily",
  txtRoleName = "node/wndQiHunInfo/imgBg/txtRoleName",
  txtNum = "node/wndQiHunInfo/imgBg/txtNum"
}
tbWnd.tbOnClick = {
  Empty4Raycast = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.pPanel:Label_SetText(self.txtFamily, i18n.EasyGet("\229\174\182\230\151\143"))
  self.pPanel:Label_SetText(self.txtRoleName, i18n.EasyGet("\232\167\146\232\137\178"))
  self.pPanel:Label_SetText(self.txtNum, i18n.EasyGet("\231\165\168\230\149\176"))
  self.tbAppraiseListScrollView = Ui:NewScrollView(self, self.Template)
end

function tbWnd:OnOpen(tbVoteInfo)
  self.tbAppraiseListScrollView:Clear()
  if not tbVoteInfo then
    return
  end
  local i = 0
  for nId, tbInfo in pairs(tbVoteInfo) do
    i = i + 1
    self.tbAppraiseListScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtName", tbInfo.szKinName)
      pPanel:Label_SetText("txtRole", tbInfo.szName)
      pPanel:Label_SetText("txtAttr", tbInfo.nVotes)
    end)
  end
end
