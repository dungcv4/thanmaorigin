local tbWnd = Ui:CreateWindow("UITongModifyName")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_TONG_QUIT,
      self.Close
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSave = "node/imgBG/Content/btnSave",
  txtCurrentName = "node/imgBG/Content/txtCurrentName",
  inputNewTongName = "node/imgBG/Content/inputNewTongName",
  txtCost = "node/imgBG/Content/Cost/txtCostNum",
  nodeCost = "node/imgBG/Content/Cost",
  txtFree = "node/imgBG/Content/txtFree"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSave = function(self)
    self:OnChangeTongName()
  end
}

function tbWnd:OnOpen(bFree)
  self.bFree = bFree
  local tbBaseInfo = Tong:GetBaseInfo() or {}
  self.pPanel:Label_SetText(self.txtCurrentName, tbBaseInfo.szName or "")
  self.pPanel:Label_SetText(self.txtCost, Tong.Cfg.nChangNameCost)
  self.pPanel:Input_SetText(self.inputNewTongName, "")
  self.pPanel:SetActive(self.txtFree, bFree)
  self.pPanel:SetActive(self.nodeCost, not bFree)
end

function tbWnd:OnChangeTongName()
  if not Tong:CheckMyAuthority(Tong.Def.Authority_ChangeName) then
    me.CenterMsg(i18n.Get(Tong.Cfg.Msg.NoAuthority))
    return
  end
  local szTongName = self.pPanel:Input_GetText(self.inputNewTongName)
  Tong:ChangeName(szTongName, self.bFree)
  self:Close()
end
