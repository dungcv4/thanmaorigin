function Tong:CheckTongName(szTongName)
  if not szTongName or "" == szTongName then
    return false, Tong.ErrorCode.TongNameEmpty
  end
  local nNameLen = Lib:StrLen(szTongName)
  if nNameLen > Tong.Cfg.nMaxTongNameLength or nNameLen < Tong.Cfg.nMinTongNameLength then
    return false, Tong.ErrorCode.TongNameLengthLimit
  end
  if string.find(szTongName, "%s") then
    return false, Tong.ErrorCode.CantContainSpace
  end
  if Tong.tbAllTongNames then
    local szTongNameLowerCase = Lib:StringToLowerSameChar(szTongName)
    if Tong.tbAllTongNames[szTongNameLowerCase] then
      return false, Tong.ErrorCode.DuplicateTongName
    end
  end
  return true
end

function Tong:CheckTongNotice(szContent)
  if "" ~= szContent and ReplaceLimitWords(szContent) then
    return false, Tong.ErrorCode.IllegalWorld
  end
  if Lib:StrLen(szContent) > Tong.Cfg.nMaxNoticeLength then
    return false, Tong.ErrorCode.ContentLengthLimit
  end
  return true
end

function Tong:CheckTongTitle(szContent)
  if "" ~= szContent and ReplaceLimitWords(szContent) then
    return false, Tong.ErrorCode.IllegalWorld
  end
  if "" ~= szContent and not CheckNameAvailable(szContent) then
    return false, Tong.ErrorCode.IllegalWorld
  end
  if Lib:StrLen(szContent) > Tong.Cfg.nMaxTitleLength then
    return false, Tong.ErrorCode.TitleLengthLimit
  end
  return true
end

function Tong:CheckMailContent(szContent)
  if not szContent then
    return false, Tong.ErrorCode.MailContentEmpty
  end
  if "" == szContent then
    return false, Tong.ErrorCode.MailContentEmpty
  end
  if "" ~= szContent and ReplaceLimitWords(szContent) then
    return false, Tong.ErrorCode.IllegalWorld
  end
  if Lib:StrLen(szContent) > Tong.Cfg.nMaxMailLength then
    return false, Tong.ErrorCode.ContentLengthLimit
  end
  return true
end

function Tong:GetMaxTongOfficialLevel()
  for i = 1, #Tong.Def.OfficialMaxLevelByTimeFrame do
    if 1 == GetTimeFrameState(Tong.Def.OfficialMaxLevelByTimeFrame[i].TimeFrame) then
      return Tong.Def.OfficialMaxLevelByTimeFrame[i].MaxLevel
    end
  end
  return 1
end
