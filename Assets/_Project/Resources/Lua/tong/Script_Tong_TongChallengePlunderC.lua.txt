Require("Script/Fuben/FubenUIController.lua")
Require("CommonScript/Tong/TongChallengeCfg.lua")
TongChallenge.PlunderMgr = TongChallenge.PlunderMgr or {}
local PlunderMgr = TongChallenge.PlunderMgr

function PlunderMgr:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    TongChallenge.nPlunderMapTemplateId
  }, self)
end

function PlunderMgr:IsNeedExitFubenButton()
  return false
end

function PlunderMgr:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenDefaultFuBenUI()
end

function PlunderMgr:OnLeave()
  FubenUIController:CloseDefaultFubenUI()
end

function PlunderMgr:IsInFubenMap()
  return true
end

PlunderMgr:RegisterFubenUIChecker()
