local tbWnd = {}
Ui:SetClass("UIMultipleButtonGroup", tbWnd)

function Ui:NewMultipleButtonGroup(tbOwner, ...)
  local tbClass = Ui:NewClass(Ui:GetClass("UIMultipleButtonGroup"))
  tbClass.tbOwner = tbOwner
  tbClass:OnCreate(...)
  return tbClass
end

function tbWnd:OnCreate(tbBtnKeyList, func, fnClickCheck)
  self.tbBtnKeyList = tbBtnKeyList
  self.funcCallback = func
  self.fnClickCheck = fnClickCheck
  self.tbKey = {}
  for k, v in pairs(self.tbBtnKeyList) do
    Ui:Button_BindEvent(self.tbOwner.pPanel, v, self.OnClickElement, self, k)
  end
end

function tbWnd:Select(key)
  self.OnClickElement(self, key)
end

function tbWnd:OnClickElement(key)
  if self.fnClickCheck and not self.fnClickCheck(self.tbOwner, key) then
    return
  end
  self.tbKey[key] = not self.tbKey[key]
  local btn = self.tbOwner.pPanel:GetObject(self.tbBtnKeyList[key])
  local trans = btn:GetComponent(Ui.BGroupTrans)
  if trans then
    if self.tbKey[key] then
      trans:OnSelected()
    else
      trans:OnUnSelected()
    end
  end
  if self.funcCallback then
    self.funcCallback(self.tbOwner, key)
  end
end

function tbWnd:SetButtonText(key, szText)
  self.tbOwner.pPanel:Button_SetText(self.tbBtnKeyList[key], szText, "Text")
end

function tbWnd:LightRedPoint(key)
  local szBtnKey = self.tbBtnKeyList[key]
  RedPointMgr:ForceLightRedPoint(self.tbOwner.pPanel, szBtnKey)
end

function tbWnd:FadeRedPoint(key)
  local szBtnKey = self.tbBtnKeyList[key]
  RedPointMgr:ForceFadeRedPoint(self.tbOwner.pPanel, szBtnKey)
end

function tbWnd:GetTbKey()
  return self.tbKey
end

function tbWnd:GetElementNum()
  return #self.tbBtnKeyList
end

function tbWnd:GetButtonKey(nIndex)
  return self.tbBtnKeyList[nIndex]
end

function tbWnd:ClearSelected()
  for key, value in pairs(self.tbKey) do
    local btn = self.tbOwner.pPanel:GetObject(self.tbBtnKeyList[key])
    local trans = btn:GetComponent(Ui.BGroupTrans)
    if trans then
      trans:OnUnSelected()
    end
  end
  self.tbKey = {}
end
