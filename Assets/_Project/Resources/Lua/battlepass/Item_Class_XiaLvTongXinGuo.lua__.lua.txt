local tbItem = Item:GetClass("XiaLvTongXinGuo")
tbItem.PARAM_AWARD_ID = 1
tbItem.PARAM_AWARD_KEY = 2

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnTimeOut(pItem, pPlayer)
  local nItemTid = pItem.dwTemplateId
  local nCount = pItem.nCount
  local szName = pItem.szName
  local szCurrencyType, nNum = Item.tbItemSell:GetSellCurrencyValue(nItemTid)
  if not szCurrencyType then
    return
  end
  if nCount ~= pPlayer.ConsumeItem(pItem, nCount, Env.LogWay_TimeOut) then
    Log("XiaLvTongXinGuo:OnTimeOut", pPlayer.szName, "pid:", pPlayer.dwID, "item:", nItemTid, "count:", nCount, "errConsumeItem")
    return
  end
  local nTotalNum = nNum * nCount
  local tbAward = {
    {szCurrencyType, nTotalNum}
  }
  pPlayer.SendAward(tbAward, true, true, Env.LogWay_TimeOut, nil, nil)
  local szMsg = i18n.EasyGet("\230\130\168\230\156\137{0}\228\184\170{1}\229\183\178\232\191\135\230\156\159\239\188\140\232\191\148\232\191\152\230\130\168{2}{3}", nCount, szName, nTotalNum, Currency:GetName(szCurrencyType))
  pPlayer.Msg(szMsg)
  Log("XiaLvTongXinGuo:OnTimeOut", pPlayer.szName or "", "pid:" .. (pPlayer.dwID or 0), "item:" .. nItemTid, "count:" .. nCount)
end

function tbItem:OnUseMany(it, nCount)
  if not (it.dwTemplateId and me) or not nCount then
    return 0
  end
  local nAwardId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARD_ID)
  local nAwardKey = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARD_KEY)
  if not nAwardId or not nAwardKey then
    return 0
  end
  local tbAwardParam = {
    nLogReason = Env.LogWay_SeventhEvening,
    bNotMerge = true
  }
  local tbAwards = Award:GetAwardWithTimes(nAwardId, nAwardKey, nCount, tbAwardParam)
  if not next(tbAwards) then
    return 0
  end
  Award:SendAward(me.dwID, tbAwards, tbAwardParam)
  Event:OnEvent(Event.emEVENT_ACHIEVEAWARD_ON_USE, me.dwID, it.dwTemplateId, nCount)
  return nCount
end

function tbItem:GetClientDefaultUseCount(nItemId)
  local pItem = me.GetItemInBag(nItemId)
  local nResult = math.min(pItem.nCount, Item.UseManyLimit.XiaLvTongXinGuo)
  return nResult
end
