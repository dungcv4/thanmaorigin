function tbWnd:OnCreate()
  self.uiTree = Ui:NewTree(self, tbControls.UITree, self.OnJingMaiGradeSelect, self.OnJingMaiSelect)
  for nJMIndex, tbV in ipairs(tbControls.tbAcupointPath) do
    for nXWIndex, szKey in ipairs(tbV) do
      Ui:Button_BindEvent(self.pPanel, szKey, self.OnXueWeiSelect, self, nJMIndex, nXWIndex)
    end
  end
  self.tbZhouTianGrid = {}
  for nJMIndex, tbV in ipairs(tbControls.tbZhouTianPath) do
    self.tbZhouTianGrid[nJMIndex] = {}
    for nZTIndex, szKey in ipairs(tbV) do
      Ui:Button_BindEvent(self.pPanel, szKey, self.OnZhouTianSelect, self, nJMIndex, nZTIndex)
      self.tbZhouTianGrid[nJMIndex][nZTIndex] = Ui:NewExistItemGrid(self, szKey .. "/ItemGrid/UIItemGrid")
      self.tbZhouTianGrid[nJMIndex][nZTIndex]:SetClickEnable(true, function()
        self:OnZhouTianSelect(nJMIndex, nZTIndex)
      end)
    end
  end
  self.tbConsumeItemGrid = Ui:NewExistItemGrid(self, tbControls.ConsumeItemGrid)
  self.tbConsumeItemGrid:SetClickEnable(true)
  self.tbConsumeItemGrid1 = Ui:NewExistItemGrid(self, tbControls.ConsumeItemGrid1)
  self.tbConsumeItemGrid1:SetClickEnable(true)
  self.tbConsumeItemGrid2 = Ui:NewExistItemGrid(self, tbControls.ConsumeItemGrid2)
  self.tbConsumeItemGrid2:SetClickEnable(true)
  self.tbHuMaiFuItemGrid = Ui:NewExistItemGrid(self, tbControls.HuMaiFuItemGrid)
  self.tbHuMaiFuItemGrid:SetClickEnable(true, function()
    local tbHuMaiFuData = JingMai:GetZhouTianHuMaiFuData(me, self.nJMIndex, self.nZTIndex, self.nGradeSelect)
    if tbHuMaiFuData[1] > 0 then
      local nTemplateId = JingMai:GetHuMaiFuTemplateId(me, self.nJMIndex, self.nZTIndex, self.nGradeSelect)
      if nTemplateId then
        local tbInfo = {
          nTemplate = nTemplateId,
          nFaction = me.nFaction,
          nFactionSect = me.nFactionSect,
          nSex = me.nSex,
          tbLinkInfo = {tbHuMaiFuData = tbHuMaiFuData}
        }
        Item:ShowItemDetail(tbInfo)
      end
    end
  end)
  self:SetClickEmptyToClose(tbControls.wndJumpTo)
  self:SetClickEmptyToClose(tbControls.wndReplace)
  self:SetClickEmptyToClose(tbControls.wndAllAttribute)
  self:SetClickEmptyToClose(tbControls.wndAcupoint)
  self:SetClickEmptyToClose(tbControls.wndUniverse)
  self.tbHuMaiFuScrollView = Ui:NewSuperListView(self, tbControls.HuMaiFuTemplate, self.SetHuMaiFuRePlaceElement)
  self.tbAcupointQualityAttrSV = Ui:NewScrollView(self, tbControls.AcupointQualityAttrTemplate)
  self.tbUniverseQualityAttrSV = Ui:NewScrollView(self, tbControls.UniverseQualityAttrTemplate)
  self.tbObtainScrollView = Ui:NewScrollView(self, tbControls.ObtainTemplate)
end

function tbWnd:OnDestroy()
  for nJMIndex, tbV in ipairs(tbControls.tbAcupointPath) do
    for nXWIndex, szKey in ipairs(tbV) do
      Ui:Button_UnBindEvent(self.pPanel, szKey)
    end
  end
  for nJMIndex, tbV in ipairs(tbControls.tbZhouTianPath) do
    for nZTIndex, szKey in ipairs(tbV) do
      Ui:Button_UnBindEvent(self.pPanel, szKey)
    end
  end
end

function tbWnd:OnOpen(nGrade, nJMIndex, nXWIndex, nZTIndex)
  nJMIndex = nJMIndex or 1
  self.nOpenJMIndex = nJMIndex
  self.nOpenXWIndex = nXWIndex
  self.nOpenZTIndex = nZTIndex
  self.nOpenGrade = nGrade or 1
  local tbBigData, tbSmallData = JingMai:GetTreeShowData(true)
  self.uiTree:SetData(tbBigData, tbSmallData)
  self.uiTree:OnClickBigButton(self.nOpenGrade, tbBigData[self.nOpenGrade].tbInfo, true)
  self.pPanel:SetActive(tbControls.btnList, false)
  self:UpdateMaiQiValue()
end

function tbWnd:OnClose()
  if self.nJMIndex then
    self.pPanel:SetActive(tbControls.tbJingMaiPath[self.nJMIndex], false)
  end
  for nGrade = 1, JingMai.nMaxGrade do
    for nJMIndex = 1, JingMai.nMaxJingMai do
      for nXWIndex = 1, JingMai.nMaxXueWei do
        JingMai:OnCheckXiuLianRedPoint(nJMIndex, nXWIndex, nGrade)
      end
    end
  end
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnActive)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnXiuLian)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnHuMaiFuAdd)
  for nJMIndex, tbV in ipairs(tbControls.tbAcupointPath) do
    for nXWIndex, szKey in ipairs(tbV) do
      RedPointMgr:DynamicUnRegist(self.pPanel, szKey)
    end
  end
  for nJMIndex, tbV in ipairs(tbControls.tbZhouTianPath) do
    for nZTIndex, szKey in ipairs(tbV) do
      RedPointMgr:DynamicUnRegist(self.pPanel, szKey)
    end
  end
  self.nGradeSelect = nil
  self.nJMIndex = nil
  self.nXWIndex = nil
  self.nZTIndex = nil
  self.nOpenXWIndex = nil
  self.nOpenZTIndex = nil
end

function tbWnd:DynamicRegisteRedPoint()
  for nJMIndex, tbV in ipairs(tbControls.tbAcupointPath) do
    for nXWIndex, szKey in ipairs(tbV) do
      RedPointMgr:DynamicUnRegist(self.pPanel, szKey)
      local nActiveIdx = JingMai:GetActiveRedPointID(nJMIndex, nXWIndex, self.nGradeSelect)
      local nXiuLianIdx = JingMai:GetXiuLianRedPointID(nJMIndex, nXWIndex, self.nGradeSelect)
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szKey, {nActiveIdx, nXiuLianIdx})
    end
  end
  for nJMIndex, tbV in ipairs(tbControls.tbZhouTianPath) do
    for nZTIndex, szKey in ipairs(tbV) do
      RedPointMgr:DynamicUnRegist(self.pPanel, szKey)
      local nZTIdx = JingMai:GetHuMaiFuRedPointID(nJMIndex, nZTIndex, self.nGradeSelect)
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szKey, {nZTIdx})
    end
  end
end

function tbWnd:UpdateObtainSV(tbData)
  self.pPanel:SetActive(tbControls.wndJumpTo, true)
  
  local function fnSetItem(tbValue, tbElement)
    local pPanel = tbElement.pPanel
    pPanel:Label_SetText("txtName", tbValue.szTitle())
    pPanel:Label_SetText("txtDesc", tbValue.szDesc())
    if tbValue.funcObtain then
      pPanel:SetActive("imgArrow", true)
      Ui:Button_BindEvent(pPanel, "Main", function()
        self.pPanel:SetActive(tbControls.wndJumpTo, false)
        if tbData.szType == "HuMaiFuOb" then
          local nTemplateId = JingMai.JMHuMaiId[self.nZTIndex][1]
          tbValue.funcObtain(nTemplateId)
        elseif tbData.szType == "YinZhenOb" then
          tbValue.funcObtain()
        else
          tbValue.funcObtain()
        end
      end)
    else
      pPanel:SetActive("imgArrow", false)
    end
  end
  
  self.tbObtainScrollView:Reset()
  for i, tbValue in ipairs(tbData) do
    self.tbObtainScrollView:SetElement(i, function(tbElement)
      fnSetItem(tbValue, tbElement)
    end)
  end
end

function tbWnd:UpdateMaiQiValue()

  self.pPanel:Label_SetText(tbControls.txtMaiQiRecoverInfo, i18n.EasyGet("\232\132\137\230\176\148\239\188\154{0}/{1}  \230\175\143{2}\229\136\134\233\146\159\229\155\158\229\164\1411\231\130\185", JingMai:GetMaiQiValue(me), tbCfg.nMaiQiUpperLimit, math.floor(tbCfg.nRestoreOnePointMaiQiTime / 60)))
end

function tbWnd:UpdateSelectJingMaiInfo()
  local szJMName = JingMai:GetJingMaiName(self.nJMIndex)
  self.pPanel:Label_SetText(tbControls.txtTittle, szJMName)
  self:UpdateFightPower()
  self.nActiveCount = JingMai:GetActiveXueWeiCount(me, self.nJMIndex, self.nGradeSelect)
  self.nXWIndex = nil
  self.nZTIndex = nil
  self:UpdateAllXueWeiInfo()
  self:UpdateAllZhouTianInfo()
  if self.nOpenXWIndex then
    self:OnXueWeiSelect(self.nJMIndex, self.nOpenXWIndex)
    self.nOpenXWIndex = nil
    self.nOpenZTIndex = nil
  elseif self.nOpenZTIndex then
    self:OnZhouTianSelect(self.nJMIndex, self.nOpenZTIndex)
    self.nOpenZTIndex = nil
  else
    self:OnXueWeiSelect(self.nJMIndex, 1)
  end
end

function tbWnd:OnShowJingMaiAllAttr()
  self.pPanel:SetActive(tbControls.wndAllAttribute, true)
  self.pPanel:Label_SetText(tbControls.txtAllAttrTitle, i18n.EasyGet("{0}\229\133\168\229\177\158\230\128\167\233\162\132\232\167\136", JingMai:GetJingMaiName(self.nJMIndex)))
  for i = 1, 6 do
    if 1 == i % 2 then
      for nIdx = 1, 3 do
        local nXWIndex = (math.ceil(i / 2) - 1) * 3 + nIdx
        local nAttrId = JingMai:GetXueWeiAttrID(self.nJMIndex, nXWIndex)
        local tbAttrCfg = JingMai.tbJMAttrCfg[nAttrId]
        local szPathBase = tbControls.AllAttrTemplate .. i .. "/Cont_Acupoint/img_Info0" .. nIdx
        local szXueWeiName = JingMai:GetXueWeiName(self.nJMIndex, nXWIndex)
        if nXWIndex <= self.nActiveCount then
          local tbAttrValue = JingMai:GetXueWeiAttrValue(me, self.nJMIndex, nXWIndex, self.nGradeSelect)
          local nXWQuality = JingMai:GetXueWeiQuality(me, self.nJMIndex, nXWIndex, self.nGradeSelect)
          local szAttrName, szAttrValue = FightSkill:GetMagicDescSplit(tbAttrCfg[nXWQuality].szAttrType, tbAttrValue)
          local szColor = Ui.tbColor:GetColorHexByDefineName(JingMai:GetQualityName(nXWQuality))
          self.pPanel:Label_SetText(szPathBase .. "/txtName", szXueWeiName)
          self.pPanel:Label_SetText(szPathBase .. "/txtInfo", i18n.EasyGet("{0}\239\188\154<color=#{1}>{2}</color>", szAttrName, szColor, szAttrValue))
          self.pPanel:SetActive(szPathBase .. "/img_Lock", false)
        else
          local tbValue = JingMai:GetXueWeiMinAttrCfg(self.nJMIndex, nXWIndex)
          local szAttribMsg = FightSkill:GetMagicDesc(tbAttrCfg[1].szAttrType, tbValue)

          self.pPanel:Label_SetText(szPathBase .. "/txtInfo", "")

          self.pPanel:Label_SetText(szPathBase .. "/img_Lock/txtInfoLock", szAttribMsg)
        end
      end
    else

      local szPathBase = tbControls.AllAttrTemplate .. i .. "/Cont_Universe/img_Bg/"
      local szAttrPathBase = tbControls.AllAttrTemplate .. i .. "/Cont_Universe/img_Bg/txtInfo0"
      local tbHuMaiFuData = JingMai:GetZhouTianHuMaiFuData(me, self.nJMIndex, nZTIndex, self.nGradeSelect)
      local tbZTAttrCfg = JingMai.JMZhouTianAttrInfo[self.nJMIndex][nZTIndex]
      local nZTQuality = JingMai:GetZhouTianQuality(me, self.nJMIndex, nZTIndex, self.nGradeSelect)
      local tbAttMsg = {}
      for nAttrIndex, nAttrId in ipairs(tbZTAttrCfg) do
        local tbAttrCfg = JingMai.tbZhouTianAttrCfg[nAttrId][JingMai.nZhouTianBaseAttrQuality]
        tbAttMsg[nAttrIndex] = ""
        local tbAttrV = {
          0,
          0,
          0
        }
        if tbAttrCfg then
          for nIndex = 1, 3 do
            tbAttrV[nIndex] = tbAttrV[nIndex] + tbAttrCfg[nIndex + 1]
          end
          local nAttrAddType = JingMai.tbZTAttrAddPercentIndexType[nAttrId]
          local tbAttrCfgQuality = JingMai.tbZhouTianAttrCfg[nAttrId][nZTQuality]
          if tbAttrCfgQuality then
            if JingMai.tbZhouTianAttrAddType.ZTADT_2 == nAttrAddType then
              tbAttrV[2] = tbAttrV[2] + tbAttrCfgQuality[3]
            else
              tbAttrV[1] = tbAttrV[1] + tbAttrCfgQuality[2]
              if JingMai.tbZhouTianAttrAddType.ZTADT_13 == nAttrAddType then
                tbAttrV[3] = tbAttrV[3] + tbAttrCfgQuality[4]
              end
            end
          end
          if nAttrIndex <= tbHuMaiFuData[1] then
            local nAddValue
            if JingMai.tbZhouTianAttrAddType.ZTADT_2 == nAttrAddType then
              nAddValue = math.floor(tbAttrCfg[3] * tbHuMaiFuData[nAttrIndex + 1] / 100)
              tbAttrV[2] = tbAttrV[2] + nAddValue
            else
              nAddValue = math.floor(tbAttrCfg[2] * tbHuMaiFuData[nAttrIndex + 1] / 100)

              end
            end
          end

          tbAttMsg[nAttrIndex] = i18n.EasyGet("{0}\239\188\154<color=#{1}>{2}</color>", szAttrName, szColor, szAttrValue)
        end
      end

        self.pPanel:SetActive(szPathBase .. "img_Lock", false)
        local szZTName = JingMai.tbZhouTianName[nZTIndex]()
        self.pPanel:Label_SetText(szPathBase .. "txtName", szZTName)
        for nAttrIndex = 1, 3 do
          self.pPanel:Label_SetText(szAttrPathBase .. nAttrIndex, tbAttMsg[nAttrIndex])
        end
      else
        self.pPanel:Label_SetText(szPathBase .. "txtName", "")
        self.pPanel:SetActive(szPathBase .. "img_Lock", true)
        local szZTName = JingMai.tbZhouTianName[nZTIndex]()
        self.pPanel:Label_SetText(szPathBase .. "img_Lock/txtName", szZTName)
        for nAttrIndex = 1, 3 do
          self.pPanel:Label_SetText(szAttrPathBase .. nAttrIndex, "")
          self.pPanel:Label_SetText(szPathBase .. "img_Lock/txtInfo0" .. nAttrIndex, tbAttMsg[nAttrIndex])
        end
      end
    end
  end
end

function tbWnd:OnXueWeiActiveSuccess(nXWIndex)

  if tbControls.tbimg_ActiveBg0Path[self.nJMIndex][nXWIndex] then
    self.pPanel:Sprite_SetFillAmount(tbControls.tbimg_ActiveBg0Path[self.nJMIndex][nXWIndex], 0, 1, 0.7)
    Timer:Register(math.floor(nTime * Env.GAME_FPS), function()
      if not self.nJMIndex then
        return false
      end
      local szPath = tbControls.tbAcupointUpgradeEffectPath[self.nJMIndex][nXWIndex]
      self.pPanel:SetActive(szPath, false)
      self.pPanel:SetActive(szPath, true)

      self:OnXueWeiInfoChange()
    end)

    self:OnXueWeiInfoChange()
  end
end

    self.pPanel:SetActive(tbControls.tbZhouTianPath[self.nJMIndex][nZTIndex] .. "/imgSelect", true)
  else
    self.pPanel:SetActive(tbControls.tbZhouTianPath[self.nJMIndex][nZTIndex] .. "/imgSelect", false)
  end
end

function tbWnd:UpdateSelectZhouTianInfo()
  self.pPanel:Label_SetText(tbControls.txtBigTittle, JingMai.tbZhouTianName[self.nZTIndex]())
  local szXWName1 = JingMai:GetXueWeiName(self.nJMIndex, (self.nZTIndex - 1) * JingMai.nZhouTianNeedXueWei + 1)
  local szXWName2 = JingMai:GetXueWeiName(self.nJMIndex, (self.nZTIndex - 1) * JingMai.nZhouTianNeedXueWei + 2)
  local szXWName3 = JingMai:GetXueWeiName(self.nJMIndex, (self.nZTIndex - 1) * JingMai.nZhouTianNeedXueWei + 3)
  local tbHuMaiFuData = JingMai:GetZhouTianHuMaiFuData(me, self.nJMIndex, self.nZTIndex, self.nGradeSelect)
  self.pPanel:SetActive(tbControls.txtItemName, true)
  if self.nActiveCount >= self.nZTIndex * JingMai.nZhouTianNeedXueWei then
    if tbHuMaiFuData[1] > 0 then
      self.pPanel:SetActive(tbControls.btnHuMaiFuRemove, true)
      self.pPanel:SetActive(tbControls.btnHuMaiFuAdd, false)
      self.pPanel:SetActive(tbControls.btnHuMaiFuReplace, true)
      local nTemplateId = JingMai:GetHuMaiFuTemplateId(me, self.nJMIndex, self.nZTIndex, self.nGradeSelect)
      if nTemplateId then
        self.tbHuMaiFuItemGrid:SetItemByTemplate(nTemplateId)
        self.pPanel:SetActive(tbControls.txtItemName, false)
      end
    else
      self.tbHuMaiFuItemGrid:SetEmpty(true)
      self.pPanel:SetActive(tbControls.btnHuMaiFuReplace, false)
      self.pPanel:SetActive(tbControls.btnHuMaiFuRemove, false)
      self.pPanel:SetActive(tbControls.btnHuMaiFuAdd, true)
    end
    self.pPanel:SetActive(tbControls.State_Open_ZhouTian, true)
    self.pPanel:SetActive(tbControls.State_Lock_ZhouTian, false)
    self.pPanel:Label_SetText(tbControls.Tet_UniverseState, i18n.EasyGet("\229\145\168\229\164\169\229\147\129\232\180\168\228\184\142\229\133\182\229\133\179\232\129\148\231\154\132\231\169\180\228\189\141\229\147\129\232\180\168\231\155\184\229\144\140"))
    self.pPanel:Label_SetColorByName(tbControls.Tet_UniverseState, "JingMaiTips")
    self.pPanel:Label_SetText(tbControls.txtZhouTianTip, "")
  else
    self.pPanel:SetActive(tbControls.btnHuMaiFuAdd, false)
    self.pPanel:SetActive(tbControls.btnHuMaiFuRemove, false)
    self.pPanel:SetActive(tbControls.btnHuMaiFuReplace, false)
    self.pPanel:SetActive(tbControls.State_Lock_ZhouTian, true)
    self.pPanel:SetActive(tbControls.State_Open_ZhouTian, false)
    self.pPanel:Label_SetText(tbControls.Tet_UniverseState, i18n.EasyGet("\233\156\128\230\191\128\230\180\187[{0}], [{1}], [{2}]", szXWName1, szXWName2, szXWName3))
    self.pPanel:Label_SetColorByName(tbControls.Tet_UniverseState, "NewRed")
    self.pPanel:Label_SetText(tbControls.txtZhouTianTip, i18n.EasyGet("\230\191\128\230\180\187\229\145\168\229\164\169\229\144\142\229\143\175\232\163\133\229\164\135"))
    self.tbHuMaiFuItemGrid:SetEmpty(true)
  end
  local tbZTAttrCfg = JingMai.JMZhouTianAttrInfo[self.nJMIndex][self.nZTIndex]
  local nZTQuality = JingMai:GetZhouTianQuality(me, self.nJMIndex, self.nZTIndex, self.nGradeSelect)
  if nZTQuality <= 0 then
    nZTQuality = 1
  end
  for nAttrIndex, nAttrId in ipairs(tbZTAttrCfg) do
    local tbAttrCfg = JingMai.tbZhouTianAttrCfg[nAttrId][JingMai.nZhouTianBaseAttrQuality]
    if tbAttrCfg then
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCfg[1], {
        table.unpack(tbAttrCfg, 2)
      })
      self.pPanel:Label_SetText(tbControls.tbZhouTianBaseAttrNamePath[nAttrIndex], szAttribMsg)
      self.pPanel:Label_SetText(tbControls.tbZhouTianBaseAttrValuePath[nAttrIndex], szValue)
      if nAttrIndex <= tbHuMaiFuData[1] then
        local nAttrAddType = JingMai.tbZTAttrAddPercentIndexType[nAttrId]
        local nAddValue
        local nAddPercent = tbHuMaiFuData[nAttrIndex + 1]
        if JingMai.tbZhouTianAttrAddType.ZTADT_2 == nAttrAddType then
          nAddValue = math.floor(tbAttrCfg[3] * nAddPercent / 100)
        else
          nAddValue = math.floor(tbAttrCfg[2] * nAddPercent / 100)
        end
        self.pPanel:Label_SetText(tbControls.tbZhouTianBaseAttrHMFAddValuePath[nAttrIndex], i18n.EasyGet("(+{0})", nAddValue))
        local nAttrQuality = JingMai:GetHuMaiFuAttrQuality(nAddPercent)
        self.pPanel:Label_SetColorByName(tbControls.tbZhouTianBaseAttrHMFAddValuePath[nAttrIndex], JingMai:GetQualityName(nAttrQuality))
        self.pPanel:SetActive(tbControls.tbZhouTianBaseAttrHMFAddValuePath[nAttrIndex], true)
      else
        self.pPanel:Label_SetText(tbControls.tbZhouTianBaseAttrHMFAddValuePath[nAttrIndex], "")
      end
    end
    local tbAttrCfgQuality = JingMai.tbZhouTianAttrCfg[nAttrId][nZTQuality]
    if tbAttrCfgQuality then
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCfgQuality[1], {
        table.unpack(tbAttrCfgQuality, 2)
      })
      self.pPanel:Label_SetText(tbControls.tbZhouTianQualitAttrNamePath[nAttrIndex], szAttribMsg)
      self.pPanel:Label_SetText(tbControls.tbZhouTianQualitAttrValuePath[nAttrIndex], szValue)
      self.pPanel:Label_SetColorByName(tbControls.tbZhouTianQualitAttrValuePath[nAttrIndex], JingMai:GetQualityName(nZTQuality))
    end
  end
end

function tbWnd:SetHuMaiFuRePlaceElement(tbElement, pItem, nIndex)
  local pPanel = tbElement.pPanel
  local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  itemGrid:SetItem(pItem.dwId, pItem.dwTemplateId)
  itemGrid:SetClickEnable(true, function()
  end)
  if pItem.szClass == "HuMaiFu" then
    pPanel:SetActive("btnIdentify", false)
    pPanel:SetActive("btnAdd", true)
    pPanel:SetActive("AttrWid", true)
    pPanel:SetActive("txtEmpty", false)
    Ui:Button_BindEvent(pPanel, "btnAdd", function()
      JingMai:C2SHuMaiFuPutOnReqC(self.nJMIndex, self.nZTIndex, pItem.dwId, self.nGradeSelect)
    end)
    local tbHuMaiFu = Item:GetClass("HuMaiFu")
    local tbItemHuMaiFuData = tbHuMaiFu:GetAddAttrData(pItem)
    local nAddAttrCount = tbItemHuMaiFuData[1]
    local tbZTAttrCfg = JingMai.JMZhouTianAttrInfo[self.nJMIndex][self.nZTIndex]
    for nAttrIndex, nAttrId in ipairs(tbZTAttrCfg) do
      if nAttrIndex <= nAddAttrCount then
        pPanel:SetActive("AttrWid/txtInfo" .. nAttrIndex, true)
        local tbAttrCfg = JingMai.tbZhouTianAttrCfg[nAttrId][JingMai.nZhouTianBaseAttrQuality]
        if tbAttrCfg then
          local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCfg[1], {
            table.unpack(tbAttrCfg, 2)
          })
          local nAttrAddType = JingMai.tbZTAttrAddPercentIndexType[nAttrId]
          local nAddValue
          local nAddPercent = tbItemHuMaiFuData[nAttrIndex + 1]
          if JingMai.tbZhouTianAttrAddType.ZTADT_2 == nAttrAddType then
            nAddValue = math.floor(tbAttrCfg[3] * nAddPercent / 100)
          else
            nAddValue = math.floor(tbAttrCfg[2] * nAddPercent / 100)
          end
          local nAttrQuality = JingMai:GetHuMaiFuAttrQuality(nAddPercent)
          local szColor = Ui.tbColor:GetColorHexByDefineName(JingMai:GetQualityName(nAttrQuality))
          pPanel:Label_SetText("AttrWid/txtInfo" .. nAttrIndex, i18n.EasyGet("<color=#{0}>{1}\239\188\154+{2}</color>", szColor, szAttribMsg, nAddValue))
        end
      else
        pPanel:SetActive("AttrWid/txtInfo" .. nAttrIndex, false)
      end
    end
  else
    pPanel:SetActive("btnIdentify", true)
    pPanel:SetActive("btnAdd", false)
    pPanel:SetActive("AttrWid", false)
    pPanel:SetActive("txtEmpty", true)
    Ui:Button_BindEvent(pPanel, "btnIdentify", function()
      Ui:OpenWindow("UIMeridians_IdentifyTip")
      self.pPanel:SetActive(tbControls.wndReplace, false)
    end)
  end
end

function tbWnd:OnJingMaiGradeSelect(tbInfo, nBigIndex)
  if self.nGradeSelect == tbInfo.nGrade then
    return
  end
  self.nGradeSelect = tbInfo.nGrade
  local nJMIdx = 1
  if self.nOpenJMIndex then
    nJMIdx = self.nOpenJMIndex
    self.nOpenJMIndex = nil
  end
  local nSmallIndex = (self.nGradeSelect - 1) * JingMai.nMaxJingMai + nJMIdx
  self.uiTree:OnClickSmallButton(nSmallIndex, self.uiTree:GetSmallDataInfoByIndex(tbInfo.nGrade, nJMIdx))
end

function tbWnd:OnJingMaiSelect(tbInfo, nSmallIndex)
  if self.nJMIndex then
    self.pPanel:SetActive(tbControls.tbJingMaiPath[self.nJMIndex], false)
  end
  self.nJMIndex = tbInfo.nJMIndex
  self.pPanel:SetActive(tbControls.tbJingMaiPath[self.nJMIndex], true)
  if self.nJMIndex and self.nXWIndex then
    local nIdx = JingMai:GetXiuLianRedPointID(self.nJMIndex, self.nXWIndex, self.nGradeSelect)
    RedPointMgr:FadeRedPoint(nIdx)
  end
  self:UpdateSelectJingMaiInfo()
  self:DynamicRegisteRedPoint()
end
