function JingMai:OpenJingMaiWindow()
  if not self:IsSystemOpen(me) then
    return
  end
  local nGrade, nJMIndex, nXWIndex = self:GetCanActiveXueWei()
  if nGrade then
    Ui:OpenWindowAsync("UIMeridians", nGrade, nJMIndex, nXWIndex)
    return
  end
  for nGrade1 = 1, self.nMaxGrade do
    for nJMIndex1 = 1, self.nMaxJingMai do
      local nActiveIndex = self:GetActiveXueWeiCount(me, nJMIndex1, nGrade1)
      if nActiveIndex >= 1 then
        nGrade, nJMIndex = nGrade1, nJMIndex1
      end
    end
  end
  Ui:OpenWindowAsync("UIMeridians", nGrade, nJMIndex)
end

function JingMai:GetCanActiveXueWei()
  for nGrade = 1, self.nMaxGrade do
    for nJMIndex = 1, self.nMaxJingMai do
      for nXWIndex = 1, self.nMaxXueWei do
        if self:CanActiveXueWei(me, nJMIndex, nXWIndex, nGrade) then
          return nGrade, nJMIndex, nXWIndex
        end
      end
    end
  end
end

function JingMai:GetTreeShowData(bNeedTips)
  local tbBigData = {}
  local tbSmallData = {}
  for nGrade, tbV in ipairs(self.JingMaiTimeFrame) do
    if 1 == GetTimeFrameState(tbV[1]) then
      local tbBigPerData = {
        nType = nGrade,
        szText = i18n.EasyGet("\231\172\172{0}\233\135\141", nGrade),
        tbInfo = {nGrade = nGrade}
      }
      if bNeedTips then
        tbBigPerData.tbRedPoint = {}
      end
      table.insert(tbBigData, tbBigPerData)
      for nJMIndex, szTimeFrame in ipairs(tbV) do
        if 1 == GetTimeFrameState(szTimeFrame) then
          local tbSmallPerData = {
            nType = nGrade,
            szText = self:GetJingMaiName(nJMIndex),
            tbInfo = {nJMIndex = nJMIndex},
            tbRedPoint = {}
          }
          if bNeedTips then
            tbSmallPerData.tbRedPoint = {}
          end
          table.insert(tbSmallData, tbSmallPerData)
          if bNeedTips then
            for nXWIndex = 1, JingMai.nMaxXueWei do
              local nActiveIdx = self:GetActiveRedPointID(nJMIndex, nXWIndex, nGrade)
              table.insert(tbBigPerData.tbRedPoint, nActiveIdx)
              table.insert(tbSmallPerData.tbRedPoint, nActiveIdx)
              local nXiuLianIdx = self:GetXiuLianRedPointID(nJMIndex, nXWIndex, nGrade)
              table.insert(tbBigPerData.tbRedPoint, nXiuLianIdx)
              table.insert(tbSmallPerData.tbRedPoint, nXiuLianIdx)
            end
            for nZTIndex = 1, JingMai.nZhouTianCount do
              local nZTIdx = self:GetHuMaiFuRedPointID(nJMIndex, nZTIndex, nGrade)
              table.insert(tbBigPerData.tbRedPoint, nZTIdx)
              table.insert(tbSmallPerData.tbRedPoint, nZTIdx)
            end
          end
        else
          break
        end
      end
    else
      break
    end
  end
  return tbBigData, tbSmallData
end

function JingMai:QuickUseMaiQiDan()
  local pItemList = me.GetItemListInBag("MaiQiDan")
  if next(pItemList) then
    local tbMaiQiDan = Item:GetClass("MaiQiDan")
    local usemaiqidan = pItemList[1]
    local nMaxAddValue = tbMaiQiDan:GetAddMaiQiValue(usemaiqidan)
    for _, pItem in ipairs(pItemList) do
      local nAddValue = tbMaiQiDan:GetAddMaiQiValue(pItem)
      if nMaxAddValue < nAddValue then
        nMaxAddValue = nAddValue
        usemaiqidan = pItem
      end
    end
    Item:UseMany(usemaiqidan)
    Client.UIModule.PushGOFront("UIItemBatchPanel")
    return true
  else
    return false
  end
end

function JingMai:C2SActiveXueWeiReqC(nJMIndex, nXWIndex, nGrade)

    if jingMaiErrorCode == JingMai.nErrorType.ET_NotEnoughMaiQi then

end

function JingMai:OnCheckRedPoint()
  self:OnCheckAllXueWeiRedPoint()
  self:OnCheckAllZhouTianRedPoint()
end

function JingMai:OnCheckAllXueWeiRedPoint()
  if Ui.bInZone then
    return
  end
  for nGrade = 1, self.nMaxGrade do
    for nJMIndex = 1, self.nMaxJingMai do
      for nXWIndex = 1, self.nMaxXueWei do
        self:OnCheckActiveRedPoint(nJMIndex, nXWIndex, nGrade)
        self:OnCheckXiuLianRedPoint(nJMIndex, nXWIndex, nGrade)
      end
    end
  end
end

function JingMai:OnCheckAllZhouTianRedPoint()
  if Ui.bInZone then
    return
  end
  for nGrade = 1, self.nMaxGrade do
    for nJMIndex = 1, self.nMaxJingMai do
      for nZTIndex = 1, self.nZhouTianCount do
        self:OnCheckZhouTianRedPoint(nJMIndex, nZTIndex, nGrade)
      end
    end
  end
end

function JingMai:OnCheckActiveRedPoint(nJMIndex, nXWIndex, nGrade)
  local nIdx = self:GetActiveRedPointID(nJMIndex, nXWIndex, nGrade)
  if self:CanActiveXueWei(me, nJMIndex, nXWIndex, nGrade) then
    RedPointMgr:LightRedPoint(nIdx)
  else
    RedPointMgr:FadeRedPoint(nIdx)
  end
end

function JingMai:CheckXiuLianRedPointTypeHadClickToday(nJMIndex, nXWIndex)
  local szSaveKey = JingMai.szXueWeiClick
  local nLocalDay = Lib:GetLocalDay(GetTime() - self.OFFSET)
  local nLastClickTime = LocalData:GetRoleNumber(szSaveKey, 0) - self.OFFSET
  local nLastClickDay = Lib:GetLocalDay(nLastClickTime)
  if nLocalDay == nLastClickDay then
    return true
  else
    return false
  end
end

function JingMai:SetNotShowValue(bNotShowToday)
  if bNotShowToday then
    LocalData:SetRoleNumber("JingMai_Save", GetTime())
  else
    LocalData:SetRoleNumber("JingMai_Save", 0)
  end
end

function JingMai:CheckXueWeiIsMaxValue(nJMIndex, nXWIndex, nGrade)
  local tbAttrValue = self:GetXueWeiAttrValue(me, nJMIndex, nXWIndex, nGrade)
  local nAttrId = self:GetXueWeiAttrID(nJMIndex, nXWIndex)
  local nCompareIndex = self.tbAttrId2CompareIndex[nAttrId] or 1
  local _, nMaxValue = self:GetXueWeiAttrValueRange(nJMIndex, nXWIndex)
  if nMaxValue == tbAttrValue[nCompareIndex] then
    return true
  else
    return false
  end
end

function JingMai:OnCheckXiuLianRedPoint(nJMIndex, nXWIndex, nGrade)
  local nIdx = self:GetXiuLianRedPointID(nJMIndex, nXWIndex, nGrade)
  local nZTIndex = self:DoXWIndex2ZTIndex(nXWIndex)
  if not self:CheckXiuLianRedPointTypeHadClickToday(nJMIndex, nXWIndex) and self:CanXiuLianXueWei(me, nJMIndex, nZTIndex, nGrade) and not self:CheckXueWeiIsMaxValue(nJMIndex, nXWIndex, nGrade) then
    RedPointMgr:LightRedPoint(nIdx)
  else
    RedPointMgr:FadeRedPoint(nIdx)
  end
end

function JingMai:OnCheckZhouTianRedPoint(nJMIndex, nZTIndex, nGrade)
  local nIdx = self:GetHuMaiFuRedPointID(nJMIndex, nZTIndex, nGrade)
  local tbHuMaiFuData = self:GetZhouTianHuMaiFuData(me, nJMIndex, nZTIndex, nGrade)
  if tbHuMaiFuData[1] > 0 then
    RedPointMgr:FadeRedPoint(nIdx)
  else
    local bRed = false
    local tbItem = self:GetHuMaiFuList(nZTIndex)
    for _, pItem in ipairs(tbItem) do
      if self:CanPutOnHuMaiFu(me, nJMIndex, nZTIndex, pItem.dwId, nGrade) then
        bRed = true
        break
      end
    end
    if bRed then
      RedPointMgr:LightRedPoint(nIdx)
    else
      RedPointMgr:FadeRedPoint(nIdx)
    end
  end
end

function JingMai:OnSyncItemLimitTime()
  if not PlayerEvent.bLogin then
    return
  end
  self:OnCheckRedPoint()
end

function JingMai:OnDelItem(nItemId, dwTemplateId)
  self:OnCheckRedPoint()
end
