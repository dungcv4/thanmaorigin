local tbWnd = Ui:CreateWindow("UIMeridians_IdentifyTip")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItemEvent
    }
  }
end

local tbControls = {
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  btnClose = "node/imgBG/btnClose",
  UIItemGrid_Silver = "node/imgBG/Content/Template1/UIItemGrid",
  txtSilverPrice = "node/imgBG/Content/txtSilverInfo",
  UIAddAndSubGroupYin = "node/imgBG/Content/Template1/UIAddAndSubGroup",
  txtSilverConsume = "node/imgBG/Content/txtSilverInfo/txtSilverTips",
  btnSilverOK = "node/imgBG/Content/btnSilverOK",
  UIItemGrid_Golden = "node/imgBG/Content/Template2/UIItemGrid",
  txtGoldenPrice = "node/imgBG/Content/txtGoldenInfo",
  btnGoldenOK = "node/imgBG/Content/btnGoldenOK",
  txtGoldenConsume = "node/imgBG/Content/txtGoldenInfo/txtGoldenTips",
  UIAddAndSubGroupJin = "node/imgBG/Content/Template2/UIAddAndSubGroup",
  txtNow = "node/imgBG/Content/txtNow"
}
tbWnd.tbControls = tbControls
tbWnd.nMaxUse = 20
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OpenWindow("UIGeneralHelp", "JMHuMaiFuIdentify")
  end,
  btnSilverOK = function(self)
    if self.nHuMaiFuNum < 1 then
      return me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", Item:GetName(JingMai.nHuMaiFuId, me)))
    end
    if self.nYinXiangExchagneCount < 1 then
      local szMoneyName = Currency:GetName(JingMai.szYinXiangExchangeMoney)
      return me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szMoneyName))
    elseif self.nYinXiangExchagneCount < 1 or 1 == self.nHuMaiFuNum then
      JingMai:C2SHuMaiExchangeReqC(JingMai.tbHMFIdentifyType.IT_YinYin, 1)
    else
      local nMaxCanUse = math.min(self.nYinXiangExchagneCount, self.nHuMaiFuNum)
      nMaxCanUse = math.min(nMaxCanUse, self.nMaxUse)
      local szMoneyName = Currency:GetName(JingMai.szYinXiangExchangeMoney)
      local nNeedMoney = JingMai.nYinXiangExchangePrice * self.nYinXiangExchagneCount
      local nHadMoney = me.GetMoney(JingMai.szYinXiangExchangeMoney)
      if nNeedMoney > nHadMoney then
        return me.CenterMsg(i18n.EasyGet("{0}\228\184\141\232\182\179", szMoneyName))
      end
      Ui:OpenWindow("UIItemBatchPanel", nil, JingMai.nHuMaiFuId, "JMHuMaiFu", nMaxCanUse, {
        JingMai.tbHMFIdentifyType.IT_YinYin
      })
    end
  end,
  btnGoldenOK = function(self)
    if self.nHuMaiFuNum < 1 then
      return me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", Item:GetName(JingMai.nHuMaiFuId, me)))
    end
    if self.nJinXiangExchagneCount < 1 then
      local szMoneyName = Currency:GetName(JingMai.szJinXiangExchangeMoney)
      return me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", Item:GetName(JingMai.nJinXiangId, me)))
    elseif self.nJinXiangExchagneCount < 1 or 1 == self.nHuMaiFuNum then
      JingMai:C2SHuMaiExchangeReqC(JingMai.tbHMFIdentifyType.IT_JinYin, 1)
    else
      local nMaxCanUse = math.min(self.nJinXiangExchagneCount, self.nHuMaiFuNum)
      nMaxCanUse = math.min(nMaxCanUse, self.nMaxUse)
      local szMoneyName = Currency:GetName(JingMai.szJinXiangExchangeMoney)
      local nNeedMoney = JingMai.nJinXiangExchangePrice * self.nJinXiangExchagneCount
      local nHadMoney = me.GetMoney(JingMai.szJinXiangExchangeMoney)
      if nNeedMoney > nHadMoney then
        return me.CenterMsg(i18n.EasyGet("{0}\228\184\141\232\182\179", szMoneyName))
      end
      Ui:OpenWindow("UIItemBatchPanel", nil, JingMai.nHuMaiFuId, "JMHuMaiFu", nMaxCanUse, {
        JingMai.tbHMFIdentifyType.IT_JinYin
      })
    end
  end
}

function tbWnd:OnCreate()
  self.tbSilverItemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid_Silver)
  self.tbSilverItemGrid:SetClickEnable(true)
  self.tbGoldenItemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid_Golden)
  self.tbGoldenItemGrid:SetClickEnable(true)
  self.AddAndSubGroupYin = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroupYin)
  self.AddAndSubGroupYin:BindEvent(function(nValue)
    self.nYinXiangExchagneCount = self:UpdateNumberInput(nValue)
    local szMoneyName = Currency:GetName(JingMai.szYinXiangExchangeMoney)
    local nNeedMoney = JingMai.nYinXiangExchangePrice * self.nYinXiangExchagneCount
    self:UpdateYinXiangConsume(nNeedMoney, szMoneyName)
    return self.nYinXiangExchagneCount
  end)
  self.AddAndSubGroupYin:SetEnable(true)
  self.AddAndSubGroupJin = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroupJin)
  self.AddAndSubGroupJin:BindEvent(function(nValue)
    self.nJinXiangExchagneCount = self:UpdateNumberInput(nValue)
    local szMoneyName = Currency:GetName(JingMai.szJinXiangExchangeMoney)
    local nNeedMoney = JingMai.nJinXiangExchangePrice * self.nJinXiangExchagneCount
    self:UpdateJinXiangConsume(nNeedMoney, szMoneyName)
    return self.nJinXiangExchagneCount
  end)
  self.AddAndSubGroupJin:SetEnable(true)
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.tbSilverItemGrid:SetItemByTemplate(JingMai.nYinXiangId)
  self.tbGoldenItemGrid:SetItemByTemplate(JingMai.nJinXiangId)
  self:InitUI()
  self:UpdateView()
end

function tbWnd:InitUI()
  local szMoneyName = Currency:GetName(JingMai.szYinXiangExchangeMoney)
  self.pPanel:Label_SetText(tbControls.txtSilverPrice, i18n.EasyGet("\229\133\145\230\141\162\229\141\149\228\187\183\239\188\154{0}{1}", JingMai.nYinXiangExchangePrice, szMoneyName))
  szMoneyName = Currency:GetName(JingMai.szJinXiangExchangeMoney)
  self.pPanel:Label_SetText(tbControls.txtGoldenPrice, i18n.EasyGet("\229\133\145\230\141\162\229\141\149\228\187\183\239\188\154{0}{1}", JingMai.nJinXiangExchangePrice, szMoneyName))
end

function tbWnd:UpdateView()
  self.nHuMaiFuNum = me.GetItemCountByTypeInBag(JingMai.nHuMaiFuId)
  local nMinValue = math.min(self.nHuMaiFuNum, 1)
  local nMaxValue = math.min(self.nHuMaiFuNum, self.nMaxUse)
  self.AddAndSubGroupYin:SetNumRange(nMinValue, nMaxValue)
  self.nYinXiangExchagneCount = self.AddAndSubGroupYin:SetInputValue(nMinValue)
  local szMoneyName = Currency:GetName(JingMai.szYinXiangExchangeMoney)
  local nNeedMoney = JingMai.nYinXiangExchangePrice * self.nYinXiangExchagneCount
  self:UpdateYinXiangConsume(nNeedMoney, szMoneyName)
  self.AddAndSubGroupJin:SetNumRange(nMinValue, nMaxValue)
  self.nJinXiangExchagneCount = self.AddAndSubGroupJin:SetInputValue(nMinValue)
  local szMoneyName = Currency:GetName(JingMai.szJinXiangExchangeMoney)
  local nNeedMoney = JingMai.nJinXiangExchangePrice * self.nJinXiangExchagneCount
  self:UpdateJinXiangConsume(nNeedMoney, szMoneyName)
  self.pPanel:Label_SetText(tbControls.txtNow, i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\230\138\164\232\132\137\231\172\166\194\183\230\156\170\233\137\180\229\174\154\230\149\176\233\135\143\239\188\154{0}", self.nHuMaiFuNum))
end

function tbWnd:UpdateYinXiangConsume(nNeedMoney, szMoneyName)
  self.pPanel:Label_SetText(tbControls.txtSilverConsume, i18n.EasyGet("\230\182\136\232\128\151\239\188\154{0}{1}", nNeedMoney, szMoneyName))
  local nHadMoney = me.GetMoney(JingMai.szYinXiangExchangeMoney)
  if nNeedMoney <= nHadMoney then
    self.pPanel:Label_SetColorByName(tbControls.txtSilverConsume, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtSilverConsume, "NotEnough")
  end
end

function tbWnd:UpdateJinXiangConsume(nNeedMoney, szMoneyName)
  self.pPanel:Label_SetText(tbControls.txtGoldenConsume, i18n.EasyGet("\230\182\136\232\128\151\239\188\154{0}{1}", nNeedMoney, szMoneyName))
  local nHadMoney = me.GetMoney(JingMai.szJinXiangExchangeMoney)
  if nNeedMoney <= nHadMoney then
    self.pPanel:Label_SetColorByName(tbControls.txtGoldenConsume, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtGoldenConsume, "NotEnough")
  end
end

function tbWnd:OnSyncItemEvent(nItemId, bUpdateAll)
  self:UpdateView()
end

function tbWnd:UpdateNumberInput(nNum)
  if nNum > self.nHuMaiFuNum then
    nNum = self.nHuMaiFuNum
    me.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\229\134\141\229\162\158\229\138\160\228\186\134"))
    return nNum
  end
  if nNum <= 0 and self.nHuMaiFuNum > 0 then
    nNum = 1
    me.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\229\134\141\229\135\143\229\176\145\228\186\134"))
    return nNum
  end
  return nNum
end
