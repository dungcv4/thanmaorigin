LocalData.FileType = {
  GLOBAL = 1,
  ACCOUNT = 2,
  ROLE = 3
}
local szLocalDataGlobalPath = "%sUser/GlobalData.json"
local szLocalDataAccountPath = "%sUser/Account/%s.json"
local szLocalDataRolePath = "%sUser/Role/%s.json"
local szLocalDataAccountDir = "%sUser/Account"
local szLocalDataRoleDir = "%sUser/Role"

function LocalData:PathInit()
  self.tbGlobalData = nil
  self.tbAccountData = nil
  self.tbRoleData = nil
  self.szCurAccount = ""
  self.nCurRoleId = 0
  local Directory = CS.System.IO.Directory
  local szAccountDir = string.format(szLocalDataAccountDir, g_szUserPath)
  local szRoleDir = string.format(szLocalDataRoleDir, g_szUserPath)
  if not Directory.Exists(szAccountDir) then
    Directory.CreateDirectory(szAccountDir)
  end
  if not Directory.Exists(szRoleDir) then
    Directory.CreateDirectory(szRoleDir)
  end
end

function LocalData:GetValue(nFileType, key, defaultVal)
  if not key then
    assert(false)
    return nil
  end
  local tbFlag = self:Update(nFileType, key)
  local value = tbFlag[key]
  if nil == value then
    value = defaultVal
  end
  return value
end

function LocalData:SetValue(nFileType, key, value)
  if not key then
    assert(false)
    return false
  end
  if nil == value then
    assert(false)
    return false
  end
  local tbFlag = self:Update(nFileType, key)
  tbFlag[key] = value
  self:Flush(nFileType)
  return true
end

function LocalData:DeleteValue(nFileType, key)
  if not key then
    assert(false)
    return false
  end
  local tbFlag = self:Update(nFileType, key)
  tbFlag[key] = nil
  self:Flush(nFileType)
  return true
end

local UpdateDataFunc = {
  [LocalData.FileType.GLOBAL] = function(self, key)
    local szFilePath = string.format(szLocalDataGlobalPath, g_szUserPath)
    if not self.tbGlobalData then
      local szFileContent, nLen = Lib:ReadFile_W(szFilePath, "r")
      if 0 == nLen then
        szFileContent = "{}"
      end
      self.tbGlobalData = Lib:DecodeJson(szFileContent)
    end
    return self.tbGlobalData
  end,
  [LocalData.FileType.ACCOUNT] = function(self, key)
    local szFilePath = string.format(szLocalDataAccountPath, g_szUserPath, GetAccountName())
    if not self.tbAccountData or self.szCurAccount ~= GetAccountName() then
      local szFileContent, nLen = Lib:ReadFile_W(szFilePath, "r")
      if 0 == nLen then
        szFileContent = "{}"
      end
      self.szCurAccount = GetAccountName()
      self.tbAccountData = Lib:DecodeJson(szFileContent)
    end
    return self.tbAccountData
  end,
  [LocalData.FileType.ROLE] = function(self, key)
    local szFilePath = string.format(szLocalDataRolePath, g_szUserPath, me.dwID)
    if not self.tbRoleData or self.nCurRoleId ~= me.dwID then
      local szFileContent, nLen = Lib:ReadFile_W(szFilePath, "r")
      if 0 == nLen then
        szFileContent = "{}"
      end
      self.nCurRoleId = me.dwID
      self.tbRoleData = Lib:DecodeJson(szFileContent)
    end
    return self.tbRoleData
  end
}

function LocalData:Update(nFileType, key)
  return UpdateDataFunc[nFileType](self, key)
end

function LocalData:Flush(nFileType)
  if not nFileType then
    assert(false)
    return
  end
  local tbData, szFilePath
  if nFileType == self.FileType.GLOBAL then
    tbData = self.tbGlobalData
    szFilePath = string.format(szLocalDataGlobalPath, g_szUserPath)
  elseif nFileType == self.FileType.ACCOUNT then
    tbData = self.tbAccountData
    szFilePath = string.format(szLocalDataAccountPath, g_szUserPath, GetAccountName())
  elseif nFileType == self.FileType.ROLE then
    tbData = self.tbRoleData
    szFilePath = string.format(szLocalDataRolePath, g_szUserPath, me.dwID)
  end
  local nRet = cjson.encode_sparse_array()
  if not nRet then
    cjson.encode_sparse_array(true)
  end
  local bResult = Lib:WriteFile_W(szFilePath, "w+", Lib:EncodeJson(tbData))
  if not nRet then
    cjson.encode_sparse_array(false)
  end
  if not bResult then
    Log("LocalData:Flush ERR ?? file is nil !!" .. szFilePath)
    return
  end
end
