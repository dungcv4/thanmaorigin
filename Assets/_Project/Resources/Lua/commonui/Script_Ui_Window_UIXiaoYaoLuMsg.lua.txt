local tbWnd = Ui:CreateClass("UIXiaoYaoLuMsg", "UIPanel")
local tbControls = {
  txtElectDaShiXiong = "ElectTitle/txtElectDaShiXiong",
  Template = "imgList/List/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  RemoteServer.RequestJoinXoyoGame("XYGLastRankReq")
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSV(tbData)
  local tbOtherData = {}
  for _, tbInfo in pairs(tbData) do
    local info = self:GetInfo(tbInfo)
    table.insert(tbOtherData, info)
  end
  
  local function fnShowItem(tbElement, nIdx)
    local info = tbOtherData[nIdx]
    if nIdx <= 3 then
      tbElement.pPanel:SetActive("txtRank", false)
      tbElement.pPanel:SetActive("UIRankNo/imgNO" .. nIdx, true)
    end
    tbElement.pPanel:Label_SetText("txtRank", nIdx)
    tbElement.pPanel:Sprite_SetSprite("imgFaction", info.spFaction)
    tbElement.pPanel:Label_SetText("txtName", info.szName)
    tbElement.pPanel:Label_SetText("txtKin", info.szKinName)
    tbElement.pPanel:Label_SetText("txtCard", info.nCardNum)
  end
  
  self.tbScrollView:Reset()
  for i = 1, #tbOtherData do
    self.tbScrollView:SetElement(i, function(tbElement)
      fnShowItem(tbElement, i)
    end)
  end
end

function tbWnd:GetInfo(info)
  if not info then
    return nil
  end
  return {
    szKinName = info.szKinName,
    szName = info.szName,
    spFaction = Faction:GetIcon(info.nFaction),
    nCardNum = info.nCardNum
  }
end
