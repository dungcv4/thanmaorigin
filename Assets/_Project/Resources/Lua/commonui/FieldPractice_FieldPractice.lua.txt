function FieldPractice:CheckCanUseXiuLianDan(pPlayer, nTime)
  if self.Cfg.nXiuLianLevel > pPlayer.nLevel then
    return false, i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}", self.Cfg.nXiuLianLevel)
  end
  if GlobalHangUp:IsGlobalHangUpMap(pPlayer.nMapTemplateId) and GlobalHangUp:IsForbiddenTime() then
    return false, GlobalHangUp:GetForbiddenUsingItemMsg()
  end
  local nStateTime = self:GetCurXiuLianTime(pPlayer)
  nStateTime = nStateTime + nTime
  if nStateTime > tbDef.nMaxOpenTime then
    return false, i18n.EasyGet("\230\156\128\229\164\154\230\139\165\230\156\137{0}\229\176\143\230\151\18210\229\128\141\228\191\174\231\130\188\230\151\182\233\151\180", math.floor(tbDef.nMaxOpenTime / 60 / 60))
  end
  local nMaxExp = math.min(tbDef.nMaxResidueExp, math.floor(tbDef.nMaxOpenTime * self.Cfg.nXiuLianExpTiLi * pPlayer.GetBaseAwardExp() / tbDef.nExpBaseTime))
  local nNewExp = pPlayer.GetUserValue(tbDef.nSaveGroupID, tbDef.nSaveResidueExp) + math.floor(self.Cfg.nXiuLianExpTiLi * pPlayer.GetBaseAwardExp() * nTime / tbDef.nExpBaseTime)
  if nMaxExp < nNewExp then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\173\137\231\186\167\230\156\128\229\164\154\230\139\165\230\156\137{0}\231\130\185\229\141\129\229\128\141\231\187\143\233\170\140", nMaxExp)
  end
  return true, ""
end

function FieldPractice:GetCurXiuLianTime(pPlayer)
  local nEndTime = pPlayer.GetUserValue(tbDef.nSaveGroupID, tbDef.nSaveEndTime)
  if self:IsPlayState(pPlayer) then
    local nNow = GetTime()
    if nEndTime <= nNow then
      nEndTime = nNow
    end
    return nEndTime - nNow
  else
    if nEndTime <= 0 then
      nEndTime = 0
    end
    return nEndTime
  end
end

function FieldPractice:IsXiuLianTime(pPlayer)
  local nStateTime = self:GetCurXiuLianTime(pPlayer)
  if nStateTime <= 0 then
    return false
  end
  return true
end

function FieldPractice:GetLeftTime(pPlayer)
  local nEndTime = pPlayer.GetUserValue(tbDef.nSaveGroupID, tbDef.nSaveEndTime)
  if self:IsPlayState(pPlayer) then
    local nNow = GetTime()
    if nEndTime <= nNow then
      return 0
    end
    return nEndTime - nNow
  else
    return nEndTime
  end
end

function FieldPractice:IsPlayState(pPlayer)
  return 1 == pPlayer.GetUserValue(tbDef.nSaveGroupID, tbDef.nSavePlayState)
end
