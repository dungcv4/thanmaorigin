CustomMusicPlayer.nCustomMusicEventId = 1000
CustomMusicPlayer.PlayerPrefs = CS.UnityEngine.PlayerPrefs
CustomMusicPlayer.sPlayerPrefsKey = "customMusic.enable"
CustomMusicPlayer.nCurIndex = 1
CustomMusicPlayer.tbBgmEventList = {}

function CustomMusicPlayer:IsVisiable()
  return false
end

function CustomMusicPlayer:IsEnable()
  if not self:IsVisiable() then
    return false
  end
  local nEnable = self.PlayerPrefs.GetInt(self.sPlayerPrefsKey, 0)
  return 1 == nEnable
end

function CustomMusicPlayer:Enable(bEnable)
  local nEnable = 0
  if bEnable then
    nEnable = 1
  end
  self.PlayerPrefs.SetInt(self.sPlayerPrefsKey, nEnable)
  if bEnable then
    CustomMusicPlayer:PlayRandomMusic()
  else
    Map:StopMapSound()
    Map:PlayMapSound(me.nMapTemplateId)
  end
end

function CustomMusicPlayer:PlayPreMusic()
  local nCount = #self.tbBgmEventList
  if self.nCurIndex <= 1 then
    self.nCurIndex = nCount
  else
    self.nCurIndex = self.nCurIndex - 1
  end
  local nEventId = self.tbBgmEventList[self.nCurIndex]
  Client.AudioModule.PlayMusic(nEventId)
end

function CustomMusicPlayer:PlayNextMusic()
  local nCount = #self.tbBgmEventList
  if nCount <= self.nCurIndex then
    self.nCurIndex = 1
  else
    self.nCurIndex = self.nCurIndex + 1
  end
  local nEventId = self.tbBgmEventList[self.nCurIndex]
  Client.AudioModule.PlayMusic(nEventId)
end

function CustomMusicPlayer:PlayRandomMusic()
  self.nCurIndex = MathRandom(1, #self.tbBgmEventList)
  local nEventId = self.tbBgmEventList[self.nCurIndex]
  Client.AudioModule.PlayMusic(nEventId)
end
