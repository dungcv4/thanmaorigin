ChangeName.OPEN_LEVEL = 11
ChangeName.SAVE_GROUP = GetGroupKey("ChangeName")
ChangeName.KEY_TIMES = 1
ChangeName.KEY_FREE = 2
ChangeName.KEY_REMAIN_TIMES = 3
ChangeName.szChangeCostType = "YuanBao"
ChangeName.tbChangeCost = {
  [1] = 800,
  [2] = 800,
  [3] = 800
}

function ChangeName:GetFreeChange(pPlayer)
  return 0 == pPlayer.GetUserValue(ChangeName.SAVE_GROUP, ChangeName.KEY_FREE)
end

function ChangeName:GetRemainTimes(pPlayer)
  return pPlayer.GetUserValue(ChangeName.SAVE_GROUP, ChangeName.KEY_REMAIN_TIMES)
end

function ChangeName:AddRemainTimes(pPlayer, nCount)
  local nCurCount = self:GetRemainTimes(pPlayer)
  pPlayer.SetUserValue(ChangeName.SAVE_GROUP, ChangeName.KEY_REMAIN_TIMES, nCurCount + nCount)
end

function ChangeName:ReduceRemainTimes(pPlayer, nCount)
  local nCurCount = self:GetRemainTimes(pPlayer)
  local nAfterReduce = nCurCount - nCount
  if nAfterReduce < 0 then
    Log(string.format("[ChangeName:ReduceRemainTimes] reduce count is larger than remain count, player id: %d, reduce count: %d, remain count: %d", pPlayer.dwID, nCount, nCurCount))
    nAfterReduce = 0
  end
  pPlayer.SetUserValue(ChangeName.SAVE_GROUP, ChangeName.KEY_REMAIN_TIMES, nAfterReduce)
end

function ChangeName:GetCostByCount(pPlayer)
  local nCurCount = self:GetChangeNameCount(pPlayer)
  local nCost = self.tbChangeCost[#self.tbChangeCost]
  if self.tbChangeCost[nCurCount + 1] then
    nCost = self.tbChangeCost[nCurCount + 1]
  end
  return self.szChangeCostType, nCost
end

function ChangeName:GetChangeNameCount(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.KEY_TIMES)
end

function ChangeName:AddChangeNameCount(pPlayer, nAdd)
  local nCurChangeCount = pPlayer.GetUserValue(self.SAVE_GROUP, self.KEY_TIMES)
  nCurChangeCount = nCurChangeCount + nAdd
  pPlayer.SetUserValue(self.SAVE_GROUP, self.KEY_TIMES, nCurChangeCount)
end
