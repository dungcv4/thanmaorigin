local function SplitClz(clz)
  local tb = {}
  for word in string.gmatch(clz, "([^.]+)") do
    table.insert(tb, word)
  end
  return tb
end

local function GetType(clz)
  local tb = SplitClz(clz)
  local cur = CS
  for k, t in ipairs(tb) do
    cur = cur[t]
    if nil == cur then
      break
    end
  end
  return cur
end

luanet = {}

function luanet.import_type(clz)
  xlua.import_type(clz)
  local tp = GetType(clz)
  return tp
end
