local tbWnd = Ui:CreateClass("UINewInfoMsg", "UIPanel")
local tbControls = {WebView = "WebView", Normal = "Normal"}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self:SetPanelBulletin()
end

function tbWnd:OnClose()
end

function tbWnd:SetPanelBulletin()
  local nIndex = self.tbOwner:GetCurPanelIndex()
  if nIndex <= ClientBulletin.MAX_REDPOINT then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ClientBulletinBegin - 1 + nIndex)
  end
  if self.tbOwner.tbJsonData[nIndex].content_type == "url" then
    self.pPanel:SetActive(tbControls.Normal, false)
    self.pPanel:SetActive(tbControls.WebView, true)
    self.pPanel:ShowUrl(tbControls.WebView, self.tbOwner.tbJsonData[nIndex].content, 0, 0)
  else
    self.pPanel:SetActive(tbControls.Normal, true)
    self.pPanel:SetActive(tbControls.WebView, false)
    self.tbOwner.bulletinContent:ShowByIndex(nIndex - 1)
  end
end
