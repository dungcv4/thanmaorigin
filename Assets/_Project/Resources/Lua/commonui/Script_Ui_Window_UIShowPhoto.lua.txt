local tbWnd = Ui:CreateWindow("UIShowPhoto")
local tbControls = {
  btnReturn = "panel2/btnReturn",
  btnSave = "panel2/buttons/btnSave",
  btnShare = "panel2/buttons/btnShare",
  RawPhoto = "panel2/photo",
  RawPhotoFrame = "panel2/frame"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnReturn = function(self)
    if Ui:WindowVisible("UITakePhoto") then
      local ui = Ui:GetWindow("UITakePhoto")
      ui:ShowPanel()
    end
    self:Close()
  end,
  btnSave = function(self)
    self:OnClickSave()
    Ui:CloseWindow("UITakePhoto")
    self:Close()
  end
}

function tbWnd:OnCreate()
  self:SetGroup(2)
  self.pPanel:SetActive(tbControls.btnShare, Client.EnableSharePhoto)
end

function tbWnd:OnOpen(fnOnReturn)
  self.pPanel:RawImage_SetImage(tbControls.RawPhoto, Ui.SceneCameraController.GetRenderTexture2D())
  self.fnOnReturn = fnOnReturn
end

function tbWnd:OnClose()
end

function tbWnd:OnClickSave()
  if Sdk:GetOsType() ~= Sdk.eOSType_iOS then
    return self:OnClickSaveHelper()
  end
  Log("UIShowPhoto:OnClickIosSave")
  local nPermission = Ui.SceneCameraController.GetWritePhotoPermissionState()
  if 1 == nPermission then
    self:OnClickSaveHelper()
  elseif 0 == nPermission then
    Ui.SceneCameraController.GetWritePhotoPermission()
  else
    me.CenterMsg(i18n.EasyGet("\230\178\161\230\156\137\228\191\157\229\173\152\229\155\190\231\137\135\230\157\131\233\153\144\239\188\140\230\139\141\231\133\167\229\164\177\232\180\165"))
  end
end
