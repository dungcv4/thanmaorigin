Require("Script/Fuben/FubenUIController.lua")
XiSuiDao = XiSuiDao or {}

function XiSuiDao:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    self.Def.nMapId
  }, self)
end

function XiSuiDao:OnFubenMapLoaded(pPlayer)
end

function XiSuiDao:OnLeave()
  FubenUIController:CloseFubenUI()
  self.IsCanExit = nil
end

function XiSuiDao:SetCanExit(bCanExit)
  self.IsCanExit = bCanExit
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function XiSuiDao:IsShowTipWhenQuit()
  return false
end

function XiSuiDao:IsNeedExitFubenButton()
  return self.IsCanExit or false
end

function XiSuiDao:RequieExit()
  local function funcOK()
    FubenUIController:RequireQuit()
  end
  
  local function funcCancel()
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.Get(XiSuiDao.Msg.LeaveConfirm), {
    {funcOK},
    {funcCancel}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

XiSuiDao:RegisterFubenUIChecker()

function XiSuiDao:BuyItem()
  local tbWare = Shop:GetAWare2("YuanBao", XiSuiDao.Def.nDefaultBuyItem)
  if not tbWare then
    return
  end
  local nRemainCount = Shop:GetShopRemainCount(tbWare, me)
  Ui:OpenWindow("UIShopBuy", tbWare, true, nRemainCount)
end
