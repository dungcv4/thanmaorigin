  end
  Ui:OpenWindow("UIWelfare", "PANEL_TYPE_MONEY_TREE")
end

function GainCurrency:JumpToGetHunShi()
  if Ui.bInZone then
    Ui:SendForbiddenMsgWhenInZone()
    return
  end
  local tbWare = Shop:GetAWare(GainCurrency.nHunshiBoxId)
  if tbWare then
    Ui:OpenWindow("UIShop", 1, GainCurrency.nHunshiBoxId)
    local nRemainCount = Shop:GetShopRemainCount(tbWare, me)
    if 0 == nRemainCount then
      return
    end
    Ui:OpenWindow("UIShopBuy", tbWare, true)
  end
end

function GainCurrency:JumpToDragonBoatGift()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.DragonBoatGift)
  if pInstance and pInstance:IsOpen() then
    Ui:CloseWindow("UIShopBuy")
    Ui:CloseWindow("UINpcShop")
    Ui:OpenWindow("UIDuanWuGift")
  else
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
  end
end

function GainCurrency:JumpToHuaHaiMiYuan()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.HuaHaiMiYuan)
  if pInstance and pInstance:IsOpen() then
    Ui:CloseWindow("UIShopBuy")
    Ui:CloseWindow("UINpcShop")
    local tbParam = Lib:SplitStrTry2Num(pInstance.tbConfig.szExtraParam, "|")
    local pFunc = ActivityMgr.tbJumpFunc[pInstance.tbConfig.szJumpType]
    pFunc(pInstance.nId, table.unpack(tbParam))
  else
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
  end
end

function GainCurrency:JumpToLimitShop()

end

function GainCurrency:AskIfJump(szName)
  Ui:OpenWindow("UIMessageBox", GainCurrency.tbCurrencyInfo[szName].fnGetDesc(), {
    {
      GainCurrency.tbCurrencyInfo[szName].fnJump
    },
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function GainCurrency:DirectJump(szName)
  if GainCurrency.tbCurrencyInfo[szName] == nil then
    return false
  end
  GainCurrency.tbCurrencyInfo[szName].fnJump()
  return true
end

function GainCurrency:ReadyForJump(szName)
  if GainCurrency.tbCurrencyInfo[szName] == nil then
    return false
  end
  if Ui.bInZone then
    Ui:SendForbiddenMsgWhenInZone()
    return false
  end
  if Ui:GetWindow("UIMessageBox") then
    Timer:Register(2, GainCurrency.AskIfJump, GainCurrency, szName)
  else
    GainCurrency:AskIfJump(szName)
  end
  return true
end
