local tbWnd = Ui:CreateClass("UIFirework", "UIPanel")
local tbControls = {
  szBigListTemplate = "List02/List/Viewport/Content/Template",
  szSmallListTemplate = "List01/List/Viewport/Content/Template",
  szTitle = "txtTitle",
  szTime = "txtTime01",
  szBigList = "List02",
  szSmallList = "List01",
  btnHelp = "txtTitle/btnHelp",
  btnGo = "btnGo"
}
tbWnd.tbControls = tbControls
tbWnd.nSmallListPerLine = 4
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.nIndex = nIndex
  self:LoadActivityData()
  self.szHelpKey = tbInfo.szHelpKey
  self.pPanel:Label_SetText(self.szTitle, tbInfo.szName)
  local szTime = string.format("<color=#1FFB24>%s</color>", tbInfo.szShowTime)
  local nStartTime = self.pInstance:GetStartTime()
  local nEndTime = self.pInstance:GetEndTime()
  if nStartTime < GetTime() then
    local nDiff = nEndTime - GetTime()
    szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", tbInfo.szShowTime, Lib:TimeDesc2(nDiff))
  end
  self.bTimeOut = nEndTime < GetTime()
  if self.bTimeOut then
    szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> \229\183\178\231\187\147\230\157\159", tbInfo.szShowTime)
  end
  self.pPanel:Label_SetText(self.szTime, szTime)
  self:UpdateListView(tbInfo.tbAward)
  self:BindFindPath()
end

function tbWnd:UpdateListView(tbAward)
  if not tbAward then
    return
  end
  local bUseSmall = #tbAward < self.nSmallListPerLine
  local szTemplate = bUseSmall and self.szSmallListTemplate or self.szBigListTemplate
  self.pPanel:SetActive(self.szBigList, not bUseSmall)
  self.pPanel:SetActive(self.szSmallList, bUseSmall)
  if self.szTemplate ~= szTemplate then
    if self.tbListView then
      self.tbListView:Clear()
    end
    self.tbListView = Ui:NewScrollView(self, szTemplate)
    self.szTemplate = szTemplate
  end
  self.pPanel:SetActive(szTemplate, bUseSmall)
  self.tbListView:Reset()
  for i = 1, #tbAward do
    self.tbListView:SetElement(i, function(tbElement)
      self:UpdateListElement(tbElement, tbAward[i])
    end, true)
  end
end

function tbWnd:UpdateListElement(tbElement, tbAward)
  local tbGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  if "Item" == tbAward[1] then
    tbGrid:SetItemByTemplate(tbAward[2], tbAward[3], nil, nil, {bShowTip = true})
    local szName = Item:GetName(tbAward[2], me)
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  else
    tbGrid:SetDigitalItem(tbAward[1], tbAward[3])
    local szName = Currency:GetName(tbAward[1])
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  end
  tbGrid:SetClickEnable(true)
end

function tbWnd:BindFindPath()
  Ui:Button_BindEvent(self.pPanel, tbControls.btnGo, self.GoTo, self)
end

function tbWnd:LoadActivityData()
  local nActId = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].nActId
  self.pInstance = ActivityMgr.EntranceSchedule:BuildInstance(nActId, ActivityMgr.Def.ClassNameList.EMeiFireWork)
end

function tbWnd:GoTo()
  if self.bTimeOut then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\183\178\231\187\147\230\157\159"))
    return
  end
  self.pInstance:DoCommonJump()
end

function tbWnd:OnClose()
  if self.tbListView then
    self.tbListView:Reset()
  end
end
