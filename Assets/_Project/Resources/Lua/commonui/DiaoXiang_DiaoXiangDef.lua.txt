Require("CommonScript/DiaoXiang/DiaoXiangType.lua")
DiaoXiang.szSaveKey = "DiaoXiang"
DiaoXiang.TASK_WORSHIP_GROUP = GetGroupKey("DiaoXiangWorship")
DiaoXiang.TASK_WORSHIP_ID = {
  [DiaoXiang.Type.WuShuangHaoJie] = 1
}

function DiaoXiang:LoadConfig()
  DiaoXiang.tbConfig = Lib:EasyLoadTabFile("Setting/DiaoXiang/DiaoXiangCfg.tab", {"nType"})
  assert(DiaoXiang.tbConfig)
  for nType, tbCfg in pairs(DiaoXiang.tbConfig) do
    local tbOrg = Lib:EasyLoadTabFile(tbCfg.szFilePath, {"nId"}, {
      "nId",
      "nTemplateID",
      "nResID",
      "nMapID",
      "nPosX",
      "nPosY",
      "szName",
      "nDirect",
      "nTitleId"
    })
    assert(tbOrg)
    local tbDiaoxiangNpc = {}
    for nId, tb in pairs(tbOrg) do
      tbDiaoxiangNpc[nId] = {
        nTemplateID = tb.nTemplateID,
        nResID = tb.nResID,
        nMapID = tb.nMapID,
        nPosX = tb.nPosX,
        nPosY = tb.nPosY,
        szName = tb.szName,
        nDirect = tb.nDirect,
        nTitleId = tb.nTitleId
      }
    end
    DiaoXiang.tbConfig[nType].tbNpc = SetReadOnly(tbDiaoxiangNpc)
  end
end

function DiaoXiang:NeedCheckFaction(nType)
  return nType == DiaoXiang.Type.MenPaiJingJi
end

function DiaoXiang:GetConfig(nType)
  return DiaoXiang.tbConfig[nType]
end

function DiaoXiang:GetNpcConfig(nType, nIndex)
  return DiaoXiang.tbConfig[nType] and DiaoXiang.tbConfig[nType].tbNpc[nIndex] or nil
end

function DiaoXiang:GetDefaultDialog(nType)
  return DiaoXiang.tbConfig[nType] and DiaoXiang.tbConfig[nType].szDefaultDlg and DiaoXiang.tbConfig[nType].szDefaultDlg or ""
end

DiaoXiang:LoadConfig()
