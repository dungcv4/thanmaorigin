local tbWnd = Ui:CreateWindow("UIBoilingWineEnter")
local tbControls = {
  txtMatchInfo = "node/imgBG/Content/Left/Txt_MatchInfo",
  itemgrid = "node/imgBG/Content/Right/GridItem/Player%d/Cont_Info/Viewport/Content/Template",
  txtNum = "node/imgBG/Content/Right/GridItem/Player%d/Txt_Num",
  txtName = "node/imgBG/Content/Right/GridItem/Player%d/Txt_Skill",
  imgGet = "node/imgBG/Content/Right/GridItem/Player%d/img_Get",
  btnGet = "node/imgBG/Content/Right/GridItem/Player%d/btnOk",
  txtBtn = "node/imgBG/Content/Right/GridItem/Player%d/btnOk/Text",
  btnStart = "node/imgBG/Content/Right/btnStart",
  btnRecruit = "node/imgBG/Content/Right/btnRecruit",
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnStart = function(self)
    me.MsgBox(i18n.EasyGet("\230\152\175\229\144\166\232\191\155\229\133\165\230\177\159\230\185\150\231\133\174\233\133\146\229\175\185\229\177\128"), {
      {
        i18n.szOk,
        JiangHuZhuJiu.CreateGameReqC
      },
      {
        i18n.szCancel
      }
    })
  end,
  btnRecruit = function(self)
    self:SendChatRecruit()
  end,
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("JiangHuZhuJiu")
  end
}

function tbWnd:OnCreate()
  self.pPanel:Label_SetText(self.txtMatchInfo, i18n.Get(JiangHuZhuJiu.TEXT.szMatchInfo))
  self.tbItemGridList = {}
  for i = 1, #JiangHuZhuJiu.AWARDS do
    self.pPanel:Label_SetText(string.format(self.txtName, i), i18n.Get(JiangHuZhuJiu.AWARDS[i].szName))
    local tbAwards = Award:GetAward(JiangHuZhuJiu.AWARDS[i].nAwardId, JiangHuZhuJiu.AWARDS[i].nAwardKey)
    self.tbItemGridList[i] = Ui:NewItemGridScrollView(self, string.format(self.itemgrid, i), 10, 1)
    self.tbItemGridList[i]:UpdateGenericItem(tbAwards, true)
    Ui:Button_BindEvent(self.pPanel, string.format(self.btnGet, i), function()
      RemoteServer.RequestJiangHuZhuJiu("RequestAwardW", i)
    end)
  end
end

function tbWnd:OnOpen()
  self:Update()
  self:RegisteRedPoint()
end

function tbWnd:OnClose()
  self:UnRegistRedPoint()
end

function tbWnd:RegisteRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, string.format(self.btnGet, 1), {
    RedPointMgr.EVENT.RedPoint_JiangHuZhuJiu_Award1
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, string.format(self.btnGet, 2), {
    RedPointMgr.EVENT.RedPoint_JiangHuZhuJiu_Award2
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, string.format(self.btnGet, 3), {
    RedPointMgr.EVENT.RedPoint_JiangHuZhuJiu_Award3
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, string.format(self.btnGet, 4), {
    RedPointMgr.EVENT.RedPoint_JiangHuZhuJiu_Award4
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, string.format(self.btnGet, 1))
  RedPointMgr:DynamicUnRegist(self.pPanel, string.format(self.btnGet, 2))
  RedPointMgr:DynamicUnRegist(self.pPanel, string.format(self.btnGet, 3))
  RedPointMgr:DynamicUnRegist(self.pPanel, string.format(self.btnGet, 4))
end

function tbWnd:Update()
  for nIndex, tbInfo in ipairs(JiangHuZhuJiu.AWARDS) do
    self.pPanel:Label_SetText(string.format(self.txtNum, nIndex), i18n.EasyGet("\230\156\172\229\145\168\229\143\175\233\162\134\229\143\150{0}/{1}", tbInfo.nWeekCount - (me.GetUserValue(JiangHuZhuJiu.nSaveGroup, tbInfo.nGotSaveKey) or 0), tbInfo.nWeekCount))
    local nCanGetNum = me.GetUserValue(JiangHuZhuJiu.nSaveGroup, tbInfo.nSaveKey)
    local nGotNum = me.GetUserValue(JiangHuZhuJiu.nSaveGroup, tbInfo.nGotSaveKey)
    if nGotNum == tbInfo.nWeekCount then
      self.pPanel:SetActive(string.format(self.btnGet, nIndex), false)
      self.pPanel:SetActive(string.format(self.imgGet, nIndex), true)
    elseif nCanGetNum > nGotNum then
      self.pPanel:Button_SetEnable(string.format(self.btnGet, nIndex), true)
      self.pPanel:SetActive(string.format(self.btnGet, nIndex), true)
      self.pPanel:SetActive(string.format(self.imgGet, nIndex), false)
      self.pPanel:SetActive(string.format(self.txtBtn, nIndex), true)
      self.pPanel:Label_SetText(string.format(self.txtBtn, nIndex), i18n.EasyGet("\229\143\175\233\162\134\229\143\150"))
    else
      self.pPanel:Button_SetEnable(string.format(self.btnGet, nIndex), true)
      self.pPanel:SetActive(string.format(self.btnGet, nIndex), true)
      self.pPanel:SetActive(string.format(self.imgGet, nIndex), false)
      self.pPanel:SetActive(string.format(self.txtBtn, nIndex), true)
      self.pPanel:Label_SetText(string.format(self.txtBtn, nIndex), i18n.EasyGet("\230\156\170\232\190\190\230\136\144"))
      self.pPanel:Button_SetEnable(string.format(self.btnGet, nIndex), false)
    end
  end
end

function tbWnd:SendChatRecruit()
  if not TeamMgr:HasTeam() then
    TeamMgr:CreateOnePersonTeam()
    local szMsg = i18n.Get(TeamMgr.Terms.RecruitMsg, math.min(TeamMgr:GetMembersCount() + 1, TeamMgr.Cfg.MAX_MEMBER_COUNT), TeamMgr.Cfg.MAX_MEMBER_COUNT, i18n.EasyGet("\230\177\159\230\185\150\231\133\174\233\133\146"))
    TeamMgr:SendRecruitMsg(szMsg)
  else
    local szMsg = i18n.Get(TeamMgr.Terms.RecruitMsg, TeamMgr:GetMembersCount(), TeamMgr.Cfg.MAX_MEMBER_COUNT, i18n.EasyGet("\230\177\159\230\185\150\231\133\174\233\133\146"))
    TeamMgr:SendRecruitMsg(szMsg)
  end
end
