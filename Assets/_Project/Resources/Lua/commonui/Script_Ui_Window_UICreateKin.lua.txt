local tbWnd = Ui:CreateWindow("UICreateKin")

function tbWnd:NotifyEvents()
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnOK = "node/imgBG/Content/btnOK",
  txtCostNum = "node/imgBG/Content/Cost/txtCostNum",
  inputKinNotice = "node/imgBG/Content/KinNotice/inputKinNotice",
  inputKinName = "node/imgBG/Content/KinName/inputKinName"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    self:OnClickOK()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self.pPanel:Input_SetText(self.inputKinName, "")
  self.pPanel:Input_SetText(self.inputKinNotice, "")
  self.pPanel:Label_SetText(self.txtCostNum, Kin.Cfg.nCreateKinNeedYuanBao)
end

function tbWnd:OnClickOK()
  local szKinName = self.pPanel:Input_GetText(self.inputKinName)
  local szKinNotice = self.pPanel:Input_GetText(self.inputKinNotice)
  szKinNotice = szKinNotice or ""
  local nRet = Kin:Create(szKinName, szKinNotice)
  if nRet then
    self:Close()
  end
end
