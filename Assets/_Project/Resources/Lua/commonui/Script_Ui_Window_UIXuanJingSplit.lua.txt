local tbWnd = Ui:CreateWindow("UIXuanJingSplit")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtName = "node/imgBG/panelSplit/imgXuanjing/txtName",
  goInItemGrid = "node/imgBG/panelSplit/imgXuanjing/UIItemGrid",
  goOutItemGrid = "node/imgBG/panelSplit/SplitOutcome/ItemList/Item",
  goAddAndSubGroup = "node/imgBG/panelSplit/imgXuanjing/UIAddAndSubGroup",
  btnSplit = "node/imgBG/panelSplit/btnSplit",
  btnHelp = "node/imgBG/BG/Text/btnHelp"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSplit = function(self)
    self:OnClickSplit()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("XuanJingSplitHelp")
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_DEL_ITEM,
      self.OnDelItem
    },
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem
    }
  }
end

function tbWnd:OnCreate()
  self.tbInItemGrid = Ui:NewExistItemGrid(self, self.goInItemGrid)
  self.tbInItemGrid:SetClickEnable(true)
  self.tbOutItemGridView = Ui:NewItemGridView(self, self.goOutItemGrid)
  self.tbAddAndSubGroup = Ui:NewExistAddAndSubGroup(self, self.goAddAndSubGroup)
  self.tbAddAndSubGroup:BindEvent(function(nValue)
    self:SetNum(nValue)
  end)
end

function tbWnd:OnOpen(nItemId)
  self.nItemId = nItemId
  self.nNum = 1
  self:InitNum()
  self:UpdateData()
end

function tbWnd:InitNum()
  local pItem = me.GetItemInAllSpace(self.nItemId)
  if pItem and pItem.nCount > 1 then
    self.pPanel:SetActive(self.goAddAndSubGroup, true)
    self.tbAddAndSubGroup:SetNumRange(1, pItem.nCount)
    self.tbAddAndSubGroup:SetInputValue(self.nNum)
  else
    self.pPanel:SetActive(self.goAddAndSubGroup, false)
  end
end

function tbWnd:SetNum(nNum)
  if self.nNum == nNum then
    return
  end
  self.nNum = nNum
  self:UpdateData()
end

function tbWnd:UpdateData()
  self:ClearData()
  local pItem = me.GetItemInAllSpace(self.nItemId)
  if not pItem then
    return
  end
  self.szName = i18n.Parse(pItem.szName)
  self.pPanel:Label_SetText(self.txtName, self.szName)
  self.tbInItemGrid:SetItem(pItem.dwId, pItem.dwTemplateId)
  if Item.EquipEnhance:CanSplit(me, self.nItemId, self.nNum) then
    local tbItemList = Item.EquipEnhance:CalcSplitLv(pItem.nLevel, self.nNum)
    for nId, tbItem in ipairs(tbItemList) do
      self.tbOutItemGridView:SetItem(nId, {
        "Item",
        Item.EquipEnhance.XUANJING_ITEM[tbItem[1]],
        tbItem[2]
      }, true)
    end
  end
  self.pPanel:Button_SetEnable(self.btnSplit, true)
end

function tbWnd:ClearData()
  self.pPanel:Label_SetText(self.txtName, "")
  self.tbInItemGrid:SetEmpty()
  self.tbOutItemGridView:Reset()
  self.pPanel:Button_SetEnable(self.btnSplit, false)
end

function tbWnd:OnClickSplit()
  if not self.nItemId then
    return
  end
  local szMsg = i18n.EasyGet("\230\152\175\229\144\166\231\161\174\229\174\154\230\139\134\232\167\163{0}", self.szName)
  
  local function fnConfirm()
    Item.EquipEnhance:SplitReq(me, self.nItemId, self.nNum)
  end
  
  me.MsgBox(szMsg, {
    {
      i18n.szOk,
      fnConfirm
    },
    {
      i18n.szCancel
    }
  })
end

function tbWnd:OnSyncItem(nItemId)
  if nItemId == self.nItemId then
    self.nNum = 1
    self:InitNum()
    self:UpdateData()
  end
end

function tbWnd:OnDelItem(nItemId)
  if nItemId == self.nItemId then
    self:Close()
  end
end
