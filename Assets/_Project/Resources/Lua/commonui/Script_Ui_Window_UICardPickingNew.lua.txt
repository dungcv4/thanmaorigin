function tbWnd:OnClickEmptyClose()
  self.pPanel:SetActive(tbControls.goWndPickPartnerLook, false)

function tbWnd:OnCreate()
  self.pPanel:SetActive(tbControls.btnHelp, true)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
  self.tbMoneyPanel:SetMoneyValue()
  self.tbLookScrollView = Ui:NewScrollView(self, tbControls.goLookListTemplate, nil, false, false)
  self:SetClickEmptyCallback(tbControls.goWndPickPartnerLook)
end

function tbWnd:OnOpen(nIndex)
  self.tbMoneyPanel:SetMoneyValue()
  self:GetPartnerPickItemInfo()
  self:UpdateSubsidySSState()
  self:UpdateTimer()
  self:RegistRedPoint()
  self.nCurCardPickType = tbSubPanelIndex.PANEL_TYPE_YUANBAO
  if CardPicker.bIsInLimitedPoolActivity then
    self.nCurCardPickType = tbSubPanelIndex.PANEL_TYPE_LIMITED
  end
  local nCoinFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_COIN_PICK_TIME)
  local nGoldFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_GOLD_PICK_TIME)
  local nLimitedFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_LIMITED_PICK_TIME)
  local nCurTime = GetTime()
  local nLeftCoinSecond = 28800 - (nCurTime - nCoinFreeTime)
  local nLeftGoldSecond = 86400 - (nCurTime - nGoldFreeTime)
  local nLeftLimitedSecond = 86400 - (nCurTime - nLimitedFreeTime)
  if nLeftCoinSecond < 0 then
    self.nCurCardPickType = tbSubPanelIndex.PANEL_TYPE_COIN
  end
  if nLeftGoldSecond < 0 then
    self.nCurCardPickType = tbSubPanelIndex.PANEL_TYPE_YUANBAO
  end
  if nLeftLimitedSecond < 0 and CardPicker.bIsInLimitedPoolActivity then
    self.nCurCardPickType = tbSubPanelIndex.PANEL_TYPE_LIMITED
  end
  if nIndex then
    self.nCurCardPickType = nIndex
  end
  self:ToggleGroupSelect(self.nCurCardPickType)
  self.pPanel:SetActive(tbControls.goNewLingPaiEffect, true)
  self.pPanel:SetActive(tbControls.effectCardPicking1, false)
  self.pPanel:Animator_SetEnable(tbControls.animCardPicking2, false)
  if CardPicker.tbLimitedPoolUpInfo then
    self.tbLookScrollView:Clear()
    for i = 1, #CardPicker.tbLimitedPoolUpInfo do
      self.tbLookScrollView:SetElement(i, function(tbElement)
        local pPanel = tbElement.pPanel
        local nPartnerId = CardPicker.tbLimitedPoolUpInfo[i]
        pPanel:Label_SetText("Partner1/txtName", i18n.Parse(Partner.tbAllPartnerInfo[nPartnerId].szName))
        pPanel:Sprite_SetSprite("Partner1/imgNameBg", Partner.tbPartnerNameBg[Partner.tbAllPartnerInfo[nPartnerId].nQualityLevel])
        pPanel:Button_BindEvent("Partner1", function()
          Ui:OpenWindow("UIPartnerGalleryDetail", nPartnerId)
        end)
      end)
    end
  end
end

function tbWnd:OnClose()
  if self.nFreeTimer then
    Timer:Close(self.nFreeTimer)
    self.nFreeTimer = nil
  end
  self:UnRegistRedPoint()
end

function tbWnd:StartCardPick(szPickType)
  if self.nEffectTimer then
    Timer:Close(self.nEffectTimer)
    self.nEffectTimer = nil
  end
  self:SetAllButtonState(false)
  self.pPanel:SetActive(tbControls.effectCardPicking1, true)
  self.pPanel:Animator_SetEnable(tbControls.animCardPicking2, true)
  self.nEffectTimer = Timer:Register(Env.GAME_FPS * 1.5, self.StartCardPick_Real, self, szPickType)
end

function tbWnd:StartCardPick_Real(szPickType)
  self.pPanel:SetActive(tbControls.effectCardPicking1, false)
  self.pPanel:Animator_SetEnable(tbControls.animCardPicking2, false)
  CardPicker:RequestCardPick(szPickType)
  self:SetAllButtonState(true)
  return false
end

function tbWnd:GetPartnerPickItemInfo()
  self:OnChangeMoney()
  self.pPanel:Label_SetText(tbControls.txtCoinPickValue, CardPicker.Def.nCoinCost)
  self.pPanel:Label_SetText(tbControls.txtCoinTenPickValue, CardPicker.Def.nCoinTenCost)
  local nLimitedCardPickTicketCount = me.GetItemCountInBags(CardPicker.Def.nPartnerLimitedPickItemId)
  self.pPanel:Label_SetText(tbControls.txtLimitedPickValue, CardPicker.Def.nLimitedCost)
  if nLimitedCardPickTicketCount > 0 and nLimitedCardPickTicketCount < CardPicker.Def.nLimitedTenCost then
    self.pPanel:Label_SetText(tbControls.txtLimitedTenPickValue, string.format("%d/%d", nLimitedCardPickTicketCount, CardPicker.Def.nLimitedTenCost))
  else
    self.pPanel:Label_SetText(tbControls.txtLimitedTenPickValue, CardPicker.Def.nLimitedTenCost)
  end
  local nCardPickTicketCount = me.GetItemCountInBags(CardPicker.Def.nPartnerPickItemId)
  local szPickItemIconPath = "UI/icon/item/partner/partnerpick.png"
  local szCoinIconPath = "UI/icon/money/icon_coin.png"
  local szYuanbaoIconPath = "UI/icon/money/icon_yuanbao.png"
  local szLimitedPickItemIconPath = "UI/icon/item/partner/partnerpick_Gold.png"
  if nCardPickTicketCount > 0 then
    self.pPanel:Sprite_SetSprite(tbControls.imgIconYuanBaoPick, szPickItemIconPath)
    self.pPanel:Sprite_SetSprite(tbControls.imgIconYuanBaoTenPick, szPickItemIconPath)
    self.pPanel:Label_SetText(tbControls.txtYuanBaoPickValue, 1)
    if nCardPickTicketCount > 0 and nCardPickTicketCount < 10 then
      self.pPanel:Label_SetText(tbControls.txtYuanBaoTenPickValue, string.format("%d/10", nCardPickTicketCount))
    else
      self.pPanel:Label_SetText(tbControls.txtYuanBaoTenPickValue, "10")
    end
  else
    self.pPanel:Sprite_SetSprite(tbControls.imgIconYuanBaoPick, szYuanbaoIconPath)
    self.pPanel:Sprite_SetSprite(tbControls.imgIconYuanBaoTenPick, szYuanbaoIconPath)
    self.pPanel:Label_SetText(tbControls.txtYuanBaoPickValue, CardPicker.Def.nGoldCost)
    self.pPanel:Label_SetText(tbControls.txtYuanBaoTenPickValue, CardPicker.Def.nGoldTenCost)
  end
end

function tbWnd:UpdateTimer()
  self:UpdateFreePickButtonState()
  if self.nFreeTimer then
    Timer:Close(self.nFreeTimer)
    self.nFreeTimer = nil
  end
  self.nFreeTimer = Timer:Register(Env.GAME_FPS, self.UpdateFreePickButtonState, self)
end

function tbWnd:UpdateFreePickButtonState()
  local nCoinFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_COIN_PICK_TIME)
  local nGoldFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_GOLD_PICK_TIME)
  local nLimitedFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_LIMITED_PICK_TIME)
  local nCurTime = GetTime()
  local nLeftCoinSecond = 28800 - (nCurTime - nCoinFreeTime)
  local nLeftGoldSecond = 86400 - (nCurTime - nGoldFreeTime)
  local nLeftLimitedSecond = 86400 - (nCurTime - nLimitedFreeTime)
  if nLeftCoinSecond > 0 then
    self.pPanel:SetActive(tbControls.btnCoinFreePick, false)
    self.pPanel:SetActive(tbControls.btnCoinPick, true)
    local nCoinHour, nCoinMin, nCoinSec = Lib:TransferSecond2NormalTime(nLeftCoinSecond)
    self.pPanel:Label_SetText(tbControls.txtCoinCountDown, i18n.EasyGet("{0}:{1}:{2}\229\144\142\229\133\141\232\180\185", nCoinHour, nCoinMin, nCoinSec))
  else
    self.pPanel:SetActive(tbControls.btnCoinFreePick, true)
    self.pPanel:SetActive(tbControls.btnCoinPick, false)
    self.pPanel:Label_SetText(tbControls.txtCoinCountDown, "")
  end
  if nLeftGoldSecond > 0 then
    self.pPanel:SetActive(tbControls.btnYuanBaoFreePick, false)
    self.pPanel:SetActive(tbControls.btnYuanBaoPick, true)
    local nGoldHour, nGoldMin, nGoldSec = Lib:TransferSecond2NormalTime(nLeftGoldSecond)
    self.pPanel:Label_SetText(tbControls.txtYuanBaoCountDown, i18n.EasyGet("{0}:{1}:{2}\229\144\142\229\133\141\232\180\185", nGoldHour, nGoldMin, nGoldSec))
  else
    self.pPanel:SetActive(tbControls.btnYuanBaoFreePick, true)
    self.pPanel:SetActive(tbControls.btnYuanBaoPick, false)
    self.pPanel:Label_SetText(tbControls.txtYuanBaoCountDown, "")
  end
  if nLeftLimitedSecond > 0 then
    self.pPanel:SetActive(tbControls.btnLimitedFreePick, false)
    self.pPanel:SetActive(tbControls.btnLimitedPick, true)
    local nLimitedHour, nLimitedMin, nLimitedSec = Lib:TransferSecond2NormalTime(nLeftLimitedSecond)
    self.pPanel:Label_SetText(tbControls.txtLimitedCountDown, i18n.EasyGet("{0}:{1}:{2}\229\144\142\229\133\141\232\180\185", nLimitedHour, nLimitedMin, nLimitedSec))
  else
    self.pPanel:SetActive(tbControls.btnLimitedFreePick, true)
    self.pPanel:SetActive(tbControls.btnLimitedPick, false)
    self.pPanel:Label_SetText(tbControls.txtLimitedCountDown, "")
  end
  return true
end

function tbWnd:OnChangeMoney()
  local nItemCount = me.GetItemCountInBags(CardPicker.Def.nPartnerPickItemId)
  local nLimitedItemCount = me.GetItemCountInBags(CardPicker.Def.nPartnerLimitedPickItemId)
  local szPickItemIconPath = "UI/icon/item/partner/partnerpick.png"
  local szLimitedPickItemIconPath = "UI/icon/item/partner/partnerpick_Gold.png"
  self.tbMoneyPanel:SetMoneyValue()
  if self.nCurCardPickType == tbSubPanelIndex.PANEL_TYPE_COIN then
    self.tbMoneyPanel:SetDynamicShow(true)
    self.tbMoneyPanel:SetDynamicCurrencyInfo("PartnerPickItem", nItemCount, szPickItemIconPath)
    self.tbMoneyPanel:HideDynamicBtnPlus()
  elseif self.nCurCardPickType == tbSubPanelIndex.PANEL_TYPE_YUANBAO then
    self.tbMoneyPanel:SetDynamicShow(true)
    self.tbMoneyPanel:SetDynamicCurrencyInfo("PartnerPickItem", nItemCount, szPickItemIconPath)
    self.tbMoneyPanel:HideDynamicBtnPlus()
  elseif self.nCurCardPickType == tbSubPanelIndex.PANEL_TYPE_LIMITED then
    self.tbMoneyPanel:SetDynamicShow(true)
    self.tbMoneyPanel:SetDynamicCurrencyInfo("PartnerPickItem", nLimitedItemCount, szLimitedPickItemIconPath)
    self.tbMoneyPanel:HideDynamicBtnPlus()
  end
end
