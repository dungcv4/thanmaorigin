local tbWnd = Ui:CreateClass("UIStoneGrid", "UIPanel")

function Ui:NewExistStoneGrid(tbOwner, szStoneGridKey)
  return Ui:CreatePanel("UIStoneGrid", szStoneGridKey, tbOwner)
end

tbWnd.tbControls = {
  UIItemGrid = "Panel/UIItemGrid",
  txtLevel = "Panel/txtLevel",
  txtNum = "Panel/txtNum"
}

function tbWnd:OnCreate()
  self.grid = Ui:NewExistItemGrid(self, self.UIItemGrid)
end

function tbWnd:SetTemplate(nTemplateId, bShowNum)
  if not Item.Stone:IsStone(nTemplateId) then
    self:Clear()
    return
  end
  self.grid:SetItemByTemplate(nTemplateId, 0)
  self.grid:SetClickEnable(false)
  local nLevel = Item.Stone:GetStoneLevel(nTemplateId)
  self.pPanel:Label_SetText(self.txtLevel, i18n.EasyGet("{0}\231\186\167", nLevel))
  local szNum = ""
  if bShowNum then
    local nCurNum = me.GetItemCountInBags(nTemplateId)
    local nNeedCount = 2
    local szColor = nCurNum >= nNeedCount and "#FFFFCC" or "red"
    szNum = string.format("<color=%s>%s</color>/%s", szColor, nCurNum, nNeedCount)
  end
  self.pPanel:Label_SetText(self.txtNum, szNum)
end

function tbWnd:Clear()
  self.grid:Clear()
  self.pPanel:Label_SetText(self.txtLevel, "")
  self.pPanel:Label_SetText(self.txtNum, "")
end
