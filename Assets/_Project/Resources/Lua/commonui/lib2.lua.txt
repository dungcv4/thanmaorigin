local tbGameObject = {}
Lib.GameObject = tbGameObject

function Lib:ConvertJsonToGameObject(jData)
  local tb = Lib:DecodeJson(jData)
  return tb
end

function Lib:GetComponent(go, comName)
  for k, v in ipairs(go.Component) do
    if v.Type == comName then
      return v
    end
  end
end

function Lib:GetVecPos(szPos)
  local tb1 = Lib:SplitStr(szPos, ">")
  local tb = Lib:SplitStr(tb1[2], ",")
  local n1 = tonumber(tb[1])
  local n2 = tonumber(tb[2])
  local n3 = tonumber(tb[3])
  return {
    n1,
    n2,
    n3
  }
end

function Lib:LogicPos2MapPos(nX, nY)
  local mapX = nX / 800
  local mapY = nY / 1600
  return {
    mapX,
    mapY,
    0
  }
end

function Lib:MapPos2LogicPos(nX, nY)
  local lx = nX * 800
  local ly = nY * 1600
  return {
    lx,
    ly,
    0
  }
end

function Lib:NormalizeDiffDeg(diffY)
  diffY = math.mod(diffY, 360)
  if diffY > 180 then
    diffY = diffY - 360
  end
  if diffY < -180 then
    diffY = 360 + diffY
  end
  return diffY
end

function Lib:RotateByTimer(pNpc, nDir, nSpeed)
  local playerDir = pNpc.GetDir()
  local deltaDir = nDir - playerDir
  deltaDir = math.fmod(deltaDir, 256)
  if deltaDir > 128 then
    deltaDir = deltaDir - 256
  elseif deltaDir < -128 then
    deltaDir = deltaDir + 256
  end
  local absDir = math.abs(deltaDir)
  local nRotateMax = nSpeed
  local bMinus = true
  if absDir > nRotateMax then
    if deltaDir < 0 then
    else
      bMinus = false
    end
  else
    bMinus = nDir < playerDir and true or false
  end
  return bMinus
end

function Lib:ConvertTabToDict(tb)
  local temp = {}
  for k, v in ipairs(tb) do
    temp[v] = true
  end
  return temp
end

function Lib:ForEachMap(tb, func)
  local ret = {}
  for k, v in pairs(tb) do
    ret[k] = func(k, v)
  end
  return ret
end

function Lib:CallFunction(tbTab, szFunc, ...)
  if nil ~= tbTab[szFunc] then
    return tbTab[szFunc](tbTab, ...)
  end
end

function Lib:TablePack(...)
  return {
    n = select("#", ...),
    ...
  }
end

function Lib:ConcatString(...)
  local arg = Lib:TablePack(...)
  local temp = {}
  for i = 1, arg.n do
    table.insert(temp, tostring(arg[i]))
  end
  return table.concat(temp)
end

function Lib:CreateDijkstraPointArr(tbReachable, nStartIndex, nEndIndex)
  local tbCost = {}
  local tbParent = {}
  local tbUnSelected = {}
  local bSuccess = false
  local nCurIndex
  if nStartIndex == nEndIndex then
    return {nEndIndex}
  end
  for i, _ in ipairs(tbReachable) do
    tbUnSelected[i] = true
  end
  nCurIndex = nStartIndex
  tbCost[nCurIndex] = 0
  while true do
    tbUnSelected[nCurIndex] = nil
    local arr = tbReachable[nCurIndex]
    for i, v in ipairs(arr) do
      if tbUnSelected[i] and v >= 0 then
        local nCost = tbCost[nCurIndex] + v
        if not tbCost[i] or nCost < tbCost[i] then
          tbCost[i] = nCost
          tbParent[i] = nCurIndex
        end
      end
    end
    local nMinCost = -1
    local nSelectIndex
    for k, _ in pairs(tbUnSelected) do
      local nCost = tbCost[k]
      if nCost and (nMinCost < 0 or nMinCost > nCost) then
        nMinCost = nCost
        nSelectIndex = k
      end
    end
    if not nSelectIndex then
      break
    end
    nCurIndex = nSelectIndex
    if nSelectIndex == nEndIndex then
      bSuccess = true
      break
    end
  end
  if bSuccess then
    local nParent = nEndIndex
    local tbPathArr = {nParent}
    while true do
      nParent = tbParent[nParent]
      if nParent then
        table.insert(tbPathArr, nParent)
      else
        break
      end
    end
    return tbPathArr
  end
end
