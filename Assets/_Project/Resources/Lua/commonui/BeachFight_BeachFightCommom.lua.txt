Require("CommonScript/BeachFight/BeachFightCfg.lua")
local BeachFightCfg = BeachFight.Cfg

function BeachFight:LoadConfig()
  self:LoadSkillNpcPos()
  self:LoadDebrisNpcPos()
  self:LoadBeginPos()
end

function BeachFight:LoadSkillNpcPos()
  local tbSkillNpcPos = Lib:EasyLoadTabFile(BeachFightCfg.PositionCfg.RefreshSkillPosFile, nil, nil, nil, false)
  if not tbSkillNpcPos then
    Lib:LogErrFormat("BeachFightCommon: LoadSkillNpcPos failed, file %s not exist", BeachFightCfg.PositionCfg.RefreshSkillPosFile)
    return false
  end
  for nIndex, tbPos in pairs(tbSkillNpcPos) do
    tbSkillNpcPos[nIndex] = {
      PosX = tonumber(tbPos.PosX),
      PosY = tonumber(tbPos.PosY)
    }
  end
  BeachFightCfg.tbSkillNpcPos = tbSkillNpcPos
  return true
end

function BeachFight:LoadDebrisNpcPos()
  local tbDebrisNpcPos = Lib:EasyLoadTabFile(BeachFightCfg.PositionCfg.RefreshDebrisPosFile, nil, nil, nil, false)
  if not tbDebrisNpcPos then
    Lib:LogErrFormat("BeachFightCommon: LoadSkillNpcPos failed, file %s not exist", BeachFightCfg.PositionCfg.RefreshDebrisPosFile)
    return false
  end
  for nIndex, tbPos in pairs(tbDebrisNpcPos) do
    tbDebrisNpcPos[nIndex] = {
      PosX = tonumber(tbPos.PosX),
      PosY = tonumber(tbPos.PosY)
    }
  end
  BeachFightCfg.tbDebrisNpcPos = tbDebrisNpcPos
  return true
end

function BeachFight:LoadBeginPos()
  local tbBeginPos = Lib:EasyLoadTabFile(BeachFightCfg.PositionCfg.BeginPosFile, nil, nil, nil, false)
  if not tbBeginPos then
    Lib:LogErrFormat("BeachFightCommon: LoadBeginPos failed, file %s not exist", BeachFightCfg.PositionCfg.BeginPosFile)
    return false
  end
  local tbBeginPosConfig = {}
  for _, tbPos in pairs(tbBeginPos) do
    tbBeginPosConfig[tbPos.nCamp] = tbBeginPosConfig[tbPos.nCamp] or {}
    table.insert(tbBeginPosConfig[tbPos.nCamp], {
      PosX = tonumber(tbPos.PosX),
      PosY = tonumber(tbPos.PosY)
    })
  end
  BeachFightCfg.tbBeginPos = tbBeginPosConfig
  return true
end

BeachFight:LoadConfig()
