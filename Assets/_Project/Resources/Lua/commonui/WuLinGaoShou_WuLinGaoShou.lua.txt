function WuLinGaoShou:LoadSetting()
  self.tbMainSetting = {}
  self.tbBossSetting = {}
  self.tbAllNpcGroup = {}
  self.tbAllPlayerDmgRank = {}
  self.tbPosListSetting = {}
  self.tbAllTMapSetting = {}
  self.tbTimeFrameTMapSetting = {}
  self.tbKinItemAwardSetting = {}
  self.tbTimeFrameSetting = {}
  self.bCanEnterWuLinGaoShou = false
  local tbFileData = Lib:LoadTabFile("Setting/WuLinGaoShou/PosList.tab", {PosID = 1})
  for _, tbInfo in pairs(tbFileData) do
    local tbPosInfo = {}
    tbPosInfo.nMapID = tbInfo.MapID
    tbPosInfo.strMapName = tbInfo.MapName
    tbPosInfo.tbPos = Lib:GetTabFromString1(tbInfo.Pos)
    tbPosInfo.nTrapIndex = tbInfo.TrapIndex
    self.tbPosListSetting[tbInfo.PosID] = tbPosInfo
  end
  tbFileData = Lib:LoadTabFile("Setting/WuLinGaoShou/Main.tab", {
    ShowAwardID = 1,
    ShowNpcID = 1,
    BossId = 1,
    BossNum = 1,
    FalseBossId = 1,
    FalseBossNum = 1,
    NpcLevel = 1,
    SeriesType = 1,
    AwardId = 1,
    TongAwardId = 0,
    MapID = 1,
    Priority = 1,
    FalsePriority = 1,
    CanNotBianYi = 1,
    Sort = 1
  })
  self.tbMainSetting = {}
  for nRow, tbInfo in pairs(tbFileData) do
    self.tbMainSetting[tbInfo.TimeFrame] = self.tbMainSetting[tbInfo.TimeFrame] or {}
    self.tbTimeFrameTMapSetting[tbInfo.TimeFrame] = self.tbTimeFrameTMapSetting[tbInfo.TimeFrame] or {}
    local tbMapInfo = {}
    local nBossGroup = tonumber(tbInfo.Group)
    tbMapInfo.nMapTID = tbInfo.MapID
    tbMapInfo.nShowAwardID = tbInfo.ShowAwardID
    tbMapInfo.nShowNpcID = tbInfo.ShowNpcID
    tbMapInfo.nBossId = tbInfo.BossId
    tbMapInfo.nBossNum = tbInfo.BossNum
    tbMapInfo.nFalseBossId = tbInfo.FalseBossId
    tbMapInfo.nFalseBossNum = tbInfo.FalseBossNum
    tbMapInfo.nNpcLevel = tbInfo.NpcLevel
    tbMapInfo.nSeries = tbInfo.SeriesType
    tbMapInfo.nAwardId = tbInfo.AwardId
    tbMapInfo.nTongAwardId = tbInfo.TongAwardId
    tbMapInfo.szActivityName = tbInfo.MapName or ""
    tbMapInfo.nSort = tbInfo.Sort
    tbMapInfo.nCanNotBianYi = tbInfo.CanNotBianYi
    tbMapInfo.tbPos = Lib:GetTabFromString1(tbInfo.Pos)
    tbMapInfo.tbPosIDs = Lib:SplitStr2Num(tbInfo.PosIDs)
    tbMapInfo.tbMapTIDs = {}
    for _, posID in pairs(tbMapInfo.tbPosIDs) do
      local posInfo = self.tbPosListSetting[posID]
      if posInfo then
        tbMapInfo.tbMapTIDs[posID] = posInfo
        self.tbAllTMapSetting[tonumber(posInfo.nMapID)] = 1
        self.tbTimeFrameTMapSetting[tbInfo.TimeFrame][tonumber(posInfo.nMapID)] = 1
      end
    end
    self.tbMainSetting[tbInfo.TimeFrame][tbMapInfo.nShowNpcID] = self.tbMainSetting[tbInfo.TimeFrame][tbMapInfo.nShowNpcID] or {}
    self.tbMainSetting[tbInfo.TimeFrame][tbMapInfo.nShowNpcID][nBossGroup] = self.tbMainSetting[tbInfo.TimeFrame][tbMapInfo.nShowNpcID][nBossGroup] or {}
    table.insert(self.tbMainSetting[tbInfo.TimeFrame][tbMapInfo.nShowNpcID][nBossGroup], tbMapInfo)
    self.tbBossSetting[tbInfo.TimeFrame] = self.tbBossSetting[tbInfo.TimeFrame] or {}
    self.tbBossSetting[tbInfo.TimeFrame][nBossGroup] = self.tbBossSetting[tbInfo.TimeFrame][nBossGroup] or {}
    self.tbBossSetting[tbInfo.TimeFrame][nBossGroup].tbTrueBoss = self.tbBossSetting[tbInfo.TimeFrame][nBossGroup].tbTrueBoss or {}
    self.tbBossSetting[tbInfo.TimeFrame][nBossGroup].tbFakeBoss = self.tbBossSetting[tbInfo.TimeFrame][nBossGroup].tbFakeBoss or {}
    if tbMapInfo.nBossId > 0 and tbMapInfo.nBossNum > 0 then
      local tbTrueBossData = {}
      tbTrueBossData.nShowNpcID = tbMapInfo.nShowNpcID
      tbTrueBossData.nNpcTId = tbMapInfo.nBossId
      tbTrueBossData.nNpcLevel = tbMapInfo.nNpcLevel
      tbTrueBossData.nSeries = tbMapInfo.nSeries
      tbTrueBossData.nAwardId = tbMapInfo.nAwardId
      tbTrueBossData.nTongAwardId = tbMapInfo.nTongAwardId
      tbTrueBossData.nCount = tbMapInfo.nBossNum
      tbTrueBossData.tbPosIDs = tbMapInfo.tbPosIDs
      tbTrueBossData.tbMapTIDs = tbMapInfo.tbMapTIDs
      table.insert(self.tbBossSetting[tbInfo.TimeFrame][nBossGroup].tbTrueBoss, tbTrueBossData)
    end
    if tbMapInfo.nFalseBossId > 0 and tbMapInfo.nFalseBossNum > 0 then
      local tbFakeBossData = {}
      tbFakeBossData.nShowNpcID = tbMapInfo.nShowNpcID
      tbFakeBossData.nNpcTId = tbMapInfo.nFalseBossId
      tbFakeBossData.nNpcLevel = tbMapInfo.nNpcLevel
      tbFakeBossData.nSeries = tbMapInfo.nSeries
      tbFakeBossData.nAwardId = tbMapInfo.nAwardId
      tbFakeBossData.nTongAwardId = tbMapInfo.nTongAwardId
      tbFakeBossData.nCount = tbMapInfo.nFalseBossNum
      tbFakeBossData.tbPosIDs = tbMapInfo.tbPosIDs
      tbFakeBossData.tbMapTIDs = tbMapInfo.tbMapTIDs
      table.insert(self.tbBossSetting[tbInfo.TimeFrame][nBossGroup].tbFakeBoss, tbFakeBossData)
    end
  end
  self.tbMainSetting = SetReadOnly(self.tbMainSetting)
  self.tbBossSetting = SetReadOnly(self.tbBossSetting)
  self.tbPosListSetting = SetReadOnly(self.tbPosListSetting)
end

WuLinGaoShou:LoadSetting()

function WuLinGaoShou:GetTimeFrameNpcGroupByPlayerCount()
  Log("------------ perpare Goldboss maps and npcs ------------")
  local szCurTimeFrame = Lib:GetMaxTimeFrame(self.tbMainSetting)
  if not self.tbBossSetting[szCurTimeFrame] then
    LogErr("WuLinGaoShou not Boss TimeFrame Config", szCurTimeFrame)
    return
  end
  if not self.tbBossSetting[szCurTimeFrame][self.nBossGroup] then
    LogErr("WuLinGaoShou not Boss Group Config", self.nBossGroup)
    return
  end
  local tbBossSetting = Lib:CopyTB(self.tbBossSetting[szCurTimeFrame][self.nBossGroup])
  if not tbBossSetting then
    LogErr("WuLinGaoShou not tbBossSetting", szCurTimeFrame, self.nBossGroup)
    return
  end
  local tbAllCreateBoss = {}
  for nIdx, tbData in pairs(tbBossSetting.tbTrueBoss) do
    tbAllCreateBoss[tbData.nNpcTId] = tbData
    Log(string.format("Create count:%s TrueBoss nShowNpcID:%d, BossTemplateID:%d ", tbData.nCount, tbData.nShowNpcID, tbData.nNpcTId))
  end
  for nIdx, tbFakeData in pairs(tbBossSetting.tbFakeBoss) do
    tbAllCreateBoss[tbFakeData.nNpcTId] = tbFakeData
    Log(string.format("Create count:%s FakeBoss nShowNpcID:%d, BossTemplateID:%d", tbFakeData.nCount, tbFakeData.nShowNpcID, tbFakeData.nNpcTId))
  end
  local tbCreateMapData = {}
  for nTId, tbData in pairs(tbAllCreateBoss) do
    for _, posID in pairs(tbData.tbPosIDs) do
      local posInfo = self.tbPosListSetting[posID]
      local nMapTId = tonumber(posInfo.nMapID)
      if posInfo and not tbCreateMapData[nMapTId] then
        tbCreateMapData[nMapTId] = 1
        Log("Create Map TID:", nMapTId)
      end
    end
  end
  Log("----------------------- over ---------------------")
  return tbCreateMapData, tbAllCreateBoss, szCurTimeFrame
end

function WuLinGaoShou:ClientCustomCheckEnter(szType, nMapTID)
  if me.nLevel < WuLinGaoShou.nJoinLevel then
    return false, i18n.EasyGet("\232\190\190\229\136\176{0}\231\186\167\230\137\141\229\143\175\229\143\130\229\138\160\230\180\187\229\138\168", WuLinGaoShou.nJoinLevel)
  end
  local bRet, szMsg = ActivityCheck:CheckCanEnterActivityEx(me, nil, nil, false, false, false, nil, {
    "UIGoldBossEntrance"
  }, false)
  return bRet, szMsg
end

function WuLinGaoShou:ClientCanEnterWuLinGaoShou(szType, nMapTID, nX, nY)
  local bRet, szMsg = WuLinGaoShou:ClientCustomCheckEnter(szType, nMapTID)
  if not bRet then
    me.CenterMsg(szMsg or "")
    return
  end
  RemoteServer.WuLinGaoShouEnterFuben(szType, nMapTID, nX, nY)
end

function WuLinGaoShou:IsWuLinGaoShouMap(nMapTID)
  if not self.tbAllTMapSetting[nMapTID] then
    return false
  end
  return true
end

function WuLinGaoShou:IsCurTimeFrameWuLinGaoShouMap(nMapTID)
  local curTimeFrameMap, nCurTimeFrame = WuLinGaoShou:GetTimeFrameMapSetting()
  if not curTimeFrameMap[nMapTID] then
    return false
  end
  return true
end

function WuLinGaoShou:GetBornPoint(nMapTID)
  local tbPos = WuLinGaoShou.tbBornPos[nMapTID]
  if not tbPos or Lib:IsEmptyTB(tbPos) then
    return
  end
  local nRandIndex = MathRandom(#tbPos)
  local tRandPos = tbPos[nRandIndex]
  if not tRandPos then
    return
  end
  return table.unpack(tRandPos)
end

function WuLinGaoShou:AddOpenTimes()
  local tbWuLinGaoShouData = WuLinGaoShou:GetCurrentData() or {}
  local nCur = tbWuLinGaoShouData.nOpenTimes or 1
  if nCur > #WuLinGaoShou.NotBianYiConfig + 1 then
    return
  end
  tbWuLinGaoShouData.nOpenTimes = nCur + 1
  ScriptData:SaveValue(WuLinGaoShou.SCRIPTDATA_NAME, 0, tbWuLinGaoShouData)
  Log("[WuLinGaoShou] AddOpenTimes ", tbWuLinGaoShouData.nOpenTimes)
  return tbWuLinGaoShouData
end

function WuLinGaoShou:RefreshOpenTimes()

function WuLinGaoShou:GetMaxValueConf()
  local szCurTimeFrame = Lib:GetMaxTimeFrame(WuLinGaoShou.PlayerCount2Value)
  local tbConf = WuLinGaoShou.PlayerCount2Value[szCurTimeFrame] or {}
  local nMaxValue = Lib:GetTableMaxKey(tbConf)
  local tbCurConf = tbConf[nMaxValue] or {}
  local nMaxPlayer = tbCurConf[2] or 0
  return tbConf, nMaxPlayer, nMaxValue
end

end

function WuLinGaoShou:RequestActivityStatus()
  if me.nLevel < WuLinGaoShou.nJoinLevel then
    return
  end
  self.nLastRequestStatus = self.nLastRequestStatus or 0
  local now = GetTime()
  if now > self.nLastRequestStatus + 2 then
    self.nLastRequestStatus = now
    RemoteServer:RequestGoldBossStatus()
  end
end

function WuLinGaoShou:OnSyncActivityStatus(bStartEnter, bStart)
  local bMapCheck = self:IsWuLinGaoShouMap(me.nMapTemplateId)
  self.bCanEnterWuLinGaoShou = bStartEnter and bMapCheck
  EventNotify.OnNotify(EventNotify.EMNOTIFY_WULINGAOSHOU_STATUS_REPLY)
end

function WuLinGaoShou:OnSyncActEnd(bNotStart)
  local msg = i18n.EasyGet("\230\180\187\229\138\168\229\183\178\231\187\147\230\157\159")
  if bNotStart then
    msg = i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  if Ui:WindowVisible("UIGoldBossEntrance") == true then
    Ui:GetWindow("UIGoldBossEntrance"):Close()
  end
  me.CenterMsg(msg)
  self.nSelectIndex = 1
end

function WuLinGaoShou:RequestWuLinGaoShouBossDmgRank()
  if Calendar:IsActivityInOpenState("WuLinGaoShou") then
    RemoteServer.RequestWuLinGaoShouBossDmgRank()
  else
    Ui:FinishHudFuben()
  end
end

function WuLinGaoShou:GoToBossPosition(nMapTemplatedID, nX, nY)
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  if AFK:IsLeadFightRunning() then
    AFK:Stop()
  end
  AutoPath:GotoAndCallEx(nMapTemplatedID, nX, nY, function()
    if not AFK:IsRunning() then
      AFK:Start()
    end
  end, nil)
end

function WuLinGaoShou:IsPlayerInActivityStart()
  if not self:IsWuLinGaoShouMap(me.nMapTemplateId) then
    return false
  end
  local tbWLGSData = self:GetSyncData("Boss")
  if not tbWLGSData then
    return false
  end
  return tbWLGSData.szCurrentState == "Start"
end

function WuLinGaoShou:GetSeriesType(type)
  if 1 == type then
    return i18n.EasyGet("\233\135\145")
  elseif 2 == type then
    return i18n.EasyGet("\230\156\168")
  elseif 3 == type then
    return i18n.EasyGet("\230\176\180")
  elseif 4 == type then
    return i18n.EasyGet("\231\129\171")
  elseif 5 == type then
    return i18n.EasyGet("\229\156\159")
  else
    return i18n.EasyGet("\230\151\160")
  end
end

function WuLinGaoShou:OnJoinWuLinGaoShou(szType)

end

function WuLinGaoShou:CheckQualification()
  local nWeekDay = Lib:GetLocalWeekDay(GetTime())
  if 2 == nWeekDay or 4 == nWeekDay then
    return true
  elseif 6 == nWeekDay then
    return not ResourceBattle:CheckIsOpen()
  end
  return false
end
