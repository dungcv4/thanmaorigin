YeLanGuan.STATE = {
  PREPARE = 1,
  START = 2,
  END = 3
}
YeLanGuan.CAMP = {UP = 3, DOWN = 4}
YeLanGuan.nLandRange = 500

function YeLanGuan:LoadNpcSetting()
  local szPath = "Setting/Map/yelanguan1/info/Buildingnpc_pos.txt"
  local tbSetting = Lib:EasyLoadTabFile(szPath)
  YeLanGuan.tbBuildings = {}
  for _, tbInfo in ipairs(tbSetting) do
    table.insert(YeLanGuan.tbBuildings, {
      nPosX = tonumber(tbInfo.PosX),
      nPosY = tonumber(tbInfo.PosY),
      nPath = tbInfo.nPath,
      nCamp = tbInfo.nCamp
    })
  end
  szPath = "Setting/Map/yelanguan1/info/npc_pos.txt"
  tbSetting = Lib:EasyLoadTabFile(szPath)
  YeLanGuan.tbPos = {}
  for _, tbInfo in ipairs(tbSetting) do
    YeLanGuan.tbPos[tbInfo.nPoint] = YeLanGuan.tbPos[tbInfo.nPoint] or {}
    table.insert(YeLanGuan.tbPos[tbInfo.nPoint], {
      nPosX = tonumber(tbInfo.PosX),
      nPosY = tonumber(tbInfo.PosY)
    })
  end
  szPath = "Setting/Map/yelanguan1/info/npc_path.txt"
  tbSetting = Lib:EasyLoadTabFile(szPath)
  YeLanGuan.tbPath = {}
  for _, tbInfo in ipairs(tbSetting) do
    YeLanGuan.tbPath[tbInfo.nPath] = YeLanGuan.tbPath[tbInfo.nPath] or {}
    YeLanGuan.tbPath[tbInfo.nPath][tbInfo.nIndex] = {
      nPosX = tonumber(tbInfo.PosX),
      nPosY = tonumber(tbInfo.PosY)
    }
  end
end

YeLanGuan:LoadNpcSetting()

function YeLanGuan:GetBuilding(nX, nY)
  for nIndex, tbPos in pairs(YeLanGuan.tbBuildings) do
    if math.abs(nX - tbPos.PosX) < TowerDefense.nLandRange and math.abs(nY - tbPos.PosY) < TowerDefense.nLandRange then
      return nIndex
    end
  end
end

function YeLanGuan:GetPosByType(nType)
  local tbPos = YeLanGuan.tbPos[nType]
  if tbPos then
    if #tbPos <= 1 then
      return tbPos[1].nPosX, tbPos[1].nPosY
    else
      local tbRandPos = Lib:RandTable(tbPos)
      return tbRandPos.nPosX, tbRandPos.nPosY
    end
  end
end
