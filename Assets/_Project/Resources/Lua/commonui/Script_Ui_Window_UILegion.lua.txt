function tbWnd:OnCreate()
  self.tbPanelList = Ui:NewButtonGroup(self, {
    [1] = self.btnLegionInfo,
    [2] = self.btnLegionMember,
    [3] = self.btnActivity
  }, true, function(self, nIndex)
    if 1 == nIndex then
      self:onClickSelectLegionInfoPanel()
    elseif 2 == nIndex then
      self:onClickSelectLegionMemberPanel()
    elseif 3 == nIndex then
      self:onClickSelectLegionDomainPanel()
    end
  end)
  self.tbLegionMemberScrollView = Ui:NewLoopScrollView(self, self.LegionMemberTemplate)
  self.tbSimpleLegionMemberScrollView = Ui:NewLoopScrollView(self, self.SimpleLegionMemberTemplate)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, self.UIMoneyPanel)
  self.tbLegionMemberScrollRect = self.pPanel:GetScrollRect(tbControls.LegionMemberList)
  self.tbLegionMemberScrollRect.onValueChanged:AddListener(function(vec2)
    self:OnSliderList(vec2)
  end)
end

tbWnd.PAGE_INDEX = {MEMBERLIST = 2, ACTIVITY = 3}

function tbWnd:OnOpen(szPage, ...)
  if 0 == me.dwLegionId then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\138\160\229\133\165\229\134\155\229\155\162"))
    return 0
  end
  self.tbMoneyPanel:SetMoneyValue()
  if not GlobalDomainBattle.bMeetTimeFrame and "ACTIVITY" == szPage then
    szPage = ""
  end
  for nMapTemplateId, tbInfo in pairs(GlobalDomainBattle.MAP_FIGHT) do
    local szPath = string.format(self.btnDomainMap, tbInfo.MAP_UI)
    Ui:Button_BindEvent(self.pPanel, szPath, self.onClickDomain, self, nMapTemplateId)
  end
  self.tbWndDomainStat = Ui:CreatePanel("UILegionDomainStat", self.wndDomainStat, self)
  self.szSortType = "Default"
  self.bIsDown = true
  local nPage = self.PAGE_INDEX[szPage]
  if not nPage then
    self.tbPanelList:Select(1)
  else
    self.tbOpenParam = {
      ...
    }
    self.tbPanelList:Select(nPage)
  end
  self:RegistRedPoint()
  self.nCurY = 1
  self.nSliderLimit = nDefaultSlideLimit
  if self.nC2sTimer then
    Timer:Close(self.nC2sTimer)
    self.nC2sTimer = nil
  end
  self:SetMonitorUI(true)
end

function tbWnd:OnClose()
  self:UnRegistRedPoint()
  self.pPanel:SetActive(tbControls.btnGoBottom, false)
  self.pPanel:SetActive(tbControls.btnGoTop, false)
  if self.nC2sTimer then
    Timer:Close(self.nC2sTimer)
    self.nC2sTimer = nil
  end
  self.nC2sTimer = Timer:Register(Env.GAME_FPS, function()
    self:SetMonitorUI(false)
  end, self)
end

function tbWnd:SetMonitorUI(bMonitor)
  Legion:AskMonitorUI(bMonitor)
  if self.nC2sTimer then
    Timer:Close(self.nC2sTimer)
    self.nC2sTimer = nil
  end
end

function tbWnd:onLegionSyncData(szDataType, nRemainTime)
  if "BaseInfo" == szDataType then
    self:OnUpdateBaseLegionInfo()
  elseif "MemberList" == szDataType then
    self:OnUpdateMemberList()
  elseif "Declare" == szDataType then
    local tbBaseInfo = Legion:GetBaseInfo() or {}
    self.pPanel:Label_SetText(self.txtLegionNotice, tbBaseInfo.szPublicDeclare or "")
  else
    if "Quit" == szDataType then
      self:Close()
    else
    end
  end
end

function tbWnd:OnUpdateBaseLegionInfo()
  local tbBaseInfo = Legion:GetBaseInfo() or {}
  self.pPanel:Label_SetText(self.txtLegionName, tbBaseInfo.szName or "")
  self.pPanel:Label_SetText(self.txtLegionMasterName, tbBaseInfo.szMasterName or "")
  local szMemberDesc = ""
  if tbBaseInfo.nMemberCount then
    szMemberDesc = string.format("%d/%d", tbBaseInfo.nMemberCount, tbBaseInfo.nMaxMemberCount)
  end
  self.pPanel:Label_SetText(self.txtMemberCount, szMemberDesc)
  self.pPanel:Label_SetText(self.txtRankingValue, tbBaseInfo.nDomainBattleRanking or 0)
  self.pPanel:Label_SetText(self.txtStarValue, tbBaseInfo.nDomainBattleStarCount or 0)
  self.pPanel:Label_SetText(self.txtLegionNotice, tbBaseInfo.szPublicDeclare or "")
  self.pPanel:SetActive(self.btnEditLegionNotice, Legion:CheckMyAuthority(Legion.Def.Authority_EditPubilcDeclare))
end

function tbWnd:OnUpdateSimpleMemberList(szSortType)
  szSortType = szSortType or "SortByCareer"
  local tbMemberList = Legion:GetMemberList() or {}
  local fSortFun = Legion:GetSortFunction(szSortType)
  if not fSortFun then
    error("UILegion MemberInfoList Sort Error, SortFun don't found!")
    return
  end
  self.tbMemberList = fSortFun(tbMemberList, false)
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local tbMember = tbMemberList[nIndex]
    local tbElem = tbLoopScroll:GetElement(nIndex)
    tbElem.tbMember = tbMember
    local pPanel = tbElem.pPanel
    local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElem, "PortraitObj")
    tbPortraitObjPanel:SetRoleIcon(tbMember.nPortrait)
    tbPortraitObjPanel:SetPortraitFrame(tbMember.nPortraitFrame)
    tbPortraitObjPanel:SetFactionIcon(tbMember.nFaction)
    tbPortraitObjPanel:SetLevel(tbMember.nLevel)
    pPanel:Label_SetText("txtName", tbMember.szName)
    Ui:SetVipLevelEffect(pPanel, tbMember.nVipLevel, "txtName")
    local szCareer = i18n.Get(Legion.Def.Career_Name[tbMember.nCareer])
    pPanel:Label_SetText("txtJob", szCareer)
  end
  
  local nCount = Lib:CountTB(tbMemberList)
  self.tbSimpleLegionMemberScrollView:SetElement(nCount, fnSetItem)
end

function tbWnd:OnUpdateMemberList(szSortType)
  if szSortType then
    if self.szSortType == szSortType then
      self.bIsDown = not self.bIsDown
    else
      self.szSortType = szSortType
      self.bIsDown = true
    end
  end
  local tbMemberList = Legion:GetMemberList() or {}
  local fSortFun = Legion:GetSortFunction(self.szSortType)
  if not fSortFun then
    error("UILegion MemberInfoList Sort Error, SortFun don't found!")
    return
  end
  self.tbMemberList = fSortFun(tbMemberList, self.bIsDown)
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local tbMember = tbMemberList[nIndex]
    if not tbMember then
      return
    end
    local tbElem = tbLoopScroll:GetElement(nIndex)
    tbElem.tbMember = tbMember
    local pPanel = tbElem.pPanel
    local szPopupType = "Legion"
    if tbMember.nServerId == SERVER_ID then
      if Legion:CheckCanKickOut(tbMember.nCareer) then
        szPopupType = "LegionKick"
      end
    elseif Legion:CheckCanKickOut(tbMember.nCareer) then
      szPopupType = "LegionOtherServerKick"
    else
      szPopupType = "LegionOtherServer"
    end
    if tbMember.nMemberId ~= me.dwID then
      Ui:Button_BindEvent(pPanel, "", function()
        Ui:OpenWindowWithCallback("UIRightPopUpMenu2", function(caller, tbWndMenu)
          local pos = self.pPanel:GetTransform("node/imgBG/Content/imgPopupPos").position
          tbWndMenu:SetPos(pos.x, pos.y)
        end, self, szPopupType, tbMember, self)
      end)
    else
      Ui:Button_UnBindEvent(pPanel, "")
    end
    pPanel:Sprite_SetSprite("imgFaction", Faction:GetIcon(tbMember.nFaction))
    pPanel:Label_SetText("txtName", tbMember.szName)
    Ui:SetVipLevelEffect(pPanel, tbMember.nVipLevel, "txtName")
    local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(tbMember.nHonorStarLevel, false)
    local obj = pPanel:GetObject("HonorTitle")
    local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
    pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
    pPanel:Label_SetText("txtLevel", Lib:GetLevelText(tbMember.nLevel))
    pPanel:Label_SetText("txtJob", i18n.Get(Legion.Def.Career_Name[tbMember.nCareer]))
    pPanel:Label_SetText("txtServer", tbMember.szServerName)
  end
  
  local nCount = Lib:CountTB(tbMemberList)
  self.tbLegionMemberScrollView:SetElement(nCount, fnSetItem)
  self.pPanel:SetActive(self.btnLegionJob, Legion:CheckMyAuthority(Legion.Def.Authority_GrantViceMaster))
  self.pPanel:SetActive(self.btnApplyerList, Legion:CheckMyAuthority(Legion.Def.Authority_Recruit))
  self.pPanel:SetActive(self.btnLegionMail, Legion:CheckMyAuthority(Legion.Def.Authority_Mail))
  self.pPanel:SetActive(self.btnLegionAnnounce, Legion:CheckMyAuthority(Legion.Def.Authority_EditRecuitInfo))
end

function tbWnd:OnUpdateActivityList()
  self.tbActiviryListScrollView:Clear()
  for i = 1, #Legion.Cfg.Salary_Channel_Type do
    self.tbActiviryListScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("ActType/txtTitle", i18n.Get(Legion.Cfg.Salary_Channel_Type[i].Desc))
      tbElement.pPanel:SetActive("ActType/MarkFull", false)
      local activityAllSalary = 0
      for j = 1, #Legion.Cfg.Salary_Channel do
        if Legion:CheckSalaryTimeFrameState(j) and i == Legion.Cfg.Salary_Channel[j].Channel_Type then
          local key = string.format("ActDetail%d", j)
          local item = tbElement.pPanel:CloneObject("ActDetail/item", key)
          tbElement.pPanel:Label_SetText(item .. "/txtActName", i18n.Get(Legion.Cfg.Salary_Channel[j].Title))
          tbElement.pPanel:Label_SetText(item .. "/txtActDesc", i18n.Get(Legion.Cfg.Salary_Channel[j].Desc))
          local degreeName = Legion.Cfg.Salary_Channel[j].DegreeName
          local maxTimes = Legion.Cfg.Salary_Channel[j].MaxTimes
          local salaryOneTimes = Legion.Cfg.Salary_Channel[j].Salary
          local curDegree = DegreeCtrl:GetDegree(me, degreeName)
          local nMaxDegree = DegreeCtrl:GetMaxDegree(degreeName, me)
          if curDegree < 0 then
            curDegree = 0
          end
          local nChange = nMaxDegree - maxTimes
          nChange = Lib:IIF(nChange > 0, nChange, 0)
          local curTimes = maxTimes + nChange - curDegree
          local times = string.format("(%d/%d)", curTimes, maxTimes)
          tbElement.pPanel:Label_SetText(item .. "/txtActName/txtActCount", times)
          local activitySalary = salaryOneTimes * curTimes
          tbElement.pPanel:Label_SetText(item .. "/txtSalary", activitySalary)
          activityAllSalary = activityAllSalary + activitySalary
          if curTimes == maxTimes then
            tbElement.pPanel:SetActive(item .. "/txtFull", true)
          else
            tbElement.pPanel:SetActive(item .. "/txtFull", false)
          end
        end
      end
      tbElement.pPanel:ContentSizeFitter_Refresh("ActDetail")
      self.pPanel:ContentSizeFitter_Refresh(tbElement.szKey)
      local nMaxSalary = Legion:GetMaxSalaryByChannelType(i)
      if activityAllSalary > nMaxSalary then
        activityAllSalary = nMaxSalary
      end
      local activitySalaryPer = math.floor(activityAllSalary * 100 / nMaxSalary)
      tbElement.pPanel:Label_SetText("ActType/txtPercent/txtPercentValue", string.format("%d", activitySalaryPer) .. "%")
      if 100 == activitySalaryPer then
        activitySalaryPer = nMaxSalary
        tbElement.pPanel:SetActive("ActType/MarkFull", true)
        tbElement.pPanel:SetActive("ActType/txtFull", true)
        tbElement.pPanel:SetActive("ActType/txtPercent", false)
        tbElement.pPanel:Label_SetText("ActType/txtFull/txtFullValue", string.format("%d", activitySalaryPer))
      else
        tbElement.pPanel:SetActive("ActType/MarkFull", false)
        tbElement.pPanel:SetActive("ActType/txtFull", false)
        tbElement.pPanel:SetActive("ActType/txtPercent", true)
      end
    end)
  end
  self:ShowLegionSalaryInfo()
end

function tbWnd:OnSliderList(vec2)
  if vec2.y <= 0 or vec2.y >= 1 then
    return
  end
  if self.bButtonChange then
    self.bButtonChange = false
    self.nSliderLimit = nDefaultSlideLimit
    return
  end
  local y = math.floor(vec2.y * 10) / 10
  local nDiff = math.abs(y - self.nCurY)
  if 0.9 == y or 0 == y then
    self.pPanel:SetActive(tbControls.btnGoTop, false)
    self.pPanel:SetActive(tbControls.btnGoBottom, false)
    return
  end
  if nDiff < self.nSliderLimit then
    return
  end
  if y < self.nCurY then
    self.pPanel:SetActive(tbControls.btnGoTop, false)
    self.pPanel:SetActive(tbControls.btnGoBottom, true)
  elseif y > self.nCurY then
    self.pPanel:SetActive(tbControls.btnGoBottom, false)
    self.pPanel:SetActive(tbControls.btnGoTop, true)
  end
  self.nCurY = y
  self.nSliderLimit = nSwitchBtnLimit
end

function tbWnd:OnUpdateDomainInfo()
  if 0 == #GlobalDomainBattle:GetDomain(me.dwLegionId) then
    self:OnUpdateDomainInfoWhenNoDomain()
  else
    self:OnUpdateDomainInfoWhenHasDomain()
  end
end

function tbWnd:OnUpdateDomainInfoWhenNoDomain()
  local szRedBgPath = "UI/atlas/Family/img_domain_bg_red.png"
  local szBlueBgPath = "UI/atlas/Family/img_domain_bg_blue.png"
  local szGreyBgPath = "UI/atlas/Family/img_domain_bg_grey.png"
  local bWeekDay = GlobalDomainBattle:IsWeekDay()
  local bClaimPhase, nClaimTimeFlag = GlobalDomainBattle:GetClaimState()
  local bGamePhase = GlobalDomainBattle:IsFightPhase()
  local tbExtLink = {}
  local tbLinkList = {}
  if bGamePhase then
    local nTarget = GlobalDomainBattle:GetTarget(me.dwLegionId) or 0
    tbLinkList = {}
    if nTarget and GlobalDomainBattle.MAP_FIGHT[nTarget] then
      tbLinkList = Lib:ListToSet(GlobalDomainBattle.MAP_FIGHT[nTarget].LINK)
    end
    if nTarget and GlobalDomainBattle.MAP_FIGHT_EXT_LINK[nTarget] then
      local tbLastDomainList = GlobalDomainBattle:GetLastDomain(me.dwLegionId)
      local bHasExtLink = false
      for _, nLastMapTemplateId in pairs(tbLastDomainList) do
        if GlobalDomainBattle.MAP_FIGHT[nLastMapTemplateId].STAR_LV >= GlobalDomainBattle.MAP_FIGHT_EXT_LINK[nTarget].MERCY_STAR_LV_REQ then
          bHasExtLink = true
          break
        end
      end
      if bHasExtLink then
        tbExtLink = Lib:ListToSet(GlobalDomainBattle.MAP_FIGHT_EXT_LINK[nTarget].LINK)
      end
    end
  end
  local tbLegendState = {}
  local szPath
  for nMapTemplateId, tbInfo in pairs(GlobalDomainBattle.MAP_FIGHT) do
    szPath = string.format(self.txtDomainOccupier, tbInfo.MAP_UI)
    self.pPanel:Label_SetText(szPath, i18n.EasyGet("\229\134\155\229\155\162\239\188\154{0}", GlobalDomainBattle:GetOccupierShow(nMapTemplateId)))
    szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
    if 1 == tbInfo.OPEN_CLAIM and tbInfo.STAR_LV <= GlobalDomainBattle.CLAIM_COND.STAR_LV_LIMIT then
      self.pPanel:Sprite_SetSprite(szPath, szRedBgPath)
    else
      self.pPanel:Sprite_SetSprite(szPath, szGreyBgPath)
    end
    szPath = string.format(self.DomainState, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, true)
    szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
    if 1 == tbInfo.OPEN_CLAIM and tbInfo.STAR_LV <= GlobalDomainBattle.CLAIM_COND.STAR_LV_LIMIT then
      self.pPanel:SetActive(szPath, true)
    else
      self.pPanel:SetActive(szPath, false)
    end
    szPath = string.format(self.DomainDeclare, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.DomainDefend, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    if bWeekDay and 1 == tbInfo.OPEN_CLAIM then
      if bGamePhase then
        local nTarget = GlobalDomainBattle:GetTarget(me.dwLegionId)
        if nTarget == nMapTemplateId or tbLinkList[nMapTemplateId] then
          szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
          self.pPanel:Sprite_SetSprite(szPath, szRedBgPath)
          szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, false)
          szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, true)
          szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, true)
        elseif tbExtLink[nMapTemplateId] then
          szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
          self.pPanel:Sprite_SetSprite(szPath, szRedBgPath)
          szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, false)
          szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, true)
        else
          szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
          self.pPanel:Sprite_SetSprite(szPath, szGreyBgPath)
          szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, false)
        end
      elseif bClaimPhase then
        if GlobalDomainBattle:GetTarget(me.dwLegionId) == nMapTemplateId then
          szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
          self.pPanel:Sprite_SetSprite(szPath, szRedBgPath)
          szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, false)
          szPath = string.format(self.DomainDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, true)
        elseif tbInfo.STAR_LV <= GlobalDomainBattle.CLAIM_COND.STAR_LV_LIMIT then
          szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
          self.pPanel:Sprite_SetSprite(szPath, szRedBgPath)
          szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, nClaimTimeFlag > 0)
        else
          szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
          self.pPanel:Sprite_SetSprite(szPath, szGreyBgPath)
          szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, false)
        end
      end
    end
    szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendCanDeclareWar] = true
    end
    szPath = string.format(self.DomainDeclare, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendDeclareWar] = true
    end
    szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendAttack] = true
    end
    szPath = string.format(self.DomainDefend, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendDefend] = true
    end
    szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendNpcInvade] = true
    end
  end
  self.pPanel:SetActive(self.LegendCanDeclareWar, tbLegendState[self.LegendCanDeclareWar] or false)
  self.pPanel:SetActive(self.LegendDeclareWar, tbLegendState[self.LegendDeclareWar] or false)
  self.pPanel:SetActive(self.LegendAttack, tbLegendState[self.LegendAttack] or false)
  self.pPanel:SetActive(self.LegendDefend, tbLegendState[self.LegendDefend] or false)
  self.pPanel:SetActive(self.LegendNpcInvade, tbLegendState[self.LegendNpcInvade] or false)
  self.pPanel:SetActive(self.LegendRoot, 0 ~= Lib:CountTB(tbLegendState))
end

function tbWnd:OnUpdateDomainInfoWhenHasDomain()
  local szRedBgPath = "UI/atlas/Family/img_domain_bg_red.png"
  local szBlueBgPath = "UI/atlas/Family/img_domain_bg_blue.png"
  local szGreyBgPath = "UI/atlas/Family/img_domain_bg_grey.png"
  local bGamePhase = GlobalDomainBattle:IsFightPhase()
  local tbDomainList = GlobalDomainBattle:GetDomain(me.dwLegionId)
  local tbExtLink = {}
  if bGamePhase then
    for _, nMapTemplateId in pairs(tbDomainList) do
      if GlobalDomainBattle.MAP_FIGHT_EXT_LINK[nMapTemplateId] and 1 == GetTimeFrameState(GlobalDomainBattle.MAP_FIGHT_EXT_LINK[nMapTemplateId].TIMEFRAME_REQ) then
        for _, nLink in pairs(GlobalDomainBattle.MAP_FIGHT_EXT_LINK[nMapTemplateId].LINK) do
          tbExtLink[nLink] = true
        end
      end
    end
  end
  local tbLegendState = {}
  local szPath
  for nMapTemplateId, tbInfo in pairs(GlobalDomainBattle.MAP_FIGHT) do
    szPath = string.format(self.txtDomainOccupier, tbInfo.MAP_UI)
    self.pPanel:Label_SetText(szPath, i18n.EasyGet("\229\134\155\229\155\162\239\188\154{0}", GlobalDomainBattle:GetOccupierShow(nMapTemplateId)))
    szPath = string.format(self.imgDomainBg, tbInfo.MAP_UI)
    if Lib:Is_include(nMapTemplateId, tbDomainList) then
      self.pPanel:Sprite_SetSprite(szPath, szBlueBgPath)
    elseif tbInfo.STAR_LV <= GlobalDomainBattle.CLAIM_COND.STAR_LV_LIMIT then
      self.pPanel:Sprite_SetSprite(szPath, szGreyBgPath)
    else
      self.pPanel:Sprite_SetSprite(szPath, szRedBgPath)
    end
    szPath = string.format(self.DomainState, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, true)
    szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.DomainDeclare, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.DomainDefend, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
    self.pPanel:SetActive(szPath, false)
    if bGamePhase and 1 == tbInfo.OPEN_CLAIM then
      if Lib:Is_include(nMapTemplateId, tbDomainList) then
        szPath = string.format(self.DomainDefend, tbInfo.MAP_UI)
        self.pPanel:SetActive(szPath, true)
        szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
        self.pPanel:SetActive(szPath, true)
      elseif tbInfo.STAR_LV > GlobalDomainBattle.CLAIM_COND.STAR_LV_LIMIT then
        local bLink = false
        for _, nSomeMapTemplateId in pairs(tbDomainList) do
          if Lib:Is_include(nMapTemplateId, GlobalDomainBattle.MAP_FIGHT[nSomeMapTemplateId].LINK) then
            bLink = true
            break
          end
        end
        if not bLink and tbExtLink[nMapTemplateId] then
          bLink = true
        end
        if bLink then
          szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, true)
          szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
          self.pPanel:SetActive(szPath, true)
        end
      end
    end
    szPath = string.format(self.DomainCanDeclare, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendCanDeclareWar] = true
    end
    szPath = string.format(self.DomainDeclare, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendDeclareWar] = true
    end
    szPath = string.format(self.DomainAttack, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendAttack] = true
    end
    szPath = string.format(self.DomainDefend, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendDefend] = true
    end
    szPath = string.format(self.imgDomainNpcInvade, tbInfo.MAP_UI)
    if self.pPanel:IsActive(szPath) then
      tbLegendState[self.LegendNpcInvade] = true
    end
  end
  self.pPanel:SetActive(self.LegendCanDeclareWar, tbLegendState[self.LegendCanDeclareWar] or false)
  self.pPanel:SetActive(self.LegendDeclareWar, tbLegendState[self.LegendDeclareWar] or false)
  self.pPanel:SetActive(self.LegendAttack, tbLegendState[self.LegendAttack] or false)
  self.pPanel:SetActive(self.LegendDefend, tbLegendState[self.LegendDefend] or false)
  self.pPanel:SetActive(self.LegendNpcInvade, tbLegendState[self.LegendNpcInvade] or false)
  self.pPanel:SetActive(self.LegendRoot, 0 ~= Lib:CountTB(tbLegendState))
end

function tbWnd:OnSyncDomainInfo()
  if not self.pPanel:IsActive(self.panelLegionActivity) then
    return
  end
  self:OnUpdateDomainInfo()
end

function tbWnd:onClickDomain(nMapTemplateId)
  if 0 == #GlobalDomainBattle:GetDomain(me.dwLegionId) then
    self:onClickDomainWhenNoDomain(nMapTemplateId)
  else
    self:onClickDomainWhenHasDomain(nMapTemplateId)
  end
end

function tbWnd:onClickDomainWhenNoDomain(nMapTemplateId)
  local bWeekDay = GlobalDomainBattle:IsWeekDay()
  local bClaimPhase, nClaimTimeFlag = GlobalDomainBattle:GetClaimState()
  local bGamePhase = GlobalDomainBattle:IsFightPhase()
  if bWeekDay then
    if bGamePhase then
      GlobalDomainBattle:AttendReq(nMapTemplateId)
      return
    elseif bClaimPhase and nClaimTimeFlag > 0 then
      GlobalDomainBattle:ClaimReq(nMapTemplateId)
      return
    elseif nClaimTimeFlag and nClaimTimeFlag < 0 then
      me.CenterMsg(i18n.EasyGet("\232\183\168\230\156\141\233\162\134\229\156\159\230\136\152\230\156\170\229\188\128\229\167\139"))
      return
    end
  end
  me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\233\157\158\229\174\163\230\136\152\230\151\182\233\151\180"))
end

function tbWnd:onClickDomainWhenHasDomain(nMapTemplateId)
  local bWeekDay = GlobalDomainBattle:IsWeekDay()
  local bGamePhase = GlobalDomainBattle:IsFightPhase()
  if bWeekDay and bGamePhase then
    GlobalDomainBattle:AttendReq(nMapTemplateId)
    return
  end
  me.CenterMsg(i18n.EasyGet("\232\183\168\230\156\141\233\162\134\229\156\159\230\136\152\230\156\170\229\188\128\229\167\139"))
end
