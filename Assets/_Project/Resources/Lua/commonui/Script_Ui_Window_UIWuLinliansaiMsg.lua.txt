local tbWnd = Ui:CreateClass("UIWuLinliansaiMsg", "UIPanel")
local tbControls = {
  Template = "imgList/List/Template",
  btnCheck = "btnCheck",
  txtInfo2 = "txtInfo2"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCheck = function(self)
    self:OpenWulinliansaiUI()
  end
}

function tbWnd:OnCreate()
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  self:UpdateView()
  self.pPanel:Label_SetText("txtInfo2", i18n.Get(MultiPvP.CommonMsg.TIP_FINAL_MATCH_MSG, MultiPvP.Cfg.FINAL_DATE, MultiPvP.Cfg.OPEN_TIME))
end

function tbWnd:UpdateView()
  local tbPreInforRank = MultiPvP:GetPreInforRank()
  if not tbPreInforRank then
    return
  end
  for i, tbData in ipairs(tbPreInforRank) do
    self.ScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtRankingNum", i)
      tbElement.pPanel:Label_SetText("txtName", tbData.szTeamName)
      tbElement.pPanel:Label_SetText("txtExceedNum", tbData.nWinCount)
      tbElement.pPanel:Label_SetText("txtTime", Lib:TimeDesc3(tbData.nBattleTime))
      tbElement.pPanel:Label_SetText("txtScoreNum", tbData.nScore)
    end)
  end
end

function tbWnd:OnClose()
  self.ScrollView:Clear()
end

function tbWnd:OpenWulinliansaiUI()
  MultiPvP:ApplyOpenLianSaiUI()
end
