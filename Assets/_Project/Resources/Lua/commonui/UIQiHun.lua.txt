function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbQiHunScrollView = Ui:NewScrollView(self, tbControls.templateQiHun, function(self, nIndex, tbElement)
    self:OnSelectQiHun(nIndex, tbElement)
  end)
  self.highRankScroll = Ui:NewScrollView(self, tbControls.highRankTemplate)
  self.lowRankScroll = Ui:NewScrollView(self, tbControls.lowRankTemplate)
  QiHun:ApplyRankData()
  self:SetClickEmptyToClose(tbControls.wndRate)
end

function tbWnd:OnOpen(nSelectPanel, nItemTemplateId)
  self.nSelectIndex = nil
  self:UpdateQiHunSV()
  if not self.tbTabControl:OnOpen(nSelectPanel) then
    return
  end
  self.nPanelType = self.tbTabControl.nPanelType
  if nItemTemplateId then
    for i = 1, #self.tbQiHunItem do
      if self.tbQiHunItem[i][1] == nItemTemplateId then
        self.tbQiHunScrollView:SelectElement(i)
        break
      end
    end
  else
    self.tbQiHunScrollView:GoTop()
  end
  Item.QiHun:ApplyRankData()
  self:RefineRankUpdate()
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  self.tbQiHunScrollView:ClearSelected()
end

function tbWnd:OnPanelChange(nPanelType)
  self.nPanelType = nPanelType
end

function tbWnd:UpdateQiHunSV(nSubLevel)
  self.tbQiHunItem = Item.QiHun:GetQiHunItemNum()
  if nSubLevel then
    for i = 1, #self.tbQiHunItem do
      if self.tbQiHunItem[i][3] == nSubLevel then
        self.tbQiHunItem[i][2] = self.tbQiHunItem[i][2] - 1
        if self.tbQiHunItem[i][2] <= 0 then
          table.remove(self.tbQiHunItem, i)
        end
        break
      end
    end
  end
  self.tbQiHunScrollView:Reset()
  for i = 1, #self.tbQiHunItem do
    self.tbQiHunScrollView:SetElement(i, function(tbElement)
      local nTemplate = self.tbQiHunItem[i][1]
      local nCount = self.tbQiHunItem[i][2]
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(nTemplate, nCount)
      itemGrid:SetClickEnable(true)
      local szName = Item:GetName(nTemplate, me)
      local nQuality = Item:GetQuality(nTemplate)
      local _, szColor = Item:GetQualityColor(nQuality)
      tbElement.pPanel:Label_SetText("txtName", i18n.EasyGet("<color=#{0}>{1}</color>", szColor, szName))
      if self.nSelectIndex == i then
        tbElement.pPanel:SetActive("imgSelected", true)
      else
        tbElement.pPanel:SetActive("imgSelected", false)
      end
    end)
  end
end

function tbWnd:OnSelectQiHun(nIndex, tbElement)
  if self.nSelectIndex then
    local tbOldElement = self.tbQiHunScrollView:GetElement(self.nSelectIndex)
    tbOldElement.pPanel:SetActive("imgSelected", false)
  end
  self.nSelectIndex = nIndex
  tbElement.pPanel:SetActive("imgSelected", true)
  self.tbTabControl:UpdateSubPanel(self.nPanelType, "OnSelectQiHun")
end

function tbWnd:GetSelectLevel()
  if not self.nSelectIndex or not self.tbQiHunItem[self.nSelectIndex] then
    return nil
  end
  local nRet = Item.QiHun:GetLevelByTemplateId(self.tbQiHunItem[self.nSelectIndex][1])
  return nRet
end

function tbWnd:ResetSelectQiHun(nLevel, nSubLevel)
  self.nSelectIndex = nil
  self.tbQiHunScrollView:ClearSelected()
  self:UpdateQiHunSV(nSubLevel)
  self.tbQiHunScrollView:GoTop()
  local nTemplateId = QiHun.tbLevelInfo[nLevel][1]
  for i, tbV in ipairs(self.tbQiHunItem) do
    if tbV[1] == nTemplateId then
      self.nSelectIndex = i
      break
    end
  end
  if self.nSelectIndex then
    self.tbQiHunScrollView:ScrollToIndex(self.nSelectIndex, true)
  end
end

function tbWnd:NotifyCompose(nOldLevel, nNewLevel)
  self:ResetSelectQiHun(nOldLevel)
  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_COMPOSE, "OnComposeRsp", nOldLevel, nNewLevel)
end

function tbWnd:NotifyReforgeGetBack(nInitLevel)
  self:ResetSelectQiHun(nInitLevel)
  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_REGORGE, "ReforgeGetBack")
end

function tbWnd:NotifyReforge(nintial_level, nOldLevel, nNewLevel)
  self:ResetSelectQiHun(nintial_level, nNewLevel)
  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_REGORGE, "UpdateResultView", nOldLevel, nNewLevel)
end

function tbWnd:NotifySplit(nLevel)
  self:ResetSelectQiHun(nLevel)
  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_SPLIT, "SplitRsp")
end

function tbWnd:OnSyncRefineRank()
  self:RefineRankUpdate()
end

function tbWnd:RefineRankUpdate()
  self.lowRankScroll:GoTop()
  self.highRankScroll:Reset()
  self.lowRankScroll:Reset()
  local tbRankData = QiHun.tbRankData or {}
  if tbRankData.tbHigh then
    for nIndex, tbInfo in ipairs(tbRankData.tbHigh) do
      self.highRankScroll:SetElement(nIndex, function(tbElement)
        local szName = tbInfo[1]
        local nOldLevel = tbInfo[2]
        local nNewLevel = tbInfo[3]
        local nServerId = tbInfo[4]
        local nOptType = tbInfo[5]
        nOptType = nOptType or Item.QiHun.ReforgeType
        local _, szColorOld = Item:GetQualityColor(Item:GetQuality(QiHun.tbLevelInfo[nOldLevel][1]))
        local _, szColorNew = Item:GetQualityColor(Item:GetQuality(QiHun.tbLevelInfo[nNewLevel][1]))
        local pPanel = tbElement.pPanel
        if nOptType == Item.QiHun.ReforgeType then
          local szRank = i18n.EasyGet("<color=#FFC107>{0}\230\156\141{1}</color>\239\188\154<color=#{2}>{3}\231\186\167</color>\233\135\141\233\147\184\229\135\186<color=#{4}>{5}\231\186\167</color>", nServerId, szName, szColorOld, nOldLevel, szColorNew, nNewLevel)
          pPanel:Label_SetText("txtMsg", szRank)
        else
          local szRank = i18n.EasyGet("<color=#FFC107>{0}\230\156\141{1}</color>\239\188\154<color=#{2}>{3}\231\186\167</color>\229\144\136\230\136\144\229\135\186<color=#{4}>{5}\231\186\167</color>", nServerId, szName, szColorOld, nOldLevel, szColorNew, nNewLevel)
          pPanel:Label_SetText("txtMsg", szRank)
        end
      end)
    end
  end
  if tbRankData.tbLow then
    for nIndex, tbInfo in ipairs(tbRankData.tbLow) do
      self.lowRankScroll:SetElement(nIndex, function(tbElement)
        local szName = tbInfo[1]
        local nOldLevel = tbInfo[2]
        local nNewLevel = tbInfo[3]
        local nServerId = tbInfo[4]
        local nOptType = tbInfo[5]
        nOptType = nOptType or Item.QiHun.ReforgeType
        local _, szColorOld = Item:GetQualityColor(Item:GetQuality(QiHun.tbLevelInfo[nOldLevel][1]))
        local _, szColorNew = Item:GetQualityColor(Item:GetQuality(QiHun.tbLevelInfo[nNewLevel][1]))
        local pPanel = tbElement.pPanel
        if nOptType == Item.QiHun.ReforgeType then
          local szRank = i18n.EasyGet("<color=#FFC107>{0}\230\156\141{1}</color>\239\188\154<color=#{2}>{3}\231\186\167</color>\233\135\141\233\147\184\229\135\186<color=#{4}>{5}\231\186\167</color>", nServerId, szName, szColorOld, nOldLevel, szColorNew, nNewLevel)
          pPanel:Label_SetText("txtMsg", szRank)
        else
          local szRank = i18n.EasyGet("<color=#FFC107>{0}\230\156\141{1}</color>\239\188\154<color=#{2}>{3}\231\186\167</color>\229\144\136\230\136\144\229\135\186<color=#{4}>{5}\231\186\167</color>", nServerId, szName, szColorOld, nOldLevel, szColorNew, nNewLevel)
          pPanel:Label_SetText("txtMsg", szRank)
        end
      end)
    end
  end
end
