Require("CommonScript/BaiHuTang/GlobalBaiHuTangCfg.lua")
Require("Script/Fuben/FubenUIController.lua")

function GlobalBaiHuTang:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenFuBenUI()
end

function GlobalBaiHuTang:IsInFubenMap()
  return true, i18n.EasyGet("\231\153\189\232\153\142\229\160\130")
end

function GlobalBaiHuTang:IsNeedExitFubenButton()
  return true
end

function GlobalBaiHuTang:RegisterFubenUIChecker()
  local tbMapList = {}
  table.insert(tbMapList, self.MAP_READY.ID)
  table.insert(tbMapList, self.MAP_FIGHT.ID)
  FubenUIController:RegisterChecker(tbMapList, self)
end

function GlobalBaiHuTang:Init()
  if not self.bInit then
    self:RegisterFubenUIChecker()
    self.bInit = true
  end
end

GlobalBaiHuTang:Init()
