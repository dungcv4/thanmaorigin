local tbWnd = Ui:CreateClass("UIPortraitObj", "UIPanel")

function Ui:NewExistPortraitObjPanel(tbOwner, szGroupKey)
  return Ui:CreatePanel("UIPortraitObj", szGroupKey, tbOwner)
end

local tbControls = {
  PortraitBg = "PortraitBg",
  LevelBg = "Level",
  imgLvBg = "Level/imgLvBg",
  LevelText = "Level/Text",
  LevelTextMaster = "Level/TextMaster",
  imgFactionIcon = "imgFactionIcon",
  imgIcon = "Portrait/Image",
  imgPortraitFrame = "PortraitFrame",
  OutlineObj = "Portrait/Image/OutlineObj",
  txtOutline = "Portrait/Image/OutlineObj/txtOutline",
  imgQinMiDuIcon = "imgQinMiDuIcon",
  txtQinMiDuLevel = "txtQinMiDuLevel"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
end

function tbWnd:OnClose()
end

function tbWnd:SetQinMiDuIconActive(bEnable)
  self.pPanel:SetActive(tbControls.imgQinMiDuIcon, bEnable)
end

function tbWnd:SetQinMiDuLevelActive(bEnable)
  self.pPanel:SetActive(tbControls.txtQinMiDuLevel, bEnable)
end

function tbWnd:SetQinMiDuLevel(nLevel)
  self.pPanel:Label_SetText(tbControls.txtQinMiDuLevel, nLevel)
end

function tbWnd:SetLevelActive(bEnable)
  self.pPanel:SetActive(tbControls.LevelBg, bEnable)
end

function tbWnd:SetLevel(nLevel)
  if nLevel >= GameSetting.MAX_PLAYER_LEVEL then
    self.pPanel:Label_SetText(tbControls.LevelTextMaster, nLevel - GameSetting.MAX_PLAYER_LEVEL)
    self.pPanel:SetActive(tbControls.LevelTextMaster, true)
    self.pPanel:SetActive(tbControls.LevelText, false)
  else
    self.pPanel:Label_SetText(tbControls.LevelText, nLevel)
    self.pPanel:SetActive(tbControls.LevelText, true)
    self.pPanel:SetActive(tbControls.LevelTextMaster, false)
  end
end

function tbWnd:SetLevelText(szLevelText)
  self.pPanel:Label_SetText(tbControls.LevelText, szLevelText)
  self.pPanel:SetActive(tbControls.LevelText, true)
  self.pPanel:SetActive(tbControls.LevelTextMaster, false)
end

function tbWnd:SetOutlineText(szText)
  self.pPanel:Label_SetText(tbControls.txtOutline, szText)
end

function tbWnd:SetRoleIcon(nPortrait)
  if not nPortrait or nPortrait <= 0 then
    self.pPanel:Sprite_SetSprite(tbControls.imgIcon, "UI/icon/portrait/unknown.png")
  else
    local szIcon = PlayerPortrait:GetIcon(nPortrait)
    self.pPanel:Sprite_SetSprite(tbControls.imgIcon, szIcon)
  end
end

function tbWnd:SetFactionIcon(nFaction)
  local szIcon = Faction:GetIcon(nFaction)
  self.pPanel:Sprite_SetSprite(tbControls.imgFactionIcon, szIcon)
end

function tbWnd:SetFactionIconActive(bEnable)
  self.pPanel:SetActive(tbControls.imgFactionIcon, bEnable)
end

function tbWnd:SetPortraitFrame(nPortraitFrameID)
  PlayerPortraitFrame:SetPortraitFrame(self.pPanel, tbControls.imgPortraitFrame, nPortraitFrameID)
end

function tbWnd:SetOutlineObjActive(bEnable)
  self.pPanel:SetActive(tbControls.OutlineObj, bEnable)
end
