ActivePoint.Def = {
  nMaxCount = 100,
  tbAwardReqCount = {
    20,
    40,
    60,
    80,
    100
  }
}
ActivePoint.nKeyGroup = GetGroupKey("ActivePoint")
ActivePoint.nCountKey = 1
ActivePoint.nAwardKey = 2
ActivePoint.nRecordBegin = 3
ActivePoint.nEndKey = 5
ActivePoint.REFRESH_OFFSET = 14400
ActivePoint.nLastFullDayKey = 6
ActivePoint.nContinuousDayNumKey = 7

function ActivePoint:LoadSetting()
  self.tbActSetting = Lib:LoadTabFileEx("Setting/Calendar/ActivePointSetting.tab", "sdd", "StringKey", {
    "StringKey",
    "SaveKey",
    "nCount"
  })
  self.tbActAwardSetting = Lib:LoadTabFileEx("Setting/Calendar/ActivePointAward.tab", "ddd", "AwardIndex", {
    "AwardIndex",
    "AwardSettingId",
    "AwardSettingKey"
  })
end

function ActivePoint:GetActSetting(szKey)
  return self.tbActSetting[szKey]
end

function ActivePoint:GetCount(pPlayer)
  return pPlayer.GetUserValue(self.nKeyGroup, self.nCountKey)
end

function ActivePoint:GetKeyAndBit(nSaveKey)
  local nKey = self.nRecordBegin + math.floor(nSaveKey / 32)
  local nBit = nSaveKey % 32
  return nKey, nBit
end

function ActivePoint:CheckFinish(pPlayer, nSaveKey)
  local nKey, nBit = self:GetKeyAndBit(nSaveKey)
  local nValue = pPlayer.GetUserValue(self.nKeyGroup, nKey)
  if 1 << nBit - 1 & nValue > 0 then
    return true
  end
  return false
end

function ActivePoint:CheckFinishByKey(pPlayer, szKey)
  local tbSetting = self:GetActSetting(szKey)
  if not tbSetting then
    assert(false, "ActivePoint:FinishTarget fail: can not find this szKey")
    return false
  end
  return self:CheckFinish(pPlayer, tbSetting.SaveKey)
end

function ActivePoint:CheckAward(pPlayer, nIndex)
  local nValue = pPlayer.GetUserValue(self.nKeyGroup, self.nAwardKey)
  if 1 << nIndex - 1 & nValue > 0 then
    return true
  end
  return false
end

ActivePoint:LoadSetting()
