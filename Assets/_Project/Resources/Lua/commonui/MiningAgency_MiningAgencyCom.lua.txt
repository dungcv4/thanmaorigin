function MiningAgency:CreateNewProcessor(nPlayerId1, nPlayerId2)
  return Lib:NewClass(self.ProcessorClass, nPlayerId1, nPlayerId2)
end

function MiningAgency:CheckItemCanMining(nItemTemplateId)
  if not self.tbItemConfig[nItemTemplateId] then
    return false, self.ERR.ITEM_CAN_NOT_MINING
  end
  if self.tbItemConfig[nItemTemplateId].szTimeFrameStart and 1 ~= GetTimeFrameState(self.tbItemConfig[nItemTemplateId].szTimeFrameStart) then
    return false, self.ERR.OPEN_TIMEFRAME
  end
  if self.tbItemConfig[nItemTemplateId].szTimeFrameEnd and 0 ~= GetTimeFrameState(self.tbItemConfig[nItemTemplateId].szTimeFrameEnd) then
    return false, self.ERR.CLOSE_TIMEFRAME
  end
  return true
end

function MiningAgency:CheckDistance(pNpc1, pNpc2)
  if not pNpc1 or not pNpc2 then
    return false
  end
  if pNpc1.nMapTemplateId ~= pNpc2.nMapTemplateId then
    return false, self.ERR.NOT_IN_SAME_MAP
  end
  local nDistance = pNpc1.GetDistance(pNpc2.nId)
  if nDistance < 0 or nDistance >= MiningAgency.nDistance then
    return false, self.ERR.TOO_FAR
  end
  return true
end
