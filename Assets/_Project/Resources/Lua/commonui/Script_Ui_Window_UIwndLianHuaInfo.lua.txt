local tbWnd = Ui:CreateClass("UIWndLianHuaInfo", "UIPanel")
tbWnd.tbControls = {
  CurTemplate = "imgBg/CurrentAttr/AttrList/Template",
  txtCurrentLv = "imgBg/CurrentAttr/txtCurrentLv",
  txtNextLv = "imgBg/NextLvAttr/txtNextLv",
  txtActiveEquip = "imgBg/NextLvAttr/txtNextLv/txtActiveEquip",
  NextTemplate = "imgBg/NextLvAttr/AttrList/Template",
  Empty4Raycast = "Empty4Raycast",
  txtTitle = "imgBg/txtTitle"
}
tbWnd.tbOnClick = {
  Empty4Raycast = function(self)
    self.pPanel:SetActive("Main", false)
  end
}

function tbWnd:OnCreate()
  self.tbCurAttrScrollView = Ui:NewScrollView(self, self.CurTemplate, nil, false, true)
  self.tbNextAttrScrollView = Ui:NewScrollView(self, self.NextTemplate, nil, false, true)
end

function tbWnd:UpdateView(pAsyncData)
  local tbInfoCfg = Item.EquipLianHua:GetLianHuaScoreCfg()
  if not tbInfoCfg then
    self.pPanel:SetActive("Main", false)
    return
  end
  local nAllLianHuaLevel = self:GetELHGradeScoreLevel(pAsyncData)
  self.tbCurAttrScrollView:Reset()
  self.tbNextAttrScrollView:Reset()
  if nAllLianHuaLevel >= tbInfoCfg.nPartnerNeedLevel then
    self.pPanel:SetActive(self.txtCurrentLv, true)
    self.pPanel:Label_SetText(self.txtCurrentLv, i18n.EasyGet("\231\130\188\229\140\150\229\147\129\233\152\182{0}", tbInfoCfg.nPartnerNeedLevel))
    self.pPanel:Label_SetText(self.txtNextLv, "")
    self:UpdateGradeScoreSV(2, nil)
  elseif nAllLianHuaLevel >= tbInfoCfg.nSecBookNeedLevel then
    self.pPanel:SetActive(self.txtCurrentLv, true)
    self.pPanel:Label_SetText(self.txtCurrentLv, i18n.EasyGet("\231\130\188\229\140\150\229\147\129\233\152\182{0}", tbInfoCfg.nSecBookNeedLevel))
    self.pPanel:Label_SetText(self.txtNextLv, i18n.EasyGet("\231\130\188\229\140\150\229\147\129\233\152\182{0}", tbInfoCfg.nPartnerNeedLevel))
    self:UpdateGradeScoreSV(1, 2)
  else
    self.pPanel:SetActive(self.txtCurrentLv, false)
    self.pPanel:Label_SetText(self.txtNextLv, i18n.EasyGet("\231\130\188\229\140\150\229\147\129\233\152\182{0}", tbInfoCfg.nSecBookNeedLevel))
    self:UpdateGradeScoreSV(nil, 1)
  end
end

function tbWnd:UpdateGradeScoreSV(nCurNum, nNextNum)
  self.tbCurAttrScrollView:Reset()
  if nCurNum then
    for i = 1, nCurNum do
      self.tbCurAttrScrollView:SetElement(i, function(tbElement)
        tbElement.pPanel:Label_SetText("txtAttr", Item.EquipLianHua.tbGradeScoreAddText[i]())
        tbElement.pPanel:Label_SetText("txtAttrValue", "+" .. Item.EquipLianHua.tbGradeScoreAddValue[i])
      end)
    end
  else
    self.tbCurAttrScrollView:SetElement(1, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttr", i18n.EasyGet("\230\156\170\230\191\128\230\180\187"))
      tbElement.pPanel:Label_SetText("txtAttrValue", "")
    end)
  end
  self.tbNextAttrScrollView:Reset()
  if nNextNum then
    for i = 1, nNextNum do
      self.tbNextAttrScrollView:SetElement(i, function(tbElement)
        tbElement.pPanel:Label_SetText("txtAttr", Item.EquipLianHua.tbGradeScoreAddText[i]())
        tbElement.pPanel:Label_SetText("txtAttrValue", "+" .. Item.EquipLianHua.tbGradeScoreAddValue[i])
      end)
    end
  else
    self.tbNextAttrScrollView:SetElement(1, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttr", i18n.EasyGet("\229\183\178\232\190\190\229\189\147\229\137\141\230\156\128\233\171\152\232\175\132\231\186\167"))
      tbElement.pPanel:Label_SetText("txtAttrValue", "")
    end)
  end
end

function tbWnd:GetELHGradeScoreLevel(pAsyncData)
  local nAllLevel = 0
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    if pAsyncData then
      local nLianHuaLeve = pAsyncData.GetLianHua(i)
      nAllLevel = nAllLevel + nLianHuaLeve
    else
      local nLianHuaLeve = Item.EquipLianHua:GetLianHuaLevel(me, i - 1)
      nAllLevel = nAllLevel + nLianHuaLeve
    end
  end
  return nAllLevel
end
