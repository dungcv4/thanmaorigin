Player.STEALSTATE_MAGIC_NAME = "attack_stealstate"
Player.tbStealStateSkillIdDict = nil

function Player:UnlockStealState()
  if not self:IsLockStealState(me) then
    me.CenterMsg(i18n.EasyGet("\232\167\163\233\148\129\231\138\182\230\128\129\239\188\140\230\148\187\229\135\187\231\155\174\230\160\135\230\151\182\228\188\154\229\129\183\229\143\150\230\149\140\228\186\186\231\138\182\230\128\129"))
    return
  end
  RemoteServer.UnlockStealState()
end

function Player:RefreshStealState()
  local tbWnd = Ui:GetWindow("UIHudRightSkill")
  if tbWnd then
    tbWnd:UpdateStealStateLock()
  end
end

function Player:IsStealStateSkillId(nSkillId)
  if not self.tbStealStateSkillIdDict then
    self:InitStealStateSkillId()
  end
  return self.tbStealStateSkillIdDict[nSkillId]
end

function Player:InitStealStateSkillId()
  self.tbStealStateSkillIdDict = {}
  for nFaction = 1, Faction.MAX_FACTION_COUNT do
    for nFactionSect = 0, 1 do
      local tbSkillList = FightSkill:GetFactionSkill(nFaction, nFactionSect)
      for _, tbInfo in pairs(tbSkillList) do
        local tbMagicList = KFightSkill.GetSkillAllMagic(tbInfo.SkillId, 1)
        local bFound = false
        for _, tbMA in pairs(tbMagicList) do
          if tbMA.szName == self.STEALSTATE_MAGIC_NAME then
            bFound = true
            break
          end
        end
        if bFound then
          self.tbStealStateSkillIdDict[tbInfo.SkillId] = true
        end
      end
    end
  end
end

function Player:GetCurStealState()
  if not me.GetNpc().GetSkillState(self.nStealStateShowSkillId) then
    return 0, 0
  end
  local nValue = me.GetUserValue(self.nStealStateSaveGroup, self.nStealStateInfoKey)
  local nSkillId = nValue >> 16
  local nSkillLevel = nValue & 65535
  return nSkillId, nSkillLevel
end
