local tbWnd = Ui:CreateWindow("UILock")
local tbControls = {
  txtTitle = "imgBG/UIBg/txtTitle",
  Input = "imgBG/UIBg/InputCell/inputNew",
  btnNarrow = "imgBG/UIBg/InputCell/btns/btnNarrow",
  btnCancel = "imgBG/UIBg/InputCell/btns/btnCancel",
  btnConfirm = "imgBG/UIBg/InputCell/btns/btnConfirm"
}
tbWnd.tbControls = tbControls
local nMaxNum = 10
local tbType2Tittle = {
  [SafeLock.OperationTypeDef.SetPassWord] = function()
    return i18n.EasyGet("\232\174\190\231\189\174\229\175\134\231\160\129")
  end,
  [SafeLock.OperationTypeDef.UnLock] = function()
    return i18n.EasyGet("\232\190\147\229\133\165\229\175\134\231\160\129")
  end,
  [SafeLock.OperationTypeDef.ChangePassWord] = function()
    return i18n.EasyGet("\232\190\147\229\133\165\230\151\167\229\175\134\231\160\129")
  end,
  [SafeLock.OperationTypeDef.CancelPassWord] = function()
    return i18n.EasyGet("\232\190\147\229\133\165\230\151\167\229\175\134\231\160\129")
  end
}
local tbNeedConfirm = {
  [SafeLock.OperationTypeDef.SetPassWord] = true
}
tbWnd.tbOnClick = {
  btnCancel = function(self)
    self:Close()
  end,
  btnConfirm = function(self)
    if Lib:IsEmptyStr(self.szInput) then
      return
    end
    if #self.szInput ~= SafeLock.Cfg.nPassWordLenth then
      me.CenterMsg(i18n.Get(SafeLock.Msg.ERR_PASSWORDSHORT, SafeLock.Cfg.nPassWordLenth))
      return
    end
    if tbNeedConfirm[self.nType] then
      if Lib:IsEmptyStr(self.szOldInput) then
        self.szOldInput = self.szInput
        self.szInput = ""
        self.pPanel:Input_SetText(tbControls.Input, "")
        self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\232\175\183\229\134\141\230\172\161\232\190\147\229\133\165\229\175\134\231\160\129"))
        return
      end
      if self.szOldInput ~= self.szInput then
        me.CenterMsg(i18n.Get(SafeLock.Msg.ERR_PASSWORDDIFF))
        self:Close()
        return
      end
    end
    SafeLock:OnInputPassWord(self.nType, self.szInput, self.bInSetPanel)
  end,
  btnNarrow = function(self)
    if Lib:IsEmptyStr(self.szInput) then
      return
    end
    self.szInput = string.sub(self.szInput, 1, #self.szInput - 1)
    self.pPanel:Input_SetText(tbControls.Input, self.szInput)
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nType, bInSetPanel)
  self.szInput = ""
  self.szOldInput = ""
  local szTittle = tbType2Tittle[nType] and tbType2Tittle[nType]() or ""
  self.pPanel:Label_SetText(tbControls.txtTitle, szTittle)
  self.pPanel:Input_SetEnable(tbControls.Input, false)
  self.nType = nType
  self.bInSetPanel = bInSetPanel
  local fn = Lib:GetRandomSelect(nMaxNum)
  for i = 0, nMaxNum - 1 do
    local nRand = fn() - 1
    local szBtnPath = "imgBG/UIBg/InputCell/btns/btn" .. tostring(nRand)
    self.pPanel:Button_SetText(szBtnPath, tostring(i), "txt")
    Ui:Button_BindEvent(self.pPanel, szBtnPath, self.AppendPassWord, self, i)
  end
end

function tbWnd:AppendPassWord(nNums)
  if not nNums or not tostring(nNums) then
    return
  end
  if #self.szInput >= SafeLock.Cfg.nPassWordLenth then
    me.CenterMsg(i18n.Get(SafeLock.Msg.ERR_PASSWORDSHORT, SafeLock.Cfg.nPassWordLenth))
    return
  end
  self.szInput = self.szInput .. nNums
  self.pPanel:Input_SetText(tbControls.Input, self.szInput)
end

function tbWnd:OnClose()
  self:ResetInput()
  for i = 1, nMaxNum do
    local nRand = i - 1
    local szBtnPath = "imgBG/UIBg/InputCell/btns/btn" .. tostring(nRand)
    Ui:Button_UnBindEvent(self.pPanel, szBtnPath)
  end
end

function tbWnd:ResetInput()
  self.szInput = ""
  self.szOldInput = ""
  self.pPanel:Input_SetText(tbControls.Input, "")
end
