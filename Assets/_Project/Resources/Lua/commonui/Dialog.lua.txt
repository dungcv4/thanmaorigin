function Dialog:SetPlayerDialog(pPlayer, tbDialogInfo)
  pPlayer.tbDialogInfo = tbDialogInfo
end

function Dialog:GetPlayerDialogInfo(pPlayer)
  pPlayer.tbDialogInfo = pPlayer.tbDialogInfo or {
    szText = "",
    tbOptList = {}
  }
  return pPlayer.tbDialogInfo
end

function Dialog:ShowNpcDialog(varDlgInfo, pPlayer, nNpcTemplateId, bNeedLastStepDelay)
  local szNpcName = Npc:GetNameByTemplateId(nNpcTemplateId)
  self:CommonShow(varDlgInfo, pPlayer, nNpcTemplateId, szNpcName, nil, bNeedLastStepDelay)
end

function Dialog:Show(varDlgInfo, pPlayer, pNpc, bNeedLastStepDelay)
  self:CommonShow(varDlgInfo, pPlayer, pNpc.nTemplateId, pNpc.szName, pNpc.nId, bNeedLastStepDelay)
end

function Dialog:CommonShow(varDlgInfo, pPlayer, nNpcTemplateId, szNpcName, nNpcId, bNeedLastStepDelay)
  if not pPlayer or not nNpcTemplateId then
    return
  end
  local tbDlgInfo
  if type(varDlgInfo) == "table" then
    tbDlgInfo = varDlgInfo
  elseif type(varDlgInfo) == "string" then
    tbDlgInfo = {
      Text = varDlgInfo,
      OptList = {}
    }
  else
    return
  end
  tbDlgInfo.NpcId = nNpcId
  local tbDlgToClient = {
    Text = tbDlgInfo.Text,
    nCountdown = tbDlgInfo.nCountdown,
    OptList = {}
  }
  for _, tbOpt in ipairs(tbDlgInfo.OptList) do
    local tbOptToClient = {
      Text = tbOpt.Text,
      DelayClose = tbOpt.DelayClose
    }
    if tbOpt.Type or tbOpt.Callback then
      tbOptToClient.Callback = 1
      if not tbOpt.Type then
        tbOpt.Type = "Script"
      end
    end
    table.insert(tbDlgToClient.OptList, tbOptToClient)
  end
  if tbDlgInfo.tbSkip then
    tbDlgToClient.bCanSkip = 1
  end
  self:SetPlayerDialog(pPlayer, tbDlgInfo)
  if MODULE_GAMESERVER then
    pPlayer.CallClientScript("Ui:UpdateDialog", tbDlgToClient, false, szNpcName, nNpcTemplateId, tbDlgInfo.SoundId, tbDlgInfo.NpcId, bNeedLastStepDelay)
  else
    Ui:UpdateDialog(tbDlgToClient, true, szNpcName, nNpcTemplateId, tbDlgInfo.SoundId, tbDlgInfo.NpcId, bNeedLastStepDelay)
  end
end

function Dialog:MsgBox(szMsg, tbCallback, szNotTipsType, nTimeout, fnTimeout, szUiName, nTimeOutBtnIndex)
  szUiName = szUiName or "UIMessageBox"
  me.tbMsgBoxInfo = {
    tbCallback = tbCallback,
    nTimerId = nil,
    szUiName = szUiName
  }
  if MODULE_GAMESERVER and nTimeout and fnTimeout then
    me.tbMsgBoxInfo.nTimerId = Timer:Register(Env.GAME_FPS * nTimeout, function(nPlayerId)
      local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
      GameSetting:SetGlobalObj(pPlayer)
      fnTimeout()
      GameSetting:RestoreGlobalObj()
      if pPlayer then
        pPlayer.CallClientScript("Ui:CloseWindow", szUiName)
        pPlayer.tbMsgBoxInfo = nil
      end
    end, me.dwID)
  end
  local tbBtnTxt = {}
  local nCallbackCount = 0
  if tbCallback then
    for _, tbInfo in ipairs(tbCallback) do
      table.insert(tbBtnTxt, tbInfo[1])
    end
    nCallbackCount = #tbCallback
  end
  me.CallClientScript("Ui:OnServerMsgBox", szMsg, nCallbackCount, tbBtnTxt, MODULE_GAMESERVER, szNotTipsType, nTimeout, szUiName, not MODULE_GAMESERVER and fnTimeout, nTimeOutBtnIndex)
end

function Dialog:OnMsgBoxSelect(nSelect, bIsServer)
  if not MODULE_GAMESERVER and bIsServer then
    RemoteServer.OnMsgBoxSelect(nSelect)
    return
  end
  local tbMsgBoxInfo = me.tbMsgBoxInfo
  me.tbMsgBoxInfo = nil
  if tbMsgBoxInfo and tbMsgBoxInfo.nTimerId then
    Timer:Close(tbMsgBoxInfo.nTimerId)
  end
  if not tbMsgBoxInfo or not nSelect then
    return
  end
  if tbMsgBoxInfo.tbCallback and tbMsgBoxInfo.tbCallback[nSelect] and tbMsgBoxInfo.tbCallback[nSelect][2] then
    Lib:CallBack({
      table.unpack(tbMsgBoxInfo.tbCallback[nSelect], 2)
    })
  end
end

function Dialog:ShowDefaultDialog(pPlayer, pNpc)
  local szText
  local tbDialogInfo = {
    OptList = {}
  }
  if pNpc then
    if pNpc.szDefaultDialogInfo and pNpc.szDefaultDialogInfo ~= "" then
      szText = pNpc.szDefaultDialogInfo
    end
    tbDialogInfo.NpcTemplateId = pNpc.nTemplateId
    Npc:AddTaskDialogOption(tbDialogInfo, pNpc.nTemplateId)
    Npc:AddExtraDialogOption(tbDialogInfo, pPlayer, pNpc)
    Npc:AddShopDialogOption(tbDialogInfo, pNpc.nTemplateId, pNpc.nMapTemplateId)
  end
  szText = szText or i18n.EasyGet("\228\189\160\230\137\190\230\136\145\228\187\128\228\185\136\228\186\139\239\188\159")
  tbDialogInfo.Text = szText
  Dialog:Show(tbDialogInfo, pPlayer, pNpc)
end

function Dialog:SendBlackBoardMsg(pPlayer, szMsg)
  pPlayer.CenterMsg(szMsg)
  pPlayer.Msg(szMsg)
end

function Dialog:OnLogout(pPlayer)
  if pPlayer.tbMsgBoxInfo then
    local nTimerId = pPlayer.tbMsgBoxInfo.nTimerId
    if not nTimerId then
      return
    end
    Timer:Close(nTimerId)
    local szUiName = pPlayer.tbMsgBoxInfo.szUiName
    pPlayer.CallClientScript("Ui:CloseWindow", szUiName)
    pPlayer.tbMsgBoxInfo = nil
  end
end
