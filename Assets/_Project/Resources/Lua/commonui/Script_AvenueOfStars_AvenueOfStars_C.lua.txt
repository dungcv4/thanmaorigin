function AvenueOfStars:QueryVoteLevel_C()
  if me.nLevel < self.MIN_LEVEL then
    return false
  end
  if not self:IsOpenDate() then
    return false
  end
  RemoteServer.QueryAvenueOfStarsVoteLevel(self.tbVoteLevel)
  return true
end

function AvenueOfStars:SyncVoteLevel(tbVoteLevel)
  for nPhase, nLevel in pairs(tbVoteLevel) do
    self.tbVoteLevel[nPhase] = nLevel
  end
  self:CheckRewardRedPoint()
  EventNotify.OnNotify(EventNotify.emNOTIFY_AVENUEOFSTARS_SYNC_VOTE_LEVEL)
end

function AvenueOfStars:GetVoteLevelAward_C(nPhase, nIndex)
  if me.nLevel < self.MIN_LEVEL then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}\231\186\167\239\188\140\230\151\160\230\179\149\229\143\130\229\138\160\232\175\165\230\180\187\229\138\168\239\188\129", self.MIN_LEVEL))
    return false
  end
  local bSettleOpen = self:IsUseVoteLevelSettleDate(GetTime(), nPhase)
  if not bSettleOpen then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\230\156\170\229\156\168\233\162\134\229\165\150\230\156\159\233\151\180"))
    return false
  end
  local nFlag = me.GetUserValue(self.SAVE_GROUP, self.USE_VOTE_LEVEL_AWARD_FLAG[nPhase])
  if 0 ~= Lib:LoadBits(nFlag, nIndex - 1, nIndex - 1) then
    me.CenterMsg(i18n.EasyGet("\229\165\150\229\138\177\229\183\178\233\162\134\229\143\150"))
    return false
  end
  if (self.tbVoteLevel[nPhase] or 0) < self.USE_VOTE_LEVEL_AWARD[nPhase].tbAward[nIndex][1] then
    me.CenterMsg(i18n.EasyGet("\230\138\149\231\165\168\230\149\176\233\135\143\228\184\141\232\182\179"))
    return false
  end
  RemoteServer.GetAvenueOfStarsVoteLevelAward(nPhase, nIndex)
  return true
end

function AvenueOfStars:CheckPhaseRedPoint()
  if me.nLevel < self.MIN_LEVEL then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.AvenueOfStarsPhase)
    return
  end
  local szDate = Lib:OSDate("%Y.%m.%d")
  if szDate == self.NOTIFY_DATE then
    if not LocalData:GetRoleBoolean("AvenueOfStars_RedPoint_Notify", false) then
      LocalData:SetRoleBoolean("AvenueOfStars_RedPoint_Notify", true)
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.AvenueOfStarsPhase)
    end
  elseif szDate == self.ENROLL_DATE_YMD[1] then
    if not LocalData:GetRoleBoolean("AvenueOfStars_RedPoint_Enroll", false) then
      LocalData:SetRoleBoolean("AvenueOfStars_RedPoint_Enroll", true)
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.AvenueOfStarsPhase)
    end
  else
    for nPhase, tbDate in pairs(self.VOTE_DATE_YMD) do
      local szKey = "AvenueOfStars_RedPoint_Phase_" .. nPhase
      if szDate == tbDate[1] then
        if not LocalData:GetRoleBoolean(szKey, false) then
          LocalData:SetRoleBoolean(szKey, true)
          RedPointMgr:LightRedPoint(RedPointMgr.EVENT.AvenueOfStarsPhase)
        end
        break
      end
    end
  end
end

function AvenueOfStars:CheckRewardRedPoint()
  if me.nLevel < self.MIN_LEVEL then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.AvenueOfStarsRewardSemi)
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.AvenueOfStarsRewardFinal)
    return
  end
  local tbRedPointList = {
    [self.PHASE_PRE] = RedPointMgr.EVENT.AvenueOfStarsRewardSemi,
    [self.PHASE_SEMI] = RedPointMgr.EVENT.AvenueOfStarsRewardFinal
  }
  for nPhase, tbAwardSetting in pairs(self.USE_VOTE_LEVEL_AWARD) do
    local bHasAward = false
    for nIndex, tbAward in ipairs(tbAwardSetting.tbAward) do
      local nFlag = me.GetUserValue(self.SAVE_GROUP, self.USE_VOTE_LEVEL_AWARD_FLAG[nPhase])
      local nVoteLevel = self.tbVoteLevel[nPhase] or 0
      if 0 == Lib:LoadBits(nFlag, nIndex - 1, nIndex - 1) and nVoteLevel >= tbAward[1] then
        bHasAward = true
        break
      end
    end
    if bHasAward then
      RedPointMgr:LightRedPoint(tbRedPointList[nPhase])
    else
      RedPointMgr:FadeRedPoint(tbRedPointList[nPhase])
    end
  end
end

function AvenueOfStars:OnLogin()
  self.tbVoteLevel = {}
  self:QueryVoteLevel_C()
  self:CheckPhaseRedPoint()
end
