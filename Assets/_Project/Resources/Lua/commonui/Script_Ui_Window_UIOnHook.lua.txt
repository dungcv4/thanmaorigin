function tbWnd:OnCreate()
  self.pPanel:SetActive(tbControls.TemplateTeammate .. "2/txtTips", true)
  self.pPanel:SetActive(tbControls.TemplateTeammate .. "3/txtTips", true)
end

function tbWnd:OnOpen()
  OfflineExp:RefreshClientMaxLevel()
  OfflineExp:RefreshOfflineExp()
  self:UpdateView()
  OfflineExp.bClickUI = true
  OfflineExp:OnCheckRedPoint()
end

function tbWnd:UpdateView()
  local nExpTime = OfflineExp:ExpTime(me)
  Lib:LogFormat("<OfflineExp>Client UpdateView, Cur ExpTime is %d", nExpTime)
  local nBaiJuWanTime = OfflineExp:BaiJuWanTime(me)
  local nSpecialBaiJuWanTime = OfflineExp:SpecialBaiJuWanTime(me)
  self.pPanel:Label_SetText(tbControls.txtTime2, i18n.EasyGet("{0}", Lib:TimeDesc8(nExpTime)))
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "2/txtTime", i18n.EasyGet("{0}", Lib:TimeDesc8(nBaiJuWanTime)))
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "3/txtTime", i18n.EasyGet("{0}", Lib:TimeDesc8(nSpecialBaiJuWanTime)))
  local nAddSpecialPayExp = OfflineExp:NowExp(me, OfflineExp.OfflineExpType.SpecialPay)
  local nAddPayExp = OfflineExp:NowExp(me, OfflineExp.OfflineExpType.Pay)
  local nAddFreeExp = OfflineExp:NowExp(me, OfflineExp.OfflineExpType.Free)
  local szFreeGetExpRate = i18n.EasyGet("\232\142\183\229\190\151{0}\229\128\141\231\187\143\233\170\140", OfflineExp.fFreeGetExpRate)
  local szPayGetExpRate = i18n.EasyGet("\232\142\183\229\190\151{0}\229\128\141\231\187\143\233\170\140", OfflineExp.fPayGetExpRate)
  local szPayTips = OfflineExp:PayTips(me, OfflineExp.OfflineExpType.Pay)
  local szSpecialPayGetExpRate = i18n.EasyGet("\232\142\183\229\190\151{0}\229\128\141\231\187\143\233\170\140", OfflineExp.fSpecialPayGetExpRate)
  local szSpecialPayTips = OfflineExp:PayTips(me, OfflineExp.OfflineExpType.SpecialPay)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "1/txtExpTip", szFreeGetExpRate)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "2/txtExpTip", szPayGetExpRate)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "3/txtExpTip", szSpecialPayGetExpRate)
  self.nAddFreeExp = nAddFreeExp
  self.nAddPayExp = nAddPayExp
  self.nAddSpecialPayExp = nAddSpecialPayExp
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "1/txtExp", nAddFreeExp)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "2/txtExp", nAddPayExp)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "3/txtExp", nAddSpecialPayExp)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "2/txtTips", szPayTips)
  self.pPanel:Label_SetText(tbControls.TemplateTeammate .. "3/txtTips", szSpecialPayTips)
  self:UpdateTip()
  self:ShowSpecialPanel()
  self:UpdateButton()
end

function tbWnd:UpdateButton()
  local nAddFreeExp = OfflineExp:NowExp(me, OfflineExp.OfflineExpType.Free)
  local nAddPayExp = OfflineExp:NowExp(me, OfflineExp.OfflineExpType.Pay)
  local nAddSpecialPayExp = OfflineExp:NowExp(me, OfflineExp.OfflineExpType.SpecialPay)
  if nAddFreeExp > 0 then
    self.pPanel:Sprite_SetSprite(tbControls.btnFreeGet, "UI/atlas/common_btn/btn_common2_2.png")
  else
    self.pPanel:Sprite_SetSprite(tbControls.btnFreeGet, Ui.szGrayPath)
  end
  if nAddPayExp > 0 then
    self.pPanel:Sprite_SetSprite(tbControls.btnPayGet, "UI/atlas/common_btn/btn_common2_2.png")
  else
    self.pPanel:Sprite_SetSprite(tbControls.btnPayGet, Ui.szGrayPath)
  end
  if nAddSpecialPayExp > 0 then
    self.pPanel:Sprite_SetSprite(tbControls.btnSpecialGet, "UI/atlas/common_btn/btn_common2_2.png")
  else
    self.pPanel:Sprite_SetSprite(tbControls.btnSpecialGet, Ui.szGrayPath)
  end
end

function tbWnd:OnClose()
end

function tbWnd:UpdateTip()
end

function tbWnd:ShowSpecialPanel()
  local bShow = OfflineExp:CheckSpecialBaiJuWanIsOpen(me)
  self.pPanel:SetActive(tbControls.TemplateTeammate .. "3", bShow)
  if bShow then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\230\137\152\231\174\161\228\184\137\233\128\137\228\184\128"))
  else
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\230\137\152\231\174\161\228\186\140\233\128\137\228\184\128"))
  end
end
