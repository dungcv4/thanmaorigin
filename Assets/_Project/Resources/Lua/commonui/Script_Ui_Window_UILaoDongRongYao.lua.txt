local tbWnd = Ui:CreateWindow("UILaoDongRongYao")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_LABORHONOR_DATA_SYNC,
      self.UpdateActivityInfo
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  txtActivtyTime = "node/imgBG/Content/ActivtyTime/txtActivtyTime",
  txtTime = "node/imgBG/Content/ActivtyTime/txtTime",
  Template = "node/imgBG/Content/AwardPreview/AwardList/Template",
  btnSubmit = "node/imgBG/Content/Detail/btnSubmit",
  SubmitCount = "node/imgBG/Content/Detail/SubmitCount",
  txtSubmitValue = "node/imgBG/Content/Detail/SubmitTips/txtSubmitValue",
  Progress = "node/imgBG/Content/Detail/Progress",
  txtProgressTip = "node/imgBG/Content/Detail/ProgressInfo/txtProgressTip",
  txtProgressValue = "node/imgBG/Content/Detail/ProgressInfo/txtProgressValue",
  imgMark5000 = "node/imgBG/Content/Detail/Progress/imgMark5000/txt5000",
  imgMark10000 = "node/imgBG/Content/Detail/Progress/imgMark10000/txt10000",
  imgMark20000 = "node/imgBG/Content/Detail/Progress/imgMark20000/txt20000",
  imgMark30000 = "node/imgBG/Content/Detail/Progress/imgMark30000/txt30000",
  TrailEffect = "node/imgBG/Content/Detail/Progress/TrailEffect"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("LaborHonor")
  end,
  btnSubmit = function(self)
    self:GotoSubmission()
  end
}

function tbWnd:OnCreate()
  self.tbMarkList = {
    tbControls.imgMark5000,
    tbControls.imgMark10000,
    tbControls.imgMark20000,
    tbControls.imgMark30000
  }
  self.tbAwardSV = Ui:NewItemGridScrollView(self, tbControls.Template, 1, 1)
end

function tbWnd:OnOpen(nId)
  self.nActivityId = nId
  self:UpdateActivityInfo(true)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnSubmit, {
    RedPointMgr.EVENT.LaborHonor
  })
end

function tbWnd:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnSubmit)
end

function tbWnd:UpdateActivityInfo(bSync)
  local pInstance = ActivityMgr:GetInstanceById(self.nActivityId)
  if not pInstance then
    return
  end
  if bSync then
    pInstance:ReqSyncData()
  end
  local tbInfo = pInstance:GetShowInfo()
  self.pPanel:Label_SetText(tbControls.txtActivtyTime, i18n.EasyGet("\230\180\187\229\138\168\230\151\182\233\151\180\239\188\154"))
  self.pPanel:Label_SetText(tbControls.txtTime, tbInfo.szShowTime)
  self.pPanel:Label_SetText(tbControls.txtSubmitValue, tbInfo.nCanSubmitValue)
  self.pPanel:Label_SetText(tbControls.SubmitCount .. "/txtSubmitCountValue", tbInfo.nCurNum)
  self.pPanel:Label_SetText(tbControls.SubmitCount .. "/txtSubmitCountTip", i18n.EasyGet("\228\189\160\229\156\168\230\156\172\229\184\174\228\188\154\230\143\144\228\186\164\230\128\187\230\149\176{0}-{1}\230\156\159\233\151\180\231\154\132\230\143\144\228\186\164\228\184\170\230\149\176\239\188\154", tbInfo.nMin, tbInfo.nMax))
  self.pPanel:Sprite_SetFillAmount(tbControls.Progress .. "/imgBarBg/imgBar", tbInfo.fBarRatio, tbInfo.fBarRatio)
  self:MoveDonkey(tbInfo)
  for nIndex, nValue in ipairs(tbInfo.tbLevel) do
    if self.tbMarkList[nIndex] then
      self.pPanel:Label_SetText(self.tbMarkList[nIndex], nValue)
    end
  end
  self.tbAwardSV:UpdateGenericItem(tbInfo.tbAward, true)
  self.tbAwardSV:ResetParent()
  self.pPanel:Label_SetText(tbControls.txtProgressTip, i18n.EasyGet("\229\189\147\229\137\141\229\184\174\228\188\154\230\143\144\228\186\164\232\191\155\229\186\166\239\188\154"))
  self.pPanel:Label_SetText(tbControls.txtProgressValue, tbInfo.nTotalNum)
end

function tbWnd:MoveDonkey(tbInfo)
  local nBegin, nEnd = -380, 380
  self.pPanel:Object_SetLocalPosition(tbControls.TrailEffect, Ui.Vector3(nBegin, 0, 0))
  local pDonkeyObj = self.pPanel:GetObject(tbControls.TrailEffect)
  if not pDonkeyObj:GetComponent(Ui.UIDoTweenAnimation) then
    pDonkeyObj:AddComponent(Ui.UIDoTweenAnimation)
  end
  local effect = pDonkeyObj:GetComponent(Ui.UIDoTweenAnimation)
  local rt = self.pPanel:GetTransform(tbControls.TrailEffect)
  effect:Init(rt, false, false)
  local tweener = effect:DOLocalMove(tbInfo.fBarRatio * (nEnd - nBegin) + nBegin, 0, 1, 0.3)
  effect:Append(tweener)
end

function tbWnd:GotoSubmission()
  local pInstance = ActivityMgr:GetInstanceById(self.nActivityId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return
  end
  local tbConfig = pInstance:GetConfig()
  ActivityMgr:GotoNpc(tbConfig.nNpcTemplateId, tbConfig.nMapTemplateId)
  self:Close()
  if Ui:GetWindow("UIXiongWangActivity") then
    Ui:CloseWindow("UIXiongWangActivity")
  end
end
