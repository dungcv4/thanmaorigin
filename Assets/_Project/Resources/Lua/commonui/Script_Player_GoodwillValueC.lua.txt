GoodwillValue.nValue = nil

function GoodwillValue:GetMyGoodwillValue()
  if self.nValue == nil then
    self.nValue = self:GetGoodwillValue(me)
  end
  return self.nValue
end

function GoodwillValue:S2CUpdateValue(nValue)
  if nil ~= nValue then
    self.nValue = nValue
    EventNotify.OnNotify(EventNotify.emNOTIFY_GOODWILL_VALUE_CHANGE)
  end
end

function GoodwillValue:S2CInjureRuduceExp(nReduceExp)
  if nil ~= nReduceExp and nReduceExp > 0 then
    ChatMgr:OnChannelMessage(ChatMgr.ChannelType.System, 0, ChatMgr.SystemMsgType.System, "", 0, 0, 0, i18n.Get(self.tbText.szInjureShowReduceExp, nReduceExp))
  end
end
