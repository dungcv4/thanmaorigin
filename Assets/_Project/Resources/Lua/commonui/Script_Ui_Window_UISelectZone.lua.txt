local tbWnd = Ui:CreateWindow("UISelectZone")
tbWnd.tbControls = {
  btnTemplate = "imgBG/ZoneList/ZoneList/btnTemplate"
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.tbScrollView = Ui:NewScrollView(self, self.btnTemplate, function(self, nIndex, tbElement)
    Login.CurrentSelectZoneID = tbElement.nZoneID
    Ui:GetWindow("UILoginChannelInner"):SetCurZoneID(tbElement.nZoneID)
    Ui.PlayerPrefs.SetInt("LastZoneID", tbElement.nZoneID)
    self:Close()
  end, false, true)
  self:initZoneButtons()
end

function tbWnd:initZoneButtons()
  self.tbScrollView:Reset()
  local tbList = Login:GetZoneList()
  for nIndex, tbZone in ipairs(tbList) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      tbElement.nZoneID = nIndex
      tbElement.pPanel:Label_SetText("Text", tbZone.Name)
    end)
  end
end
