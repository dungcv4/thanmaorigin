local tbWnd = Ui:CreateWindow("UIShowTitle")
local tbControls = {
  btnClose = "panel2/btnClose",
  btnSave = "panel2/btnSave",
  txtFightPowerValue = "panel2/Bottom/FightPower/txtFightPowerValue",
  txtLevel = "panel2/Bottom/PlayerInfo/txtLevel",
  txtPlayerName = "panel2/Bottom/PlayerInfo/txtPlayerName",
  txtServer2 = "panel2/Bottom/Server/txtServer2",
  HonorTitle = "panel2/TitleShow/HonorTitle"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSave = function(self)
    self:OnClickSave()
  end
}

function tbWnd:OnCreate()
  self:SetGroup(2)
  self.pHonorTitle = self.pPanel:GetObject(tbControls.HonorTitle):GetComponent(Ui.UIHonorTitle)
end

function tbWnd:OnOpen()
  Client.UIModule.SetGroupActive(1, false)
  self.pPanel:SetActive(self.btnClose, true)
  self.pPanel:SetActive(self.btnSave, true)
  self:UpdateUI(true)
end

function tbWnd:OnClose()
  Client.UIModule.SetGroupActive(1, true)
  EventNotify.OnNotify(EventNotify.emNOTIFY_SUPER_VIEW_REFRESH)
end

function tbWnd:UpdateUI(bInit)
  self.pPanel:Label_SetText(self.txtFightPowerValue, me.GetNpc().GetFightPower())
  if me.nLevel < GameSetting.MAX_PLAYER_LEVEL then
    self.pPanel:Label_SetText(self.txtLevel, i18n.EasyGet("{0}\231\186\167", me.nLevel))
  else
    self.pPanel:Label_SetText(self.txtLevel, i18n.EasyGet("{0}", Lib:GetLevelText(me.nTotalLevel)))
  end
  self.pPanel:Label_SetText(self.txtPlayerName, me.szName)
  self.pPanel:Label_SetText(self.txtServer2, Login.ServerName)
  self:UpdateHonorTitle()
end

function tbWnd:UpdateHonorTitle()
  local nStarLevel = HonorLevel:GetStarLevel(me)
  if 0 == nStarLevel then
    return
  end
  local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(nStarLevel, false)
  self.pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
end

function tbWnd:OnClickSaveHelper()
  local function fnSave()
    Ui:SetCameraCaptureCallback(function()
      self:Close()
      
      local path = Ui.SceneCameraController.SaveRenderTexture()
      if Sdk:GetOsType() == Sdk.eOSType_Windows then
        me.Msg(i18n.EasyGet("\230\136\144\229\138\159\228\191\157\229\173\152, \230\150\135\228\187\182\229\144\141\239\188\154{0}", path))
      else
        me.Msg(i18n.EasyGet("\230\136\144\229\138\159\228\191\157\229\173\152\232\135\179\231\155\184\229\134\140\228\184\173"))
      end
      me.CenterMsg(i18n.EasyGet("\230\136\144\229\138\159\228\191\157\229\173\152"))
    end)
    self.pPanel:SetActive(self.btnClose, false)
    self.pPanel:SetActive(self.btnSave, false)
    Ui.SceneCameraController.StartCapture()
    EventNotify.OnNotify(EventNotify.emNOTIFY_SUPER_VIEW_REFRESH)
  end
  
  local nRet = Ui.SceneCameraController.CheckExternalWriteSDCardPermission()
  if Ui.GameEnv.EnableSDK then
    if -1 == nRet or 0 == nRet then
      Ui:OpenWindow("UIRequestPermission", i18n.EasyGet("\233\156\128\232\166\129\232\174\191\233\151\174\232\174\190\229\164\135\228\184\138\231\154\132\231\133\167\231\137\135\229\146\140\229\170\146\228\189\147\231\154\132\230\157\131\233\153\144\231\148\168\228\186\142\229\176\134\230\184\184\230\136\143\228\184\173\231\154\132\231\133\167\231\137\135\228\191\157\229\173\152\229\136\176\231\155\184\229\134\140\228\184\173\227\128\130"), "WriteSDCard")
    else
      fnSave()
    end
  elseif -1 == nRet then
    Ui:OpenWindow("UIRequestPermission", i18n.EasyGet("\233\156\128\232\166\129\232\174\191\233\151\174\232\174\190\229\164\135\228\184\138\231\154\132\231\133\167\231\137\135\229\146\140\229\170\146\228\189\147\231\154\132\230\157\131\233\153\144\231\148\168\228\186\142\229\176\134\230\184\184\230\136\143\228\184\173\231\154\132\231\133\167\231\137\135\228\191\157\229\173\152\229\136\176\231\155\184\229\134\140\228\184\173\227\128\130"), "WriteSDCard")
  elseif 0 == nRet then
    Ui:OpenWindow("UIRequestPermission", i18n.EasyGet("\233\156\128\232\166\129\232\174\191\233\151\174\232\174\190\229\164\135\228\184\138\231\154\132\231\133\167\231\137\135\229\146\140\229\170\146\228\189\147\231\154\132\230\157\131\233\153\144\231\148\168\228\186\142\229\176\134\230\184\184\230\136\143\228\184\173\231\154\132\231\133\167\231\137\135\228\191\157\229\173\152\229\136\176\231\155\184\229\134\140\228\184\173\227\128\130"), "WriteSDCard")
  else
    fnSave()
  end
end

function tbWnd:OnClickSave()
  if Sdk:GetOsType() ~= Sdk.eOSType_iOS then
    return self:OnClickSaveHelper()
  end
  Log("UIShowTitle:OnClickIosSave")
  local nPermission = Ui.SceneCameraController.GetWritePhotoPermissionState()
  if 1 == nPermission then
    self:OnClickSaveHelper()
  elseif 0 == nPermission then
    Ui.SceneCameraController.GetWritePhotoPermission()
  else
    self:Close()
    me.CenterMsg(i18n.EasyGet("\230\178\161\230\156\137\228\191\157\229\173\152\229\155\190\231\137\135\230\157\131\233\153\144\239\188\140\228\191\157\229\173\152\229\164\177\232\180\165"))
  end
end
