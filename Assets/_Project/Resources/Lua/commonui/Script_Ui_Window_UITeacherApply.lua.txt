local tbUi = Ui:CreateWindow("UITeacherApply")
tbUi.ItemTemplate = {
  RoleTemplete = "node/imgBG/Content/imgApplyList/ApplyList/ApplyTemplate"
}
tbUi.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnClearAllApply = "node/imgBG/Content/btnClearAllApply",
  NotInfo = "node/imgBG/Content/NotInfo"
}

function tbUi:OnCreate()
  self.tbItemScrollView = Ui:NewScrollView(self, self.ItemTemplate.RoleTemplete)
end

function tbUi:OnOpenEnd()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MP_TeacherApply)
  MasterPrentice:GetApplies()
end

function tbUi:Refresh(tbAllRequet)
  if not tbAllRequet or 0 == #tbAllRequet then
    self.pPanel:SetActive(self.NotInfo, true)
    self.tbItemScrollView:Clear()
    return
  end
  self.tbAllRequet = tbAllRequet
  self.tbItemScrollView:Clear()
  for i = 1, #tbAllRequet do
    self.tbItemScrollView:SetElement(i, function(tbElement)
      if tbAllRequet[i] then
        self:SetScrollViewData(i, tbElement, tbAllRequet[i])
      end
    end)
  end
  self.pPanel:SetActive(self.NotInfo, not self.tbAllRequet or Lib:IsEmptyTB(self.tbAllRequet))
end

function tbUi:SetScrollViewData(nIndex, tbElement, tbApplyInfo)
  local panel = tbElement.pPanel
  self.tbApplyInfo = tbApplyInfo
  local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
  tbPortraitObjPanel:SetRoleIcon(tbApplyInfo[6])
  tbPortraitObjPanel:SetPortraitFrame(tbApplyInfo[11] or 0)
  tbPortraitObjPanel:SetFactionIcon(tbApplyInfo[10])
  tbPortraitObjPanel:SetLevel(tbApplyInfo[4])
  local nHonorStarLevel = tbApplyInfo[8] or 0
  if nHonorStarLevel > 0 then
    panel:SetActive("txtPlayerTitle", true)
    local szHonorName, nFontColor = HonorLevel:GetHonorNameByStarLevel(nHonorStarLevel, true)
    panel:Label_SetText("txtPlayerTitle", i18n.Parse(szHonorName))
    panel:Label_SetColorByID("txtPlayerTitle", nFontColor)
  else
    panel:SetActive("txtPlayerTitle", false)
  end
  local nVipLevel = tbApplyInfo[9] or 0
  if 0 ~= nVipLevel then
  else
    panel:SetActive("txtName/imgVIP", false)
  end
  panel:Label_SetText("txtName", tbApplyInfo[5])
  if 1 == tbApplyInfo[3] then
    panel:SetActive("txtFindTeacher", false)
    panel:SetActive("txtFindStudent", true)
  else
    panel:SetActive("txtFindTeacher", true)
    panel:SetActive("txtFindStudent", false)
  end
  Ui:Button_BindEvent(panel, "btnAgree", function()
    MasterPrentice:AgreeApprentice(tbApplyInfo[1])
  end)
  Ui:Button_BindEvent(panel, "PortraitObj", function()
    local tbData = {
      dwRoleId = tbApplyInfo[2],
      szName = tbApplyInfo[5],
      nPortrait = tbApplyInfo[6],
      nFaction = tbApplyInfo[10],
      nLevel = tbApplyInfo[4],
      szKinOrTongName = "" == tbApplyInfo[7] and i18n.EasyGet("\230\154\130\230\151\160") or tbApplyInfo[7]
    }
    Ui:OpenWindow("UIRightPopUpMenu", "roleselectFromUI", tbData, self, 317, -170)
  end)
end

tbUi.tbOnClick = {}

function tbUi.tbOnClick:btnClose()
  self:Close()
end

function tbUi.tbOnClick:btnClearAllApply()
  if not self.tbAllRequet or 0 == #self.tbAllRequet then
    me.CenterMsg(i18n.EasyGet("\230\154\130\230\151\160\229\143\175\230\184\133\231\169\186\231\154\132\228\191\161\230\129\175"))
    return
  end
  
  local function fnYes()
    MasterPrentice:DisagreeAllApprentice()
    self:Refresh()
    self:Close()
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\130\168\231\161\174\229\174\154\232\166\129\230\184\133\231\169\186\231\148\179\232\175\183\229\136\151\232\161\168\229\144\151\239\188\159"), {
    {fnYes},
    {}
  }, {
    i18n.EasyGet("\231\161\174\229\174\154"),
    i18n.EasyGet("\229\143\150\230\182\136")
  })
end
