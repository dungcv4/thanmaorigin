GoodwillValue.GROUP_KEY = GetGroupKey("GoodwillValue")
GoodwillValue.nSaveKey = 1

function GoodwillValue:GetGoodwillValue(pPlayer)
  return self.Cfg.nMaxValue - pPlayer.GetUserValue(self.GROUP_KEY, self.nSaveKey)
end

function GoodwillValue:CheckValueCanUseMedicine(pPlayer)
  return self:GetGoodwillValue(pPlayer) > self.Cfg.nLessThanOrEqualValueCanNotUseMedicine
end
