local tbWnd = Ui:CreateWindow("UICenterMsg")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_CENTER_MSG,
      self.UpdateCenterMsg
    },
    {
      EventNotify.emNOTIFY_APPLICATION_ACTIVE,
      self.ChkNotice
    }
  }
end

tbWnd.tbControls = {
  MsgTemplate = "Content/MsgTemplate"
}

function tbWnd:OnCreate()
  self.pPanel:SetActive(self.MsgTemplate, false)
  self.lastShowMsgTime = 0
  self.destroyMsgInterval = 3
  self.showMsgInterval = 0.7
  self.tbAllMsg = {}
  self.tbDestroyList = {}
  self.nMaxLatestNum = 5
  self.nMaxShowNum = 3
end

function tbWnd:OnFixedUpdate()
  local toDestory
  while next(self.tbDestroyList) do
    toDestory = self.tbDestroyList[1]
    if Ui.Time.time - toDestory[2] >= self.destroyMsgInterval or #self.tbDestroyList > self.nMaxShowNum then
      self.pPanel:DeleteObject(toDestory[1])
      table.remove(self.tbDestroyList, 1)
    else
      break
    end
  end
  if Ui.Time.time - self.lastShowMsgTime >= self.showMsgInterval then
    if #self.tbAllMsg > self.nMaxLatestNum then
      for i = #self.tbAllMsg, self.nMaxLatestNum + 1, -1 do
        table.remove(self.tbAllMsg, 1)
      end
    end
    if #self.tbAllMsg > 0 then
      local szMsg = self.tbAllMsg[1]
      table.remove(self.tbAllMsg, 1)
      local szNewKey = self.pPanel:CloneObject(self.MsgTemplate, "")
      self.pPanel:Label_SetChildText(szNewKey, "txtMsg", szMsg or "")
      Sound:PlaySound("GetCenterMsg", _)
      self.lastShowMsgTime = Ui.Time.time
      table.insert(self.tbDestroyList, {
        szNewKey,
        self.lastShowMsgTime
      })
    end
  end
end

function tbWnd:UpdateCenterMsg()
  local tbMsg = Ui:FetchCenterMsg()
  for _, szMsg in ipairs(tbMsg) do
    table.insert(self.tbAllMsg, szMsg)
  end
end

function tbWnd:ChkNotice()
end
