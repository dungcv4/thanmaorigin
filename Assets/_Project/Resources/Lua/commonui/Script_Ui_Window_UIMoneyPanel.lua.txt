local tbWnd = Ui:CreateClass("UIMoneyPanel", "UIPanel")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_LOGIN_ZONE,
      self.OnLoginZone
    },
    {
      EventNotify.emNOTIFY_LOGOUT_ZONE,
      self.OnLogoutZone
    }
  }
end

function Ui:NewExistMoneyPanel(tbOwner, szGroupKey)
  return Ui:CreatePanel("UIMoneyPanel", szGroupKey, tbOwner)
end

local tbControls = {
  txtCoinValue = "panelList/panelCoin/txtCoinValue",
  btnPlusCoin = "panelList/panelCoin/btnPlusCoin",
  txtYuanBaoValue = "panelList/panelYuanBao/txtYuanBaoValue",
  btnPlusYuanBao = "panelList/panelYuanBao/btnPlusYuanBao",
  txtGoldValue = "panelList/panelGold/txtGoldValue",
  btnPlusGold = "panelList/panelGold/btnPlusGold",
  txtDynamic = "panelList/panelDynamic/txtDynamic",
  imgDynamic = "panelList/panelDynamic/imgDynamic",
  btnPlusDynamic = "panelList/panelDynamic/btnPlusDynamic",
  panelDynamic = "panelList/panelDynamic",
  panelCoin = "panelList/panelCoin",
  panelYuanBao = "panelList/panelYuanBao",
  panelGold = "panelList/panelGold"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnPlusCoin = function(self)
    Ui:OpenWindow("UIWelfare", "PANEL_TYPE_MONEY_TREE")
  end,
  btnPlusYuanBao = function(self)
    GainCurrency:DirectJump("YuanBao")
  end,
  btnPlusGold = function(self)
    GainCurrency:DirectJump("Gold")
  end,
  btnPlusDynamic = function(self)
    if self.tbCallBack then
      Lib:CallBack(self.tbCallBack)
    end
  end
}

function tbWnd:OnCreate()
  if not IsPayOpen() then
    self.pPanel:SetActive(tbControls.btnPlusYuanBao, false)
    self.pPanel:SetActive(tbControls.btnPlusGold, false)
  end
  self.tbFunc = {
    self.SetCoinShow,
    self.SetYuanBaoShow,
    self.SetGoldShow,
    self.SetDynamicShow
  }
  self.pPanel:SetActive(tbControls.btnPlusDynamic, false)
end

function tbWnd:SetMoneyValue(bNeedNotDealZone)
  self.pPanel:Label_SetText(tbControls.txtCoinValue, me.GetMoney("Coin"))
  self.pPanel:Label_SetText(tbControls.txtYuanBaoValue, me.GetMoney("YuanBao"))
  self.pPanel:Label_SetText(tbControls.txtGoldValue, me.GetMoney("Gold"))
  if not bNeedNotDealZone then
    self:SetIconsShow()
  end
end

function tbWnd:SetCoinShow(bIsShow)
  self.pPanel:SetActive(tbControls.panelCoin, bIsShow)
end

function tbWnd:SetYuanBaoShow(bIsShow)
  self.pPanel:SetActive(tbControls.panelYuanBao, bIsShow)
end

function tbWnd:SetGoldShow(bIsShow)
  self.pPanel:SetActive(tbControls.panelGold, bIsShow)
end

function tbWnd:SetDynamicShow(bIsShow)
  self.pPanel:SetActive(tbControls.panelDynamic, bIsShow)
end

function tbWnd:SetCurrencyView(isDefaultType)
  for i, func in ipairs(self.tbFunc) do
    if i <= 3 then
      func(self, isDefaultType)
    else
      func(self, not isDefaultType)
    end
  end
end

function tbWnd:SetDefaultCurrencyView()
  self:SetCurrencyView(true)
  self:SetIconsShow()
end

function tbWnd:SetDynamicCurrencyType(szType, bNeedNotDealZone)
  self.szDynamicType = szType
  self:SetCurrencyView(false)
  self:SetDynamicCurrencyValue()
  if not bNeedNotDealZone then
    self:SetIconsShow()
  end
end

function tbWnd:SetDynamicCurrencyValue()
  if not self.szDynamicType then
    return
  end
  local nCurrencyValue = me.GetMoney(self.szDynamicType)
  self.pPanel:Label_SetText(tbControls.txtDynamic, nCurrencyValue)
  local szIcon = Currency:GetIcon(self.szDynamicType)
  self.pPanel:Sprite_SetSprite(tbControls.imgDynamic, szIcon)
end

function tbWnd:SetDynamicCurrencyInfo(szType, nValue, szIconPath)
  self.szDynamicType = szType
  self.pPanel:Label_SetText(tbControls.txtDynamic, nValue)
  self.pPanel:Sprite_SetSprite(tbControls.imgDynamic, szIconPath)
end

function tbWnd:HideDynamicBtnPlus()
  self.tbCallBack = nil
  self.pPanel:SetActive(tbControls.btnPlusDynamic, false)
end

function tbWnd:SetDynamicBtnPlusCallBack(tbCallBack, ...)
  self.tbCallBack = {
    tbCallBack,
    ...
  }
  self.pPanel:SetActive(tbControls.btnPlusDynamic, true)
  self:SetIconsShow()
end

function tbWnd:SetIconsShow()
  local bShow = not Ui.bInZone
  local tbBtns = {
    tbControls.btnPlusCoin,
    tbControls.btnPlusYuanBao,
    tbControls.btnPlusGold,
    tbControls.btnPlusDynamic
  }
  for _, szPath in ipairs(tbBtns) do
    self.pPanel:SetActive(szPath, bShow)
  end
end

function tbWnd:OnLoginZone()
  self:SetIconsShow()
end

function tbWnd:OnLogoutZone()
  self:SetIconsShow()
end

function tbWnd:OnOpen()
  self:SetIconsShow()
end
