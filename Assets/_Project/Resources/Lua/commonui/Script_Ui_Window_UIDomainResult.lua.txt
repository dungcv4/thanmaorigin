local tbWnd = Ui:CreateWindow("UIDomainResult")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  goResultTemplate = "node/imgBG/Content/ScrollView/Viewport/Content/Template",
  goMyRank = "node/imgBG/Content/MyRank"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_TONG_DOMAIN_RESULT,
      self.OnSyncDomainResult
    }
  }
end

function tbWnd:OnCreate()
  self.tbResultScroll = Ui:NewScrollView(self, self.goResultTemplate, nil, nil, true)
  self.tbResultList = {}
  self.nMyRank = 0
  self.nQueryDay = 0
end

function tbWnd:OnOpen()
  local nNowDay = Lib:GetLocalDay()
  if self.nQueryDay ~= nNowDay then
    self.tbResultList = {}
    self.nMyRank = 0
    self.nQueryDay = nNowDay
  end
  if 0 == #self.tbResultList then
    DomainBattle:QueryResultReq()
  end
  self:UpdateShow()
end

function tbWnd:UpdateShow()
  self.tbResultScroll:Reset()
  for nIndex, tbInfo in ipairs(self.tbResultList) do
    self.tbResultScroll:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtRank", tostring(nIndex))
      pPanel:Sprite_SetSprite("imgFaction", Faction:GetIcon(tbInfo.nFaction))
      pPanel:Label_SetText("txtName", tbInfo.szName)
      pPanel:Label_SetText("txtLv", tostring(tbInfo.nCombo))
      pPanel:Label_SetText("txtScore", tostring(tbInfo.nScore))
    end)
  end
  local tbInfo = self.tbResultList[self.nMyRank]
  if tbInfo then
    self.pPanel:Label_SetText(self.goMyRank .. "/txtRank", tostring(self.nMyRank))
    self.pPanel:Sprite_SetSprite(self.goMyRank .. "/imgFaction", Faction:GetIcon(tbInfo.nFaction))
    self.pPanel:Label_SetText(self.goMyRank .. "/txtName", tbInfo.szName)
    self.pPanel:Label_SetText(self.goMyRank .. "/txtLv", tostring(tbInfo.nCombo))
    self.pPanel:Label_SetText(self.goMyRank .. "/txtScore", tostring(tbInfo.nScore))
    self.pPanel:SetActive(self.goMyRank, true)
  else
    self.pPanel:SetActive(self.goMyRank, false)
  end
end

function tbWnd:OnSyncDomainResult(tbResultList, nMyRank)
  self.tbResultList = tbResultList
  self.nMyRank = nMyRank
  if Ui:WindowVisible(self.ClassName) then
    self:UpdateShow()
  end
end
