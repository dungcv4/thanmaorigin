local tbWnd = Ui:CreateWindow("UIGiftMoney")
local nNumPerPage = 4
local tbControls = {
  txtPage = "node/imgBG/Content/panelFriend/txtPage",
  btnPrePage = "node/imgBG/Content/panelFriend/btnPrePage",
  btnNextPage = "node/imgBG/Content/panelFriend/btnNextPage",
  btnFirstPage = "node/imgBG/Content/panelFriend/btnFirstPage",
  btnLastPage = "node/imgBG/Content/panelFriend/btnLastPage",
  Template = "node/imgBG/Content/panelFriend/Scroll View/Viewport/Content/Template",
  btnClose = "node/imgBG/btnClose",
  txtNowNum = "node/imgBG/Content/panelFriend/txtNowNum"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnPrePage = function(self)
    self:EnterPrePage()
  end,
  btnNextPage = function(self)
    self:EnterNextPage()
  end,
  btnFirstPage = function(self)
    self:EnterFirstPage()
  end,
  btnLastPage = function(self)
    self:EnterLastPage()
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbRankScrollView = Ui:NewScrollView(tbWnd, tbControls.Template)
end

function tbWnd:OnOpen()
  self.nCurPage = 1
  self.bNeedUpdate = true
  Wedding:TryReqCashMoneyData()
end

function tbWnd:OnUpdate()
  if self.bNeedUpdate then
    self:DoUpdateRank()
    self.bNeedUpdate = false
  end
end

function tbWnd:DoUpdateRank()
  self.pPanel:Label_SetText(tbControls.txtPage, tbWnd.nCurPage .. "/" .. self:GetMaxPageNum())
  self.pPanel:Label_SetText(tbControls.txtNowNum, i18n.EasyGet("\229\189\147\229\137\141\229\133\177\230\148\182\231\164\188\233\135\145\239\188\154{0}", self:GetCurCashMoneySum()))
  self.tbRankScrollView:Clear()
  local tbCurPageData = self:GetCurPageData()
  for index, tbInfo in ipairs(tbCurPageData) do
    self.tbRankScrollView:SetElement(index, function(tbElement)
      self:SetElement(tbElement, tbInfo)
    end)
  end
end

function tbWnd:UpdateRank()
  self.bNeedUpdate = true
end

function tbWnd:EnterPrePage()
  if self.nCurPage > 1 then
    self.nCurPage = self.nCurPage - 1
    self:UpdateRank()
  end
end

function tbWnd:EnterNextPage()
  local nMaxPage = self:GetMaxPageNum()
  if nMaxPage > self.nCurPage then
    self.nCurPage = self.nCurPage + 1
    self:UpdateRank()
  end
end

function tbWnd:EnterFirstPage()
  self.nCurPage = 1
  self:UpdateRank()
end

function tbWnd:EnterLastPage()
  self.nCurPage = self:GetMaxPageNum()
  self:UpdateRank()
end

function tbWnd:SetElement(tbElement, tbInfo)
  if not tbInfo then
    return
  end
  local pPanel = tbElement.pPanel
  if tbInfo.nRank <= 3 then
    pPanel:SetActive("txtOrder", false)
    pPanel:SetActive("UIRankNo", true)
    pPanel:SetActive("UIRankNo/imgNO1", 1 == tbInfo.nRank)
    pPanel:SetActive("UIRankNo/imgNO2", 2 == tbInfo.nRank)
    pPanel:SetActive("UIRankNo/imgNO3", 3 == tbInfo.nRank)
  else
    pPanel:SetActive("UIRankNo", false)
    pPanel:SetActive("txtOrder", true)
    pPanel:Label_SetText("txtOrder", tbInfo.nRank)
  end
  pPanel:Label_SetText("txtName", tbInfo.szName)
  pPanel:Label_SetText("txtNum", tbInfo.nCashMoney)
end

function tbWnd:GetMaxPageNum()
  if not Wedding.tbCashMoneyRank or Lib:IsEmptyTB(Wedding.tbCashMoneyRank.Data) then
    return 1
  end
  return math.ceil(#Wedding.tbCashMoneyRank.Data / nNumPerPage)
end

function tbWnd:GetCurCashMoneySum()
  if not Wedding.tbCashMoneyRank then
    return 0
  end
  return Wedding.tbCashMoneyRank.nCashSum
end

function tbWnd:GetCurPageData()
  if not Wedding.tbCashMoneyRank then
    return {}
  end
  local nShowRankStart = (self.nCurPage - 1) * nNumPerPage + 1
  local nShowRankEnd = nShowRankStart + nNumPerPage - 1
  if nShowRankEnd > #Wedding.tbCashMoneyRank.Data then
    nShowRankEnd = #Wedding.tbCashMoneyRank.Data
  end
  local tbCurPageData = {}
  for i = 1, nShowRankEnd - nShowRankStart + 1 do
    local tbInfo = Wedding.tbCashMoneyRank.Data[nShowRankStart + i - 1]
    tbCurPageData[i] = {
      nRank = nShowRankStart + i - 1,
      szName = tbInfo.szName,
      nCashMoney = tbInfo.nCashMoney
    }
  end
  return tbCurPageData
end
