function tbWnd:OnCreate()
  self.tbBaseAttribScrollView = Ui:NewScrollView(self, tbControls.txtBaseAttribTemplate, function(self, nIndex, tbElement)
    self.pPanel:SetActive(self.btnCloseAttributeInfo, true)
    self.pPanel:SetActive(tbControls.btnAttributeInfo, true)
    local szText = tbBaseAttribute[nIndex].fnDesc(self.pAsyncRole, self.tbAsyncAttr)
    self.pPanel:Label_SetText(tbControls.btnAttributeInfo .. "/imgBg/txtInfo", szText)
    local tbPosition = tbElement.pPanel:GetSelfPosition()
    self.pPanel:Object_SetPosition(tbControls.btnAttributeInfo, Ui.Vector3(tbPosition.x + 4, tbPosition.y, tbPosition.z))
    self.tbBaseAttribScrollView:ClearSelected()
  end)
  self.tbPotentialPoint = {}
  self.tbBasePotentialPoint = {}
  self.tbPotencyGroup = {
    [PotentialPoint.PotencyType.Strength] = Ui:NewExistAddAndSubGroup(self, tbControls.StrengthUIAddAndSubGroup),
    [PotentialPoint.PotencyType.Dexterity] = Ui:NewExistAddAndSubGroup(self, tbControls.DexterityUIAddAndSubGroup),
    [PotentialPoint.PotencyType.Vitality] = Ui:NewExistAddAndSubGroup(self, tbControls.VitalityUIAddAndSubGroup),
    [PotentialPoint.PotencyType.Energy] = Ui:NewExistAddAndSubGroup(self, tbControls.EnergyUIAddAndSubGroup)
  }
  for nIndex = 1, #self.tbPotencyGroup do
    self.tbPotencyGroup[nIndex]:SetNumPadPos(-140, 0)
    self.tbPotencyGroup[nIndex]:SetLongPressLoop(true, true, true)
    self.tbPotencyGroup[nIndex]:BindEvent(function(nValue)
      local nResult, szResult = self:AddPotencyChange(nIndex, nValue)
      return nResult, szResult
    end)
  end
  self.pPanel:Label_SetText(tbControls.txtRoleName, me.szName)
  self.pPanel:Label_SetText(tbControls.txtRoleID, me.dwID)
end

function tbWnd:UpdateAttackValue()

    self.pPanel:Label_SetText(tbControls.txtAttackValue, string.format("%d-%d", nMin, nMax))
  else
    self.pPanel:Label_SetText(tbControls.txtAttackValue, nMin)

    self.tbPotencyGroup[i]:SetNumRange(1, self.tbBasePotentialPoint[i] + self.nLeftPoint)
    self.tbPotencyGroup[i]:SetInputValue(self.tbPotentialPoint[i])
    self.tbPotencyGroup[i]:SetNumPadPos(-580, -100)

function tbWnd:UpdateTitle()
  local pNpc = me.GetNpc()
  local nTitleId = pNpc.nTitleID
  if not nTitleId or nTitleId <= 0 then
    self.pPanel:Label_SetText(tbControls.txtTitleName, i18n.EasyGet("\230\151\160"))
    return
  end
  local tbInfo = PlayerTitle:GetTitle(nTitleId)
  if tbInfo and tbInfo.szTitle and tbInfo.szTitle ~= "" then
    self.pPanel:Label_SetText(tbControls.txtTitleName, i18n.Parse(tbInfo.szTitle))
    return
  end
  local tbTitleTemp = PlayerTitle:GetTitleTemplate(nTitleId)
  if not tbTitleTemp or tbTitleTemp.bShow <= 0 then
    self.pPanel:Label_SetText(tbControls.txtTitleName, i18n.EasyGet("\230\151\160"))
    return
  end
  self.pPanel:Label_SetText(tbControls.txtTitleName, i18n.Parse(tbTitleTemp.Name))
end

function tbWnd:UpdatePotentialPoint()
  self.nLeftPoint = me.nLeftPotentialPoint
  self.tbPotentialPoint[PotentialPoint.PotencyType.Strength] = me.nBaseStrength
  self.tbPotentialPoint[PotentialPoint.PotencyType.Vitality] = me.nBaseVitality
  self.tbPotentialPoint[PotentialPoint.PotencyType.Energy] = me.nBaseEnergy
  self.tbPotentialPoint[PotentialPoint.PotencyType.Dexterity] = me.nBaseDexterity
  self.tbBasePotentialPoint[PotentialPoint.PotencyType.Strength] = me.nBaseStrength
  self.tbBasePotentialPoint[PotentialPoint.PotencyType.Vitality] = me.nBaseVitality
  self.tbBasePotentialPoint[PotentialPoint.PotencyType.Energy] = me.nBaseEnergy
  self.tbBasePotentialPoint[PotentialPoint.PotencyType.Dexterity] = me.nBaseDexterity
  self.nBaseTotalPoint = me.nBaseStrength + me.nBaseVitality + me.nBaseEnergy + me.nBaseDexterity
end

function tbWnd:AddPotencyChange(nIndex, nValue)
  if 0 == self.nLeftPoint and nValue > self.tbPotentialPoint[nIndex] then
    me.CenterMsg(i18n.EasyGet("\230\178\161\230\156\137\229\143\175\229\136\134\233\133\141\231\154\132\230\189\156\232\131\189\231\130\185"))
    nValue = self.tbPotentialPoint[nIndex]
  elseif nValue < self.tbBasePotentialPoint[nIndex] then
    me.CenterMsg(i18n.EasyGet("\232\190\147\229\133\165\228\184\141\232\131\189\228\189\142\228\186\142\229\159\186\231\161\128\230\189\156\232\131\189\231\130\185"))
    nValue = self.tbPotentialPoint[nIndex]
  else
    local nTmpTotalPoint = nValue
    for i = 1, #self.tbPotentialPoint do
      if i ~= nIndex then
        nTmpTotalPoint = nTmpTotalPoint + self.tbPotentialPoint[i]
      end
    end
    if nTmpTotalPoint - self.nBaseTotalPoint > me.nLeftPotentialPoint then
      me.CenterMsg(i18n.EasyGet("\232\190\147\229\133\165\228\184\141\232\131\189\232\182\133\232\191\135\230\189\156\232\131\189\231\130\185\228\184\138\233\153\144"))
      nValue = self.tbPotentialPoint[nIndex] + self.nLeftPoint
    end
  end
  local bOverFlow = false
  local nAllTotal = me.nBaseStrength + me.nBaseVitality + me.nBaseEnergy + me.nBaseDexterity + me.nLeftPotentialPoint
  local nMaxPoint = math.floor(nAllTotal * PotentialPoint.SINGLE_ATTR_MAX_PERCENT / 100)
  if nValue > nMaxPoint then
    nValue = nMaxPoint
    me.CenterMsg(i18n.EasyGet("\229\141\149\228\184\170\229\177\158\230\128\167\232\182\133\232\191\135\228\186\13460%\231\154\132\233\153\144\229\136\182"))
  end
  local nCurTotalDiff = nValue - self.tbPotentialPoint[nIndex]
  self.nLeftPoint = self.nLeftPoint - nCurTotalDiff
  self.pPanel:Label_SetText(tbControls.txtPotentialValue, self.nLeftPoint)
  self.tbPotentialPoint[nIndex] = nValue
  local nCurDiff = self.tbPotentialPoint[nIndex] - self.tbBasePotentialPoint[nIndex]
  local szValue
  if 0 == nCurDiff then
    szValue = nValue
  else
    szValue = string.format("%d(%d<color=#28ff28>+%d</color>)", nValue, self.tbBasePotentialPoint[nIndex], nCurDiff)
  end
  return nValue, szValue
end

function tbWnd:OnAddPotencyConfirm()
  if me.IsMaskPlayerPotential() then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\228\184\141\229\143\175\229\138\160\231\130\185"))
    return
  end
  local nStrengthDiff = self.tbPotentialPoint[PotentialPoint.PotencyType.Strength] - self.tbBasePotentialPoint[PotentialPoint.PotencyType.Strength]
  local nVitalityDiff = self.tbPotentialPoint[PotentialPoint.PotencyType.Vitality] - self.tbBasePotentialPoint[PotentialPoint.PotencyType.Vitality]
  local nEnergyDiff = self.tbPotentialPoint[PotentialPoint.PotencyType.Energy] - self.tbBasePotentialPoint[PotentialPoint.PotencyType.Energy]
  local nDexterityDiff = self.tbPotentialPoint[PotentialPoint.PotencyType.Dexterity] - self.tbBasePotentialPoint[PotentialPoint.PotencyType.Dexterity]
  if 0 == nStrengthDiff and 0 == nVitalityDiff and 0 == nEnergyDiff and 0 == nDexterityDiff then
    me.CenterMsg(i18n.EasyGet("\230\130\168\230\178\161\230\156\137\232\174\190\231\189\174\229\138\160\231\130\185"))
    return
  end
  RemoteServer.AllocPotentialPointByScheme(nStrengthDiff, nVitalityDiff, nEnergyDiff, nDexterityDiff)
end

function tbWnd:RecommendPotentialScheme()
  local tbOriginalPotency = me.GetFactionPotencyByLevel(1)
  local tbPotency = me.GetFactionPotencyByLevel(2)
  local nTotalRatio = tbPotency.nStrength - tbOriginalPotency.nStrength + (tbPotency.nVitality - tbOriginalPotency.nVitality) + (tbPotency.nEnergy - tbOriginalPotency.nEnergy) + (tbPotency.nDexterity - tbOriginalPotency.nDexterity)
  local nTotalPoint = me.nLeftPotentialPoint - me.nLeftPotentialPoint % nTotalRatio
  if 0 == me.nLeftPotentialPoint then
    me.CenterMsg(i18n.EasyGet("\229\143\175\229\136\134\233\133\141\230\189\156\232\131\189\228\184\141\232\182\179"))
    return
  elseif me.nLeftPotentialPoint < 10 then
    me.CenterMsg(i18n.EasyGet("\229\143\175\229\136\134\233\133\141\230\189\156\232\131\189\228\184\141\232\182\17910\231\130\185\239\188\140\232\175\183\232\135\170\232\161\140\230\137\139\229\138\168\229\136\134\233\133\141"))
    return
  end
  local fStrengthRatio = (tbPotency.nStrength - tbOriginalPotency.nStrength) / nTotalRatio
  local fVitalityRatio = (tbPotency.nVitality - tbOriginalPotency.nVitality) / nTotalRatio
  local fEnergyRatio = (tbPotency.nEnergy - tbOriginalPotency.nEnergy) / nTotalRatio
  local fDexterityRatio = (tbPotency.nDexterity - tbOriginalPotency.nDexterity) / nTotalRatio
  local tbDiffPoint = {
    [PotentialPoint.PotencyType.Strength] = math.floor(nTotalPoint * fStrengthRatio),
    [PotentialPoint.PotencyType.Vitality] = math.floor(nTotalPoint * fVitalityRatio),
    [PotentialPoint.PotencyType.Energy] = math.floor(nTotalPoint * fEnergyRatio),
    [PotentialPoint.PotencyType.Dexterity] = math.floor(nTotalPoint * fDexterityRatio)
  }
  for i = 1, #self.tbPotencyGroup do
    local nPoint = self.tbBasePotentialPoint[i] + tbDiffPoint[i]
    self.tbPotencyGroup[i]:InputValueChanged(nPoint)
  end
end
