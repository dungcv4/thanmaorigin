Require("CommonScript/CrossTimeRoom/CrossTimeRoom_Common.lua")
CrossTimeRoom = CrossTimeRoom or {}
CrossTimeRoom.Fuben = CrossTimeRoom.Fuben or {}
local Fuben = CrossTimeRoom.Fuben

function Fuben:RegisterFubenUIChecker()
  local tbMap = {}
  for index, tbInfo in ipairs(CrossTimeRoom.tbStage) do
    table.insert(tbMap, tbInfo.nTemplateID)
  end
  FubenUIController:RegisterChecker(tbMap, self)
end

Fuben:RegisterFubenUIChecker()

function Fuben:OnFubenMapLoaded()
  FubenUIController:OpenFuBenUI()
end

function Fuben:OnLeave()
  if self.IsCanExit then
    FubenUIController:CloseFubenUI()
    self.IsCanExit = nil
  end
end

function Fuben:RequieExit()
  FubenUIController:RequireQuit()
end

function Fuben:SetCanExit(bCanExit)
  self.IsCanExit = bCanExit
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function Fuben:IsNeedExitFubenButton()
  return self.IsCanExit or false
end

function CrossTimeRoom:ShowTeamUI()
  if not TeamMgr:HasTeam() then
    Ui:OpenWindow("UITeam", TeamMgr.ACTIVITY_ID.CrossTimeRoom)
    return
  elseif not TeamMgr:IsCaptain(me.dwID) then
    me.CenterMsg(TeamMgr.OnlyCaptainCanJoinHintMsg())
    return
  end
  local tbMembers = TeamMgr:GetMembers()
  local nActivityId = TeamMgr:GetActivityId("CrossTimeRoom", "CrossTimeRoom")
  if TeamMgr:HasTeam() then
    TeamMgr:SetTeamActivity(nActivityId)
    Ui:OpenWindow("UITeam")
  else
    if not TeamMgr:CanTeam(me.nMapTemplateId) then
      me.CenterMsg(i18n.Get(TeamMgr.Terms.MapLimitWhenJoin))
      return
    end
    TeamMgr:CreateOnePersonTeam(nActivityId)
    Ui:OpenWindow("UITeam")
  end
end
