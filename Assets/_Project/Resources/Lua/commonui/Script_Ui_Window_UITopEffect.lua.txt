local tbWnd = Ui:CreateWindow("UITopEffect")
tbWnd.tbControls = {
  BossWarning = "BossWarning"
}

function tbWnd:OnCreate()
  self:HideBossWarning()
end

function tbWnd:OnOpen()
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:ShowBossWarning(nCloseSecond, nR, nG, nB)
  self.pPanel:SetActive(self.BossWarning, true)
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  if nCloseSecond and nCloseSecond > 0 then
    self.nTimer = Timer:Register(Env.GAME_FPS * nCloseSecond, function()
      self:HideBossWarning()
      Timer:Close(self.nTimer)
      self.nTimer = nil
    end)
  end
  if nR and nG and nB then
    self.pPanel:Sprite_SetColor(self.BossWarning, nR, nG, nB, 255)
  end
end

function tbWnd:HideBossWarning()
  self.pPanel:SetActive(self.BossWarning, false)
end
