function Ui._tbClassMeta.__index(tb, key)
  local szParent = rawget(tb, "__Parent")
  if nil == szParent then
    return Ui._BaseClass[key]
  end
  if nil == Ui.tbClass[szParent] then
    return
  end
  return Ui.tbClass[szParent][key]
end

function Ui:OnWindowOpened(szUiName, ...)
  local tbWnd = self:GetWindow(szUiName)
  if not tbWnd then
    return
  end
  local nRetCode = 1
  if tbWnd.OnOpen then
    local bRet, nOpenRetCode = Lib:CallBack({
      tbWnd.OnOpen,
      tbWnd,
      ...
    })
    nOpenRetCode = nOpenRetCode or 1
    nRetCode = bRet and nOpenRetCode or 0
    if not bRet then
      Log(debug.traceback())
    end
  end
  if 1 == nRetCode then
    self.tbWndState[szUiName] = true
    if tbWnd.OnOpenEnd then
      local bRet, nOpenRetCode = Lib:CallBack({
        tbWnd.OnOpenEnd,
        tbWnd,
        ...
      })
      if not bRet then
        Log(szUiName, debug.traceback())
      end
    end
    EventNotify.OnNotify(EventNotify.emNOTIFY_WND_OPENED, szUiName)
  else
    self:CloseWindow(szUiName, unpack(arg))
  end
  return tbWnd
end

function Ui:OpenWindow(szUiName, ...)
  if self.bIsInDestroyAllWindowState == true then
    return
  end
  if self:WindowVisible(szUiName) then
    self:CloseWindow(szUiName)
  end
  Sound:PlaySound("OpenWindow", szUiName)
  local arg = {
    ...
  }
  Client.UIModule.PreloadUI(szUiName, function(self)
    self:OnWindowOpened(szUiName, unpack(arg))
  end, self)
end

function Ui:OpenWindowAsync(szUiName, ...)
  if self.bIsInDestroyAllWindowState == true then
    return
  end
  if self:WindowVisible(szUiName) then
    self:CloseWindow(szUiName)
  end
  if self.tbOpeningWndList[szUiName] then
    return
  end
  self.tbOpeningWndList[szUiName] = true
  Sound:PlaySound("OpenWindow", szUiName)
  local arg = {
    ...
  }
  Client.UIModule.PreloadUIAsync(szUiName, function(self)
    self.tbOpeningWndList[szUiName] = nil
    self:OnWindowOpened(szUiName, unpack(arg))
    if self.tbClosingWndList[szUiName] then
      self:CloseWindow(szUiName, unpack(self.tbClosingWndList[szUiName]))
      self.tbClosingWndList[szUiName] = nil
    end
  end, self)
end

function Ui:CloseWindow(szUiName, ...)
  local bVisiable = self:WindowVisible(szUiName)
  if self.tbOpeningWndList[szUiName] and not bVisiable then
    self.tbClosingWndList[szUiName] = {
      ...
    }
  end
  if bVisiable then
    Sound:PlaySound("CloseWindow", szUiName)
    local arg = {
      ...
    }
    local tbWnd = self:GetWindow(szUiName)
    if tbWnd.OnClose then
      local bRet, nCloseRetCode = Lib:CallBack({
        tbWnd.OnClose,
        tbWnd,
        unpack(arg)
      })
      if not bRet then
        Log(debug.traceback())
      end
    end
    self.tbWndState[szUiName] = false
    Client.UIModule.Close(szUiName)
    EventNotify.OnNotify(EventNotify.emNOTIFY_WND_CLOSED, szUiName)
  end
end

end

function Ui:DestroyWindow(szUiName)
  self:CloseWindow(szUiName)
  if self.tbUi[szUiName] then
    self.tbUi[szUiName]:OnPreDestroy()
  end
  Client.UIModule.DestroyUI(szUiName)
  self.tbUi[szUiName] = nil
end

function Ui:WindowVisible(szUiName)
  local tbWnd = self:GetWindow(szUiName)
  -- DEVIATION 2026-04-27 (canonicalization Day 9.12): defensive nil-check on
  -- tbWnd.view before .gameObject access. Gốc gốc raw `tbWnd.view.gameObject.activeSelf`
  -- crashes if Unity destroyed UIView while tbUi cache still holds the entry
  -- (e.g. UILoadingTips auto-closed via UIModule.DestroyUI without going through
  -- Ui:DestroyWindow which would clear the cache via OnPreDestroy).
  -- Safe-check prevents "object of type 'UIView' has been destroyed" exception
  -- and self-heals cache. Equivalent observable behavior in non-stale cases.
  if not tbWnd then return false end
  -- Use tostring null-check (Unity-destroyed objects tostring="null").
  if tbWnd.view == nil or tostring(tbWnd.view) == 'null' then
    self.tbUi[szUiName] = nil  -- self-heal stale cache
    return false
  end
  return tbWnd.view.gameObject.activeSelf
end

function Ui:CheckIdle()
  if not me.bOfflineExpOnline then
    return
  end
  local nTouchSecond = Client.InputModule.GetDeltaTimeFromLastTouch()
  local nValue = math.min(math.floor(GetTime() - self.nIdleTime), math.floor(nTouchSecond))
  if nValue < Client.nMaxIdleTime then
    return
  end
  if Ui.bTimeKickOffline then
    return
  end
  if me.nFightMode ~= Npc.FIGHT_MODE.emFightMode_None then
    return
  end
  Ui:KickOffLine()
end

function Ui:_UpdateMainUi()
  if not self.bInDelayUpdateMain then
    return
  end
  local bFigthUi = self:IsFightUiState()
  local tbShowUi = bFigthUi and tbWndForFightState[1] or tbWndForFightState[0]
  local tbHideUi = bFigthUi and tbWndForFightState[0] or tbWndForFightState[1]
  local tbShowUiState = {}
  for szUi, _ in pairs(tbShowUi) do
    local bOpenWindow = false
    if not self.tbHideStateUi then
      bOpenWindow = true
    elseif self.tbHideStateUi[szUi] and not self.bHideStateWnd then
      bOpenWindow = true
    elseif not self.tbHideStateUi[szUi] and self.bHideStateWnd then
      bOpenWindow = true
    end
    if bOpenWindow then
      if not Ui:WindowVisible(szUi) and not self:IsWindowOpening(szUi) then
        self:OpenWindow(szUi)
      end
    else
      self:CloseWindow(szUi)
    end
    tbShowUiState[szUi] = bOpenWindow
  end
  for szUi, _ in pairs(tbHideUi) do
    if not tbShowUiState[szUi] then
      self:CloseWindow(szUi)
    end
  end
  self.tbShowUiState = tbShowUiState
  if bFigthUi then
    self:RefreshHomeScreenState(false, self.HomeScreenState.Battle)
  elseif self.nHomeScreenState ~= self.HomeScreenState.VerySimple then
    self:RefreshHomeScreenState(false, self.HomeScreenState.TopButton)
  elseif bForbiddenVerySimple then
    self:RefreshHomeScreenState(false, self.HomeScreenState.TopButton)
  else
    self:RefreshHomeScreenState(false, self.HomeScreenState.VerySimple)
  end
  self:UpdateHomeScreenState()
end

function Ui:UpdateMainUi(forceInstant)
  if forceInstant then
    self:_UpdateMainUi()
  end
  if not forceInstant then
    if self.bInDelayUpdateMain then
      return
    end
    self.bInDelayUpdateMain = true
    Timer:Register(1, function()
      Ui:_UpdateMainUi()
      Ui.bInDelayUpdateMain = nil
    end)
  end
end

function Ui:RefreshHomeScreenState(bMoveNext, nExpectState)
  if self.tbShowUiState.HomeScreenBattle then
    self.nHomeScreenState = self.HomeScreenState.Battle
  else
    self.nHomeScreenState = self.HomeScreenState.TopButton
  end
  do return end
  if not self.tbShowUiState.HomeScreenBattle and not self.tbShowUiState.TopButton then
    self.nHomeScreenState = self.HomeScreenState.UiStateHideButNotVerySimple
    return
  end
  if not self.tbShowUiState.TopButton then
    self.nHomeScreenState = self.HomeScreenState.Battle
    return
  end
  if not self.tbShowUiState.HomeScreenBattle then
    if nExpectState == self.HomeScreenState.VerySimple or nExpectState == self.HomeScreenState.TopButton then
      self.nHomeScreenState = nExpectState
    else
      self.nHomeScreenState = bMoveNext and self.nHomeScreenState == self.HomeScreenState.TopButton and self.HomeScreenState.VerySimple or self.HomeScreenState.TopButton
    end
    if bForbiddenVerySimple and self.nHomeScreenState == self.HomeScreenState.VerySimple then
      self.nHomeScreenState = self.HomeScreenState.TopButton
    end
    return
  end
  local bBattleVisible = false
  local bBattleInHide = false
  if nExpectState == self.HomeScreenState.VerySimple or nExpectState == self.HomeScreenState.TopButton or nExpectState == self.HomeScreenState.Battle then
    self.nHomeScreenState = nExpectState
  else
    if not bBattleInHide and bBattleVisible then
      self.nHomeScreenState = self.HomeScreenState.Battle
    elseif bBattleVisible then
      if self.nHomeScreenState ~= self.HomeScreenState.VerySimple then
        self.nHomeScreenState = self.HomeScreenState.TopButton
      end
    else
      self.nHomeScreenState = self.HomeScreenState.TopButton
    end
    if bMoveNext then
      self.nHomeScreenState = (self.nHomeScreenState + 1) % 3
    end
    if bForbiddenVerySimple and self.nHomeScreenState == self.HomeScreenState.VerySimple then
      self.nHomeScreenState = self.HomeScreenState.TopButton
    end
  end
end

function Ui:DropAward(nPosX, nPosY, tbAward, bRandom)
  local tbDropInfo
  if nil == bRandom then
    bRandom = #tbAward > 0
  end
  local nDis = bRandom and Item.Obj.MAX_DROP_RANGE * Item.Obj.CELL_LENGTH
  for _, tbInfo in pairs(tbAward) do
    tbDropInfo = {}
    local awardType = Player.AwardType[tbInfo[1] or "nil"]
    if awardType == Player.award_type_money then
    elseif awardType == Player.award_type_item then
      tbDropInfo.tbItem = tbDropInfo.tbItem or {}
      table.insert(tbDropInfo.tbItem, {
        nItemId = tbInfo[2],
        nCount = tbInfo[3]

    local nY = bRandom and Lib:random(nPosY - nDis, nPosY + nDis) or nPosY
    me.DropItemInPos(nX, nY, 0, 0, tbDropInfo)
  end

  Log("StartProcessTask************", nActEventId, nNpcShowId, nNpcShowActId, nNpcShowActEventId, nMeSkillId, nShowNpcSkillId)
  if nCollectActId and nCollectActId > 0 then
    local pNpc = me.GetNpc()
    if pNpc then
      pNpc.DoCommonAct(nCollectActId, nActEventId or 0, 0, nInterval)
    end
  end
  local pShowNpc
  if nNpcShowId and nNpcShowId > 0 then
    return
  end
  local bRet
  if nMeSkillId and nMeSkillId > 0 then
    local nSkillId = nMeSkillId
    local nMpsX, nMpsY = -1, -1
    local bSelf = true
    local bSelectTarget = false
    local pNpc = me.GetNpc()
    if pNpc then
      local pMeNpc = me.GetNpc()
      local eDoing = me.GetDoing()
      if nil ~= pShowNpc and eDoing == Npc.Doing.stand then
        pMeNpc.SetDirToNpc(pShowNpc.nId or 0)
      end
      bRet = pNpc.AddSkill(nSkillId, 1, 0, 1)
      pNpc.SetForceSkill(1)
      if bSelf then
        nMpsY = pNpc.nId
      end
      me.UseSkill(nSkillId, nMpsX, nMpsY)
      pNpc.SetForceSkill(0)
    end
  end
  if pShowNpc and nShowNpcSkillId and nShowNpcSkillId > 0 then
    local nSkillId = nShowNpcSkillId
    local nMpsX, nMpsY = -1, -1
    local bSelf = true
    local bSelectTarget = false
    local pNpc = pShowNpc
    if pNpc then
      bRet = pNpc.AddSkill(nSkillId, 1, 0, 1)
      pNpc.SetForceSkill(1)
      if bSelf then
        nMpsY = pNpc.nId
      end
      pNpc.UseSkill(nSkillId, nMpsX, nMpsY)
      pNpc.SetForceSkill(0)
    end
  end

function Ui:NpcBubbleTalk(npc, szContent, nDuration, nMaxCount, nDealyTime)

  end

    tbNpcInfo = npc

  for i, nNpcId in pairs(tbNpcInfo) do

      ChatMgr:OnNpcBubbleTalk(pNpc.szName, szContent)
      nCount = nCount + 1

function Ui:OnConnectLost()
  Log("<color=#D00000>Ui:OnConnectLost</color>")
  for _, v in pairs(tbNeed2CloseWhenDisconnect) do
    local wnd = self:GetWindow(v)
    if wnd and self:WindowVisible(v) then
      wnd:Close()
    end
  end
  if not self.bForRetrunLogin and not self.bKickOffline and not self.bTimeKickOffline and not Ui.bForRetrunLogin_ByUi then
    local bSkip = false
    for _, v in ipairs(tbSkipReconnectWhenThisUiOpen) do
      if Ui:WindowVisible(v) == true then
        bSkip = true
        break
      end
    end
    if true == Ui:WindowVisible("UIMessageBoxBig") then
      local wnd = self:GetWindow("UIMessageBoxBig")
      if wnd and wnd.bSkipConnect then
        bSkip = true
      end
    end
    if not bSkip then
      self:TryReconnect()
    end
  end
  Log("[Ui:OnConnectLost] enter Ui:OnConnectLost param is ", not self.bForRetrunLogin, not self.bKickOffline)
  Lib:CallBack({
    PlayerEvent.OnEvent,
    PlayerEvent,
    PlayerEvent.EventType.OnConnectLost
  })
end

  if self.nReconnectAttemptCount >= self.UIReconnectSetting.MAXATTEMPT then
    self.nReconnectAttemptCount = 0
    Log("[Ui:OnConnectLost] Stop Reconnect Attempt")
  elseif PlayerEvent.bLogin and nTimeNow >= self.nNextReconnectTime then
    self.nNextReconnectTime = nTimeNow + self.UIReconnectSetting.INTERVAL
    self.nReconnectAttemptCount = self.nReconnectAttemptCount + 1
    self:ReconnectServer()
    Log("[Ui:OnConnectLost] ReconnectServer param is ", self.nNextReconnectTime, nTimeNow)
  else
    Ui:OpenWindow("UIMessageBoxBig", i18n.Get("DisconnectedFromServer"), nil, nil, nil, nil, true)
    Log("[Ui:OnConnectLost] ReconnectBox show , wait player select option")
  end
end

function Ui:SynNotifyMsg(tbData)
  self.__Notify_Msg_Index__ = self.__Notify_Msg_Index__ or 0
  self.__Notify_Msg_Index__ = self.__Notify_Msg_Index__ + 1
  tbData.__Notify_Msg_Index__ = self.__Notify_Msg_Index__
  local szType = tbData.szType
  local Key = tbUniqueNotifyMsg[szType]
  if Key then
    for i, v in ipairs(self.tbNotifyMsgDatas) do
      if v.szType == szType and v[Key] == tbData[Key] then
        self.tbNotifyMsgDatas[i] = tbData
        return
      end
    end
  end
  table.insert(self.tbNotifyMsgDatas, 1, tbData)
  self.nUnReadNotifyMsgNum = self.nUnReadNotifyMsgNum + 1
  EventNotify.OnNotify(EventNotify.emNOTIFY_NOTIFY_PROCESS_MSG)
  return self.__Notify_Msg_Index__
end

function Ui:CancelNotifyMsg(nMsgIndex)
  for nIndex, tbData in pairs(self.tbNotifyMsgDatas) do
    if tbData.__Notify_Msg_Index__ == nMsgIndex then
      table.remove(self.tbNotifyMsgDatas, nIndex)
      self.nUnReadNotifyMsgNum = self.nUnReadNotifyMsgNum > 1 and self.nUnReadNotifyMsgNum - 1 or 0
      EventNotify.OnNotify(EventNotify.emNOTIFY_NOTIFY_PROCESS_MSG)
      return
    end
  end
end

function Ui:OnSystemBackCall()
  local nMaxLayer = -1
  local szTarget = Client.UIModule.GetTopWindow()
  if nil ~= szTarget then
    local wnd = Ui:GetWindow(szTarget)
    if wnd then
      wnd:CloseByDefault()
    end
    return true
  end
  for k, v in pairs(self.tbUi) do
    if Ui:WindowVisible(k) and Ui.UISetting.IsAffectedBySystemBack(k) then
      local nLayer = Ui.UISetting.GetUISortingOrder(k)
      if nMaxLayer < nLayer then
        nMaxLayer = nLayer
        szTarget = k
      end
    end
  end
  if szTarget then
    local wnd = Ui:GetWindow(szTarget)
    if wnd then
      wnd:CloseByDefault()
    end
    return true
  end
  return false
end

function Ui:Button_BindEvent(pPanel, szButtonKey, func, ...)
  local btn = pPanel:GetButton(szButtonKey)
  local args = {
    ...
  }
  if btn then
    btn.onClick:RemoveAllListeners()
    local szUIPath = pPanel.UIPath
    btn.onClick:AddListener(function()
      local nSoundID = Client.AudioModule.GetSoundID("OnButtonClick")
      Client.UIModule.PlaySound(nSoundID)
      local szKey
      if Guide:IsLocked() and nil ~= szUIPath then
        szKey = szUIPath .. "/" .. szButtonKey
      end
      func(unpack(args))
      if Guide:IsLocked() and nil ~= szKey then
        EventNotify.OnNotify(EventNotify.emNOTIFY_BUTTON_EVENT, szKey)
      end
    end)
  else
    LogErr(string.format("ButtonBindEvent Error! Wnd=%s, key=%s not exist !", pPanel.ClassName, szButtonKey))
  end
end

function Ui:Button_UnBindEvent(pPanel, szButtonKey)
  local btn = pPanel:GetButton(szButtonKey)
  if btn then
    btn.onClick:RemoveAllListeners()
  else
    LogErr(string.format("Button_UnBindEvent Error! Wnd=%s, key=%s not exist !", pPanel.ClassName, szButtonKey))
  end
end

function Ui:Toggle_BindEvent(pPanel, szToggleKey, func, tbPanelSelf)
  local toggle = pPanel:GetToggle(szToggleKey)
  if toggle then
    toggle.onValueChanged:RemoveAllListeners()
    local szUIPath = pPanel.UIPath
    toggle.onValueChanged:AddListener(function(isOn)
      local trans = toggle.gameObject:GetComponent(Ui.BGroupTrans)
      if true == isOn then
        local nSoundID = Client.AudioModule.GetSoundID("OnToggleClick")
        Client.UIModule.PlaySound(nSoundID)
        if trans then
          trans:OnSelected()
        end
      elseif trans then
        trans:OnUnSelected()
      end
      local szKey
      if Guide:IsLocked() and nil ~= szUIPath then
        szKey = szUIPath .. "/" .. szToggleKey
      end
      func(tbPanelSelf, isOn)
      if Guide:IsLocked() and nil ~= szKey then
        EventNotify.OnNotify(EventNotify.emNOTIFY_TOGGLE_EVENT, szKey, isOn)
      end
    end)
  else
    LogErr(string.format("ToggleBindEvent Error! Wnd=%s, key=%s not exist !", pPanel.ClassName, szToggleKey))
  end
end

function Ui:Toggle_UnBindEvent(pPanel, szToggleKey, func, tbPanelSelf)

function Ui:GetUIGameObjectKey(gameObject)
  local tbNodeName = {}
  local bIsFindUIModule = false
  while true do
    if gameObject:GetComponent(typeof(CS.UIModule)) ~= nil then
      bIsFindUIModule = true
      break
    end
    table.insert(tbNodeName, 1, gameObject.transform.name)
    local parentTransform = gameObject.transform.parent
    if nil ~= parentTransform then
      gameObject = parentTransform.gameObject
    else
      break
    end
  end
  if bIsFindUIModule then
    local nNeedRemoveNodeNum = 1
    return table.concat(tbNodeName, "/", nNeedRemoveNodeNum + 1)
  else
    return nil
  end
end

function Ui:IsDlgAutoNext()
  if self.__Dlg_AutoNext__ == nil then
    self.__Dlg_AutoNext__ = true
  end
  return self.__Dlg_AutoNext__
end

function Ui:SetIsDlgAutoNext(bValue)
  local bOldValue = self:IsDlgAutoNext()
  self.__Dlg_AutoNext__ = bValue
  if bOldValue ~= bValue then
    EventNotify.OnNotify(EventNotify.emNOTIFY_DIALOG_AUTO_PLAY_CHANGE)
  end
end

function Ui:AddUIAnimScaleDown(gameObj, AnimTime, DownToScale)
  if not gameObj then
    return
  end
  local compObj = gameObj:GetComponent(Ui.UIAnimScaleDown)
  if not compObj then
    compObj = gameObj:AddComponent(Ui.UIAnimScaleDown)
    if compObj then
      compObj.AnimTime = AnimTime
      compObj.DownToScale = DownToScale
    end
  end
end

function Ui.ClearUIPanel(tbParent)
  if tbParent._tbBtnGroup then
    for tbBtnG, _ in pairs(tbParent._tbBtnGroup) do
      tbBtnG:OnDestroy()
    end
    tbParent._tbBtnGroup = nil
  end
  if tbParent._tbTabCtrl then
    for tbTabCtrl, _ in pairs(tbParent._tbTabCtrl) do
      tbTabCtrl:OnDestroy()
    end
    tbParent._tbTabCtrl = nil
  end
  if tbParent._tbScrollViewList then
    for tbSV, _ in pairs(tbParent._tbScrollViewList) do
      if tbSV.OnDestroy then
        tbSV:OnDestroy()
      end
    end
    tbParent._tbScrollViewList = nil
  end
  if tbParent._tbTree then
    for tbTree, _ in pairs(tbParent._tbTree) do
      tbTree:OnDestroy()
    end
    tbParent._tbTree = nil
  end
  if tbParent.tbSubPanelList then
    for tbPanel, _ in pairs(tbParent.tbSubPanelList) do
      tbPanel:UnInitPanel()
    end
    tbParent.tbSubPanelList = nil
  end
end
