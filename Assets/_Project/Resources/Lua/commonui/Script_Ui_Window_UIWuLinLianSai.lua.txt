function tbWnd:OnCreate()
  self.tbModelList = {}
  for nIndex = 1, self.MAX_MEMBER do
    local pObj = self.pPanel:GetObject(string.format(tbControls.NpcModel, nIndex))
    local pModel = pObj:GetComponent(Ui.UINpcModel)
    self.tbModelList[nIndex] = pModel
  end
  for nPhase, tbGroup in pairs(self.tbUIGroup) do
    for nIndex, szGroup in ipairs(tbGroup) do
      Ui:Button_BindEvent(self.pPanel, szGroup .. "/btnView", function()
        MultiPvP:C2SWatchBattle(nPhase, nIndex)
      end)
    end
  end
end

function tbWnd:OnClose()
  for _, pModel in ipairs(self.tbModelList) do
    pModel:Release()
  end
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self:UnRegistRedPoint()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MultiPvPView)
end

function tbWnd:OnOpen(bShowTeamInfo, bShowViewRedPoint)
  self.bShowTeamInfo = bShowTeamInfo
  self.bShowSelf = false
  self.bShowFinal = MultiPvP:IsShowFinalMatchDate()
  bShowViewRedPoint = true == bShowViewRedPoint and true or false
  if self.bShowFinal and not self.bShowTeamInfo then
    self:UpdateFinalPage()
  else
    self:UpdateTeamPage()
  end
  self:RegistRedPoint()
  self:UpdateRedPoint()
  if bShowViewRedPoint then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MultiPvPView)
  end
end

function tbWnd:UpdateTeamPage()
  self.pPanel:SetActive(tbControls.Team, true)
  self.pPanel:SetActive(tbControls.Guessing, false)
  self.m_tbTeamInfo = MultiPvP:GetCurTeamInfo()
  self.m_tbMembers = MultiPvP:GetCurTeamDetail()
  if not self.m_tbTeamInfo or not self.m_tbMembers then
    self.bShowSelf = true
    self:UpdateCreateTeam()
  else
    self.bShowSelf = self.m_tbTeamInfo[MPTeam.TEAM_ID] == MultiPvP:GetBattleTeamId(me.dwID)
    self.pPanel:SetActive(tbControls.ToggleAutoTeam, self.bShowSelf)
    local bAutoFight = 0 == me.GetUserValue(MultiPvP.SAVE_GROUP, MultiPvP.SAVE_NOT_AUTOFIGHT)
    self.pPanel:Toggle_SetChecked(tbControls.ToggleAutoTeam, bAutoFight)
    self:SetOtherState(self.bShowSelf, true)
    self:UpdateBottomInfo()
    self:UpdateTeamInfo()
  end
end

    self.pPanel:SetActive(self.tbMember[i] .. "/imgFaction", false)

  end
end

function tbWnd:UpdateTeamInfo()
  self.pPanel:Label_SetText(tbControls.txtTeamName, self.m_tbMembers.szTeamName)
  self:CleanTeamMember()
  for i = 1, 3 do
    if self.m_tbMembers.tbMember[i] then
      local tbInfo = self.m_tbMembers.tbMember[i]
      self.pPanel:Button_SetEnable(self.tbMember[i], true)
      Ui:Button_BindEvent(self.pPanel, self.tbMember[i], function()
        ViewRole:OpenWindow("UIViewRole", tbInfo.nPlayerId)
      end)
      self.pPanel:SetActive(self.tbMember[i] .. "/imgEmpty", false)
      self.pPanel:SetActive(self.tbMember[i] .. "/txtName", true)
      self.pPanel:SetActive(self.tbMember[i] .. "/HonorTitle", true)
      self.pPanel:SetActive(self.tbMember[i] .. "/imgFaction", true)
      self.pPanel:SetActive(self.tbMember[i] .. "/imgModel", true)
      self.pPanel:Label_SetText(self.tbMember[i] .. "/txtName", tbInfo.szPlayerName)
      local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(tbInfo.nHonorStarLevel, false)
      local obj = self.pPanel:GetObject(self.tbMember[i] .. "/HonorTitle")
      if obj then
        local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
        if pHonorTitle then
          pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
        end
      end
      self.pPanel:Sprite_SetSprite(self.tbMember[i] .. "/imgFaction", Faction:GetIcon(tbInfo.nFaction))
      self:ShowModel(i, tbInfo.nPlayerId)
    else
      local nTeamSize = MultiPvP:GetSeasonSettingValue("nMemberNeed", true)
      local nCurMember = #self.m_tbMembers.tbMember
      if self.bShowSelf and nTeamSize > nCurMember then
        self.pPanel:Button_SetEnable(self.tbMember[i], true)
        Ui:Button_BindEvent(self.pPanel, self.tbMember[i], function()
          Ui:OpenWindow("UITeamInviteWuLin", "MultiPvP")
        end)
        self.pPanel:SetActive(self.tbMember[i] .. "/imgEmpty", true)
      else
        self.pPanel:Button_SetEnable(self.tbMember[i], false)
        self.pPanel:SetActive(self.tbMember[i] .. "/imgEmpty", false)
      end
      self.pPanel:SetActive(self.tbMember[i] .. "/txtName", false)
      self.pPanel:SetActive(self.tbMember[i] .. "/HonorTitle", false)
      self.pPanel:SetActive(self.tbMember[i] .. "/imgFaction", false)
      self.pPanel:SetActive(self.tbMember[i] .. "/imgModel", false)
    end
  end
end

function tbWnd:ShowModel(i, nPlayerId)
  if self.m_tbTeamInfo[MPTeam.MEMBER] then
    for nIdx, tbData in pairs(self.m_tbTeamInfo[MPTeam.MEMBER]) do
      if nPlayerId == tbData.nPlayerId and tbData.tbFeature then
        self.tbModelList[i]:ShowModel(tbData.tbFeature.nSex, 1, tbData.tbFeature.nBodyResID, tbData.tbFeature.nWeaponResID, tbData.tbFeature.nMantleResID, 0, tbData.tbFeature.nHeadResID, tbData.tbFeature.nWindID, tbData.tbFeature.nWeaponEffectResID or 0)
        return
      end
    end
  end
end

function tbWnd:ShowResult()
  self.m_tbTeamInfo = MultiPvP:GetCurTeamInfo()
  if not self.m_tbTeamInfo then
    me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\230\178\161\230\156\137\230\136\152\233\152\159"))
    return
  end
  self.pPanel:SetActive(tbControls.wndResult, true)
  local nWin = MPTeam:GetWinCount(self.m_tbTeamInfo)
  local nFail = MPTeam:GetFailCount(self.m_tbTeamInfo)
  local nDraw = MPTeam:GetDrawCount(self.m_tbTeamInfo)
  local nAll = #self.m_tbTeamInfo[MPTeam.WIN_COUNT]
  local nWinRate = 50
  if nAll > 0 then
    nWinRate = math.floor(nWin / nAll * 100)
  end
  local nNextWinRate = math.floor((nWin + 1) / (nAll + 1) * 100)
  local nNextFailRate = math.floor(nWin / (nAll + 1) * 100)
  local nScore = self.m_tbTeamInfo[MPTeam.SCORE]
  local szText = i18n.EasyGet("\230\136\152\233\152\159\230\136\152\231\187\169\239\188\154{0}\232\131\156 {1}\229\185\179 {2}\232\180\159\n\230\136\152\233\152\159\232\131\156\231\142\135\239\188\154{3}%\n\228\184\139\229\156\186\232\142\183\232\131\156\229\144\142\232\131\156\231\142\135\239\188\154{4}%\n\228\184\139\229\156\186\229\164\177\232\180\165\232\131\156\231\142\135\239\188\154{5}%\n\230\175\148\232\181\155\231\167\175\229\136\134\239\188\154{6}", nWin, nDraw, nFail, nWinRate, nNextWinRate, nNextFailRate, nScore)
  self.pPanel:Label_SetText(tbControls.txtTemplate, szText)
end

function tbWnd:UpdateFinalPage()
  self.pPanel:SetActive(tbControls.Guessing, true)
  self.pPanel:SetActive(tbControls.Team, false)
  self:SetOtherState(true, true)
  self:UpdateBottomInfo()
  self:UpdateLeftTime()
  self.tbMatchList, self.tbTeamData, self.tbIndexMapId, self.bBattleBegin = MultiPvP:GetFinalData()
  if not self.tbMatchList or Lib:IsEmptyTB(self.tbMatchList) then
    return
  end
  for nPhase, tbTeamData in pairs(self.tbMatchList) do
    if self.tbUIGroup[nPhase] then
      for i = 1, nPhase / 2 do
        local nIdx1 = i
        if tbTeamData[nIdx1] and self.tbTeamData[tbTeamData[nIdx1].nTeamId] then
          self.pPanel:Label_SetText(self.tbUIGroup[nPhase][nIdx1] .. "/txtTeam1", self.tbTeamData[tbTeamData[nIdx1].nTeamId].szTeamName)
          self.pPanel:SetActive(self.tbUIGroup[nPhase][nIdx1] .. "/MarkWin1", tbTeamData[nIdx1].nResult == MPTeam.Win)
          Ui:Button_BindEvent(self.pPanel, self.tbUIGroup[nPhase][nIdx1] .. "/txtTeam1", function()
            MultiPvP:OpenGuessUI(tbTeamData[nIdx1].nTeamId)
          end)
        else
          self.pPanel:Label_SetText(self.tbUIGroup[nPhase][nIdx1] .. "/txtTeam1", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
          self.pPanel:SetActive(self.tbUIGroup[nPhase][nIdx1] .. "/MarkWin1", false)
        end
        local nIdx2 = nPhase - i + 1
        if tbTeamData[nIdx2] and self.tbTeamData[tbTeamData[nIdx2].nTeamId] then
          self.pPanel:Label_SetText(self.tbUIGroup[nPhase][nIdx1] .. "/txtTeam2", self.tbTeamData[tbTeamData[nIdx2].nTeamId].szTeamName)
          self.pPanel:SetActive(self.tbUIGroup[nPhase][nIdx1] .. "/MarkWin2", tbTeamData[nIdx2].nResult == MPTeam.Win)
          Ui:Button_BindEvent(self.pPanel, self.tbUIGroup[nPhase][nIdx1] .. "/txtTeam2", function()
            MultiPvP:OpenGuessUI(tbTeamData[nIdx2].nTeamId)
          end)
        else
          self.pPanel:Label_SetText(self.tbUIGroup[nPhase][nIdx1] .. "/txtTeam2", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
          self.pPanel:SetActive(self.tbUIGroup[nPhase][nIdx1] .. "/MarkWin2", false)
        end
      end
    end
  end
  local szText = i18n.EasyGet("\229\137\141\229\190\128\232\167\130\230\136\152")
  for i = 1, MultiPvP.PHASE_INIT do
    if self.tbMatchList[i] and Lib:CountTB(self.tbMatchList[i]) > 0 then
      if MultiPvP:IsInMatchList(self.tbMatchList[i], MultiPvP:GetBattleTeamId(me.dwID)) then
        szText = i18n.EasyGet("\229\143\130\229\138\160\230\180\187\229\138\168")
      end
      break
    end
  end
  local bShowWatch = MPTeam:ChkPlayerInMapEx(me) and 1 == me.GetUserValue(MultiPvP.SAVE_GROUP, MultiPvP.SAVE_WATCH)
  for nPhase, tbGroup in pairs(self.tbUIGroup) do
    for nIndex, szGroup in ipairs(tbGroup) do
      local bHaveMatch = false
      if self.tbIndexMapId and self.tbIndexMapId[nPhase] and self.tbIndexMapId[nPhase][nIndex] then
        bHaveMatch = true
      end
      local bShowViewBtn = bHaveMatch and bShowWatch or false
      self.pPanel:SetActive(szGroup .. "/btnView", bShowViewBtn)
    end
  end
  self.pPanel:Button_SetText(tbControls.btnJoin, szText, "Text")
  self:UpdateGuessTeam()
end

function tbWnd:UpdateGuessTeam()
  local nTeamId = me.GetUserValue(MultiPvP.SAVE_GROUP, MultiPvP.SAVE_GUESS)
  local szTeam = i18n.EasyGet("\231\171\158\231\140\156\233\152\159\228\188\141")
  self.pPanel:SetActive(tbControls.imgGuessingBg, true)
  if nTeamId >= 0 and self.tbTeamData[nTeamId] then
    szTeam = self.tbTeamData[nTeamId].szTeamName
  elseif not MultiPvP:IsGuessOpen() or self.bBattleBegin then
    szTeam = ""
    self.pPanel:SetActive(tbControls.imgGuessingBg, false)
  end
  self.pPanel:Label_SetText(tbControls.txtGuessing, szTeam)
  self.pPanel:SetActive(tbControls.btnGuess, MultiPvP:IsGuessOpen() and not self.bBattleBegin and Lib:IsEmptyTB(self.tbMatchList[1]))
end

function tbWnd:UpdateLeftTime()
  local szTime = i18n.EasyGet("\231\171\158\231\140\156\229\183\178\231\187\147\230\157\159")
  local nTimeRemaining
  if MultiPvP:IsFinalOpen() then
    nTimeRemaining = Lib:ParseTodayTime(MultiPvP.Cfg.OPEN_TIME) - Lib:GetTodaySec()
  else
    nTimeRemaining = 86400 - Lib:GetTodaySec() + Lib:ParseTodayTime(MultiPvP.Cfg.OPEN_TIME)
  end
  if MultiPvP:IsGuessOpen() and nTimeRemaining > 0 then
    local nHour, nMin = Lib:TransferSecond2NormalTime(nTimeRemaining)
    szTime = i18n.EasyGet("\231\173\137\229\190\133\230\175\148\232\181\155\229\188\128\229\144\175\229\128\146\232\174\161\230\151\182\239\188\154{0}\229\176\143\230\151\182{1}\229\136\134", nHour, nMin)
  end
  self.pPanel:Label_SetText(tbControls.txtCountdown, szTime)
  local nCurTime = tonumber(os.date("%Y%m%d%H%M", GetTime()))
  local nNextTime = Lib:GetDate2Time(nCurTime + 1) - GetTime()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  if nTimeRemaining > 0 then
    self.nTimer = Timer:Register(nNextTime * Env.GAME_FPS, self.UpdateLeftTime, self)
  end
  return false
end

function tbWnd:SendChatRecruit()
  local tbCondition = {
    [MultiPvP.emMsgNeedBattleTeam] = 1
  }
  if not MultiPvP:ChkConditionAndSendMsg(me, tbCondition) then
    return false
  end
  local nTeamSize = MultiPvP:GetSeasonSettingValue("nMemberNeed", true)
  local nCurMember = #self.m_tbMembers.tbMember
  if nTeamSize <= nCurMember then
    me.CenterMsg(i18n.EasyGet("\230\136\152\233\152\159\228\186\186\230\149\176\229\183\178\232\190\190\229\136\176\230\156\172\232\181\155\229\173\163\231\187\132\233\152\159\228\186\186\230\149\176\233\153\144\229\136\182"))
    return
  end
  Ui:CloseWindow(self.ClassName)
  Ui:OpenWindowWithCallback("UIChat", function(_, tbWnd)
    tbWnd:AddMultiPvPRecruit(self.m_tbMembers.nTeamId, self.m_tbMembers.szTeamName, nCurMember, nTeamSize)
  end, nil, ChatMgr.ChannelType.Public)
end

function tbWnd:UpdateRedPoint()
  local nCurPartnerId = me.GetUserValue(MultiPvP.SAVE_GROUP, MultiPvP.SAVE_PARTNER)
  if 0 == nCurPartnerId then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MultiPvPPartner)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MultiPvPPartner)
  end
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnPartner, {
    RedPointMgr.EVENT.MultiPvPPartner
  })
  for nPhase, tbGroup in pairs(self.tbUIGroup) do
    for nIndex, szGroup in ipairs(tbGroup) do
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szGroup .. "/btnView", {
        RedPointMgr.EVENT.MultiPvPView
      })
    end
  end
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnPartner)
  for nPhase, tbGroup in pairs(self.tbUIGroup) do
    for nIndex, szGroup in ipairs(tbGroup) do
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szGroup .. "/btnView")
    end
  end
end
