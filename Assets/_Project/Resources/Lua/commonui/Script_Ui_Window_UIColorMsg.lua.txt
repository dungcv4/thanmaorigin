local tbWnd = Ui:CreateWindow("UIColorMsg")
tbWnd.tbControls = {
  txtColorMsgCount = "imgBG/Frame/InnerBg/txtColorMsgCount",
  txtInput = "imgBG/Frame/InnerBg/txtInput",
  btnBuy = "imgBG/Frame/InnerBg/btnBuy",
  btnEmoji = "imgBG/Frame/InnerBg/btnEmoji",
  btnSend = "imgBG/Frame/InnerBg/btnSend",
  btnClose = "imgBG/Frame/btnClose"
}
tbWnd.tbOnClick = {
  btnBuy = function(self)
    self:BtnBuy()
  end,
  btnEmoji = function(self)
    self:BtnEmotionLink()
  end,
  btnSend = function(self)
    self:BtnSend()
  end,
  btnClose = function(self)
    self:CloseByDefault()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UPDATE_COLORMSG_COUNT,
      self.Update,
      self
    },
    {
      EventNotify.emNOTIFY_WND_CLOSED,
      self.OnCloseEmotionLink,
      self
    },
    {
      EventNotify.emNOTIFY_WND_OPENED,
      self.OnOpenShop,
      self
    }
  }
end

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.pPanel:Input_SetPlaceHolderText(self.txtInput, i18n.Get(ChatMgr.Terms.InputPlaceHolder, ChatMgr.nMaxColorMsgLen))
end

function tbWnd:OnOpen()
  self:Update()
end

function tbWnd:Update()
  local nColorTimes = ChatMgr:GetColorChatLeftCount(me)
  self.pPanel:Label_SetText(self.txtColorMsgCount, nColorTimes)
end

function tbWnd:OnCloseEmotionLink(szWndName)
end

function tbWnd:OnOpenShop(szWndName)
  if "UIShop" == szWndName then
    self:Close()
  end
end

function tbWnd:OuterInput(szCurTab, szMsg, tbChatLink)
  local szContent = self.pPanel:Input_GetText(self.txtInput) or ""
  if "Position" == szCurTab and string.match(szMsg, "^(%[.+%])(.*)$") then
    szContent = szMsg .. string.gsub(szContent, "^(%[.+%])(.*)$", "%2")
  elseif "Emoji" == szCurTab then
    szContent = szContent .. szMsg
  elseif "History" == szCurTab then
    szContent = szMsg
  else
    szContent = szContent .. szMsg
  end
  self.pPanel:Input_SetText(self.txtInput, szContent)
  if tbChatLink then
    ChatMgr:SetChatLink(tbChatLink.nLinkType, tbChatLink.linkParam)
  end
end

function tbWnd:CloseByDefault()
  Ui:CloseWindow("UIChatPlus")
  self:Close()
end

function tbWnd:BtnSend()
  if not ChatMgr:CheckSendMsg(ChatMgr.ChannelType.Color, nil, "check", true) then
    self:BtnBuy()
    return
  end
  local szMsg = self.pPanel:Input_GetText(self.txtInput) or ""
  local bSucceed = ChatMgr:SendMsg(ChatMgr.ChannelType.Color, nil, szMsg)
  if bSucceed then
    self.pPanel:Input_SetText(self.txtInput, "")
    me.CenterMsg(i18n.Get(ChatMgr.Terms.SendSuccess))
    Ui:CloseWindow("UIChatPlus")
    self:Close()
  end
end

function tbWnd:BtnEmotionLink()
  if Ui:WindowVisible("UIChatPlus") then
    Ui:CloseWindow("UIChatPlus")
  end
  Ui:OpenWindow("UIChatPlus", function(...)
    self:OuterInput(...)
  end, "Emoji")
end

function tbWnd:BtnBuy()
  local nWareId = Shop:GetWareIDByItemID(ChatMgr.nSpeakerColorItemId)
  local tbWare = Shop:GetAWare(nWareId)
  local nRemainCount = Shop:GetShopRemainCount(tbWare, me)
  Ui:OpenWindow("UIShopBuy", tbWare, true, nRemainCount)
end
