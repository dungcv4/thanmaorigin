local tbWnd = {}
Ui:SetClass("UIObjectGroup", tbWnd)

function Ui:NewObjectGroup(tbOwner, ...)
  local tbClass = Ui:NewClass(Ui:GetClass("UIObjectGroup"))
  tbClass.tbOwner = tbOwner
  tbClass:OnCreate(...)
  return tbClass
end

function tbWnd:OnCreate(tbKeyPrefix, nCount)
  self.tbObjectKeyList = {}
  for i = 1, nCount do
    local tbKey = {}
    for j = 1, #tbKeyPrefix do
      tbKey[j] = tbKeyPrefix[j] .. i
    end
    self.tbObjectKeyList[i] = tbKey
  end
end

function tbWnd:SetActive(i, j, bActive)
  self.tbOwner.pPanel:SetActive(self.tbObjectKeyList[i][j], bActive)
end

function tbWnd:SetActiveAll(bActive)
  for _, v in pairs(self.tbObjectKeyList) do
    for _, v2 in pairs(v) do
      self.tbOwner.pPanel:SetActive(v2, bActive)
    end
  end
end
