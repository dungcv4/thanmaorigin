local tbWnd = Ui:CreateWindow("UIQuickItem")
tbWnd.tbControls = {
  btnMain = "btnMain",
  btnClose = "btnClose",
  txtBtnMain = "btnMain/Text",
  txtTitle = "txtTitle",
  txtUp = "txtUp",
  txtDown = "txtDown"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnMain = function(self)
    self:OnClickMainButton()
  end
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.itemGrid = Ui:NewExistItemGrid(self, "CommitItem/UIItemGrid")
end

function tbWnd:OnOpen(szTitle, szUp, szDown, szMainButton, fnSetItemGrid, fnOnMainButton, bAutoClick, nClickDelayTime)
  self.pPanel:Label_SetText(self.txtTitle, szTitle)
  self.pPanel:Label_SetText(self.txtUp, szUp)
  self.pPanel:Label_SetText(self.txtDown, szDown)
  self.pPanel:Label_SetText(self.txtBtnMain, szMainButton)
  fnSetItemGrid(self.itemGrid)
  self.fnOnMainButton = fnOnMainButton
  if bAutoClick and self.fnOnMainButton then
    nClickDelayTime = nClickDelayTime or 1
    self.nTimer = Timer:Register(Env.GAME_FPS, self.fnOnMainButton, self)
  end
end

function tbWnd:OnClickMainButton()
  self:fnOnMainButton()
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
end
