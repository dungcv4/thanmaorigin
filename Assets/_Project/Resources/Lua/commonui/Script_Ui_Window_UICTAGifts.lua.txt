local tbWnd = Ui:CreateWindow("UICTAGifts")

function tbWnd:NotifyEvents()
  local tbRegEvent = {
    {
      EventNotify.emNOTIFY_SYN_DLC_GIFT_DATA,
      self.UpdatePanel,
      self
    }
  }
  return tbRegEvent
end

tbWnd.tbControls = {
  btnClose = "imgBG/btnClose",
  imgGiftBagS = "imgBG/StartGift/imgGiftBag",
  imgGiftBag = "imgBG/StartGift/imgGiftBag",
  GiftBag3D = "imgBG/StartGift/imgGiftBag/GiftBag3D",
  btnGetFirst = "imgBG/StartGift/BtnGet/btnGetFirst",
  btnGet20 = "imgBG/Gift20/btnGet",
  btnGet60 = "imgBG/Gift60/btnGet",
  btnGet100 = "imgBG/Gift100/btnGet",
  imgOpen = "imgBG/StartGift/BtnGet/imgOpen",
  txtReceived20 = "imgBG/Gift20/txtReceived",
  txtReceived60 = "imgBG/Gift60/txtReceived",
  txtReceived100 = "imgBG/Gift100/txtReceived",
  UIItemGrid20 = "imgBG/Gift20/item/UIItemGrid",
  UIItemGrid60 = "imgBG/Gift60/item/UIItemGrid",
  UIItemGrid100 = "imgBG/Gift100/item/UIItemGrid",
  imgForeground = "imgBG/imgground/imgForeground",
  Foreground_Effect = "imgBG/imgground/imgForeground/Foreground_Effect",
  txtDownloadPercent = "imgBG/TxtDownload/txtDownloadPercent",
  txtDownloadSpeed = "imgBG/TxtDownload/txtDownloadSpeed",
  txtDownloadSize = "imgBG/TxtDownload/txtDownloadSize"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.DlcAward)
    self:Close()
  end,
  btnGetFirst = function(self)
    RemoteServer.GetDLCGift(4)
  end,
  btnGet20 = function(self)
    RemoteServer.GetDLCGift(1)
  end,
  btnGet60 = function(self)
    RemoteServer.GetDLCGift(2)
  end,
  btnGet100 = function(self)
    RemoteServer.GetDLCGift(3)
  end
}

function tbWnd:OnCreate()
  self.itemGift20Reward = Ui:NewExistItemGrid(self, self.UIItemGrid20)
  self.itemGift60Reward = Ui:NewExistItemGrid(self, self.UIItemGrid60)
  self.itemGift100Reward = Ui:NewExistItemGrid(self, self.UIItemGrid100)
  local tbItem20 = Award:GetAward(Gift.DlcAwardID, 1)
  local tbItem60 = Award:GetAward(Gift.DlcAwardID, 2)
  local tbItem100 = Award:GetAward(Gift.DlcAwardID, 3)
  if nil ~= tbItem20 and nil ~= tbItem20[1] then
    self.itemGift20Reward:SetGenericItemTemplate(tbItem20[1][2], tbItem20[1][3])
    self.itemGift20Reward:SetClickEnable(true)
  end
  if nil ~= tbItem60 and nil ~= tbItem60[1] then
    self.itemGift60Reward:SetGenericItemTemplate(tbItem60[1][2], tbItem60[1][3])
    self.itemGift60Reward:SetClickEnable(true)
  end
  if nil ~= tbItem100 and nil ~= tbItem100[1] then
    self.itemGift100Reward:SetGenericItemTemplate(tbItem100[1][2], tbItem100[1][3])
    self.itemGift100Reward:SetClickEnable(true)
  end
  self.pPanel:Button_SetEnable(self.btnGetFirst, false)
  self.pPanel:Button_SetEnable(self.btnGet20, false)
  self.pPanel:Button_SetEnable(self.btnGet60, false)
  self.pPanel:Button_SetEnable(self.btnGet100, false)
end

function tbWnd:OnOpen()
  self:UpdatePanel()
  self:StartCalcProgressBar()
end

function tbWnd:UpdatePanel()
  local tbDlcAward = Gift:GetDLCGift()
  local nDefaultAward1 = tbDlcAward[1]
  if nDefaultAward1 and 1 == nDefaultAward1 then
    self.pPanel:SetActive(self.btnGet20, false)
    self.pPanel:SetActive(self.txtReceived20, true)
  end
  local nDefaultAward2 = tbDlcAward[2]
  if nDefaultAward2 and 1 == nDefaultAward2 then
    self.pPanel:SetActive(self.btnGet60, false)
    self.pPanel:SetActive(self.txtReceived60, true)
  end
  local nDefaultAward3 = tbDlcAward[3]
  if nDefaultAward3 and 1 == nDefaultAward3 then
    self.pPanel:SetActive(self.btnGet100, false)
    self.pPanel:SetActive(self.txtReceived100, true)
  end
  local nDefaultAward = tbDlcAward[4]
  if nDefaultAward and 1 == nDefaultAward then
    self.pPanel:SetActive(self.btnGetFirst, false)
    self.pPanel:SetActive(self.GiftBag3D, false)
    self.pPanel:SetActive(self.imgGiftBagS, true)
    self.pPanel:SetActive(self.imgOpen, true)
  end
end

function tbWnd:SetPercent(nPercent)
  nPercent = 0.2 + nPercent * 0.8
  self.pPanel:Label_SetText(self.txtDownloadPercent, math.floor(nPercent * 100) .. "%")
  self.pPanel:Sprite_SetFill(self.imgForeground, nPercent)
  self.pPanel:Sprite_SetFill(self.Foreground_Effect, nPercent)
  self.nCurPercent = nPercent
  if self.nCurPercent >= 0 then
    self.pPanel:Button_SetEnable(self.btnGet20, true)
  end
  if self.nCurPercent >= 0.2 then
    self.pPanel:Button_SetEnable(self.btnGet60, true)
  end
  if self.nCurPercent >= 0.6 then
    self.pPanel:Button_SetEnable(self.btnGet100, true)
  end
  if self.nCurPercent >= 1 then
    self.pPanel:Button_SetEnable(self.btnGetFirst, true)
  end
end

function tbWnd:CloseProgressTimer()
  if self.nProgressTimer ~= nil then
    Timer:Close(self.nProgressTimer)
    self.nProgressTimer = nil
  end
end

function tbWnd:StartCalcProgressBar()
  self:CloseProgressTimer()
  
  local function funProgress()
    local nDLCTatalSize = Client.UpdateModule.GetDLCTotalSize() or 0
    local nNeedDownloadDLCSize = Client.UpdateModule.GetNeedDownDLCSize()
    local nReadDownLoadSize = Client.UpdateModule.GetCurDownloadedDLCSize()
    local nSavedSize = nReadDownLoadSize + Client.UpdateModule.GetDownloadedDLCSize()
    local nDownTime = Client.UpdateModule.GetDownloadTime()
    self.pPanel:Label_SetText(self.txtDownloadSize, math.ceil(nNeedDownloadDLCSize / 1024 / 1024) .. "MB")
    local nPer = 1
    if 0 ~= nDLCTatalSize then
      nPer = nSavedSize / nDLCTatalSize
    end
    self:SetPercent(nPer)
    local nSpeed = 0
    if nPer < 1 then
      nSpeed = nReadDownLoadSize / nDownTime
    end
    self.pPanel:Label_SetText(self.txtDownloadSpeed, "(" .. math.ceil(nSpeed / 1024) .. "KB/s)")
    return true
  end
  
  funProgress()
  self.nProgressTimer = Timer:Register(Env.GAME_FPS, funProgress)
end

function tbWnd:OnClose()
  self:CloseProgressTimer()
end

function tbWnd:CloseByDefault()
  self:Close()
end
