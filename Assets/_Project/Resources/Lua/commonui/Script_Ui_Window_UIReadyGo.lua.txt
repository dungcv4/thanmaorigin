local tbWnd = Ui:CreateWindow("UIReadyGo")
tbWnd.tbImgReady = {
  [0] = "Start",
  [1] = "01",
  [2] = "02",
  [3] = "03",
  [4] = "04",
  [5] = "05"
}

function tbWnd:CloseByDefault()
end

function tbWnd:OnCreate()
  self:SetAllImg()
end

function tbWnd:OnOpen(nSec, szStartText)
  if not nSec or nSec <= 0 then
    nSec = 3
  end
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
  self.TotalSec = nSec
  self.nTimer = Timer:Register(Env.GAME_FPS, self.OnTimer, self)
  self:ShowCount(self.TotalSec)
  szStartText = szStartText or i18n.EasyGet("\229\188\128 \229\167\139")
  if szStartText then
    self.pPanel:Label_SetText("Start/txtStart02", szStartText)
    self.pPanel:Label_SetText("Start/txtStart", szStartText)
  end
end

function tbWnd:OnTimer()
  if self.TotalSec < 1 then
    if self.nTimer then
      Timer:Close(self.nTimer)
      self.nTimer = nil
    end
    Timer:Register(Env.GAME_FPS * 1.25, function(self)
      self:Close()
    end, self)
    return false
  end
  self.TotalSec = self.TotalSec - 1
  self:ShowCount(self.TotalSec)
  return true
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self:SetAllImg()
end

function tbWnd:ShowCount(nSec)
  local nKey = nSec <= 0 and 0 or nSec
  self:SetAllImg(nKey)
end

function tbWnd:SetAllImg(nKey)
  for i = 0, Lib:CountTB(self.tbImgReady) - 1 do
    self.pPanel:SetActive(self.tbImgReady[i], nKey == i)
  end
end
