Require("Script/Fuben/FubenUIController.lua")
XoyoGame.tbMapTemplateIds = XoyoGame.tbMapTemplateIds or {}

function XoyoGame:RegisterFubenUIChecker()
  local tbMapIds = self:GetXoyoAllMapTemplateId()
  for _, nMapId in pairs(tbMapIds) do
    self.tbMapTemplateIds[nMapId] = true
  end
  FubenUIController:RegisterChecker(tbMapIds, self)
end

XoyoGame:RegisterFubenUIChecker()

function XoyoGame:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenDefaultFuBenUI()
end

function XoyoGame:OpenTeamView()
  Ui:OpenWindow("UITeam")
end

function XoyoGame:OnJoin()
  local nDifficultyLevel = 1
  if not TeamMgr:HasTeam() then
    self:OpenXoyoGameTeam(nDifficultyLevel)
    return
  end
  if not TeamMgr:IsCaptain() then
    me.CenterMsg(TeamMgr.OnlyCaptainCanJoinHintMsg())
    return
  end
  self:OpenXoyoGameTeam(nDifficultyLevel, true)
end

function XoyoGame:XYGCheckCanEnter(pPlayer, nIndex)
  local tbFubenSetting = self:GetFubenSetting(nIndex)
  local nLevelLimit = tbFubenSetting.nMinLevel
  if nLevelLimit > pPlayer.nLevel then
    return false, i18n.EasyGet("{0}\229\189\147\229\137\141\231\173\137\231\186\167\228\184\141\232\182\179{1}\239\188\140\230\151\160\230\179\149\232\191\155\229\133\165{2}\233\154\190\229\186\166", pPlayer.szName, nLevelLimit, tbFubenSetting.szName)
  end
  return true
end

function XoyoGame:OpenXoyoGameTeam(nDifficultyLevel, bIsHadTeam)
  local szSubType = XoyoGame:GetXoyoGameActivityType(nDifficultyLevel)
  local nActivityId = TeamMgr:GetActivityId("XoyoGame", szSubType)
  if bIsHadTeam then
    if TeamMgr:TeamFull() then
      TeamMgr:AutoSetAndEnterActivity(nActivityId)
    else
      Ui:OpenWindowWithCallback("UITeam", function(_, tbWnd)
        TeamMgr:SetTeamActivity(nActivityId)
        tbWnd:SelectBigButton("myTeam")
      end)
    end
  else
    Ui:OpenWindowWithCallback("UITeam", function(_, tbWnd)
      tbWnd:SelectBigAndSmallButton("XoyoGame", "XoyoGame" .. nDifficultyLevel)
    end)
  end
end

function XoyoGame:IsMapTemplateId(nMapTemplateId)
  return self.tbMapTemplateIds[nMapTemplateId] or false
end

function XoyoGame:IsNeedExitFubenButton()
  return self.bstatus
end

function XoyoGame:IsShowTipWhenQuit()
  return false
end

function XoyoGame:SCXoyoGameFubenEndStatus(bstatus)
  self.bstatus = bstatus
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function XoyoGame:XYGRecordRankReqC()
  RemoteServer.RequestJoinXoyoGame("XYGRecordRankReq")
end

function XoyoGame:XYGRecordRankRsp(nPosition)
  if Ui:WindowVisible("UIXiaoYaoLu") then
    Ui:GetWindow("UIXiaoYaoLu"):UpdateMyRank(nPosition)
  end
end

function XoyoGame:XYGLastRankRsp(tbShowInfo)
  if tbShowInfo and Lib:CountTB(tbShowInfo) > 0 and Ui:WindowVisible("UINewInformation") then
    Ui:GetWindow("UINewInformation"):UpdateXoyoGameRecordPanel(tbShowInfo)
  end
end

function XoyoGame:IsHadXYGRecordRankInfo(bIsHadLastRankInfo)
  self.bIsHadLastRankInfo = bIsHadLastRankInfo
end
