local tbWnd = Ui:CreateWindow("UIWuLinLianSaiAward")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnPreliminary = "node/imgBG/Content/btnList/btnPreliminary",
  btnFinal = "node/imgBG/Content/btnList/btnFinal",
  btnRate = "node/imgBG/Content/btnList/btnRate",
  Template = "node/imgBG/Content/AwardList/ScrollView/Viewport/Content/Template",
  txtRank = "node/imgBG/Content/TitleList/txtRank"
}
tbWnd.tbControls = tbControls
local tbPageBtnIndex = {
  [1] = tbControls.btnPreliminary,
  [2] = tbControls.btnFinal,
  [3] = tbControls.btnRate
}
tbWnd.tbTextTitle = {
  [1] = function()
    return i18n.EasyGet("\230\136\152\233\152\159\230\142\146\229\144\141")
  end,
  [2] = function()
    return i18n.EasyGet("\230\136\152\233\152\159\230\142\146\229\144\141")
  end,
  [3] = function()
    return i18n.EasyGet("\232\131\156\231\142\135\229\140\186\233\151\180")
  end
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbPageGroup = Ui:NewButtonGroup(self, tbPageBtnIndex, true, function(self, nIndex)
    self:OnSelect(nIndex)
  end)
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.tbItemListGroup = {}
end

function tbWnd:OnClose()
  for _, tbItemList in pairs(self.tbItemListGroup) do
    tbItemList:Clear()
  end
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(tbControls.btnPreliminary, false)
  self.pPanel:SetActive(tbControls.btnFinal, true)
  self.pPanel:SetActive(tbControls.btnRate, false)
  self.pPanel:Button_SetText(tbControls.btnFinal, i18n.EasyGet("\230\142\146\229\144\141\229\165\150\229\138\177"), "Text")
  local nDefaultType = 2
  self.tbPageGroup:Select(nDefaultType)
end

function tbWnd:OnSelect(nIndex)
  self.tbAwards = MultiPvP:GetAwardSetting(nIndex)
  self.pPanel:Label_SetText(tbControls.txtRank, self.tbTextTitle[nIndex]())
  self:UpdateView()
end

function tbWnd:UpdateView()
  self:SetTemplate()
end

function tbWnd:SetTemplate()
  self.ScrollView:Reset()
  for i, tbAward in ipairs(self.tbAwards) do
    self.ScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtRank", i18n.Parse(tbAward.szDesc))
      local tbAwards = Award:GetAward(tbAward.nAwardId, tbAward.nAwardKey)
      local tbItemList = self.tbItemListGroup[i] or Ui:NewItemGridView(self, tbElement.szKey .. "/ItemList/Item")
      self.tbItemListGroup[i] = tbItemList
      for j = 1, #tbAwards do
        tbItemList:SetItem(j, tbAwards[j], true)
      end
      tbItemList:ResetFrom(#tbAwards + 1)
    end)
  end
end
