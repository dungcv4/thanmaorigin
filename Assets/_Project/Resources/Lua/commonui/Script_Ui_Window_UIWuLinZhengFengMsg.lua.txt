local tbWnd = Ui:CreateClass("UIWuLinZhengFengMsg", "UIPanel")
local tbControls = {
  btnRanking = "btnRanking",
  txtTips = "txtTips",
  List = "imgList/List",
  RankTemplate = "imgList/List/Template",
  txtInfo = "txtInfo"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnRanking = function(self)
    Ui:OpenWindow("UIRanking", "FightPower")
  end
}

function tbWnd:OnCreate()
  self.pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.WuLinZhengFeng)
end

function tbWnd:OnDestroy()
  if self.pInstance then
    self.pInstance = nil
  end
end

function tbWnd:OnOpen()
  if not self.bReg then
    EventNotify:RegistNotify(EventNotify.emNOTIFY_WULINZHENGFENG_AWARD_UI, self.OnOpenAwardUI, self)
    EventNotify:RegistNotify(EventNotify.emNOTIFY_WULINZHENGFENG_NOT_AWARD_UI, self.OnOpenNotAwardUI, self)
    self.bReg = true
  end
  if RedPointMgr:CheckEvent(RedPointMgr.EVENT.WuLinZhengFeng) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.WuLinZhengFeng)
  end
  self:InitPanel()
  if self.pInstance and self.pInstance:IsOpen() then
    self.pInstance:ReqAwardPlayerList()
  end
end

function tbWnd:OnClose()
  if self.bReg then
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_WULINZHENGFENG_AWARD_UI, self)
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_WULINZHENGFENG_NOT_AWARD_UI, self)
    self.bReg = nil
  end
  if self.RankScrollView then
    self.RankScrollView:Clear()
    self.RankScrollView = nil
  end
  if self.tbRankInfo then
    self.tbRankInfo = nil
  end
end

function tbWnd:InitPanel()
  self.pPanel:SetActive(tbControls.txtTips, false)
  self.pPanel:SetActive(tbControls.List, false)
  self.pPanel:SetActive(tbControls.btnRanking, false)
end

local ITEM_PER_LINE = 1
local MIN_LINE = 1

function tbWnd:OnOpenNotAwardUI()
  self.pPanel:SetActive(tbControls.txtTips, true)
  self.pPanel:Label_SetText(tbControls.txtTips, i18n.EasyGet("\230\156\172\230\156\141\230\136\152\229\138\155\230\142\146\229\144\141\229\137\141 <color=#54AB4D>{0}</color> \230\136\150\233\151\168\230\180\190\230\136\152\229\138\155\230\142\146\229\144\141\231\172\172 <color=#54AB4D>{1}</color> \231\154\132\228\190\160\229\163\171\229\176\134\232\142\183\229\190\151\228\184\176\229\142\154\229\165\150\229\138\177", self.pInstance.tbConfig.nPersonRank, self.pInstance.tbConfig.nFactionRank))
  self.pPanel:SetActive(tbControls.btnRanking, true)
  self:UpdateAwardList()
  self.pPanel:SetActive(tbControls.List, true)
  self.pPanel:SetActive(tbControls.txtInfo, true)
  self.pPanel:Label_SetText(tbControls.txtInfo, i18n.EasyGet("\230\136\170\232\135\179{0}\231\187\147\231\174\151\230\142\146\229\144\141\229\185\182\229\143\145\230\148\190\230\180\187\229\138\168\229\165\150\229\138\177", self.pInstance.tbConfig.szAwardTime))
end

function tbWnd:UpdateAwardList()
  self.RankScrollView = Ui:NewScrollView(self, tbControls.RankTemplate, nil, false, false)
  self.RankScrollView:Clear()
  for i = 1, self.pInstance.tbConfig.nPersonRank do
    self.RankScrollView:SetElement(i, function(tbElement)
      self:SetPersonalAwardListElement(tbElement, i)
    end)
  end
  for i = self.pInstance.tbConfig.nPersonRank + 1, self.pInstance.tbConfig.nPersonRank + self.pInstance.tbConfig.nFactionRank do
    self.RankScrollView:SetElement(i, function(tbElement)
      self:SetFactionAwardListElement(tbElement, i - self.pInstance.tbConfig.nPersonRank)
    end)
  end
end

function tbWnd:SetPersonalAwardListElement(tbElement, nIndex)
  tbElement.pPanel:Label_SetText("RankingInfo/txtTips", i18n.EasyGet("\230\156\172\230\156\141\230\136\152\229\138\155\230\142\146\229\144\141\231\172\172{0}\229\144\141", nIndex))
  tbElement.AwardListScrollView = Ui:NewItemGridScrollView(tbElement, "AwardList/Viewport/Content/ItemTemplate", ITEM_PER_LINE, MIN_LINE)
  local tbAwards = Award:GetAward(self.pInstance.tbConfig.nAwardId or 0, self.pInstance.tbConfig.tbPersonalAward[nIndex] or 0)
  tbElement.AwardListScrollView:UpdateGenericItem(tbAwards, true)
end

function tbWnd:SetFactionAwardListElement(tbElement, nIndex)
  tbElement.pPanel:Label_SetText("RankingInfo/txtTips", i18n.EasyGet("\233\151\168\230\180\190\230\136\152\229\138\155\230\142\146\229\144\141\231\172\172{0}\229\144\141", nIndex))
  tbElement.AwardListScrollView = Ui:NewItemGridScrollView(tbElement, "AwardList/Viewport/Content/ItemTemplate", ITEM_PER_LINE, MIN_LINE)
  local tbAwards = Award:GetAward(self.pInstance.tbConfig.nAwardId or 0, self.pInstance.tbConfig.nShowFactionAwardKey or 0)
  tbElement.AwardListScrollView:UpdateGenericItem(tbAwards, true)
end

local RANK_ICON_NUM = 3

function tbWnd:OnOpenAwardUI(tbRankInfo)
  self.tbRankInfo = tbRankInfo
  self.pPanel:SetActive(tbControls.txtTips, true)
  self.pPanel:Label_SetText(tbControls.txtTips, i18n.EasyGet("\230\173\166\230\158\151\228\186\137\233\148\139\230\180\187\229\138\168\232\142\183\229\165\150\229\144\141\229\141\149"))
  self.pPanel:SetActive(tbControls.btnRanking, false)
  self:UpdateRankList(tbRankInfo)
  self.pPanel:SetActive(tbControls.List, true)
  self.pPanel:SetActive(tbControls.txtInfo, true)
  self.pPanel:Label_SetText(tbControls.txtInfo, i18n.EasyGet("\229\165\150\229\138\177\229\144\141\229\141\149\229\177\149\231\164\186\229\176\134\228\186\142{0}\231\187\147\230\157\159", Lib:TimeDesc16(Lib:CalcTimeFrameOpenTime(self.pInstance:GetScheduleCfg().szCloseTimeFrame))))
end

function tbWnd:UpdateRankList(tbRankInfo)
  self.RankScrollView = Ui:NewScrollView(self, tbControls.RankTemplate, nil, false, false)
  self.RankScrollView:Clear()
  for i = 1, #tbRankInfo.tbPersonalRank do
    self.RankScrollView:SetElement(i, function(tbElement)
      self:SetPersonalRankElement(tbElement, i)
    end)
  end
  for i = #tbRankInfo.tbPersonalRank + 1, #tbRankInfo.tbFactionRank + #tbRankInfo.tbPersonalRank do
    local nFaction = i - #tbRankInfo.tbPersonalRank
    if 0 ~= #tbRankInfo.tbFactionRank[nFaction] then
      self.RankScrollView:SetElement(i, function(tbElement)
        self:SetFactionRankElement(tbElement, nFaction)
      end)
    end
  end
end

function tbWnd:SetPersonalRankElement(tbElement, nIndex)
  if nIndex <= RANK_ICON_NUM then
    tbElement.pPanel:SetActive("RankingInfo/UIRankNo", true)
    for i = 1, RANK_ICON_NUM do
      tbElement.pPanel:SetActive("RankingInfo/UIRankNo/imgNO" .. i, i == nIndex)
    end
  else
    tbElement.pPanel:SetActive("RankingInfo/UIRankNo", false)
  end
  local tbPlayerInfo = self.tbRankInfo.tbPersonalRank[nIndex]
  tbElement.pPanel:Label_SetText("RankingInfo/txtName", tbPlayerInfo.szName)
  tbElement.pPanel:SetActive("RankingInfo/txtName", true)
  tbElement.pPanel:SetActive("RankingInfo/txtTips", false)
  local SpFaction = Faction:GetIcon(tbPlayerInfo.nFaction)
  tbElement.pPanel:Sprite_SetSprite("RankingInfo/imgFaction", SpFaction)
  tbElement.pPanel:SetActive("RankingInfo/imgFaction", true)
  tbElement.AwardListScrollView = Ui:NewItemGridScrollView(tbElement, "AwardList/Viewport/Content/ItemTemplate", ITEM_PER_LINE, MIN_LINE)
  local tbAwards = Award:GetAward(self.pInstance.tbConfig.nAwardId or 0, self.pInstance.tbConfig.tbPersonalAward[nIndex] or 0)
  tbElement.AwardListScrollView:UpdateGenericItem(tbAwards, true)
end

function tbWnd:SetFactionRankElement(tbElement, nFaction)
  if not self.tbRankInfo.tbFactionRank[nFaction] then
    tbElement.pPanel:SetActive(false)
    return
  end
  for i = 1, #self.tbRankInfo.tbFactionRank[nFaction] do
    local tbPlayerInfo = self.tbRankInfo.tbFactionRank[nFaction][i]
    tbElement.pPanel:SetActive("RankingInfo/UIRankNo", false)
    local szFactionName = Faction:GetName(tbPlayerInfo.nFaction)
    tbElement.pPanel:Label_SetText("RankingInfo/txtFaction", i18n.EasyGet("{0}\231\172\172{1}", szFactionName, i))
    tbElement.pPanel:SetActive("RankingInfo/txtFaction", true)
    tbElement.pPanel:Label_SetText("RankingInfo/txtName", tbPlayerInfo.szName)
    tbElement.pPanel:SetActive("RankingInfo/txtName", true)
    tbElement.pPanel:SetActive("RankingInfo/txtTips", false)
    local SpFaction = Faction:GetIcon(tbPlayerInfo.nFaction)
    tbElement.pPanel:Sprite_SetSprite("RankingInfo/imgFaction", SpFaction)
    tbElement.pPanel:SetActive("RankingInfo/imgFaction", true)
    tbElement.AwardListScrollView = Ui:NewItemGridScrollView(tbElement, "AwardList/Viewport/Content/ItemTemplate", ITEM_PER_LINE, MIN_LINE)
    local tbAwards = Award:GetAward(self.pInstance.tbConfig.nAwardId or 0, self.pInstance.tbConfig.tbFactionAward[nFaction][i] or 0)
    tbElement.AwardListScrollView:UpdateGenericItem(tbAwards, true)
  end
end
