function Lib:random(nBegin, nEnd)
  self.nRandomSeed = self.nRandomSeed or GetTime()
  self.nRandomSeed = (self.nRandomSeed * 3877 + 29573) % 4294967295
  if nEnd < nBegin then
    nBegin, nEnd = nEnd, nBegin
  end
  return nBegin + self.nRandomSeed % (nEnd - nBegin + 1)
end

function Lib:RandomWithWeights(tbWeights)
  local nTotal = 0
  for k, v in pairs(tbWeights) do
    nTotal = nTotal + v
  end
  local nResult = MathRandom(nTotal)
  for k, v in pairs(tbWeights) do
    if nResult >= nTotal - v then
      return k
    else
      nTotal = nTotal - v
    end
  end
end

function Lib:RandomWithWeights2(tbWeights)
  local nTotal = 0
  for k, tbV in pairs(tbWeights) do
    nTotal = nTotal + tbV.nWeight
  end
  local nResult = MathRandom(nTotal)
  for k, tbV in pairs(tbWeights) do
    if nResult >= nTotal - tbV.nWeight then
      return k, tbV
    else
      nTotal = nTotal - tbV.nWeight
    end
  end
end

  local tbCurInfo = {}
  return function()
    if #tbCurInfo <= 0 then

    local nRandom = MathRandom(#tbCurInfo)
    local nIdx = tbCurInfo[nRandom]
    table.remove(tbCurInfo, nRandom)

function Lib:CopyTinyTB(tb, layers, tbIgnore, curlayers)
  local tbCopy = {}
  tb = tb or {}
  tbIgnore = tbIgnore or {_tbBase = 1}
  layers = tonumber(layers) or 0
  curlayers = tonumber(curlayers) or 1
  if layers > 0 and layers < curlayers then
    return {"?"}
  end
  for k, v in pairs(tb) do
    local szType = type(v)
    if "table" == szType then
      if not tbIgnore[k] then
        tbCopy[k] = self:CopyTinyTB(v, layers, tbIgnore, curlayers + 1)
      end
    elseif "number" == szType or "string" == szType or "boolean" == szType then
      tbCopy[k] = v
    else
      tbCopy[k] = szType
    end
  end
  return tbCopy
end

function Lib:ShowTB(tbVar, szBlank, tbTravelledTables)
  szBlank = szBlank or ""
  if not tbVar or 0 == Lib:CountTB(tbVar) then
    print(szBlank .. "tbVar is nil or count is 0")
    return
  end
  tbTravelledTables = tbTravelledTables or {}
  if tbTravelledTables[tbVar] then
    print("ERROE~~ InShowTB Recursive reference \239\188\129\239\188\129\239\188\129")
    return 0
  end
  tbTravelledTables[tbVar] = true
  local tbType = {}
  for k, v in pairs(tbVar) do
    local nType = self:TypeId(type(v))
    if not tbType[nType] then
      tbType[nType] = {
        n = 0,
        name = type(v)
      }
    end
    local tbTmp = tbType[nType]
    tbTmp.n = tbTmp.n + 1
    tbTmp[tbTmp.n] = k
  end
  for i = 1, self.TYPE_COUNT do
    if tbType[i] then
      local tbTmp = tbType[i]
      local szType = tbTmp.name
      table.sort(tbTmp, self.Comp)
      for i = 1, tbTmp.n do
        local key = tbTmp[i]
        local value = tbVar[key]
        local str
        if type(key) == "number" then
          str = szBlank .. "[" .. key .. "]"
        else
          str = szBlank .. "." .. key
        end
        if "nil" == szType then
          print(str .. "\t= nil")
        elseif "number" == szType then
          print(str .. "\t= " .. tbVar[key])
        elseif "string" == szType then
          print(str .. "\t= \"" .. tbVar[key] .. "\"")
        elseif "function" == szType then
          print(str .. "()")
        elseif "table" == szType then
          if tbVar[key] == tbVar then
            print(str .. "\t= {...}(self)")
          else
            print(str .. ":")
            if 0 == self:ShowTB(tbVar[key], str, tbTravelledTables) then
              print("recursively reference key:", key, " value:", tbVar[key])
            end
          end
        elseif "userdata" == szType then
          print(str .. "*")
        else
          print(str .. "\t= " .. tostring(tbVar[key]))
        end
      end
    end
  end
  tbTravelledTables[tbVar] = nil
end

function Lib:GetSubTable(tbA, tbB)
  local function funContain(tbTemp, value)
    for _, v in pairs(tbTemp) do
      if v == value then
        return true
      end
    end
    return false
  end
  
  local tbSub = {}
  for _, v in pairs(tbA) do
    if not funContain(tbB, v) then
      table.insert(tbSub, v)
    end
  end
  return tbSub
end

function Lib:GetSubTable2(tbA, tbB)
  local function funContain(tbTemp, key)
    if tbTemp[key] then
      return true
    else
      return false
    end
  end
  
  local tbSub = {}
  for k, v in pairs(tbA) do
    if not funContain(tbB, k) then
      tbSub[k] = v
    end
  end
  return tbSub
end

function Lib:StrTrim(szDes, szTrimChar)
  szTrimChar = szTrimChar or " "
  if 1 ~= string.len(szTrimChar) then
    return szDes
  end
  local szRet, nCount = string.gsub(szDes, "(" .. szTrimChar .. "*)([^" .. szTrimChar .. "]*.*[^" .. szTrimChar .. "])(" .. szTrimChar .. "*)", "%2")
  if 0 == nCount then
    return ""
  end
  return szRet
end

function Lib:Val2Str(var, szBlank)
  local szType = type(var)
  if "nil" == szType then
    return "nil"
  elseif "number" == szType then
    return tostring(var)
  elseif "string" == szType then
    return self:StrVal2Str(var)
  elseif "function" == szType then
    local szCode = string.dump(var)
    local arByte = {
      string.byte(szCode, i, #szCode)
    }
    szCode = ""
    for i = 1, #arByte do
      szCode = szCode .. "\\" .. arByte[i]
    end
    return "load(\"" .. szCode .. "\")"
  elseif "table" == szType then
    szBlank = szBlank or ""
    local szTbBlank = szBlank .. "  "
    local szCode = ""
    for k, v in pairs(var) do
      local szPair = szTbBlank .. "[" .. self:Val2Str(k) .. "]\t= " .. self:Val2Str(v, szTbBlank) .. ",\n"
      szCode = szCode .. szPair
    end
    if "" == szCode then
      return "{}"
    else
      return "\n" .. szBlank .. "{\n" .. szCode .. szBlank .. "}"
    end
  elseif "boolean" == szType then
    return var and "true" or "false"
  else
    return "\"" .. tostring(var) .. "\""
  end
end

function Lib:NewClass(tbBase, ...)
  local tbNew = {_tbBase = tbBase}
  setmetatable(tbNew, self._tbCommonMetatable)
  local tbRoot = tbNew
  local tbInit = {}
  repeat
    tbRoot = rawget(tbRoot, "_tbBase")
    local fnInit = rawget(tbRoot, "init")
    if type(fnInit) == "function" then
      table.insert(tbInit, fnInit)
    end
  until not rawget(tbRoot, "_tbBase")
  for i = #tbInit, 1, -1 do
    local fnInit = tbInit[i]
    if fnInit then
      fnInit(tbNew, ...)
    end
  end
  return tbNew
end

function Lib:SplitStr(szStrConcat, szSep)
  szSep = szSep or ","
  local tbStrElem = {}
  local tbSpeSep = {
    ["%."] = 1
  }
  local nSepLen = tbSpeSep[szSep] or #szSep
  local nStart = 1
  local nAt = string.find(szStrConcat, szSep)
  while nAt do
    tbStrElem[#tbStrElem + 1] = string.sub(szStrConcat, nStart, nAt - 1)
    nStart = nAt + nSepLen
    nAt = string.find(szStrConcat, szSep, nStart)
  end
  tbStrElem[#tbStrElem + 1] = string.sub(szStrConcat, nStart)
  return tbStrElem
end

function Lib:GetTableFromString(szValue, bKeyNotNumber, bValueNotNumber)
  local tbResult = {}
  local tbLines = Lib:SplitStr(szValue, ";")
  for _, szCell in ipairs(tbLines) do
    if "" ~= szCell then
      local Key, Value = string.match(szCell, "^([^|]+)|([^|]+)$")
      if not Key then
        return
      end
      if not bKeyNotNumber then
        Key = tonumber(Key)
      end
      if not bValueNotNumber then
        Value = tonumber(Value)
      end
      if not Key or not Value then
        return
      end
      tbResult[Key] = Value
    end
  end
  return tbResult
end

function Lib:CallBack(tbCallBack)
  local varFunc = tbCallBack[1]
  local szType = type(varFunc)
  if "function" == szType then
    return xpcall(tbCallBack[1], Lib.ShowStack, table.unpack(tbCallBack, 2))
  elseif "string" == szType then
    local fnFunc, tbSelf = KLib.GetValByStr(varFunc)
    if fnFunc then
      if tbSelf then
        return xpcall(fnFunc, Lib.ShowStack, tbSelf, table.unpack(tbCallBack, 2))
      else
        return xpcall(fnFunc, Lib.ShowStack, table.unpack(tbCallBack, 2))
      end
    end
  elseif WINDOWS then
    Lib.ShowStack()
  end
  return false, "Wrong name string : " .. tostring(varFunc)
end

end

function Lib:TransferSecond2NormalTime(nSecondTime)
  local nHour, nMinute, nSecond = 0, 0, 0
  if nSecondTime >= 3600 then
    nHour = math.floor(nSecondTime / 3600)
    nSecondTime = math.floor(nSecondTime % 3600)
  end
  if nSecondTime >= 60 then
    nMinute = math.floor(nSecondTime / 60)
    nSecondTime = math.floor(nSecondTime % 60)
  end
  nSecond = math.floor(nSecondTime)
  return nHour, nMinute, nSecond
end

  local szCnNum = ""
  local szNumTmp = ""
  if 0 == nDigit then
    return
  end
  if nDigit >= 10 and nDigit < 20 then
    if 10 == nDigit then
      szCnNum = tb4LenCnNum[2]
    else
      szCnNum = tb4LenCnNum[2] .. tbCnNum[math.floor(nDigit % 10)]
    end
    return szCnNum
  end
  for i = 1, #tb4LenCnNum do
    szNumTmp = ""
    nModel = math.floor(nDigitTmp % 10)
    if 0 ~= nModel then
      szNumTmp = szNumTmp .. tbCnNum[nModel] .. tb4LenCnNum[i]
      if 0 == nPreNum and bOneEver then
        szNumTmp = szNumTmp .. i18n.EasyGet("\233\155\182")
      end
      bOneEver = true
    end
    szCnNum = szNumTmp .. szCnNum
    nPreNum = nModel
    nDigitTmp = math.floor(nDigitTmp / 10)
    if 0 == nDigitTmp then
      break
    end
  end
  return szCnNum
end

function Lib:LoadTabFile(szFileName, tbNumColName, bOutsidePackage, bReadOnly)
  if nil == bReadOnly then
    bReadOnly = true
  end
  local tbData = KLib.LoadTabFileEx(szFileName, bOutsidePackage or 0)
  if not tbData then
    return
  end
  tbNumColName = tbNumColName or {}
  local tbColName = tbData[1]
  tbData[1] = nil
  local tbRet = {}
  for nRow, tbDataRow in pairs(tbData) do
    local tbRow = {}
    tbRet[nRow - 1] = tbRow
    for nCol, szName in pairs(tbColName) do
      if tbNumColName[szName] then
        tbRow[szName] = tonumber(tbDataRow[nCol]) or 0
      else
        tbRow[szName] = tbDataRow[nCol]
      end
    end
  end
  if bReadOnly then
    tbRet = SetReadOnly(tbRet)
  end
  return tbRet
end

function Lib:EasyLoadTabFile(szFile, tbIndex, tbField, bOutsidePackage, bReadOnly)
  if nil ~= tbField then
    tbField = self:ListToSet(tbField)
  end
  if nil == bOutsidePackage then
    bOutsidePackage = 0
  end
  if nil == bReadOnly then
    bReadOnly = true
  end
  local tbFile = KLib.LoadTabFileEx(szFile, bOutsidePackage)
  if not tbFile then
    return
  end
  local tbHeader = tbFile[1]
  tbFile[1] = nil
  local tbFnParse = {}
  for nCol, szField in pairs(tbHeader) do
    if string.match(szField, "^n") then
      tbFnParse[nCol] = tonumber
    elseif string.match(szField, "^tbsz") then
      tbFnParse[nCol] = function(szVal)
        return Lib:SplitStr(szVal, "|")
      end
    elseif string.match(szField, "^tbn") then
      tbFnParse[nCol] = function(szVal)
        return Lib:SplitStr2Num(szVal, "|")
      end
    else
      tbFnParse[nCol] = function(szVal)
        return szVal
      end
    end
  end
  local tbTemp = {}
  for nRow, tbRow in pairs(tbFile) do
    local tbData = {}
    for nCol, szField in pairs(tbHeader) do
      if not tbField or tbField[szField] then
        tbData[szField] = tbFnParse[nCol](tbRow[nCol])
      end
    end
    tbTemp[nRow - 1] = tbData
  end
  local tbRet
  if tbIndex and 1 == #tbIndex then
    tbRet = {}
    local szKey = tbIndex[1]
    for _, tbData in pairs(tbTemp) do
      tbRet[tbData[szKey]] = tbData
    end
  elseif tbIndex and 2 == #tbIndex then
    tbRet = {}
    local szPrimaryKey = tbIndex[1]
    local szSecondaryKey = tbIndex[2]
    for _, tbData in pairs(tbTemp) do
      tbRet[tbData[szPrimaryKey]] = tbRet[tbData[szPrimaryKey]] or {}
      tbRet[tbData[szPrimaryKey]][tbData[szSecondaryKey]] = tbData
    end
  elseif tbIndex and 3 == #tbIndex then
    tbRet = {}
    local szPrimaryKey = tbIndex[1]
    local szSecondaryKey = tbIndex[2]
    local szThirdKey = tbIndex[3]
    for _, tbData in pairs(tbTemp) do
      tbRet[tbData[szPrimaryKey]] = tbRet[tbData[szPrimaryKey]] or {}
      tbRet[tbData[szPrimaryKey]][tbData[szSecondaryKey]] = tbRet[tbData[szPrimaryKey]][tbData[szSecondaryKey]] or {}
      tbRet[tbData[szPrimaryKey]][tbData[szSecondaryKey]][tbData[szThirdKey]] = tbData
    end
  else
    tbRet = tbTemp
  end
  if bReadOnly then
    tbRet = SetReadOnly(tbRet)
  end
  return tbRet
end

function Lib:LoadBits(nInt32, nBegin, nEnd)
  if nEnd < nBegin then
    local _ = nBegin
    nBegin = nEnd
    nEnd = _
  end
  if nBegin < 0 or nEnd >= 32 then
    return 0
  end
  nInt32 = nInt32 % 2 ^ (nEnd + 1)
  nInt32 = nInt32 / 2 ^ nBegin
  return math.floor(nInt32)
end

function Lib:SetBits(nInt32, nBits, nBegin, nEnd)
  if nEnd < nBegin then
    local _ = nBegin
    nBegin = nEnd
    nEnd = _
  end
  nBits = nBits % 2 ^ (nEnd - nBegin + 1)
  nBits = nBits * 2 ^ nBegin
  nInt32 = nInt32 % 2 ^ nBegin + nInt32 - nInt32 % 2 ^ (nEnd + 1)
  nInt32 = nInt32 + nBits
  return nInt32
end

function Lib:GetTableBit(tb, nPos)
  assert(nPos >= 0, "Lib:GetTableBit nPos = " .. nPos)
  local nIndex = math.floor(nPos / 32)
  local nRealPos = nPos % 32
  if not tb or not tb[nIndex] then
    return 0
  end
  return self:LoadBits(tb[nIndex], nRealPos, nRealPos)
end

function Lib:SetTableBit(tb, nPos, nBit)
  assert(nPos >= 0, "Lib:GetTableBit nPos = " .. nPos)
  local nIndex = math.floor(nPos / 32)
  local nRealPos = nPos % 32
  if not nBit or 0 ~= nBit then
    nBit = 1
  end
  tb[nIndex] = tb[nIndex] or 0
  tb[nIndex] = self:SetBits(tb[nIndex], nBit, nRealPos, nRealPos)
end

function Lib:GetDate2Time(nDate)
  local nDate = tonumber(nDate)
  if nil == nDate then
    return
  end
  local nSecd = 0
  local nMin = 0
  local nHour = 0
  local nDay = 0
  local nMon = 0
  local nYear = 0
  if 8 == string.len(nDate) then
    nDay = math.mod(nDate, 100)
    nMon = math.mod(math.floor(nDate / 100), 100)
    nYear = math.mod(math.floor(nDate / 10000), 10000)
  elseif 10 == string.len(nDate) then
    nHour = math.mod(nDate, 100)
    nDay = math.mod(math.floor(nDate / 100), 100)
    nMon = math.mod(math.floor(nDate / 10000), 100)
    nYear = math.mod(math.floor(nDate / 1000000), 10000)
  elseif 12 == string.len(nDate) then
    nMin = math.mod(nDate, 100)
    nHour = math.mod(math.floor(nDate / 100), 100)
    nDay = math.mod(math.floor(nDate / 10000), 100)
    nMon = math.mod(math.floor(nDate / 1000000), 100)
    nYear = math.mod(math.floor(nDate / 100000000), 10000)
  elseif 14 == string.len(nDate) then
    nSecd = math.mod(nDate, 100)
    nMin = math.mod(math.floor(nDate / 100), 100)
    nHour = math.mod(math.floor(nDate / 10000), 100)
    nDay = math.mod(math.floor(nDate / 1000000), 100)
    nMon = math.mod(math.floor(nDate / 100000000), 100)
    nYear = math.mod(math.floor(nDate / 10000000000), 10000)
  else
    return 0
  end
  local tbData = {
    year = nYear,
    month = nMon,
    day = nDay,
    hour = nHour,
    min = nMin,
    sec = nSecd
  }
  local nSec = Lib:GetSecFromNowData(tbData)
  return nSec
end

end

function Lib:ParseDateTime(szDateTime)
  local year, month, day, hour, minute, second = string.match(szDateTime, "(%d+)/(%d+)/(%d+) (%d+):(%d+):(%d+)")
  if not year then
    year, month, day, hour, minute, second = string.match(szDateTime, "(%d+)-(%d+)-(%d+) (%d+):(%d+):(%d+)")
  end
  if not year then
    year, month, day, hour, minute, second = string.match(szDateTime, "(%d+).(%d+).(%d+) (%d+):(%d+):(%d+)")
  end
  if not year then
    year, month, day, hour, minute, second = string.match(szDateTime, "(%d%d%d%d)(%d%d)(%d%d)(%d%d)(%d%d)(%d%d)")
  end
  if not year then
    year, month, day = string.match(szDateTime, "(%d+)/(%d+)/(%d+)")
    hour, minute, second = 0, 0, 0
  end
  if not year then
    year, month, day = string.match(szDateTime, "(%d+)-(%d+)-(%d+)")
    hour, minute, second = 0, 0, 0
  end
  if not year then
    Log("Lib:ParseDateTime Illegal time string format" .. szDateTime, debug.traceback())
    return
  end
  return os.time({
    year = year,
    month = month,
    day = day,
    hour = hour,
    min = minute,
    sec = second
  })
end

function Lib:GetUft8Chars(s)
  local nTotalLen = string.len(s)
  local nCurIdx = 1
  local tbResult = {}
  while nTotalLen >= nCurIdx do
    local c = string.byte(s, nCurIdx)
    if c > 0 and c <= 127 then
      table.insert(tbResult, string.sub(s, nCurIdx, nCurIdx))
      nCurIdx = nCurIdx + 1
    elseif c >= 194 and c <= 223 then
      table.insert(tbResult, string.sub(s, nCurIdx, nCurIdx + 1))
      nCurIdx = nCurIdx + 2
    elseif c >= 224 and c <= 239 then
      table.insert(tbResult, string.sub(s, nCurIdx, nCurIdx + 2))
      nCurIdx = nCurIdx + 3
    elseif c >= 240 and c <= 244 then
      table.insert(tbResult, string.sub(s, nCurIdx, nCurIdx + 3))
      nCurIdx = nCurIdx + 4
    else
      break
    end
  end
  return tbResult
end

function Lib:CalRotOffPos(tbPos, tbOffPos, nDir)
  local nDir2 = 450 - nDir * 360 / Env.LOGIC_MAX_DIR
  local nRealDir = -math.rad(nDir2)
  local cosTheta = math.cos(nRealDir)
  local sinTheta = math.sin(nRealDir)
  local tbRelativePos = {
    tbOffPos[1] * 512 / 100,
    tbOffPos[2] * 512 / 100
  }
  local rotX = cosTheta * tbRelativePos[1] - sinTheta * tbRelativePos[2]
  local rotY = sinTheta * tbRelativePos[1] + cosTheta * tbRelativePos[2]
  local newPosX, newPosY = tbPos[1] + rotX, tbPos[2] + rotY
  newPosX = math.floor(newPosX)
  newPosY = math.floor(newPosY)
  return newPosX, newPosY
end

function Lib:PrintLuaTable(tbSomething, szTbName, szDeepSign)
  if not szDeepSign or "" == szDeepSign then
    szDeepSign = ""
    print(string.format("------------- start print table[%s]-------------", szTbName or "empty_table"))
    print(string.format("%s{", szDeepSign))
  end
  local oldDeepSign = szDeepSign
  szDeepSign = szDeepSign .. "    "
  if tbSomething then
    if type(tbSomething) == "table" then
      for key, value in pairs(tbSomething) do
        local szLine = ""
        local szFormat = ""
        if "string" == type(key) then
          szFormat = "%s[\"%s\"]"
        elseif type(key) == "number" then
          szFormat = "%s[%d]"
        end
        szLine = string.format(szFormat, szDeepSign, key)
        if not value then
          szLine = szLine .. " = nil,"
        elseif type(value) == "number" then
          szLine = szLine .. string.format(" = %d,", value)
        elseif "string" == type(value) then
          szLine = szLine .. string.format(" = \"%s\",", value)
        elseif type(value) == "boolean" then
          if true == value then
            szLine = szLine .. " = true,"
          else
            szLine = szLine .. " = false,"
          end
        end
        if type(value) == "table" then
          print(szLine .. " = {")
          self:PrintLuaTable(value, "subTb", szDeepSign)
        else
          print(szLine)
        end
      end
    end
  else
    print("nil")
  end
  print(string.format("%s},", oldDeepSign))
  if not oldDeepSign or "" == oldDeepSign then
    print("------------------ end ---------------------")
  end
end

function Lib:TrackEvent(dwPlayerId, tbLogReason, ...)

  end

      Log("Tracking_Dbg", dwPlayerId, szMainEvent, szSubEvent, szEasyExtParam)

function Lib:CalcTimeFrameOpenTime(szEvent, nOpenServerTime)
  local tbMapTimeFrame = Calendar.tbMapTimeFrame
  local nDay, nTime = 0, 0
  local nOpenTime = 0
  local bRet = false
  if tbMapTimeFrame[szEvent] then
    nDay = tbMapTimeFrame[szEvent].OpenDay
    nTime = tbMapTimeFrame[szEvent].OpenTime
    bRet = true
  end
  if not bRet then
    return nOpenTime
  end
  nDay = nDay - 1
  local nHour = math.floor(nTime / 100)
  local nMinute = math.mod(nTime, 100)
  nOpenServerTime = nOpenServerTime or GetServerCreateTime()
  nOpenServerTime = nOpenServerTime - math.mod(nOpenServerTime + Lib:GetGMTSec(), 86400)
  nOpenTime = nOpenServerTime + nDay * 24 * 3600 + nHour * 3600 + nMinute * 60
  return nOpenTime
end

function Lib:ParseNumbericDateToTime(szDateTime)
  if 14 ~= string.len(szDateTime) then
    LogErr("Lib:ParseNumbericDateToTimem format error! szDateTime = ", szDateTime)
    return
  end
  local nTmp = tonumber(szDateTime)
  if not nTmp then
    LogErr("Lib:ParseNumbericDateToTimem format error! szDateTime = ", szDateTime)
    return
  end
  local nYear = string.sub(szDateTime, 1, 4)
  local nMonth = string.sub(szDateTime, 5, 6)
  local nDay = string.sub(szDateTime, 7, 8)
  local nHour = string.sub(szDateTime, 9, 10)
  local nMinute = string.sub(szDateTime, 11, 12)
  local nSecond = string.sub(szDateTime, 13, 14)
  if not (nYear and nMonth and nDay and nHour and nMinute) or not nSecond then
    LogErr("Lib:ParseNumbericDateToTimem format error! szDateTime = ", szDateTime)
    return
  end
  local nTime = os.time({
    year = nYear,
    month = nMonth,
    day = nDay,
    hour = nHour,
    min = nMinute,
    sec = nSecond
  })
  if MODULE_GAMECLIENT then
    nTime = nTime - GetZoneTimeSecDiff()
  end
  return nTime
end

function Lib:GetLastWeekActiveDay(nCreatedTime)

function Lib:DumpTableLine(obj)
  local getIndent, quoteStr, wrapKey, wrapVal, dumpObj
  
  function getIndent(level)
    return ""
  end
  
  function quoteStr(str)
    return table.concat({
      "\"",
      string.gsub(str, "\"", "\\\""),
      "\""
    })
  end
  
  function wrapKey(val)
    if type(val) == "number" then
      return table.concat({
        "[",
        val,
        "]"
      })
    elseif type(val) == "string" then
      return table.concat({
        "[",
        quoteStr(val),
        "]"
      })
    else
      return table.concat({
        "[",
        tostring(val),
        "]"
      })
    end
  end
  
  function wrapVal(val, level)
    if type(val) == "table" then
      return dumpObj(val, level)
    elseif type(val) == "number" then
      return val
    elseif type(val) == "string" then
      return quoteStr(val)
    else
      return tostring(val)
    end
  end
  
  function dumpObj(obj, level)
    if type(obj) ~= "table" then
      return wrapVal(obj)
    end
    level = level + 1
    local tokens = {}
    tokens[#tokens + 1] = "{"
    for k, v in pairs(obj) do
      tokens[#tokens + 1] = table.concat({
        getIndent(level),
        wrapKey(k),
        " = ",
        wrapVal(v, level),
        ","
      })
    end
    tokens[#tokens + 1] = getIndent(level - 1) .. "}"
    return table.concat(tokens, "")
  end
  
  return dumpObj(obj, 0)
end

function Lib:GetTimeFrameOpenTime(szTimeFrame)
  if not MODULE_GAMESERVER then
    return 0
  end
  local nTimeFrameOpenTime = 0
  local tbOpenLevelTimeFrame = Calendar.tbOpenLevelTimeFrame
  if not tbOpenLevelTimeFrame then
    return nTimeFrameOpenTime
  end
  for i = 1, #tbOpenLevelTimeFrame do
    local v = tbOpenLevelTimeFrame[i]
    if v.Event == szTimeFrame then
      local nPassed = v.OpenDay * 24 * 3600
      local nOpenServerTime = Lib:GetTodayZeroHour(GetServerCreateTime()) + 14400
      nTimeFrameOpenTime = nOpenServerTime + nPassed
      break
    end
  end
  return nTimeFrameOpenTime
end

function Lib:GetZoneMaxTimeFrame(szCheckTimeFrame)
  local tbZoneClientInfoList = Server:GetAllZoneClientInfo()
  local nMinSvrOpenTime, szTimeFrame
  local nNowTime = GetTime()
  if tbZoneClientInfoList then
    for _, tbInfo in pairs(tbZoneClientInfoList) do
      if (not nMinSvrOpenTime or nMinSvrOpenTime > tbInfo.nServerOpenTime) and (not szCheckTimeFrame or nNowTime > Lib:CalcTimeFrameOpenTime(szCheckTimeFrame, tbInfo.nServerOpenTime)) then
        nMinSvrOpenTime = tbInfo.nServerOpenTime
      end
    end
  end
  for nIndex, tbInfo in ipairs(Calendar.tbOpenLevelTimeFrame) do
    local szEvent = tbInfo.Event
    if nNowTime > Lib:CalcTimeFrameOpenTime(szEvent, nMinSvrOpenTime) then
      szTimeFrame = szEvent
    end
  end
  return szTimeFrame
end
