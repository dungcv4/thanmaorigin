local tbWnd = Ui:CreateWindow("UIKinBossLootScore")
tbWnd.tbControls = {
  txtMsg = "Content/MsgTemplate/txtMsg"
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
end

function tbWnd:OnOpen(nMultiple)
  local nCloseTimeCount = 3
  self.pPanel:Label_SetText(self.txtMsg, i18n.EasyGet("\231\167\175\229\136\134 \195\151{0} \229\128\141", nMultiple))
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * nCloseTimeCount, function()
    self:Close()
  end)
end

function tbWnd:OnClose()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
end
