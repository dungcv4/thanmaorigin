Require("CommonScript/WuLinHonor/WuLinHonorDef.lua")
WuLinHonor.Protocol = {RequestRankData = 1, RequestRiseGradeAward = 2}

function WuLinHonor:GetNowSeasonMonth()
  return Lib:GetLocalMonth(GetTime() - 14400)
end

function WuLinHonor:GetGrade(pPlayer, nActivityId)
  return pPlayer.GetUserValue(self.GroupKey_Grade, nActivityId)
end

function WuLinHonor:GetAwardGrade(pPlayer, nActivityId)
  return pPlayer.GetUserValue(self.GroupKey_AwardGrade, nActivityId)
end

function WuLinHonor:GetRecordTimes(pPlayer, nRecordTypeId)
  return pPlayer.GetUserValue(self.GroupKey_RecordTimes, nRecordTypeId)
end

function WuLinHonor:HaveQualification(pPlayer, nQualificationId)
  local nValue = pPlayer.GetUserValue(self.GroupKey_Qualification, nQualificationId)
  return 0 ~= nValue
end

function WuLinHonor:IsReachOpenLevel(pPlayer)
  return pPlayer.nLevel >= self.nOpenLevel
end

function WuLinHonor:IsTimeFrameOpen()
  return 0 ~= GetTimeFrameState(self.szOpenTimeFrame)
end

function WuLinHonor:IsLastDayOfLastMonthTimeFrameOpen()
  local tbNowSeasonTime = os.date("*t", GetTime() - 14400)
  local nNowYear = tbNowSeasonTime.year
  local nNowMonth = tbNowSeasonTime.month
  local nLastDayOfLastMonthTime = os.time({
    year = nNowYear,
    month = nNowMonth,
    day = 1,
    hour = 4,
    min = 0,
    sec = 0
  }) - 86400
  return nLastDayOfLastMonthTime >= Lib:CalcTimeFrameOpenTime(self.szOpenTimeFrame)
end

function WuLinHonor:GetLastDayOfMonthTime(szDateTime)
  local nHour, nMinute, nSecond = string.match(szDateTime, "(%d+):(%d+):(%d+)")
  local tbNowSeasonTime = os.date("*t", GetTime() - 14400)
  local nNowYear = tbNowSeasonTime.year
  local nNowMonth = tbNowSeasonTime.month
  local nLastDayOfLastMonthTime = os.time({
    year = nNowYear,
    month = nNowMonth,
    day = Lib:GetThisMonthTotalDay(GetTime() - 14400),
    hour = nHour,
    min = nMinute,
    sec = nSecond
  })
  return nLastDayOfLastMonthTime
end

function WuLinHonor:GetNowSeasonRankBoardKey()
  local nSeasonMonth = self:GetNowSeasonMonth()
  if 0 == nSeasonMonth % 2 then
    return self.szEvenMonthRankBoardKey
  else
    return self.szOddMonthRankBoardKey
  end
end

function WuLinHonor:GetLastSeasonRankBoardKey()
  local nSeasonMonth = self:GetNowSeasonMonth() - 1
  if 0 == nSeasonMonth % 2 then
    return self.szEvenMonthRankBoardKey
  else
    return self.szOddMonthRankBoardKey
  end
end

function WuLinHonor:GetPlatinaGradeNum(pPlayer)
  local nNum = 0
  for nActivityId = 1, Lib:CountTB(self.tbActivityGroup) do
    if self:GetGrade(pPlayer, nActivityId) == self.tbGradeGroup.Platina then
      nNum = nNum + 1
    end
  end
  return nNum
end

function WuLinHonor:GetSumHonorScore(pPlayer)
  local nScore = 0
  for nActivityId = 1, Lib:CountTB(self.tbActivityGroup) do
    local nGrade = self:GetGrade(pPlayer, nActivityId)
    if nGrade > 0 then
      nScore = nScore + self.tbEachGradeCfg[nGrade].nHonorScore
    end
  end
  return nScore
end

function WuLinHonor:IsInShowPlatinaMasterRankTime()
  if not self:IsLastDayOfLastMonthTimeFrameOpen() then
    return false
  else
    local nDayInMonthDiff4Hours = Lib:GetLocalMonthDay(GetTime() - 14400)
    return nDayInMonthDiff4Hours < self.nShowRankEndDayInMonth
  end
end
