local tbWnd = Ui:CreateWindow("UIWuShuangHaoJieReport")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_WULINDAHUI_SYNC_MID,
      self.UpdateShow,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  GroupTemplate = "node/imgBG/Content/ReportList/Viewport/Content/GroupTemplate"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbGroupScrollView = Ui:NewScrollView(tbWnd, tbControls.GroupTemplate)
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen()
  self.nWarZone = WuLinDaHui.nWarZone
  WuLinDaHui:RequestMidMatch()
  self:UpdateShow()
end

function tbWnd:UpdateShow()
  if not WuLinDaHui.tbMidMatchList then
    return
  end
  self.tbMidMatchList = WuLinDaHui.tbMidMatchList
  self.tbGroupScrollView:Reset()
  for i, tbTeamList in ipairs(self.tbMidMatchList) do
    self.tbGroupScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("GroupTitle/txtGroup", i18n.Get(WuLinDaHui.GroupName[i]))
      tbElement.tbTeamScrollView = tbElement.tbTeamScrollView or Ui:NewScrollView(tbElement, "TeamList/TeamTemplate")
      tbElement.tbTeamScrollView:Reset()
      for nIdx, tbTeamData in ipairs(tbTeamList) do
        tbElement.tbTeamScrollView:SetElement(nIdx, function(tbTeamElement)
          tbTeamElement.pPanel:Label_SetText("txtName", tbTeamData.szTeamName or "")
          tbTeamElement.pPanel:Label_SetText("txtServer", tbTeamData.nServer and GetServerName(tbTeamData.nServer) or "")
          tbTeamElement.pPanel:Label_SetText("txtVictory", tbTeamData.nWin or "")
          tbTeamElement.pPanel:Label_SetText("txtTime", tbTeamData.nTime or "")
          tbTeamElement.pPanel:Label_SetText("txtScore", tbTeamData.nScore or "")
          tbTeamElement.pPanel:Label_SetText("txtRank", tbTeamData.nSort or "")
        end)
      end
    end)
  end
end
