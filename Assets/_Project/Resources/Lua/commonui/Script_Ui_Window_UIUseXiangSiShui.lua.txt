local tbWnd = Ui:CreateWindow("UIUseXiangSiShui")
local tbControls = {
  btnClose = "btnClose",
  txtTitle = "txtTitle",
  UIItemGrid = "CommitItem/UIItemGrid",
  btnMain = "btnMain",
  UIAddAndSubGroup = "UIAddAndSubGroup"
}
tbWnd.tbControls = tbControls

function tbWnd.NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      tbWnd.OnItemUpdate
    },
    {
      EventNotify.emNOTIFY_DEL_ITEM,
      tbWnd.OnItemDelete
    }
  }
end

tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnMain = function(self)
    self:UseXiangSiShui()
  end
}

function tbWnd:OnCreate()
  self.itemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroup)
  self.AddAndSubGroup:SetNumRange(1, 20, i18n.EasyGet("\229\183\178\232\190\190\228\184\138\233\153\144"))
  self.AddAndSubGroup:BindEvent(function(nValue)
    local nResult = self:UpdateNumberInput(nValue)
    return nResult
  end, nil, nil, nil, nil)
end

function tbWnd:OnOpen(nTemplateId, nMaxValue, nNpcId)
  self.nTemplateId = nTemplateId
  self.nMaxValue = nMaxValue or 20
  self.AddAndSubGroup:SetInputValue(1)
  self.nNpcId = nNpcId
  self:Update()
end

function tbWnd:Update()
  local tbItem = me.FindItem(self.nTemplateId or 0)
  self.tbItemIdList = {}
  for _, pItem in ipairs(tbItem or {}) do
    self.tbItemIdList[pItem.dwId] = true
  end
  if not next(self.tbItemIdList) then
    Timer:Register(1, Ui.CloseWindow, Ui, "UIUseXiangSiShui")
    return
  end
  local nItemCount = me.GetItemCountByTypeInBag(self.nTemplateId)
  local nMax = math.min(nItemCount, self.nMaxValue)
  self.AddAndSubGroup:SetNumRange(1, nMax)
  self.itemGrid:SetItemByTemplate(self.nTemplateId, nItemCount, nil, nil, {bShowTip = true})
  self.itemGrid:SetClickEnable(true)
  if nMax < self.AddAndSubGroup.nValue then
    self.AddAndSubGroup:SetInputValue(nMax)
  end
  self.nUseNum = self.AddAndSubGroup.nValue
end

function tbWnd:OnClose()
  self.nTemplateId = 0
  self.nActivityId = nil
end

function tbWnd:OnItemUpdate(nItemId)
  if self.tbItemIdList[nItemId] then
    self:Update()
  end
end

function tbWnd:OnItemDelete(nItemId)
  if self.tbItemIdList[nItemId] then
    self:Update()
  end
end

function tbWnd:UseXiangSiShui()
  if self.nNpcId and self.nUseNum then
    RemoteServer.RequestTongXinShu("UseSpecialWater", self.nNpcId, self.nUseNum)
  end
  self:Close()
end

function tbWnd:UpdateNumberInput(nNum)
  local nItemCount = me.GetItemCountByTypeInBag(self.nTemplateId)
  nNum = nNum > nItemCount and nItemCount or nNum
  self.nUseNum = nNum
  return self.nUseNum
end
