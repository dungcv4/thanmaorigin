function RedPointMgr:DynamicRegisteRedPoint(pPanel, szObjectPath, tbEventList)
  local obj = pPanel:GetObject(szObjectPath)
  local redPoint = obj:GetComponent(Ui.UIRedPoint)
  redPoint = redPoint or obj:AddComponent(Ui.UIRedPoint)
  redPoint:RegistEventList(tbEventList)
end

function RedPointMgr:DynamicUnRegist(pPanel, szObjectPath)
  local obj = pPanel:GetObject(szObjectPath)
  local redPoint = obj:GetComponent(Ui.UIRedPoint)
  if not redPoint then
    return
  end
  redPoint:UnRegist()
end

function RedPointMgr:DynamicAddEvent(pPanel, szObjectPath, nEventId)
  local obj = pPanel:GetObject(szObjectPath)
  local redPoint = obj:GetComponent(Ui.UIRedPoint)
  redPoint:AddEvent(nEventId)
end

function RedPointMgr:LightRedPoint(nEventId)
  local ins = self:GetInstance(nEventId)
  if not ins.light then
    ins.light = true
    if ins.ObjList then
      for key, obj in pairs(ins.ObjList) do
        if not Lib:IsUnityObjNil(obj) then
          obj:Light()
        end
      end
    end
  end
end

function RedPointMgr:FadeRedPoint(nEventId)
  local ins = self:GetInstance(nEventId)
  if ins.light then
    ins.light = false
    if ins.ObjList then
      for key, obj in pairs(ins.ObjList) do
        if not Lib:IsUnityObjNil(obj) then
          obj:Fade()
        end
      end
    end
  end
end

function RedPointMgr:CheckEvent(nEventId)
  return self:GetInstance(nEventId).light or false
end

function RedPointMgr:GetInstance(nEventId)
  if not RedPointMgr.Ins[nEventId] then
    RedPointMgr.Ins[nEventId] = {}
  end
  return RedPointMgr.Ins[nEventId]
end

function RedPointMgr:csharpRegistRedPoint(nEventId, nObjInstanceID, objRedPoint)
  if objRedPoint and objRedPoint.gameObject and objRedPoint.gameObject.activeInHierarchy == true then
    local newIns = self:GetInstance(nEventId)
    newIns.ObjList = newIns.ObjList or {}
    newIns.ObjList[nObjInstanceID] = objRedPoint
  end
end

function RedPointMgr:csharpUnRegistRedPoint(nEventId, nObjInstanceID)
  local newIns = self:GetInstance(nEventId)
  newIns.ObjList = newIns.ObjList or {}
  newIns.ObjList[nObjInstanceID] = nil
end

function RedPointMgr:OnLogout()
  RedPointMgr.Ins = {}
end

function RedPointMgr:DebugShow()
  Lib:Tree(RedPointMgr.Ins)
end

function RedPointMgr:CheckRedpointStatus(pPanel, szPath)
  local obj = pPanel:GetObject(szPath)
  if not obj then
    return
  end
  local redPoint = obj:GetComponent(Ui.UIRedPoint)
  if redPoint then
    redPoint:DebugShowRelativeEvent()
  end
end

function RedPointMgr:ForceLightRedPoint(pPanel, szObjectPath)
  local obj = pPanel:GetObject(szObjectPath)
  local redPoint = obj:GetComponent(Ui.UIRedPoint)
  redPoint = redPoint or obj:AddComponent(Ui.UIRedPoint)
  redPoint:Light()
end

function RedPointMgr:ForceFadeRedPoint(pPanel, szObjectPath)
  local obj = pPanel:GetObject(szObjectPath)
  local redPoint = obj:GetComponent(Ui.UIRedPoint)
  if redPoint then
    redPoint:Fade()
  end
end

function RedPointMgr:ChangeRedPointPositionLeft(pPanel, redPointObjPath)
  local redPointObj = pPanel:GetObject(redPointObjPath)
  if nil == redPointObj then
    return
  end
  local redPointRectTransform = redPointObj:GetComponent("RectTransform")
  redPointRectTransform:SetInsetAndSizeFromParentEdge(0, 0, 20)
  redPointRectTransform:SetInsetAndSizeFromParentEdge(2, 0, 20)
  redPointRectTransform.anchoredPosition3D = Ui.Vector3(15, -10, 0)
end

function RedPointMgr:SetRedPointPos(pPanel, szObjectPath, dwX, dwY, dwZ)
  local redPointObj = pPanel:GetObject(szObjectPath)
  if nil == redPointObj then
    return
  end
  local redPointComp = redPointObj:GetComponent(Ui.UIRedPoint)
  redPointComp:SetRedPointPos(dwX, dwY)
end
