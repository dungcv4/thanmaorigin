local tbWnd = Ui:CreateWindow("UIJinWangJiuXi")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnClose = "node/imgBg/btnClose",
  btnGo = "node/imgBg/Content/TimeLimit/Detail/HuoDeCaiLiaoDai/btnGo",
  btnMake = "node/imgBg/Content/TimeLimit/Detail/BaoZongZi/btnMake",
  btnGoHandIn = "node/imgBg/Content/TimeLimit/Detail/ShangGongZongZi/btnGo",
  txtAcitivityName = "node/imgBg/Content/TimeLimit/Detail/ShangGongZongZi/txtAcitivityName"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnMake = function(self)
    Ui:OpenWindow("UIBaoZongZi", self.nId)
  end,
  btnGoHandIn = function(self)
    if not self.pInstance then
      LogErr("UIBaoZongZi pInstance nil")
      return
    end
    self.pInstance:GotoHandInZongzi()
    self:CloseAllWindow()
  end,
  btnGo = function(self)
    Ui:OpenWindow("UIActivity", Calendar.TIMELIMIT, Calendar:GetActivityId("GongPinRobber"))
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nId)
  self:RegRedPoint()
  self.nId = nId
  self.pInstance = ActivityMgr:GetInstanceById(nId)
  self.pPanel:Label_SetText(tbControls.txtAcitivityName, i18n.EasyGet("\228\184\138\232\180\161\231\178\189\229\173\144"))
end

function tbWnd:OnClose()
  self:UnregRedPoint()
end

function tbWnd:RegRedPoint()
  local tbRedPointEvent = {
    RedPointMgr.EVENT.JingWangJiuxiShangdengMake,
    RedPointMgr.EVENT.JingWangJiuxiLvZongziMake,
    RedPointMgr.EVENT.JingWangJiuxiExtraAward
  }
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnMake, tbRedPointEvent)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGoHandIn, {
    RedPointMgr.EVENT.JingWangJiuxiZongziHandIn
  })
end

function tbWnd:UnregRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnMake)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnGoHandIn)
end

function tbWnd:CloseAllWindow()
  self:Close()
  if Ui:GetWindow("UIXiongWangActivity") then
    Ui:CloseWindow("UIXiongWangActivity")
  end
end
