local tbWnd = Ui:CreateWindow("UIActivationCode")
tbWnd.tbControls = {
  InputField = "imgBG/InputField",
  btnOk = "imgBG/btnOK",
  btnCancel = "imgBG/btnCancel"
}
tbWnd.tbOnClick = {
  btnOk = function(self)
    self:OnExchangeGiftCode()
  end,
  btnCancel = function(self)
    self:CloseView()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
end

function tbWnd:OnExchangeGiftCode()
  local szCode = self.pPanel:Input_GetText(self.InputField) or ""
  if not szCode or "" == szCode then
    me.CenterMsg(i18n.EasyGet("\232\175\183\232\190\147\229\133\165\231\164\188\229\147\129\231\160\129"))
    return
  end
  Sdk:exchangeGiftCode(szCode)
end

function tbWnd:CloseView()
  self:Close()
end
