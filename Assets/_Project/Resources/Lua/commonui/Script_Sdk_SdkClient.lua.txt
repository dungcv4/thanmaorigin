function Sdk:onRoleLevelup()
  if Ui.GameEnv.EnableSDK then
    Client.ChannelModule.onRoleLevelup()
    if me then
      local tbArgs = {
        vng_ge_user_id = GetAccountName(),
        role_id = tostring(me.dwID),
        server_id = tostring(Sdk:GetServerId()),
        role_class = tostring(me.nFaction),
        level = me.nLevel
      }
      local tbCleverTapArgs = {
        user_id = GetAccountName(),
        server_id = tostring(Sdk:GetServerId()),
        role_id = tostring(me.dwID),
        level = me.nLevel
      }
      if 13 == me.nLevel then
        Sdk:RecordAppsflyerEvent("af_level_achieved_13", tbArgs)
        Sdk:RecordFirebaseEvent("level_up_13", tbArgs)
        Sdk:RecordCleverTapEvent("level_up_13", tbCleverTapArgs)
      elseif 27 == me.nLevel then
        Sdk:RecordAppsflyerEvent("af_level_achieved_27", tbArgs)
        Sdk:RecordFirebaseEvent("level_up_27", tbArgs)
        Sdk:RecordCleverTapEvent("level_up_27", tbCleverTapArgs)
      end
      Sdk:RecordAppsflyerEvent("af_level_achieved", tbArgs)
      Sdk:RecordFirebaseEvent("level_up", tbArgs)
    end
  end
end

function Sdk:RecordAppsflyerEvent(szEvent, tbArgs)
  if not Ui.GameEnv.EnableSDK then
    return
  end
  if not Client.EnableAppsflyer then
    return
  end
  if not szEvent or "" == szEvent then
    LogErr("Sdk:RecordAppsflyerEvent Err, szEvent is nil")
    return
  end
  if not tbArgs or not next(tbArgs) then
    tbArgs = {}
  end
  tbArgs.eventType = szEvent
  local szArgs = Lib:EncodeJson(tbArgs)
  if "" == szArgs then
    szArgs = nil
  end
  Log("Sdk:RecordAppsflyerEvent", szArgs)
  Client.ChannelModule.CallXGMethod("appsflyerCustomEvent", tbArgs, "")
end

function Sdk:OnShopBuySuccess(nWareId, nCount)
  if not Ui.GameEnv.EnableSDK then
    return
  end
  local tbWare = Shop:GetAWare(nWareId)
  local nPrice = Shop:GetPrice(tbWare)
  local amount = nCount
  local customParams
  local currencyName = tbWare.szCurrencyType
  local currencyTotal = nPrice * nCount
  local itemType = tbWare.nItemId
  local itemName = Item:GetName(itemType, me)
  local itemNum = nCount
  self:onVirtualCurrencyConsume(itemName, amount, customParams, currencyName, currencyTotal, itemNum, itemType)
  local tbArgs = {
    vng_ge_user_id = GetAccountName(),
    role_id = tostring(me.dwID),
    server_id = tostring(Sdk:GetServerId()),
    role_class = tostring(me.nFaction),
    currency = currencyName,
    shopID = nWareId,
    itemname = itemName,
    itemID = tostring(tbWare.nItemId),
    amount = nCount,
    currencyprice = nPrice,
    total_spend = currencyTotal
  }
  local tbCleverTapArgs = {
    user_id = GetAccountName(),
    server_id = tostring(Sdk:GetServerId()),
    role_id = tostring(me.dwID),
    item_id = tostring(tbWare.nItemId),
    item_name = itemName,
    item_price = nPrice,
    item_currency = currencyName,
    item_quantity = nCount
  }
  Sdk:RecordAppsflyerEvent("af_spend_virtual_currency", tbArgs)
  Sdk:RecordFirebaseEvent("spend_virtual_currency", tbArgs)
  Sdk:RecordCleverTapEvent("spend_virtual_currency", tbCleverTapArgs)
end
