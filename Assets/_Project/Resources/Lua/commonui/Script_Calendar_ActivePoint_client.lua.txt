function ActivePoint:OnSyncInfo()
  self:CheckRedPoint()
  
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_ACTIVEPOINT_INFO)
end

function ActivePoint:TryGetAward(nIndex)
  RemoteServer.GetActivePointAward(nIndex)
end

function ActivePoint:CheckActivityFinish(szKey)
  local nSaveKey = ActivePoint:GetActSetting(szKey).SaveKey
  return ActivePoint:CheckFinish(me, nSaveKey)
end

function ActivePoint:CheckRedPoint()
  local tbRequireCount = ActivePoint.Def.tbAwardReqCount
  local nCount = ActivePoint:GetCount(me)
  local bNeedRedPoint = false
  for i = 1, #tbRequireCount do
    if nCount >= tbRequireCount[i] and not ActivePoint:CheckAward(me, i) then
      bNeedRedPoint = true
      break
    end
  end
  if bNeedRedPoint then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ActivePoint)
  elseif RedPointMgr:CheckEvent(RedPointMgr.EVENT.ActivePoint) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ActivePoint)
  end
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, ActivePoint.CheckRedPoint, ActivePoint)
