local tbWnd = Ui:CreateWindow("UIGetAwardShow")
tbWnd.tbControls = {
  ItemTemplate = "imgBG/ItemPanel/Scroll View/Viewport/Content/Template",
  Text = "imgBG/Text",
  txtTips = "imgBG/txtTips"
}

function tbWnd:OnCreate()
  self:SetClickEmptyCallback()
  self.tbItemScrollView = Ui:NewScrollView(self, self.ItemTemplate)
end

function tbWnd:OnOpen(tbAwards)
  self.bShowAll = false
  self.nPage = 1
  self:InitAwards(tbAwards)
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * 5, self.UpdateAwardShow, self)
end

function tbWnd:InitAwards(tbAwards)
  if #tbAwards > 21 then
    tbAwards = Award:MergetSameTypeAward(tbAwards)
  end
  self.tbItemScrollView:ResetFrom(#tbAwards)
  for i, tbAward in ipairs(tbAwards) do
    self.tbItemScrollView:SetElement(i, function(tbElement)
      local tbItemGrid = tbElement.tbItemGrid
      if not tbItemGrid then
        tbItemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        tbItemGrid:SetClickEnable(true)
      end
      tbElement.pPanel:SetActive("Effect/Quality1", false)
      tbElement.pPanel:SetActive("Effect/Quality2", false)
      local szType = tbAward[1]
      if szType == Player.AwardType2Name[Player.award_type_item] or szType == Player.AwardType2Name[Player.award_type_item_Unsafe] or szType == Player.AwardType2Name[Player.award_type_itemshow] then
        local szName, nIcon, nView, nQuality = KItem.GetItemShowInfo(tbAward[2], me.nFaction, me.nFactionSect)
        if nQuality >= 6 then
          tbElement.pPanel:SetActive("Effect/Quality2", true)
        elseif nQuality >= 4 then
          tbElement.pPanel:SetActive("Effect/Quality1", true)
        end
        if tbAward.dwItemId then
          tbItemGrid:SetItem(tbAward.dwItemId, tbAward[2])
        else
          tbItemGrid:SetItemByTemplate(tbAward[2], tbAward[3], nil, nil, nil, nil, nil, tbAward[4])
        end
      else
        tbItemGrid:SetDigitalItem(szType, tbAward[2])
      end
    end)
  end
  if self.nPage > 1 then
    self.pPanel:SetActive(self.txtTips, true)
    local szPageName = i18n.EasyGet("<color=#c5b505>\231\172\172 {0} \233\161\181</color>", self.nPage)
    self.pPanel:Label_SetText(self.txtTips, szPageName)
  else
    self.pPanel:SetActive(self.txtTips, false)
  end
  self.tbItemScrollView:PlayItemGridAnimation()
end

function tbWnd:UpdateAwardShow()
  if 0 == #Ui.tbCachedSendAwards then
    self:Close()
    return false
  end
  self.bShowAll = false
  self.nPage = self.nPage + 1
  Ui:UpdateAwardShow()
  return true
end

function tbWnd:OnClose()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
  Ui:OpenGetAwardShow()
  Guide:TryShowQueueHeadEquipWearUI()
end

function tbWnd:OnClickEmptyClose()
  if self.bShowAll then
    if self.nCloseTimer then
      Timer:Close(self.nCloseTimer)
      self.nCloseTimer = nil
    end
    self:UpdateAwardShow()
    self.nCloseTimer = Timer:Register(Env.GAME_FPS * 5, self.UpdateAwardShow, self)
  end
  self.bShowAll = true
  self.tbItemScrollView:StopAnimation()
end
