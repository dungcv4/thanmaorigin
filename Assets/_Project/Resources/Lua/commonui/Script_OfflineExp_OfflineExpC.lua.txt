function OfflineExp:OnGetExpFinish(nAddExp, nUseExpTime, nGetType)

function OfflineExp:RefreshClientMaxLevel()
  self.nCurMaxLevel = Calendar:GetWorldLevel()
end

function OfflineExp:GetClientMaxLevel()
  self:RefreshClientMaxLevel()
  return self.nCurMaxLevel
end

function OfflineExp:RefreshOfflineExp()
  if self:IsOfflineExp() then
    EventNotify:RegistNotify(EventNotify.emNOTIFY_HOOK_ONLINE_STOP, function(self)
      EventNotify:UnRegistNotify(EventNotify.emNOTIFY_HOOK_ONLINE_STOP, OfflineExp)
      self:HookStart()
    end, OfflineExp)
    self:HookStop()
  end
end

function OfflineExp:GetOfflineExpTime(pPlayer)
  if pPlayer.nLastLoginTime then
    local bIsCross = self:IsCrossDay(pPlayer, GetTime(), pPlayer.nLastLoginTime)
    if bIsCross then
      return self.OfflineExpTimePerDay
    end
  end
  return self:OfflineExpTime(pPlayer)
end

function OfflineExp:OnLogin(bIsReconnect)
  if not bIsReconnect then
    self:CreateReadyTimer()
  end
end

function OfflineExp:CreateReadyTimer()
  if self.nHookReadyTimer then
    Timer:Close(self.nHookReadyTimer)
  end
  self.nHookReadyCount = 0
  self.nHookReadyTimer = Timer:Register(Env.GAME_FPS * 5, function()
    self.nHookReadyCount = self.nHookReadyCount + 5
    if self.nHookReadyCount > self.nHookReadyNum then
      OfflineExp:HookStart()
    end
    return true
  end)
end

function OfflineExp:RemoveReadyTimer()
  if self.nHookReadyTimer then
    Timer:Close(self.nHookReadyTimer)
    self.nHookReadyTimer = nil
  end
  self.nHookReadyCount = 0
end

function OfflineExp:OnOpenWindow()
end

function OfflineExp:HookStart(bNotice)
  if self:IsOfflineExp() then
    return
  end
  local bOpen, szMsg = self:CheckIsOpen(me, true)
  if not bOpen then
    if bNotice then
      me.CenterMsg(szMsg)
    end
    return
  end
  RemoteServer.OfflineExpRequest("OnlineHookStart")
end

function OfflineExp:OnDisConnect()
  if self:IsOfflineExp() then
    me.bOfflineExpOnline = nil
  end
end

function OfflineExp:SCOfflineExpStart()
  local nOfflineExpTime = self:GetOfflineExpTime(me)
  local szTips = i18n.EasyGet("\230\137\152\231\174\161\229\188\128\229\167\139\239\188\140\231\155\174\229\137\141\229\137\169\228\189\153\229\143\175\230\137\152\231\174\161\230\151\182\233\151\180: {0}", Lib:TimeDesc8(nOfflineExpTime))
  me.bOfflineExpOnline = true
  self:RemoveReadyTimer()
  ChatMgr:OnChannelMessage(ChatMgr.ChannelType.System, 0, ChatMgr.SystemMsgType.System, "", 0, 0, 0, szTips)
  EventNotify.OnNotify(EventNotify.emNOTIFY_HOOK_ONLINE_START)
end

function OfflineExp:HookStop()
  self.nHookReadyCount = 0
  if not self:IsOfflineExp() then
    return
  end
  RemoteServer.OfflineExpRequest("OnlineHookStop")
end

function OfflineExp:SCOfflineExpStop()
  me.bOfflineExpOnline = nil
  self.nHookReadyCount = 0
  EventNotify.OnNotify(EventNotify.emNOTIFY_HOOK_ONLINE_STOP)
  self:CreateReadyTimer()
  self:OnCheckRedPoint()
end

function OfflineExp:OfflineExpFailed()
end

function OfflineExp:IsOfflineExp()
  return me.bOfflineExpOnline
end

function OfflineExp:OnExpChange()
  self:HookStop()
end

function OfflineExp:OnFightStatusChange()
  if me.nFightMode == Npc.FIGHT_MODE.emFightMode_Fight then
    self:HookStop()
  end
end

function OfflineExp:OnMapEnter(nTemplateID)
  if not Map:CanOnHook(nTemplateID) then
    self:HookStop()
  end
end

function OfflineExp:OnCheckRedPoint()
  if self.bClickUI then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.OfflineExp)
    return
  end
  local nExpTime = OfflineExp:ExpTime(me)
  if nExpTime > 21600 then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.OfflineExp)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.OfflineExp)
  end
end
