local tbWnd = Ui:CreateClass("UIwndJingZhuInfo", "UIPanel")
tbWnd.tbControls = {
  CurTemplate = "imgBg/CurrentAttr/AttrList/Template",
  txtCurrentLv = "imgBg/CurrentAttr/txtCurrentLv",
  txtNextLv = "imgBg/NextLvAttr/txtNextLv",
  txtActiveEquip = "imgBg/NextLvAttr/txtNextLv/txtActiveEquip",
  NextTemplate = "imgBg/NextLvAttr/AttrList/Template",
  Empty4Raycast = "Empty4Raycast",
  txtTitle = "imgBg/txtTitle"
}
tbWnd.tbOnClick = {
  Empty4Raycast = function(self)
    self.pPanel:SetActive("Main", false)
  end
}

function tbWnd:OnCreate()
  self.tbCurAttrScrollView = Ui:NewScrollView(self, self.CurTemplate, nil, false, true)
  self.tbNextAttrScrollView = Ui:NewScrollView(self, self.NextTemplate, nil, false, true)
end

function tbWnd:UpdateView(szGSType, nCurGradeScoreLevel, pAsyncData)
  if "JingZhu" == szGSType then
    self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\231\178\190\233\147\184\232\175\132\231\186\167"))
    local tbCurCfg = Item.EquipJingZhu.tbGradeScoreInfo[nCurGradeScoreLevel]
    local tbNextCfg = Item.EquipJingZhu.tbGradeScoreInfo[nCurGradeScoreLevel + 1] or tbCurCfg
    if tbCurCfg then
      self.pPanel:SetActive(self.txtCurrentLv, true)
      local nCurNeedNum = tbCurCfg[0][1]
      local nCurNeedLevel = tbCurCfg[0][2]
      self.pPanel:Label_SetText(self.txtCurrentLv, i18n.EasyGet("{0}\228\187\182{1}\231\186\167\232\163\133\229\164\135", nCurNeedNum, nCurNeedLevel))
    else
      self.pPanel:SetActive(self.txtCurrentLv, false)
    end
    if tbNextCfg then
      self.pPanel:SetActive(self.txtNextLv, true)
      local nCurNeedNum = tbNextCfg[0][1]
      local nCurNeedLevel = tbNextCfg[0][2]
      self.pPanel:Label_SetText(self.txtNextLv, i18n.EasyGet("{0}\228\187\182{1}\231\186\167\232\163\133\229\164\135", nCurNeedNum, nCurNeedLevel))
      local nCurNum = self:GetEJHGradeScoreLevelCount(nCurNeedLevel, pAsyncData)
      self.pPanel:Label_SetText(self.txtActiveEquip, "(" .. nCurNum .. "/" .. nCurNeedNum .. ")")
    end
    self:UpdateGradeScoreSV(tbCurCfg, tbNextCfg)
  elseif "Stone" == szGSType then
    self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\229\174\157\231\159\179\232\175\132\231\186\167"))
    local tbCurCfg = Item.Stone.tbGradeScoreInfo[nCurGradeScoreLevel]
    local tbNextCfg = Item.Stone.tbGradeScoreInfo[nCurGradeScoreLevel + 1] or tbCurCfg
    if tbCurCfg then
      self.pPanel:SetActive(self.txtCurrentLv, true)
      local nCurNeedNum = tbCurCfg[0][1]
      local nCurNeedLevel = tbCurCfg[0][2]
      self.pPanel:Label_SetText(self.txtCurrentLv, i18n.EasyGet("{0}\228\187\182{1}\231\186\167\229\174\157\231\159\179", nCurNeedNum, nCurNeedLevel))
    else
      self.pPanel:SetActive(self.txtCurrentLv, false)
    end
    if tbNextCfg then
      self.pPanel:SetActive(self.txtNextLv, true)
      local nCurNeedNum = tbNextCfg[0][1]
      local nCurNeedLevel = tbNextCfg[0][2]
      self.pPanel:Label_SetText(self.txtNextLv, i18n.EasyGet("{0}\228\187\182{1}\231\186\167\229\174\157\231\159\179", nCurNeedNum, nCurNeedLevel))
      local nCurNum = self:GetStoneGradeScoreLevelCount(nCurNeedLevel, pAsyncData)
      self.pPanel:Label_SetText(self.txtActiveEquip, "(" .. nCurNum .. "/" .. nCurNeedNum .. ")")
    end
    self:UpdateGradeScoreSV(tbCurCfg, tbNextCfg)
  end
end

function tbWnd:UpdateGradeScoreSV(tbCurCfg, tbNextCfg)
  self.tbCurAttrScrollView:Reset()
  if tbCurCfg then
    for i = 1, #tbCurCfg do
      self.tbCurAttrScrollView:SetElement(i, function(tbElement)
        local szDesc, szValue = FightSkill:GetMagicDescSplit(tbCurCfg[i][1], {
          table.unpack(tbCurCfg[i], 2)
        })
        tbElement.pPanel:Label_SetText("txtAttr", szDesc)
        tbElement.pPanel:Label_SetText("txtAttrValue", szValue)
      end)
    end
  else
    self.tbCurAttrScrollView:SetElement(1, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttr", i18n.EasyGet("\230\156\170\230\191\128\230\180\187"))
      tbElement.pPanel:Label_SetText("txtAttrValue", "")
    end)
  end
  if tbNextCfg then
    self.tbNextAttrScrollView:Reset()
    for i = 1, #tbNextCfg do
      self.tbNextAttrScrollView:SetElement(i, function(tbElement)
        local szDesc, szValue = FightSkill:GetMagicDescSplit(tbNextCfg[i][1], {
          table.unpack(tbNextCfg[i], 2)
        })
        tbElement.pPanel:Label_SetText("txtAttr", szDesc)
        tbElement.pPanel:Label_SetText("txtAttrValue", szValue)
      end)
    end
  end
end

function tbWnd:GetEJHGradeScoreLevelCount(nLevel, pAsyncData)
  local nCount = 0
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    if pAsyncData then
      local nJingZhuLevel = pAsyncData.GetJingZhu(i)
      if nLevel <= nJingZhuLevel then
        nCount = nCount + 1
      end
    else
      local nJingZhuLevel = Item.EquipJingZhu:GetJingZhuLevel(me, i - 1)
      if nLevel <= nJingZhuLevel then
        nCount = nCount + 1
      end
    end
  end
  return nCount
end

function tbWnd:GetStoneGradeScoreLevelCount(nLevel, pAsyncData)
  local nCount = 0
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    for j = 1, Item.Stone.INSET_COUNT_MAX do
      local nTemplateId
      if pAsyncData then
        nTemplateId = pAsyncData.GetStone((j - 1) * Item.EQUIPPOS_MAIN_NUM + i)
      else
        nTemplateId = Item.Stone:GetStoneHoleTemplateId(me, i - 1, j)
      end
      if nTemplateId and nTemplateId > 0 then
        local tbInfo = Item:GetItemBaseProp(nTemplateId)
        if tbInfo and nLevel <= tbInfo.nLevel then
          nCount = nCount + 1
        end
      end
    end
  end
  return nCount
end
