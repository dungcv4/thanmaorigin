Require("CommonScript/BlissSharing/BlissSharingCfg.lua")
BlissSharing.IS_SYS_OPEN = true
BlissSharing.SAVE_GROUP = GetGroupKey("BlissSharing")
BlissSharing.SAVE_ID_LEVEL_AWARD = 1
BlissSharing.SAVE_ID_RECHARGE_AWARD = 2
BlissSharing.SAVE_ID_INVITER = 3
BlissSharing.SAVE_ID_INVITE_LEVEL_AWARD = 4
BlissSharing.SAVE_ID_INVITE_RECHARGE_AWARD = 7
BlissSharing.INVITEE_NUM_LIMIT = 3
BlissSharing.AWARD_TYPE = {LEVEL = "LEVEL", RECHARGE = "RECHARGE"}
BlissSharing.AWARD_CHECKER = {
  [BlissSharing.AWARD_TYPE.LEVEL] = function(pPlayer, nCond)
    return nCond <= pPlayer.nLevel
  end,
  [BlissSharing.AWARD_TYPE.RECHARGE] = function(pPlayer, nCond)
    return nCond <= math.floor(Recharge:GetTotoalRecharge(pPlayer) / Recharge.PRICE_RATE)
  end
}
BlissSharing.AWARD_SAVE_ID = {
  [BlissSharing.AWARD_TYPE.LEVEL] = BlissSharing.SAVE_ID_LEVEL_AWARD,
  [BlissSharing.AWARD_TYPE.RECHARGE] = BlissSharing.SAVE_ID_RECHARGE_AWARD
}
BlissSharing.INVITE_AWARD_SAVE_ID = {
  [BlissSharing.AWARD_TYPE.LEVEL] = BlissSharing.SAVE_ID_INVITE_LEVEL_AWARD,
  [BlissSharing.AWARD_TYPE.RECHARGE] = BlissSharing.SAVE_ID_INVITE_RECHARGE_AWARD
}
BlissSharing.RED_POINT_TYPE = {SELF_AWARD = 1, INVITE_AWARD = 2}
