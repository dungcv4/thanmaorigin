function ShiYueDao:LoadIslandNpcSetting()
  local tbNPCList = Lib:LoadTabFileEx(ShiYueDao.Def.FLOORTWO_NPC_SETTING, "sddd", nil, {
    "Type",
    "nIndex",
    "PosX",
    "PosY"
  })
  for _, tbInfo in ipairs(tbNPCList) do
    if not (tbInfo.Type and tbInfo.nIndex and tbInfo.PosX) or not tbInfo.PosY then
      LogErr("ShiYueDao:LoadNpcSetting() load error!")
      return
    end
    if tbInfo.Type == "TransferFrom" then
      ShiYueDao.Def.TransferFromPOS = ShiYueDao.Def.TransferFromPOS or {}
      ShiYueDao.Def.TransferFromPOS[tbInfo.nIndex] = {
        nIndex = tbInfo.nIndex,
        PosX = tbInfo.PosX,
        PosY = tbInfo.PosY
      }
    elseif tbInfo.Type == "TransferTo" then
      ShiYueDao.Def.TransferToPOS = ShiYueDao.Def.TransferToPOS or {}
      ShiYueDao.Def.TransferToPOS[tbInfo.nIndex] = {
        nIndex = tbInfo.nIndex,
        PosX = tbInfo.PosX,
        PosY = tbInfo.PosY
      }
    elseif tbInfo.Type == "ShiYueDong" then
      ShiYueDao.Def.ShiYueDongPOS = ShiYueDao.Def.ShiYueDongPOS or {}
      ShiYueDao.Def.ShiYueDongPOS[tbInfo.nIndex] = {
        nIndex = tbInfo.nIndex,
        PosX = tbInfo.PosX,
        PosY = tbInfo.PosY
      }
    end
  end
end

function ShiYueDao:LoadHoleNpcSetting()
  local tbNPCList = Lib:LoadTabFileEx(ShiYueDao.Def.FLOORTHREE_NPC_SETTING, "sdd", nil, {
    "Type",
    "PosX",
    "PosY"
  })
  for _, tbInfo in ipairs(tbNPCList) do
    if not (tbInfo.Type and tbInfo.PosX) or not tbInfo.PosY then
      LogErr("ShiYueDao:LoadNpcSetting() load error!")
      return
    end
    local tbLocal = {
      PosX = tbInfo.PosX,
      PosY = tbInfo.PosY
    }
    if tbInfo.Type == "Boss" then
      ShiYueDao.Def.FloorThreeBossPos = ShiYueDao.Def.FloorThreeBossPos or {}
      table.insert(ShiYueDao.Def.FloorThreeBossPos, tbLocal)
    elseif tbInfo.Type == "Box" then
      ShiYueDao.Def.FloorThreeBoxPos = ShiYueDao.Def.FloorThreeBoxPos or {}
      table.insert(ShiYueDao.Def.FloorThreeBoxPos, tbLocal)
    elseif tbInfo.Type == "Monster" then
      ShiYueDao.Def.FloorThreeMonsterPos = ShiYueDao.Def.FloorThreeMonsterPos or {}
      table.insert(ShiYueDao.Def.FloorThreeMonsterPos, tbLocal)
    elseif tbInfo.Type == "MonsterHead" then
      ShiYueDao.Def.FloorThreeMonsterHeadPos = ShiYueDao.Def.FloorThreeMonsterHeadPos or {}
      table.insert(ShiYueDao.Def.FloorThreeMonsterHeadPos, tbLocal)
    end
  end
end

function ShiYueDao:LoadIslandMonsterSetting()
  local tbMonsterList = Lib:LoadTabFileEx(ShiYueDao.Def.FLOORTWO_MONSTER_SETTING, "sdd", nil, {
    "Type",
    "PosX",
    "PosY"
  })
  for _, tbInfo in ipairs(tbMonsterList) do
    if not (tbInfo.Type and tbInfo.PosX) or not tbInfo.PosY then
      LogErr("ShiYueDao:LoadFloorTwoMonsterSetting() load error!")
      return
    end
    local tbLocal = {
      PosX = tbInfo.PosX,
      PosY = tbInfo.PosY
    }
    if tbInfo.Type == "Monster" then
      ShiYueDao.Def.ShiYueDaoMonster = ShiYueDao.Def.ShiYueDaoMonster or {}
      table.insert(ShiYueDao.Def.ShiYueDaoMonster, tbLocal)
    end
  end
end

function ShiYueDao:LoadSetting()
  self:LoadIslandNpcSetting()
  self:LoadIslandMonsterSetting()
  self:LoadHoleNpcSetting()
end

ShiYueDao:LoadSetting()
