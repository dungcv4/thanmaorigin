local tbWnd = Ui:CreateClass("UIYuanXiaoFlower", "UIFestival_GetYuanXiaoFlower")
local tbControls = {
  szTitle = "txtTitle",
  txtTime01 = "Right/txtTime01",
  Template = "Right/List/Viewport/Content/Template",
  btnRank = "Right/btnRank",
  btnHelp = "Right/txtTitle/btnHelp",
  txtInfo = "Right/txtInfo",
  txtSecondTittle = "Right/txtSecondTittle",
  txtLV = "Right/txtLV"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbActivitySetting = {
    [self.tbType.Person] = 199,
    [self.tbType.Rank] = 199
  }
  self.tbLeftElementPath = {
    [self.tbType.Person] = {
      txtActivityName = "Left/List01/List/Viewport/Content/Template/txtActivityName",
      Template = "Left/List01/List/Viewport/Content/Template",
      txtSecondTittle = "Left/List01/txtSecondTittle"
    },
    [self.tbType.Rank] = {
      txtActivityName = "Left/List02/List/Viewport/Content/Template/txtActivityName",
      Template = "Left/List02/List/Viewport/Content/Template",
      txtSecondTittle = "Left/List02/txtSecondTittle"
    }
  }
  self.tbItemElementPath = {
    UIItemGrid = "UIItemGrid",
    szName = "txtItemCount",
    Nums = "txtItemCountValue"
  }
  self.szTime = tbControls.txtTime01
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.tbListView = {}
end

function tbWnd:UpdateListElement(tbElement, tbAward)
  local tbGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  if "Item" == tbAward[1] then
    tbGrid:SetItemByTemplate(tbAward[2], tbAward[3], nil, nil, {bShowTip = true})
    local szName = Item:GetName(tbAward[2], me)
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  else
    tbGrid:SetDigitalItem(tbAward[1], tbAward[3])
    local szName = Currency:GetName(tbAward[1])
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  end
  tbGrid:SetClickEnable(true)
end
