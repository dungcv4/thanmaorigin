function OfflineExp:CheckType(nGetType)
  for _, nType in pairs(OfflineExp.OfflineExpType) do
    if nType == tonumber(nGetType) then
      return true
    end
  end
end

function OfflineExp:IsHaveHookTime(pPlayer)
  return self:OfflineExpTime(pPlayer) > 0
end

function OfflineExp:IsHaveExpTime(pPlayer)
  local nExpTime = self:ExpTime(pPlayer)
  if nExpTime > 0 then
    local nHour, nMin = Lib:TransferSecond2NormalTime(nExpTime)
    if nHour * 60 + nMin > 0 then
      return true
    end
  end
end

function OfflineExp:ExpTime2Exp(pPlayer, nExpTime, nType, nRate)
  local nHour, nMin = Lib:TransferSecond2NormalTime(nExpTime)
  local nRet = math.ceil((nHour * 60 + nMin) * self.nBaseRate * self.GetExpRate[nType] * nRate)
  nRet = self:AddVipExp(pPlayer, nRet)
  return nRet
end

function OfflineExp:Exp2ExpTime(pPlayer, nExp, nType, nRate)
  local nCurVipPer = Vip:GetVipAddOnHookPercent(pPlayer.GetVipLevel())
  nExp = math.ceil(nExp / (1 + nCurVipPer))
  return math.ceil(nExp / self.nBaseRate / self.GetExpRate[nType] / nRate) * 60
end

function OfflineExp:NowExp(pPlayer, nType)
  local nExpTime = self:ExpTime(pPlayer)
  local nRate = pPlayer.GetBaseAwardExp()
  return self:ExpTime2Exp(pPlayer, nExpTime, nType, nRate)
end

function OfflineExp:MaxOpenLevel()
  if MODULE_GAMESERVER then
    return GetMaxTotalLevel()
  end
  if not MODULE_GAMESERVER and SERVER_RUNNING_MODE ~= SERVER_MODE.ZONE_SERVER then
    return self.nCurMaxLevel

  return 1000
end

function OfflineExp:IsCrossDay(pPlayer, nEndTime, nStartTime)
  nEndTime = nEndTime or GetTime()
  return nEndTime > self:CrossTime(pPlayer, nStartTime)
end

function OfflineExp:CrossTime(pPlayer, nStartTime)
  local nStartZeroTime = Lib:GetTodayZeroHour(nStartTime)
  local nStartHour, nStartMin, nStartSec = Lib:TransferSecond2NormalTime(nStartTime - nStartZeroTime)
  nStartTime = nStartTime - nStartMin * 60 - nStartSec
  local nCrossHour = (24 - (nStartHour - self.OfflineExpTimeReset_Hour)) % 24
  if 0 == nCrossHour then
    nCrossHour = 24
  end
  local nCrossTime = nStartTime + nCrossHour * 60 * 60
  return nCrossTime
end

function OfflineExp:CalcPassExpAndOfflineExpTime(pPlayer, nEndTime, nHookStartTime)
  local nPassExpTime = 0
  local nRemainOfflineExpTime = 0
  local nOfflineExpTime = self:OfflineExpTime(pPlayer)
  local nExpTime = self:ExpTime(pPlayer)
  local bIsCross = self:IsCrossDay(pPlayer, nEndTime, nHookStartTime)
  if nExpTime >= self.MaxExpTime then
    nRemainOfflineExpTime = nOfflineExpTime
    if bIsCross then
      nRemainOfflineExpTime = self.OfflineExpTimePerDay
    end
    return nPassExpTime, nRemainOfflineExpTime
  end
  if bIsCross then
    local nCrossTime = self:CrossTime(pPlayer, nHookStartTime)
    local nPassBeforeTime = nCrossTime - nHookStartTime
    nPassExpTime = nOfflineExpTime < nPassBeforeTime and nOfflineExpTime or nPassBeforeTime
    local nPassAfterTime = nEndTime - nCrossTime
    local nDay = math.floor(nPassAfterTime / 86400)
    nPassExpTime = nPassExpTime + nDay * self.OfflineExpTimePerDay
    nPassAfterTime = nPassAfterTime % 86400
    nPassExpTime = nPassExpTime + (nPassAfterTime > self.OfflineExpTimePerDay and self.OfflineExpTimePerDay or nPassAfterTime)
    nPassExpTime = nPassExpTime - self.nDelayTime
    nPassAfterTime = self:DragOutSecond(nPassAfterTime)
    nRemainOfflineExpTime = nPassAfterTime > self.OfflineExpTimePerDay and 0 or self.OfflineExpTimePerDay - nPassAfterTime
    if nExpTime + nPassExpTime > self.MaxExpTime then
      local nOverTime = nExpTime + nPassExpTime - self.MaxExpTime
      nRemainOfflineExpTime = nRemainOfflineExpTime + nOverTime
      if nRemainOfflineExpTime > self.OfflineExpTimePerDay then
        nRemainOfflineExpTime = self.OfflineExpTimePerDay or nRemainOfflineExpTime
      end
    end
  else
    local nPassTime = nEndTime - nHookStartTime - self.nDelayTime
    nPassTime = nPassTime < 0 and 0 or nPassTime
    nPassTime = self:DragOutSecond(nPassTime)
    nPassExpTime = nOfflineExpTime < nPassTime and nOfflineExpTime or nPassTime
    nRemainOfflineExpTime = nOfflineExpTime < nPassTime and 0 or nOfflineExpTime - nPassTime
    if nExpTime + nPassExpTime > self.MaxExpTime then
      local nOverTime = nExpTime + nPassExpTime - self.MaxExpTime
      nRemainOfflineExpTime = nRemainOfflineExpTime + nOverTime
      nRemainOfflineExpTime = nRemainOfflineExpTime > self.OfflineExpTimePerDay and self.OfflineExpTimePerDay or nRemainOfflineExpTime
    end
  end
  nPassExpTime = self:DragOutSecond(nPassExpTime)
  return nPassExpTime, nRemainOfflineExpTime
end

function OfflineExp:DragOutSecond(nTime)
  local nHour, nMin = Lib:TransferSecond2NormalTime(nTime)
  return nHour * 60 * 60 + nMin * 60
end

function OfflineExp:CheckCommond(pPlayer)
  local bOpen, szMsg = self:CheckIsOpen(pPlayer)
  if not bOpen then
    return bOpen, szMsg
  end
  local nExpTime = self:ExpTime(pPlayer)
  if nExpTime < 60 then
    return false, i18n.EasyGet("\230\178\161\230\156\137\229\143\175\233\162\134\229\143\150\231\154\132\231\166\187\231\186\191\231\187\143\233\170\140")
  end
  return true
end

function OfflineExp:GetNextNeedExp(nCurLevel)
  if self.tbOfflineExp[nCurLevel] then
    return self.tbOfflineExp[nCurLevel].nExpUpGrade
  end
  if ZongShiSkill.tbZongShiLevelSetting[nCurLevel] then
    return ZongShiSkill.tbZongShiLevelSetting[nCurLevel].ExpUpGrade
  end
  return -1
end

function OfflineExp:CheckIsMaxOpenLevel(pPlayer, nAddExp)
  local nHaveExp = pPlayer.GetExp()
  local nNowLevel = pPlayer.nTotalLevel
  local nNextExp = self:GetNextNeedExp(nNowLevel)
  return nNowLevel >= self:MaxOpenLevel() and nNextExp <= nHaveExp + nAddExp
end

function OfflineExp:PayTips(pPlayer, nPayType)
  local szTip = ""
  local nExpTime = self:ExpTime(pPlayer)
  local nBaiJuWanTime = self:GetBaiJuWanTime(pPlayer, nPayType)
  if nExpTime > nBaiJuWanTime then
    local nExtraExpTime = nExpTime - nBaiJuWanTime
    local nPerBaiJuWanTime, nPerBaiJuWanPrice, nItemId = self:GetPerBaiJuWanInfo(nPayType, pPlayer)
    local nHaveNum = pPlayer.GetItemCountByTypeInBag(nItemId)
    local nNeedNum = math.ceil(nExtraExpTime / nPerBaiJuWanTime)
    local nUseNum = nHaveNum > nNeedNum and nNeedNum or nHaveNum
    local nLackNum = nNeedNum - nHaveNum
    local szName = KItem.GetItemShowInfo(nItemId, pPlayer.nFaction, pPlayer.nFactionSect)
    szName = i18n.Parse(szName)
    local szColor = Ui.tbColor:GetColorHexBySettingId(502)
    if nLackNum > 0 then
      local nNeedPay = nLackNum * nPerBaiJuWanPrice
      if nHaveNum > 0 then
        szTip = i18n.EasyGet("\229\183\178\230\139\165\230\156\137<color=#{0}>{1}\228\184\170</color>\239\188\140\233\156\128\232\138\177\232\180\185<color=#{2}>{3}</color>\229\133\131\229\174\157\239\188\140\229\134\141\232\180\173\228\185\176<color=#{4}>{5}\228\184\170</color>{6}", szColor, nHaveNum, szColor, nNeedPay, szColor, nLackNum, szName)
      else
        szTip = i18n.EasyGet("\232\138\177\232\180\185<color=#{0}>{1}</color>\229\133\131\229\174\157\239\188\140\232\180\173\228\185\176<color=#{2}>{3}\228\184\170</color>{4}", szColor, nNeedPay, szColor, nLackNum, szName)
      end
    else
      szTip = i18n.EasyGet("\229\183\178\230\139\165\230\156\137<color=#{0}>{1}\228\184\170</color>{2}", szColor, nHaveNum, szName)
    end
  end
  return szTip
end

function OfflineExp:MessageBoxTips(pPlayer, nPayType)
  local szTip = ""
  local nExpTime = self:ExpTime(pPlayer)
  local nBaiJuWanTime = self:GetBaiJuWanTime(pPlayer, nPayType)
  if nExpTime > nBaiJuWanTime then
    local nExtraExpTime = nExpTime - nBaiJuWanTime
    local nPerBaiJuWanTime, nPerBaiJuWanPrice, nItemId = self:GetPerBaiJuWanInfo(nPayType, pPlayer)
    local nHaveNum = pPlayer.GetItemCountByTypeInBag(nItemId)
    local nNeedNum = math.ceil(nExtraExpTime / nPerBaiJuWanTime)
    local nUseNum = nHaveNum > nNeedNum and nNeedNum or nHaveNum
    local nLackNum = nNeedNum - nHaveNum
    if nLackNum <= 0 then
      return szTip
    else
      local nNeedPay = nLackNum * nPerBaiJuWanPrice
      local szColor = Ui.tbColor:GetColorHexBySettingId(502)
      local szNeedPay = i18n.EasyGet("\232\138\177\232\180\185<color=#{0}>{1}</color>\229\133\131\229\174\157\n\232\180\173\228\185\176<color=#{2}>{3}\228\184\170</color>{4}", szColor, nNeedPay, szColor, nLackNum, KItem.GetItemShowInfo(nItemId, pPlayer.nFaction, pPlayer.nFactionSect))
      szTip = szTip .. szNeedPay
    end
  end
  return szTip
end

function OfflineExp:CheckSpecialBaiJuWanIsOpen(pPlayer)
  local nVipLevel = pPlayer.GetVipLevel()
  return nVipLevel >= Vip:GetVipCanUseSpecialBaiJuWWanLevel()
end

function OfflineExp:GetBaiJuWanTime(pPlayer, nPayType)
  if nPayType == OfflineExp.OfflineExpType.Pay then
    return self:BaiJuWanTime(pPlayer)
  elseif nPayType == OfflineExp.OfflineExpType.SpecialPay then
    return self:SpecialBaiJuWanTime(pPlayer)
  end
  return 0
end

function OfflineExp:GetBaiJuWanSaveKey(nPayType)
  if nPayType == OfflineExp.OfflineExpType.Pay then
    return self.BaiJuWan_Time
  elseif nPayType == OfflineExp.OfflineExpType.SpecialPay then
    return self.SpecialBaiJuWan_Time
  end
end

function OfflineExp:GetPerBaiJuWanInfo(nPayType, pPlayer)
  if nPayType == OfflineExp.OfflineExpType.Pay then
    local tbWare = Shop:GetAWare2(Shop.ShopTypeStr.YuanBao, self.nBaiJuWanId)
    local nPrice = tbWare and Shop:GetPrice(tbWare, Shop.ShopTypeStr.YuanBao, pPlayer) or Item:GetClass("BaiJuWan").nPrice
    return Item:GetClass("BaiJuWan").nAddTime, nPrice, self.nBaiJuWanId
  elseif nPayType == OfflineExp.OfflineExpType.SpecialPay then
    local tbWare = Shop:GetAWare2(Shop.ShopTypeStr.YuanBao, self.nSpecialBaiJuWanId)
    local nPrice = tbWare and Shop:GetPrice(tbWare, Shop.ShopTypeStr.YuanBao, pPlayer) or Item:GetClass("SpecialBaiJuWan").nPrice
    return Item:GetClass("SpecialBaiJuWan").nAddTime, nPrice, self.nSpecialBaiJuWanId
  end
end
