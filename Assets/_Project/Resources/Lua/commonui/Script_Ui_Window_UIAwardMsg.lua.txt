local tbWnd = Ui:CreateWindow("UIAwardMsg")
local tbControls = {
  MsgTemplate = "Content/MsgTemplate/Msg/txtMsg",
  txtMsg = "Content/MsgTemplate/Msg/txtMsg",
  UIItemGrid = "Content/MsgTemplate/Msg/UIItemGrid"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbItemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  self.tbItemGrid:SetClickEnable(false)
end

function tbWnd:OnOpen(tbAward)
  self:UpdateView(tbAward)
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
  self.nTimer = nil
end

function tbWnd:UpdateView(tbAward)
  self.tbItemGrid:SetItemByTemplate(tbAward[2], tbAward[3])
  local szName = Item:GetName(tbAward[2], me)
  self.pPanel:Label_SetText(tbControls.txtMsg, szName)
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
  self.nTimer = Timer:Register(Env.GAME_FPS * 3, function()
    self:Close()
  end)
end
