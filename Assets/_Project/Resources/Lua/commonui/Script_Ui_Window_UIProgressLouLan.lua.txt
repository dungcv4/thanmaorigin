local tbWnd = Ui:CreateWindow("UIProgressLoulan")
local tbSetting = {
  [2] = "AwardEvent1",
  [3] = "AwardEvent2",
  [4] = "AwardEvent3",
  [5] = "AwardEvent4"
}
tbWnd.tbControls = {
  imgBar = "Cont/imgBar",
  ImgFire = "Cont/btnDay%d/Img_Fire",
  ImgGottenFire = "Cont/btnDay%d/Img_Lock",
  btnBox = "Cont/btnDay%d",
  txtNum = "Cont/txtNum",
  TrailEffect = "Cont/TrailEffect",
  txtStart = "Cont/Time",
  wndAwards = "wndDailyAwards",
  wndAwardsContent = "wndDailyAwards/imgBg",
  AwardItemTemp = "wndDailyAwards/imgBg/AwardList/Content/ItemTemplate"
}
tbWnd.tbOnClick = {}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_LOULANMIZONG_SYNC_BOSSSTART,
      self.UpdateTip
    }
  }
end

function tbWnd:OnCreate()
  self:SetClickEmptyToClose(self.wndAwards)
  self.tbAwardList = Ui:NewItemGridView(self, self.AwardItemTemp)
  self.pPanel:SetActive(self.txtStart, false)
  self.pPanel:Label_SetColorByName(self.txtNum, "White")
end

function tbWnd:OnOpen(nExplore)
  self:Update(nExplore)
  self.pPanel:SetActive(self.wndAwards, false)
end

function tbWnd:OnClose()
end

function tbWnd:Update(nExplore)
  nExplore = nExplore or 0
  self.pPanel:Label_SetText(self.txtNum, string.format("%s/%s", nExplore, LouLanMiZong.EXPLOREEVENT.AwardEvent4))
  local nBarRatio = math.min(1, nExplore / LouLanMiZong.EXPLOREEVENT.AwardEvent4)
  self.pPanel:Sprite_SetFill(self.imgBar, nBarRatio)
  local nBegin, nEnd = -232, 252
  self.pPanel:Object_SetLocalPosition(self.TrailEffect, Ui.Vector3(nBegin, -45, 0))
  local pDonkeyObj = self.pPanel:GetObject(self.TrailEffect)
  if not pDonkeyObj:GetComponent(Ui.UIDoTweenAnimation) then
    pDonkeyObj:AddComponent(Ui.UIDoTweenAnimation)
  end
  local effect = pDonkeyObj:GetComponent(Ui.UIDoTweenAnimation)
  local rt = self.pPanel:GetTransform(self.TrailEffect)
  effect:Init(rt, false, false)
  local tweener = effect:DOLocalMove(nBarRatio * (nEnd - nBegin) + nBegin, -45, 1, 0.3)
  self.pPanel:Object_SetLocalPosition(self.TrailEffect, Ui.Vector3(nBarRatio * (nEnd - nBegin) + nBegin, -45, 0))
  effect:Append(tweener)
  for i = 2, 5 do
    local bSatisfy = nExplore < LouLanMiZong.EXPLOREEVENT[tbSetting[i]]
    self.pPanel:SetActive(string.format(self.ImgFire, i), bSatisfy)
    self.pPanel:SetActive(string.format(self.ImgGottenFire, i), not bSatisfy)
  end
end

function tbWnd:UpdateTip(bShow)
  self.pPanel:SetActive(self.txtStart, bShow)
end
