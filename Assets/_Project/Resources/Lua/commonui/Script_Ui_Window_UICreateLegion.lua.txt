local tbWnd = Ui:CreateWindow("UICreateLegion")

function tbWnd:NotifyEvents()
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnOK = "node/imgBG/Content/btnOK",
  inputLegionNotice = "node/imgBG/Content/LegionNotice/inputLegionNotice",
  inputLegionName = "node/imgBG/Content/LegionName/inputLegionName",
  moneyCreate = "node/imgBG/Content/Cost/Create",
  imgJinbi = "node/imgBG/Content/Cost/Create/imgJinbi",
  txtCostNum = "node/imgBG/Content/Cost/Create/txtCostNum",
  ItemCreate = "node/imgBG/Content/Cost/CostCell",
  imgItem = "node/imgBG/Content/Cost/CostCell/imgJinbi",
  txtItemNum = "node/imgBG/Content/Cost/CostCell/txtReturnNum"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    self:OnClickOK()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self.pPanel:Input_SetText(self.inputLegionName, "")
  self.pPanel:Input_SetText(self.inputLegionNotice, "")
  self:UpdateCost()
end

function tbWnd:OnClickOK()
  local szLegionName = self.pPanel:Input_GetText(self.inputLegionName)
  local szLegionNotice = self.pPanel:Input_GetText(self.inputLegionNotice)
  szLegionNotice = szLegionNotice or ""
  local nRet = Legion:Create(szLegionName, szLegionNotice)
  if nRet then
    self:Close()
  end
end

function tbWnd:UpdateCost()
  local nItemCount = me.GetItemCountInBags(Legion.Cfg.nCreateLegionPriorityItem)
  if nItemCount > 0 then
    self.pPanel:SetActive(self.moneyCreate, false)
    self.pPanel:SetActive(self.ItemCreate, true)
    local szName, nIcon, nView, nQuality = KItem.GetItemShowInfo(Legion.Cfg.nCreateLegionPriorityItem, me.nFaction)
    local szIconAtlas, szIconSprite = Item:GetIcon(nIcon)
    local nNums = 1
    self.pPanel:Sprite_SetSprite(self.imgItem, szIconSprite)
    self.pPanel:Label_SetText(self.txtItemNum, nItemCount .. "/" .. nNums)
  else
    self.pPanel:SetActive(self.moneyCreate, true)
    self.pPanel:SetActive(self.ItemCreate, false)
    local szCurrency = "YuanBao"
    local nNums = Legion.Cfg.nCreateLegionNeedYuanBao
    local szIcon = Currency:GetIcon(szCurrency)
    self.pPanel:Sprite_SetSprite(self.imgJinbi, szIcon)
    self.pPanel:Label_SetText(self.txtCostNum, nNums)
  end
end
