  Empty4Raycast = function(self)
    self.pPanel:SetActive("Main", false)

function tbWnd:OnCreate()
  self.tbAttribScrollView = Ui:NewScrollView(self, self.goAttribTemplate, nil, false, true)
end

function tbWnd:UpdateView(pAsyncData)
  self.tbAttribScrollView:Clear()
  local tbSeriesList
  if pAsyncData then
    tbSeriesList = Item.EquipSeriesActive:StatSeriesByAsyncData(pAsyncData)
  else
    tbSeriesList = Item.EquipSeriesActive:StatSeries(me)
  end
  for nSeries = Item.emSERIES_TYPE.SERIES_NONE + 1, Item.emSERIES_TYPE.SERIES_NUM - 1 do
    local bActiveAttrib = false
    if tbSeriesList[nSeries] >= 2 then
      bActiveAttrib = true
    end
    self.pPanel:SetActive(string.format(self.imgUnActive, tbSeriesXiangSheng[nSeries]), true)
    self.pPanel:SetActive(string.format(self.imgUnactiveArrow, tbSeriesXiangSheng[nSeries]), true)
    self.pPanel:SetActive(string.format(self.imgActive, tbSeriesXiangSheng[nSeries]), bActiveAttrib)
    self.pPanel:SetActive(string.format(self.imgActiveArrow, tbSeriesXiangSheng[nSeries]), bActiveAttrib)
    local nSeriesLink = tbSeriesActiveLink[nSeries]
    self.tbAttribScrollView:SetElement(nSeriesLink, function(tbElement)
      self:SetAttribInfo(tbElement, nSeriesLink, tbSeriesList[nSeriesLink])
    end)
  end
end

function tbWnd:SetAttribInfo(tbElement, nSeriesIndex, nSeriesWeHave)
  local tbSeriesAttrib = Item.EquipSeriesActive.tbSeriesAttribList[nSeriesIndex]
  local szDescName, szDescValue = FightSkill:GetMagicDescSplit(tbSeriesAttrib[1], {
    tbSeriesAttrib[2],
    tbSeriesAttrib[3],
    tbSeriesAttrib[4]
  })
  tbElement.pPanel:Label_SetText("txtAttribType", tbSeriesDesc[nSeriesIndex](nSeriesWeHave, 2, tbSeriesXiangShengText[nSeriesIndex]()))
  tbElement.pPanel:Label_SetText("txtAttribValue", string.format("%s: %s", szDescName, szDescValue))
end
