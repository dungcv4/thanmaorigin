local tbWnd = Ui:CreateClass("UICP_StateBg", "UIPanel")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  imgNotStarted = "panel/img_NotStarted",
  imgBegin = "panel/img_Begin",
  imgEnd = "panel/img_End",
  imgInProgress = "panel/img_InProgress"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
end

function tbWnd:ShowNotStarted()
  self.pPanel:SetActive(tbControls.imgNotStarted, true)
  self.pPanel:SetActive(tbControls.imgBegin, false)
  self.pPanel:SetActive(tbControls.imgEnd, false)
  self.pPanel:SetActive(tbControls.imgInProgress, false)
end

function tbWnd:ShowBegin()
  self.pPanel:SetActive(tbControls.imgNotStarted, false)
  self.pPanel:SetActive(tbControls.imgBegin, true)
  self.pPanel:SetActive(tbControls.imgEnd, false)
  self.pPanel:SetActive(tbControls.imgInProgress, false)
end

function tbWnd:ShowEnd()
  self.pPanel:SetActive(tbControls.imgNotStarted, false)
  self.pPanel:SetActive(tbControls.imgBegin, false)
  self.pPanel:SetActive(tbControls.imgEnd, true)
  self.pPanel:SetActive(tbControls.imgInProgress, false)
end

function tbWnd:ShowInProgress()
  self.pPanel:SetActive(tbControls.imgNotStarted, false)
  self.pPanel:SetActive(tbControls.imgBegin, false)
  self.pPanel:SetActive(tbControls.imgEnd, false)
  self.pPanel:SetActive(tbControls.imgInProgress, true)
end

function tbWnd:HideAll()
  self.pPanel:SetActive(tbControls.imgNotStarted, false)
  self.pPanel:SetActive(tbControls.imgBegin, false)
  self.pPanel:SetActive(tbControls.imgEnd, false)
  self.pPanel:SetActive(tbControls.imgInProgress, false)
end

function tbWnd:OnOpen()
  self:HideAll()
end
