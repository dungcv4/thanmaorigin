function Profiler:BeginSample(szName)
  CS.UnityEngine.Profiling.Profiler.BeginSample(szName)
end

function Profiler:EndSample()
  CS.UnityEngine.Profiling.Profiler.EndSample()
end

function Profiler:UnpackReturnValue(...)
  self:EndSample()
  return ...
end

function Profiler:CollectTableMemory()
  local tbChecked = {}
  
  local function fnGetSize(tb)
    local nSize = debug.gettablesize(tb)
    for _, value in pairs(tb) do
      if type(value) == "table" and not tbChecked[value] then
        tbChecked[value] = true
        nSize = nSize + fnGetSize(value)
      end
    end
    return nSize
  end
  
  for n, v in pairs(_G) do
    if type(v) == "table" and not tbChecked[v] then
      local size = fnGetSize(v)
      tbChecked[v] = true
      CS.DebugModule.AddLuaTableData(n, size)
    end
  end
end
