    [1] = tbControls.btnCommonGM,
    [2] = tbControls.btnAllGM,
    [3] = tbControls.btnScriptGM,
    [4] = tbControls.btnTestGM,
    [5] = tbControls.btnSearchGM
  }, true, function(self, nIndex)
    if 1 == nIndex then
      self:SwitchPanel(self.panelCommonGM)
    elseif 2 == nIndex then
      self:SwitchPanel(self.panelAllGM)
    elseif 3 == nIndex then
      self:SwitchPanel(self.panelScriptGM)
      local file = io.open("./tempscript.lua", "rb")
      if file then
        local content = file:read("*a")

function tbWnd:OnSyncGMCMD(tbCMD)
  if tbCMD then
    self.SERVER_CMD = self.SERVER_CMD or {}
    table.insert(self.SERVER_CMD, tbCMD)
  end
  if self.nUpdateTimer then
    Timer:Close(self.nUpdateTimer)
    self.nUpdateTimer = nil
  end
  
  local function fnCall()
    self:UpdateCMDView()
  end
  
  self.nUpdateTimer = Timer:Register(9, fnCall)
end

function tbWnd:UpdateCMDView()
  local nCount = 1
  self.CMD = {}
  self.tbMainCmdScrollView:Clear()
  for i = 1, #self.CLIENT_CMD do
    self.tbMainCmdScrollView:SetElement(nCount, function(tbElement)
      tbElement.pPanel:Label_SetText("Text", self.CLIENT_CMD[i].MainName)
    end)
    self.CMD[nCount] = self.CLIENT_CMD[i]
    nCount = nCount + 1
  end
  if self.SERVER_CMD then
    for i = 1, #self.SERVER_CMD do
      self.tbMainCmdScrollView:SetElement(nCount, function(tbElement)
        tbElement.pPanel:Label_SetText("Text", self.SERVER_CMD[i].MainName)
      end)
      self.CMD[nCount] = self.SERVER_CMD[i]
      nCount = nCount + 1
    end
  end
end

function tbWnd:UpdateScheduleTaskViewList(szSearch)
  local tbInfoList = {}
  if not szSearch then
    tbInfoList = self.tbScheduleTaskInfo
  else
    for k, v in ipairs(self.tbScheduleTaskInfo) do
      if string.find(i18n.Parse(v._Desc), szSearch) then
        table.insert(tbInfoList, v)
      end
    end
  end
  self.tbScheduleTaskScrollView:Clear()
  for k, v in ipairs(tbInfoList) do
    self.tbScheduleTaskScrollView:SetElement(k, function(tbElement)
      tbElement.pPanel:Label_SetText("txtName", i18n.EasyGet("{0} (\229\143\130\230\149\176\239\188\154 {1})", i18n.Parse(v._Desc), v.Param))
      Ui:Button_BindEvent(tbElement.pPanel, "btnExecute", self.ExecuteScheduleTask, self, v.ScriptFunc, v.Param)
    end)
  end
end

  end

    self.tbFactionScrollView:SetElement(nFaction, function(tbElement)
      tbElement.pPanel:Label_SetText("txtFaction", Faction:GetName(nFaction))
      for j = 0, 1 do
        local btnName = string.format("btnSection%d", j)
        tbElement.pPanel:Button_SetText(btnName, Faction:GetSectName(nFaction, j), "Text")

      end
    end)
  end
  self.tbMainCmdScrollView:Clear()
  self:UpdateCMDView()
  self.tbCurrencyScrollView:Clear()
  local tbkey2type = {}
  for szCurrencyType, tbSettings in pairs(Currency.tbSetting) do
    if "TongFund" ~= szCurrencyType then
      tbkey2type[tbSettings.SaveKey] = szCurrencyType
    end
  end
  for key, szCurrencyType in ipairs(tbkey2type) do
    self.tbCurrencyScrollView:SetElement(key, function(tbElement)
      tbElement.pPanel:Input_SetText("InputField", "10000")
      local szCurrencyName = Currency:GetName(szCurrencyType)
      tbElement.pPanel:Label_SetText("txtFaction", szCurrencyName)
      tbElement.pPanel:Button_SetText("btnSection", "Add", "Text")
      Ui:Button_BindEvent(tbElement.pPanel, "btnSection", self.OnClickAddCurrency, self, tbElement, szCurrencyType)
    end)
  end
  self:InitSafeAreaInfo()
  self:InitScheduleTaskInfo()
  self:InitScheduleTaskView()
end

function tbWnd:SwitchPanel(pPanel)

    return
  end

  end

function tbWnd:InitCommandList(nIndex)
  if not self.CMD[nIndex] then
    return
  end
  local tbCmdList = self.CMD[nIndex].CmdList
  self.tbCommandScrollView:Reset()
  for i = 1, #tbCmdList do
    local tbData = tbCmdList[i]
    self.tbCommandScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbParams = tbData.Params or {}
      pPanel:Label_SetText("txtDesc", tbData.Name)
      pPanel:SetActive("btnExecute", true)
      for nId = 1, #tbParams do
        local szTextKey = string.format("txtParamTip%d", nId)
        pPanel:SetActive(szTextKey, true)
        pPanel:Label_SetText(szTextKey, tbParams[nId][1])
        local szInputKey = string.format("txtParamTip%d/inputParam%d", nId, nId)
        pPanel:Input_SetText(szInputKey, tbParams[nId][2])
      end
      for nId = #tbParams + 1, nCmdParamLimit do
        pPanel:SetActive("txtParamTip" .. nId, false)
      end
      Ui:Button_BindEvent(pPanel, "btnExecute", self.onClickCmd, self, tbData, pPanel)
    end)
  end
end

function tbWnd:OnSearchClick()
  local szText = self.pPanel:Input_GetText(tbControls.inputCustomCmd)
  if not szText or Lib:IsEmptyStr(szText) then
    return
  end
  self.tbSearchCmd = {}
  for _, tbCmdInfo in pairs(self.CMD) do
    local tbCmdList = tbCmdInfo.CmdList
    for _, tbCmdDetail in pairs(tbCmdList) do
      if string.find(tbCmdDetail.Name, szText) then
        local szMainName = tbCmdInfo.MainName
        self.tbSearchCmd[szMainName] = self.tbSearchCmd[szMainName] or {}
        table.insert(self.tbSearchCmd[szMainName], tbCmdDetail)
      end
    end
  end
  self.pPanel:SetActive(tbControls.btnSearchTab, true)
  self.pPanel:Button_SetEnable(tbControls.btnSearchTab, false)
  self.tbMainCmdScrollView:ClearSelected()
  self.tbCommandScrollView:Reset()
  if 0 == Lib:CountTB(self.tbSearchCmd) then
    me.CenterMsg("not found")
    return
  end
  
  local function fnSetCMD(tbData, nIndex, bNotDesc)
    self.tbCommandScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbParams = tbData.Params or {}
      pPanel:Label_SetText("txtDesc", tbData.Name)
      for nId = 1, nCmdParamLimit do
        pPanel:SetActive("txtParamTip" .. nId, bNotDesc)
      end
      pPanel:SetActive("btnExecute", bNotDesc)
      for nId = 1, #tbParams do
        local szTextKey = string.format("txtParamTip%d", nId)
        pPanel:SetActive(szTextKey, true)
        pPanel:Label_SetText(szTextKey, tbParams[nId][1])
        local szInputKey = string.format("txtParamTip%d/inputParam%d", nId, nId)
        pPanel:Input_SetText(szInputKey, tbParams[nId][2])
      end
      for nId = #tbParams + 1, nCmdParamLimit do
        pPanel:SetActive("txtParamTip" .. nId, false)
      end
      Ui:Button_BindEvent(pPanel, "btnExecute", self.onClickCmd, self, tbData, pPanel)
    end)
  end
  
  local nCount = 1
  for szMainName, tbData in pairs(self.tbSearchCmd) do
    fnSetCMD({
      Name = string.format("<color=#2FF666>Group\239\188\154%s</color>", szMainName)
    }, nCount)
    nCount = nCount + 1
    for i = 1, Lib:CountTB(tbData) do
      fnSetCMD(tbData[i], nCount, true)
      nCount = nCount + 1
    end
  end
end

function tbWnd:onClickCmd(tbCmd, pPanel)
  local tbParams = tbCmd.Params or {}
  local tbInputs = {}
  for nId = 1, #tbParams do
    table.insert(tbInputs, pPanel:Input_GetText(string.format("txtParamTip%d/inputParam%d", nId, nId)))
  end
  local szCommand = ""
  if tbCmd.StrFunc ~= nil then
    szCommand = tbCmd.StrFunc(table.unpack(tbInputs))
  elseif nil == tbCmd.ExecFunc then
    szCommand = string.format(tbCmd.Cmd, table.unpack(tbInputs))
  else
    tbCmd.ExecFunc(table.unpack(tbInputs))
    me.CenterMsg(string.format("execute [%s] [%s]", tbCmd.Name, table.concat(tbInputs, ",")))
  end
  if "" ~= szCommand then
    local _, _, szTarget, szCmd = string.find(szCommand, "^([/?][?wzc]+)%s*(.*)")
    print(szTarget, szCmd)
    if "" == szCmd then
      return szTarget
    end
    if "??" == szTarget then
      local fnCmd, szMsg = load(szCmd, "[GmCmd]")
      if not fnCmd then
        Log("Error" .. szMsg)
        error("Do GmCmd failed:" .. szMsg)
        return
      end
      local bRet, nRetCode = pcall(fnCmd)
      if not bRet then
        print(nRetCode)
        local szInfo = debug.traceback()
        print(szInfo)
        return
      end
      return
    elseif "/?" == szTarget then
      GMCommand(szCmd)
    elseif "/w" == szTarget then
      GMCommand(szCmd, 1)
    elseif "/z" == szTarget then
      GMCommand(szCmd, 2, 1)
    elseif "/c" == szTarget then
      GMCommand(szCmd, 2, 2)
    end

function tbWnd:OnItemSearchClick()
  local sInput = self.pPanel:Input_GetText(tbControls.itemSearchInputText)
  if nil == sInput or Lib:IsEmptyStr(sInput) then
    return
  end
  if nil == self.tbItems then
    self:LoadItemAssets()
  end
  local tbSearchResult = self:SearchItem(sInput)
  self.tbItemScrollView:ClearSelected()
  self.tbItemScrollView:Reset()
  if 0 == #tbSearchResult then
    me.CenterMsg("not found")
    return
  end
  for i = 1, #tbSearchResult do
    local tbData = tbSearchResult[i]
    self.tbItemScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtDesc", i18n.Parse(tbData.Name))
      pPanel:SetActive("txtParamTip", true)
      pPanel:SetActive("btnExecute", true)
      pPanel:Label_SetText("txtParamTip", i18n.EasyGet("\230\149\176\233\135\143"))
      pPanel:Input_SetText("InputField", "1")
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(tbData.TemplateId)
      itemGrid:SetClickEnable(true)
      Ui:Button_BindEvent(pPanel, "btnExecute", function()
        local nAddNumStr = pPanel:Input_GetText("InputField")
        local num = tonumber(nAddNumStr)
        if type(num) == "number" then
          local sAddItemCommand = string.format("GM:AddItem(me, %d, %d)", tbData.TemplateId, math.floor(num))
          GMCommand(sAddItemCommand)
          me.CenterMsg(string.format("execute %s", sAddItemCommand))
        end
      end)
    end)
  end
end

function tbWnd:SearchItem(str)
  local tbSearchResult = {}
  local nameMatchCount = 0
  for _, value in pairs(self.tbItems) do
    if string.find(i18n.Parse(value.Name), str) ~= nil then
      nameMatchCount = nameMatchCount + 1
      table.insert(tbSearchResult, nameMatchCount, value)
    elseif nil ~= string.find(i18n.Parse(value.Intro), str) then
      table.insert(tbSearchResult, value)
    end
  end
  return tbSearchResult
end

function tbWnd:InitScheduleTaskInfo()
  local tbKey = {
    "ScriptFunc",
    "_Desc",
    "Param"
  }
  local szFile = "Setting/ScheduleTask.tab"
  self.tbScheduleTaskInfo = {}
  local tbFilter = {}
  local tbScheduleSetting = Lib:EasyLoadTabFile(szFile, nil, tbKey)
  for k, v in ipairs(tbScheduleSetting) do
    if not tbFilter[v.ScriptFunc] then
      tbFilter[v.ScriptFunc] = {}
      tbFilter[v.ScriptFunc][v.Param] = 1
      table.insert(self.tbScheduleTaskInfo, v)
    elseif not tbFilter[v.ScriptFunc][v.Param] then
      tbFilter[v.ScriptFunc][v.Param] = 1
      table.insert(self.tbScheduleTaskInfo, v)
    end
  end
end
