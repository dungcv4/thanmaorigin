function Legion:CheckMyAuthority(authority)
  local nCareer
  if Legion:AmIMaster() then
    nCareer = Legion.Def.Career_Master
  else
    nCareer = self:GetMyCareer()
  end
  if not nCareer then
    return
  end
  local careerAuthority = Legion.Def.Career_Authority[nCareer] or {}
  if careerAuthority[authority] then
    return true
  end
  return false
end

function Legion:CheckCanKickOut(nCareer)
  if not Legion:CheckMyAuthority(Legion.Def.Authority_KickOut) then
    return false

    return false
  end
  local nMyCareer = self:GetMyCareer()
  if nMyCareer ~= Legion.Def.Career_Master and nCareer < Legion.Def.Career_Normal then

function Legion:GetMemberCareerInfo(nCareer)
  local tbMemberList = self:GetMemberList() or {}
  
  local function fnEqual(tbMember, nCareer)
    return tbMember.nCareer == nCareer
  end
  
  local nCount = 0
  local tbCareerPos = {}
  for _, tbItem in pairs(tbMemberList) do
    if fnEqual(tbItem, nCareer) then
      nCount = nCount + 1
      if tbItem.nCareerPos then
        tbCareerPos[tbItem.nCareerPos] = tbItem.nMemberId
      end
    end
  end
  return nCount, tbCareerPos
end

function Legion:InsertClientMember(tbCacheMember, tbCacheMap, tbMember)
  if tbCacheMap[tbMember.nMemberId] then
    for k, v in pairs(tbCacheMember) do
      if v.nMemberId == tbMember.nMemberId then
        table.remove(tbCacheMember, k)
        break
      end
    end
  end
  table.insert(tbCacheMember, tbMember)
  tbCacheMap[tbMember.nMemberId] = tbMember
end

function Legion:DeleteClientMember(tbCacheMember, tbCacheMap, tbMember)
  if tbCacheMap[tbMember.nMemberId] then
    for k, v in pairs(tbCacheMember) do
      if v.nMemberId == tbMember.nMemberId then
        table.remove(tbCacheMember, k)
        break
      end
    end
  end
  tbCacheMap[tbMember.nMemberId] = nil
end

function Legion:OnSyncMemberList(tbMemberList)
  local tbCacheMember, nVersion, tbCacheMap = self:GetMemberList()
  tbCacheMember = tbCacheMember or {}
  nVersion = nVersion or 0
  tbCacheMap = tbCacheMap or {}
  for _, tbMember in pairs(tbMemberList) do
    if not tbMember.bDelete then
      self:InsertClientMember(tbCacheMember, tbCacheMap, tbMember)
    else
      self:DeleteClientMember(tbCacheMember, tbCacheMap, tbMember)
    end
  end
  Legion:CacheData("MemberList", tbCacheMember, nVersion)
  Legion:CacheData("MemberMap", tbCacheMap, 0)
end

function Legion:OnSyncMemberListEnd(nSyncVersion)
  local tbCacheMember, _ = self:GetMemberList()
  Legion:CacheData("MemberList", tbCacheMember, nSyncVersion)
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_LEGION_BASEINFO, "MemberList")
end

    me.CenterMsg(i18n.Get(Legion.Cfg.Msg.MailCountLimit))

local SortFunction = {
  Default = function(tbMemberList, bIsDown)
    table.sort(tbMemberList, function(a, b)
      if a.nMemberId == me.dwID then
        return true
      end
      if b.nMemberId == me.dwID then
        return false
      end
      local bOnlineA = 1 == a.nIsOnLine
      local bOnlineB = 1 == b.nIsOnLine
      if bOnlineA == bOnlineB then
        if a.nCareer == b.nCareer then
          return a.nLevel > b.nLevel
        else
          return a.nCareer < b.nCareer

function Legion:GetErrorMsg(MsgCode)
  local szMsg
  if type(MsgCode) == "table" then
    MsgCode[1] = tbMsgCode2Msg[MsgCode[1]]
    if MsgCode[1] then
      if type(MsgCode[1]) == "function" then
        szMsg = MsgCode[1](table.unpack(MsgCode, 2))
      else
        szMsg = i18n.Get(table.unpack(MsgCode))
      end
    else
      error("Legion:OnServerResult  Unknow MsgCode")
    end
  else
    szMsg = tbMsgCode2Msg[MsgCode]
    if szMsg then
      szMsg = i18n.Get(szMsg)
    else
      error("Legion:OnServerResult  Unknow MsgCode")
    end
  end
  return szMsg
end

function Legion:OnServerResult(szRequestType, bSuccess, MsgCode)
  if MsgCode then
    local szMsg = Legion:GetErrorMsg(MsgCode)
    if szMsg then
      me.CenterMsg(szMsg)
    end
  end
  if bSuccess and "Apply" == szRequestType then
    EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_LEGION_BASEINFO, szRequestType)
  end
end

function Legion:IsNeedLightPointByApplyNewMember()
  local emCareer = Legion:GetMyCareer()
  if not emCareer then
    return false
  end
  if emCareer == Legion.Def.Career_Master or emCareer == Legion.Def.Career_ViceMaster then
    return true
  end
  return false
end

function Legion:GetMaxLegionOfficialLevel()
  for i = 1, #self.Def.OfficialMaxLevelByTimeFrame do
    if not GlobalDomainBattle.szZoneTimeFrame then
      return #self.Def.OfficialTable
    end
    if Lib:CalcTimeFrameOpenDay(GlobalDomainBattle.szZoneTimeFrame) >= Lib:CalcTimeFrameOpenDay(self.Def.OfficialMaxLevelByTimeFrame[i].TimeFrame) then
      return self.Def.OfficialMaxLevelByTimeFrame[i].MaxLevel
    end
  end
  return 1
end
