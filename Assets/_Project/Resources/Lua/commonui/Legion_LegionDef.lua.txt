function Legion:GetMaxLegionOfficialLevel()
  for i = 1, #self.Def.OfficialMaxLevelByTimeFrame do
    if 1 == GetTimeFrameState(self.Def.OfficialMaxLevelByTimeFrame[i].TimeFrame) then
      return self.Def.OfficialMaxLevelByTimeFrame[i].MaxLevel
    end
  end
  return 1
end

function Legion:GetBestWarZoneType()
  return 1
end

function Legion:GetWarZoneTypeByPlayerId(nPlayerId)
  local tbPlayerData = self.tbScriptPlayerDataList
  if not tbPlayerData or not tbPlayerData[nPlayerId] then
    if self.tbWarZoneList and 1 == #self.tbWarZoneList then
      return self:GetBestWarZoneType()
    end
    local nBestWarZoneType = self:GetBestWarZoneType()
    if not (self.tbWarZoneList and self.tbWarZoneList[nBestWarZoneType]) or not self.tbWarZoneList[nBestWarZoneType].tbLegions then
      return 0
    end
    if Lib:IsEmptyTB(self.tbWarZoneList) or Lib:IsEmptyTB(self.tbWarZoneList[nBestWarZoneType].tbLegions) then
      return 0
    end
    local nTotal = #self.tbWarZoneList[nBestWarZoneType].tbLegions
    local nLastArgFightPower = self.tbWarZoneList[nBestWarZoneType].tbLegions[nTotal].nArgFightPower
    local nPlayerAvgFightPower = FightPower:GetPlayerAverageFightPower(nPlayerId)
    if SERVER_RUNNING_MODE == SERVER_MODE.ZONE_SERVER then
      nPlayerAvgFightPower = Legion.tbAvgFightPower[nPlayerId] or 0
    end
    if nLastArgFightPower <= nPlayerAvgFightPower then
      return nBestWarZoneType
    end
    return 0
  end
  return tbPlayerData[nPlayerId].nLegionType
end

function Legion:GetWarZoneType(nLegionId)
  for nWarZoneType, tbList in pairs(self.tbWarZoneList or {}) do
    for nIndex, tbInfo in pairs(tbList.tbLegions) do
      if tbInfo.nLegionId == nLegionId then
        return nWarZoneType
      end
    end
  end
end

function Legion:GetCreateWarZoneType(nPlayerId)
  local nWarZoneType = Legion:GetWarZoneTypeByPlayerId(nPlayerId)
  if nWarZoneType and 0 ~= nWarZoneType then
    return nWarZoneType
  end
  return #self.tbWarZoneList
end
