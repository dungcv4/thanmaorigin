function tbWnd:GoFunction(nIndex)
  if not self.nSelectIndex then
    return
  end
  if not self.tbCurListItem then
    return
  end
  local tbInfo = self.tbCurListItem[self.nSelectIndex]
  if not tbInfo then
    return
  end
  local nPosID = tbInfo.tbPosIDs[nIndex]
  local tbPosInfo = tbInfo.tbMapTIDs[nPosID]
  local nMapTID = tonumber(tbPosInfo.nMapID)
  local nMapType = Map:GetMapType(nMapTID)
  local nX = tonumber(tbPosInfo.tbPos[1][1])
  local nY = tonumber(tbPosInfo.tbPos[1][2])
  WuLinGaoShou:ClientCanEnterWuLinGaoShou(self.szType, nMapTID, nX, nY)
  Ui:CloseWindow("UIActivity")
  self:Close()
end

function tbWnd:OnCreate()

  table.insert(self.tbGoButtonGroup, 3, self.btnGo3)
  table.insert(self.tbGoButtonGroup, 4, self.btnGo4)
  table.insert(self.tbGoButtonGroup, 5, self.btnGo5)
  table.insert(self.tbGoButtonGroup, 6, self.btnGo6)
  for i = 1, #self.tbGoButtonGroup do
    self.pPanel:SetActive(self.tbGoButtonGroup[i], false)
  end
  self.tbGoTextGroup = {}
  table.insert(self.tbGoTextGroup, 1, self.txtGo1)
  table.insert(self.tbGoTextGroup, 2, self.txtGo2)
  table.insert(self.tbGoTextGroup, 3, self.txtGo3)
  table.insert(self.tbGoTextGroup, 4, self.txtGo4)
  table.insert(self.tbGoTextGroup, 5, self.txtGo5)
  table.insert(self.tbGoTextGroup, 6, self.txtGo6)
  self.tbBossScrollView = Ui:NewScrollView(self, self.BossTemplate, function(self, nIndex, tbElement)
    self.nSelectIndex = nIndex
    WuLinGaoShou.nSelectIndex = nIndex
    self:UpdateListItemMap(self.tbListItem[nIndex])
  end)
  self.nSelectIndex = 1
  self.tbItemView = Ui:NewItemGridView(self, self.ItemTemplate)
end

function tbWnd:OnOpen(szType)
  self.szType = szType or "Boss"
  self.nCalendarID = tbActivityID[szType]
  local tbMapNpcGroup = WuLinGaoShou:GetTimeFrameNpcGroup()
  if not tbMapNpcGroup then
    me.CenterMsg(i18n.EasyGet("\229\176\154\230\156\170\229\188\128\229\144\175"))
    return
  end
  local szUiName = tbTitleName[self.szType]()
  self.pPanel:Label_SetText(self.txtTitle, szUiName)
  self:UpdateSyncData()
end

function tbWnd:SetBossItem(tbElement, tbInfo)
  local tbSyncData = WuLinGaoShou:GetSyncData(self.szType)
  local pPanel = tbElement.pPanel
  local nSeriesType = tbInfo.nSeries
  pPanel:Label_SetText("txtName", i18n.Parse(tbInfo.szNpcName))
  pPanel:SetActive("Series/" .. tbSeriesItem[1], false)
  pPanel:SetActive("Series/" .. tbSeriesItem[nSeriesType], true)
  if tbSyncData.szCurrentState == "Start" then
    pPanel:SetActive("imgClosed", false)
    pPanel:Label_SetText("imgRefresh/Text", i18n.EasyGet("\229\183\178\229\136\183\230\150\176"))
    pPanel:SetActive("imgRefresh", true)
    local tbAllBossInfo = tbSyncData[10][tbInfo.nShowNpcID]
    if nil ~= tbAllBossInfo then
      local alldead = true
      for key, value in pairs(tbAllBossInfo) do
        if 0 == value.IsDead then
          alldead = false
        end
      end
      if alldead then
        pPanel:Label_SetText("imgClosed/Text", i18n.EasyGet("\229\183\178\229\135\187\233\128\128"))
        pPanel:SetActive("imgRefresh", false)
        pPanel:SetActive("imgClosed", true)
      end
    end
  elseif tbSyncData.szCurrentState == "NotStart" then
    pPanel:Label_SetText("imgClosed/Text", i18n.EasyGet("\230\156\170\229\136\183\230\150\176"))
    pPanel:SetActive("imgRefresh", false)
    pPanel:SetActive("imgClosed", true)
  else
    pPanel:Label_SetText("imgClosed/Text", i18n.EasyGet("\229\183\178\231\187\147\230\157\159"))
    pPanel:SetActive("imgRefresh", false)
    pPanel:SetActive("imgClosed", true)
  end
end

function tbWnd:UpdateListLeft(tbListItem)
  if not tbListItem then
    return
  end
  self.tbBossScrollView:Reset()
  self.tbListItem = tbListItem
  for i = 1, #tbListItem do
    self.tbBossScrollView:SetElement(i, function(tbElement)
      self:SetBossItem(tbElement, tbListItem[i])
    end)
  end
  self.nSelectIndex = WuLinGaoShou.nSelectIndex
  self.tbBossScrollView:SelectElement(WuLinGaoShou.nSelectIndex)
end

function tbWnd:GetOrgShowInfo()
  local tbMapNpcGroup, tbBossData = self:GetSyncMapInfo()
  local tbListItem = {}
  for _, tbAllMapInfo in pairs(tbMapNpcGroup) do
    for _, tbInfo in pairs(tbAllMapInfo) do
      local nNpcTID = tbInfo.nBossId
      local tbSubInfo = {}
      tbSubInfo.nNpcTID = nNpcTID
      tbSubInfo.tbPosIDs = tbInfo.tbPosIDs
      tbSubInfo.tbMapTIDs = tbInfo.tbMapTIDs
      tbSubInfo.nNpcCount = WuLinGaoShou:GetCount(tbInfo, tbBossData)
      tbSubInfo.nNpcLevel = tbInfo.nNpcLevel
      tbSubInfo.nShowNpcID = tbInfo.nShowNpcID
      tbSubInfo.nSeries = tbInfo.nSeries
      tbSubInfo.szNpcName = KNpc.GetNameByTemplateId(tbInfo.nShowNpcID)
      tbSubInfo.tbShowAward = Award:GetAllAwardNoRand(WuLinGaoShou.SHOW_AWARD_ID, tbInfo.nShowAwardID)
      tbSubInfo.nSort = tbInfo.nSort
      table.insert(tbListItem, tbSubInfo)
    end
  end
  table.sort(tbListItem, function(a, b)
    return a.nSort < b.nSort
  end)
  return tbListItem
end

function tbWnd:GetSyncMapBianYi(nMapTID)
  local tbSyncData = WuLinGaoShou:GetSyncData(self.szType)
  if not tbSyncData[5] or not tbSyncData[5][nMapTID] then
    return true
  end
  return false
end

function tbWnd:IsPeaceMap(nMapTID)
  local tbSyncData = WuLinGaoShou:GetSyncData(self.szType)
  if tbSyncData[7] and tbSyncData[7][nMapTID] then
    return true
  end
  return false
end

function tbWnd:GetSyncState(tbTime)
  local tbSyncData = WuLinGaoShou:GetSyncData(self.szType)
  if not tbSyncData[4] then
    return
  end
  local nLocalDay = Lib:GetLocalDay(tbSyncData[4])
  if nLocalDay ~= Lib:GetLocalDay() then
    return
  end
  local nDaySec = Lib:GetTodaySec(tbSyncData[4])
  if nDaySec < tbTime[1] or nDaySec > tbTime[2] then
    return
  end
  return tbSyncData[3]
end

function tbWnd:GetNpcDeathInfo(nMapTID)
  local tbSyncData = WuLinGaoShou:GetSyncData(self.szType)
  local tbNpcDeath = tbSyncData[2]
  if not tbNpcDeath or not tbNpcDeath[nMapTID] then
    return {
      trueBoss = {},
      fakeBoss = {}
    }
  end
  return tbNpcDeath[nMapTID]
end

  end
  local nCount = 0
  for _, nCurNpcTID in pairs(tbSyncData[1][nMapTID]) do
    if nNpcTID then
      if nCurNpcTID == nNpcTID then
        nCount = nCount + 1
      end
    else
      nCount = nCount + 1
    end
  end
  return nCount
end

function tbWnd:IsOpenTime(tbTime)
  local nCurTime = Lib:GetTodaySec()
  if nCurTime >= tbTime[1] and nCurTime <= tbTime[2] then
    return true
  end
  return false
end

function tbWnd:GetActivityState(tbTime)
  local nCurTime = Lib:GetTodaySec()
  if nCurTime >= tbTime[1] and nCurTime < tbTime[2] then
    local nState = self:GetSyncState(tbTime)
    if 1 == nState then
      return "Start"
    elseif 2 == nState then
      return "End"
    end
    return "NotStart"
  end
  if nCurTime < tbTime[1] then
    return "NotStart"
  end
  if nCurTime > tbTime[2] then
    return "End"
  end
end

function tbWnd:GetSyncMapInfo()
  local tbSyncData = WuLinGaoShou:GetSyncData(self.szType)
  if not tbSyncData[6] then
    return {}
  end
  local tbMapNpcGroup = WuLinGaoShou:GetTimeFrameNpcGroup()
  if not tbMapNpcGroup then
    return {}
  end
  local tbCreateMapData = {}
  local nBossGroup = tbSyncData[9]
  for nTId, tbData in pairs(tbSyncData[6]) do
    local tbMapTIDs = tbData.tbMapTIDs
    local nShowNpcID = tbData.nShowNpcID
    local tbMapData = tbMapNpcGroup[nShowNpcID][nBossGroup]
    if not tbCreateMapData[nShowNpcID] then
      tbCreateMapData[nShowNpcID] = tbMapData
    end
  end
  return tbCreateMapData, tbSyncData[6]
end

function tbWnd:UpdateSyncData()
  self.tbCurListItem = self:GetOrgShowInfo()
  self:UpdateListLeft(self.tbCurListItem)
end
