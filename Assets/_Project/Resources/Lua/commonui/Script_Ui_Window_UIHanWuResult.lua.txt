local tbWnd = Ui:CreateWindow("UIHanWuResult")
local tbControls = {
  goVictory = "imgBG/Victory",
  goDefeat = "imgBG/Defeat",
  txtTips = "imgBG/txtTips",
  goPlayerTemplate = "imgBG/PlayerList/PlayerTemplate"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.tbPlayerScrollView = Ui:NewScrollView(self, tbControls.goPlayerTemplate)
end

function tbWnd:OnOpen(nPhase, bWin, tbKillNumList, nAwardKey)
  self:ShowResult(nPhase, bWin, tbKillNumList, nAwardKey)
end

function tbWnd:ShowResult(nPhase, bWin, tbKillNumList, nAwardKey)
  self.pPanel:SetActive(tbControls.goVictory, bWin)
  self.pPanel:SetActive(tbControls.goDefeat, not bWin)
  self.pPanel:SetActive(tbControls.txtTips, bWin)
  self.tbPlayerScrollView:Reset()
  local tbMemberInfo, szName, nLevel, nPortrait, nSex, nPortraitFrame, tbFactionInfo, nFaction, nFactionSect
  local nCnt = 0
  for nMemberId, nKillNum in pairs(tbKillNumList) do
    nCnt = nCnt + 1
    if me.dwID == nMemberId then
      szName, nLevel, nPortrait, nSex = me.szName, me.nLevel, me.nPortrait, me.nSex
      nFaction, nFactionSect = me.nFaction, me.nFactionSect
    else
      tbMemberInfo = HanWuYiJi.tbMemberList[nMemberId]
      if tbMemberInfo then
        szName, nLevel, nPortrait, nSex = tbMemberInfo.szName, tbMemberInfo.nLevel, tbMemberInfo.nPortrait, tbMemberInfo.nSex
        nPortraitFrame = tbMemberInfo.nPortraitFrame
      else
        szName, nLevel, nPortrait, nSex = nil, nil, nil, nil
        nPortraitFrame = nil
      end
      tbFactionInfo = HanWuYiJi.tbSelFactionList[nMemberId]
      if tbFactionInfo then
        nFaction, nFactionSect = tbFactionInfo.nFaction, tbFactionInfo.nFactionSect
      else
        nFaction, nFactionSect = nil, nil
      end
    end
    self.tbPlayerScrollView:SetElement(nCnt, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtPlayerName", szName or "")
      pPanel:Label_SetText("PortraitObj/Level/Text", nLevel or "")
      pPanel:Sprite_SetSprite("PortraitObj/Portrait/Image", nPortrait and PlayerPortrait:GetIcon(nPortrait) or "")
      PlayerPortraitFrame:SetPortraitFrame(pPanel, "PortraitObj/PortraitFrame", nPortraitFrame)
      if nFaction then
        pPanel:Sprite_SetSprite("PortraitObj/imgFactionIcon", Faction:GetIcon(nFaction))
      else
        pPanel:Sprite_SetSprite("PortraitObj/imgFactionIcon", "")
      end
      pPanel:Label_SetText("txtStage", i18n.EasyGet("\231\172\172{0}\233\152\182\230\174\181", nPhase))
      pPanel:Label_SetText("txtKillCount", nKillNum)
      pPanel:Label_SetText("txtGrade", HanWuYiJi.tbGradeNameFun[nAwardKey]())
    end)
  end
end
