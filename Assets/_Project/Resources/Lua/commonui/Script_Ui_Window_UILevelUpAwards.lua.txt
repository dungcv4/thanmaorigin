local tbWnd = Ui:CreateWindow("UILevelUpAwards")
local tbControls = {
  btnClose = "node/imgBg/btnClose",
  btnJinjie = "node/imgBg/content/btnJinjie",
  txtLevel = "node/imgBg/content/Awards/imgUpBg/txtLevel",
  txtExplain = "node/imgBg/content/Awards/imgUpBg/txtExplain",
  ItemTemplate = "node/imgBg/content/Awards/imgAwardBg/AwardsList/itemTemplate",
  txtTitle = "node/imgBg/txtTitle",
  txtUp = "node/imgBg/content/Awards/imgUpBg/imgTitleBg/txtUp"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnJinjie = function(self)
    LevelUpAwards:OnChargeLevelUpAwardsReq()
  end
}
local ITEM_PER_LINE = 1
local MIN_LINE = 1

function tbWnd:OnCreate()
  self.tbItemScrollView = Ui:NewItemGridScrollView(self, tbControls.ItemTemplate, ITEM_PER_LINE, MIN_LINE, "LevelUpAwards")
  self.tbItemScrollView:Clean()
end

function tbWnd:OnOpen()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.LevelUpAwards)
  self:UpdateView()
end

function tbWnd:UpdateView()
  local tbSetting = LevelUpAwards:GetAwardSetting()
  if not tbSetting then
    return
  end
  local nLevel = tbSetting.nDirectLevel
  if tbSetting.nDirectLevel >= GameSetting.MAX_PLAYER_LEVEL then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\229\174\151\229\184\136\231\155\180\229\141\135"))
    self.pPanel:Label_SetText(tbControls.txtUp, i18n.EasyGet("\231\155\180\229\141\135\229\174\151\229\184\136\231\173\137\231\186\167"))
    nLevel = tbSetting.nDirectLevel - GameSetting.MAX_PLAYER_LEVEL
  else
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\231\173\137\231\186\167\231\155\180\229\141\135"))
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\231\155\180\229\141\135"))
  end
  self.pPanel:Label_SetText(tbControls.txtLevel, i18n.EasyGet("{0}", nLevel))
  self.pPanel:Label_SetText(tbControls.txtExplain, i18n.EasyGet("\232\180\173\228\185\176\229\144\142\231\173\137\231\186\167\231\155\180\229\141\135\229\136\176{0}\231\186\167", Lib:GetLevelText(tbSetting.nDirectLevel)))
  local bCanBuy = LevelUpAwards:CanBuyAwards(me)
  self.pPanel:Button_SetEnable(tbControls.btnJinjie, bCanBuy)
  local szButton = i18n.EasyGet("\232\180\173\228\185\176")
  local nProductId = LevelUpAwards:GetProductId()
  if nProductId then
    local tbProductSetting = Recharge:GetProductSettingById(nProductId)
    local szShowPrice = Recharge:GetShowPrice(tbProductSetting)
    szButton = szShowPrice
  end
  local nBuyLevel = me.GetUserValue(LevelUpAwards.SAVE_GROUP, LevelUpAwards.SAVE_BUY_LEVEL)
  if nBuyLevel >= tbSetting.nOpenLevel then
    szButton = i18n.EasyGet("\229\183\178\232\180\173\228\185\176")
  end
  self.pPanel:Button_SetText(tbControls.btnJinjie, szButton, "Text")
  local tbAwards = Award:GetAward(tbSetting.nAwardId, tbSetting.nAwardKey)
  self.tbItemScrollView:UpdateGenericItem(tbAwards, true)
end
