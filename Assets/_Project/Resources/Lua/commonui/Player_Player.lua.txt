local function AddItemUnsafe(pPlayer, nLogReason, nLogReason2, szMsg, szType, nItemTemplateId, nCount, tbItemExtParams)
  tbItemExtParams = tbItemExtParams or {}
  nLogReason2 = nLogReason2 or 0
  if not tbItemExtParams.nTradeLimitType then
    local nBindType = tbItemExtParams.nBindType or Item.eBINDTYPE.BIND
    if nBindType == Item.eBINDTYPE.BIND or nBindType == Item.eBINDTYPE.MONTHCARD_UNBIND and not WeekMonthCard:CheckMonthCardState(pPlayer) then
      tbItemExtParams.nTradeLimitType = Item.eTRADELIMITTYPE.NONE
      tbItemExtParams.nTradeLimitValue = 0
    end
  end
  local bAutoUse, tbAwards = Item:AutoUseRandItem(pPlayer, nItemTemplateId, nCount, szMsg, tbItemExtParams, nLogReason, nLogReason2)
  if bAutoUse then
    return tbAwards
  else
    local pItem = pPlayer.AddItemEx_Unsafe(nItemTemplateId, nCount, tbItemExtParams, nLogReason, nLogReason2)
  end
end

local function AddMoney(pPlayer, nLogReason, nLogReason2, szMsg, szType, nCount, nUselessParam)
  if not nCount or 0 == nCount then
    LogErr("AwardFunc Err AddMoney Fail", nLogReason, nUselessParam, nCount)
    return false
  end
  if not szType or not Currency:Valid(szType) then
    Log("_LuaPlayer.AddMoney fail !! unknow money type ", pPlayer.szName, pPlayer.dwID, szType, nCount)
  elseif nLogReason == Env.LogWay_RandomItem then
    pPlayer.AddMoney(szType, nCount, nLogReason, nLogReason2, nLogReason2, 1)
  else
    pPlayer.AddMoney(szType, nCount, nLogReason, nLogReason2)
  end
end

local function AddExp(pPlayer, nLogReason, nLogReason2, szMsg, szType, nCount, nUselessParam)
  if not nCount or nCount <= 0 then
    LogErr("AwardFunc Err AddExp Fail", nLogReason, nUselessParam, nCount)
    return false
  end
  return pPlayer.AddExperience(nCount, nLogReason)
end

local function AddTimeTitle(pPlayer, nLogReason, nLogReason2, szMsg, szType, nTitleID, nEndTime, bActive, bShowInfo, szTitle)
  if not nTitleID or nTitleID <= 0 then
    return
  end
  if -1 == nEndTime then
    pPlayer.AddTitle(nTitleID, -1, bActive, bShowInfo, szTitle)
  else
    pPlayer.AddTimeTitle(nTitleID, nEndTime, bActive, bShowInfo, szTitle)
  end
end

local function AddDurationTitle(pPlayer, nLogReason, nLogReason2, szMsg, szType, nTitleID, nDuration, bActive, bShowInfo, szTitle)
  if not nTitleID or nTitleID <= 0 then
    return
  end
  if not nDuration or 0 == nDuration then
    pPlayer.AddTitle(nTitleID, nil, bActive, bShowInfo, szTitle)
  elseif nDuration < 0 then
    pPlayer.AddTitle(nTitleID, -1, bActive, bShowInfo, szTitle)
  else
    local nNow = GetTime()
    local nEndTime = nNow + nDuration
    pPlayer.AddTimeTitle(nTitleID, nEndTime, bActive, bShowInfo, szTitle)
  end
end

local function AddBasicExp(pPlayer, nLogReason, nLogReason2, szMsg, szType, nCount, nUselessParam)

local function AddTongStock(pPlayer, nLogReason, nLogReason2, szMsg, szType, nCount, nUselessParam)
  if not nCount or 0 == nCount then
    LogErr("AwardFunc Err AddTongStock Fail", nLogReason, nUselessParam, nCount)
    return false
  end
  if not Currency:Valid(szType) or nCount <= 0 then
    LogErr("AddTongStock szType", szType, "nCount", nCount)
    return false
  end
  if not nLogReason or 0 == nLogReason then
    LogErr("nLogReason == 0 ", debug.traceback())
  end
  local nOrgPoint = Tong:GetPersonalStock(pPlayer.dwID)
  local bRet = Tong:ChangePersonalStock(pPlayer.dwID, nCount, nLogReason, nLogReason2)
  if not bRet then
    return false
  end
  local nAfterChangePoint = Tong:GetPersonalStock(pPlayer.dwID)
  Log("AddTongStock", pPlayer.szName, pPlayer.dwID, "TongFund", nOrgPoint, nCount, nAfterChangePoint, Env.tbLogWaySubWays[nLogReason] or "unkonwn", Env.tbLogWayDesc[nLogReason] or "unkonwn", nLogReason2)
  return true
end

local function AddPartner(pPlayer, nLogReason, nLogReason2, szMsg, szType, nPartnerTemplateId, nPartnerCount)
  for i = 1, nPartnerCount do
    local bHaveMainPartner, nMainPartnerId = Partner:GetMainPartnerId(pPlayer, nPartnerTemplateId)
    if not bHaveMainPartner then
      local nPartnerId = pPlayer.AddPartner(nPartnerTemplateId, nLogReason or 0)
      if nPartnerId then
        Log("player add partner", pPlayer.szName, "partnerTemplateId:" .. nPartnerTemplateId, "partnerId:" .. nPartnerId)
      end
    elseif 0 ~= nMainPartnerId then
      local nPartnerRedundanceNum = Partner:GetMainPartnerRedundanceNumById(pPlayer, nMainPartnerId) or 0
      Partner:UpdateMainPartnerRedundanceNum(pPlayer, nMainPartnerId, nPartnerRedundanceNum + 1)
      Log("player add partner redundance num", pPlayer.szName, "partnerTemplateId:" .. nPartnerTemplateId, "redundance num:" .. nPartnerRedundanceNum + 1)
    end
  end
end

function Player:GetPlayerSex(pPlayer)
  if not pPlayer then
    return
  end
  local pNpc = pPlayer.GetNpc()
  local nResult
  if pNpc then
    local nBodySex = pNpc.GetBodySex()
    nResult = self:GetPlayerRealSex(nBodySex)
  end
  return nResult
end

function Player:GetPlayerSexById(nDwId)
  local pPlayer = KPlayer.GetPlayerObjById(nDwId)
  if not pPlayer then
    return
  end
  Player:GetPlayerSex(pPlayer)
end

function Player:GetPlayerRealSex(nBodySex)
  local nResult
  if nBodySex and nBodySex >= 0 then
    local nLast = nBodySex % 2
    if 1 == nLast then
      nResult = 1
    else
      nResult = 0
    end
  end
  return nResult
end

function Player:MyMsgBox(pPlayer, szMsg, tbCallback, szNotTipsType, nTime, fnTimeOut, szUiName)
  local tbLight = {}
  local tbBtnTxt = {}
  local nCallbackCount = 0
  szUiName = szUiName or "UIMessageBox"
  pPlayer.tbMsgBoxCallback = tbCallback
  for _, tbInfo in ipairs(tbCallback or {}) do
    table.insert(tbBtnTxt, tbInfo[1])
    table.insert(tbLight, tbInfo.bLight and true or false)
  end
  if MODULE_GAMESERVER and nTime and nTime > 0 and pPlayer.tbMsgBoxCallback then
    pPlayer.tbMsgBoxCallback.nTimerId = Timer:Register(Env.GAME_FPS * nTime, function(nPlayerId)
      local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
      local old_me = pPlayer
      fnTimeOut()
      if pPlayer then
        if pPlayer.tbMsgBoxCallback then
          pPlayer.tbMsgBoxCallback.nTimerId = nil
        end
        pPlayer.CallClientScript("Ui:CloseWindow", szUiName)
      end
    end, pPlayer.dwID)
  end
  pPlayer.CallClientScript("Ui:OnServerMsgBox", szMsg, tbCallback and #tbCallback or 0, tbBtnTxt, MODULE_GAMESERVER, szNotTipsType, nTime, tbLight, szUiName, not MODULE_GAMESERVER and fnTimeOut)
end

function Player:LoadExpReduceSetting()
  self.tbExpReduceSetting = Lib:LoadTabFileEx("Setting/Player/ExpReduce.tab", "ddddd", "", {
    "LevelMin",
    "LevelMax",
    "ExpMin",
    "ExpMax",
    "ReduceRate"
  }, nil, nil, true)
end

function Player:GetExpReduceRate(pPlayer)
  local nCurLevel = pPlayer.nLevel
  local nHaveExp = pPlayer.GetExp()
  local nNextExp = OfflineExp.tbOfflineExp[nCurLevel].nExpUpGrade
  local nExpTimes = math.floor(nHaveExp / nNextExp)
  local nRate = 1
  for _, tbCfg in pairs(self.tbExpReduceSetting) do
    if nCurLevel >= tbCfg.LevelMin and nCurLevel <= tbCfg.LevelMax and nExpTimes >= tbCfg.ExpMin and nExpTimes <= tbCfg.ExpMax then
      nRate = tbCfg.ReduceRate
      break
    end
  end
  return nRate
end

function Player:LoadLevelExpAdditionSetting()
  local tbFile = Lib:LoadTabFileEx("Setting/Player/LevelExpAddition.tab", "dd", nil, {
    "nLowerLevel",
    "nPercent"
  })
  self.tbLevelExpAddition = {}
  local nLowerLevel, nPercent = tbFile[#tbFile].nLowerLevel, tbFile[#tbFile].nPercent
  for i = GameSetting.MAX_PLAYER_TOTAL_LEVEL, nLowerLevel, -1 do
    self.tbLevelExpAddition[i] = nPercent
  end
  local nIndex = 1
  for i = 1, #tbFile do
    nLowerLevel, nPercent = tbFile[i].nLowerLevel, tbFile[i].nPercent
    for nIdx = nIndex, nLowerLevel do
      self.tbLevelExpAddition[nIdx] = nPercent
      nIndex = nIndex + 1
    end
  end
end

Player:LoadLevelExpAdditionSetting()

end
