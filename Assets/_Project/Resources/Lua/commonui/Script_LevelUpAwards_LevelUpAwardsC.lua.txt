function LevelUpAwards:OnChargeLevelUpAwardsReq()
  local bCanBuy, szMsg = LevelUpAwards:CanBuyAwards(me)
  
  if not bCanBuy then
    me.CenterMsg(szMsg, true)
    return false
  end
  local nProductId = LevelUpAwards:GetProductId()
  if not nProductId then
    return false
  end
  Recharge:RequestCreateRechargeOrder(nProductId)
end

function LevelUpAwards:BuyAwardScucess()
  Ui:CloseWindow("UILevelUpAwards")
end
