local tbWnd = {}
Ui:SetClass("UITinyScrollView", tbWnd)

function Ui:NewTinyScrollView(tbOwner, ...)
  local tbClass = Ui:NewClass(Ui:GetClass("UITinyScrollView"))
  tbClass.tbOwner = tbOwner
  tbClass:OnCreate(...)
  return tbClass
end

function tbWnd:OnCreate(szTemplateKey, func, nMaxElementNum)
  self.szTemplateKey = szTemplateKey
  self.tbOwner.pPanel:SetActive(szTemplateKey, false)
  self.tbElementList = {}
  self.funcCallback = func
  self.nMaxElementNum = nMaxElementNum or 999
  self.nShownCount = 0
end

function tbWnd:AddElement(funcSet)
  self.nShownCount = self.nShownCount + 1
  if self.nShownCount <= #self.tbElementList then
    local curElement = self.tbElementList[self.nShownCount]
    self.tbOwner.pPanel:SetActive(curElement.szKey, true)
    funcSet(curElement)
  else
    local szKey = self.tbOwner.pPanel:CloneObject(self.szTemplateKey, "Element" .. #self.tbElementList + 1)
    local tbElement = Ui:CreateSimplePanel(self.tbOwner, szKey)
    funcSet(tbElement)
    self.tbElementList[#self.tbElementList + 1] = tbElement
    if self.funcCallback then
      Ui:Button_BindEvent(self.tbOwner.pPanel, szKey, self.OnClickElement, self, tbElement)
    end
  end
end

function tbWnd:OnClickElement(tbElement)
  if self.funcCallback then
    self.funcCallback(tbElement)
  end
end

function tbWnd:Reset()
  self.nShownCount = 0
  for i, val in ipairs(self.tbElementList) do
    self.tbOwner.pPanel:SetActive(val.szKey, false)
  end
end

function tbWnd:ResetShowCount()
  self.nShownCount = 0
end

function tbWnd:ResetFrom(nStartIndex)
  for i = nStartIndex, #self.tbElementList do
    self.tbOwner.pPanel:SetActive(self.tbElementList[i].szKey, false)
  end
end
