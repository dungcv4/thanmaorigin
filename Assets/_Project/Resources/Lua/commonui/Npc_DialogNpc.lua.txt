Require("CommonScript/Npc/Npc.lua")
Npc.DialogNpc = Npc.DialogNpc or {}
local DialogNpc = Npc.DialogNpc
DialogNpc.szConfigFileName = "Setting/Npc/DialogNpcCfg.tab"
DialogNpc.FnCallBack = {
  OpenWindow = function(pPlayer, tbParam)
    if pPlayer then
      pPlayer.CallClientScript("Ui:OpenWindow", table.unpack(tbParam))
    end
  end,
  DoNothing = function()
    return
  end,
  DoCustomFunc = function(pPlayer, tbParam)
    if pPlayer and #tbParam > 0 then
      Lib:StringCallBack(tbParam[1], pPlayer, table.unpack(tbParam, 2))
    end
  end
}

function DialogNpc:LoadConfig()
  DialogNpc.tbDialogNpcSetting = {}
  local tbSetting = Lib:EasyLoadTabFile(DialogNpc.szConfigFileName, nil)
  if not tbSetting then
    Lib:LogErrFormat("DialogNpc: LoadConfig failed, FileName = %s", DialogNpc.szConfigFileName)
    return
  end
  for _, tbInfo in ipairs(tbSetting) do
    if DialogNpc.FnCallBack[tbInfo.szCallBack] then
      DialogNpc.tbDialogNpcSetting[tbInfo.nNpcTemplateId] = DialogNpc.tbDialogNpcSetting[tbInfo.nNpcTemplateId] or {}
      local tbParam = Lib:SplitStrTry2Num(tbInfo.szParam or "", "|")
      table.insert(DialogNpc.tbDialogNpcSetting[tbInfo.nNpcTemplateId], {
        szOption = tbInfo.szOption,
        szCallBack = tbInfo.szCallBack,
        tbParam = tbParam
      })
    else
      Lib:LogErrFormat("DialogNpc: LoadConfig occur error, unexist callback type = %s, nNpcTemplateId = %d", tbInfo.szCallBack, tbInfo.nNpcTemplateId)
    end
  end
end

DialogNpc:LoadConfig()

function DialogNpc:GetParam(nNpcTemplateId, nOptionOrder)
  if DialogNpc.tbDialogNpcSetting[nNpcTemplateId] and DialogNpc.tbDialogNpcSetting[nNpcTemplateId][nOptionOrder] then
    return DialogNpc.tbDialogNpcSetting[nNpcTemplateId][nOptionOrder].tbParam
  end
end
