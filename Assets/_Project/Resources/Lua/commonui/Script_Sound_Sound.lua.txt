Require("CommonScript/lib.lua")
Require("CommonScript/Sound/SoundCfg.lua")
Sound = Sound or {}
local tbKeyNeedCheck = {
  OpenWindow = Sound.tbCfg.tbOpenWindowNotPlay,
  CloseWindow = Sound.tbCfg.tbCloseWindowNotPlay
}

function Sound:GetSound(szKey)
  local nSoundId = Client.AudioModule.GetSoundID(szKey)
  if 0 == nSoundId then
    LogErr(string.format("Sound.tab,szKey:%s not exsit nSoundID!", szKey))
  end
  return nSoundId
end

function Sound:PlaySound(szKey, szName)
  if tbKeyNeedCheck[szKey] then
    local bRet = self:KeyCheck(szKey, szName)
    if not bRet then
      return
    end
  end
  local nSoundId = self:GetSound(szKey)
  if 0 ~= nSoundId then
    Client.UIModule.PlaySound(nSoundId)
  end
  if Sound.GMOpenLog then
    Log(string.format("PlaySound.szKey:%s,SoundId:%d,UiName:%s", szKey, nSoundId, szName))
  end
end

function Sound:KeyCheck(szKey, szName)
  if tbKeyNeedCheck[szKey][szName] then
    return false
  end
  return true
end

function Sound:StopSound(szKey)
  local nSoundId = self:GetSound(szKey)
  if 0 ~= nSoundId then
    Client.UIModule.StopSound(nSoundId)
  end
end
