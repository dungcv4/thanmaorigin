local tbWnd = Ui:CreateWindow("UIAgreementNew")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/imgTitle/txtTitle",
  WebView = "node/imgBG/WebView",
  NormalView = "node/imgBG/NormalView",
  Scroll = "node/imgBG/NormalView/Scroll",
  btnTemplate = "node/imgBG/btnList/btnList/paneBG/btnTemplate",
  btnOK = "node/imgBG/btnOK"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    self:OnClickOK()
  end
}
tbWnd.UserAgreement = 1
tbWnd.PrivacyPolicy = 2
tbWnd.URLList = {
  [tbWnd.UserAgreement] = "https://dieukhoantrochoi.zing.vn/kiemtheorigin",
  [tbWnd.PrivacyPolicy] = "https://dieukhoantrochoi.zing.vn/kto-baomat"
}

function tbWnd:OnCreate()
  self.tbBtnScrollView = Ui:NewScrollView(self, tbControls.btnTemplate, function(this, nIndex, tbElement)
    this:UpdateCommonView(nIndex, tbElement)
  end)
end

function tbWnd:OnOpen(nSelectIndex)
  self.pPanel:ClearCamera()
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\231\148\168\230\136\183\233\161\187\231\159\165"))
  self.pPanel:Label_SetText(tbControls.btnOK .. "/Text", i18n.EasyGet("\229\144\140\230\132\143"))
  self.pPanel:SetActive(tbControls.NormalView, false)
  self.pPanel:SetActive(tbControls.WebView, true)
  self:UpdateBtnScrollView()
  self.tbBtnScrollView:SelectElement(nSelectIndex or self.UserAgreement)
end

function tbWnd:OnClickOK()
  local wnd = Ui:GetWindow("UILoginServer")
  if wnd then
    wnd:ToggleAgreement(true)
  end
  self:Close()
end

function tbWnd:UpdateBtnScrollView()
  local tbBtnList = {
    {
      szName = i18n.EasyGet("\231\148\168\230\136\183\229\141\143\232\174\174"),
      nType = self.UserAgreement
    },
    {
      szName = i18n.EasyGet("\233\154\144\231\167\129\230\148\191\231\173\150"),
      nType = self.PrivacyPolicy
    }
  }
  for nIndex, tbInfo in ipairs(tbBtnList) do
    self.tbBtnScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("Text", tbInfo.szName)
      tbElement.nType = tbInfo.nType
    end)
  end
end

function tbWnd:UpdateCommonView(nIndex, tbElement)
  self.pPanel:ShowUrl(tbControls.WebView, self.URLList[tbElement.nType], 0, 0)
end
