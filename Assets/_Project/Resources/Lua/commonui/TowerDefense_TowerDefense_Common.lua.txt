Require("CommonScript/TowerDefense/TowerDefenseCfg.lua")

function TowerDefense:LoadSettings()
  local szSettingFile = "Setting/Map/mogutafan/info/NpcPosPoint.txt"
  TowerDefense.tbNpcSetting = Lib:EasyLoadTabFile(szSettingFile, {"szType", "nIndex"})
  for szType, value in pairs(TowerDefense.tbNpcSetting) do
    for nIndex, tbInfo in pairs(value) do
      tbInfo.PosX = tonumber(tbInfo.PosX)
      tbInfo.PosY = tonumber(tbInfo.PosY)
    end
  end
  TowerDefense.tbNpcSetting = SetReadOnly(TowerDefense.tbNpcSetting)
  szSettingFile = "Setting/TowerDefense/NpcRefreshCfg.tab"
  TowerDefense.tbBatchSetting = Lib:EasyLoadTabFile(szSettingFile, {"nBatch", "nSort"})
  szSettingFile = "Setting/Map/mogutafan/info/NpcPathPoint.txt"
  local tbPathSetting = Lib:EasyLoadTabFile(szSettingFile, {"nPath", "nIndex"})
  TowerDefense.tbPathSetting = {}
  for nPath, tbInfo in pairs(tbPathSetting) do
    TowerDefense.tbPathSetting[nPath] = TowerDefense.tbNpcSetting[nPath] or {}
    for nIndex, tbPos in ipairs(tbInfo) do
      local tbTemp = {
        PosX = tonumber(tbPos.PosX),
        PosY = tonumber(tbPos.PosY)
      }
      TowerDefense.tbPathSetting[nPath][nIndex] = tbTemp
    end
  end
  TowerDefense.tbPathSetting = SetReadOnly(TowerDefense.tbPathSetting)
end

TowerDefense:LoadSettings()

function TowerDefense:GetNpcPosByType(szType)
  return self.tbNpcSetting[szType] or {}
end

function TowerDefense:GetNpcByBatchAndIndex(nBatch, nIndex)
  if self.tbBatchSetting[nBatch] and self.tbBatchSetting[nBatch][nIndex] then
    return self.tbBatchSetting[nBatch][nIndex].nNpcID, self.tbBatchSetting[nBatch][nIndex].nLevel, self.tbBatchSetting[nBatch][nIndex].nPath
  end
end

function TowerDefense:GetTotalNpcByBatch(nBatch)
  local nTotal = 0
  if self.tbBatchSetting[nBatch] then
    nTotal = #self.tbBatchSetting[nBatch]
  end
  return nTotal
end

function TowerDefense:GetNpcPath(nPath)
  return self.tbPathSetting[nPath]
end

function TowerDefense:GetBaseScore(nRank)
  return self.Cfg.RandAward[nRank].nScore
end

function TowerDefense:GetAwardKey(nRank)
  return self.Cfg.RandAward[nRank].nAwardkey
end

function TowerDefense:GetExtScoreRate(nNums)
  for index, tbInfo in ipairs(self.Cfg.ExtScoreRate) do
    if nNums <= tbInfo.nMaxNums then
      return tbInfo.nRate
    end
  end
end
