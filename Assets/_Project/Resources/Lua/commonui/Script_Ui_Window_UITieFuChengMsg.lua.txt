local tbWnd = Ui:CreateClass("UITieFuChengMsg", "UIPanel")
local tbControls = {
  btnCheckAll = "btnCheckAll",
  Template = "imgList/List/Template",
  txtDesc = "txt",
  txtBtn = "btnCheckAll/Text",
  txtTitleScore = "ListTitle/txtScore",
  txtFloor = "ListTitle/txtFloor"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCheckAll = function(self)
    if 1 == self.nType then
      self.nType = 2
    else
      self.nType = 1
    end
    self:Update()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.nType = 1
end

function tbWnd:OnOpen()
  RemoteServer.RequestTieFuCheng("GetLastResult")
  self:Update()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSV(tbResult)
  self.tbLastResult = tbResult.tbLastResult
  self.tbSeasonResult = tbResult.tbSeasonResult
  self:Update()
end

function tbWnd:Update()
  self.tbScrollView:Reset()
  if 1 == self.nType then
    self.pPanel:Label_SetText(self.txtBtn, i18n.EasyGet("\230\159\165\231\156\139\230\128\187\230\166\156"))
    self.pPanel:Label_SetText(self.txtDesc, i18n.EasyGet("\230\156\172\229\156\186\229\144\132\229\184\174\228\188\154\233\147\129\230\181\174\229\159\142\230\142\146\229\144\141\229\166\130\228\184\139"))
    self.pPanel:Label_SetText(self.txtTitleScore, i18n.EasyGet("\230\156\172\229\156\186\231\167\175\229\136\134"))
    self.pPanel:SetActive(self.txtFloor, true)
  elseif 2 == self.nType then
    self.pPanel:Label_SetText(self.txtBtn, i18n.EasyGet("\230\159\165\231\156\139\230\156\172\229\156\186"))
    self.pPanel:Label_SetText(self.txtDesc, i18n.EasyGet("\230\156\172\232\181\155\229\173\163\229\144\132\229\184\174\228\188\154\233\147\129\230\181\174\229\159\142\230\142\146\229\144\141\229\166\130\228\184\139"))
    self.pPanel:Label_SetText(self.txtTitleScore, i18n.EasyGet("\232\181\155\229\173\163\231\167\175\229\136\134"))
    self.pPanel:SetActive(self.txtFloor, false)
  end
  if 1 == self.nType then
    local tbData = self.tbLastResult
    if not tbData then
      return
    end
    for i = 1, #tbData do
      self.tbScrollView:SetElement(i, function(tbElement)
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("txtRank", tbData[i].nRank)
        pPanel:Label_SetText("txtTong", string.format("S%s-%s", tbData[i].nServerId or "", tbData[i].szName))
        pPanel:Label_SetText("txtScore", tbData[i].nFieldScore)
        pPanel:Label_SetText("txtBoss", tbData[i].szLeaderName)
        pPanel:SetActive("txtFloor", true)
        pPanel:Label_SetText("txtFloor", tbData[i].nFloor)
      end)
    end
  else
    local tbData = self.tbSeasonResult
    if not tbData then
      return
    end
    for i = 1, #tbData do
      self.tbScrollView:SetElement(i, function(tbElement)
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("txtRank", i)
        pPanel:Label_SetText("txtTong", string.format("S%s-%s", tbData[i].nServerId or "", tbData[i].szName))
        pPanel:Label_SetText("txtScore", tbData[i].nScore)
        pPanel:Label_SetText("txtBoss", tbData[i].szLeaderName)
        pPanel:SetActive("txtFloor", false)
      end)
    end
  end
end
