local tbWnd = Ui:CreateWindow("UIGeneralHelp")
tbWnd.tbControls = {
  btnClose = "imgBG/btnClose",
  txtBigTitle = "imgBG/BG/txtTitle",
  helpTemplate = "imgBG/panelContentList/maskLayer/paneBG/HelpTemplate"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.tbScrollView = Ui:NewScrollView(self, self.helpTemplate)
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen(szHelpKey)
  self.szHelpKey = szHelpKey
  if not self.szHelpKey or not GeneralHelp.tbHelpInfo[self.szHelpKey] then
    return
  end
  self.tbScrollView:GoTop()
  local tbInfo = GeneralHelp.tbHelpInfo[self.szHelpKey]
  self.pPanel:Label_SetText(self.txtBigTitle, i18n.Parse(tbInfo.szBigTitle))
  self.tbScrollView:Reset()
  for i = 1, #tbInfo.tbContent do
    self.tbScrollView:SetElement(i, function(tbElement)
      self:SetItem(tbElement, i, tbInfo.tbContent[i])
    end)
  end
end

function tbWnd:SetItem(tbElement, i, tbContent)
  tbElement.pPanel:Label_SetText("txtNumber", i)
  tbElement.pPanel:Label_SetText("txtTitle", i18n.Parse(tbContent.szTitle))
  tbElement.pPanel:Label_SetText("txtContent", string.gsub(i18n.Parse(tbContent.szContent), "\\n", "\n"))
end
