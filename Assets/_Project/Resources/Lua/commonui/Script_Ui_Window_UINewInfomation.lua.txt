function tbWnd:OnCreate()
  self:CreateBulletin()
  self.tbButtonListScrollView = Ui:NewScrollView(self, tbControls.btnTemplate, function(self, nIndex, tbElement)
  end)
  self.pPanel:SetActive(tbControls.NewInfoMsg, false)
end

function tbWnd:OnOpen(tbJsonData, szPanelClass)
  self:OnChangeWebViewOnOff()
  self.tbJsonData = tbJsonData
  self:UpdateLeftTabScrollView()
  if self.tbButtonListScrollView:GetCount() <= 0 then
    me.CenterMsg(i18n.EasyGet("\230\154\130\230\151\182\230\151\160\230\150\176\230\182\136\230\129\175"))
    self.pPanel:SetActive(tbControls.txtEmpty, true)
    return
  else
    self.pPanel:SetActive(tbControls.txtEmpty, false)
  end
  self:UpdatePanelSetting()
  if self.tbTabControl then
    self.tbTabControl:UpdateData(self.ALL_PANEL_SETTING)
  else
    self.tbTabControl = Ui:NewTabControl(self, self.ALL_PANEL_SETTING)
  end
  local nOpenType
  for nPanelType, tbSetting in ipairs(self.ALL_PANEL_SETTING) do
    if szPanelClass == tbSetting.CLASS and tbSetting.OpenFunc and tbSetting.OpenFunc() then
      nOpenType = nPanelType
    end
  end
  if not self.tbTabControl:OnOpen(nOpenType) then
    return
  end
  self.tbButtonListScrollView:ScrollToIndex(self.tbTabControl.nPanelType, true)
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  for nId, szPath in ipairs(self.tbRedPointList) do
    RedPointMgr:DynamicUnRegist(self.pPanel, szPath)
  end
end

function tbWnd:UpdatePanelSetting()
  self.ALL_PANEL_SETTING = {
    PARENT_PATH = "node/imgBG/Content/InfoContent"
  }
  self.tbRedPointList = {}
  for i = 1, self.nActivityOffSet do
    self.ALL_PANEL_SETTING[i] = {
      CLASS = "UINewInfoMsg",
      OBJ = tbControls.panelNewInfoMsg,
      BTN = string.format(tbControls.BtnElement, i)
    }
    if i <= ClientBulletin.MAX_REDPOINT then
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, string.format(tbControls.BtnElement, i), {
        RedPointMgr.EVENT.ClientBulletinBegin - 1 + i
      })
      self.tbRedPointList[i] = string.format(tbControls.BtnElement, i)
    end
  end
  for i = 1, #self.ACTIVITY_PANEL_SETTING do
    self.ALL_PANEL_SETTING[i + self.nActivityOffSet] = self.ACTIVITY_PANEL_SETTING[i]
    local szPath = string.format(tbControls.BtnElement, i + self.nActivityOffSet)
    self.ALL_PANEL_SETTING[i + self.nActivityOffSet].BTN = szPath
    if self.ACTIVITY_PANEL_SETTING[i] and self.ACTIVITY_PANEL_SETTING[i].RedPointEvent then
      self.tbRedPointList[i + self.nActivityOffSet] = szPath
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szPath, self.ACTIVITY_PANEL_SETTING[i].RedPointEvent)
    end
    self.tbRedPointList[i + self.nActivityOffSet] = szPath
  end
end

function tbWnd:UpdateLeftTabScrollView()
  self.tbButtonListScrollView:Reset()
  self.bulletinContent:ClearBulletin()
  self.nActivityOffSet = 0
  if self.tbJsonData then
    if #self.tbJsonData > ClientBulletin.MAX_REDPOINT then
      LogErr("ClientBulletin RedPoint overflow!Check information nums and Max_RedPoint!")
    end
    self.nActivityOffSet = #self.tbJsonData
    for nIdx, tbInfo in ipairs(self.tbJsonData) do
      self.tbButtonListScrollView:SetElement(nIdx, function(tbElement)
        tbElement.pPanel:Label_SetText("Text", tbInfo.title)
        tbElement.pPanel:SetActive("Main", true)
      end)
      local szPath = string.format(tbControls.BtnElement, nIdx)
      self.bulletinContent:AddBulletin(tbInfo.title, tbInfo.content)
    end
  end
  for i = 1, #self.ACTIVITY_PANEL_SETTING do
    self.tbButtonListScrollView:SetElement(i + self.nActivityOffSet, function(tbElement)
      tbElement.pPanel:Label_SetText("Text", self.ACTIVITY_PANEL_SETTING[i].pFuncBtnText())
      tbElement.pPanel:SetActive("Main", true)
    end)
  end
end

end
