local tbItem = Item:GetClass("DingFengPKGmCard")

function tbItem:OnClientUse(it)
  Ui:CloseWindow("UIBag")
  Ui:CloseWindow("UIItemTips")
  return 1
end

function tbItem:OnUse(it)
  local tbOptList = {}
  self:GetDingFengPKWatch(tbOptList)
  table.insert(tbOptList, {
    Text = i18n.EasyGet("\230\136\145\229\143\170\230\152\175\230\157\165\231\156\139\231\156\139")
  })
  Dialog:ShowNpcDialog({
    Text = i18n.EasyGet("\232\175\183\233\128\137\230\139\169\232\167\130\230\136\152\231\154\132\230\175\148\232\181\155"),
    OptList = tbOptList
  }, me, DingFengPK.JieChuYingCaiCfg.NPC_ID)
  return 0
end

function tbItem:GetDingFengPKWatch(tbOptList)
  if not DingFengPK.tbMapId2FuBen then
    return
  end
  for nMapId, tbFuben in pairs(DingFengPK.tbMapId2FuBen) do
    if nMapId ~= DingFengPK.nCurReadyMap and tbFuben:IsOpen() and not tbFuben:IsFinish() then
      local szMatch = string.format("%s vs %s", tbFuben.tbBattleTeamDatas[1].szTeamName, tbFuben.tbBattleTeamDatas[2].szTeamName)
      table.insert(tbOptList, {
        Text = szMatch,
        Callback = DingFengPK.WatchMatch,
        Param = {
          DingFengPK,
          me,
          nMapId
        }
      })
    end
  end
end
