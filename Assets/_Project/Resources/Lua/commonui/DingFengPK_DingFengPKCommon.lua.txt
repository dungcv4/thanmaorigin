function DingFengPK:IsSysOpen()
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local nNowTime = GetTime()
  if nNowTime < Lib:GetConfigDate2Time(self.Cfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.Cfg.END_DATE) then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  return true
end

function DingFengPK:GetOpenType()
  if not self.IS_SYS_OPEN then
    return
  end
  if self:CheckInPreTime() then
    return self.OPEN_TYPE.PRE
  elseif self:CheckInMidTime() then
    return self.OPEN_TYPE.MID
  elseif Lib:IsSameDay(0, GetTime(), Lib:GetDate2Time(self.TimeCfg.FINAL_1)) then
    return self.OPEN_TYPE.FINAL_DAY1
  elseif Lib:IsSameDay(0, GetTime(), Lib:GetDate2Time(self.TimeCfg.FINAL_2)) then
    return self.OPEN_TYPE.FINAL_DAY2
  elseif Lib:IsSameDay(0, GetTime(), Lib:GetDate2Time(self.TimeCfg.FINAL_3)) then
    return self.OPEN_TYPE.FINAL_DAY3
  elseif Lib:IsSameDay(0, GetTime(), Lib:GetDate2Time(self.TimeCfg.FINAL_4)) then
    return self.OPEN_TYPE.FINAL_DAY4
  end
end

function DingFengPK:ChkPlayerInMapEx(pPlayer)
  if not pPlayer then
    return
  end
  if pPlayer.nMapTemplateId == DingFengPK.Cfg.READY_MAP_ID or pPlayer.nMapTemplateId == DingFengPK.Cfg.FIGHT_MAP_ID then
    return true
  end
  return false
end

function DingFengPK:ChkCondition(pPlayer, tbCondition)
  if tbCondition[DingFengPK.emMsgNotBegin] and not DingFengPK:IsSysOpen() then
    return false, DingFengPK.emMsgNotBegin
  end
  if tbCondition[DingFengPK.emMsgSignTime] and (GetTime() < Lib:GetConfigDate2Time(self.TimeCfg.PREVIEW_TIME) or GetTime() > Lib:GetConfigDate2Time(self.TimeCfg.PRE_END)) then
    return false, DingFengPK.emMsgSignTime
  end
  if tbCondition[DingFengPK.emMsgMinLevel] and pPlayer.nLevel < DingFengPK.Cfg.MIN_LEVEL then
    return false, DingFengPK.emMsgMinLevel, {
      DingFengPK.Cfg.MIN_LEVEL
    }
  end
  if tbCondition[DingFengPK.emMsgNotTeamCaptain] and pPlayer.dwTeamID > 0 and not TeamMgr:IsCaptain(pPlayer) then
    return false, DingFengPK.emMsgNotTeamCaptain
  end
  if tbCondition[DingFengPK.emMsgTeamMemeberErr] and pPlayer.dwTeamID > 0 then
    local tbMember = TeamMgr:GetMembers(pPlayer.dwTeamID)
    if tbCondition[DingFengPK.emMsgTeamMemeberErr] and #tbMember > DingFengPK.TeamCfg.MAX_LEAGUE_MEMBER then
      return false, DingFengPK.emMsgTeamMemeberErr, {
        DingFengPK.TeamCfg.MAX_LEAGUE_MEMBER
      }
    end
  end
  if tbCondition[DingFengPK.emMsgNeedBattleTeam] and self:GetBattleTeamId(pPlayer.dwID) <= 0 then
    return false, DingFengPK.emMsgNeedBattleTeam
  end
  if tbCondition[DingFengPK.emMsgSafeMap] and not Map:IsCityMap(pPlayer.nMapTemplateId) and not Map:IsFieldFightMap(pPlayer.nMapTemplateId) then
    return false, DingFengPK.emMsgSafeMap
  end
  if tbCondition[DingFengPK.emMsgNotSafePoint] and Map:IsFieldFightMap(pPlayer.nMapTemplateId) and pPlayer.nFightMode ~= Npc.FIGHT_MODE.emFightMode_None then
    return false, DingFengPK.emMsgNotSafePoint
  end
  if tbCondition[DingFengPK.emMsgAloneState] and not pPlayer.nState == Player.emPLAYER_STATE_ALONE then
    return false, DingFengPK.emMsgAloneState
  end
  if tbCondition[self.emMsgQualification] and not self:CheckQualification(pPlayer) then
    return false, self.emMsgQualification
  end
  if MODULE_GAMESERVER then
    if tbCondition[DingFengPK.emMsgHaveTeam] and self:ResetTeamGroup(pPlayer.dwID) and self:GetBattleTeamId(pPlayer.dwID) > 0 then
      return false, DingFengPK.emMsgHaveTeam
    end
    if pPlayer.dwTeamID > 0 then
      local tbMember = TeamMgr:GetMembers(pPlayer.dwTeamID)
      if tbMember then
        for _, nPlayerId in pairs(tbMember) do
          if nPlayerId ~= pPlayer.dwID then
            local pMember = KPlayer.GetPlayerObjById(nPlayerId)
            if not pMember then
              if tbCondition[DingFengPK.emMsgMemberOffline] then
                return false, DingFengPK.emMsgMemberOffline
              end
            else
              if tbCondition[DingFengPK.emMsgMemberMinLevel] and pMember.nLevel < DingFengPK.Cfg.MIN_LEVEL then
                return false, DingFengPK.emMsgMemberMinLevel, {
                  pMember.szName,
                  DingFengPK.Cfg.MIN_LEVEL
                }
              end
              if tbCondition[DingFengPK.emMsgMemberSafeMap] and not Map:IsCityMap(pMember.nMapTemplateId) and not Map:IsFieldFightMap(pMember.nMapTemplateId) then
                return false, DingFengPK.emMsgMemberSafeMap, {
                  pMember.szName
                }
              end
              if tbCondition[DingFengPK.emMsgMemberNotSafePoint] and Map:IsFieldFightMap(pMember.nMapTemplateId) and pMember.nFightMode ~= Npc.FIGHT_MODE.emFightMode_None then
                return false, DingFengPK.emMsgMemberNotSafePoint, {
                  pMember.szName
                }
              end
              if tbCondition[DingFengPK.emMsgMemberAloneState] and not pMember.nState == Player.emPLAYER_STATE_ALONE then
                return false, DingFengPK.emMsgMemberAloneState, {
                  pMember.szName
                }
              end
              if tbCondition[DingFengPK.emMsgMemberHaveTeam] and self:ResetTeamGroup(pMember.dwID) and self:GetBattleTeamId(pMember.dwID) > 0 then
                return false, DingFengPK.emMsgMemberHaveTeam, {
                  pMember.szName
                }
              end
              if tbCondition[self.emMsgMemberQualification] and not self:CheckQualification(pMember) then
                return false, self.emMsgMemberQualification, {
                  pMember.szName
                }
              end
            end
          end
        end
      end
    end
    if tbCondition[DingFengPK.emMsgWeekOverCount] and self:ChkWeekJoinEnough(pPlayer.dwID) then
      return false, DingFengPK.emMsgWeekOverCount
    end
    if tbCondition[DingFengPK.emMsgTotalOverCount] and self:ChkTotalJoinEnough(pPlayer.dwID) then
      return false, DingFengPK.emMsgTotalOverCount
    end
    if tbCondition[DingFengPK.emMsgJoinOverCount] and self:ChkDailyJoinEnough(pPlayer.dwID) then
      return false, DingFengPK.emMsgJoinOverCount
    end
  end
  return true
end

function DingFengPK:ChkConditionAndSendMsg(pPlayer, tbCondition)
  if not pPlayer then
    LogErr("DingFengPK:ChkConditionAndSendMsg pPlayer is nil ")
    return false
  end
  local bRet = true
  local nCode, tbParam
  bRet, nCode, tbParam = self:ChkCondition(pPlayer, tbCondition)
  if not bRet then
    if MODULE_GAMECLIENT then
      self:MsgCode(pPlayer.dwID, nCode, table.unpack(tbParam or {}))
    end
    if MODULE_GAMESERVER then
      if not pPlayer.dwTeamID or pPlayer.dwTeamID <= 0 then
        self:MsgCode(pPlayer.dwID, nCode, table.unpack(tbParam or {}))
      else
        local tbMember = TeamMgr:GetMembers(pPlayer.dwTeamID) or {}
        if #tbMember <= 0 then
          self:MsgCode(pPlayer.dwID, nCode, table.unpack(tbParam or {}))
        end
        for _, nMemberId in pairs(tbMember) do
          self:MsgCode(nMemberId, nCode, table.unpack(tbParam or {}))
        end
      end
    end
  end
  return bRet
end

function DingFengPK:MsgCode(nPlayerId, nCode, ...)
  if MODULE_GAMESERVER then
    local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
    if not pPlayer then
      return
    end
    pPlayer.CallClientScript("DingFengPK:MsgCode", nPlayerId, nCode, ...)
  elseif MODULE_GAMECLIENT then
    local szMsg = ""
    if (...) then
      szMsg = i18n.Get(DingFengPK.tbMsg[nCode](), ...)
    else
      szMsg = DingFengPK.tbMsg[nCode]()
    end
    if nCode == self.emMsgQualification then
      local function funcOK()
        Shop:OpenShopBuyWidDefault(DingFengPK.Cfg.QUALIFICATION_ID)
      end
      
      local function funcCancel()
      end
      
      Ui:OpenWindow("UIMessageBox", szMsg, {
        {funcOK},
        {funcCancel}
      }, {
        i18n.szOk,
        i18n.szCancel
      })
    else
      Ui:OpenWindow("UIMessageBox", szMsg)
    end
  end
end

function DingFengPK:GetBattleTeamId(nPlayerId)
  local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
  if not pPlayer then
    return 0
  end
  if MODULE_GAMESERVER then
    DingFengPK:ResetTeamGroup(nPlayerId)
  end
  return pPlayer.GetUserValue(DingFengPK.SAVE_GROUP, DingFengPK.SAVE_TEAMID)
end

if MODULE_GAMESERVER then
  function DingFengPK:ChkWeekJoinEnough(nPlayerId)
    local tbTeamData = self.DingFengTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    
    if not tbTeamData then
      return
    end
    return self.DingFengTeam:IsWeekOverCount(tbTeamData)
  end
  
  function DingFengPK:ChkTotalJoinEnough(nPlayerId)
    local tbTeamData = self.DingFengTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    if not tbTeamData then
      return true
    end
    if DingFengPK.nOpenType == DingFengPK.OPEN_TYPE.PRE then
      return self.DingFengTeam:IsPreTotalJoinEnough(tbTeamData)
    end
    return self.DingFengTeam:IsTotalJoinEnough(tbTeamData)
  end
  
  function DingFengPK:ChkDailyJoinEnough(nPlayerId)
    local tbTeamData = self.DingFengTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    if not tbTeamData then
      return false
    end
    return self.DingFengTeam:IsDailyJoinEnough(tbTeamData)
  end
end

function DingFengPK:CheckGmWarch(pPlayer)
  local nGmCard = pPlayer.GetItemCountByTypeInBag(self.Cfg.GM_CARD_ID)
  return nGmCard > 0
end

function DingFengPK:GetRankAward(nStageAwardType, nPlayerRank)
  local tbAward
  local tbAllAward = self:GetAwardSetting(nStageAwardType)
  if not tbAllAward then
    return
  end
  for _, tbData in ipairs(tbAllAward) do
    if nPlayerRank <= tbData.nRank then
      tbAward = {
        tbData.nAwardId,
        tbData.nAwardKey
      }
      break
    end
  end
  return tbAward
end

function DingFengPK:CheckQualification(pPlayer)
  local nNum = pPlayer.GetItemCountByTypeInBag(self.Cfg.QUALIFICATION_ID)
  return nNum > 0
end
