HanWuYiJi.pHanWuYiJiEnhanceCtrl = HanWuYiJi.pHanWuYiJiEnhanceCtrl or {}
local HanWuEnhanceCtrl = HanWuYiJi.pHanWuYiJiEnhanceCtrl

function HanWuEnhanceCtrl.ChangeTo(nPlayerId, nFactionId, nFactionSectId, nEnhancePart, nOldLevel, nNewLevel)
  local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
  local bCanChange, szReason = HanWuEnhanceCtrl._CanChangeTo(pPlayer, nFactionId, nFactionSectId, nEnhancePart, nOldLevel, nNewLevel)
  if not bCanChange then
    return false, szReason
  end
  if nOldLevel > 0 then
    HanWuEnhanceCtrl._RemoveAttrByLevel(pPlayer, nFactionId, nFactionSectId, nEnhancePart, nOldLevel)
  end
  if nNewLevel > 0 then
    HanWuEnhanceCtrl._ApplyAttrByLevel(pPlayer, nFactionId, nFactionSectId, nEnhancePart, nNewLevel)
  end
  if SERVER_RUNNING_MODE == SERVER_MODE.ZONE_SERVER then
    pPlayer.CallClientScript("HanWuYiJi.pHanWuYiJiEnhanceCtrl.ChangeTo", nPlayerId, nFactionId, nFactionSectId, nEnhancePart, nOldLevel, nNewLevel)
  end
  return true, szReason
end

function HanWuEnhanceCtrl._CanChangeTo(pPlayer, nFactionId, nFactionSectId, nEnhancePart, nOldLevel, nNewLevel)
  if not pPlayer then
    return false, "[HanWuEnhanceCtrl._CanChangeTo] player offline!"
  end
  local tbNewAttrListCfg = HanWuEnhanceCtrl._GetAttrList(nFactionId, nFactionSectId, nEnhancePart, nNewLevel)
  local tbOldAttrListCfg = HanWuEnhanceCtrl._GetAttrList(nFactionId, nFactionSectId, nEnhancePart, nOldLevel)
  if not tbNewAttrListCfg and nNewLevel > 0 then
    return false, "[HanWuEnhanceCtrl._CanChangeTo] new level cfg not exist!"
  end
  if not tbOldAttrListCfg and nOldLevel > 0 then
    return false, "[HanWuEnhanceCtrl._CanChangeTo] old level cfg not exist!"
  end
  if nEnhancePart < HanWuYiJi.tbEquipPosIndex.Weapon or nEnhancePart > HanWuYiJi.tbEquipPosIndex.ShouShi then
    return false, "[HanWuEnhanceCtrl._CanChangeTo] bad enhance part!"
  end
  if nOldLevel < 0 or nOldLevel > HanWuYiJi.DEF.MAX_ENCHANCE_LV or nNewLevel < 0 or nNewLevel > HanWuYiJi.DEF.MAX_ENCHANCE_LV then
    return false, "[HanWuEnhanceCtrl._CanChangeTo] bad enhance level!"
  end
  return true, ""
end

function HanWuEnhanceCtrl._ApplyAttrByLevel(pPlayer, nFactionId, nFactionSectId, nEnhancePart, nLevel)
  local tbAttrListCfg = HanWuEnhanceCtrl._GetAttrList(nFactionId, nFactionSectId, nEnhancePart, nLevel)
  if not tbAttrListCfg then
    LogErr("[HanWuEnhanceCtrl._ApplyAttrByLevel] can`t found config! playerid, faction, faction sect, enhance level", pPlayer.dwID, nFactionId, nFactionSectId, nLevel)
    return
  end
  for _, tbAttrInfo in ipairs(tbAttrListCfg) do
    pPlayer.ApplyMagicAttrib(tbAttrInfo[1], tbAttrInfo[2], tbAttrInfo[3], tbAttrInfo[4])
  end
end

function HanWuEnhanceCtrl._RemoveAttrByLevel(pPlayer, nFactionId, nFactionSectId, nEnhancePart, nLevel)
  local tbAttrListCfg = HanWuEnhanceCtrl._GetAttrList(nFactionId, nFactionSectId, nEnhancePart, nLevel)
  if not tbAttrListCfg then
    LogErr("[HanWuEnhanceCtrl._RemoveAttrByLevel] can`t found config! playerid, faction, faction sect, enhance level", pPlayer.dwID, nFactionId, nFactionSectId, nLevel)
    return
  end
  for _, tbAttrInfo in ipairs(tbAttrListCfg) do
    pPlayer.RemoveMagicAttrib(tbAttrInfo[1], tbAttrInfo[2], tbAttrInfo[3], tbAttrInfo[4])
  end
end

function HanWuEnhanceCtrl._GetAttrList(nFactionId, nFactionSectId, nEnhancePart, nLevel)
  local tbList = {}
  for _, tbCfg in pairs(HanWuYiJi.tbEnhanceLevelCfg) do
    if tbCfg.nFactionId == nFactionId and tbCfg.nFactionSectId == nFactionSectId then
      if not tbCfg.tbLevelAttr[nEnhancePart] then
        return nil
      end
      return tbCfg.tbLevelAttr[nEnhancePart][nLevel]
    end
  end
  return tbList
end
