Calendar.Def = {
  OPEN_LEVEL = 8,
  HELP_GUIDETIPS_ACTID = 8,
  HELP_GUIDETIPS_KEY = "ActivityTipHelp"
}
Calendar.nCurMaxLevel = Calendar.nCurMaxLevel or 0
Calendar.nNextOpenDay = Calendar.nNextOpenDay or 0

function Calendar:LoadTimeFrame()
  self.tbOpenLevelTimeFrame = {}
  self.tbMapTimeFrame = {}
  local tbFile = Lib:LoadTabFileEx("Setting/timeframe.tab", "sdddddds", nil, {
    "Event",
    "nLowerLimitLevel",
    "NewMaxLevel",
    "nLowerLimitZongShiLevel",
    "NewMaxZongShiLevel",
    "OpenDay",
    "OpenTime",
    "Name"
  })
  local nDay = 0
  for i, v in ipairs(tbFile) do
    assert(nDay <= v.OpenDay)
    nDay = v.OpenDay
    if 0 ~= v.NewMaxLevel and 0 ~= v.nLowerLimitLevel then
      table.insert(self.tbOpenLevelTimeFrame, v)
    end
    self.tbMapTimeFrame[v.Event] = v
  end
  self.tbOpenLevelTimeFrame = SetReadOnly(self.tbOpenLevelTimeFrame)
  self.tbMapTimeFrame = SetReadOnly(self.tbMapTimeFrame)
end

function Calendar:GetWorldLevelByTimeframe(szTimeFrame)
  if self.tbMapTimeFrame[szTimeFrame] then
    return self.tbMapTimeFrame[szTimeFrame].NewMaxLevel + self.tbMapTimeFrame[szTimeFrame].NewMaxZongShiLevel
  end
  return 0
end

function Calendar:GetTimeframeName(szTimeFrame)
  if self.tbMapTimeFrame[szTimeFrame] then
    return i18n.Parse(self.tbMapTimeFrame[szTimeFrame].Name)
  else
    return ""
  end
end

Calendar:LoadTimeFrame()

function Calendar:GetLeftTimeByTimeframe(szTimeFrame)
  if not self.tbMapTimeFrame[szTimeFrame] then
    return 0
  end
  local OpenDay = self.tbMapTimeFrame[szTimeFrame].OpenDay
  local nKaiFuDay = Lib:GetServerOpenDay()
  if OpenDay < nKaiFuDay then
    return 0
  end
  local OpenTime = self.tbMapTimeFrame[szTimeFrame].OpenTime
  local nCreateTime = GetServerCreateTime()
  local tbTime = os.date("*t", nCreateTime)
  local nTimestamp = os.time({
    day = tbTime.day,
    month = tbTime.month,
    year = tbTime.year,
    hour = 0,
    minute = 0,
    second = 0
  })
  local nOpenTimestamp = nTimestamp + (OpenDay - 1) * 3600 * 24 + math.ceil(OpenTime / 100) * 3600
  local nLeftTime = nOpenTimestamp - GetTime()
  return nLeftTime
end

function Calendar:GetCurLevelOpenDay(szTimeFrame, nOpenTime)
  if not self.tbMapTimeFrame[szTimeFrame] then
    return nil
  end
  local nCreateTime = nOpenTime or GetServerCreateTime()
  local nToday = Lib:GetLocalDay(GetTime() - 14400)
  local nOpenedDay = nToday - Lib:GetLocalDay(nCreateTime) + 1
  return nOpenedDay - self.tbMapTimeFrame[szTimeFrame].OpenDay + 1
end
