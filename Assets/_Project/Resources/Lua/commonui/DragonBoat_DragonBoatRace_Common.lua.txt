Require("CommonScript/DragonBoat/DragonBoatRaceCfg.lua")
DragonBoatRace = DragonBoatRace or {}
DragonBoatRace.GAME_STATE = {START = 1, CLOSE = 0}

function DragonBoatRace:Load()
  self.Cfg.nOpenTime = Lib:ParseNumbericDateToTime(self.Cfg.nOpenTime)
  self.Cfg.nEndTime = Lib:ParseNumbericDateToTime(self.Cfg.nEndTime)
  local szFilePath = "Setting/DragonBoat/Skill.tab"
  local tbField = {
    "nSkillId",
    "szDesc",
    "szIconAltlas",
    "nSortInUI",
    "CastType"
  }
  local tbSkills = Lib:LoadTabFileEx(szFilePath, "dssdd", nil, tbField, nil, nil, true)
  self.tbSkills = {}
  self.tbSkillIdList = {}
  for _, tbInfo in ipairs(tbSkills or {}) do
    table.insert(self.tbSkills, {
      SkillId = tbInfo.nSkillId,
      SortInUI = tbInfo.nSortInUI,
      szDesc = tbInfo.szDesc,
      szIconAltlas = tbInfo.szIconAltlas,
      CastType = tbInfo.CastType
    })
    self.tbSkillIdList[tbInfo.nSkillId] = true
  end
  table.sort(self.tbSkills, function(tbFirst, tbSecond)
    return tbFirst.SortInUI < tbSecond.SortInUI
  end)
  self.tbSkills = SetReadOnly(self.tbSkills)
end

DragonBoatRace:Load()

function DragonBoatRace:GetSkillList()
  return self.tbSkills or {}
end
