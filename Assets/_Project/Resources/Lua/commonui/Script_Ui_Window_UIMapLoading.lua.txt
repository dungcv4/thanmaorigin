local tbWnd = Ui:CreateWindow("UIMapLoading")
local tbLerp = {}
tbLerp.MapLoading_BeforePointMaxPercent = 1
tbLerp.MapLoading_NeedArriveTime = 1
tbLerp.MapLoading_NeedArriveValue = 0.9
tbLerp.MapLoading_NeedArriveTime2 = 1
tbLerp.MapLoading_NeedArriveValue2 = 1

function tbLerp:Start()
  self.fPastTime = nil
  self.fStartTime = Ui.Time.time
  self:CalcCoe()
end

function tbLerp:PassPoint()
  if self.fPastTime == nil then
    self.fPastTime = Ui.Time.time
  end
end

function tbLerp:GetPercent()
  local t = Ui.Time.time
  if self.fPastTime ~= nil then
    return self:Calc(self.fSecCoe * (t - self.fPastTime)) * (1 - tbLerp.MapLoading_BeforePointMaxPercent) + tbLerp.MapLoading_BeforePointMaxPercent
  end
  if nil ~= self.fStartTime then
    return self:Calc(self.fFirstCoe * (t - self.fStartTime)) * tbLerp.MapLoading_BeforePointMaxPercent
  end
  return 0
end

function tbLerp:Calc(fx)
  return 2 / math.pi * math.atan(fx)
end

function tbLerp:CalcCoe()
  self.fFirstCoe = self:_CalcAtan(tbLerp.MapLoading_NeedArriveTime, tbLerp.MapLoading_NeedArriveValue / tbLerp.MapLoading_BeforePointMaxPercent)
  self.fSecCoe = self:_CalcAtan(tbLerp.MapLoading_NeedArriveTime2, (tbLerp.MapLoading_NeedArriveValue2 - tbLerp.MapLoading_BeforePointMaxPercent) / (1 - tbLerp.MapLoading_BeforePointMaxPercent))
end

function tbLerp:_CalcAtan(fx, fy)
  return math.tan(fy * math.pi / 2) / fx
end

function tbWnd:OnOpen(nMapId)
  if nMapId then
    self.szMapName = Map:GetMapResName(nMapId)
    CS.SceneModule.InitMapDownloadInfo(self.szMapName)
    self.bNeedDownloadMap = CS.SceneModule.NeedDownloadMap(self.szMapName)
    if self.bNeedDownloadMap then
      self.nMapSize = tonumber(CS.SceneModule.GetMapTotalSize(self.szMapName))
      self.nCurDownload = 0
      self.nLastDownloadTime = GetTime()
    end
  end
  self.pPanel:SetActive(self.txtLoadingPercent, not Ui.GameEnv.IsShenHe)
  self.pPanel:SetActive(self.txtLoadingTips, not Ui.GameEnv.IsShenHe)
  self:SetTips(tbTips[math.random(1, #tbTips)]())
  self:SetPercent(0)
  self:StartCalcProgressBar()
  HeartBeatToWorldServer()
  self.nHeartBeatTimer = Timer:Register(Env.GAME_FPS * 3, function()
    HeartBeatToWorldServer()
    return true
  end)
  Client.AudioModule.StopMusic()
  Ui:CloseWindow("UITakePhoto")
  self:SetRandomBG()
end

function tbWnd:OnClose()
  self:CloseProgressTimer()
  Timer:Close(self.nHeartBeatTimer)
  self.nHeartBeatTimer = nil
end

function tbWnd:SetTips(szTips)
  self.pPanel:Label_SetText(self.txtLoadingTips, szTips)
end

function tbWnd:SetPercent(nPercent)
  self.pPanel:Label_SetText(self.txtLoadingPercent, math.floor(nPercent * 100) .. "%")
  self.pPanel:Sprite_SetFill(self.imgProgress, nPercent)
end

function tbWnd:CloseProgressTimer()
  if self.nProgressTimer ~= nil then
    Timer:Close(self.nProgressTimer)
    self.nProgressTimer = nil
  end
end

function tbWnd:StartCalcProgressBar()
  self:CloseProgressTimer()
  if not self.bNeedDownloadMap then
    tbLerp:Start()
    self.pPanel:SetActive(self.txtMapDownloadTips, false)
  else
    self.pPanel:SetActive(self.txtMapDownloadTips, true)
  end
  tbLerp:Start()
  self:SetPercent(0)
  self.nProgressTimer = Timer:Register(1, function()
    if self.bNeedDownloadMap then
      local nowTime = GetTime()
      local nDownload = tonumber(CS.SceneModule.GetDownloadedSize(self.szMapName))
      if nDownload > 0 and nDownload > self.nCurDownload then
        self.nCurDownload = nDownload
        self.nLastDownloadTime = nowTime
      end
      local changeTime = nowTime - self.nLastDownloadTime
      if changeTime > 180 then
        Client.SceneModule.OnLoaderCallBack()
        return false
      end
      if not Ui.GameEnv.IsShenHe then
        self.pPanel:SetActive(self.txtMapDownloadTips, true)
        self.pPanel:Label_SetText(self.txtMapDownloadTips, i18n.EasyGet("\229\156\176\229\155\190\228\184\139\232\189\189\228\184\173({0}kb/{1}kb)...", math.ceil(nDownload / 1024), math.ceil(self.nMapSize / 1024)))
      else
        self.pPanel:SetActive(self.txtMapDownloadTips, false)
      end
      self.bNeedDownloadMap = CS.SceneModule.NeedDownloadMap(self.szMapName)
      if not self.bNeedDownloadMap then
        tbLerp:Start()
        self.pPanel:SetActive(self.txtMapDownloadTips, false)
      end
    else
      local p = tbLerp:GetPercent()
      self:SetPercent(p)
    end
    return true
  end)
end
