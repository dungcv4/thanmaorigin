Require("Script/EventSystem/EventNotify.lua")

local function ClearEffect(nNpcId, isStop)
  local meNpcId = me.GetNpc().nId
  local pNpc = KNpc.GetById(nNpcId)
  local doing = pNpc.GetAction()
  if doing == Npc.Doing.common then
    local pNpc = KNpc.GetById(nNpcId)
    pNpc.RestoreAction()
  end
  if nNpcId == meNpcId then
    Loading:CloseLoadWindow()
  end
end

local function PlayNpcEffect(nNpcId, nEffectId)
  Client.RepresentEvent.PlayNpcEffect(nNpcId, nEffectId, 0, 0, 0)
end

  end

end

function ClientUtil:GetNpcDefaultRes(nNpcId)
  local pNpc = KNpc.GetById(nNpcId)
  if nil ~= pNpc then
    local tbSexFactionInitInfo = GetPlayerInitInfo(pNpc.nFaction, pNpc.GetBodySex())
    if nil == tbSexFactionInitInfo then
      return nil
    end
    local nBodyResId = tbSexFactionInitInfo.nBodyResId
    local nHeadResId = tbSexFactionInitInfo.nHeadResId
    local nFaceId = tbSexFactionInitInfo.nFaceId
    local nWeaponResId = tbSexFactionInitInfo.nWeaponResId
    local nSex = tbSexFactionInitInfo.nSex
    return nBodyResId, nHeadResId, nFaceId, nWeaponResId, nSex
  end
  return nil
end

function ClientUtil:GetNpcLogicData(nNpcId)
  local pNpc = KNpc.GetById(nNpcId)
  if nil ~= pNpc then
    return pNpc.nTemplateId, pNpc.nLevel, pNpc.i64MaxLife, pNpc.i64CurLife
  end
  return nil
end

function ClientUtil:GetHorseResId(nNpcId)
  local pNpc = KNpc.GetById(nNpcId)
  local nHorseId = pNpc.GetHorseResId()
  return nHorseId or 0
end

function ClientUtil:DoCommonAct(nNpcId, nActID, nLoop, nFrame)
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc then
    return
  end
  pNpc.DoCommonAct(nActID, 0, nLoop or 0, nFrame or 0)
end

function ClientUtil:RemoveNpc(nNpcId)
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc then
    return
  end
  pNpc.Delete()
end

function ClientUtil:MemoryWarning()
  self.bMemoryWarning = true
  Client:DeepGC()
  Client.AssetResourceModule.ReleaseCacheLoader(100)
  Client.AssetResourceModule.ReleaseAudioCacheLoader(100)
  Client:AutoGC(900)
end

function ClientUtil:OnMainPlayerRevive()
  Ui.SceneCameraController.SetGray(false)
  local pMeNpc = me.GetNpc()
  if not pMeNpc then
    return
  end
  local pNpc = Client.NpcManager.GetNpc(pMeNpc.nId)
  if not pNpc then
    return
  end
  if not Map:IsTongMap(me.nMapTemplateId) then
    self:StopAutoGoto()
  end
  local bIsLeadFightRunning = AFK:IsLeadFightRunning()
  if bIsLeadFightRunning then
    local bIsFuBen = Map:IsFuben(me.nMapTemplateId)
    if not bIsFuBen then
      if Map:IsTongMap(me.nMapTemplateId) and TongChallenge:IsChanllengeRunning() then
        return
      elseif WuLinGaoShou:IsPlayerInActivityStart() then
        return
      elseif Ui:GetIsAutoBackFightPosAfterRevive() then
        return
      else
        AFK:Stop()
      end
    end
  end
end

function ClientUtil:OnMainPlayerLevelUp()
  local pMeNpc = me.GetNpc()
  if not pMeNpc then
    return
  end
  local pNpc = Client.NpcManager.GetNpc(pMeNpc.nId)
  if not pNpc then
    return
  end
  pNpc:PlayCommonEffect(115)
end

function ClientUtil:SetMusicVolume(nValue, bSave)
  local nOldValue = Ui.PlayerPrefs.GetFloat("MusicVolume", 100)
  nValue = nValue or nOldValue
  Client.AudioModule.SetMusicVolume(nValue)
  if bSave then
    Ui.PlayerPrefs.SetFloat("MusicVolume", nValue)
    if Ui.GameEnv.OpenAudioDisable then
      if 0 == nValue and 0 == Ui.PlayerPrefs.GetFloat("SoundVolume", 100) then
        Map:StopMapSound()
        if not Client.AudioModule.SetEnable(false) then
          Map:RestartPlayMapSound()
          Log("Failed to disable AudioModule")
        end
      elseif not Client.AudioModule.IsEnable() and Client.AudioModule.SetEnable(true) then
        Map:RestartPlayMapSound()
      end
    end
  end
  return nOldValue
end

function ClientUtil:SetSoundVolume(nValue, bSave)
  local nOldValue = Ui.PlayerPrefs.GetFloat("SoundVolume", 100)
  nValue = nValue or nOldValue
  Client.AudioModule.SetSoundVolume(nValue)
  if bSave then
    Ui.PlayerPrefs.SetFloat("SoundVolume", nValue)
    if Ui.GameEnv.OpenAudioDisable then
      if 0 == Ui.PlayerPrefs.GetFloat("MusicVolume", 100) and 0 == nValue then
        Map:StopMapSound()
        if not Client.AudioModule.SetEnable(false) then
          Map:RestartPlayMapSound()
          Log("Failed to disable AudioModule")
        end
      elseif not Client.AudioModule.IsEnable() and Client.AudioModule.SetEnable(true) then
        Map:RestartPlayMapSound()
      end
    end
  end
  return nOldValue
end

function ClientUtil:OnQualitySettingChanged()
  if Sdk:GetOsType() == Sdk.eOSType_iOS and self.bMemoryWarning then
    ClientUtil.QualityModule.SetOnceLoadResourceCount(1)
  end
end
