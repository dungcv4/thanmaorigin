local tbWnd = Ui:CreateWindow("UIResourceTips")
local tbControls = {
  Template = "imgBg/Scroll View/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self:SetClickEmptyCallback()
  self.tbListView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(tbRank)
  self:Update(tbRank)
end

function tbWnd:Update(tbRank)
  self.tbListView:Clear()
  if 0 == #tbRank then
    return
  end
  for i = 1, #tbRank do
    self.tbListView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtRank", i)
      tbElement.pPanel:Label_SetText("txtNum", tbRank[i].nResource)
      tbElement.pPanel:Label_SetText("txtName", tbRank[i].szName)
    end)
  end
end

function tbWnd:OnClickEmptyClose()
  self:Close()
end

function tbWnd:OnClose()
end
