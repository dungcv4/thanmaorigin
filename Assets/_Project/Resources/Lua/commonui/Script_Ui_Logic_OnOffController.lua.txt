Require("Script/EventSystem/EventNotify.lua")
Require("CommonScript/OnOffControllerDef.lua")
OnOffController.tbGMOnOff = {}

function OnOffController:Init(tbOnOff)
  self.tbGMOnOff = tbOnOff
end

local tbEvent = {
  [OnOffController.KEY.CADPA] = EventNotify.emNOTIFY_CADPA_ONOFF,
  [OnOffController.KEY.KeFu] = EventNotify.emNOTIFY_KEFU_ONOFF,
  [OnOffController.KEY.LimitTimeLevelUp] = EventNotify.emNOTIFY_LIMITTIME_LEVELUP_ONOFF,
  [OnOffController.KEY.SevenDayWelfare] = EventNotify.emNOTIFY_SEVENDAYWELFARE_ONOFF,
  [OnOffController.KEY.GiftCode] = EventNotify.emNOTIFY_GIFTCODE_ONOFF,
  [OnOffController.KEY.ShopLevelLimit] = EventNotify.emNOTIFY_SHOP_LEVEL_LIMIT_ONOFF,
  [OnOffController.KEY.GeneralWebView] = EventNotify.emNOTIFY_GENERAL_WEB_VIEW_ONOFF,
  [OnOffController.KEY.RankingServer] = EventNotify.emNOTIFY_RANKING_SERVER_ONOFF,
  [OnOffController.KEY.TreasureChest] = EventNotify.emNOTIFY_TREASURE_CHEST_ONOFF
}

function OnOffController:Handle(tbOnOff)
  for nKey, bOnOff in pairs(tbOnOff) do
    if self.tbGMOnOff[nKey] ~= nil and self.tbGMOnOff[nKey] == bOnOff then
      Log("GM try to change same on-off key:", nKey)
    elseif self.tbGMOnOff[nKey] ~= nil and self.tbGMOnOff[nKey] ~= bOnOff then
      self.tbGMOnOff[nKey] = bOnOff
      Lib:CallBack({
        EventNotify.OnNotify,
        tbEvent[nKey]
      })
    else
      LogErr("GM find on-off key illegal ", nKey)
      return
    end
  end
end

function OnOffController:Check(nKey)
  return self.tbGMOnOff[nKey]
end
