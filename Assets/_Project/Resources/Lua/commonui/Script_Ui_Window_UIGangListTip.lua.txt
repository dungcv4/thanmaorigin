local tbWnd = Ui:CreateWindow("UIGangListTip")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_FIRSTTONG_SYNC_FINAL,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_FIRSTTONG_SYNC_CONTRACT,
      self.UpdatePanel,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  Template = "node/imgBG/Content/Scroll View/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  self:UpdatePanel()
  FirstTong:RequestSyncContract()
end

function tbWnd:OnClose()
end

function tbWnd:UpdatePanel()
  if FirstTong.tbContractInviteList then
    for _, tbTong in ipairs(FirstTong.tbContractInviteList) do
      if tbTong[1] == me.dwTongId or tbTong[2] == me.dwTongId then
        return
      end
    end
  end
  self.tbTongRankList = {}
  if FirstTong.tbFinalMatchData then
    self.tbTongRankList = FirstTong.tbFinalMatchData[FirstTong.FINAL_PAHSE.PAHSE64]
  end
  local nIndex = 0
  self.tbScrollView:Clear()
  for nInviteTongId, tbData in pairs(FirstTong.tbContractInviteList or {}) do
    local bInvite, bInvalid = false, false
    for nTongId, nResult in pairs(tbData) do
      if nTongId == me.dwTongId then
        bInvite = true
      end
      if 1 == nResult then
        bInvalid = true
      end
    end
    if bInvite and not bInvalid then
      for nRank, tbTongData in ipairs(self.tbTongRankList) do
        if tbTongData.nTongId == nInviteTongId then
          nIndex = nIndex + 1
          self.tbScrollView:SetElement(nIndex, function(tbElement)
            tbElement.pPanel:Label_SetText("txtName", tbTongData.szTongName or "")
            tbElement.pPanel:Label_SetText("txtDistinguish", tbTongData.szServerName or "")
            tbElement.pPanel:Label_SetText("txtRank", nRank)
            Ui:Button_BindEvent(tbElement.pPanel, "btnFight", function()
              FirstTong:RequestAgreeContract(nInviteTongId)
            end)
          end)
        end
      end
    end
  end
end
