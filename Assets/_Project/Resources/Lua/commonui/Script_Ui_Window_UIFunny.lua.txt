local tbWnd = Ui:CreateClass("UIFunny", "UIPanel")
local tbControls = {
  btnHelp = "txtTitle/btnHelp",
  Template = "List02/List/Viewport/Content/Template",
  txtTime = "txtTime01",
  btnGo = "btnGo",
  txtSecondTittle = "List02/txtSecondTittle",
  TexBuyNum = "TexBuyNum",
  txtHasSubmitItemNum = "CostItem/txtNum",
  UIItemGrid02 = "UIItemGrid02",
  Tex_Info01 = "Tex_Info01",
  btnView = "btnView"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end,
  btnGo = function(self)
    local tbExtraData = ActivityMgr.EntranceSchedule:GetExtraData(self.nEntranceIndex, self.nIndex)
    if tbExtraData then
      Shop:OpenShopBuyWid(tbExtraData.nItemId, true, "LimitBuy")
    end
  end,
  btnView = function(self)
    local tbExtraData = ActivityMgr.EntranceSchedule:GetExtraData(self.nEntranceIndex, self.nIndex)
    if tbExtraData then
      Ui:OpenWindow("UIReward_Newyear", tbExtraData.nAchieveActId)
    end
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  self.nIndex = self.nIndex or nIndex
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return 0
  end
  self.nActivityId = tbInfo.nId
  self.szHelpKey = tbInfo.szHelpKey
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnView, {
    RedPointMgr.EVENT.EarlySummerGift
  })
  self.pPanel:SetActive(tbControls.UIItemGrid02, false)
  self.pPanel:SetActive(tbControls.Tex_Info01, false)
  self.pPanel:SetActive(tbControls.TexBuyNum, false)
  self.pPanel:Button_SetText(tbControls.btnGo, i18n.EasyGet("\229\137\141\229\190\128\232\180\173\228\185\176"), "Text")
  self:UpdateActivityInfo()
end

function tbWnd:UpdateActivityInfo()
  local pInstance = ActivityMgr:GetInstanceById(self.nActivityId)
  if pInstance.nStartTime > GetTime() then
    local nDiff = pInstance.nStartTime - GetTime()
    self.pPanel:Label_SetText(tbControls.txtTime, i18n.EasyGet("<color=#1FFB24>{0}</color>", i18n.Parse(pInstance.tbConfig.szShowTime)))
    self.pPanel:Label_SetText(tbControls.txtTime .. "/txtTimeTittle01", i18n.EasyGet("{0}\229\144\142\231\187\147\230\157\159", Lib:TimeDesc2(nDiff)))
  else
    local nRestTime = pInstance:GetEndTime() - GetTime()
    self.pPanel:Label_SetText(tbControls.txtTime, i18n.EasyGet("<color=#1FFB24>{0}</color>", i18n.Parse(pInstance.tbConfig.szShowTime)))
    self.pPanel:Label_SetText(tbControls.txtTime .. "/txtTimeTittle01", i18n.EasyGet("{0}\229\144\142\231\187\147\230\157\159", Lib:TimeDesc2(nRestTime)))
  end
  local tbExtraData = ActivityMgr.EntranceSchedule:GetExtraData(self.nEntranceIndex, self.nIndex)
  if tbExtraData then
    self.pPanel:Label_SetText(tbControls.txtSecondTittle, i18n.EasyGet("{0}", i18n.Get(tbExtraData.szInfo)))
    if tbExtraData.nShowAwardKey and tbExtraData.nShowItemId then
      local tbAward = Award:GetAward(tbExtraData.nShowItemId, tbExtraData.nShowAwardKey)
      self:UpdateListView(tbAward, tbControls.Template)
    end
  end
end

function tbWnd:UpdateListView(tbAward, szTemplate)
  if not tbAward then
    return
  end
  if not self.tbListView then
    self.tbListView = Ui:NewScrollView(self, szTemplate)
  end
  self.tbListView:Clear()
  for i = 1, #tbAward do
    self.tbListView:SetElement(i, function(tbElement)
      self:UpdateListElement(tbElement, tbAward[i])
    end, true)
  end
end

function tbWnd:UpdateListElement(tbElement, tbAward)
  local tbGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  tbGrid:SetGenericItem(tbAward)
  if "Item" == tbAward[1] then
    local szName = Item:GetName(tbAward[2], me)
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  else
    local szName = Currency:GetName(tbAward[1])
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  end
  tbGrid:SetClickEnable(true)
end

function tbWnd:OnClose()
end
