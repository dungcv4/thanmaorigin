function GeneralProcess:JoinGamePlayFormMessageBox(szMsg, tbNeedCloseUi, fnContinue, fnClose, nCloseTime)
  if nil == nCloseTime then
    nCloseTime = 15
  end
  if nil == szMsg then
    if GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
      szMsg = i18n.EasyGet("\229\189\147\229\137\141\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186\239\188\140\230\152\175\229\144\166\229\137\141\229\190\128\229\174\137\229\133\168\229\140\186\231\166\187\229\188\128\239\188\159")
    else
      szMsg = i18n.EasyGet("\229\189\147\229\137\141\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186\239\188\140\230\152\175\229\144\166\229\137\141\229\190\128\229\174\137\229\133\168\229\140\186\229\143\130\229\138\160\239\188\159")
    end
  end
  if nil == fnClose then
    function fnClose()
      Ui:CloseWindow("UIMessageBox")
    end
  end
  if nil == fnContinue then
    function fnContinue()
    end
  end
  
  local function fnAgree()
    if TeamMgr:IsFollowTeammate() then
      TeamMgr:StopFollowMember()
    end
    AFK:Stop()
    local nX, nY = Map:GetDefaultPos(me.nMapTemplateId)
    AutoPath:GotoAndCallEx(me.nMapTemplateId, nX, nY, fnContinue)
    Ui:CloseWindow("UIMessageBox")
    if tbNeedCloseUi then
      for _, szUiName in pairs(tbNeedCloseUi) do
        Ui:CloseWindow(szUiName)
      end
    end
  end
  
  Ui:OpenWindow("UIMessageBox", szMsg, {
    {fnAgree},
    {fnClose}
  }, {
    i18n.szOk,
    i18n.szCancel
  }, nil, nCloseTime)
end
