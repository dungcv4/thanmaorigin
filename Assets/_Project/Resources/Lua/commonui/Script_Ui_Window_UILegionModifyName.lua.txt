local tbWnd = Ui:CreateWindow("UILegionModifyName")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSave = "node/imgBG/Content/btnSave",
  txtCurrentName = "node/imgBG/Content/txtCurrentName",
  inputNewLegionName = "node/imgBG/Content/inputNewLegionName",
  txtCost = "node/imgBG/Content/Cost/txtCostNum",
  nodeCost = "node/imgBG/Content/Cost",
  txtFree = "node/imgBG/Content/txtFree"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSave = function(self)
    self:OnChangeLegionName()
  end
}

function tbWnd:OnOpen(bFree)
  self.bFree = bFree
  local tbBaseInfo = Legion:GetBaseInfo() or {}
  self.pPanel:Label_SetText(self.txtCurrentName, tbBaseInfo.szName or "")
  self.pPanel:Label_SetText(self.txtCost, Legion.Cfg.nChangNameCost)
  self.pPanel:Input_SetText(self.inputNewLegionName, "")
  self.pPanel:SetActive(self.txtFree, bFree)
  self.pPanel:SetActive(self.nodeCost, not bFree)
end

function tbWnd:OnChangeLegionName()
  if not Legion:CheckMyAuthority(Legion.Def.Authority_ChangeName) then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\230\157\131\233\153\144\232\191\155\232\161\140\230\147\141\228\189\156"))
    return
  end
  local szLegionName = self.pPanel:Input_GetText(self.inputNewLegionName)
  Legion:ChangeName(szLegionName, self.bFree)
  self:Close()
end
