local tbWnd = Ui:CreateWindow("UIWuLin102Course")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnBaoMing = "node/imgBG/Content/BaoMingCell",
  btnYuXuanSai = "node/imgBG/Content/YuXuanSai",
  btnYuXuanSaiSettle = "node/imgBG/Content/YuXuanSaiSettle",
  btnBanJueSai = "node/imgBG/Content/BanJueSai",
  btnBanJueSaiSettle = "node/imgBG/Content/BanJueSaiSettle",
  btnJueSai = "node/imgBG/Content/JueSai",
  btnJueSaiSettle = "node/imgBG/Content/JueSaiSettle",
  txtBanJueSai = "node/imgBG/Content/BanJueSai/imgNomal/txtBanJueSai",
  txtBanJueSaiSettle = "node/imgBG/Content/BanJueSaiSettle/imgNomal/txtBanJueSai",
  JueSai = "node/imgBG/Content/JueSai",
  imgJueSaiArrow = "node/imgBG/Content/imgJueSaiArrow",
  JueSaiSettle = "node/imgBG/Content/JueSaiSettle",
  imgJueSaiSettleArrow = "node/imgBG/Content/imgJueSaiSettleArrow"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnBaoMing = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "Rule", "Enroll")
    self:Close()
  end,
  btnYuXuanSai = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "Rule", "Vote" .. AvenueOfStars.PHASE_PRE)
    self:Close()
  end,
  btnBanJueSai = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "Rule", "Vote" .. AvenueOfStars.PHASE_SEMI)
    self:Close()
  end,
  btnJueSai = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "Rule", "Vote" .. AvenueOfStars.PHASE_FINAL)
    self:Close()
  end,
  btnYuXuanSaiSettle = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "GetVoteRank", AvenueOfStars.PHASE_PRE)
    self:Close()
  end,
  btnBanJueSaiSettle = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "GetVoteRank", AvenueOfStars.PHASE_SEMI)
    self:Close()
  end,
  btnJueSaiSettle = function(self)
    Ui:OpenWindow("UIWuLin102Rule", "GetVoteRank", AvenueOfStars.PHASE_FINAL)
    self:Close()
  end
}

function tbWnd:OnOpen()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.nTimer = Timer:Register(Env.GAME_FPS, self.UpdateTime, self)
  self:InitView()
  self:InitTime()
  self:UpdateTime()
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:InitView()
  self.pPanel:SetActive(tbControls.JueSai, false)
  self.pPanel:SetActive(tbControls.JueSaiSettle, false)
  self.pPanel:SetActive(tbControls.imgJueSaiArrow, false)
  self.pPanel:SetActive(tbControls.imgJueSaiSettleArrow, false)
  self.pPanel:Label_SetText(tbControls.txtBanJueSai, i18n.EasyGet("\229\134\179\232\181\155\n\232\175\132\230\175\148"))
  self.pPanel:Label_SetText(tbControls.txtBanJueSaiSettle, i18n.EasyGet("\229\134\179\232\181\155\n\231\187\147\231\174\151"))
end

function tbWnd:InitTime()
  self:InitSingleTime(tbControls.btnBaoMing, AvenueOfStars.ENROLL_DATE_MD)
  self:InitSingleTime(tbControls.btnYuXuanSai, AvenueOfStars.VOTE_DATE_MD[AvenueOfStars.PHASE_PRE])
  self:InitSingleTime(tbControls.btnYuXuanSaiSettle, AvenueOfStars.SETTLE_DATE_MD[AvenueOfStars.PHASE_PRE])
  self:InitSingleTime(tbControls.btnBanJueSai, AvenueOfStars.VOTE_DATE_MD[AvenueOfStars.PHASE_SEMI])
  self:InitSingleTime(tbControls.btnBanJueSaiSettle, AvenueOfStars.SETTLE_DATE_MD[AvenueOfStars.PHASE_SEMI])
  self:InitSingleTime(tbControls.btnJueSai, AvenueOfStars.VOTE_DATE_MD[AvenueOfStars.PHASE_FINAL])
  self:InitSingleTime(tbControls.btnJueSaiSettle, AvenueOfStars.SETTLE_DATE_MD[AvenueOfStars.PHASE_FINAL])
end

function tbWnd:UpdateTime()
  local bContinue = false
  local nGameTime = GetTime()
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnBaoMing, AvenueOfStars.ENROLL_DATE) or bContinue
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnYuXuanSai, AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_PRE]) or bContinue
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnYuXuanSaiSettle, AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_PRE]) or bContinue
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnBanJueSai, AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_SEMI]) or bContinue
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnBanJueSaiSettle, AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_SEMI]) or bContinue
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnJueSai, AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_FINAL]) or bContinue
  bContinue = self:UpdateSingleTime(nGameTime, tbControls.btnJueSaiSettle, AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_FINAL]) or bContinue
  if not bContinue then
    self.nTimer = nil
  end
  return bContinue
end

function tbWnd:InitSingleTime(szControl, tbDate)
  self.pPanel:Label_SetText(string.format("%s/imgNomal/txtTime", szControl), string.format("%s-%s", tbDate[1], tbDate[2]))
end

function tbWnd:UpdateSingleTime(nGameTime, szControl, tbDate)
  local nStartTime = Lib:ParseConfigDateTime(tbDate[1])
  local nEndTime = Lib:ParseConfigDateTime(tbDate[2])
  local bNormal = false
  local bHL = false
  local bGrey = false
  if nGameTime < nStartTime then
    bNormal = true
  elseif nGameTime < nEndTime then
    self.pPanel:Label_SetText(string.format("%s/imgHL/txtTime", szControl), Lib:TimeDesc2(nEndTime - nGameTime))
    bHL = true
  else
    bGrey = true
  end
  self.pPanel:SetActive(string.format("%s/imgNomal", szControl), bNormal)
  self.pPanel:SetActive(string.format("%s/imgHL", szControl), bHL)
  self.pPanel:SetActive(string.format("%s/imgGrey", szControl), bGrey)
  if bGrey then
    return false
  end
  return true
end
