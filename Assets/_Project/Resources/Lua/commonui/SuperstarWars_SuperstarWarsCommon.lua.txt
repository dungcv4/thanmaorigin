function SuperstarWars:IsSysOpen()
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local nNowTime = GetTime()
  if nNowTime < Lib:GetConfigDate2Time(self.Cfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.Cfg.END_DATE) then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  return true
end

function SuperstarWars:IsStageOpen(nStage)
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local tbTimeSetting = self.tbTimeSettings[nStage]
  if not tbTimeSetting then
    return false, i18n.EasyGet("\228\184\141\229\156\168\230\180\187\229\138\168\230\156\159\233\151\180")
  end
  local nNowTime = GetTime()
  if nNowTime < Lib:GetConfigDate2Time(tbTimeSetting.nTimeStart) or nNowTime > Lib:GetConfigDate2Time(tbTimeSetting.nTimeEnd) then
    return false, i18n.EasyGet("\228\184\141\229\156\168{0}\230\156\159\233\151\180", i18n.Parse(tbTimeSetting.szDesc))
  end
  return true
end

function SuperstarWars:GetStage()
  if not self.IS_SYS_OPEN then
    return
  end
  local nNowTime = GetTime()
  for nStage, tbTimeSetting in ipairs(self.tbTimeSettings) do
    if nNowTime >= Lib:GetConfigDate2Time(tbTimeSetting.nTimeStart) and nNowTime <= Lib:GetConfigDate2Time(tbTimeSetting.nTimeEnd) then
      return nStage
    end
  end
end

function SuperstarWars:GetWinningRate(tbData)
  local nWinCount = self:GetWinCount(tbData)
  local nWinningRate = 50
  if #tbData[self.PLAYER_DATA.WIN_COUNT] > 0 then
    nWinningRate = math.floor(nWinCount / #tbData[self.PLAYER_DATA.WIN_COUNT] * 100)
  end
  return nWinningRate
end

function SuperstarWars:GetWinCount(tbData)
  local nWinCount = 0
  for _, nWin in pairs(tbData[self.PLAYER_DATA.WIN_COUNT]) do
    if nWin == self.RESULT.WIN then
      nWinCount = nWinCount + 1
    end
  end
  return nWinCount
end

function SuperstarWars:GetFailCount(tbData)
  local nFailCount = 0
  for _, nFail in pairs(tbData[self.PLAYER_DATA.WIN_COUNT]) do
    if nFail == self.RESULT.FAIL then
      nFailCount = nFailCount + 1
    end
  end
  return nFailCount
end

function SuperstarWars:GetDrawCount(tbData)
  local nDrawCount = 0
  for _, nDraw in pairs(tbData[self.PLAYER_DATA.WIN_COUNT]) do
    if nDraw == self.RESULT.DRAW then
      nDrawCount = nDrawCount + 1
    end
  end
  return nDrawCount
end

function SuperstarWars:CheckGmWarch(pPlayer)
  local nGmCard = pPlayer.GetItemCountByTypeInBag(self.Cfg.GM_CARD_ID)
  return nGmCard > 0
end
