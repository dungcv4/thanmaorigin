CanaryRelease.tbSwitchSetting = Lib:EasyLoadTabFile("Setting/CanaryRelease/CanarySwitch.tab", {"nId"})

function CanaryRelease:CheckSwitch(nId, nWeekDay)
  local tbSetting = self.tbSwitchSetting[nId]
  if not tbSetting then
    return false
  end
  if not OnOffController:Check(OnOffController.KEY[tbSetting.szOnOffKey]) then
    return false
  end
  nWeekDay = nWeekDay or Lib:GetLocalWeekDay()
  if not string.find(tbSetting.szWeekDay, tostring(nWeekDay)) then
    return false
  end
  return true
end
