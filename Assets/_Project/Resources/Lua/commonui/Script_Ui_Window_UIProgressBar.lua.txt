local WND_NAME = "UIProgressBar"
local tbWnd = Ui:CreateWindow(WND_NAME)
Ui.ProgressBar = {}
Ui.ProgressBar._instance = tbWnd

function Ui.ProgressBar:OpenAutoProgressBar(szTitleName, nFinishTime, funCallBack, argsCallback)
  if nFinishTime < 0 then
    LogErr("Error! nFinishTime Can not less than 0 !!")
    return
  end
  self._funCallBack = funCallBack
  self._argsCallBack = argsCallback
  Ui:OpenWindowWithCallback(WND_NAME, function()
    local ins = Ui:GetWindow(WND_NAME)
    local MaxProcess = ins.cscript:GetMaxProcess()
    local nSpeed = MaxProcess / nFinishTime
    if szTitleName and "" ~= szTitleName then
      ins.cscript:SetTitle(i18n.Parse(szTitleName))
    end
    ins.cscript:Init()
    ins.cscript:SetSpeed(nSpeed)
    ins.cscript:SetTarget(MaxProcess)
    AFK:WaitTime(nFinishTime / 1000 + 1)
  end)
end

function Ui.ProgressBar:OpenManualProgressBar(szTitleName)
  Ui:OpenWindowWithCallback(WND_NAME, function()
    local ins = Ui:GetWindow(WND_NAME)
    if szTitleName and "" ~= szTitleName then
      ins.cscript:SetTitle(szTitleName)
    end
    ins.cscript:Init()
  end)
end

function Ui.ProgressBar:SetProgressBarWithTarget(nTarget, nFinishTime, funCallBack, argsCallback)
  if nFinishTime < 0 then
    LogErr("Error! nFinishTime Can not less than 0 !!")
    return
  end
  if nTarget < 0 then
    LogErr("Error! nTarget Can not less than 0 !!")
    return
  end
  self._funCallBack = funCallBack
  self._argsCallBack = argsCallback
  Ui:OpenWindow(WND_NAME)
  local ins = Ui:GetWindow(WND_NAME)
  if not ins then
    return
  end
  local MaxProcess = ins.cscript:GetMaxProcess()
  if nTarget > MaxProcess then
    LogErr("Error! nTarget Can not larger than MAX_PROCESS !!", nTarget, MaxProcess)
    return
  end
  local currentProcess = ins.cscript:GetCurrentProcess()
  local nSpeed = math.abs(nTarget - currentProcess) / nFinishTime
  ins.cscript:SetSpeed(nSpeed)
  ins.cscript:SetTarget(nTarget)
end

function Ui.ProgressBar:GetInstance()
  return self._instance
end

function Ui.ProgressBar:GetCallBackFunction()
  return self._funCallBack
end

function Ui.ProgressBar:Close()
  self._funCallBack = nil
  self._argsCallBack = nil
  Ui:CloseWindow(WND_NAME)
  AFK:WaitTime(1)
end

function Ui.ProgressBar:OnProgressFinished()
  if self._funCallBack ~= nil then
    if nil ~= self._argsCallBack then
      self._funCallBack(table.unpack(self._argsCallBack))
    else
      self._funCallBack()
    end
    self._funCallBack = nil
    self._argsCallBack = nil
  end
end

function tbWnd:OnOpen()
  tbWnd.cscript = self.pPanel:GetComponent(WND_NAME)
end

function tbWnd:Close()
  Ui:CloseWindow(WND_NAME)
end
