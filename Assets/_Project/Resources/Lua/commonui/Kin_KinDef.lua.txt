function Kin:GetKinActivitySalaryFaction(nWeekActive)
  local bRet = 0
  for nActiveNum, nMultiple in pairs(Kin.Cfg.tbAllActivitySalary) do
    if nActiveNum <= nWeekActive and nMultiple > bRet then
      bRet = nMultiple
    end
  end
  return bRet
end

local function Array2Map(params)
  local map = {}
  for _, key in pairs(params) do
    map[key] = true
  end
  return map
end
