function YeLanGuan:TryUseItem(dwTemplateID)
  local tbInfo = YeLanGuan.Cfg.Item[dwTemplateID]
  if not tbInfo then
    return
  end
  if 0 == tbInfo.nLandType then
    RemoteServer.RequestYeLanGuan("UseItem", dwTemplateID)
    return
  end
  local nMapId, nX, nY = me.GetWorldPos()
  for nIndex, tbPos in ipairs(YeLanGuan.tbBuildings) do
    if math.abs(nX - tbPos.nPosX) < YeLanGuan.nLandRange and math.abs(nY - tbPos.nPosY) < YeLanGuan.nLandRange then
      RemoteServer.RequestYeLanGuan("UseItem", dwTemplateID)
      return
    end
  end
  me.CenterMsg(i18n.Get(YeLanGuan.Cfg.Msg.CANNOTUSE))
end

function YeLanGuan:RequestSignUp()
  if DegreeCtrl:GetDegree(me, "YeLanGuan") > 0 then
    local function fCall()
      if TeamMgr:HasTeam() then
        local function fLeaveTeamAndSignUp()
          TeamMgr:Quite()
          
          RemoteServer.RequestYeLanGuan("TrySignUp")
        end
        
        Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\232\191\155\229\133\165\229\164\156\233\152\145\229\133\179\233\156\128\232\166\129\233\128\128\229\135\186\233\152\159\228\188\141\239\188\140\231\161\174\229\174\154\231\166\187\233\152\159\239\188\159"), {
          {fLeaveTeamAndSignUp},
          {}
        }, {
          i18n.szOk,
          i18n.szCancel
        })
      else
        RemoteServer.RequestYeLanGuan("TrySignUp")
      end
    end
    
    local bRet, szErr = ActivityCheck:CheckCanEnterActivityEx(me, nil, fCall, false, false, true, nil, {
      "UIBattleEntrance_Common"
    }, false)
    if not bRet then
      me.CenterMsg(szErr)
      return
    end
    return true
  else
    me.CenterMsg(i18n.EasyGet("\229\137\169\228\189\153\229\143\130\228\184\142\230\172\161\230\149\176\228\184\141\232\182\179"))
  end
end

function RaceMap:OpenItemList()
  if not Ui:WindowVisible("UIHUD_Tip") then
    local tbData = {}
    for dwTemplateID, tbInfo in pairs(YeLanGuan.Cfg.Item) do
      local tbTemp = {
        dwTemplateID = dwTemplateID,
        szIcon = "UI/icon/item/other/junxiang.png",
        nPrice = tbInfo.nPrice,
        nSort = tbInfo.nSort
      }
      table.insert(tbData, tbTemp)
    end
    table.sort(tbData, function(tbTempA, tbTempB)
      return tbTempA.nSort < tbTempB.nSort
    end)
    Ui:OpenWindow("UIHUD_Tip", tbData, YeLanGuan.nMoney, YeLanGuan.TryUseItem, YeLanGuan)
  end
end

local function SortByScore(tbDataA, tbDataB)
  if tbDataA.nScore ~= tbDataB.nScore then
    return tbDataA.nScore > tbDataB.nScore
  end
  return tbDataA.dwPlayerId > tbDataB.dwPlayerId
end

function RaceMap:OnSynData(nTime, nMoney, tbRank, nState)
  YeLanGuan.nMoney = nMoney
  table.sort(tbRank, SortByScore)
  if nState == YeLanGuan.STATE.PREPARE then
    self:ShowPrepareUI(nTime)
  else
    local tbTime, tbContent = self:FormatContent(nTime, nMoney, tbRank, nState)
    self.tbTime = tbTime
    self.tbContent = tbContent
    FubenUIController:OpenDefaultFuBenUI("YeLanGuanHelp", {tbTime}, tbContent)
    EventNotify.OnNotify(EventNotify.emNOTIFY_UI_UIHUDTIP_UPDATE, nMoney)
  end
  if nState == YeLanGuan.STATE.START then
    self:OpenItemList()
  end
  if nState == YeLanGuan.STATE.END then
    Ui:CloseWindow("UIHUD_Tip")
  end
end

function RaceMap:FormatContent(nTime, nMoney, tbRank, nState)
  nState = nState or 1
  local tbTime = {}
  table.insert(tbTime, nTime)
  table.insert(tbTime, self.tbState2Tittle[nState]())
  local tbContent = {}
  local szMoney = i18n.EasyGet("<color=#{0}>\230\139\165\230\156\137\229\134\155\233\165\183\230\149\176\239\188\154{1}</color>", Ui.tbColor:GetColorHexByDefineName("BrightYellow"), nMoney)
  table.insert(tbContent, szMoney)
  local szTeamTitle = i18n.EasyGet("\233\152\159\228\188\141\231\167\175\229\136\134\230\142\146\229\144\141")
  local szPlayerTitle = i18n.EasyGet("\231\142\169\229\174\182\231\167\175\229\136\134\230\142\146\229\144\141")
  local tbCampScore = {
    [YeLanGuan.CAMP.UP] = {nTotalScore = 0, nMaxScore = 0},
    [YeLanGuan.CAMP.DOWN] = {nTotalScore = 0, nMaxScore = 0}
  }
  local nUpMax = 0
  local nDownMax = 0
  local nMyCamp = 0
  for nRank, tbInfo in ipairs(tbRank) do
    local szTemp = "\n" .. nRank .. "." .. string.format("%s    %d", tbInfo.szName, tbInfo.nScore)
    if tbInfo.szName == me.szName then
      nMyCamp = tbInfo.nCamp
      szTemp = string.format("<color=#%s>%s</color>", Ui.tbColor:GetColorHexByDefineName("NewOrange"), szTemp)
    end
    szPlayerTitle = szPlayerTitle .. szTemp
    tbCampScore[tbInfo.nCamp].nTotalScore = tbCampScore[tbInfo.nCamp].nTotalScore + tbInfo.nScore
    if tbInfo.nScore > tbCampScore[tbInfo.nCamp].nMaxScore then
      tbCampScore[tbInfo.nCamp].nMaxScore = tbInfo.nScore
    end
  end
  local szUpDesc = self.tbCamp2Desc[YeLanGuan.CAMP.UP](tbCampScore[YeLanGuan.CAMP.UP].nTotalScore)
  local szDownDesc = self.tbCamp2Desc[YeLanGuan.CAMP.DOWN](tbCampScore[YeLanGuan.CAMP.DOWN].nTotalScore)
  if nMyCamp == YeLanGuan.CAMP.UP then
    szUpDesc = string.format("<color=#%s>%s</color>", Ui.tbColor:GetColorHexByDefineName("BrightYellow"), szUpDesc)
  else
    szDownDesc = string.format("<color=#%s>%s</color>", Ui.tbColor:GetColorHexByDefineName("BrightYellow"), szDownDesc)
  end
  local szResult = szTeamTitle .. "\n" .. szUpDesc .. "\n" .. szDownDesc
  if tbCampScore[YeLanGuan.CAMP.UP].nTotalScore < tbCampScore[YeLanGuan.CAMP.DOWN].nTotalScore then
    szResult = szTeamTitle .. "\n" .. szDownDesc .. "\n" .. szUpDesc
  elseif tbCampScore[YeLanGuan.CAMP.UP].nTotalScore == tbCampScore[YeLanGuan.CAMP.DOWN].nTotalScore and tbCampScore[YeLanGuan.CAMP.UP].nMaxScore < tbCampScore[YeLanGuan.CAMP.DOWN].nMaxScore then
    szResult = szTeamTitle .. "\n" .. szDownDesc .. "\n" .. szUpDesc
  end
  table.insert(tbContent, szResult)
  table.insert(tbContent, szPlayerTitle)
  return tbTime, tbContent
end

function RaceMap:ShowRank(tbRank)
  local tbData = {}
  tbData.tbTitle = {
    i18n.EasyGet("\230\142\146\229\144\141"),
    i18n.EasyGet("\229\144\141\229\173\151"),
    i18n.EasyGet("\230\156\128\229\144\142\228\184\128\229\135\187"),
    i18n.EasyGet("\231\167\175\229\136\134")
  }
  tbData.szPanelName = i18n.EasyGet("\229\164\156\233\152\145\229\133\179")
  table.sort(tbRank, SortByScore)
  tbData.tbRankData = {}
  for nRank, tbInfo in ipairs(tbRank) do
    tbData.tbRankData[nRank] = tbData.tbRankData[nRank] or {}
    tbData.tbRankData[nRank].nParam1 = nRank
    tbData.tbRankData[nRank].nParam2 = tbInfo.szName
    tbData.tbRankData[nRank].nParam3 = tbInfo.nKillNums
    tbData.tbRankData[nRank].nParam4 = tbInfo.nScore
    if tbInfo.szName == me.szName then
      tbData.nMyRank = nRank
    end
  end
  Ui:OpenWindow("UIBattleRankQingming", tbData)
end

function RaceMap:StartCheckShowTitle(nMyTitle)
  self.nMyTitle = nMyTitle
  if self.nCheckTimer then
    Timer:Close(self.nCheckTimer)
  end
  self:OnCheckShowTitle()
  self.nCheckTimer = Timer:Register(Env.GAME_FPS, self.OnCheckShowTitle, self)
end

function RaceMap:OnCheckShowTitle()
  if not self.nMyTitle then
    return true
  end
  if me.nMapTemplateId ~= YeLanGuan.Cfg.nRaceMapTemplate then
    self.nCheckTimer = nil
    return false
  end
  local tbNpc = KNpc.GetNpcListInCurrentMap()
  if not tbNpc then
    return true
  end
  for nIndex, pNpc in ipairs(tbNpc) do
    if pNpc.nTitleID ~= self.nMyTitle and 0 == pNpc.nPlayerID then
      pNpc.SetTitle("")
    end
  end
  return true
end
