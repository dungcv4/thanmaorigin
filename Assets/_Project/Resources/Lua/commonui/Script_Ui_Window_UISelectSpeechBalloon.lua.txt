local tbWnd = Ui:CreateWindow("UISelectSpeechBalloon")
local tbControls = {
  BalloonTemplate = "Scroll View/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls
local tbIndex2BalloonID = {}

function tbWnd:OnCreate()
  self.nCurIndex = 0
  self.tbScrollView = Ui:NewScrollView(self, tbControls.BalloonTemplate, function(self, nIndex, tbElement)
    self:OnSelectBalloon(nIndex)
  end, true, true)
end

function tbWnd:OnOpen()
  SpeechBalloon:SyncDataReq()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.NewSpeechBalloon)
end

function tbWnd:Update()
  local bShowLocked = true
  local tbWnd = Ui:GetWindow("UISelectPortrait")
  if tbWnd then
    bShowLocked = tbWnd:GetLockedStatus()
  end
  self:InitAllBalloon(bShowLocked)
end

function tbWnd:OnClose()
  if self.nCurIndex and 0 ~= self.nCurIndex then
    local pCurPanel = self.tbScrollView:GetElement(self.nCurIndex).pPanel
    pCurPanel:SetActive("imgSelected", false)
  end
  self.nCurIndex = nil
  self.tbRedPointIndex = {}
  self.tbScrollView:Clear()
end

function tbWnd:OnSelectBalloon(nIndex)
  if not nIndex then
    Log("PortraitFrame nIndex is Nil")
  end
  local pPanel = self.tbScrollView:GetElement(nIndex).pPanel
  local nBalloonID = tbIndex2BalloonID[nIndex].nFrameIndex
  local bLocked = tbIndex2BalloonID[nIndex].bLocked
  if self.nCurIndex and 0 ~= self.nCurIndex then
    local pCurPanel = self.tbScrollView:GetElement(self.nCurIndex).pPanel
    pCurPanel:SetActive("imgSelected", false)
  end
  pPanel:SetActive("imgSelected", true)
  self.nCurIndex = nIndex
  if not bLocked then
    pPanel:SetActive("imgRedPoint", false)
    SpeechBalloon.tbRedPoint[nBalloonID] = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_PORTRAITUI_UPDATE)
end

function tbWnd:InitAllBalloon(bShowLocked)
  self.tbScrollView:Clear()
  local tbResult = {}
  local tbLockedResult = {}
  for nId, tbInfo in pairs(SpeechBalloon.tbSetting) do
    if 1 ~= tbInfo.nBackup then
      if SpeechBalloon:CheckValid(me, nId, SpeechBalloon.tbData) then
        table.insert(tbResult, {nFrameIndex = nId, bLocked = false})
      elseif bShowLocked then
        table.insert(tbLockedResult, {nFrameIndex = nId, bLocked = true})
      end
    end
  end
  if bShowLocked then
    for i = 1, #tbLockedResult do
      table.insert(tbResult, tbLockedResult[i])
    end
  end
  tbIndex2BalloonID = tbResult
  self:_SetElement(tbResult)
end

function tbWnd:_SetElement(tbShowList)
  for index, tbInfo in ipairs(tbShowList) do
    self.tbScrollView:SetElement(index, function(tbElement)
      local tbSetting = SpeechBalloon.tbSetting[tbInfo.nFrameIndex]
      local bLocked = tbInfo.bLocked
      local pPanel = tbElement.pPanel
      pPanel:Sprite_SetSprite("ImgItem", "UI/atlas/UIChatBox/" .. tbSetting.Total)
      pPanel:Label_SetText("txtName", i18n.Parse(tbSetting.szName))
      if bLocked then
        pPanel:SetActive("imgLocked", true)
      else
        pPanel:SetActive("imgLocked", false)
      end
      if (not self.nCurIndex or 0 == self.nCurIndex) and tbSetting.nIndex == SpeechBalloon:GetPlayerSpeechBalloonId(me) then
        self.nCurIndex = index
      end
      if index == self.nCurIndex then
        pPanel:SetActive("imgSelected", true)
      else
        pPanel:SetActive("imgSelected", false)
      end
      pPanel:SetActive("imgRedPoint", false)
      if SpeechBalloon.tbRedPoint[tbInfo.nFrameIndex] then
        pPanel:SetActive("imgRedPoint", true)
      end
    end)
  end
  if self.nCurIndex and self.nCurIndex > #tbShowList then
    self.nCurIndex = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_PORTRAITUI_UPDATE)
end

function tbWnd:ReqChange()
  if not self.nCurIndex or 0 == self.nCurIndex then
    return
  end
  SpeechBalloon:ChangeReq(tbIndex2BalloonID[self.nCurIndex].nFrameIndex)
end

function tbWnd:GetCurBalloonInfo()
  if self.nCurIndex then
    return tbIndex2BalloonID[self.nCurIndex]
  end
end
