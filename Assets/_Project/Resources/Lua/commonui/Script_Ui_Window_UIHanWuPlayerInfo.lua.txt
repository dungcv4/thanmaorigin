local tbWnd = Ui:CreateWindow("UIHanWuPlayerInfo")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  panelEquip = "node/imgBG/Content/PlayerInfo/Equip/UIEquipPanel",
  imgRolePortrait = "node/imgBG/Content/PlayerInfo/Attrib/Info/PortraitObj/Portrait/Image",
  txtLevel = "node/imgBG/Content/PlayerInfo/Attrib/Info/PortraitObj/Level/Text",
  imgFactionIcon = "node/imgBG/Content/PlayerInfo/Attrib/Info/PortraitObj/imgFactionIcon",
  txtRoleName = "node/imgBG/Content/PlayerInfo/Attrib/Info/txtRoleName",
  honorTitle = "node/imgBG/Content/PlayerInfo/Attrib/Info/txtHonorTip/HonorTitle",
  txtHonorTip = "node/imgBG/Content/PlayerInfo/Attrib/Info/txtHonorTip",
  txtHP = "node/imgBG/Content/PlayerInfo/Attrib/Info/txtHPTip/txtHP",
  txtMP = "node/imgBG/Content/PlayerInfo/Attrib/Info/txtMPTip/txtMP",
  txtAttackValue = "node/imgBG/Content/PlayerInfo/Attrib/Info/txtAttackTip/txtAttackValue",
  btnClose = "node/imgBG/btnClose",
  npcModel = "node/imgBG/Content/PlayerInfo/Equip/NpcModel"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbEquipPanel = Ui:CreatePanel("UIEquipPanel", tbControls.panelEquip, self, me.GetNpc().GetBodySex())
  self.tbEquipPanel:UnRegistRedPoint()
  local honorTitleObj = self.pPanel:GetObject(tbControls.honorTitle)
  self.pHonorTitle = honorTitleObj:GetComponent(Ui.UIHonorTitle)
  local npcObj = self.pPanel:GetObject(tbControls.npcModel)
  self.npcModel = npcObj:GetComponent(Ui.UINpcModel)
end

function tbWnd:OnOpen()
  self.tbEquipPanel:OnOpen()
  self:UpdateFightPower()
  self:DoUpdateEquip()
  self:ShowModel()
  self:UpdatePlayerInfo()
end

function tbWnd:OnClose()
  self.npcModel:Release()
  self.tbEquipPanel:OnClose()
end

function tbWnd:DoUpdateEquip()
  self.tbEquipPanel:Reset()
  local tbEquip = me.GetEquips(1)
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    if 0 ~= tbEquip[i - 1] then
      self.tbEquipPanel:SetEquip(i, tbEquip[i - 1])
    end
  end
  if 0 ~= tbEquip[Item.EQUIPPOS_GUANYIN] then
    self.tbEquipPanel:SetEquip(Item.EQUIPPOS_GUANYIN + 1, tbEquip[Item.EQUIPPOS_GUANYIN])
  end
  if 0 ~= tbEquip[Item.EQUIPPOS_MANTLE] then
    self.tbEquipPanel:SetEquip(Item.EQUIPPOS_MANTLE + 1, tbEquip[Item.EQUIPPOS_MANTLE])
  end
  if 0 ~= tbEquip[Item.EQUIPPOS_WUXINGYIN] then
    self.tbEquipPanel:SetEquip(Item.EQUIPPOS_WUXINGYIN + 1, tbEquip[Item.EQUIPPOS_WUXINGYIN])
  end
  self.tbEquipPanel:ResetFrom(Item.EQUIPPOS_MAIN_NUM + 1)
end

function tbWnd:UpdatePlayerInfo()
  local szIcon = PlayerPortrait:GetIcon(me.nPortrait)
  self.pPanel:Sprite_SetSprite(tbControls.imgRolePortrait, szIcon)
  self.pPanel:Label_SetText(tbControls.txtLevel, me.nLevel)
  self.pPanel:Sprite_SetSprite(tbControls.imgFactionIcon, Faction:GetIcon(me.nFaction))
  self.pPanel:Label_SetText(tbControls.txtRoleName, me.szName)
  self:UpdateHonor(me.nHonorStarLevel)
  self:ChangePlayerHp()
  self:ChangePlayerMp()
  self:UpdateAttackValue()
end

function tbWnd:UpdateHonor(nHonorStarLevel)
  if nHonorStarLevel > 0 then
    self.pPanel:Label_SetText(tbControls.txtHonorTip, i18n.EasyGet("\229\164\180\232\161\148"))
  else
    self.pPanel:Label_SetText(tbControls.txtHonorTip, i18n.EasyGet("\230\154\130\230\151\160\229\164\180\232\161\148"))
  end
  local szName, szColor = HonorLevel:GetHonorNameByStarLevel(nHonorStarLevel, false)
  self.pHonorTitle:SetHonorTitle(i18n.Parse(szName), szColor)
end

function tbWnd:UpdateAttackValue()
  local nMin, nMax = me.GetBaseDamage()
  if nMin ~= nMax then
    self.pPanel:Label_SetText(tbControls.txtAttackValue, string.format("%d-%d", nMin, nMax))
  else
    self.pPanel:Label_SetText(tbControls.txtAttackValue, nMin)
  end
end

function tbWnd:ChangePlayerHp()
  self.pPanel:Label_SetText(tbControls.txtHP, me.GetNpc().i64MaxLife)
end

function tbWnd:ChangePlayerMp()
  self.pPanel:Label_SetText(tbControls.txtMP, me.GetNpc().i64MaxMana)
end

function tbWnd:ShowModel()
  local tbFeature, tbEffect = me.GetNpcResInfo()
  self.nNpcModelDir = 0
  self.npcModel:ShowModel(me.nSex, 1, tbFeature[Npc.NpcResPartsDef.npc_part_body], tbFeature[Npc.NpcResPartsDef.npc_part_weapon], tbFeature[Npc.NpcResPartsDef.npc_part_mantle], 0, tbFeature[Npc.NpcResPartsDef.npc_part_head], tbFeature[Npc.NpcResPartsDef.npc_part_wind], tbEffect[Npc.NpcResPartsDef.npc_part_weapon])
end

function tbWnd:UpdateFightPower()
  self.tbEquipPanel:SetFightPower(Stronger.nFightPower or 0)
end
