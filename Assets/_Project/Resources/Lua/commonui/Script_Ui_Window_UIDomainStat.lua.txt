local tbWnd = Ui:CreateClass("UIDomainStat", "UIPanel")
tbWnd.tbControls = {
  goEmpty4Raycast = "Empty4Raycast",
  txtDomainInfo = "imgBg/txtDomainInfo"
}
tbWnd.tbOnClick = {
  goEmpty4Raycast = function(self)
    self:Hide()
  end
}

function tbWnd:Show()
  local tbDomainList = DomainBattle:GetDomain(me.dwTongId)
  if 0 == #tbDomainList then
    self.pPanel:Label_SetText(self.txtDomainInfo, i18n.EasyGet("\229\189\147\229\137\141\230\156\170\229\141\160\233\162\134\233\162\134\229\156\159\239\188\140\232\175\183\229\184\174\228\184\187/\229\137\175\229\184\174\228\184\187\229\156\168\228\184\139\228\184\128\229\156\186\233\162\134\229\156\159\230\136\152\229\189\147\229\164\1694:00-20:50\233\128\137\230\139\169\233\162\134\229\156\159\229\174\163\230\136\152"))
  else
    local tbMsg = {}
    table.insert(tbMsg, i18n.EasyGet("\230\156\172\229\184\174\233\162\134\229\156\176\230\149\176\233\135\143\239\188\154{0}", #tbDomainList))
    local nStarLv = 0
    for _, nMapTid in pairs(tbDomainList) do
      nStarLv = nStarLv + DomainBattle.MAP_FIGHT[nMapTid].STAR_LV
    end
    table.insert(tbMsg, i18n.EasyGet("\230\156\172\229\184\174\233\162\134\229\156\176\230\152\159\231\186\167\239\188\154{0}", nStarLv))
    if DomainBattle.nRank then
      table.insert(tbMsg, i18n.EasyGet("\229\184\174\228\188\154\230\142\146\229\144\141\239\188\154{0}", DomainBattle.nRank))
    end
    self.pPanel:Label_SetText(self.txtDomainInfo, table.concat(tbMsg, "\n"))
  end
  self.pPanel:SetActive("Main", true)
end

function tbWnd:Hide()
  self.pPanel:SetActive("Main", false)
end
