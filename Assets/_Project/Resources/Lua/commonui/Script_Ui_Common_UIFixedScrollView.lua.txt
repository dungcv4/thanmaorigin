local tbWnd = {}
Ui:SetClass("UIFixedScrollView", tbWnd)

function Ui:NewFixedScrollView(tbOwner, ...)
  local tbClass = Ui:NewClass(Ui:GetClass("UIFixedScrollView"))
  tbClass.tbOwner = tbOwner
  tbClass:OnCreate(...)
  return tbClass
end

function tbWnd:OnCreate(tbElementKey, funcInit, funcClick, bNoHighlight)
  self.tbElementList = {}
  self.funcInit = funcInit
  self.funcClick = funcClick
  self.bNoHighlight = bNoHighlight
  for k, v in pairs(tbElementKey) do
    self.tbElementList[k] = Ui:CreateSimplePanel(self.tbOwner, v)
    if self.funcInit then
      self.funcInit(self.tbOwner, k, self.tbElementList[k].pPanel)
    end
  end
end

function tbWnd:SetAllElementEnable(bEnable)
  for k, v in pairs(self.tbElementList) do
    self.tbOwner.pPanel:SetActive(v.szKey, bEnable)
  end
end

function tbWnd:SetElement(key, funcSet)
  local tbElement = self.tbElementList[key]
  funcSet(tbElement)
  self.tbElementList[key] = tbElement
  if self.funcClick then
    Ui:Button_BindEvent(self.tbOwner.pPanel, self.tbElementList[key].szKey, self.OnClickElement, self, key)
  end
end

function tbWnd:SelectElement(key)
  self.OnClickElement(self.tbOwner, key)
end

function tbWnd:OnClickElement(key)
  if not self.bNoHighlight then
    for k, v in pairs(self.tbElementList) do
      self.tbOwner.pPanel:Button_SetEnable(v.szKey, k ~= key)
    end
  end
  if self.funcClick then
    self.funcClick(self.tbOwner, key, self.tbElementList[key])
  end
end
