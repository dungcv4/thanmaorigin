local tbWnd = Ui:CreateClass("UIMoneyTree", "UIPanel")
local tbControls = {
  btnOne = "btnOne",
  btnTen = "btnTen",
  btnHelp = "btnHelp",
  txtFree = "txtFree",
  imgOneCost = "imgOneCost",
  txtOneCost = "imgOneCost/txtOneCost",
  txtTenCost = "imgTenCost/txtTenCost",
  GlodPot = "GlodPot",
  Once_Effect = "GlodPot/Once_Effect",
  Ten_Effect = "GlodPot/Ten_Effect",
  btnFortuneRecord = "btnFortuneRecord"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnOne = function(self)
    MoneyTree:UICallShakeOne()
  end,
  btnTen = function(self)
    MoneyTree:UICallShakeTen()
  end,
  btnHelp = function()
    Ui:OnHelpClicked("MoneyTree")
  end,
  btnFortuneRecord = function(self)
    self.tbOwner:ShowFortuneRecord()
  end
}

function tbWnd:OnCreate()
  self.pPanel:Label_SetText(tbControls.txtOneCost, tostring(MoneyTree.Cfg.nCostYuanBaoCountOnce))
  self.pPanel:Label_SetText(tbControls.txtTenCost, tostring(MoneyTree.Cfg.nCostYuanBaoCountOnce * MoneyTree.TEN_COUNT))
  self:UpdateView()
end

function tbWnd:UpdateView(tbRandomIndexArray)
  local bHaveFreeTimes = MoneyTree:HaveFreeTimes()
  self.pPanel:SetActive(tbControls.txtFree, bHaveFreeTimes)
  self.pPanel:SetActive(tbControls.imgOneCost, not bHaveFreeTimes)
  if tbRandomIndexArray then
    if 1 == #tbRandomIndexArray then
      self:ShowEffect("one")
    elseif #tbRandomIndexArray == MoneyTree.TEN_COUNT then
      self:ShowEffect("ten")
    end
  end
end

function tbWnd:OnOpen()
  self:UpdateView()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnOne, {
    RedPointMgr.EVENT.MoneyTreeFree
  })
  self.pPanel:SetActive(tbControls.Once_Effect, false)
  self.pPanel:SetActive(tbControls.Ten_Effect, false)
  local Animator = self.pPanel:GetAnimator(tbControls.GlodPot)
  Animator:Play("Await")
  self.pPanel:SetActive(tbControls.btnFortuneRecord, true)
end

function tbWnd:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnOne)
end

function tbWnd:ShowEffect(szType)
  local Animator = self.pPanel:GetAnimator(tbControls.GlodPot)
  if self.nEffectTimer then
    Timer:Close(self.nEffectTimer)
    self.nEffectTimer = nil
    self.pPanel:SetActive(tbControls.Once_Effect, false)
    self.pPanel:SetActive(tbControls.Ten_Effect, false)
  end
  if "one" == szType then
    Animator:Play("Work", -1, 0)
    self.pPanel:SetActive(tbControls.Once_Effect, true)
    self.nEffectTimer = Timer:Register(Env.GAME_FPS * 3, function()
      self.pPanel:SetActive(tbControls.Once_Effect, false)
      Animator:Play("Await")
    end)
  elseif "ten" == szType then
    self.pPanel:SetActive(tbControls.Ten_Effect, true)
    Animator:Play("Work", -1, 0)
    self.nEffectTimer = Timer:Register(Env.GAME_FPS * 5, function()
      self.pPanel:SetActive(tbControls.Ten_Effect, false)
      Animator:Play("Await")
    end)
  end
end
