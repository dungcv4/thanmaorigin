local tbWnd = Ui:CreateWindow("UILowEffect")
tbWnd.tbControls = {SceneColor = "SceneColor"}
tbWnd.tbSceneColorParams = {
  [1] = {
    0,
    30,
    56,
    100,
    100,
    20,
    100
  },
  [2] = {
    0,
    60,
    255,
    150,
    60,
    2.5,
    10
  }
}

function tbWnd:OnCreate()
  self:SetGroup(2)
  self:HideSceneColor()
end

function tbWnd:OnOpen()
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:ShowSceneColor(nStartR, nStartG, nStartB, nStartA, nEndA, nDuration, nCloseSecond)
  self.pPanel:SetActive(self.SceneColor, true)
  self.pPanel:Sprite_SetColor(self.SceneColor, nStartR, nStartG, nStartB, nStartA)
  self.pPanel:Sprite_SetAlphaAnimation(self.SceneColor, nStartA, nEndA, nDuration)
  if not nCloseSecond or nCloseSecond <= 0 then
    return
  end
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.nTimer = Timer:Register(Env.GAME_FPS * nCloseSecond, function()
    self:HideSceneColor()
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end)
end

function tbWnd:ShowSceneColorByID(nResId)
  if not self.tbSceneColorParams[nResId] then
    return
  end
  local tbParam = self.tbSceneColorParams[nResId]
  self:ShowSceneColor(tbParam[1], tbParam[2], tbParam[3], tbParam[4], tbParam[5], tbParam[6], tbParam[7])
end

function tbWnd:HideSceneColor()
  self.pPanel:SetActive(self.SceneColor, false)
end
