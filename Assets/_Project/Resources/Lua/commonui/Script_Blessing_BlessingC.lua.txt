function Blessing:ApplyNewBlessing(bReplace)
  RemoteServer.RequestBlessing("C2S_ApplyNewBlessing", bReplace)
end

function Blessing:ApplyNewBlessingSuccess()
  EventNotify.OnNotify(EventNotify.emNOTIFY_BLESSING_CHANGE)
  self:StartTimer()
end

function Blessing:ClientApplyPlayerAttrib()
  local nBlessingId = me.GetUserValue(self.KEY_GROUP, self.KEY_BLESSING_ID)
  local nLevel = me.GetUserValue(self.KEY_GROUP, self.KEY_PLAYER_LEVEL)
  self:ApplyAttrib(me.GetNpc(), nBlessingId, nLevel)
end

function Blessing:ClientRemovePlayerAttrib()
  local nBlessingId = me.GetUserValue(self.KEY_GROUP, self.KEY_BLESSING_ID)
  local nLevel = me.GetUserValue(self.KEY_GROUP, self.KEY_PLAYER_LEVEL)
  self:RemoveAttrib(me.GetNpc(), nBlessingId, nLevel)
end

function Blessing:OnPlayerFirstLoginToday()
  if 1 ~= GetTimeFrameState(self.Cfg.OPEN_TIMEFRAME) then
    return
  end
  if me.nLevel < self.Cfg.LEVEL_LIMIT then
    return
  end
  if self.bShowNotifyMsg ~= false then
    self:NotifyMsg()
  end
end

function Blessing:NotifyMsg()
  local tbMsgData = {}
  tbMsgData.szType = "Blessing"
  local nNextDayBlessTime = self:GetNextDayBlessTime()
  local nTimeOut = nNextDayBlessTime > 43200 and 43200 or nNextDayBlessTime
  tbMsgData.nTimeOut = GetTime() + nTimeOut
  Ui:SynNotifyMsg(tbMsgData)
end

function Blessing:NotifyTimeOut()
  self.bShowNotifyMsg = true
end

function Blessing:Init()
  if self.bHasInitialized then
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self)
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, Blessing)
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, Blessing)
  end
  EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self.OnPlayerFirstLoginToday, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, Blessing.OnPlayerReturnToLogin, Blessing)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, Blessing.OnPlayerLogin, Blessing)
  self.bHasInitialized = true
end

function Blessing:OnPlayerLogin()
  if 1 ~= GetTimeFrameState(self.Cfg.OPEN_TIMEFRAME) then
    return
  end
  if me.nLevel < self.Cfg.LEVEL_LIMIT then
    return
  end
  if 0 ~= self:GetBlessingId(me) then
    self:StartTimer()
  end
  if self.bShowNotifyMsg then
    self:NotifyMsg()
    self.bShowNotifyMsg = false
  end
end

function Blessing:OnPlayerReturnToLogin()
  self:CloseTimer()
  self.bShowNotifyMsg = nil
end

function Blessing:CloseTimer()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function Blessing:StartTimer()
  self:CloseTimer()
  self.nTimer = Timer:Register(Env.GAME_FPS, function()
    if Blessing:CheckBlessingTimeOut(me) then
      RemoteServer.RequestBlessing("C2S_ClientCheckTimeOut")
    end
    if 0 == self:GetBlessingId(me) then
      self:NotifyMsg()
      return false
    else
      return true
    end
  end)
end

Blessing:Init()
