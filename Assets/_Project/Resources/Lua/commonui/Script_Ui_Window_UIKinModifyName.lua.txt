local tbWnd = Ui:CreateWindow("UIKinModifyName")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSave = "node/imgBG/Content/btnSave",
  txtCurrentName = "node/imgBG/Content/txtCurrentName",
  inputNewKinName = "node/imgBG/Content/inputNewKinName",
  txtCost = "node/imgBG/Content/Cost/txtCostNum",
  nodeCost = "node/imgBG/Content/Cost",
  txtFree = "node/imgBG/Content/txtFree"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSave = function(self)
    self:OnChangeKinName()
  end
}

function tbWnd:OnOpen(bFree)
  self.bFree = bFree
  local tbBaseInfo = Kin:GetBaseInfo() or {}
  self.pPanel:Label_SetText(self.txtCurrentName, tbBaseInfo.szName or "")
  self.pPanel:Label_SetText(self.txtCost, Kin.Cfg.nChangNameCost)
  self.pPanel:Input_SetText(self.inputNewKinName, "")
  self.pPanel:SetActive(self.txtFree, bFree)
  self.pPanel:SetActive(self.nodeCost, not bFree)
end

function tbWnd:OnChangeKinName()
  if not Kin:CheckMyAuthority(Kin.Def.Authority_ChangeName) then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\230\157\131\233\153\144\232\191\155\232\161\140\230\147\141\228\189\156"))
    return
  end
  local szKinName = self.pPanel:Input_GetText(self.inputNewKinName)
  Kin:ChangeName(szKinName, self.bFree)
  self:Close()
end
