local tbWnd = Ui:CreateWindow("UIKinSetAnnounce")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_KIN_DATA,
      self.onKinSyncData
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSaveChange = "node/imgBG/Content/btnSaveChange",
  inputKinNotice = "node/imgBG/Content/KinNotice/inputKinNotice",
  txtTitle = "node/imgBG/txtTitle"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSaveChange = function(self)
    if self.szType == "PublicDeclare" then
      self:OnChangePublicDeclare()
    elseif self.szType == "AddDeclare" then
      self:OnChangeAddDeclare()
    end
  end
}

function tbWnd:OnOpen(szTitle, szType)
  self.szType = szType
  self.pPanel:Label_SetText(self.txtTitle, szTitle)
  self.pPanel:Input_SetText(self.inputKinNotice, "")
  if self.szType == "PublicDeclare" then
    local tbBaseInfo = Kin:GetBaseInfo()
    if tbBaseInfo then
      self.pPanel:Input_SetText(self.inputKinNotice, tbBaseInfo.szPublicDeclare or "")
    end
  elseif self.szType == "AddDeclare" then
    Kin:UpdateRecruitSetting()
    local tbRecruitSetting = Kin:GetRecruitSetting()
    if tbRecruitSetting then
      self.pPanel:Input_SetText(self.inputKinNotice, tbRecruitSetting.szAddDeclare or "")
    end
  end
end

function tbWnd:onKinSyncData(szDataType, nRemainTime)
  if "RecruitSetting" == szDataType and Ui:WindowVisible("UIKinSetAnnounce") and self.szType == "AddDeclare" then
    local tbRecruitSetting = Kin:GetRecruitSetting()
    self.pPanel:Input_SetText(self.inputKinNotice, tbRecruitSetting.szAddDeclare)
  end
end

function tbWnd:OnChangePublicDeclare()
  if not Kin:CheckMyAuthority(Kin.Def.Authority_EditPubilcDeclare) then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\230\157\131\233\153\144\232\191\155\232\161\140\230\147\141\228\189\156"))
    return
  end
  local szKinNotice = self.pPanel:Input_GetText(self.inputKinNotice)
  Kin:ChangePublicDeclare(szKinNotice)
  self:Close()
end

function tbWnd:OnChangeAddDeclare()
  if not Kin:CheckMyAuthority(Kin.Def.Authority_EditRecuitInfo) then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\230\157\131\233\153\144\232\191\155\232\161\140\230\147\141\228\189\156"))
    return
  end
  local szKinNotice = self.pPanel:Input_GetText(self.inputKinNotice)
  Kin:ChangeAddDeclare(szKinNotice)
  self:Close()
end
