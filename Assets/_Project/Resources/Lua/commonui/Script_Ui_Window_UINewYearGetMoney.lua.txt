local tbWnd = Ui:CreateClass("UINewYearGetMoney", "UIFestival_GetMoneyBase")
local tbControls = {
  btnGo = "btnGo",
  szTitle = "txtTitle",
  szTime = "txtTime",
  btnReceive = "btnReceive",
  btnAdd = "btnAdd",
  txtBtnInfoTime = "txtBtnInfoTime",
  btnHelp = "txtTitle/btnHelp",
  btnView = "btnView"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnGo = function(self)
    local pInstance = ActivityMgr:GetInstanceById(self.nAddMonsterActId)
    local tbMapPos = pInstance and pInstance.tbConfig and pInstance.tbConfig.tbMapPos
    if tbMapPos then
      for nMapId, tbInfo in pairs(tbMapPos) do
        for key, tbPos in pairs(tbInfo) do
          ActivityMgr:GotoDynamicNpc(nMapId, tbPos.nPosX, tbPos.nPosY)
          break
        end
      end
    end
  end,
  btnReceive = function(self)
    Ui:OpenWindow("UIFeedBack_Tip", self.nCollectPointActId)
  end,
  btnAdd = function(self)
    Ui:OpenWindow("UIAccess_Tip", nil, table.unpack(self.tbItemList))
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end,
  btnView = function(self)
    Ui:OpenWindow("UIReward_Newyear", self.nHandInTongAwardActId)
  end
}

function tbWnd:OnCreate()
  self.tbActivitySetting = {
    [self.tbType.Person] = 154,
    [self.tbType.Tong] = 155
  }
  self.tbElementPath = {
    [self.tbType.Person] = {
      txtActivityName = "List/Viewport/Content/Template/txtActivityName",
      txtTimes = "List/Viewport/Content/Template/txtTimes",
      txtInfo = "List/Viewport/Content/Template/txtInfo",
      Template = "List/Viewport/Content/Template/ContList/Viewport/Content/Template",
      btnView = "List/Viewport/Content/Template/btnView"
    },
    [self.tbType.Tong] = {
      txtActivityName = "List/Viewport/Content/Template1/txtActivityName",
      txtTimes = "List/Viewport/Content/Template1/txtTimes",
      txtInfo = "List/Viewport/Content/Template1/txtInfo",
      Template = "List/Viewport/Content/Template1/ContList/Viewport/Content/Template",
      btnView = "List/Viewport/Content/Template1/btnView"
    }
  }
  self.tbItemList = {1350, 1351}
  self.tbItemPath = {
    [1] = "Scroll View/Viewport/Content/Template1",
    [2] = "Scroll View/Viewport/Content/Template2"
  }
  self.tbItemElementPath = {
    UIItemGrid = "UIItemGrid",
    szName = "txtItemCount",
    Nums = "txtItemCountValue"
  }
end

function tbWnd:LoadActivityData()
  self.nCollectPointActId = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].nCollectPointActId
  self.nAddMonsterActId = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].nAddMonsterActId
  self.nHandInTongAwardActId = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].nHandInTongAwardActId
end

function tbWnd:UpdateOtherInfo()
  local pInstance = ActivityMgr:GetInstanceById(self.nCollectPointActId)
  local bOpenExchange = pInstance and pInstance:IsOpenExchange()
  local szStart = ""
  if bOpenExchange then
    local nDiff = Lib:GetDate2Time(pInstance.tbConfig.nEndExchangeTime) - GetTime()
    szStart = i18n.EasyGet("\231\180\175\232\174\161\229\155\158\233\166\136<color=#1FFB24>{0}</color>\229\144\142\229\133\179\233\151\173", Lib:TimeDesc2(nDiff))
  else
    local nDiff = Lib:GetDate2Time(pInstance.tbConfig.nStartExchangeTime) - GetTime()
    szStart = i18n.EasyGet("\231\180\175\232\174\161\229\155\158\233\166\136<color=#1FFB24>{0}</color>\229\144\142\229\188\128\229\144\175", Lib:TimeDesc2(nDiff))
  end
  self.pPanel:Label_SetText(tbControls.txtBtnInfoTime, szStart)
end

function tbWnd:DynamicRegisteRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGo, {
    RedPointMgr.EVENT.NewYearHandIn
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnReceive, {
    RedPointMgr.EVENT.CollectExchange
  })
end

function tbWnd:UnRegisteRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnGo)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnReceive)
end
