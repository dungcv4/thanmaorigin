function Forbid:IsForbidAward()
  local nEndTime = self.nEndTime or 0
  
  return nEndTime < 0 or nEndTime > 0 and nEndTime > GetTime()
end

function Forbid:IsForbidTrade()
  local nEndTime = self.nTradeEndTime or 0
  return nEndTime < 0 or nEndTime > 0 and nEndTime > GetTime()
end

function Forbid:IsForbidFriend()
  local nEndTime = self.nFriendEndTime or 0
  return nEndTime < 0 or nEndTime > 0 and nEndTime > GetTime()
end

function Forbid:OnSyncForbidAwardState(nEndTime, szMsg, bHideBox)
  self.nEndTime = nEndTime
  self.szMsg = szMsg or i18n.EasyGet("\228\184\141\230\173\163\229\189\147\231\154\132\230\184\184\230\136\143\232\161\140\228\184\186")
  local bIsForbid = self:IsForbidAward()
  EventNotify.OnNotify(EventNotify.emNOTIFY_FORBID_STATE_CHANGE, bIsForbid)
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nLeaveTime = self.nEndTime - GetTime() + 1
  self.nTimer = Timer:Register(nLeaveTime * Env.GAME_FPS, function()
    EventNotify.OnNotify(EventNotify.emNOTIFY_FORBID_STATE_CHANGE)
    self.nTimer = nil
    return false
  end)
  if not bHideBox then
    Ui:OpenWindow("MessageBoxBig", self:GetDesc(bIsForbid), {
      {}
    }, {
      i18n.szOk
    }, 3)
  end
end
