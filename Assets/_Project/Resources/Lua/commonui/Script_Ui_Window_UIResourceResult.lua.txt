local tbWnd = Ui:CreateWindow("UIResourceResult")
local tbControls = {
  Template = "imgBg/Scroll View/Viewport/Content/Template",
  OneLegionPanel = "imgBg/Template",
  MoreLegionPanel = "imgBg/Scroll View",
  txtTime = "imgBg/txtTime01"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbListView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(nEndTime, tbBelongData)
  self:Update(nEndTime, tbBelongData)
  self:SetTime()
  self.nMainTimer = Timer:Register(Env.GAME_FPS, self.SetTime, self)
end

function tbWnd:Update(nEndTime, tbBelongData)
  self.tbListView:Clear()
  self.pPanel:SetActive(tbControls.OneLegionPanel, false)
  self.pPanel:SetActive(tbControls.MoreLegionPanel, false)
  if nEndTime then
    if nEndTime < GetTime() then
      return
    else
      self.nEndTime = nEndTime
    end
  end
  if 0 == #tbBelongData then
    return
  end
  if 1 == #tbBelongData then
    self.pPanel:SetActive(tbControls.OneLegionPanel, true)
    self.pPanel:Label_SetText(tbControls.OneLegionPanel .. "/Image/txtName", tbBelongData[1].szName)
    self.pPanel:Label_SetText(tbControls.OneLegionPanel .. "/Image/txtNum", i18n.EasyGet("{0}\228\186\186", tbBelongData[1].nNums))
    return
  end
  self.pPanel:SetActive(tbControls.MoreLegionPanel, true)
  for i = 1, #tbBelongData do
    self.tbListView:SetElement(i, function(tbElement)
      tbElement.pPanel:SetActive("Image/txtState", tbBelongData[i].bBelong)
      tbElement.pPanel:Label_SetText("Image/txtNum", i18n.EasyGet("{0}\228\186\186", tbBelongData[i].nNums))
      tbElement.pPanel:Label_SetText("Image/txtName", tbBelongData[i].szName)
    end)
  end
end

function tbWnd:SetTime()
  local nNow = GetTime()
  if self.nEndTime and nNow < self.nEndTime then
    self.pPanel:SetActive(tbControls.txtTime, true)
    self.pPanel:Label_SetText(tbControls.txtTime, i18n.EasyGet("{0}\231\167\146", self.nEndTime - nNow))
  else
    self.pPanel:SetActive(tbControls.txtTime, false)
  end
  return true
end

function tbWnd:OnClose()
  if self.nMainTimer then
    Timer:Close(self.nMainTimer)
  end
  self.nMainTimer = nil
  self.nEndTime = nil
end
