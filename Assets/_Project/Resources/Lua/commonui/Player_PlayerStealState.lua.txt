Player.nStealStateSaveGroup = GetGroupKey("StealState")
Player.nStealStateLockKey = 1
Player.nStealStateInfoKey = 2
local tbFile = Lib:LoadIniFile("Setting/Skill/SkillSetting.ini")
Player.nStealStateShowSkillId = tonumber(tbFile.StealState.StateShow)

function Player:IsLockStealState(pPlayer)
  return 0 ~= pPlayer.GetUserValue(self.nStealStateSaveGroup, self.nStealStateLockKey)
end
