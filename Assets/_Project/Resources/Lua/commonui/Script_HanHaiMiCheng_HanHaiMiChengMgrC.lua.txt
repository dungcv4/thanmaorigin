function HanHaiMiCheng:AttendReq()
  local bRet, szErr = self:CanAttend()
  
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestHanHaiMiCheng("AttendReq")
end

function HanHaiMiCheng:CanAttend()
  if 1 ~= GetTimeFrameState(self.ATTEND_COND.TIMEFRAME_REQ) then
    return false, i18n.EasyGet("\231\179\187\231\187\159\230\156\170\229\188\128\230\148\190")
  end
  if 0 == me.dwTeamID then
    return false, i18n.EasyGet("\232\175\183\231\187\132\233\152\159\229\143\130\229\138\160")
  end
  local bRet, szErr = ActivityCheck:CheckCanEnterActivityEx(me, nil, nil, nil, false, true)
  if not bRet then
    return false, szErr
  end
  return true
end

function HanHaiMiCheng:AutoPathToKey()
  if not self.tbNearestKeyPos then
    return
  end
  local nTargetX = self.tbNearestKeyPos[1]
  local nTargetY = self.tbNearestKeyPos[2]
  self:CheckAndGoto(me.nMapTemplateId, nTargetX, nTargetY)
end

function HanHaiMiCheng:CheckAndGoto(nMapTID, XPos, YPos)
  if nMapTID and HanHaiMiCheng:IsKeyFightMap(nMapTID) then
    self:GetPaths(XPos, YPos)
    self:GoToNextPos()
    return true
  end
  return false
end

function HanHaiMiCheng:GetPaths(XPos, YPos)
  self.tbPathArr = {}
  table.insert(self.tbPathArr, {XPos = XPos, YPos = YPos})
  if me.nFightMode == Npc.FIGHT_MODE.emFightMode_None then
    table.insert(self.tbPathArr, self:GetNearestTPPoint(XPos, YPos))
  end
end

function HanHaiMiCheng:GetNearestTPPoint(XPos, YPos)
  local tbTrapTargets = self.TRAPSETTING
  local nMinDis, szTrapName
  for szName, tbPos in pairs(tbTrapTargets) do
    local nDis = Lib:GetDistance(XPos, YPos, tbPos.XPos, tbPos.YPos)
    if not nMinDis or nMinDis > nDis then
      nMinDis = nDis
      szTrapName = szName
    end
  end
  return self.TRAPNPCSETTING[szTrapName]
end

function HanHaiMiCheng:PauseAutoCtrl()
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  if AFK:IsLeadFightRunning() then
    AFK:Stop()
  end
end

function HanHaiMiCheng:GoToNextPos()
  if self:IsGoingTo() then
    local tbPos = table.remove(self.tbPathArr)
    if tbPos then
      self:PauseAutoCtrl()
      AutoPath:GotoAndCallEx(me.nMapId, tbPos.XPos, tbPos.YPos, nil, 0, nil, nil, nil, nil, nil, function()
        self:GoToNextPos()
      end)
    end
  end
end

function HanHaiMiCheng:IsGoingTo()
  local tb = self.tbPathArr
  return nil ~= tb and #tb > 0
end

function HanHaiMiCheng:GetLevelName(nLevel)
  if nLevel == self.tbTeamState.BossFight then
    return i18n.EasyGet("\228\184\128\229\177\130")
  elseif nLevel == self.tbTeamState.KeyFight then
    return i18n.EasyGet("\228\186\140\229\177\130")
  elseif nLevel == self.tbTeamState.BossCompete or nLevel == self.tbTeamState.BossCompeteSucc then
    return i18n.EasyGet("\228\184\137\229\177\130")
  end
  return i18n.EasyGet("\233\155\182\229\177\130")
end

function HanHaiMiCheng:LoadExternSetting()
  local tbTrapList = Lib:LoadTabFileEx("Setting/Map/hanhaimicheng_2/info/trap_setting.txt", "sdd", nil, {
    "Name",
    "PosX",
    "PosY"
  }) or {}
  local tbTrapSetting = {}
  for _, tbPos in ipairs(tbTrapList) do
    tbTrapSetting[tbPos.Name] = {
      XPos = tbPos.PosX,
      YPos = tbPos.PosY
    }
  end
  self.TRAPSETTING = tbTrapSetting
  local tbTrapNpcList = Lib:LoadTabFileEx("Setting/Map/hanhaimicheng_2/info/npc_info.txt", "ssdd", nil, {
    "NpcName",
    "ShowName",
    "XPos",
    "YPos"
  }) or {}
  local tbTrapNpcSetting = {}
  for _, tbPos in ipairs(tbTrapNpcList) do
    tbTrapNpcSetting[tbPos.ShowName] = {
      XPos = tbPos.XPos,
      YPos = tbPos.YPos
    }
  end
  self.TRAPNPCSETTING = tbTrapNpcSetting
end

HanHaiMiCheng:LoadExternSetting()
