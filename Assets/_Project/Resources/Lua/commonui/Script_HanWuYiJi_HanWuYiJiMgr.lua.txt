function HanWuYiJi:CanAttend(pPlayer, bHasTeam)
  if 1 ~= GetTimeFrameState(self.ATTEND_COND.TIMEFRAME_REQ) then
    return false, i18n.EasyGet("\231\179\187\231\187\159\230\156\170\229\188\128\230\148\190")
  end
  if GetTime() >= self.nGameEndTime then
    return false, i18n.EasyGet("\230\180\187\229\138\168\229\183\178\231\187\147\230\157\159")
  end
  if pPlayer.nLevel < self.ATTEND_COND.PLAYER_LEVEL_REQ then
    return false, i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}\231\186\167", self.ATTEND_COND.PLAYER_LEVEL_REQ)
  end
  local bRet, szErr = ActivityCheck:CheckCanEnterActivityEx(pPlayer, nil, nil, nil, nil, true)
  if not bRet then
    return false, szErr
  end
  if bHasTeam then
    if not TeamMgr:HasTeam() then
      return false, i18n.EasyGet("\232\175\183\231\187\132\233\152\159\229\144\142\229\143\130\229\138\160")
    end
    if not TeamMgr:IsCaptain() then
      return false, i18n.EasyGet("\233\152\159\233\149\191\230\137\141\232\131\189\230\138\165\229\144\141")
    end
    local tbMemberList = TeamMgr:GetMembers()
    if #tbMemberList > self.ATTEND_COND.TEAM_MEMBER_NUM_LIMIT then
      return false, i18n.EasyGet("\233\152\159\228\188\141\228\186\186\230\149\176\232\182\133\232\191\135{0}\228\186\186", self.ATTEND_COND.TEAM_MEMBER_NUM_LIMIT)
    end
    for _, tbMember in pairs(tbMemberList) do
      local bRet, szErr = self:CanMemberAttend(pPlayer, tbMember)
      if not bRet then
        return false, szErr
      end
    end
  end
  return true
end

function HanWuYiJi:CanMemberAttend(pPlayer, tbMember)
  local nCapMapId, nCapPosX, nCapPosY = pPlayer.GetWorldPos()
  local nMemMapId, nMemPosX, nMemPosY = tbMember.nMapId, tbMember.nPosX, tbMember.nPosY
  if nMemMapId ~= nCapMapId or (nMemPosX - nCapPosX) ^ 2 + (nMemPosY - nCapPosY) ^ 2 > self.ATTEND_COND.TEAM_MEMBER_DIST_LIMIT ^ 2 then
    return false, i18n.EasyGet("\233\152\159\229\145\152{0}\228\184\141\229\156\168\233\153\132\232\191\145", tbMember.szName)
  end
  if tbMember.nLevel < self.ATTEND_COND.PLAYER_LEVEL_REQ then
    return false, i18n.EasyGet("\233\152\159\229\145\152{0}\231\173\137\231\186\167\228\184\141\232\182\179{1}\231\186\167", tbMember.szName, self.ATTEND_COND.PLAYER_LEVEL_REQ)
  end
  return true
end

function HanWuYiJi:OnEnhanceEquip(nEquipPosIndex, tbHanWuYiJiEquipEnhanceData)
  local bNearNpc = false
  local tbNpcList = KNpc.GetAroundNpcList(me.GetNpc(), GetNpcDialogDistance(self.SHOP_NPCID[1])) or {}
  for _, pNpc in pairs(tbNpcList) do
    if pNpc.nTemplateId == self.SHOP_NPCID[1] then
      bNearNpc = true
      break
    end
  end
  if bNearNpc or me.GetItemCountByTypeInBag(tbHanWuYiJiEquipEnhanceData.nCostItemTemplatedId) >= tbHanWuYiJiEquipEnhanceData.nCostItemNum then
    RemoteServer.RequestHanWuYiJi("ReqEnhanceEquip", nEquipPosIndex)
  else
    me.CenterMsg(i18n.EasyGet("\230\178\161\230\156\137\232\182\179\229\164\159\231\154\132{0}", (KItem.GetItemShowInfo(tbHanWuYiJiEquipEnhanceData.nCostItemTemplatedId))))
  end
end

function HanWuYiJi:RefreshEnhanceRedPoint()
  for nEquipPosIndex, nEquipPos in ipairs(HanWuYiJi.EquipPosMap) do
    local tbEquipPos = HanWuYiJi.EquipPosMap[nEquipPosIndex]
    local pItem
    for i, nEquipPos in pairs(tbEquipPos) do
      pItem = me.GetEquipByPos(nEquipPos)
      if nil ~= pItem then
        break
      end
    end
    local nEnhanceLv = 0
    if nil ~= pItem then
      nEnhanceLv = Item.EquipEnhance:GetEquipEnhanceLevel(me, pItem.nEquipPos)
    end
    local bIsEnhanceMax = nEnhanceLv >= HanWuYiJi.DEF.MAX_ENCHANCE_LV and true or false
    local tbHanWuYiJiEquipEnhanceData
    if false == bIsEnhanceMax and nil ~= HanWuYiJi.tbHanWuYiJiEquipEnhanceCfg and nil ~= HanWuYiJi.tbHanWuYiJiEquipEnhanceCfg[nEquipPosIndex] then
      for i = nEnhanceLv, HanWuYiJi.DEF.MAX_ENCHANCE_LV do
        if nil ~= HanWuYiJi.tbHanWuYiJiEquipEnhanceCfg[nEquipPosIndex][i + 1] then
          tbHanWuYiJiEquipEnhanceData = HanWuYiJi.tbHanWuYiJiEquipEnhanceCfg[nEquipPosIndex][i + 1]
          break
        end
      end
      if nil ~= tbHanWuYiJiEquipEnhanceData and me.GetItemCountByTypeInBag(tbHanWuYiJiEquipEnhanceData.nCostItemTemplatedId) >= tbHanWuYiJiEquipEnhanceData.nCostItemNum then
        RedPointMgr:LightRedPoint(RedPointMgr.EVENT.HanWuEnhance)
        return
      end
    end
  end
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.HanWuEnhance)
end
