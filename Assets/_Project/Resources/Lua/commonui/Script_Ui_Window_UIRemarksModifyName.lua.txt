local tbWnd = Ui:CreateWindow("UIRemarksModifyName")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSave = "node/imgBG/Content/btnSave",
  inputRemarkName = "node/imgBG/Content/inputRemarkName",
  txtCurrentName = "node/imgBG/Content/txtCurrentName"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSave = function(self)
    self:OnChangeRemarkName()
  end
}

function tbWnd:OnOpen(dwRoleId, szRoleName, szOldRemarkName)
  self.dwRoleId = dwRoleId
  self.pPanel:Label_SetText(self.txtCurrentName, szRoleName)
  self.pPanel:Input_SetText(self.inputRemarkName, szOldRemarkName)
end

function tbWnd:OnChangeRemarkName()
  if not self.dwRoleId then
    self:Close()
    return
  end
  local szRemarkName = self.pPanel:Input_GetText(self.inputRemarkName)
  local nlen = Lib:StrLen(szRemarkName)
  if nlen > 13 then
    local szMsg = i18n.EasyGet("\233\149\191\229\186\166\232\182\133\232\191\135\228\184\138\233\153\144")
    me.CenterMsg(szMsg)
    return
  end
  if nlen > 0 and ReplaceLimitWords(szRemarkName) then
    me.CenterMsg(i18n.EasyGet("\229\144\171\230\156\137\230\149\143\230\132\159\229\173\151\231\172\166\239\188\140\232\175\183\228\191\174\230\148\185\229\144\142\233\135\141\232\175\149"))
    return
  end
  FriendShip:RemarkPlayer(self.dwRoleId, szRemarkName)
  local tbWindow = Ui:GetWindow("UISocial")
  if tbWindow then
    tbWindow:Refresh()
  end
  self:Close()
  me.CenterMsg(i18n.EasyGet("\232\174\190\231\189\174\230\136\144\229\138\159"))
end
