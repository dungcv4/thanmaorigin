local tbWnd = Ui:CreateClass("UISubAttribute", "UIPanel")
local tbControls = {
  Empty4Raycast = "Empty4Raycast",
  txtOtherAttribTemplate = "imgBg/ScrollView/Viewport/Content/Template",
  btnAttributeInfo = "AttributeInfo"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  Empty4Raycast = function(self)
    if self.pPanel:IsActive(tbControls.btnAttributeInfo) then
      self.pPanel:SetActive(tbControls.btnAttributeInfo, false)
      return
    end
    self.pPanel:SetActive("Main", false)
  end
}

function tbWnd:OnCreate()
  self.tbOtherAttribScrollView = Ui:NewScrollView(self, tbControls.txtOtherAttribTemplate, function(self, nIndex, tbElement)
    self.pPanel:SetActive(tbControls.btnAttributeInfo, true)
    local szText = tbOtherAttribute[nIndex].fnDesc(self.pAsyncRole, self.tbAsyncAttr)
    self.pPanel:Label_SetText(tbControls.btnAttributeInfo .. "/imgBg/txtInfo", szText)
    local tbPosition = tbElement.pPanel:GetSelfPosition()
    self.pPanel:Object_SetPosition(tbControls.btnAttributeInfo, Ui.Vector3(tbPosition.x + 3.8, tbPosition.y, tbPosition.z))
    self.tbOtherAttribScrollView:ClearSelected()
  end)
end

function tbWnd:UpdateView(pAsyncData)
  for i = 1, #tbOtherAttribute do
    self.tbOtherAttribScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtAttributeName", string.format("%s:", tbOtherAttribute[i].fnName()))
      pPanel:Label_SetText("txtAttributeValue", tbOtherAttribute[i].func(self.pAsyncRole, self.tbAsyncAttr))
    end)
  end
end
