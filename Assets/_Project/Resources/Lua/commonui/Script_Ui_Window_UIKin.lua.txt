function tbWnd:OnCreate()
  self.tbPanelList = Ui:NewButtonGroup(self, {
    [1] = self.btnKinInfo,
    [2] = self.btnKinMember,
    [3] = self.btnKinRedBag,
    [4] = self.btnKinSalary
  }, true, function(self, nIndex)
    if 1 == nIndex then
      self:onClickSelectKinInfoPanel()
    elseif 2 == nIndex then
      self:onClickSelectKinMemberPanel()
    elseif 3 == nIndex then
      self:onClickSelectKinRedBagPanel()
    elseif 4 == nIndex then
      self:onClickSelectKinSalaryPanel()
    end
  end)
  self.tbRedBagPanel = Ui:CreatePanel("UIRedBagPanel", self.panelKinRedBag, self)
  self.tbKinMemberScrollView = Ui:NewLoopScrollView(self, self.KinMemberTemplate)
  self.tbActiviryListScrollView = Ui:NewScrollView(self, self.listActivityTemplate)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, self.UIMoneyPanel)
  self.itemContribute = Ui:NewExistItemGrid(self, self.DonationItemGrid)
  self.itemGiftBoxReward = Ui:NewExistItemGrid(self, self.GiftBoxItemGrid)
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, self.AddAndSub)
  self.AddAndSubGroup:BindEvent(function(nValue)
    return self:UpdateNumberInput(nValue)
  end)
  self.AddAndSubGroup:SetNumPadPos(-200, -100)
  self.AddAndSubGroup:SetInputValue(1)
  self.tbKinMemberScrollRect = self.pPanel:GetScrollRect(tbControls.KinMemberList)
  self.tbKinMemberScrollRect.onValueChanged:AddListener(function(vec2)
    self:OnSliderList(vec2)
  end)
  self:DealDonateRecordWithOnCreate()
end

function tbWnd:OnOpen(szPage, ...)
  if 0 == me.dwKinId then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\138\160\229\133\165\229\174\182\230\151\143"))
    return 0
  end
  self.pPanel:SetActive(tbControls.btnPrestigeHelp, false)
  self.pPanel:SetActive(tbControls.wndKinPrestigeDetails, false)
  self.nDonateCount = 1
  self.tbMoneyPanel:SetMoneyValue()
  self.szSortType = "Default"
  self.bIsDown = true
  local nPage = self.PAGE_INDEX[szPage]
  if not nPage then
    self.tbPanelList:Select(1)
  else
    self.tbOpenParam = {
      ...
    }
    self.tbPanelList:Select(nPage)
  end
  self:RegistRedPoint()
  self.nCurY = 1
  self.nSliderLimit = nDefaultSlideLimit
  self:OnUpdateQuitBtn()
  if self.nC2sTimer then
    Timer:Close(self.nC2sTimer)
    self.nC2sTimer = nil
  end
  self:SetMonitorUI(true)
end

function tbWnd:OnClose()
  self:UnRegistRedPoint()
  self.pPanel:SetActive(tbControls.btnGoBottom, false)
  self.pPanel:SetActive(tbControls.btnGoTop, false)
  self.pPanel:SetActive(tbControls.wndDonateRecord, false)
  if self.nC2sTimer then
    Timer:Close(self.nC2sTimer)
    self.nC2sTimer = nil
  end
  self.nC2sTimer = Timer:Register(Env.GAME_FPS, function()
    self:SetMonitorUI(false)
  end, self)
end

function tbWnd:SetMonitorUI(bMonitor)
  Kin:AskMonitorUI(bMonitor)
  if self.nC2sTimer then
    Timer:Close(self.nC2sTimer)
    self.nC2sTimer = nil
  end
end

function tbWnd:onKinSyncData(szDataType, nRemainTime)
  if "BaseInfo" == szDataType then
    self:OnUpdateBaseKinInfo()
  elseif "MemberList" == szDataType then
    self:OnUpdateMemberList()
  elseif "Declare" == szDataType then
    local tbBaseInfo = Kin:GetBaseInfo() or {}
    self.pPanel:Label_SetText(self.txtKinNotice, tbBaseInfo.szPublicDeclare or "")
  elseif "GetKinSalary" == szDataType then
    if nRemainTime > 0 then
      self.pPanel:Button_SetEnable(self.btnGetSalary, false)
      self.pPanel:Label_SetText(self.txtLastWeekSalary, 0)
    end
  elseif "Building" == szDataType then
    self:OnUpdateBuildingInfo()
  elseif "Quit" == szDataType then
    self:Close()
  else
    self:UpdateCommData(szDataType)
  end
end

function tbWnd:OnUpdateBaseKinInfo()
  local tbBaseInfo = Kin:GetBaseInfo() or {}
  self.pPanel:Label_SetText(self.txtKinName, tbBaseInfo.szName or "")
  self.pPanel:Label_SetText(self.txtKinMasterName, tbBaseInfo.szMasterName or "")
  local nKinLevel = tbBaseInfo.nLevel or 0
  self.pPanel:Label_SetText(self.txtKinLevel, i18n.EasyGet("{0}\231\186\167", nKinLevel))
  local szMemberDesc = ""
  local szNewerDesc = ""
  if tbBaseInfo.nMemberCount then
    szMemberDesc = string.format("%d/%d", tbBaseInfo.nMemberCount, tbBaseInfo.nMaxMemberCount)
    szNewerDesc = string.format("%d/%d", tbBaseInfo.nNewerCount, tbBaseInfo.nMaxNewerCount)
  end
  self.pPanel:Label_SetText(self.txtMemberCount, szMemberDesc)
  self.pPanel:Label_SetText(self.txtTraineeCount, szNewerDesc)
  self.pPanel:Label_SetText(self.txtKinNotice, tbBaseInfo.szPublicDeclare or "")
  self.pPanel:Label_SetText(self.txtKinFund, tbBaseInfo.nFund)
  self.pPanel:Label_SetText(self.txtOnlineMemberCount, Lib:GetTableLength(Kin:GetMemberState() or {}))
  for nType, szPath in pairs(self.tbBuilding2Txt) do
    self.pPanel:Label_SetText(szPath, i18n.EasyGet("{0}\231\186\167", tbBaseInfo.tbBuilding[nType].nLevel))
  end
  self.pPanel:SetActive(self.btnEditKinNotice, Kin:CheckMyAuthority(Kin.Def.Authority_EditPubilcDeclare))
  self.pPanel:SetActive(self.btnKinAnnounce, Kin:CheckMyAuthority(Kin.Def.Authority_EditRecuitInfo))
end

function tbWnd:OnUpdateMemberList(szSortType)
  if szSortType then
    if self.szSortType == szSortType then
      self.bIsDown = not self.bIsDown
    else
      self.szSortType = szSortType
      self.bIsDown = true
    end
  end
  local tbMemberList = Kin:GetMemberList() or {}
  local tbOnlineMembers = Kin:GetMemberState() or {}
  self.pPanel:Label_SetText(self.txtOnlineMemberCount, Lib:GetTableLength(tbOnlineMembers or {}))
  local fSortFun = Kin:GetSortFunction(self.szSortType)
  if not fSortFun then
    error("UIKin MemberInfoList Sort Error, SortFun don't found!")
    return
  end
  self.tbMemberList = fSortFun(tbMemberList, self.bIsDown)
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local tbMember = tbMemberList[nIndex]
    if not tbMember then
      return
    end
    local tbElem = tbLoopScroll:GetElement(nIndex)
    tbElem.tbMember = tbMember
    local pPanel = tbElem.pPanel
    local bOnLine = tbOnlineMembers[tbMember.nMemberId]
    if tbMember.nMemberId ~= me.dwID then
      Ui:Button_BindEvent(pPanel, "", function()
        Ui:OpenWindowWithCallback("UIRightPopUpMenu2", function(caller, tbWndMenu)
          local pos = self.pPanel:GetTransform("node/imgBG/Content/KinMemberPanel/imgPopupPos").position
          tbWndMenu:SetPos(pos.x, pos.y)
        end, self, "kin", tbMember, self)
      end)
    else
      Ui:Button_UnBindEvent(pPanel, "")
    end
    pPanel:Sprite_SetSprite("imgFaction", Faction:GetIcon(tbMember.nFaction))

function tbWnd:OnUpdateActivityList()
  self.tbActiviryListScrollView:Clear()
  for i = 1, #Kin.Cfg.Salary_Channel_Type do
    self.tbActiviryListScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("ActType/txtTitle", i18n.Get(Kin.Cfg.Salary_Channel_Type[i].Desc))
      tbElement.pPanel:SetActive("ActType/MarkFull", false)
      local activityAllSalary = 0
      for j = 1, #Kin.Cfg.Salary_Channel do
        if Kin:CheckSalaryTimeFrameState(j) and i == Kin.Cfg.Salary_Channel[j].Channel_Type then
          local key = string.format("ActDetail%d", j)
          local item = tbElement.pPanel:CloneObject("ActDetail/item", key)
          tbElement.pPanel:Label_SetText(item .. "/txtActName", i18n.Get(Kin.Cfg.Salary_Channel[j].Title))
          tbElement.pPanel:Label_SetText(item .. "/txtActDesc", i18n.Get(Kin.Cfg.Salary_Channel[j].Desc))
          local degreeName = Kin.Cfg.Salary_Channel[j].DegreeName
          local maxTimes = Kin.Cfg.Salary_Channel[j].MaxTimes
          local salaryOneTimes = Kin.Cfg.Salary_Channel[j].Salary
          local curDegree = DegreeCtrl:GetDegree(me, degreeName)
          local nMaxDegree = DegreeCtrl:GetMaxDegree(degreeName, me)
          if curDegree < 0 then
            curDegree = 0
          end
          local nChange = nMaxDegree - maxTimes
          nChange = Lib:IIF(nChange > 0, nChange, 0)
          local curTimes = maxTimes + nChange - curDegree
          local times = string.format("(%d/%d)", curTimes, maxTimes)
          tbElement.pPanel:Label_SetText(item .. "/txtActName/txtActCount", times)
          local activitySalary = salaryOneTimes * curTimes
          tbElement.pPanel:Label_SetText(item .. "/txtSalary", activitySalary)
          activityAllSalary = activityAllSalary + activitySalary
          if curTimes == maxTimes then
            tbElement.pPanel:SetActive(item .. "/txtFull", true)
          else
            tbElement.pPanel:SetActive(item .. "/txtFull", false)
          end
        end
      end
      tbElement.pPanel:ContentSizeFitter_Refresh("ActDetail")
      self.pPanel:ContentSizeFitter_Refresh(tbElement.szKey)
      local nMaxSalary = Kin:GetMaxSalaryByChannelType(i)
      if activityAllSalary > nMaxSalary then
        activityAllSalary = nMaxSalary
      end
      local activitySalaryPer = math.floor(activityAllSalary * 100 / nMaxSalary)
      tbElement.pPanel:Label_SetText("ActType/txtPercent/txtPercentValue", string.format("%d", activitySalaryPer) .. "%")
      if 100 == activitySalaryPer then
        activitySalaryPer = nMaxSalary
        tbElement.pPanel:SetActive("ActType/MarkFull", true)
        tbElement.pPanel:SetActive("ActType/txtFull", true)
        tbElement.pPanel:SetActive("ActType/txtPercent", false)
        tbElement.pPanel:Label_SetText("ActType/txtFull/txtFullValue", string.format("%d", activitySalaryPer))
      else
        tbElement.pPanel:SetActive("ActType/MarkFull", false)
        tbElement.pPanel:SetActive("ActType/txtFull", false)
        tbElement.pPanel:SetActive("ActType/txtPercent", true)
      end
    end)
  end
  self:ShowKinSalaryInfo()
end

function tbWnd:ShowKinSalaryInfo()
  local myMemberData = Kin:GetMyMemberData()
  local fCareerFaction = 0.0
  local curSalary = 0
  if myMemberData then
    curSalary = myMemberData.nCurWeekSalary or 0
    self.pPanel:Label_SetText(self.txtCurWeekSalary, curSalary)
    local lastWeekSalary = myMemberData.nLastWeekSalary or 0
    self.pPanel:Label_SetText(self.txtLastWeekSalary, lastWeekSalary)
    if lastWeekSalary > 0 then
      self.pPanel:SetActive(self.btnGetSalary, true)
      self.pPanel:Button_SetEnable(self.btnGetSalary, true)
    else
      self.pPanel:SetActive(self.btnGetSalary, true)
      self.pPanel:Button_SetEnable(self.btnGetSalary, false)
    end
    local nCareerTitle = i18n.EasyGet("{0}\229\183\165\232\181\132\229\138\160\230\136\144", i18n.Get(Kin.Def.Career_Name[myMemberData.nCareer]))
    self.pPanel:Label_SetText(self.txtKinCareerFactionTitle, nCareerTitle)
    fCareerFaction = Kin.tbActivityCareerSalary[myMemberData.nCareer] or 0
    self.pPanel:Label_SetText(self.txtKinCareerFaction, string.format("%0.1f", fCareerFaction) .. i18n.EasyGet("\229\128\141"))
  end
  local myKinData = Kin:GetBaseInfo()
  local fWeekActiveFaction = 0.0
  if myKinData then
    local nKinWeekActive = myKinData.nWeekActive
    fWeekActiveFaction = Kin:GetKinActivitySalaryFaction(nKinWeekActive)
    local szWeekActiveFaction = "0"
    if fWeekActiveFaction > 0 then
      szWeekActiveFaction = string.format("%0.1f", fWeekActiveFaction)
    end
    self.pPanel:Label_SetText(self.txtKinWeekActive, string.format("%d", nKinWeekActive))
    self.pPanel:Label_SetText(self.txtKinWeekActiveFaction, szWeekActiveFaction .. i18n.EasyGet("\229\128\141"))
  end
  local finalSalary = curSalary * (fCareerFaction * 100 + fWeekActiveFaction * 100) / 100
  self.pPanel:Label_SetText(self.txtFinalWeekSalary, finalSalary)
end

function tbWnd:UpdateDonation()
  local nLeftDonateCount, nDonationTotalCount = Kin:GetDonationData()
  self.pPanel:Label_SetText(self.txtDonationCount, string.format("%d/%d", nLeftDonateCount, nDonationTotalCount))
  local nCurDonateCount = nDonationTotalCount - nLeftDonateCount
  self.AddAndSubGroup:SetNumRange(1, nLeftDonateCount)
  local nJewelryLevel = Kin:GetBuildingLevel(Kin.Def.Building_Jewelry)
  local nContribut = Kin:GetDonationsContrib(nJewelryLevel, nCurDonateCount + 1, nCurDonateCount + self.nDonateCount)
  local fVipInc = 0
  nContribut = math.ceil(nContribut * (1 + fVipInc))
  self.itemContribute:SetDigitalItem("KinContrib", nContribut)
  local bHasLeftCount = nLeftDonateCount > 0
  self.nDonateCount = bHasLeftCount and math.min(self.nDonateCount, nLeftDonateCount) or 0
  self.AddAndSubGroup:SetInputValue(self.nDonateCount)
  self.nDonateCost = Kin:GetDonationsCost(nJewelryLevel, nCurDonateCount + 1, nCurDonateCount + self.nDonateCount)
  if bHasLeftCount then
    self:UpdateNumberInput(self.nDonateCount)
  end
  self.pPanel:Label_SetText(self.txtDonationCost, self.nDonateCost)
  self.itemContribute.fnClick = self.itemContribute.DefaultClick
  self.pPanel:Button_SetEnable(self.btnDonate, bHasLeftCount)
end

function tbWnd:UpdateNumberInput(nNum)
  if not self:UpdateDonationCount(nNum) then
    local nLeftDonateCount = Kin:GetDonationData()
    if nNum < 0 then
      self:UpdateDonationCount(self.nDonateCount)
      return self.nDonateCount
    elseif nNum > nLeftDonateCount then
      me.CenterMsg(i18n.EasyGet("\230\141\144\231\140\174\230\172\161\230\149\176\228\184\141\232\182\179"))
      self.nDonateCount = nLeftDonateCount
      self:UpdateDonationCount(nLeftDonateCount)
      return nLeftDonateCount
    end
  end
  return nNum
end

function tbWnd:UpdateGiftBox()
  local nNow = GetTime()
  local nLeftCount, nNextBuyTime = Kin:GetGiftBoxData()
  nLeftCount = nLeftCount or 0
  nNextBuyTime = nNextBuyTime or nNow
  self.pPanel:Label_SetText(self.txtGiftBoxCost, Kin.Def.nGiftBoxCost)
  self.pPanel:Label_SetText(self.txtGiftBoxLeftCount, string.format("%d/%d", nLeftCount, Kin.Def.nMaxGiftBoxPerDay + Vip.tbSetting[me.GetVipLevel()].nKinGiftNum))
  self.pPanel:Button_SetEnable(self.btnBuyGiftBox, nLeftCount > 0 and nNow > nNextBuyTime)
  local nJewelryLevel = Kin:GetBuildingLevel(Kin.Def.Building_Jewelry)
  local nItemID = Kin:GetJewelryGiftBoxItemID(nJewelryLevel)
  self.itemGiftBoxReward:SetItemByTemplate(nItemID, 1)
  self.itemGiftBoxReward.fnClick = self.itemGiftBoxReward.DefaultClick
  self.pPanel:Label_SetText(self.txtGiftCountDownTime, i18n.EasyGet("\229\133\145\230\141\162"))

    return
  end

end

function tbWnd:OnSliderList(vec2)
  if vec2.y <= 0 or vec2.y >= 1 then
    return
  end
  if self.bButtonChange then
    self.bButtonChange = false
    self.nSliderLimit = nDefaultSlideLimit
    return
  end
  local y = math.floor(vec2.y * 10) / 10
  local nDiff = math.abs(y - self.nCurY)
  if 0.9 == y or 0 == y then
    self.pPanel:SetActive(tbControls.btnGoTop, false)
    self.pPanel:SetActive(tbControls.btnGoBottom, false)
    return
  end
  if nDiff < self.nSliderLimit then
    return
  end
  if y < self.nCurY then
    self.pPanel:SetActive(tbControls.btnGoTop, false)
    self.pPanel:SetActive(tbControls.btnGoBottom, true)
  elseif y > self.nCurY then
    self.pPanel:SetActive(tbControls.btnGoBottom, false)
    self.pPanel:SetActive(tbControls.btnGoTop, true)
  end
  self.nCurY = y
  self.nSliderLimit = nSwitchBtnLimit
end
