MasterPrentice.lastQueryApply = 0

function MasterPrentice:QueryMyApplies()
  local now = GetTime()
  if now - self.lastQueryApply < self.nQueryInterval then
    return
  end
  self.lastQueryApply = now
  RemoteServer.MasterPrentictRequest("DoQueryMyApplies")
end

function MasterPrentice:OnSyncAllApplies(tbApplies)
  self.Applies = tbApplies
  if Ui:WindowVisible("UITeacherApply") == true then
    Ui:GetClass("UITeacherApply"):Refresh(tbApplies)
  end
end

function MasterPrentice:GetApplies()
  self:QueryMyApplies()
end

function MasterPrentice:OnSyncApply(tbApply)
  local nApplyId = tbApply.nApplyId
  self.Applies = self.Applies or {}
  self.Applies[nApplyId] = tbApply
  if Ui:WindowVisible("UITeacherApply") == true then
    Ui:GetClass("UITeacherApply"):OnOpenEnd()
  else
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MP_TeacherApply)
  end
end

function MasterPrentice:OnSyncDeleteApply(nApplyId)
  for k, v in pairs(self.Applies or {}) do
    if v[1] == nApplyId then
      self.Applies[k] = nil
    end
  end
  if Ui:WindowVisible("UITeacherApply") == true then
    Ui:GetClass("UITeacherApply"):OnOpenEnd()
  end
end

function MasterPrentice:GetAllApply()
  return self.Applies
end

function MasterPrentice:CheckPopUpMenuCanUse(dwRoleId, nLevel, nOtherVipLevel)
  if not dwRoleId then
    return false
  end
  local Relation = self:GetRelation(me, dwRoleId)
  if Relation > 0 then
    return false
  end
  local nDiffLevel = self:GetMinLevelDiff(me.GetVipLevel())
  local nOtherDiffLevel = self:GetMinLevelDiff(nOtherVipLevel or 0)
  nDiffLevel = math.min(nDiffLevel, nOtherDiffLevel)
  if nDiffLevel <= me.nTotalLevel - nLevel then
    local tbLimitLevel = self.Cfg.MASTER_LEVEL
    local nLimitLevel = 0
    for szOpenTime, nLevel in pairs(tbLimitLevel) do
      if 1 == GetTimeFrameState(szOpenTime) then
        nLimitLevel = math.max(nLimitLevel, nLevel)
      end
    end
    if nLimitLevel > me.nTotalLevel then
      return false
    end
    local masterCtrl = self:GetMPInfo()
    if masterCtrl and masterCtrl.tbMyPrentices and #masterCtrl.tbMyPrentices[1] >= self.Def.MAX_PRENTICE_COUNT then
      return false
    end
    return true
  elseif nDiffLevel <= nLevel - me.nTotalLevel then
    if me.nTotalLevel < self.Cfg.PRENTICE_LEVEL then
      return false
    end
    local masterCtrl = self:GetMPInfo()
    if masterCtrl and masterCtrl.tbMyMaster and #masterCtrl.tbMyMaster >= self.Def.MAX_MASTER_COUNT then
      return false
    end
    return true
  end
  return false
end
