local tbWnd = Ui:CreateWindow("UISupplementBox")
local tbControls = {
  AddAndSub = "imgBG/UIAddAndSubGroup",
  btnClose = "imgBG/btnClose",
  btnCancel = "imgBG/btnCancel",
  btnSure = "imgBG/btnSure",
  txtDesc = "imgBG/txtDesc",
  txtCost = "imgBG/Cost/txtCost",
  imgMoney = "imgBG/Cost/imgMoney"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSure = function(self)
    SupplementAward:TrySupplement(self.szKey, self.nCurValue, self.bNormal, self.nPrice * self.nCurValue)
    self:Close()
  end,
  btnCancel = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.AddAndSub)
  self.AddAndSubGroup:BindEvent(function(nValue)
    return self:OnValueUpdate(nValue)
  end)
  self.AddAndSubGroup:SetNumPadPos(-274, 0)
  self.AddAndSubGroup:SetLongPressLoop(true, true, true)
end

function tbWnd:OnOpen(szKey, nMaxCount, nPrice, bNormal)
  local tbInfo = SupplementAward:GetCfgByKey(szKey) or {}
  self.szKey = szKey
  self.nMaxCount = nMaxCount
  self.nCurValue = nMaxCount
  self.bNormal = bNormal
  self.nPrice = nPrice
  self.szName = i18n.Parse(tbInfo.szName)
  self:Update()
end

function tbWnd:Update()
  local txtDesc = i18n.EasyGet("{0}\229\143\175\230\137\190\229\155\158\230\172\161\230\149\176{1}\230\172\161", self.szName, self.nMaxCount)
  self.pPanel:Label_SetText(tbControls.txtDesc, txtDesc)
  if self.bNormal then
    local szIcon = Currency:GetIcon("Coin")
    self.pPanel:Sprite_SetSprite(tbControls.imgMoney, szIcon)
  else
    local szIcon = Currency:GetIcon("YuanBao")
    self.pPanel:Sprite_SetSprite(tbControls.imgMoney, szIcon)
  end
  self.AddAndSubGroup:SetNumRange(1, self.nMaxCount)
  self.AddAndSubGroup:SetInputValue(self.nCurValue)
  self.pPanel:Label_SetText(tbControls.txtCost, self.nCurValue * self.nPrice)
end

function tbWnd:OnValueUpdate(nValue)
  if nValue < 1 then
    nValue = 1
  end
  if nValue > self.nMaxCount then
    nValue = self.nMaxCount
  end
  self.nCurValue = nValue
  self.AddAndSubGroup:SetInputValue(self.nCurValue)
  self.pPanel:Label_SetText(tbControls.txtCost, nValue * self.nPrice)
  return self.nCurValue
end
