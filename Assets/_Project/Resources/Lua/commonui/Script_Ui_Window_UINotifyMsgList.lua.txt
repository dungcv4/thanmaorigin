tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnClear = function(self)
    for i = #Ui.tbNotifyMsgDatas, 1, -1 do
      local tbMsg = Ui.tbNotifyMsgDatas[i]
      if tbMsg.nTimeOut > 0 and self.tbFnProcess[tbMsg.szType] and type(self.tbFnProcess[tbMsg.szType]) == "table" and self.tbFnProcess[tbMsg.szType].fnTimeOutOrClear then
        self.tbFnProcess[tbMsg.szType].fnTimeOutOrClear(self, tbMsg)
      end
    end
    Ui.tbNotifyMsgDatas = {}
    self:RefreshMsg()
    self:Close()
  end
}

function tbWnd:FnProcessChuanGong(tbData, emRelation)
  local emRet, szMsg = ChuanGong:AcceptChuanGong(tbData.dwOtherId, tbData.nType, emRelation, {
    "UINotifyMsgList"
  }, function()
    tbData.nTimeOut = 0
    self:RemoveTimeOutMsg()
    self:Close()
  end)
  if szMsg then
    me.CenterMsg(szMsg)
  end
  return emRet ~= ChuanGong.tbAcceptResultEnum.NeedTimeReturnSafeArea
end

function tbWnd:InitCheck()
  for k, v in pairs(self.tbFnSetData) do
    assert(self.tbFnProcess[k], k)
  end
end

tbWnd:InitCheck()

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.tbScrollView = Ui:NewScrollView(self, self.MsgTemplate, nil, false, true)
end

function tbWnd:OnOpen()
  Ui.nUnReadNotifyMsgNum = 0
  self:RefreshMsg()
  if 0 == #Ui.tbNotifyMsgDatas then
    EventNotify.OnNotify(EventNotify.emNOTIFY_NOTIFY_PROCESS_MSG)
  end
end

function tbWnd:OnClose()
  EventNotify.OnNotify(EventNotify.emNOTIFY_NOTIFY_PROCESS_MSG)
end

function tbWnd:OnFixedUpdate()
  self:RemoveTimeOutMsg()
end

function tbWnd:RefreshMsg()
  self.tbScrollView:Reset()
  for i = 1, #Ui.tbNotifyMsgDatas do
    local tbData = Ui.tbNotifyMsgDatas[i]
    self.tbScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtTitle", "")
      pPanel:Label_SetText("txtContent", "")
      pPanel:Button_SetText("btnView", i18n.EasyGet("\229\137\141\229\190\128"), "Text")
      Ui:Button_BindEvent(pPanel, "btnView", self.onClickView, self, 1, tbData, false)
      Ui:Button_BindEvent(pPanel, "btnView1", self.onClickView, self, 2, tbData, false)
      pPanel:SetActive("btnView", true)
      pPanel:SetActive("btnView1", false)
      pPanel:SetActive("txtTarget", false)
      local fnSetData = self.tbFnSetData[tbData.szType]
      if not fnSetData then
        Log("Error!! NotifyMsgList, szType:", tbData.szType)
        return
      end
      fnSetData(self, pPanel, tbData)
    end)
  end
end

tbWnd.tbNeedFriendDataType = {}

function tbWnd:RemoveTimeOutMsg()
  local nTimeNow = GetTime()
  for i = #Ui.tbNotifyMsgDatas, 1, -1 do
    local tbMsg = Ui.tbNotifyMsgDatas[i]
    if nTimeNow >= tbMsg.nTimeOut then
      if tbMsg.nTimeOut > 0 and self.tbFnProcess[tbMsg.szType] and type(self.tbFnProcess[tbMsg.szType]) == "table" and self.tbFnProcess[tbMsg.szType].fnTimeOutOrClear then
        self.tbFnProcess[tbMsg.szType].fnTimeOutOrClear(self, tbMsg)
      end
      table.remove(Ui.tbNotifyMsgDatas, i)
      self:RefreshMsg()
    elseif self.tbNeedFriendDataType[tbMsg.szType] and not FriendShip:GetFriendDataInfo(tbMsg.dwID) then
      table.remove(Ui.tbNotifyMsgDatas, i)
      self:RefreshMsg()
    end
  end
end

function tbWnd:GetMsgNum()
  return #Ui.tbNotifyMsgDatas, Ui.nUnReadNotifyMsgNum
end

tbWnd.tbSpecialCheck = {}

function tbWnd:onClickView(nButtonIdx, tbData, bNotCheckSafeZone)
  if Ui.bInZone then
    me.CenterMsg(i18n.EasyGet("\232\183\168\230\156\141\230\151\160\230\179\149\232\191\155\232\161\140\229\189\147\229\137\141\230\147\141\228\189\156"))
  end
  if tbData.nTimeOut <= GetTime() then
    me.CenterMsg(i18n.EasyGet("\230\182\136\230\129\175\229\183\178\232\191\135\230\156\159"))
    self:RefreshMsg()
    return
  end
  if 1 == IsAlone() then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\228\184\141\229\133\129\232\174\184\229\143\130\228\184\142\239\188\140\232\175\183\229\133\136\232\191\148\229\155\158\229\174\137\229\133\168\229\140\186\229\134\141\229\176\157\232\175\149\239\188\129"))
    return
  end
  if self.tbSpecialCheck[tbData.szType] then
    local fnCheck = self.tbSpecialCheck[tbData.szType]
    local bRet, szMsg = fnCheck()
    if not bRet then
      me.CenterMsg(szMsg)
      return
    end
  end
  local fnProcess
  local bRet = false
  if type(self.tbFnProcess[tbData.szType]) == "table" then
    if 1 == nButtonIdx then
      fnProcess = self.tbFnProcess[tbData.szType].fn
    elseif 2 == nButtonIdx then
      fnProcess = self.tbFnProcess[tbData.szType].fn1
    end
  else
    fnProcess = self.tbFnProcess[tbData.szType]
  end
  if fnProcess then
    if self.tbNeedCancelFollowMsg[tbData.szType] and TeamMgr:IsFollowTeammate() then
      TeamMgr:StopFollowMember()
    end
    bRet = fnProcess(self, tbData)
  end
  if not bRet then
    return
  end
  tbData.nTimeOut = 0
  print(tbData.nTimeOut)
  self:RemoveTimeOutMsg()
  self:Close()
end
