Require("CommonScript/BlissSharing/BlissSharingCom.lua")
BlissSharing.tbInviteInfo = BlissSharing.tbInviteInfo or {}

function BlissSharing:QueryInviteInfo()
  RemoteServer.RequestBlissSharing("QueryInviteInfo")
end

function BlissSharing:OnQueryInviteInfo(tbResult)
  self.tbInviteInfo = tbResult
  EventNotify.OnNotify(EventNotify.emNOTIFY_BLISSSHARING_QUERY_INVITE_INFO)
end

function BlissSharing:GetAward(szType, nIndex)
  local bRet, szErr = self:CanGetAward(me, szType, nIndex)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestBlissSharing("GetAward", szType, nIndex)
end

function BlissSharing:OnGetAward(szType, nIndex)
  EventNotify.OnNotify(EventNotify.emNOTIFY_BLISSSHARING_GET_AWARD)
end

function BlissSharing:GetInviteAward(szType, nIndex)
  RemoteServer.RequestBlissSharing("GetInviteAward", szType, nIndex)
end

function BlissSharing:GetAllInviteAward()
  RemoteServer.RequestBlissSharing("GetAllInviteAward")
end

function BlissSharing:GenInviteCode()
  local bRet, szErr = self:CanGenInviteCode(me)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestBlissSharing("GenInviteCode")
end

function BlissSharing:UseInviteCode(szInviteCode)
  local bRet, szErr = self:CanPreUseInviteCode(me)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestBlissSharing("UseInviteCode", szInviteCode)
end

function BlissSharing:OnCheckRedPoint(tbFlag)
  if tbFlag[self.RED_POINT_TYPE.SELF_AWARD] then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.BlissSharingSelfAward)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.BlissSharingSelfAward)
  end
  if tbFlag[self.RED_POINT_TYPE.INVITE_AWARD] then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.BlissSharingInviteAward)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.BlissSharingInviteAward)
  end
end

function BlissSharing:OnLogout()
  self.tbInviteInfo = {}
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, BlissSharing)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_LEVEL, BlissSharing)
EventNotify:RegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, BlissSharing.OnLogout, BlissSharing)
