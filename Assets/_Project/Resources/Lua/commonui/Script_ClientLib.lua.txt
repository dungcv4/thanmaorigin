function Lib:ParamsToXLua(tb)
  return table.unpack(tb)
end

local profiler = require("perf.profiler")

function Lib:StartP()
  profiler.start()
end

function Lib:ReportP()
  local ret = profiler.report()
  Log(ret)
end

function Lib:StopP()
  profiler.stop()
end

local memory = require("perf.memory")

function Lib:GetTotalMem()
  Log("TotalMemory:", memory.total())
end

function Lib:SnapShot()
  Log("SnapShot:", memory.snapshot())
end
