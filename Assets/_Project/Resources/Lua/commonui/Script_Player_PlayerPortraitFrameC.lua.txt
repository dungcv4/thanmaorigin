PlayerPortraitFrame.tbRedPointIndex = {}
PlayerPortraitFrame.tbPortraitData = {}

function PlayerPortraitFrame:OnSynData(tbData, nNewPortrait)
  self.tbPortraitData = tbData
  if nNewPortrait then
    self.tbRedPointIndex[nNewPortrait] = true
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.NewPortraitFrame)
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_PROTARIT_FRAME_UPDATE)
end

function PlayerPortraitFrame:OnChangePortraitFrame()
  EventNotify.OnNotify(EventNotify.emNOTIFY_PORTRAIT_FRAME_CHANGE)
end

function PlayerPortraitFrame:ApplyGroupPlayersPFWithCB(szClass, szCBFunc, tbRet)
  local uiClass = Ui:GetClass(szClass)
  if uiClass then
    uiClass[szCBFunc](uiClass, tbRet)
  end
end

function PlayerPortraitFrame:SetPortraitFrame(pPanel, szUiPath, nPortraitFrameID)
  if not nPortraitFrameID then
    pPanel:SetActive(szUiPath, false)
    return
  end
  local tbInfo = self.tbPortraitFrameSetting[nPortraitFrameID]
  if not tbInfo or Lib:IsEmptyStr(tbInfo.szIcon) then
    pPanel:SetActive(szUiPath, false)
  else
    local szFrameIcon = tbInfo.szIcon
    local nInterval = tbInfo.nPlayInterval
    local obj = pPanel:GetObject(szUiPath)
    if obj then
      if not nInterval then
        pPanel:UIAnimation_Clear(szUiPath)
        pPanel:Sprite_SetSprite(szUiPath, szFrameIcon)
      else
        pPanel:Sprite_SetSprite(szUiPath, string.format("%s%02d.png", szFrameIcon, 1))
        pPanel:UIAnimation_SetSprites(szUiPath, szFrameIcon, 2, 1, 16, "png", nInterval)
      end
    end
    pPanel:SetActive(szUiPath, true)
  end
end

function PlayerPortraitFrame:ChangePortraitFrame(nPortraitFrameID)
  local bRet, szRet = self:CanChangePortraitFrame(nPortraitFrameID)
  if not bRet then
    if szRet then
      me.CenterMsg(szRet)
    end
    return
  end
  RemoteServer.PortraitFrameCmd("ChangePortraitFrame", nPortraitFrameID)
end

function PlayerPortraitFrame:CanChangePortraitFrame(nPortraitFrameID)
  if not self:CheckPFIdExist(nPortraitFrameID) then
    return false, i18n.EasyGet("\232\175\165\229\164\180\229\131\143\230\161\134\228\184\141\229\173\152\229\156\168\239\188\129")
  end
  if not self:CheckIsInValidTimeAndInSameDay(me, nPortraitFrameID, self.tbPortraitData) then
    return false, i18n.EasyGet("\229\164\180\229\131\143\230\161\134\230\156\170\232\167\163\233\148\129\239\188\129")
  end
  return true
end
