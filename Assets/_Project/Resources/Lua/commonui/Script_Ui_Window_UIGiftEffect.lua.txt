local tbWnd = Ui:CreateWindow("UIGiftEffect")
local tbControls = {
  ClickEmptyToClose = "ClickEmptyToClose"
}
tbWnd.tbOnClick = {
  ClickEmptyToClose = function(self)
    self:Close()
  end
}

function tbWnd:OnOpen(nTime)
  self:CloseTimer()
  self.nTimer = Timer:Register(nTime * Env.GAME_FPS, self.OnTimeOut, self)
end

function tbWnd:OnTimeOut()
  self.nTimer = nil
  self:Close()
end

function tbWnd:CloseTimer()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:OnClose()
  return self:CloseTimer()
end

function tbWnd:CloseByDefault()
end
