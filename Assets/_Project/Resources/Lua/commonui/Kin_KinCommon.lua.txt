local function GetBuildingSetting()
  local tbResult = {}
  local tbMaxLevel = {}
  local szBuildingPath = "Setting/Kin/KinBuilding.tab"
  local tbSetting = Lib:LoadTabFileEx(szBuildingPath, "ddddsd", nil, {
    "BuildingId",
    "Level",
    "MainBuildingLimit",
    "LevelUpCost",
    "Desc",
    "nLimitedValue"
  })
  for _, tbLineData in ipairs(tbSetting) do
    local nBuildingId = tbLineData.BuildingId
    local nBuildingLevel = tbLineData.Level
    tbResult[nBuildingId] = tbResult[nBuildingId] or {}
    tbResult[nBuildingId][nBuildingLevel] = {
      nMainBuildingRequire = tbLineData.MainBuildingLimit,
      nLevelUpCost = tbLineData.LevelUpCost,
      szDesc = tbLineData.Desc,
      nLimitedValue = tbLineData.nLimitedValue
    }
    tbMaxLevel[nBuildingId] = tbMaxLevel[nBuildingId] or 1
    if tbLineData.Level > tbMaxLevel[nBuildingId] then
      tbMaxLevel[nBuildingId] = tbLineData.Level
    end
  end
  tbResult = SetReadOnly(tbResult)
  tbMaxLevel = SetReadOnly(tbMaxLevel)
  return tbResult, tbMaxLevel
end

local tbBuildingSetting, tbBuildingMaxLevel = GetBuildingSetting()

function Kin:RefreshMaxSalaryCfg()
  for nType, tbSalaryType in ipairs(Kin.Cfg.Salary_Channel_Type) do
    local nMaxSalary = 0
    for _, tbSalary in ipairs(Kin.Cfg.Salary_Channel) do
      if nType == tbSalary.Channel_Type and (tbSalary.szOpenTimeFrame == "" or 1 == GetTimeFrameState(tbSalary.szOpenTimeFrame)) and ("" == tbSalary.szCloseTimeFrame or 0 == GetTimeFrameState(tbSalary.szCloseTimeFrame)) then
        nMaxSalary = nMaxSalary + tbSalary.Salary * tbSalary.MaxTimes
      end
    end
    tbSalaryType.Max_Salary = nMaxSalary
  end
end

function Kin:GetMaxSalaryByChannelType(nType)
  if not nType or not Kin.Cfg.Salary_Channel_Type[nType] then
    LogErr("Kin:GetMaxSalaryByChannelType the nType:", nType, "is illegitimate.")
    return 0
  end
  if not Kin.Cfg.Salary_Channel_Type[nType].Max_Salary or Kin.Cfg.Salary_Channel_Type[nType].Max_Salary <= 0 then
    self:RefreshMaxSalaryCfg()
  end
  return Kin.Cfg.Salary_Channel_Type[nType].Max_Salary
end

function Kin:GetMainBuildingOpenLevel()
  local nMaxMainBuildingLevel = Kin.Def.BuildingInitData[Kin.Def.Building_Main].nLevel
  for nLevel = nMaxMainBuildingLevel + 1, Kin.Def.nMaxBuildingLevel do
    if 1 ~= GetTimeFrameState(GetTimeFrameKeyByLevel(nLevel)) then
      break
    end
    nMaxMainBuildingLevel = nLevel
  end
  local szNextLevelTimeFrame = GetTimeFrameKeyByLevel(nMaxMainBuildingLevel + 1)
  return nMaxMainBuildingLevel, szNextLevelTimeFrame
end

function Kin:GetBuildingLevel(nBuildingType)
  if SERVER_RUNNING_MODE then
    if 0 == me.dwKinId then
      return
    end
    local kinData = self:GetKinById(me.dwKinId)
    if not kinData then
      return
    end
    return kinData:GetBuildingLevel(nBuildingType)
  end
  local tbBuildingData = self:GetBuildingData(nBuildingType) or {}
  return tbBuildingData.nLevel or 1
end

function Kin:GetCurMaxBuildingLevel(nBuildingId)
  if nBuildingId == Kin.Def.Building_Main then
    local tbTimeFrame = Kin:GetKinBuildingTimeFrame()
    local szMaxTimeFrame = Lib:GetMaxTimeFrame(tbTimeFrame)
    if not szMaxTimeFrame or "" == szMaxTimeFrame then
      return Kin.Def.BuildingInitData[nBuildingId].nLevel
    end
    return tbTimeFrame[szMaxTimeFrame].nMaxMainLevel
  end
  local tbSetting = self:GetBuildUpgradeSetting(nBuildingId)
  local nMainLevel = self:GetBuildingLevel(Kin.Def.Building_Main)
  local nMaxLevel = Kin.Def.BuildingInitData[nBuildingId].nLevel
  for nLevel, tbInfo in pairs(tbSetting) do
    if nLevel > nMaxLevel and nMainLevel >= tbInfo.nMainBuildingRequire then
      nMaxLevel = nLevel
    end
  end
  return nMaxLevel
end

function Kin:GetBuildingUpgradeCost(nBuildingId, nLevel)
  if nLevel > Kin:GetBuildingMaxLevel(nBuildingId) then
    return 0, 0
  end
  local tbBuildingInfo = Kin:GetBuildingInfo(nBuildingId, nLevel) or {}
  local nCost = tbBuildingInfo.nLevelUpCost
  assert(nCost, string.format("KinBuilding.tab write error %d, %d", nBuildingId, nLevel))
  local nMaxLevel = Kin:GetMainBuildingOpenLevel()
  local nDiscountRate = Kin.Def.BuildingDiscountRate[nMaxLevel - nLevel] or 1
  nCost = math.floor(nCost * nDiscountRate)
  return nCost, nDiscountRate
end

function Kin:LoadKinSalarySetting()
  local szPath = "Setting/Kin/KinSalaryMaxTime.tab"
  local tbKinSalary = Lib:EasyLoadTabFile(szPath)
  self.tbKinSalaryMaxTime = {}
  for _, tbInfo in ipairs(tbKinSalary) do
    self.tbKinSalaryMaxTime[tbInfo.szLinkKey] = self.tbKinSalaryMaxTime[tbInfo.szLinkKey] or {}
    table.insert(self.tbKinSalaryMaxTime[tbInfo.szLinkKey], tbInfo)
  end
  local nFirstTime, nSecondTime
  for szKey, tbList in pairs(self.tbKinSalaryMaxTime) do
    table.sort(tbList, function(tbFirst, tbSecond)
      nFirstTime = Lib:IsEmptyStr(tbFirst.szTimeFrameStart) and 0 or Lib:CalcTimeFrameOpenTime(tbFirst.szTimeFrameStart)
      nSecondTime = Lib:IsEmptyStr(tbSecond.szTimeFrameStart) and 0 or Lib:CalcTimeFrameOpenTime(tbSecond.szTimeFrameStart)
      return nFirstTime > nSecondTime
    end)
  end
  self.tbKinSalaryMaxTime = SetReadOnly(self.tbKinSalaryMaxTime)
end

Kin:LoadKinSalarySetting()

function Kin:GetKinSalaryMaxTime(szKey, nMaxNum)
  if not self.tbKinSalaryMaxTime[szKey] then
    return nMaxNum
  end
  for _, tbInfo in ipairs(self.tbKinSalaryMaxTime[szKey] or {}) do
    if (Lib:IsEmptyStr(tbInfo.szTimeFrameStart) or 1 == GetTimeFrameState(tbInfo.szTimeFrameStart)) and (Lib:IsEmptyStr(tbInfo.szTimeFrameEnd) or 0 == GetTimeFrameState(tbInfo.szTimeFrameEnd)) then
      return tbInfo.nMaxTime
    end
  end
  return nMaxNum
end

function Kin:GetDonateSetting(nJewelryLevel, nCount)
  if not nJewelryLevel or not nCount then
    Lib.ShowStack("Kin:GetDonateSetting args are invalid")
  end
  for _, tbSetting in ipairs(self.tbDonateSettings[nJewelryLevel] or self.tbDonateSettings[self.nMaxDonateLevel]) do
    if nCount <= tbSetting.nMax then
      return tbSetting
    end
  end
  local tbSettings = self.tbDonateSettings[nJewelryLevel] or self.tbDonateSettings[self.nMaxDonateLevel]
  return tbSettings[#tbSettings]
end

function Kin:InitQuality(tbEventsList)
  local tbEventQuality = {}
  local nCount = self:GetKinEventTableCount(tbEventsList)
  local nIndex = 1
  for _, tbEvent in pairs(tbEventsList) do
    local nProbability = 0
    if nIndex > 1 then
      nProbability = tbEventQuality[nIndex - 1].nRatio + tbEvent.Ratio
    else
      nProbability = tbEvent.Ratio
    end
    local nTempEventId = tbEvent.Id
    table.insert(tbEventQuality, {nEventId = nTempEventId, nRatio = nProbability})
    nIndex = nIndex + 1
  end
  return tbEventQuality
end

function Kin:GetRandomEventId(tbEventList)
  local tbEventQuality = self.InitQuality(self, tbEventList)
  local nRanVal = MathRandom(1, tbEventQuality[#tbEventQuality].nRatio)
  local nEventId = 0
  for _, tbEvent in pairs(tbEventQuality) do
    if nRanVal <= tbEvent.nRatio then
      nEventId = tbEvent.nEventId
      break
    end
  end
  return nEventId
end

function Kin:GetEventAllowIdByString(sEventPlayerStr)
  if "" == sEventPlayerStr then
    return nil
  end
  local sTempEvent = string.match(sEventPlayerStr, "%[.*%]")
  sTempEvent = string.sub(sTempEvent, 2, #sTempEvent - 1)
  local tbAllowId = Lib:SplitStr(sTempEvent)
  local tbResult = {}
  for k, v in pairs(tbAllowId) do
    tbResult[tonumber(v)] = 1
  end
  return tbResult
end

function Kin:GetAllowEventId(tbEventsListByRatio, tbRunNames, nIdx)
  local tbTempEventList = Lib:CopyTB(tbEventsListByRatio)
  Log("GetAllowEventId = ", nIdx)
  local nRunnerId = tbRunNames[nIdx].nId
  local nIndex = 0
  local nCount = self.GetKinEventTableCount(self, tbTempEventList)
  Log("#tbTempEventList=", nCount)
  while 0 ~= nCount do
    nIndex = self.GetRandomEventId(self, tbTempEventList)
    local tbEvent = self.GetGatherRunEvent(self, nIndex)
    local tbAllowId = self.GetEventAllowIdByString(self, tbEvent.PlayerId)
    if nil == tbAllowId or nil ~= tbAllowId and nil ~= tbAllowId[nRunnerId] then
      break
    end
    for k, tbEvent in pairs(tbTempEventList) do
      if nIndex == tonumber(tbEvent.Id) then
        tbTempEventList[tbEvent.Id] = nil
        nCount = nCount - 1
        break
      end
    end
  end
  if 0 == nCount then
    nIndex = 0
  end
  if 0 ~= nIndex then
    for k, tbEvent in pairs(tbEventsListByRatio) do
      if nIndex == tbEvent.Id then
        tbEventsListByRatio[k] = nil
        Log("the Choosed EventId  is", k)
        break
      end
    end
  end
  return nIndex
end

function Kin:GetKinEventTableCount(tbTempEventList)
  local nCount = 0
  for k, v in pairs(tbTempEventList) do
    nCount = nCount + 1
  end
  return nCount
end

function Kin:CheckMailContent(szContent)
  if not szContent or "" == szContent then
    return false, Kin.Def.tbMsgCode.PleaseEnterContent
  end
  if "" ~= szContent and ReplaceLimitWords(szContent) then
    return false, Kin.Def.tbMsgCode.IllegalChar
  end
  if Lib:StrLen(szContent) > Kin.Cfg.nMaxMailLength then
    return false, Kin.Def.tbMsgCode.OutOfLengthLimit
  end
  return true
end
