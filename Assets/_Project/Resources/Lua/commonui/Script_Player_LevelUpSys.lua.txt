LevelUpSys.LockList = {}
LevelUpSys.DoSpecialFuncList = {}

function LevelUpSys:OnLevelUp(nNewLevel)
  local tbInfo = LevelUpSys.LockList[nNewLevel]
  if tbInfo then
    Ui:OpenWindow("LevelUpSysPanel", nNewLevel)
  end
  local tbFunc = LevelUpSys.DoSpecialFuncList[nNewLevel]
  if tbFunc then
    Lib:CallBack({tbFunc, self})
  end
end
