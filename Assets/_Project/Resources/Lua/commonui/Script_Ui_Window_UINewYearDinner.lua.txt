local tbWnd = Ui:CreateClass("UINewYearDinner", "UIFestival_DinnerBase")
local tbControls = {
  szBigListTemplate = "List02/List/Viewport/Content/Template",
  szSmallListTemplate = "List01/List/Viewport/Content/Template",
  szTitle = "txtTitle",
  szTime = "txtTime01",
  szBigList = "List02",
  szSmallList = "List01",
  btnHelp = "txtTitle/btnHelp"
}
tbWnd.tbControls = tbControls

function tbWnd:UpdateListElement(tbElement, tbAward)
  local tbGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  if "Item" == tbAward[1] then
    tbGrid:SetItemByTemplate(tbAward[2], tbAward[3], nil, nil, {bShowTip = true})
    local szName = Item:GetName(tbAward[2], me)
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  else
    tbGrid:SetDigitalItem(tbAward[1], tbAward[3])
    local szName = Currency:GetName(tbAward[1])
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  end
  tbGrid:SetClickEnable(true)
end

function tbWnd:BindFindPath()
  local tbMapId2BtnList = {
    [17] = "BtnGrid/btnLinAn",
    [66] = "BtnGrid/btnXiangyang",
    [95] = "BtnGrid/btnFengxiangfu"
  }
  for nMapId, szPath in pairs(tbMapId2BtnList) do
    Ui:Button_BindEvent(self.pPanel, szPath, self.GoTo, self, nMapId)
  end
end

function tbWnd:LoadActivityData()
  local nDinnerActId = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].nDinnerActId
  self.pInstance = ActivityMgr.EntranceSchedule:BuildInstance(nDinnerActId, ActivityMgr.Def.ClassNameList.ZhongQiuTuanYuan)
end

function tbWnd:GoTo(nMapId)
  self.pInstance:GoToCity(nMapId)
  Ui:CloseWindow("UIActivity_Anniversary")
end
