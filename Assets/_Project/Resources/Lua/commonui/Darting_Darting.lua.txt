  DartingTeamId = 4,
  DartingSuccessTime = 5,
  DartingFailTime = 6,
  RobDartingTime = 7,
  DartingColor = 8,
  DartingFloorTime = 9,

  RobDartingLastAttackTime = 11,

  TeamDelete = 1,
  TimeOut = 2,
  BadCarDeath = 3
}

function Darting:LoadNpcPath()
  local szFilePath = Map:GetMapPosFile(self.Cfg.nMapTemplateId, self.Cfg.szNpcPathFileName)
  local tbNpcPath = Lib:LoadTabFileEx(szFilePath, "dddd", nil, {
    "Path",
    "Index",
    "PosX",
    "PosY"
  })
  self.tbNpcPath = {}
  for _, tbInfo in ipairs(tbNpcPath or {}) do
    self.tbNpcPath[tbInfo.Path] = self.tbNpcPath[tbInfo.Path] or {}
    table.insert(self.tbNpcPath[tbInfo.Path], tbInfo)
  end
  for _, tbList in pairs(self.tbNpcPath) do
    table.sort(tbList, function(tbFirst, tbSecond)
      return tbFirst.Index < tbSecond.Index
    end)
  end
  self.tbNpcPath = SetReadOnly(self.tbNpcPath)
end

Darting:LoadNpcPath()

function Darting:HasDartingTime(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_KEY, self.SAVE_ID_LIST.DartingTime) > 0
end
