AvenueOfStars.GM = AvenueOfStars.GM or {}
local GM = AvenueOfStars.GM

function GM:SetEnrollDate(szStartDate, szEndDate)
  AvenueOfStars.ENROLL_DATE[1] = szStartDate
  AvenueOfStars.ENROLL_DATE[2] = szEndDate
  self:FixOtherDate()
end

function GM:PrintEnrollDate()
  local szMsg = string.format("\230\138\165\229\144\141\230\151\182\233\151\180\239\188\154%s - %s", AvenueOfStars.ENROLL_DATE[1], AvenueOfStars.ENROLL_DATE[2])
  me.Msg(szMsg)
  me.CenterMsg(szMsg)
end

function GM:SetVoteDate(nIndex, szStartDate, szEndDate)
  AvenueOfStars.VOTE_DATE[nIndex][1] = szStartDate
  AvenueOfStars.VOTE_DATE[nIndex][2] = szEndDate
  self:FixOtherDate()
end

function GM:SetSettleDate(nIndex, szStartDate, szEndDate)
  AvenueOfStars.SETTLE_DATE[nIndex][1] = szStartDate
  AvenueOfStars.SETTLE_DATE[nIndex][2] = szEndDate
  self:FixOtherDate()
end

function GM:PrintActDate(nIndex)
  local szMsg = string.format("\230\138\165\229\144\141\239\188\154%s - %s", AvenueOfStars.ENROLL_DATE[1], AvenueOfStars.ENROLL_DATE[2])
  me.Msg(szMsg)
  local tbRankNameList = {
    [AvenueOfStars.PHASE_PRE] = "\233\162\132\233\128\137\232\181\155",
    [AvenueOfStars.PHASE_SEMI] = "\229\141\138\229\134\179\232\181\155",
    [AvenueOfStars.PHASE_FINAL] = "\229\134\179\232\181\155"
  }
  for nPhase, szName in pairs(tbRankNameList) do
    local szMsg = string.format("%s\230\138\149\231\165\168\239\188\154%s - %s", szName, AvenueOfStars.VOTE_DATE[nPhase][1], AvenueOfStars.VOTE_DATE[nPhase][2])
    me.Msg(szMsg)
    local szMsg = string.format("%s\231\187\147\231\174\151\239\188\154%s - %s", szName, AvenueOfStars.SETTLE_DATE[nPhase][1], AvenueOfStars.SETTLE_DATE[nPhase][2])
    me.Msg(szMsg)
  end
  local tbLevelNameList = {
    [AvenueOfStars.PHASE_SEMI] = "\233\162\132\233\128\137\232\181\155\229\146\140\229\141\138\229\134\179\232\181\155",
    [AvenueOfStars.PHASE_FINAL] = "\229\134\179\232\181\155"
  }
  for nPhase, szName in pairs(tbLevelNameList) do
    local szMsg = string.format("%s\231\187\147\231\174\151\239\188\154%s - %s", szName, AvenueOfStars.USE_VOTE_LEVEL_SETTLE_DATE[nPhase][1], AvenueOfStars.USE_VOTE_LEVEL_SETTLE_DATE[nPhase][2])
    me.Msg(szMsg)
  end
end

function GM:FixOtherDate()
  AvenueOfStars.ACT_DATE[1] = AvenueOfStars.ENROLL_DATE[1]
  AvenueOfStars.ACT_DATE[2] = AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_FINAL][2] > AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_FINAL][2] and AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_FINAL][2] or AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_FINAL][2]
  AvenueOfStars.VOTE_ITEM_GET_DATE[AvenueOfStars.VOTE_ITEM_IN_GAME][1] = AvenueOfStars.ENROLL_DATE[1]
  AvenueOfStars.VOTE_ITEM_GET_DATE[AvenueOfStars.VOTE_ITEM_IN_GAME][2] = AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_SEMI][2]
  AvenueOfStars.VOTE_ITEM_GET_DATE[AvenueOfStars.VOTE_ITEM_ACROSS_GAME][1] = AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_FINAL][1]
  AvenueOfStars.VOTE_ITEM_GET_DATE[AvenueOfStars.VOTE_ITEM_ACROSS_GAME][2] = AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_FINAL][2]
  AvenueOfStars.USE_VOTE_LEVEL_SETTLE_DATE[AvenueOfStars.PHASE_SEMI][1] = AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_PRE][1]
  AvenueOfStars.USE_VOTE_LEVEL_SETTLE_DATE[AvenueOfStars.PHASE_SEMI][2] = AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_SEMI][2]
  AvenueOfStars.USE_VOTE_LEVEL_SETTLE_DATE[AvenueOfStars.PHASE_FINAL][1] = AvenueOfStars.VOTE_DATE[AvenueOfStars.PHASE_FINAL][1]
  AvenueOfStars.USE_VOTE_LEVEL_SETTLE_DATE[AvenueOfStars.PHASE_FINAL][2] = AvenueOfStars.SETTLE_DATE[AvenueOfStars.PHASE_FINAL][2]
  if CLIENT then
    AvenueOfStars:GenShortDate("ACT_DATE")
    AvenueOfStars:GenShortDate("ENROLL_DATE")
    AvenueOfStars:GenShortDate("VOTE_DATE")
    AvenueOfStars:GenShortDate("SETTLE_DATE")
  end
end
