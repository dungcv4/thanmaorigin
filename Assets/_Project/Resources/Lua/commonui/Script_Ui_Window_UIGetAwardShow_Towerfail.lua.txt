local tbWnd = Ui:CreateWindow("UIGetAwardShow_Towerfail")
local tbControls = {
  LevelName = "imgBG/txt_Name",
  Grid_LV = "imgBG/Grid_LV",
  PersonalRank = "imgBG/txt_Rank",
  AutoExitTime = "imgBG/txt_ExitTime",
  FirstPassNum = "imgBG/txt_ExitNum",
  btnExit = "imgBG/btnExit",
  btnNext = "imgBG/btnNext"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnExit = function(self)
    self:OnClickExitButton()
  end,
  btnNext = function(self)
    self:OnClickNextButton()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MAP_SWITCH,
      self.CloseByDefault,
      self
    },
    {
      EventNotify.emNOTIFY_UPDATE_SETTLEUIBTN,
      self.RefreshExitBtn,
      self
    }
  }
end

function tbWnd:OnOpen(nLevel, bShowExitBtn, bShowNextLevelBtn)
  self.nLevel = nLevel
  local szMainName, szSubName = LevelTower:GetLevelName(nLevel)
  self.pPanel:Label_SetText(tbControls.LevelName, string.format("%s\194\183%s", szMainName, szSubName))
  local nRemainFirstPassCount = LevelTower:GetRemainFirstPassCount(me)
  if nRemainFirstPassCount > 0 then
    self.pPanel:Label_SetText(tbControls.FirstPassNum, i18n.EasyGet("\233\166\150\233\128\154\230\172\161\230\149\176\239\188\154{0}", nRemainFirstPassCount))
  else
    self.pPanel:Label_SetText(tbControls.FirstPassNum, i18n.EasyGet("\233\166\150\233\128\154\230\172\161\230\149\176\239\188\1540"))
  end
  self:RefreshExitBtn(bShowExitBtn)
  self:RefreshNextBtn(bShowNextLevelBtn)
end

function tbWnd:OnClose()
  if self.nDelayTimer then
    Timer:Close(self.nDelayTimer)
    self.nDelayTimer = nil
  end
end

function tbWnd:RefreshExitBtn(bShowExitBtn)
  self.pPanel:SetActive(tbControls.AutoExitTime, bShowExitBtn or false)
  self.pPanel:SetActive(tbControls.btnExit, bShowExitBtn or false)
  self.pPanel:Button_SetEnable(tbControls.btnExit, bShowExitBtn or false)
  if true == bShowExitBtn then
    self:__StartDelayTimer()
  elseif self.nDelayTimer then
    Timer:Close(self.nDelayTimer)
  end
end

function tbWnd:RefreshNextBtn(bShowNextLevelBtn)
  self.pPanel:SetActive(tbControls.btnNext, bShowNextLevelBtn or false)
  self.pPanel:Button_SetEnable(tbControls.btnNext, bShowNextLevelBtn or false)
end

function tbWnd:__StartDelayTimer()
  if self.nDelayTimer then
    Timer:Close(self.nDelayTimer)
  end
  self.nPassTime = 0
  self.pPanel:Label_SetText(tbControls.AutoExitTime, i18n.EasyGet("{0}\231\167\146\229\144\142\232\135\170\229\138\168\233\128\128\229\135\186", LevelTower.Cfg.nAutoExitTime))
  self.nDelayTimer = Timer:Register(Env.GAME_FPS, function()
    if self.nPassTime >= LevelTower.Cfg.nAutoExitTime then
      Timer:Close(self.nDelayTimer)
      self.nDelayTimer = nil
      self:OnClickExitButton()
      return false
    else
      self.nPassTime = self.nPassTime + 1
      self.pPanel:Label_SetText(tbControls.AutoExitTime, i18n.EasyGet("{0}\231\167\146\229\144\142\232\135\170\229\138\168\233\128\128\229\135\186", LevelTower.Cfg.nAutoExitTime - self.nPassTime))
      return true
    end
  end)
end

function tbWnd:OnClickExitButton()
  if TeamMgr:IsCaptain(me.dwID) then
    LevelTower.LevelTowerC:DoCaptainConfirm(false)
  else
    LevelTower.LevelTowerC:DoRequestExitFuBen()
  end
  self:Close()
end

function tbWnd:OnClickNextButton()
  LevelTower.LevelTowerC:DoCaptainConfirm(true)
end
