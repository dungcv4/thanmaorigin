function Repute:OnOpenMessageBox(nReputeIndex)
  local function funcOk()
    Ui:OpenWindow("UIStore", "Repute", nReputeIndex)
  end
  
  local function funcCancle()
    Ui:CloseWindow("UIMessageBox")
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\228\189\191\231\148\168\229\144\142\229\163\176\230\156\155\229\176\134\230\186\162\229\135\186\239\188\140\233\128\160\230\136\144\230\181\170\232\180\185\239\188\140\229\133\136\229\142\187\231\148\168\230\142\137\228\184\128\231\130\185\229\144\167\239\188\129"), {
    {funcOk},
    {funcCancle}
  }, {
    "\229\163\176\230\156\155\229\149\134\229\186\151",
    i18n.szCancel
  })
end

function Repute:OnCheckRedPoint()
  for nReputeType, szCurrencyType in pairs(self.tbReputeType2CurrencyType) do
    local nCurNum = me.GetMoney(szCurrencyType)
    local nMaxNum = Currency:GetCfgMaxValue(szCurrencyType)
    if nCurNum >= nMaxNum then
      RedPointMgr:LightRedPoint(self.tbReputeType2RedPointType[nReputeType])
    else
      RedPointMgr:FadeRedPoint(self.tbReputeType2RedPointType[nReputeType])
    end
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, Player)
EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, Repute.OnCheckRedPoint, Repute)
