Require("Script/Item/EquipJingZhu/AutoJingZhu.lua")
local tbWnd = Ui:CreateWindow("UIAutoJingzhu")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SHOP_BUY_RESULT,
      self.OnResponseBuy,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnStop = "node/imgBG/Content/btnStop",
  btnStopText = "node/imgBG/Content/btnStop/Text",
  messageTemplate = "node/imgBG/Content/imgList/ViewPort/List/Template",
  NotInfo = "node/imgBG/Content/imgList/NotInfo",
  InfoList = "node/imgBG/Content/imgList/ViewPort/List",
  txtTips = "node/imgBG/Content/txtTips",
  JingZhuStoneItemGrid = "node/imgBG/Content/ItemJingzhushi/UIItemGrid",
  btnAddJingZhuStone = "node/imgBG/Content/ItemJingzhushi/btnAdd",
  MeteoricIronItemGrid = "node/imgBG/Content/ItemYuntie/UIItemGrid",
  btnAddMeteoricIron = "node/imgBG/Content/ItemYuntie/btnAdd"
}
local EquipJingZhu = Item.EquipJingZhu
local AutoJingZhu = Item.AutoJingZhu
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnStop = function(self)
    if AutoJingZhu.bInAutoJingZhu then
      AutoJingZhu:StopAutoJingZhu()
    else
      AutoJingZhu:StartAutoJingZhu(nil)
    end
  end,
  btnAddJingZhuStone = function(self)
    Shop:OpenShopBuyWid(self.nJingZhuStoneId, true, "YuanBao")
  end,
  btnAddMeteoricIron = function(self)
    Shop:OpenShopBuyWid(self.nMeteoricIronId, true, "YuanBao")
  end
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewSuperListView(self, tbControls.messageTemplate, self.SetElementInfo)
  self.tbJingZhuStoneItemGrid = Ui:NewExistItemGrid(self, tbControls.JingZhuStoneItemGrid)
  self.tbJingZhuStoneItemGrid:SetClickEnable(true)
  self.tbMeteoricIronItemGrid = Ui:NewExistItemGrid(self, tbControls.MeteoricIronItemGrid)
  self.tbMeteoricIronItemGrid:SetClickEnable(true)
end

function tbWnd:OnOpen()
  self.nMsgCount = 0
  self.tbResultData = {}
  self:UpdatePanelByHaveInfo(false)
  self:UpdateItemBuy()
  self.pPanel:Label_SetText(tbControls.txtTips, "")
end

function tbWnd:OnClose()
  AutoJingZhu:StopAutoJingZhu()
end

function tbWnd:UpdateItemBuy()
  self.nJingZhuStoneId = AutoJingZhu:GetNeedShowJingZhuStoneId()
  self.nMeteoricIronId = AutoJingZhu:GetNeedShowMeteoricIronId()
  self:UpdateItemCount()
end

function tbWnd:UpdateItemCount()
  local nJingZhuStoneCurNum = me.GetItemCountByTypeInBag(self.nJingZhuStoneId)
  local nJingZhuStoneSynthesisNum = EquipJingZhu:GetCanSynthesisNum(me, self.nJingZhuStoneId)
  self.tbJingZhuStoneItemGrid:SetItemByTemplate(self.nJingZhuStoneId, nJingZhuStoneCurNum + nJingZhuStoneSynthesisNum)
  local nMeteoricIronCurNum = me.GetItemCountByTypeInBag(self.nMeteoricIronId)
  local nMeteoricIronSynthesisNum = EquipJingZhu:GetCanSynthesisNum(me, self.nMeteoricIronId)
  self.tbMeteoricIronItemGrid:SetItemByTemplate(self.nMeteoricIronId, nMeteoricIronCurNum + nMeteoricIronSynthesisNum)
end

function tbWnd:OnSyncJingZhuResult(bSuccess, nPos, nRate, nSynthesisCostCoin, nNowLevel)
  self:AddJingZhuInfo(bSuccess, nPos, nRate, nSynthesisCostCoin, nNowLevel)
  self:UpdateItemBuy()
end

function tbWnd:OnSyncBreakResult(nPos, nSynthesisCostCoin, nNowLevel)
  self:AddBreakInfo(nPos, nSynthesisCostCoin, nNowLevel)
  self:UpdateItemBuy()
end

function tbWnd:OnResponseBuy()
  self:UpdateItemCount()
end

function tbWnd:AddJingZhuInfo(bSuccess, nPos, nRate, nSynthesisCostCoin, nNowLevel)
  self.nMsgCount = self.nMsgCount + 1
  self.tbResultData[self.nMsgCount] = {
    bSuccess = bSuccess,
    nNowLevel = nNowLevel,
    nPos = nPos,
    nRate = nRate,
    nSynthesisCostCoin = nSynthesisCostCoin,
    bBreak = false
  }
  self:UpdatePanelByHaveInfo(true)
  self.tbScrollView:SetAllElement(self.tbResultData)
  self.tbScrollView:MovePanelToLastItem()
end

function tbWnd:AddBreakInfo(nPos, nSynthesisCostCoin, nNowLevel)
  self.nMsgCount = self.nMsgCount + 1
  self.tbResultData[self.nMsgCount] = {
    nPos = nPos,
    nNowLevel = nNowLevel,
    nSynthesisCostCoin = nSynthesisCostCoin,
    bBreak = true
  }
  self:UpdatePanelByHaveInfo(true)
  self.tbScrollView:SetAllElement(self.tbResultData)
  self.tbScrollView:MovePanelToLastItem()
end

function tbWnd:ShowEquipNameAndLevel(pPanel, nPos, nNowLevel)
  local pEquip = me.GetEquipByPos(nPos)
  if nil ~= pEquip then
    local szEquipName = i18n.Parse(pEquip.GetItemShowInfo(me.nFaction, me.nFactionSect, me.nSex))
    pPanel:Label_SetText("txtEquipName", string.format("%s +%d", szEquipName, nNowLevel))
  end
end

function tbWnd:ShowCostMaterial(pPanel, nItemTemplateId, nItemNum)
  local szMaterialName = KItem.GetItemShowInfo(nItemTemplateId, me.nFaction, me.nFactionSect, me.nSex)
  pPanel:Label_SetText("txtMaterialsJingzhu", string.format("%s x%d", i18n.Parse(szMaterialName), nItemNum))
end

function tbWnd:OnIsAutoChange(bIsAuto)
  self.pPanel:Label_SetText(tbControls.btnStopText, bIsAuto and i18n.EasyGet("\229\129\156\230\173\162\231\178\190\233\147\184") or i18n.EasyGet("\231\187\167\231\187\173\231\178\190\233\147\184"))
end

function tbWnd:UpdatePanelByHaveInfo(bHaveInfo)
  self.pPanel:SetActive(tbControls.NotInfo, not bHaveInfo)
  self.pPanel:SetActive(tbControls.InfoList, bHaveInfo)
end

function tbWnd:AddStopReasonText(szText)
  self.pPanel:Label_SetText(tbControls.txtTips, szText)
end

function tbWnd:RemoveStopReasonText()
  self.pPanel:Label_SetText(tbControls.txtTips, "")
end

function tbWnd:SetElementInfo(tbElement, tbData, nIndex)
  local pPanel = tbElement.pPanel
  local nPos = tbData.nPos
  local nNowLevel = tbData.nNowLevel
  local nSynthesisCostCoin = tbData.nSynthesisCostCoin
  if tbData.bBreak then
    local pPanel = tbElement.pPanel
    self:ShowEquipNameAndLevel(pPanel, nPos, nNowLevel)
    local nItemTemplateId, nBreakItemNum = EquipJingZhu:GetJingZhuBreakConfig(nNowLevel, nPos)
    self:ShowCostMaterial(pPanel, nItemTemplateId, nBreakItemNum)
    pPanel:SetActive("CostJingzhu", false)
    pPanel:Label_SetText("CostCompound/txtCostCompound", tostring(nSynthesisCostCoin))
    pPanel:Label_SetText("txtRate", "")
    pPanel:Label_SetText("txtResult", "")
  else
    local bSuccess = tbData.bSuccess
    local nRate = tbData.nRate
    self:ShowEquipNameAndLevel(pPanel, nPos, nNowLevel)
    local nTargetLevel
    if bSuccess then
      nTargetLevel = nNowLevel
    else
      nTargetLevel = nNowLevel + 1
    end
    local nItemTemplateId, nItemNum, nCoin = EquipJingZhu:GetJingZhuConsumeInfo(nTargetLevel, nPos)
    self:ShowCostMaterial(pPanel, nItemTemplateId, nItemNum)
    pPanel:SetActive("CostJingzhu", true)
    pPanel:Label_SetText("CostJingzhu/txtCostJingzhu", tostring(nCoin))
    pPanel:Label_SetText("CostCompound/txtCostCompound", tostring(nSynthesisCostCoin))
    pPanel:Label_SetText("txtRate", string.format("%d%%", nRate))
    pPanel:Label_SetText("txtResult", bSuccess and i18n.EasyGet("\230\136\144\229\138\159") or i18n.EasyGet("\229\164\177\232\180\165"))
    pPanel:Label_SetColorByName("txtResult", bSuccess and "NewGreen" or "NewRed")
  end
end
