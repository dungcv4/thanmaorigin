function Mail:LoadSetting()
  local szFilePath = "Setting/Mail/Mail.tab"
  
  local tbField = {
    "nKey",
    "szTitle",
    "szText",
    "szLogWay"
  }
  self.tbMailCfg = Lib:LoadTabFileEx(szFilePath, "dsss", "nKey", tbField)
end

Mail:LoadSetting()

function Mail:GetMailCfg(nMailKey)
  return self.tbMailCfg[nMailKey]
end
