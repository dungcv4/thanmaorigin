function tbWnd:OnCreate()
  for nIndex = 1, 4 do
    self.pPanel:Button_BindEvent(string.format(self.btnSkill, nIndex), function()
      if 1 == nIndex then
        if self.tbInfo.tbPlayerList[me.dwID].tbSpecialCards[nIndex] then
          self:UseItemC(nIndex)
        end
      else
        self:UpdateChooseIndex(nIndex)
      end
    end)
  end
  self.tbTexts = {}
  self.tbSelectPlayerList = Ui:NewScrollView(self, self.tempSelectPlayer, nil, false, false)
  self.tbShortChatList = Ui:NewScrollView(self, self.tempShortChat)
  self.tbLongChatList = Ui:NewSuperListView(self, self.tempLongChat, self.SetChatElement)
  self.pPanel:SetActive(self.imgVoice, false)
  self.pPanel:Label_SetText(self.txtGuide, i18n.EasyGet("\229\133\136\230\137\139\232\128\133\229\176\134\228\188\152\229\133\136\229\175\185\230\136\152\230\177\159\230\185\150\229\144\141\228\187\149"))
  self.bChatOpen = true
end

function tbWnd:SetChatElement(tbElement, tbData)
  local pPanel = tbElement.pPanel
  pPanel:Label_SetText("txtChatContent", i18n.Parse(tbData))
  local objText = tbElement.pPanel:GetObject("txtChatContent")
  objText:GetComponent(Ui.ContentSizeFitter):SetLayoutVertical()
  tbElement.pPanel:ContentSizeFitter_Refresh("txtChatContent")
  local objElem = self.pPanel:GetObject(tbElement.szKey)
  local transformElem = objElem:GetComponent(Ui.RectTransform)
  local transformText = objText:GetComponent(Ui.RectTransform)
  transformElem.sizeDelta = Ui.Vector2(transformElem.sizeDelta.x, transformText.sizeDelta.y)
end

function tbWnd:OnOpen()
  RemoteServer.RequestJiangHuZhuJiu("RequestGameInfo")
  self.pPanel:SetActive(self.wndSelectPlayer, false)
  self.pPanel:Label_SetText(self.txtTime, i18n.EasyGet("\229\128\146\232\174\161\230\151\182\239\188\154{0}\231\167\146", self.nTime or 0))
  if not self.nTimer then
    self.nTimer = Timer:Register(Env.GAME_FPS, self.UpdateTime, self)
  end
  self:UpdateChooseIndex()
end

function tbWnd:OnClose()
  self.pPanel:SetActive(self.wndSelectPlayer, false)
  self.tbInfo = nil
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.tbTexts = {}
  self:UpdateChat()
  if self.tbBossModelList then
    for nIndex, pNpcModel in pairs(self.tbBossModelList) do
      pNpcModel:Release()
    end
    self.tbBossModelList = nil
  end
end

function tbWnd:OpenItemWindow(nItemIndex)
  self.pPanel:SetActive(self.wndSelectPlayer, false)
  self.pPanel:SetActive(self.wndSelectPlayer, true)
  self.nItemIndex = nItemIndex
  self:UpdateItemWindow()
end

function tbWnd:CloseItemSelectWindow()
  self.pPanel:SetActive(self.wndSelectPlayer, false)
  self.nItemIndex = nil
  self.nTargetId = nil
end

function tbWnd:Update(tbInfo)
  self.tbInfo = tbInfo
  self.pPanel:SetActive(self.btnStart, not tbInfo.bStart)
  self.pPanel:SetActive(self.listPlay, tbInfo.bStart)
  self:UpdatePlayerStatus()
  self:UpdateBossStatus()
  self:UpdateFightInfo()
  self:UpdateItemWindow()
end

function tbWnd:UpdateItemWindow()
  if not self.tbInfo then
    return
  end
  self.tbSelectPlayerList:Clear()
  local tbUIIndex = {}
  for nPlayerId, tbInfo in pairs(self.tbInfo.tbPlayerList) do
    if nPlayerId ~= me.dwID then
      table.insert(tbUIIndex, nPlayerId)
    end
  end
  table.sort(tbUIIndex, function(nPlayerIdA, nPlayerIdB)
    return nPlayerIdA < nPlayerIdB
  end)
  table.insert(tbUIIndex, me.dwID)
  for nIndex, nPlayerId in pairs(tbUIIndex) do
    self.tbSelectPlayerList:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbPlayerInfo = self.tbInfo.tbPlayerList[nPlayerId]
      pPanel:Label_SetText("HeadFrame/PortraitObj/Level/Text", tbPlayerInfo.nLevel)
      pPanel:SetActive("txtPlayerName", true)
      pPanel:Label_SetText("txtPlayerName", tbPlayerInfo.szName)
      local nHP = tbPlayerInfo.nHP
      if self.nTargetId == nPlayerId then
        if 0 == nHP then
          nHP = 1
        else
          nHP = math.min(nHP + JiangHuZhuJiu.PLAYERCARDSETTING[4].nParam, JiangHuZhuJiu.CONSTANTS.nPlayerHP)
        end
      end
      local nMaxHP = JiangHuZhuJiu.CONSTANTS.nPlayerHP
      pPanel:Sprite_SetFill("Img_SliderBg/Img_Slider", nHP / nMaxHP)
      pPanel:Sprite_SetSprite("HeadFrame/PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbPlayerInfo.nPortrait))
      if tbPlayerInfo.nPortraitFrame then
        pPanel:SetActive("HeadFrame/PortraitObj/PortraitFrame", true)
        PlayerPortraitFrame:SetPortraitFrame(pPanel, "HeadFrame/PortraitObj/PortraitFrame", tbPlayerInfo.nPortraitFrame)
      end
      pPanel:SetActive("Img_State1", self.tbInfo.nCardPlayerId == nPlayerId)
      pPanel:SetActive("Img_State2", 0 == self.tbInfo.tbPlayerList[nPlayerId].nHP)
      pPanel:SetActive("btnSwitch/imgChoose", self.nTargetId == nPlayerId)
      Ui:Button_BindEvent(pPanel, "btnSwitch", self.OnClickSelectPlayer, self, nPlayerId)
    end)
  end
end

function tbWnd:OnClickSelectPlayer(nTargetId)
  if nTargetId == self.nTargetId then
    self.nTargetId = nil
  else
    self.nTargetId = nTargetId
  end
  self:UpdateItemWindow()
end

function tbWnd:UpdateChooseIndex(nIndex)
  for i = 1, 4 do
    if nIndex ~= i then
      self.pPanel:SetActive(string.format(self.imgSkillSelect, i), false)
    else
      self.pPanel:SetActive(string.format(self.imgSkillSelect, i), true)
    end
  end
  self.nChooseSkillIndex = nIndex
end

function tbWnd:UpdatePlayerStatus()
  if not self.tbInfo then
    return
  end
  local tbUIIndex = {}
  for nPlayerId, tbInfo in pairs(self.tbInfo.tbPlayerList) do
    if nPlayerId ~= me.dwID then
      table.insert(tbUIIndex, nPlayerId)
    end
  end
  table.sort(tbUIIndex, function(nPlayerIdA, nPlayerIdB)
    return nPlayerIdA < nPlayerIdB
  end)
  tbUIIndex[4] = me.dwID
  for nIndex, nPlayerId in pairs(tbUIIndex) do
    local tbPlayerInfo = self.tbInfo.tbPlayerList[nPlayerId]
    self.pPanel:SetActive(string.format(self.txtPlayerName, nIndex), true)
    self.pPanel:Label_SetText(string.format(self.txtPlayerName, nIndex), tbPlayerInfo.szName)
    local nHP = tbPlayerInfo.nHP
    local nMaxHP = JiangHuZhuJiu.CONSTANTS.nPlayerHP
    self.pPanel:Label_SetText(string.format(self.txtPlayerLevel, nIndex), tbPlayerInfo.nLevel)
    self.pPanel:Label_SetText(string.format(self.txtPlayerHP, nIndex), string.format("%d/%d", nHP, nMaxHP))
    self.pPanel:Sprite_SetFill(string.format(self.imgPlayerHP, nIndex), nHP / nMaxHP)
    self.pPanel:Sprite_SetSprite(string.format(self.imgPlayerPortrait, nIndex), PlayerPortrait:GetIcon(tbPlayerInfo.nPortrait))
    if tbPlayerInfo.nPortraitFrame then
      self.pPanel:SetActive(string.format(self.imgPlayerPortraitFrame, nIndex), true)
      PlayerPortraitFrame:SetPortraitFrame(self.pPanel, string.format(self.imgPlayerPortraitFrame, nIndex), tbPlayerInfo.nPortraitFrame)
    end
    self.pPanel:SetActive(string.format(self.imgPlayState, nIndex), self.tbInfo.nCardPlayerId == nPlayerId)
    self.pPanel:SetActive(string.format(self.imgDrunkState, nIndex), 0 == self.tbInfo.tbPlayerList[nPlayerId].nHP)
    for i = 1, 4 do
      self.pPanel:SetActive(string.format(self.imgPlayerSkill, nIndex, i), false)
      if nPlayerId == me.dwID then
        self.pPanel:SetActive(string.format(self.imgMineSkill, i), false)
      end
    end
    local nSepcialCardIndex = 1
    local nNormalCardIndex = 2
    for i, nCardIndex in ipairs(tbPlayerInfo.tbSpecialCards or {}) do
      self.pPanel:SetActive(string.format(self.imgPlayerSkill, nIndex, nSepcialCardIndex), true)
      local szSkillAtalas = JiangHuZhuJiu.CARDDESC[nCardIndex].szSkillAtalas
      self.pPanel:Sprite_SetSprite(string.format(self.imgPlayerSkill, nIndex, nSepcialCardIndex), szSkillAtalas)
      if nPlayerId == me.dwID then
        self.pPanel:SetActive(string.format(self.imgMineSkill, nSepcialCardIndex), true)
        self.pPanel:Sprite_SetSprite(string.format(self.imgMineSkill, nSepcialCardIndex), szSkillAtalas)
      end
      nSepcialCardIndex = nSepcialCardIndex + 1
    end
    for i, nCardIndex in ipairs(tbPlayerInfo.tbNormalCards or {}) do
      self.pPanel:SetActive(string.format(self.imgPlayerSkill, nIndex, nNormalCardIndex), true)
      local szSkillAtalas = JiangHuZhuJiu.CARDDESC[nCardIndex].szSkillAtalas
      self.pPanel:Sprite_SetSprite(string.format(self.imgPlayerSkill, nIndex, nNormalCardIndex), szSkillAtalas)
      if nPlayerId == me.dwID then
        self.pPanel:SetActive(string.format(self.imgMineSkill, nNormalCardIndex), true)
        self.pPanel:Sprite_SetSprite(string.format(self.imgMineSkill, nNormalCardIndex), szSkillAtalas)
      end
      nNormalCardIndex = nNormalCardIndex + 1
    end
  end
  for nIndex = 1, 4 do
    if not tbUIIndex[nIndex] then
      self.pPanel:SetActive(string.format(self.szPlayerPath, nIndex), false)
    else
      self.pPanel:SetActive(string.format(self.szPlayerPath, nIndex), true)
    end
  end
end

function tbWnd:UpdateBossStatus()
  if not self.tbInfo or not self.tbInfo.bStart then
    self.pPanel:SetActive(self.txtCurInfo, false)
    self.pPanel:SetActive(self.modelCurBoss, false)
    self.pPanel:SetActive(self.listNpc, false)
    if self.tbBossModelList then
      return
    end
    self.tbBossModelList = {}
    self.pPanel:SetActive(self.goModelLeft, true)
    for i = 1, #JiangHuZhuJiu.BOSSSETTING do
      self.pPanel:SetActive(string.format(self.modelBoss, i), true)
      local pPartnerModelObj = self.pPanel:GetObject(string.format(self.modelBoss, i))
      local nResId = JiangHuZhuJiu.BOSSSETTING[i].nResId
      self.tbBossModelList[i] = pPartnerModelObj:GetComponent(Ui.UINpcModel)
      self.tbBossModelList[i]:ShowNpc(nResId)
      if Partner.tbPartnerScaleInfo[nResId].Scale ~= "" then
        self.tbBossModelList[i].fModelScale = 125 * Partner.tbPartnerScaleInfo[nResId].Scale
      else
        self.tbBossModelList[i].fModelScale = 125
      end
      self.pPanel:SetActive(string.format(self.txtModelName, i), true)
      self.pPanel:Label_SetText(string.format(self.txtModelName, i), i18n.Get(JiangHuZhuJiu.BOSSSETTING[i].szName))
    end
    return
  end
  for i = 1, #JiangHuZhuJiu.BOSSSETTING do
    self.pPanel:SetActive(string.format(self.txtModelName, i), false)
    self.pPanel:SetActive(string.format(self.modelBoss, i), false)
  end
  self.pPanel:SetActive(self.listNpc, true)
  for i, nCardIndex in ipairs(self.tbInfo.tbBossCards or {}) do
    local szSkillAtalas = JiangHuZhuJiu.CARDDESC[nCardIndex].szSkillAtalas
    self.pPanel:SetActive(string.format(self.imgBossSkill, i), true)
    self.pPanel:Sprite_SetSprite(string.format(self.imgBossSkill, i), szSkillAtalas)
  end
  for i = #self.tbInfo.tbBossCards + 1, 5 do
    self.pPanel:SetActive(string.format(self.imgBossSkill, i), false)
  end
  local nHP = self.tbInfo.nBossHP or 0
  local nMaxHP = self.tbInfo.nBossMaxHP or 0
  self.pPanel:Label_SetText(self.txtBossHP, string.format("%d/%d", nHP, nMaxHP))
  self.pPanel:Sprite_SetFill(self.imgBossHP, nHP / nMaxHP)
  local nCurBossIndex = self.tbInfo.nCurBoss or 1
  for i = 1, #JiangHuZhuJiu.BOSSSETTING do
    self.pPanel:SetActive(string.format(self.imgBossNormal, i), i >= nCurBossIndex)
    self.pPanel:SetActive(string.format(self.imgBossSelect, i), i == nCurBossIndex)
    self.pPanel:SetActive(string.format(self.imgBossFinish, i), i < nCurBossIndex)
    self.pPanel:Label_SetText(string.format(self.txtBossName, i), i18n.Get(JiangHuZhuJiu.BOSSSETTING[i].szName))
  end
  if self.tbCurBossInfo and self.tbCurBossInfo.nCurBossIndex ~= nCurBossIndex then
    self.tbCurBossInfo.pModel:Release()
    self.tbCurBossInfo = nil
  end
  self.pPanel:SetActive(self.modelCurBoss, true)
  local pPartnerModelObj = self.pPanel:GetObject(self.modelCurBoss)
  self.tbCurBossInfo = {
    pModel = pPartnerModelObj:GetComponent(Ui.UINpcModel),
    nCurBossIndex = nCurBossIndex
  }
  self.pPanel:SetActive(self.txtCurInfo, true)
  local nResId = JiangHuZhuJiu.BOSSSETTING[nCurBossIndex].nResId
  self.tbCurBossInfo.pModel:ShowNpc(nResId)
  if Partner.tbPartnerScaleInfo[nResId].Scale ~= "" then
    self.tbCurBossInfo.pModel.fModelScale = 125 * Partner.tbPartnerScaleInfo[nResId].Scale
  else
    self.tbCurBossInfo.pModel.fModelScale = 125
  end
  self.pPanel:Label_SetText(self.txtCurModelName, i18n.Get(JiangHuZhuJiu.BOSSSETTING[nCurBossIndex].szName))
  self.pPanel:Label_SetText(self.txtCurRound, i18n.EasyGet("\231\172\172{0}\229\155\158\229\144\136", self.tbInfo.nCurBoss))
end

function tbWnd:UpdateFightInfo()
  self.nTime = self.tbInfo.nTime
  self.nLeftTime = self.tbInfo.nLeftTime
  if not self.tbInfo.bStart then
    self.pPanel:SetActive(self.goCenterUp, false)
    return
  else
    self.pPanel:SetActive(self.goCenterUp, true)
  end
  local nPlayerCard = self.tbInfo.nPlayerCard
  if nPlayerCard then
    self.pPanel:SetActive(self.imgPlayerUseSkill, true)
    self.pPanel:Sprite_SetSprite(self.imgPlayerUseSkill, JiangHuZhuJiu.CARDDESC[nPlayerCard].szSkillAtalas)
  else
    self.pPanel:SetActive(self.imgPlayerUseSkill, false)
  end
  local nBossCard = self.tbInfo.nBossCard
  if nBossCard then
    self.pPanel:SetActive(self.imgBossUseSkill, true)
    self.pPanel:Sprite_SetSprite(self.imgBossUseSkill, JiangHuZhuJiu.CARDDESC[nBossCard].szSkillAtalas)
  else
    self.pPanel:SetActive(self.imgBossUseSkill, false)
  end
  if self.tbInfo.nResult then
    self.pPanel:SetActive(self.txtResult, true)
    if -1 == self.tbInfo.nResult then
      self.pPanel:Label_SetText(self.txtResult, i18n.EasyGet("\229\164\177\232\180\165"))
      self.pPanel:SetActive(self.aniFail, false)
      self.pPanel:SetActive(self.aniFail, true)
    elseif 0 == self.tbInfo.nResult then
      self.pPanel:Label_SetText(self.txtResult, i18n.EasyGet("\229\185\179\229\177\128"))
      self.pPanel:SetActive(self.aniDogFall, false)
      self.pPanel:SetActive(self.aniDogFall, true)
    else
      self.pPanel:Label_SetText(self.txtResult, i18n.EasyGet("\232\131\156\229\136\169"))
      self.pPanel:SetActive(self.aniVictory, false)
      self.pPanel:SetActive(self.aniVictory, true)
    end
  else
    self.pPanel:SetActive(self.txtResult, false)
  end
end

end

function tbWnd:UpdateChat()
  if not self.bChatOpen then
    self.pPanel:SetActive(self.szShortChat, true)
    self.pPanel:SetActive(self.szLongChat, false)
    local szText = self.tbTexts[#self.tbTexts]
    if not szText then
      return
    end
    self.tbShortChatList:Clear()
    self.tbShortChatList:SetElement(1, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtChatContent", i18n.Parse(szText))
    end)
  else
    self.pPanel:SetActive(self.szShortChat, false)
    self.pPanel:SetActive(self.szLongChat, true)
    self.tbLongChatList:SetAllElement(self.tbTexts)
  end
end

function tbWnd:UpdateTime()
  self.pPanel:Label_SetText(self.txtTime, i18n.EasyGet("\229\128\146\232\174\161\230\151\182\239\188\154{0}\231\167\146", self.nTime or 0))
  self.nTime = math.max(0, (self.nTime or 0) - 1)
  self.pPanel:Label_SetText(self.txtLeftTime, i18n.EasyGet("\229\128\146\232\174\161\230\151\182\239\188\154{0}\231\167\146", self.nLeftTime or 0))
  self.nLeftTime = math.max(0, (self.nLeftTime or 0) - 1)
  return true
end
