local tbWnd = Ui:CreateWindow("UILegionSetAnnounce")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_LEGION_BASEINFO,
      self.onLegionSyncData
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSaveChange = "node/imgBG/Content/btnSaveChange",
  inputLegionNotice = "node/imgBG/Content/LegionNotice/inputLegionNotice",
  txtTitle = "node/imgBG/txtTitle"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSaveChange = function(self)
    if self.szType == "PublicDeclare" then
      self:OnChangePublicDeclare()
    elseif self.szType == "AddDeclare" then
      self:OnChangeAddDeclare()
    end
  end
}

function tbWnd:OnOpen(szTitle, szType)
  self.szType = szType
  self.pPanel:Label_SetText(self.txtTitle, szTitle)
  self.pPanel:Input_SetText(self.inputLegionNotice, "")
  if self.szType == "PublicDeclare" then
    local tbBaseInfo = Legion:GetBaseInfo()
    if tbBaseInfo then
      self.pPanel:Input_SetText(self.inputLegionNotice, tbBaseInfo.szPublicDeclare or "")
    end
  elseif self.szType == "AddDeclare" then
    Legion:UpdateRecruitSetting()
    local tbRecruitSetting = Legion:GetRecruitSetting()
    if tbRecruitSetting then
      self.pPanel:Input_SetText(self.inputLegionNotice, tbRecruitSetting.szAddDeclare or "")
    end
  end
end

function tbWnd:onLegionSyncData(szDataType, nRemainTime)
  if "RecruitSetting" == szDataType and Ui:WindowVisible("UILegionSetAnnounce") and self.szType == "AddDeclare" then
    local tbRecruitSetting = Legion:GetRecruitSetting()
    self.pPanel:Input_SetText(self.inputLegionNotice, tbRecruitSetting.szAddDeclare)
  end
end

function tbWnd:OnChangePublicDeclare()
  if not Legion:CheckMyAuthority(Legion.Def.Authority_EditPubilcDeclare) then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\230\157\131\233\153\144\232\191\155\232\161\140\230\147\141\228\189\156"))
    return
  end
  local szLegionNotice = self.pPanel:Input_GetText(self.inputLegionNotice)
  Legion:ChangePublicDeclare(szLegionNotice)
  self:Close()
end

function tbWnd:OnChangeAddDeclare()
  if not Legion:CheckMyAuthority(Legion.Def.Authority_EditRecuitInfo) then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\230\157\131\233\153\144\232\191\155\232\161\140\230\147\141\228\189\156"))
    return
  end
  local szLegionNotice = self.pPanel:Input_GetText(self.inputLegionNotice)
  Legion:ChangeAddDeclare(szLegionNotice)
  self:Close()
end
