local tbWnd = Ui:CreateWindow("UIXiangyang")
local tbControls = {
  Time = "Cont/Time",
  txtStart = "Cont/Time/Start/txtStart",
  txtStart02 = "Cont/Time/Start/txtStart02"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
end

function tbWnd:OnDestroy()
end

function tbWnd:OnOpen(szTitle)
  if szTitle then
    self.pPanel:Label_SetText(tbControls.txtStart, szTitle)
    self.pPanel:Label_SetText(tbControls.txtStart02, szTitle)
  end
  self.pPanel:SetActive(tbControls.Time, true)
  self:ClearTimer()
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * 2, function()
    if Ui:WindowVisible("UIXiangyang") then
      Ui:CloseWindow("UIXiangyang")
    end
  end)
end

function tbWnd:OnClose()
  self:ClearTimer()
end

function tbWnd:ClearTimer()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
end
