setmetatable(Achievement.WelfareChecker, {
  __index = function(tb, szPanelType)
    if "PANEL_TYPE_MONEY_TREE" == szPanelType then
      tb[szPanelType] = function()
        return MoneyTree:CheckIsOpen(me)
      end
    elseif "PANEL_TYPE_WEEK_MONTH_CARD" == szPanelType then
      tb[szPanelType] = function()
        return WeekMonthCard:CheckIsOpen()
      end
    else
      tb[szPanelType] = function()
        return true
      end
    end
    return tb[szPanelType]
  end
})

function Achievement:OnAchievementComplete(nID)
  Timer:Register(self.nDisplayDelayTime * Env.GAME_FPS, self.OnDisplayDelayTimerOut, self, nID)
  EventNotify.OnNotify(EventNotify.emNOTIFY_ACHIEVEMENT_DATA_SYNC)
  local nUnlockLevel = self:GetAchievementUnlockLevel()
  if not nUnlockLevel or nUnlockLevel <= me.nLevel then
    local cfg = self:GetCfgByID(nID)

    me.CenterMsg(szMsg)
    me.Msg(szMsg)
  end
end

function Achievement:GetTreeShowData()
  local tbBigData = {}
  for nID, tbInfo in pairs(self.tbGroupCfg) do
    table.insert(tbBigData, {
      nType = nID,
      szText = i18n.Parse(tbInfo.szName),
      tbInfo = {nID = nID}
    })
  end
  table.sort(tbBigData, function(tbFirst, tbSecond)
    return tbFirst.nType < tbSecond.nType
  end)
  local tbSmallData = {}
  for szKind, tbInfo in pairs(self.tbMainSetting) do
    table.insert(tbSmallData, {
      nType = tbInfo.nGroupID,
      szText = i18n.Parse(tbInfo.szDesc),
      tbInfo = {szKind = szKind}
    })
  end
  table.sort(tbSmallData, function(tbFirst, tbSecond)
    return self.tbMainSetting[tbFirst.tbInfo.szKind].nPriority < self.tbMainSetting[tbSecond.tbInfo.szKind].nPriority
  end)
  return tbBigData, tbSmallData
end

function Achievement:GetShowListByKind(szKind)
  local tbList = {}
  local tbCountList, tbCfg, nSubKindID = {}
  for _, nPileID in ipairs(self.tbKindList[szKind] or {}) do
    for nIndex, nID in ipairs(self.tbPileList[nPileID] or {}) do
      tbCfg = self:GetCfgByID(nID)
      nSubKindID = tbCfg.nSubKindID
      tbCountList[nSubKindID] = tbCountList[nSubKindID] or self:GetSubKindCount(me, nSubKindID)
      if tbCountList[nSubKindID] < tbCfg.nFinishCount or nIndex == #self.tbPileList[nPileID] or not self:HasAwardGained(me, nID) then
        table.insert(tbList, {
          nID = nID,
          szName = i18n.Parse(tbCfg.szTitle),
          szDesc = i18n.Parse(tbCfg.szDesc),
          szParentKind = i18n.Parse(self.tbMainSetting[self.tbRelationCfg[nPileID].szParentKind].szDesc),
          nCurCount = tbCountList[nSubKindID],
          nFinishCount = tbCfg.nFinishCount,
          nFinished = self:HasAwardGained(me, nID),
          nScore = tbCfg.nScore,
          szAward = tbCfg.szAward,
          bShowBtnGo = tbCfg.szFuncName ~= "" and tbCfg.szFuncName ~= "NotJump"
        })
        break
      end
    end
  end
  local tbFinishedBool = {
    [true] = 2,
    [false] = 1
  }
  local bCurFirstFinished, bCurSecondFinished
  table.sort(tbList, function(tbFirst, tbSecond)
    if tbFirst.nFinished ~= tbSecond.nFinished then
      return tbFinishedBool[tbFirst.nFinished] < tbFinishedBool[tbSecond.nFinished]
    end
    bCurFirstFinished = tbFirst.nCurCount >= tbFirst.nFinishCount
    bCurSecondFinished = tbSecond.nCurCount >= tbSecond.nFinishCount
    if bCurFirstFinished ~= bCurSecondFinished then
      return tbFinishedBool[bCurFirstFinished] > tbFinishedBool[bCurSecondFinished]
    end
    return self.tbAchievementCfg[tbFirst.nID].nSortInUI < self.tbAchievementCfg[tbSecond.nID].nSortInUI
  end)
  return tbList
end

function Achievement:JumpAchievementDetail(tbLinkInfo)
  local nUnlockLevel = self:GetAchievementUnlockLevel()
  if nUnlockLevel and nUnlockLevel > me.nLevel then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}\231\186\167\239\188\140\230\136\144\229\176\177\230\156\170\232\167\163\233\148\129", nUnlockLevel))
    return
  end
  if not tbLinkInfo or not tbLinkInfo.nID then
    return
  end
  local tbCfg = self:GetCfgByID(tbLinkInfo.nID)
  if not tbCfg then
    return
  end
  Ui:OpenWindow("UIAchievement", tbLinkInfo.nID)
end

function Achievement:GetRedPointAchievementList()
  local pPlayer = me
  local tbKindList = {}
  local tbCountList = {}
  local tbCfg
  for szKind, _ in pairs(self.tbMainSetting) do
    for _, nPileID in ipairs(self.tbKindList[szKind] or {}) do
      for _, nID in ipairs(self.tbPileList[nPileID] or {}) do
        tbCfg = self:GetCfgByID(nID)
        tbCountList[tbCfg.nSubKindID] = tbCountList[tbCfg.nSubKindID] or self:GetSubKindCount(pPlayer, tbCfg.nSubKindID)
        if tbCountList[tbCfg.nSubKindID] < tbCfg.nFinishCount then
          break
        end
        if not self:HasAwardGained(pPlayer, nID) then
          tbKindList[szKind] = true
          break
        end
      end
      if tbKindList[szKind] then
        break
      end
    end
  end
  local tbGroupIDList = {}
  for szKind in pairs(tbKindList) do
    tbGroupIDList[self.tbMainSetting[szKind].nGroupID] = true
  end
  return tbGroupIDList, tbKindList
end

function Achievement:CheckRedPoint()
  local tbGroupIDList, tbKindList = self:GetRedPointAchievementList()
  if next(tbGroupIDList) or next(tbKindList) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.Achievement)
  end
end

function Achievement:DealAchievementEventC(szEvent)
  if not self.ClientEventLimitedList[szEvent] then
    return
  end
  local nSubKindID = self:GetSubKindIDBySubKind(szEvent)
  if not nSubKindID or nSubKindID <= 0 then
    return
  end
  local nCurCount = self:GetSubKindCount(me, nSubKindID)
  if nCurCount >= self.ClientEventLimitedList[szEvent] then
    return
  end
  RemoteServer.RequestAchievement(self.Protocol.DealAchievementEvent, szEvent)
end
