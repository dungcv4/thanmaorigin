local tbWnd = Ui:CreateWindow("UILoadingTips")
tbWnd.tbControls = {txtMsg = "imgBG/Text"}

function tbWnd:OnClose()
  if self.nTimeoutTimer then
    Timer:Close(self.nTimeoutTimer)
  end
  self.nTimeoutTimer = nil
end

function tbWnd:OnOpen(szTips, nTimeout, fnTimeout)
  if self.nTimeoutTimer then
    Timer:Close(self.nTimeoutTimer)
  end
  self.OnCb = fnTimeout or self.OnDefaultCb
  if nTimeout then
    self.nTimeoutTimer = Timer:Register(Env.GAME_FPS * nTimeout, self.OnTimeout, self)
  end
  self.pPanel:Label_SetText(self.txtMsg, szTips or i18n.Get("Connecting"))
end

function tbWnd.OnDefaultCb()
end

function tbWnd:OnTimeout()
  self.OnCb()
  self.nTimeoutTimer = nil
  self:Close()
end
