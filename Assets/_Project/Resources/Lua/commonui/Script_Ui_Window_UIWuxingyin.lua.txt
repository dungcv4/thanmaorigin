local tbWnd = Ui:CreateWindow("UIWuxingyin")
local tbControls = {
  panelFixAttrUp = "node/imgBG/Content/AttributeUp",
  panelEquipLvUp = "node/imgBG/Content/LevelUp",
  panelWashType = "node/imgBG/Content/AttributeReset",
  panelWashValue = "node/imgBG/Content/AttributeResetNum",
  btnEquipLvUp = "node/imgBG/Content/btnList/btnLevelUp",
  btnFixAttrUp = "node/imgBG/Content/btnList/btnAttributeUp",
  btnWashType = "node/imgBG/Content/btnList/btnAttributeReset",
  btnWashValue = "node/imgBG/Content/btnList/btnAttributeResetNum",
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp"
}
tbWnd.tbControls = tbControls

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_WUXINGYIN_SYNC,
      self.Update,
      self
    }
  }
end

tbWnd.PANEL_TYPE_FIXATTRUP = Item.WuXingYin.UISORT.FixAttrUp
tbWnd.PANEL_TYPE_EQUIPLVUP = Item.WuXingYin.UISORT.EquipLvUp
tbWnd.PANEL_TYPE_WASHTYPE = Item.WuXingYin.UISORT.WashType
tbWnd.PANEL_TYPE_WASHVALUE = Item.WuXingYin.UISORT.WashValue
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_TYPE_FIXATTRUP] = {
    CLASS = "UIWuxingyinFixAttrUp",
    OBJ = tbControls.panelFixAttrUp,
    BTN = tbControls.btnFixAttrUp,
    OpenFunc = function()
      return Item.WuXingYin:IsOpen(me, Item.WuXingYin.UISORT.FixAttrUp)
    end
  },
  [tbWnd.PANEL_TYPE_EQUIPLVUP] = {
    CLASS = "UIWuxingyinEquipLvUp",
    OBJ = tbControls.panelEquipLvUp,
    BTN = tbControls.btnEquipLvUp,
    OpenFunc = function()
      return Item.WuXingYin:IsOpen(me, Item.WuXingYin.UISORT.EquipLvUp)
    end
  },
  [tbWnd.PANEL_TYPE_WASHTYPE] = {
    CLASS = "UiWuXingYinWashType",
    OBJ = tbControls.panelWashType,
    BTN = tbControls.btnWashType,
    OpenFunc = function()
      return Item.WuXingYin:IsOpen(me, Item.WuXingYin.UISORT.WashType)
    end
  },
  [tbWnd.PANEL_TYPE_WASHVALUE] = {
    CLASS = "UiWuXingYinWashValue",
    OBJ = tbControls.panelWashValue,
    BTN = tbControls.btnWashValue,
    OpenFunc = function()
      return Item.WuXingYin:IsOpen(me, Item.WuXingYin.UISORT.WashValue)
    end
  }
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
    Guide:ClickWindowCloseCallSkipGuide(Guide.tbClickCloseButtonSkipGuide.UIWuxingyin)
  end,
  btnHelp = function()
    Ui:OnHelpClicked("WuXingYinHelp")
  end
}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
end

function tbWnd:OnOpen(nSelectPanel, nItemId)
  self.nItemId = nItemId
  self.nSelect = nSelectPanel or 1
  if not self.tbTabControl:OnOpen(nSelectPanel, nItemId) then
    return
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnFixAttrUp, {
    RedPointMgr.EVENT.WuXingYinEnhanceLvUp,
    RedPointMgr.EVENT.WuXingYinabateLvUp
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnEquipLvUp, {
    RedPointMgr.EVENT.WuXingYinEquipLvUp
  })
end

function tbWnd:SwitchPanel(nPanelType)
  self.tbTabControl:OnClose()
  self.nSelect = nPanelType
  if not self.tbTabControl:OnOpen(nPanelType, self.nItemId) then
    return
  end
end

function tbWnd:Update(nItemId, bUpdateBtn)
  self.nItemId = nItemId
  if bUpdateBtn then
    self.tbTabControl:UpdateButtonGroup()
  end
  self.tbTabControl:UpdateSubPanel(self.nSelect, "Update", nItemId)
end

function tbWnd:OnClose()
  self.nItemId = nil
  self.tbTabControl:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnFixAttrUp)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnEquipLvUp)
end
