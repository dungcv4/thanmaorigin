Weather.tbRainRate = {
  [1] = {
    [0] = 0,
    [1] = 20,
    [2] = 5,
    [3] = 3,
    [4] = 1,
    [5] = 0
  },
  [2] = {
    [0] = 0,
    [1] = 0,
    [2] = 10,
    [3] = 5,
    [4] = 0,
    [5] = 1
  }
}

function Weather:ShouldRain(nMapTemplateId)
  local tbMapSetting = Map:GetMapSetting(nMapTemplateId)
  if not tbMapSetting then
    return 0
  end
  local nMapTemplateWeather = tbMapSetting.Weather
  if not nMapTemplateWeather or 0 == nMapTemplateWeather then
    return 0
  end
  local nHour = Lib:GetLocalDayHour(GetTime()) + 1
  local nDate = Lib:GetTimeNum()
  local nTotal = nHour + nDate + nMapTemplateId
  local nRainType = 0
  local nLowRainMod = Weather.tbRainRate[1][nMapTemplateWeather] or 0
  local nHighRainMod = Weather.tbRainRate[2][nMapTemplateWeather] or 0
  if nLowRainMod > 0 and 0 == math.mod(nTotal, nLowRainMod) then
    nRainType = 1
  end
  if nHighRainMod > 0 and 0 == math.mod(nTotal, nHighRainMod) then
    nRainType = 2
  end
  return nRainType
end

function Weather:OpenTimer()
  if not self.nCheckTimerID then
    self.nCheckTimerID = Timer:Register(1800 * Env.GAME_FPS, self.CheckWeather, self)
    self:CheckWeather()
  end
end

function Weather:StopTimer()
  if self.nCheckTimerID then
    Timer:Close(self.nCheckTimerID)
    self.nCheckTimerID = nil
    Ui:StopRaining()
  end
end

function Weather:CheckWeather()
  if 1 == Ui:GetFbrbidWeatherSetting() then
    return
  end
  local nRainType = self:ShouldRain(me.nMapTemplateId)
  if 0 == nRainType then
    Ui:StopRaining()
  elseif 1 == nRainType then
    Ui:StartRaining(1)
  elseif 2 == nRainType then
    Ui:StartRaining(2)
  end
end
