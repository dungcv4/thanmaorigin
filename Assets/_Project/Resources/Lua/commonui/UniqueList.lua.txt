function UniqueList:new()
  return {
    tbSet = {},
    
    tbList = List.new()
  }
end

function UniqueList:pushFront(list, value, key)
  if list.tbSet[key] then
    return
  end
  List.pushFront(list.tbList, {value = value, key = key})
  list.tbSet[key] = true
end

function UniqueList:pushBack(list, value, key)
  if list.tbSet[key] then
    return
  end
  List.pushBack(list.tbList, {value = value, key = key})
  list.tbSet[key] = true
end

function UniqueList:popFront(list)
  local param = List.popFront(list.tbList)
  list.tbSet[param.key] = nil
  return param.value
end

function UniqueList:popBack(list)
  local param = List.popBack(list.tbList)
  list.tbSet[param.key] = nil
  return param.value
end

function UniqueList:GetFront(list)
  local first = list.tbList.first
  if first > list.tbList.last then
    return nil
  end
  local param = list.tbList[first]
  return param.value
end

function UniqueList:GetBack(list)
  local last = list.tbList.last
  if last < list.tbList.first then
    return nil
  end
  local param = list.tbList[last]
  return param.value
end

function UniqueList:Count(list)
  return List.Count(list.tbList)
end

function UniqueList:IsEmpty(list)
  return List.IsEmpty(list.tbList)
end

function UniqueList:IsQueueOnlyKey(list, key)
  if list.tbSet[key] then
    return false
  end
  return true
end
