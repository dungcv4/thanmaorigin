local SHOW_TIME = 10
local tbWnd = Ui:CreateWindow("UIComboKillCount")
tbWnd.tbControls = {
  txtComboCount = "imgBG/lianzhan/txtCount"
}

function tbWnd:OnOpen(nCombo, nShowTime)
  self:CloseComboTimer()
  if 0 == nCombo then
    return false
  end
  if not nShowTime or nShowTime <= 0 then
    nShowTime = SHOW_TIME
  end
  self.nComboTimer = Timer:Register(nShowTime * Env.GAME_FPS, self.CloseLianZhen, self)
  self:UpdateComboNum(nCombo)
end

function tbWnd:CloseLianZhen()
  self.nComboTimer = nil
  self:Close()
end

function tbWnd:UpdateComboNum(nCombo)
  self.pPanel:Label_SetText(self.txtComboCount, nCombo)
end

function tbWnd:CloseComboTimer()
  if self.nComboTimer then
    Timer:Close(self.nComboTimer)
    self.nComboTimer = nil
  end
end

function tbWnd:OnClose()
  self:CloseComboTimer()
  self.pPanel:Label_SetText(self.txtComboCount, "")
end
