local tbWnd = Ui:CreateWindow("UINumKeyboard")
local DEC = 10
tbWnd.tbControls = {
  mainPanel = "BG",
  btnConfirm = "BG/NumList/btnConfirm",
  btnDelete = "BG/NumList/btnDelete"
}
tbWnd.tbOnClick = {
  btnConfirm = function(self)
    self:OnComfirm()
  end,
  btnDelete = function(self)
    self:OnDelete()
  end
}

function tbWnd:OnCreate()
  self.nInput = 0
  for i = 0, DEC - 1 do
    Ui:Button_BindEvent(self.pPanel, string.format("BG/NumList/Num%s", i), self.OnPress, self, i)
  end
  self.rectTransform = self.pPanel:GetRectTransform(self.mainPanel)
  self:SetClickEmptyToClose()
end

function tbWnd:OnOpen(fnUpdate, fnUpdateNum, tbOpenPos, nCurNum)
  self.fnUpdate = fnUpdate
  self.fnUpdateNum = fnUpdateNum
  self.nInput = nCurNum or 0
  self.bManualChanged = false
  if tbOpenPos then
    self.rectTransform.anchoredPosition = Ui.Vector2(tbOpenPos.x, tbOpenPos.y)
  end
end

function tbWnd:OnPress(nPress)
  self.nInput = self.nInput * DEC + nPress
  self:UpdateNum()
end

function tbWnd:OnDelete()
  self.nInput = math.floor(self.nInput / DEC)
  self:UpdateNum()
end

function tbWnd:OnComfirm()
  self:Close()
end

function tbWnd:OnClose()
  self:UpdateListener()
end

function tbWnd:UpdateListener()
  self.nInput = self.nInput or 0
  if self.fnUpdate and self.bManualChanged then
    local nResult = self.fnUpdate(self.nInput) or self.nInput
    self.nInput = nResult
  end
end

function tbWnd:UpdateNum()
  if self.fnUpdateNum then
    local nResult = self.fnUpdateNum(self.nInput)
    self.nInput = nResult
  end
  self.bManualChanged = true
end
