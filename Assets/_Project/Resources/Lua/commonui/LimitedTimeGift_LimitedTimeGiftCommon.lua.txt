function LimitedTimeGift:IsTrigger(nTriggerId)
  nTriggerId = math.floor(nTriggerId)
  if nTriggerId < self.TRIGGER.LEVEL or nTriggerId >= self.TRIGGER.MAX then
    return false
  end
  return true
end

function LimitedTimeGift:LoadSetting()
  self.tbGiftSettingsByGiftType = {}
  self.tbGiftSettingsByGiftId = {}
  self.tbTimeFrame = {}
  local tbSettings = Lib:LoadTabFileEx("Setting/LimitedTimeGift/LimitedTimeGiftSetting.tab", "ddsddsddsddd", "", {
    "nGiftType",
    "nGiftId",
    "szGiftName",
    "nAwardId",
    "nAwardKey",
    "szTimeFrame",
    "nGiftTriggerParam",
    "nGiftLifeTime",
    "szMoneyType",
    "nPrice",
    "nFullPrice",
    "nMaxBuyCount"
  }, nil, nil, false)
  for _, tbInfo in pairs(tbSettings) do
    self.tbGiftSettingsByGiftType[tbInfo.nGiftType] = self.tbGiftSettingsByGiftType[tbInfo.nGiftType] or {}
    self.tbGiftSettingsByGiftType[tbInfo.nGiftType][tbInfo.szTimeFrame] = self.tbGiftSettingsByGiftType[tbInfo.nGiftType][tbInfo.szTimeFrame] or {}
    table.insert(self.tbGiftSettingsByGiftType[tbInfo.nGiftType][tbInfo.szTimeFrame], tbInfo)
    if "" == tbInfo.szTimeFrame then
      for szTimeFrame, tbSettings1 in pairs(self.tbGiftSettingsByGiftType[tbInfo.nGiftType]) do
        if "" ~= szTimeFrame then
          table.insert(tbSettings1, tbInfo)
        end
      end
    else
      self.tbTimeFrame[tbInfo.szTimeFrame] = true
    end
    self.tbGiftSettingsByGiftId[tbInfo.nGiftType] = self.tbGiftSettingsByGiftId[tbInfo.nGiftType] or {}
    self.tbGiftSettingsByGiftId[tbInfo.nGiftType][tbInfo.nGiftId] = tbInfo
  end
  for _, tbInfo in pairs(self.tbGiftSettingsByGiftType) do
    for _, tbSettings1 in pairs(tbInfo) do
      table.sort(tbSettings1, function(a, b)
        return a.nGiftTriggerParam < b.nGiftTriggerParam
      end)
    end
  end
  self.tbGiftSettingsByGiftType = SetReadOnly(self.tbGiftSettingsByGiftType)
  self.tbGiftSettingsByGiftId = SetReadOnly(self.tbGiftSettingsByGiftId)
end

function LimitedTimeGift:GetGiftSettingByGiftType(nGiftType)
  local tbSettings = self.tbGiftSettingsByGiftType[nGiftType]
  if not tbSettings then
    return
  end
  if 1 == Lib:CountTB(tbSettings) and tbSettings[""] then
    return tbSettings[""]
  end
  local szCurTimeFrame = Lib:GetMaxTimeFrame(self.tbTimeFrame)
  return tbSettings[szCurTimeFrame]
end

function LimitedTimeGift:GetGiftSettingByGiftId(nGiftType, nGiftId)
  local tbSettings = self.tbGiftSettingsByGiftId[nGiftType]
  if not tbSettings then
    return
  end
  return tbSettings[nGiftId]
end

LimitedTimeGift:LoadSetting()
