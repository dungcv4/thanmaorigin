function Npc:BeforeDoDeath(pNpc, pKiller)
  local nNpcId = pNpc.nId
  
  Npc:BossBeforeDoDeath(nNpcId)
  Npc:CallClassFunc(pNpc, "BeforeDoDeath", pKiller)
end

function Npc:DelayDoDeath(nNpcId)
end

function Npc:BossBeforeDoDeath(nNpcId)
  local pNpc = KNpc.GetById(nNpcId)
  if pNpc then
    local nBloodStyle = pNpc.GetBloodStyle()
    if nBloodStyle ~= Npc.BloodStyle.blood_style_boss then
      return
    end
  end
  Npc:DelayDoDeath(nNpcId)
end
