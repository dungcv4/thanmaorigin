DingFengPK.GuessGrade = DingFengPK.GuessGrade or {}
local GuessGrade = DingFengPK.GuessGrade

function GuessGrade:UpdataGuessData()
  local nMyCount = 0
  for nPhase, tbGuessData in pairs(DingFengPK.tbGuessData) do
    for nIndex, tbData in pairs(tbGuessData) do
      if tbData.nResult == DingFengPK.Win and tbData.tbGuessCount[DingFengPK.GuessCfg.nLuoGuItemId] then
        nMyCount = nMyCount + tbData.tbGuessCount[DingFengPK.GuessCfg.nLuoGuItemId]
      end
    end
  end
  local tbCacheData = {nMyCount = nMyCount}
  self.tbCacheData = tbCacheData
  EventNotify.OnNotify(EventNotify.emNOTIFY_UPDATE_EXTRA_AWARD)
end

function GuessGrade:GetMyCount()
  if not self.tbCacheData or not self.tbCacheData.nMyCount then
    return 0
  end
  return self.tbCacheData.nMyCount
end

function GuessGrade:UpdateLevelList(tbWnd)
  local nMyCount = self:GetMyCount()
  local tbLevel = DingFengPK.GuessGradeAward
  tbWnd.tbLevelList:Clear()
  for nIndex, tbInfo in ipairs(tbLevel) do
    tbWnd.tbLevelList:SetElement(nIndex, function(tbElement)
      local tbAward = {
        {
          "YuanBao",
          tbInfo.nYuanBao
        }
      }
      local tbShowAwardScrollView = Ui:NewItemGridScrollView(tbElement, tbWnd.tbTemplateElement.reward, 1, 1)
      tbShowAwardScrollView:UpdateGenericItem(tbAward, true)
      tbElement.pPanel:Label_SetText(tbWnd.tbTemplateElement.txtNumber, i18n.EasyGet("\230\136\144\229\138\159\229\138\169\229\168\129{0}\228\184\170", tbInfo.nNums))
      local bCompleted = nMyCount >= tbInfo.nNums
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.Prg, not bCompleted)
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.imgBar, not bCompleted)
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.txtPrg, not bCompleted)
      tbElement.pPanel:SetActive(tbWnd.tbTemplateElement.imgComplete, bCompleted)
      tbElement.pPanel:Label_SetText(tbWnd.tbTemplateElement.CompleteText, i18n.EasyGet("\229\183\178\232\190\190\230\136\144"))
      if not bCompleted then
        tbElement.pPanel:Label_SetText(tbWnd.tbTemplateElement.txtPrg, string.format("%d/%d", nMyCount, tbInfo.nNums))
        tbElement.pPanel:Sprite_SetFillAmount(tbWnd.tbTemplateElement.imgBar, 0, nMyCount / tbInfo.nNums, 1, true)
      end
    end)
  end
end

function GuessGrade:UpdateOtherInfo(tbWnd)
  local nMyCount = self:GetMyCount()
  tbWnd.pPanel:Label_SetText(tbWnd.szMineNumber, i18n.EasyGet("\230\136\145\230\173\163\231\161\174\229\138\169\229\168\129\233\148\163\233\188\147\230\128\187\230\149\176\239\188\154{0}", nMyCount))
  tbWnd.pPanel:SetActive(tbWnd.szTongNumber, false)
  tbWnd.pPanel:Label_SetText(tbWnd.txtTitleNumber, i18n.EasyGet("\230\173\163\231\161\174\229\138\169\229\168\129\233\148\163\233\188\147\230\149\176\233\135\143"))
  tbWnd.pPanel:SetActive(tbWnd.Template, false)
  tbWnd.pPanel:SetActive(tbWnd.btnAdd, false)
  tbWnd.pPanel:SetActive(tbWnd.btnHelp, false)
  tbWnd.pPanel:Label_SetText(tbWnd.txtInfo, i18n.EasyGet("\229\165\150\229\138\177\233\128\154\232\191\135\233\130\174\228\187\182\229\143\145\230\148\190"))
  tbWnd.pPanel:Button_SetText(tbWnd.btnGo, i18n.EasyGet("\229\137\141\229\190\128\229\138\169\229\168\129"), "Text")
end

function GuessGrade:OnClickbtnGo(tbWnd)
  tbWnd:Close()
end

function GuessGrade:OnUiEvent(tbWnd, szEvent, ...)
  if tbWnd.ClassName ~= "UIReward_Newyear" then
    return
  end
  if "OnOpen" == szEvent then
    DingFengPK:RequestGuessData()
    self:UpdateLevelList(tbWnd)
    self:UpdateOtherInfo(tbWnd)
  end
  if "OnClick" == szEvent then
    local tbParam = {
      ...
    }
    if "btnGo" == tbParam[1] then
      self:OnClickbtnGo(tbWnd)
    end
  end
  if "UpdateReward" == szEvent then
    self:UpdateLevelList(tbWnd)
    self:UpdateOtherInfo(tbWnd)
  end
end
