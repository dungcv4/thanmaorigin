local tbWnd = Ui:CreateWindow("UITieFuChengResult")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  goResultTemplate = "node/imgBG/Content/ScrollView/Viewport/Content/Template",
  goMyRank = "node/imgBG/Content/MyRank"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbResultScroll = Ui:NewScrollView(self, self.goResultTemplate, nil, nil, true)
end

function tbWnd:OnOpen(tbData)
  if tbData then
    self:Update(tbData)
  end
end

function tbWnd:Update(tbData)
  self.tbResultScroll:Reset()
  for nIndex, tbInfo in ipairs(tbData) do
    self.tbResultScroll:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtRank", nIndex)
      pPanel:Sprite_SetSprite("imgFaction", Faction:GetIcon(tbInfo.nFaction))
      pPanel:Label_SetText("txtName", tbInfo.szName)
      pPanel:Label_SetText("txtLv", tostring(tbInfo.nKill))
      pPanel:Label_SetText("txtScore", tbInfo.nScore)
    end)
  end
  if tbData.myIndex and tbData.myInfo then
    self.pPanel:Label_SetText(self.goMyRank .. "/txtRank", tbData.myIndex or "")
    self.pPanel:Sprite_SetSprite(self.goMyRank .. "/imgFaction", Faction:GetIcon(tbData.myInfo.nFaction or 0))
    self.pPanel:Label_SetText(self.goMyRank .. "/txtName", tbData.myInfo.szName or "")
    self.pPanel:Label_SetText(self.goMyRank .. "/txtLv", tostring(tbData.myInfo.nKill or ""))
    self.pPanel:Label_SetText(self.goMyRank .. "/txtScore", tostring(tbData.myInfo.nScore or ""))
    self.pPanel:SetActive(self.goMyRank, true)
  else
    self.pPanel:SetActive(self.goMyRank, false)
  end
end
