local self

function _LuaPlayer.CenterMsg(szMsg, bSysMsg)
  Ui:AddCenterMsg(szMsg, bSysMsg)
end

function _LuaPlayer.CallClientScript(szFunc, ...)
  if not szFunc then
    return
  end
  if type(szFunc) == "number" then
    local szFuncMaped = s2c.id2func[szFunc]
    if not szFuncMaped then
      Log("[Error] CallClientScript cant mapping id to function", tostring(szFunc))
    end
    szFunc = szFuncMaped
  end
  self = self
  if string.find(szFunc, ":") then
    local szTable, szFunc = string.match(szFunc, "^(.*):(.*)$")
    local tb = load("return " .. szTable)()
    if tb and nil ~= tb[szFunc] then
      tb[szFunc](tb, ...)
    else
      Log("CallClientScript", szTable, szFunc, "is nil")
    end
  else
    local func = load("return " .. szFunc)()
    func(...)
  end
end

function _LuaPlayer.Msg(szMsg, nSenderId, tbLinkInfo)
  self = self
  nSenderId = nSenderId or ChatMgr.SystemMsgType.Tip
  ChatMgr:OnChannelMessage(ChatMgr.ChannelType.System, 0, nSenderId, "", 0, 0, 0, szMsg, nil, nil, nil, tbLinkInfo)
end

function _LuaPlayer.SwitchRide(state)
  self = self
  local nDoing = me.GetDoing()
  if nDoing == Npc.Doing.sit then
    me.CenterMsg(i18n.Get("CantRide4Sit"))
    return
  end
  if nDoing == Npc.Doing.do_qingkung or nDoing == Npc.Doing.do_hover then
    me.CenterMsg(i18n.Get("CantRide4QingKung"))
    return
  end
  local pNpc = me.GetNpc()
  if not pNpc then
    return
  end
  if 0 == state then
    SetRideState(state)
    return 1
  end
  if Map:IsForbidRide(me.nMapTemplateId) then
    return
  end
  local nResult = pNpc.CanDoRideHorse()
  if 2 == nResult then
    return
  end
  if 1 == nResult then
    SetRideState(state)
    return 1
  end
end
