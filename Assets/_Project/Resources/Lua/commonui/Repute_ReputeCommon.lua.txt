function Repute:GetMaxShowEquipLevel()
  local nRet = 1
  for index, value in ipairs(Repute.tbPlayerLevel2EquipLevel) do
    local nCurLevel = me.nLevel
    local nZongShiLevel = ZongShiSkill:GetPlayerZongShiLevel(me)
    nCurLevel = nCurLevel + nZongShiLevel
    if nCurLevel >= value.nLevel then
      nRet = value.nEquipLevel
    end
  end
  return nRet
end

Require("CommonScript/Shop/Shop.lua")

function Shop:LoadReputeShop()
  self.tbReputeShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/ReputeShop.tab", "ddsdd", nil, {
    "nWareId",
    "nSort",
    "szReputeType",
    "nEquipPos",
    "nLevel"
  })
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.Repute then
      print("Load ReputeShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.Repute, tbWareInfo and tbWareInfo.szShopType)
    else
      self.tbReputeShopWares[v.szReputeType] = self.tbReputeShopWares[v.szReputeType] or {}
      local tbInfo = {
        nWareId = v.nWareId,
        nSort = v.nSort,
        szReputeType = v.szReputeType,
        nEquipPos = v.nEquipPos,
        nLevel = v.nLevel
      }
      table.insert(self.tbReputeShopWares[v.szReputeType], tbInfo)
    end
  end
end

Shop:LoadReputeShop()

function Shop:GetReputeWareAvaliableList(szReputeType)
  local tbWareList = self.tbReputeShopWares[szReputeType] or {}
  local tbRtn = {}
  for _, v in pairs(tbWareList) do
    local tbWare = self:GetAWare(v.nWareId)
    if tbWare and self:CheckWareAvaliable(tbWare, me) then
      table.insert(tbRtn, v)
    end
  end
  if not Lib:IsEmptyTB(tbRtn) then
    table.sort(tbRtn, function(a, b)
      return a.nSort < b.nSort
    end)
  end
  return tbRtn
end

function Shop:FindReputeShopWare(szReputeType, nItemTemplateId)
  local tbList = self.tbReputeShopWares[szReputeType]
  for _, v in pairs(tbList) do
    local tbWare = Shop:GetAWare(v.nWareId)
    if tbWare and tbWare.nItemId == nItemTemplateId then
      return tbWare
    end
  end
  return nil
end
