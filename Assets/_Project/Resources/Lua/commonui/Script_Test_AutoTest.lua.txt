local snapshot = xlua.snapshot
local print = _ENV.print
local tconcat = table.concat
local tinsert = table.insert
local srep = string.rep
local type = _ENV.type
local pairs = _ENV.pairs
local tostring = _ENV.tostring
local next = _ENV.next

function AutoTest:UseItembyItemId(nItemId)
  Item:ClientUseItem(nItemId)
end

function AutoTest:UseItembyTemplateId(dwTemplateId)
  local tbItem = me.GetItemListInBag()
  for nIdx, pItem in ipairs(tbItem) do
    if dwTemplateId == pItem.dwTemplateId then
      Item:ClientUseItem(pItem.dwId)
    end
  end
end

function AutoTest:GetItemCount(dwTemplateId)
  local tbItem = me.GetItemListInBag()
  for nIdx, pItem in ipairs(tbItem) do
    if dwTemplateId == pItem.dwTemplateId then
      return pItem.nCount
    end
  end
  return 0
end

function AutoTest:tprint(tbl, indent)
  indent = indent or 0
  for k, v in pairs(tbl) do
    formatting = string.rep("  ", indent) .. k .. ": "
    if "table" == type(v) then
      print(formatting)
      tprint(v, indent + 1)
    elseif "boolean" == type(v) then
      print(formatting .. tostring(v))
    else
      print(formatting .. v)
    end
  end
end

function AutoTest:BeginSnap()

end

local function print_r(root)
  local cache = {
    [root] = "."
  }
  
  local function _dump(t, space, name)
    local temp = {}
    for k, v in pairs(t) do
      local key = tostring(k)
      if cache[v] then
        tinsert(temp, "+" .. key .. " {" .. cache[v] .. "}")
      elseif "table" == type(v) then
        local new_key = name .. "." .. key
        cache[v] = new_key
        tinsert(temp, "+" .. key .. _dump(v, space .. (next(t, k) and "|" or " ") .. srep(" ", #key), new_key))
      else
        tinsert(temp, "+" .. key .. " [" .. tostring(v) .. "]")
      end
    end
    return tconcat(temp, "\n" .. space)
  end
  
  print(_dump(root, "", ""))
end

function AutoTest:EndSnap()
  collectgarbage("collect")
  collectgarbage("collect")
  collectgarbage("collect")
  local S2 = snapshot()
  print("--------------------------------")
  local diff = {}
  for k, v in pairs(S2) do
    if not self.S1[k] then
      diff[k] = v
    end
  end
  print_r(diff)
  return true
end
