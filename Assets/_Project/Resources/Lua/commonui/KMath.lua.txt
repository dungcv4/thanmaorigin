function KMath:g_Cos(nDirection)
  if -1 == nDirection then
    nDirection = 0
  end
  return KMath:g_Sin(KMath.DIRECTION_COUNT / 4 - nDirection)
end

function KMath:g_Sin(nDirection)
  if -1 == nDirection then
    nDirection = 0
  end
  local nResult = 0
  nDirection = nDirection % KMath.DIRECTION_COUNT
  if nDirection < 0 then
    nDirection = nDirection + KMath.DIRECTION_COUNT
  end
  if nDirection <= KMath.DIRECTION_COUNT / 4 then
    nResult = KMath.g_nSinValues[nDirection + 1]
  elseif nDirection <= KMath.DIRECTION_COUNT / 2 then
    nResult = KMath.g_nSinValues[KMath.DIRECTION_COUNT / 2 - nDirection + 1]
  elseif nDirection <= KMath.DIRECTION_COUNT * 3 / 4 then
    nResult = -KMath.g_nSinValues[nDirection - KMath.DIRECTION_COUNT / 2 + 1]
  else
    nResult = -KMath.g_nSinValues[KMath.DIRECTION_COUNT - nDirection + 1]
  end
  return nResult
end

function KMath:GetPosByLenDir(nX, nY, nLen, nDir)
  nDir = math.floor(nDir)
  local nDestX = nX + KMath:g_Cos(nDir) * nLen / KMath.SIN_COS_NUMBER
  local nDestY = nY + KMath:g_Sin(nDir) * nLen / KMath.SIN_COS_NUMBER
  return nDestX, nDestY
end

function KMath:Clamp(v, min, max)
  if v < min then
    return min
  end
  if max < v then
    return max
  end
  return v
end
