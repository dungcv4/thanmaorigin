local tbUi = Ui:CreateWindow("UITeacherFind")

function tbUi:NotifyEvents()
  local tbRegEvent = {
    {
      EventNotify.emNoTIFY_MASTERPRENTICE_SyncRegisterInfo,
      self.UpdateView
    },
    {
      EventNotify.emNoTIFY_MASTERPRENTICE_SyncMPData,
      self.UpdateView
    },
    {
      EventNotify.emNoTIFY_MASTERPRENTICE_SyncMPInfo,
      self.UpdateView
    }
  }
  return tbRegEvent
end

tbUi.ItemTemplate = {
  RoleTemplete = "node/imgBG/Content/ScrollViewList/Viewport/Grid/RoleTemplete"
}
tbUi.tbControls = {
  btnFindTea = "node/imgBG/Content/btnList/btnFindTea",
  btnFindStu = "node/imgBG/Content/btnList/btnFindStu",
  toggleHide = "node/imgBG/Content/SatisfyObj/toggleHide",
  btnRefresh = "node/imgBG/Content/SatisfyObj/btnFresh",
  btnRelease = "node/imgBG/Content/DeclarationObj/btnRelease",
  btnClose = "node/imgBG/btnClose",
  txtDeclaration = "node/imgBG/Content/DeclarationObj/inputDeclaration"
}

function tbUi:OnCreate()
  self.tbFindBtnGroup = Ui:NewButtonGroup(self, {
    [0] = self.btnFindTea,
    [1] = self.btnFindStu
  }, true, function(self, nIndex)
    self.nPage = 1
    self.bMaster = nIndex
    MasterPrentice:SearchRegisterInfo(1 - self.bMaster, self.nPage)
  end)
  self.tbItemScrollView = Ui:NewScrollView(self, self.ItemTemplate.RoleTemplete)
end

function tbUi:OnOpen(bMaster)
  self.nPage = 1
  self.bMaster = bMaster or 0
  MasterPrentice:SearchRegisterInfo(1 - self.bMaster, self.nPage)
  self.tbFindBtnGroup:Select(self.bMaster)
end

function tbUi:UpdateView()
  local tInfos
  local masterCtrl = MasterPrentice:GetMPInfo()
  self.tbItemScrollView:Clear()
  if 1 == self.bMaster then
    tInfos = MasterPrentice.RegisterInfo.tbPrentices
    self.nPage = MasterPrentice.RegisterInfo.nPrenticePage
    local szDeclaration
    if masterCtrl.szAnnouncement1 and masterCtrl.szAnnouncement1 ~= "" then
      szDeclaration = masterCtrl.szAnnouncement1
    else
      szDeclaration = i18n.EasyGet("\229\175\187\228\184\128\231\136\177\229\190\146\239\188\140\229\128\190\229\155\138\231\155\184\230\142\136\239\188\140\229\133\177\233\151\175\229\137\145\228\190\160\239\188\129")
    end
    self.pPanel:Input_SetText(self.txtDeclaration, szDeclaration)
    self.bChange = 0
    self.pPanel:Toggle_SetChecked("node/imgBG/Content/SatisfyObj/toggleHide", not masterCtrl.bRegisterMaster)
    self.bChange = 1
    self.pPanel:Label_SetText("node/imgBG/txtTitle", i18n.EasyGet("\229\175\187\230\137\190\229\190\146\229\188\159"))
    self.pPanel:Label_SetText("node/imgBG/Content/DeclarationObj/txtDeclaration", i18n.EasyGet("\230\148\182\229\190\146\229\174\163\232\168\128"))
    local tbLimitLevel = MasterPrentice.Cfg.MASTER_LEVEL
    local nLimitLevel = 0
    for szOpenTime, nLevel in pairs(tbLimitLevel) do
      if 1 == GetTimeFrameState(szOpenTime) then
        nLimitLevel = math.max(nLimitLevel, nLevel)
      end
    end
    if nLimitLevel > me.nTotalLevel then
      self.pPanel:SetActive("node/imgBG/Content/DeclarationObj", false)
      self.pPanel:SetActive("node/imgBG/Content/SatisfyObj", false)
      self.pPanel:SetActive("node/imgBG/Content/txtEmpty", false)
      self.pPanel:SetActive("node/imgBG/Content/EmptyObj/StudentRulesObj", true)
      self.pPanel:Label_SetText("node/imgBG/Content/EmptyObj/StudentRulesObj/txtStuRule01", i18n.EasyGet("\231\142\169\229\174\182\232\167\146\232\137\178\232\190\190\229\136\176{0}\231\186\167", nLimitLevel))
      self.pPanel:SetActive("node/imgBG/Content/EmptyObj/StudentRulesObj/txtStuRule01/imgRight", false)
      return
    else
      self.pPanel:SetActive("node/imgBG/Content/DeclarationObj", true)
      self.pPanel:SetActive("node/imgBG/Content/SatisfyObj", true)
      self.pPanel:SetActive("node/imgBG/Content/EmptyObj/StudentRulesObj", false)
      if #tInfos < 1 then
        self.pPanel:SetActive("node/imgBG/Content/txtEmpty", true)
        self.pPanel:Label_SetText("node/imgBG/Content/txtEmpty", i18n.EasyGet("\230\154\130\230\151\182\230\178\161\230\156\137\229\143\175\228\187\165\230\139\156\229\184\136\231\154\132\229\176\145\228\190\160"))
        self.pPanel:SetActive("node/imgBG/Content/ScrollViewList", false)
        return
      else
        self.pPanel:SetActive("node/imgBG/Content/txtEmpty", false)
        self.pPanel:SetActive("node/imgBG/Content/ScrollViewList", true)
      end
    end
  else
    tInfos = MasterPrentice.RegisterInfo.tbMasters
    self.nPage = MasterPrentice.RegisterInfo.nMasterPage
    local szDeclaration
    if masterCtrl.szAnnouncement2 and "" ~= masterCtrl.szAnnouncement2 then
      szDeclaration = masterCtrl.szAnnouncement2
    else
      szDeclaration = i18n.EasyGet("\229\175\187\228\184\128\229\144\141\229\184\136\239\188\140\229\133\137\232\128\128\229\184\136\233\151\168\239\188\129")
    end
    self.pPanel:Input_SetText(self.txtDeclaration, szDeclaration)
    self.bChange = 0
    self.pPanel:Toggle_SetChecked("node/imgBG/Content/SatisfyObj/toggleHide", not masterCtrl.bRegisterPrentice)
    self.bChange = 1
    self.pPanel:Label_SetText("node/imgBG/txtTitle", i18n.EasyGet("\229\175\187\230\137\190\229\184\136\231\136\182"))
    self.pPanel:Label_SetText("node/imgBG/Content/DeclarationObj/txtDeclaration", i18n.EasyGet("\230\139\156\229\184\136\229\174\163\232\168\128"))
    self.pPanel:SetActive("node/imgBG/Content/DeclarationObj", true)
    self.pPanel:SetActive("node/imgBG/Content/SatisfyObj", true)
    self.pPanel:SetActive("node/imgBG/Content/EmptyObj/StudentRulesObj", false)
    if #tInfos < 1 then
      self.pPanel:SetActive("node/imgBG/Content/txtEmpty", true)
      self.pPanel:Label_SetText("node/imgBG/Content/txtEmpty", i18n.EasyGet("\230\154\130\230\151\182\230\178\161\230\156\137\229\143\175\228\187\165\230\148\182\229\190\146\231\154\132\229\164\167\228\190\160"))
      self.pPanel:SetActive("node/imgBG/Content/ScrollViewList", false)
      return
    else
      self.pPanel:SetActive("node/imgBG/Content/txtEmpty", false)
      self.pPanel:SetActive("node/imgBG/Content/ScrollViewList", true)
    end
  end
  for i = 1, #tInfos do
    self.tbItemScrollView:SetElement(i, function(tbElement)
      self:SetScrollViewData(tbElement, i, tInfos[i], self.bMaster)
    end)
  end
  self.tbItemScrollView:GoTop()
end

function tbUi:SetScrollViewData(tbElement, nIdx, tbInfo, bMaster)
  local panel = tbElement.pPanel
  self.tbData = {}
  self.tbData.dwRoleId = tbInfo[1]
  self.tbData.nLevel = tbInfo[2]
  self.tbData.szName = tbInfo[3]
  self.tbData.szKinOrTongName = "" == tbInfo[5] and i18n.EasyGet("\230\154\130\230\151\160") or tbInfo[5]
  self.tbData.nPortrait = tbInfo[4]
  self.tbData.nHonorLevel = tbInfo[7]
  self.tbData.nFaction = tbInfo[8]
  if 1 == bMaster then
    panel:Label_SetText("btnFind/Text", i18n.EasyGet("\230\148\182\229\190\146"))
  else
    panel:Label_SetText("btnFind/Text", i18n.EasyGet("\230\139\156\229\184\136"))
  end
  panel:Label_SetText("info/txtPlayerDec", tbInfo[6] or "")
  local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
  tbPortraitObjPanel:SetRoleIcon(self.tbData.nPortrait or self.tbData.nFaction)
  tbPortraitObjPanel:SetPortraitFrame(self.tbData.nPortraitFrame)
  tbPortraitObjPanel:SetFactionIcon(self.tbData.nFaction)
  tbPortraitObjPanel:SetLevel(self.tbData.nLevel)
  panel:Label_SetText("info/txtTongName/txtTong", i18n.EasyGet("\229\174\182\230\151\143\239\188\154"))
  if self.tbData.szKinOrTongName then
    panel:SetActive("info/txtTongName", true)
    panel:Label_SetText("info/txtTongName", self.tbData.szKinOrTongName)
  else
    panel:SetActive("info/txtTongName", false)
  end
  panel:Label_SetText("info/txtPlayerName", self.tbData.szName)
  Ui:Button_BindEvent(panel, "PortraitObj", function()
    Ui:OpenWindow("UIRightPopUpMenu", "roleselectFromUI", tbInfo, self, 317, -170)
  end)
  Ui:Button_BindEvent(panel, "btnFind", function()
    MasterPrentice:ApplyApprentice(tbInfo[1], bMaster)
  end)
end

tbUi.tbOnClick = tbUi.tbOnClick or {}
tbUi.tbOnToggleClick = {
  toggleHide = function(self, isOn)
    if 1 ~= self.bChange then
      return
    end
    if isOn then
      MasterPrentice:Unregister(self.bMaster)
    else
      local szDeclaration = self.pPanel:Input_GetText(self.txtDeclaration)
      MasterPrentice:Register(self.bMaster, szDeclaration)
    end
  end
}

function tbUi.tbOnClick:btnRefresh()
  if not self.nPage then
    return
  end
  MasterPrentice:SearchRegisterInfo(1 - self.bMaster, self.nPage + 1)
end

function tbUi.tbOnClick:btnRelease()
  local szDeclaration = self.pPanel:Input_GetText(self.txtDeclaration)
  local nlen = Lib:StrLen(szDeclaration)
  if nlen > MasterPrentice.Cfg.MAX_DECLARATION_LENGTH then
    me.CenterMsg(i18n.EasyGet("\232\182\133\232\191\135\230\156\128\229\164\167\229\173\151\230\149\176\233\153\144\229\136\182"))
  elseif 0 == nlen then
    me.CenterMsg(i18n.EasyGet("\228\189\160\229\176\154\230\156\170\232\190\147\229\133\165\229\174\163\232\168\128\239\188\140\232\175\183\233\135\141\232\175\149"))
  elseif ReplaceLimitWords(szDeclaration) then
    me.CenterMsg(i18n.EasyGet("\229\174\163\232\168\128\230\156\137\230\149\143\230\132\159\229\173\151\231\172\166\239\188\140\232\175\183\233\135\141\232\175\149"))
  else
    local masterCtrl = MasterPrentice:GetMPInfo()
    local bRegisted
    if 1 == tbUi.bMaster then
      bRegisted = masterCtrl.bRegisterMaster
    else
      bRegisted = masterCtrl.bRegisterPrentice
    end
    if bRegisted then
      MasterPrentice:SetAnnouncement(tbUi.bMaster, szDeclaration)
    else
      MasterPrentice:Register(self.bMaster, szDeclaration)
    end
  end
end

function tbUi.tbOnClick:btnClose()
  self:Close()
end

function tbUi:onClickRole()
  Ui:OpenWindow("UIRightPopUpMenu", "roleselectFromUI", tbUi.tbData, self, 317, -170)
end
