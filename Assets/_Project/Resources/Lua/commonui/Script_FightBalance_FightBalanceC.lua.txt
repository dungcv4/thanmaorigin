function FightBalance:OnAdjustAttr(tbAttrList, nDevPercent)
  me.tbFightBalanceCache = {tbAttrList = tbAttrList, nDevPercent = nDevPercent}
  
  for _, tbAttr in pairs(tbAttrList) do
    me.ApplyMagicAttrib(tbAttr.szAttr, table.unpack(tbAttr.tbValue))
  end
end

function FightBalance:GetDevPercent()
  local tbCache = me.tbFightBalanceCache
  return tbCache and tbCache.nDevPercent or nil
end
