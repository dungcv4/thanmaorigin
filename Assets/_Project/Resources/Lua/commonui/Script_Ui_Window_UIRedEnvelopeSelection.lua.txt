local tbWnd = Ui:CreateWindow("UIRedEnvelopeSelection")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_CHANGE_MONEY,
      self.OnChangeMoney,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  UIMoneyPanel = "node/UIMoneyPanel",
  BtnJinbi = "node/imgBG/TongRedBagPanel/Tab/BtnJinbi",
  BtnYuanbao = "node/imgBG/TongRedBagPanel/Tab/BtnYuanbao",
  YuanbaoPanel = "node/imgBG/TongRedBagPanel/YuanbaoPanel",
  JinbiPanel = "node/imgBG/TongRedBagPanel/JinbiPanel",
  YuanbaoTxtInput = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/RedBagMoney/UIAddAndSubGroup/btnInput/txtInput",
  YuanbaoCountUIAddAndSubGroup = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/RedBagNumber/UIAddAndSubGroup",
  YuanbaoCounttxtInput = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/RedBagNumber/UIAddAndSubGroup/btnInput/txtInput",
  YuanbaotxtCoinCost = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/Spend/txtCoinCost",
  YuanbaoInputField = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/InputField",
  GoldValueUIAddAndSubGroup = "node/imgBG/TongRedBagPanel/JinbiPanel/Send/NormalPanel/RedBagMoney/UIAddAndSubGroup",

  YuanbaoBtnGrant = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/BtnGrant",

  YuanBaoPlaceholder = "node/imgBG/TongRedBagPanel/YuanbaoPanel/SendSetting/SettingPanel/InputField/Placeholder",
  GoldPlaceholder = "node/imgBG/TongRedBagPanel/JinbiPanel/Send/NormalPanel/InputField/Placeholder"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  GoldBtnGrant = function(self)
    local szDesc = self.pPanel:Input_GetText(tbControls.GoldInputField)
    self:SendRedBad(Currency.TYPE.Gold, self.nGoldRedBagValue, self.nGoldRedBagCount, szDesc)
  end,
  YuanbaoBtnGrant = function(self)
    local nValue = Wedding.Cfg.nYuanbaoRedBagValue
    local szDesc = self.pPanel:Input_GetText(tbControls.YuanbaoInputField)
    self:SendRedBad(Currency.TYPE.YuanBao, nValue, self.nYuanbaoRedBagCount, szDesc)
  end
}

function tbWnd:OnCreate()
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, self.UIMoneyPanel)
  self.tbRedBagPanelList = Ui:NewButtonGroup(self, {
    [1] = tbControls.BtnJinbi,
    [2] = tbControls.BtnYuanbao
  }, true, function(self, nIndex)
    self:SelectRedBagPanel(nIndex)
  end)
  self.pPanel:Label_SetText(tbControls.YuanbaoTxtInput, tostring(Wedding.Cfg.nYuanbaoRedBagValue))
  self.pPanel:Label_SetText(tbControls.YuanbaotxtCoinCost, tostring(Wedding.Cfg.nYuanbaoRedBagValue))
end

function tbWnd:OnOpen()
  self.tbMoneyPanel:SetMoneyValue()
  self.tbRedBagPanelList:Select(1)
  self.pPanel:Label_SetText(tbControls.YuanBaoPlaceholder, "Chia Th\198\176\225\187\159ng (Nh\225\186\165p v\195\160o \196\145\225\187\131 nh\225\186\173p n\225\187\153i dung)")
  self.pPanel:Label_SetText(tbControls.GoldPlaceholder, "Chia Th\198\176\225\187\159ng (Nh\225\186\165p v\195\160o \196\145\225\187\131 nh\225\186\173p n\225\187\153i dung)")
end

function tbWnd:SelectRedBagPanel(nIndex)
  if 1 == nIndex then
    self.pPanel:SetActive(tbControls.YuanbaoPanel, false)
    self.pPanel:SetActive(tbControls.JinbiPanel, true)
    self:ResetGoldRedBagPanel()
  elseif 2 == nIndex then
    self.pPanel:SetActive(tbControls.YuanbaoPanel, true)
    self.pPanel:SetActive(tbControls.JinbiPanel, false)
    self:ResetYuanBaoRedBagPanel()
  end
end

function tbWnd:ResetGoldRedBagPanel()
  self.nGoldRedBagCount = 10
  self.nGoldRedBagValue = 200
  self.GoldValueAddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.GoldValueUIAddAndSubGroup)
  self:UpdateGoldValueInput(self.nGoldRedBagValue)
  self.GoldValueAddAndSubGroup:BindEvent(function(nValue)
    self.nGoldRedBagValue = self:UpdateGoldValueInput(nValue)
    return self.nGoldRedBagValue
  end)
  self.GoldValueAddAndSubGroup:SetNumPadPos(-500, -100)
  self.GoldValueAddAndSubGroup:SetEnable(true)
  self.GoldValueAddAndSubGroup:SetNumRange(RedPacketMgr.Cfg.nMinAmount, RedPacketMgr.Cfg.nMaxAmount)
  self.GoldValueAddAndSubGroup:SetInputValue(self.nGoldRedBagValue)
  self.GoldCountAddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.GoldCountUIAddAndSubGroup)
  self.GoldCountAddAndSubGroup:BindEvent(function(nValue)
    self.nGoldRedBagCount = self:UpdateGoldCountInput(nValue)
    return self.nGoldRedBagCount
  end)
  self.GoldCountAddAndSubGroup:SetNumPadPos(-500, -100)
  self.GoldCountAddAndSubGroup:SetEnable(true)
  self.GoldCountAddAndSubGroup:SetNumRange(1, 100)
  self.GoldCountAddAndSubGroup:SetInputValue(self.nGoldRedBagCount)
end

function tbWnd:UpdateGoldValueInput(nNum)
  nNum = self:CheckGoldValueInput(nNum)
  self.pPanel:Label_SetText(tbControls.GoldValuetxtInput, nNum)
  self.pPanel:Label_SetText(tbControls.GoldtxtCoinCost, nNum)
  return nNum
end

function tbWnd:CheckGoldValueInput(nNum)
  local nMinNum = RedPacketMgr.Cfg.nMinAmount
  local nMaxNum = RedPacketMgr.Cfg.nMaxAmount
  if nNum > nMaxNum then
    me.CenterMsg(i18n.Get(Shop.Terms.InputLimit))
    nNum = nMaxNum
  end
  if nMinNum > nNum then
    me.CenterMsg(i18n.Get(Shop.Terms.InputLimit))
    nNum = nMinNum
  end
  return nNum
end

function tbWnd:UpdateGoldCountInput(nNum)
  local nMaxNum = 100
  if 0 == nMaxNum then
    nMaxNum = 1
  end
  if nNum > nMaxNum then
    me.CenterMsg(i18n.Get(Shop.Terms.InputLimit))
    nNum = nMaxNum
    return nNum
  end
  self.pPanel:Label_SetText(tbControls.GoldCounttxtInput, nNum)
  return nNum
end

function tbWnd:ResetYuanBaoRedBagPanel()
  self.nYuanbaoRedBagCount = 10
  self.YuanbaoCountAddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.YuanbaoCountUIAddAndSubGroup)
  self.YuanbaoCountAddAndSubGroup:BindEvent(function(nValue)
    self.nYuanbaoRedBagCount = self:UpdateYuanbaoNumValueInput(nValue)
    return self.nYuanbaoRedBagCount
  end)
  self.YuanbaoCountAddAndSubGroup:SetEnable(true)
  self.YuanbaoCountAddAndSubGroup:SetNumRange(1, 100)
  self.YuanbaoCountAddAndSubGroup:SetInputValue(self.nYuanbaoRedBagCount)
end

function tbWnd:UpdateYuanbaoNumValueInput(nNum)
  local nMaxNum = 100
  if 0 == nMaxNum then
    nMaxNum = 1
  end
  if nNum > nMaxNum then
    me.CenterMsg(i18n.Get(Shop.Terms.InputLimit))
    nNum = nMaxNum
    return nNum
  end
  self.pPanel:Label_SetText(tbControls.YuanbaoCounttxtInput, nNum)
  return nNum
end

function tbWnd:OnChangeMoney()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:SendRedBad(szCurrency, nValue, nCount, szDesc)
  if nValue > me.GetMoney(szCurrency) then
    me.CenterMsg(i18n.EasyGet("{0}\228\184\141\232\182\179", Currency:GetName(szCurrency)))
    return
  end
  if not szDesc or "" == szDesc then
    szDesc = i18n.EasyGet("\232\128\129\230\157\191\230\137\147\232\181\143")
  end
  local bOk, nMin, nMax = Lib:CheckLength(szDesc, i18n.CurrentLanguageCode(), 0, RedPacketMgr.Cfg.nDescLenght)
  if not bOk then
    me.CenterMsg(i18n.EasyGet("\231\186\162\229\140\133\230\143\143\232\191\176\232\182\133\232\191\135\228\186\134\230\156\128\229\164\167{0}\228\184\170\229\173\151\230\149\176\231\154\132\233\153\144\229\136\182", nMax))
    return
  end
  if "" ~= szDesc and ReplaceLimitWords(szDesc) then
    me.CenterMsg(i18n.EasyGet("\231\186\162\229\140\133\230\143\143\232\191\176\229\144\171\230\156\137\230\149\143\230\132\159\232\175\141"))
    return
  end
  
  local function fnOK()
    if nValue < nCount then
      me.CenterMsg(i18n.EasyGet("\231\186\162\229\140\133\230\149\176\233\135\143\228\184\141\232\131\189\228\189\142\228\186\142\233\135\145\233\162\157\230\149\176!"))
      return
    end
    RemoteServer.OnWeddingRequest("RequestSendRedPacket", szCurrency, nValue, nCount, szDesc)
    self:Close()
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\232\138\177\232\180\185{0}{1}\229\143\145\233\128\129\230\150\176\228\186\186\230\139\188\230\137\139\230\176\148\231\186\162\229\140\133?", nValue, Currency:GetName(szCurrency)), {
    {fnOK},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end
