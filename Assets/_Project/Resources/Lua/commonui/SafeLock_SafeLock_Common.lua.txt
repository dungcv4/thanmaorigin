Require("CommonScript/SafeLock/SafeLockCfg.lua")
SafeLock.nSaveKey = GetGroupKey("SafeLock")
SafeLock.nUnLockKey = 1
SafeLock.ModuleDef = {
  Develop = 1,
  Social = 2,
  Currency = 3,
  Player = 4
}
SafeLock.OperationTypeDef = {
  SetPassWord = 1,
  UnLock = 2,
  ChangePassWord = 3,
  CancelPassWord = 4
}

function SafeLock:IsLock(pPlayer)
  return 0 == pPlayer.GetUserValue(SafeLock.nSaveKey, SafeLock.nUnLockKey)
end
