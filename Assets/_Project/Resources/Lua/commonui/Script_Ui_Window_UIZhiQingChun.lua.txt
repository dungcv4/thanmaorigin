local tbWnd = Ui:CreateWindow("UIZhiQingChun")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnClose = "node/imgBg/btnClose",
  btnGo = "node/imgBg/Content/TimeLimit/Detail/HuoDeCaiLiaoDai/btnGo",
  btnMake = "node/imgBg/Content/TimeLimit/Detail/HeChen/btnMake",
  btnGoHandIn = "node/imgBg/Content/TimeLimit/Detail/ShiYong/btnGo"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnMake = function(self)
    Ui:OpenWindow("UIZhiQingChunCompose", self.nId)
  end,
  btnGoHandIn = function(self)
    self:OpenBag()
  end,
  btnGo = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nId)
  self:RegRedPoint()
  self.nId = nId
end

function tbWnd:OnClose()
  self:UnregRedPoint()
end

function tbWnd:RegRedPoint()
  local tbRedPointEvent = {
    RedPointMgr.EVENT.QingChunQuanExtraAward,
    RedPointMgr.EVENT.QingChunQuanXingFuQuanMake,
    RedPointMgr.EVENT.QingChunQuanHuanLeQuanMake
  }
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnMake, tbRedPointEvent)
end

function tbWnd:UnregRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnMake)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnGoHandIn)
end

function tbWnd:OpenBag()
  self:Close()
  Ui:OpenWindow("UIBag")
end
