Require("CommonScript/UiDef.lua")
Ui.tbColor = Ui.tbColor or {}
local tbColor = Ui.tbColor

function tbColor:LoadColor()
  local tbColorSetting = {
    "ColorID",
    "Red",
    "Green",
    "Blue",
    "Alpha",
    "Hex"
  }
  local szType = "ddddds"
  self.tbColorSetting = {}
  local tbD = Lib:LoadTabFileEx("Setting/ColorSetting.tab", szType, nil, tbColorSetting)
  for _, tbData in pairs(tbD) do
    self.tbColorSetting[tbData.ColorID] = tbData
  end
  local tbColorDefine = {
    "ColorName",
    "nR",
    "nG",
    "nB",
    "nA",
    "Hex"
  }
  local szType2 = "sdddds"
  self.tbColorDefine = {}
  tbD = Lib:LoadTabFileEx("Setting/UI/ColorDefine.tab", szType2, "ColorName", tbColorDefine)
  for _, tbData in pairs(tbD) do
    self.tbColorDefine[tbData.ColorName] = tbData
  end
end

tbColor:LoadColor()

function tbColor:GetColorHexBySettingId(nColorId)
  if self.tbColorSetting[nColorId] then
    return self.tbColorSetting[nColorId].Hex
  end
  return self.tbColorSetting[100].Hex
end

function tbColor:GetColorHexByDefineName(szColor)
  if self.tbColorDefine[szColor] then
    return self.tbColorDefine[szColor].Hex
  end
  return self.tbColorDefine.WhiteTop.Hex
end

function tbColor:GetColorByType(nType)
  local tbColor = {}
  tbColor.Player = 502
  tbColor.Npc = 504
  tbColor.Kin = 507
  tbColor.System = 505
  tbColor.Remind = 600
  tbColor.Faction = 506
  tbColor.Friend = 502
  tbColor.Nearby = 600
  tbColor.Private = 507
  tbColor.Team = 504
  tbColor.Tong = 501
  tbColor.World = 500
  tbColor.Classify = 600
  tbColor.ClassifyParameter = 500
  tbColor.Intro = 501
  tbColor.Default = 100
  if nil ~= tbColor[nType] then
    return Ui.tbColor:GetColorHexBySettingId(tbColor[nType])
  end
  return Ui.tbColor:GetColorHexBySettingId(tbColor.Default)
end

function tbColor:GetUnityColorByDefineName(szColor)
  local Color = self.tbColorDefine[szColor]
  Color = Color or self.tbColorDefine.WhiteTop
  return CS.UnityEngine.Color(Color.nR / 255, Color.nG / 255, Color.nB / 255, Color.nA / 255)
end
