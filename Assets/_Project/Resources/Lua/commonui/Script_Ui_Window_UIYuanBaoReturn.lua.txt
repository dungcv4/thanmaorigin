local tbWnd = Ui:CreateClass("UIYuanBaoReturn", "UIPanel")
local tbControls = {
  txttips = "txttips",
  txtRemainNum = "imgTips/txtRemainGroup/txtnum",
  txtNowNum = "imgTips/txtNowGroup/txtnum",
  txtTomorrowTip = "txtTomorrowTip",
  btnAccept = "btnAccept",
  txtNowGroup = "imgTips/txtNowGroup"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnAccept = function(self)
    YuanBaoReturn:TryGainAward()
  end
}

function tbWnd:OnCreate()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnAccept, {
    RedPointMgr.EVENT.YuanBaoReturn
  })
end

function tbWnd:OnOpen()
  self:UpdatePanel()
end

function tbWnd:UpdatePanel()
  self.pPanel:Label_SetText(tbControls.txttips, i18n.EasyGet("\229\176\145\228\190\160\229\143\175\233\162\134\229\143\150\229\156\168AT2\228\184\173\229\133\133\229\128\188\231\154\132\229\133\131\229\174\157\232\191\148\232\191\152\239\188\140\229\143\170\232\131\189\233\128\137\230\139\169\228\184\128\228\184\170\232\167\146\232\137\178\233\162\134\229\143\150\229\133\131\229\174\157\232\191\148\232\191\152\231\154\132\229\165\150\229\138\177"))
  local nRemainNum, nTodayNum = YuanBaoReturn:GetRemainAndTodayNum()
  self.pPanel:Label_SetText(tbControls.txtRemainNum, nRemainNum)
  self.pPanel:Label_SetText(tbControls.txtNowNum, nTodayNum)
  if YuanBaoReturn:HasTodayGotten() then
    self.pPanel:SetActive(tbControls.txtTomorrowTip, nRemainNum > 0)
    self.pPanel:Label_SetText(tbControls.btnAccept .. "/Text", i18n.EasyGet("\229\183\178\233\162\134\229\143\150"))
    self.pPanel:Button_SetEnable(tbControls.btnAccept, false)
    self.pPanel:SetActive(tbControls.txtNowGroup, false)
  else
    self.pPanel:SetActive(tbControls.txtTomorrowTip, false)
    self.pPanel:Label_SetText(tbControls.btnAccept .. "/Text", i18n.EasyGet("\233\162\134\229\143\150"))
    self.pPanel:Button_SetEnable(tbControls.btnAccept, true)
    self.pPanel:SetActive(tbControls.txtNowGroup, true)
  end
end

function tbWnd:UpdateView()
  self:UpdatePanel()
end

function tbWnd:OnClose()
end
