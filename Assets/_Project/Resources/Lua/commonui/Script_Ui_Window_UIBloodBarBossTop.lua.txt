local tbWnd = Ui:CreateWindow("UIBloodBarBossTop")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_WELFARE_PANEL_OPEN,
      self.OnWelfarePanelOpen
    },
    {
      EventNotify.emNOTIFY_WELFARE_PANEL_CLOSE,
      self.OnWelfarePanelClose
    }
  }
end

tbWnd.tbControls = {
  Root = "Root",
  Blood = "Root/Blood",
  Name = "Root/Name",
  BloodCount = "Root/BloodCount",
  Series = "Root/Series",
  btnTarget = "Root/btnTarget",
  btnTargetUnlock = "Root/btnTargetUnlock",
  BloodBg = "Root/BloodBg"
}
tbWnd.tbOnClick = {
  btnTarget = function(self)
    self:OnClickLockBoss()
  end,
  btnTargetUnlock = function(self)
    self:OnClickUnLockBoss()
  end
}

function tbWnd:OnCreate()
  local seriesObj = self.pPanel:GetObject(self.Series)
  self.series = seriesObj:GetComponent(Ui.UISeriesSetting)
end

function tbWnd:OnOpen(tbBossHpData)
  self.pPanel:SetActive(self.btnTarget, true)
  self.pPanel:SetActive(self.btnTargetUnlock, false)
  self.pPanel:SetActive(self.Root, true)
  self:RefreshView(tbBossHpData)
  self.pNpc = KNpc.GetById(tbBossHpData.nBossId)
end

function tbWnd:RefreshView(tbBossHpData)
  self.pPanel:Label_SetText(self.BloodCount, i18n.EasyGet("X{0}", tbBossHpData.nHpNum))
  self.pPanel:Label_SetText(self.Name, i18n.EasyGet("{0}", i18n.Parse(tbBossHpData.szName)))
  if tbBossHpData.nSeries ~= nil then
    self.series:SetSeries(tbBossHpData.nSeries)
  else
    self.series:SetSeries(0)
  end
  local _, nFillAmount = math.modf(tbBossHpData.nHpRate * 10)
  if 1 == tbBossHpData.nHpRate then
    nFillAmount = 1
  end
  self.pPanel:Sprite_SetFill(self.Blood, nFillAmount)
  local nBloodColorIndex = tbBossHpData.nHpNum % 2 + 1
  local nBloodBgColorIndex = 2 - tbBossHpData.nHpNum % 2
  self.pPanel:Sprite_SetColor(self.Blood, BossHp.bloodColor[nBloodColorIndex].nR, BossHp.bloodColor[nBloodColorIndex].nG, BossHp.bloodColor[nBloodColorIndex].nB, BossHp.bloodColor[nBloodColorIndex].nA)
  self.pPanel:Sprite_SetColor(self.BloodBg, BossHp.bloodColor[nBloodBgColorIndex].nR, BossHp.bloodColor[nBloodBgColorIndex].nG, BossHp.bloodColor[nBloodBgColorIndex].nB, BossHp.bloodColor[nBloodBgColorIndex].nA)
  if tbBossHpData.nHpNum <= 1 then
    self.pPanel:SetActive(self.BloodBg, false)
  else
    self.pPanel:SetActive(self.BloodBg, true)
  end
end

function tbWnd:CloseView()
  Operation:ClearTargetNpcId()
  BossHp:ClearData()
  self:Close()
end

function tbWnd:OnClickLockBoss()
  Operation:SetLockTargetNpcId(self.pNpc.nId)
  Operation:SimpleTapEnemy(self.pNpc)
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  self.pPanel:SetActive(self.btnTarget, false)
  self.pPanel:SetActive(self.btnTargetUnlock, true)
end

function tbWnd:OnClickUnLockBoss()
  self.pPanel:SetActive(self.btnTarget, true)
  self.pPanel:SetActive(self.btnTargetUnlock, false)
  Operation:ClearTargetNpcId()
end

function tbWnd:OnWelfarePanelOpen()
  self.pPanel:SetActive(self.Root, false)
end

function tbWnd:OnWelfarePanelClose()
  self.pPanel:SetActive(self.Root, true)
end
