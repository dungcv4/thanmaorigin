local tbWnd = Ui:CreateClass("UIReputePanel", "UIPanel")
local tbControls = {
  ReputatTemplate = "ReputationList/Viewport/Content/Template",
  btnReputationShop = "btnReputationShop"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnReputationShop = function(self)
    Ui:OpenWindow("UIStore", Shop.ShopTypeStr.Repute)
  end
}

function tbWnd:OnCreate()
  self.tbReputeScrollView = Ui:NewScrollView(self, tbControls.ReputatTemplate, function(self, nIndex, tbElement)
  end)
end

function tbWnd:OnOpen()
  self:UpdataeReputeScrollView()
end

function tbWnd:OnClose()
end

function tbWnd:UpdataeReputeScrollView()
  self.tbReputeScrollView:Reset()
  for i = 1, #Repute.tbCfgReputeInfo do
    self.tbReputeScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Sprite_SetSprite("imgIcon", Repute.tbCfgReputeInfo[i].szIconRes)
      pPanel:Label_SetText("txtName", i18n.Get(Repute.tbCfgReputeInfo[i].szName))
      pPanel:Label_SetText("txtDesc", i18n.Get(Repute.tbCfgReputeInfo[i].szDesc))
      pPanel:Label_SetText("txtTips", i18n.Get(Repute.tbCfgReputeInfo[i].szTips))
      local szCurrencyType = Repute.tbReputeType2CurrencyType[i]
      local nCurNum = me.GetMoney(szCurrencyType)
      local nMaxValue = Currency:GetCfgMaxValue(szCurrencyType)
      pPanel:Label_SetText("txtProgress/txtProgressValue", nCurNum .. "/" .. nMaxValue)
      pPanel:SetActive("MarkFull", nCurNum >= nMaxValue)
      Ui:Button_BindEvent(pPanel, "btnUse", function()
        self.tbOwner:Close()
        Ui:OpenWindow("UIStore", "Repute", i)
      end, self)
    end)
  end
end
