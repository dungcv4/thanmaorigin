function PlayerHeadDataLib:OriganizeData(nPlayerId)
  local tbData = {}
  
  local pMember = KPlayer.GetPlayerObjById(nPlayerId) or KPlayer.GetRoleStayInfo(nPlayerId)
  if pMember then
    tbData.szName = pMember.szName
    tbData.nLevel = pMember.nLevel
    tbData.nFaction = pMember.nFaction
    tbData.nPortrait = pMember.nPortrait
    tbData.nHonorLevel = pMember.nHonorLevel
    tbData.nVipLevel = pMember.nVipLevel
    if pMember.dwKinId then
      tbData.szKinName = Kin:GetKinName(pMember.dwKinId)
    end
    tbData.nFightPower = FightPower:GetFightPower(nPlayerId)
    tbData.nId = nPlayerId
  end
  return tbData
end
