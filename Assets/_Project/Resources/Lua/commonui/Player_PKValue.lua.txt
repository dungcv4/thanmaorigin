PKValue.PKValue_GroupId = 37
PKValue.PKValue_ValueId = 1
PKValue.MAX_VALUE = 100
PKValue.tbSetting = PKValue.tbSetting or {}

function PKValue:LoadSetting()
  self.tbSetting = Lib:LoadTabFileEx("Setting/Player/PKValue.tab", "ddd", "nValue", {
    "nValue",
    "nExpPercent",
    "nMaxExp"
  })
  for i = 1, self.MAX_VALUE do
    assert(self.tbSetting[i])
  end
end

PKValue:LoadSetting()

function PKValue:GetPKValue(pPlayer)
  return 0
end

function PKValue:AddValue(pPlayer)
  local nCurValue = self:GetPKValue(pPlayer)
  local nAddValue = nCurValue + 1 > self.MAX_VALUE and 0 or 1
  nCurValue = nCurValue + nAddValue
  local szMsg = i18n.EasyGet("PK\229\128\188\229\162\158\229\138\160\239\188\140\229\189\147\229\137\141PK\229\128\188") .. nCurValue
  pPlayer.Msg(szMsg)
  if nCurValue >= self.MAX_VALUE and pPlayer.nPkMode == Npc.PK_MODE.emPK_MODE_KILLER then
    pPlayer.SetPkMode(Npc.PK_MODE.emPK_MODE_KIN)
  end
  pPlayer.CallClientScript("PKValue:ValueChangeEvent")
  YeSouTask:FinishTask(pPlayer, "pkvalue", {nAddValue})
end

function PKValue:ReduceValue(pPlayer)
  local nCurValue = self:GetPKValue(pPlayer)
  if nCurValue <= 0 then
    return
  end
  nCurValue = nCurValue - 1
  local szMsg = i18n.EasyGet("PK\229\128\188\229\135\143\229\176\145\239\188\140\229\189\147\229\137\141PK\229\128\188") .. nCurValue
  pPlayer.Msg(szMsg)
  pPlayer.CallClientScript("PKValue:ValueChangeEvent")
end

function PKValue:ClearValue(pPlayer)
  local nCurValue = self:GetPKValue(pPlayer)
  if nCurValue <= 0 then
    return
  end
  nCurValue = nCurValue > 10 and nCurValue - 10 or 0
  local szMsg = i18n.EasyGet("\229\183\178\230\184\133\233\153\16410\231\130\185PK\229\128\188\239\188\140\229\189\147\229\137\141PK\229\128\188") .. nCurValue
  pPlayer.Msg(szMsg)
  pPlayer.CenterMsg(i18n.EasyGet("\229\183\178\230\184\133\233\153\16410\231\130\185PK\229\128\188"))
  pPlayer.CallClientScript("PKValue:ValueChangeEvent")
end

function PKValue:CheckMaxValue(pPlayer)
  return self:GetPKValue(pPlayer) >= self.MAX_VALUE
end

function PKValue:ValueChangeEvent()
  EventNotify.OnNotify(EventNotify.emNOTIFY_GOODWILL_VALUE_CHANGE)
end
