function JiangHuZhuJiu:OpenWindow()
  if not Ui:WindowVisible("UIBoilingWine") then
    Ui:OpenWindow("UIBoilingWineEnter")
  end
end

function JiangHuZhuJiu:OpenGameWindow(nLeftTime)
  Ui:OpenWindow("UIBoilingWine")
  if Ui:WindowVisible("UIBoilingWineEnter") then
    Ui:CloseWindow("UIBoilingWineEnter")
  end
  self:OnStartGame(nLeftTime)
end

function JiangHuZhuJiu:ReOpenGameWindow()
  Ui:OpenWindow("UIBoilingWine")
end

function JiangHuZhuJiu:CreateGameReqC()
  if not TeamMgr:HasTeam(me) then
    TeamMgr:CreateOnePersonTeam()
  end
  RemoteServer.RequestJiangHuZhuJiu("CreateGameReqW")
end

function JiangHuZhuJiu:StartC()
  RemoteServer.RequestJiangHuZhuJiu("StartReqW")
end

function JiangHuZhuJiu:PlayC(nIndex)
  if not nIndex then
    return
  end
  RemoteServer.RequestJiangHuZhuJiu("PlayReqW", nIndex)
end

function JiangHuZhuJiu:UseItemC(nItemIndex, nTargetId)
  RemoteServer.RequestJiangHuZhuJiu("UseItemReqW", nItemIndex, nTargetId)
end

function JiangHuZhuJiu:SendTextC(szText)
  if "" ~= szText then
    RemoteServer.RequestJiangHuZhuJiu("SendTextW", szText)
  end
end

function JiangHuZhuJiu:LeaveGameC()
  RemoteServer.RequestJiangHuZhuJiu("LeaveGameW")
  if Ui:WindowVisible("UIBoilingWine") then
    Ui:CloseWindow("UIBoilingWine")
  end
end

function JiangHuZhuJiu:OnLogin()
  self:CheckRedPoint(me)
end

function JiangHuZhuJiu:OnSyncAwardInfo()
  JiangHuZhuJiu:CheckRedPoint(me)
  if Ui:WindowVisible("UIBoilingWineEnter") then
    Ui:GetWindow("UIBoilingWineEnter"):Update()
  end
end

function JiangHuZhuJiu:OnSyncGameInfo(tbInfo)
  if Ui:WindowVisible("UIBoilingWine") then
    Ui:GetWindow("UIBoilingWine"):Update(tbInfo)
  end
end

function JiangHuZhuJiu:OnStartGame(nLeftTime)
  self.bGameOpen = true
  self.nLeftTime = nLeftTime
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.nTimer = Timer:Register(Env.GAME_FPS, self.UpdateHud, self)
  if Ui:WindowVisible("UIHud") then
    Ui:GetWindow("UIHud"):UpdateJiangHuZhuJiuHudTime(self.nLeftTime)
  end
end

function JiangHuZhuJiu:OnUseItem()
  if Ui:WindowVisible("UIBoilingWine") then
    local tbWnd = Ui:GetWindow("UIBoilingWine")
    tbWnd:CloseItemSelectWindow()
  end
end

function JiangHuZhuJiu:OnSyncText(szText)
  if Ui:WindowVisible("UIBoilingWine") then
    local tbWnd = Ui:GetWindow("UIBoilingWine")
    tbWnd:OnSyncText(szText)
  end
end

function JiangHuZhuJiu:ShowResult(nSucc)
  if Ui:WindowVisible("UIBoilingWine") then
    Ui:OpenWindow("UIArenaResult", nSucc, 3)
    self.nResultTimer = Timer:Register(Env.GAME_FPS * 3, function()
      if Ui:WindowVisible("UIBoilingWine") then
        Ui:CloseWindow("UIBoilingWine")
        Ui:OpenWindow("UIBoilingWineEnter")
      end
    end)
  end
end

function JiangHuZhuJiu:OnCloseGame()
  self.bGameOpen = false
  self.nLeftTime = 0
  Ui:GetWindow("UIHud"):UpdateJiangHuZhuJiuHudTime(self.nLeftTime)
  self:CheckRedPoint(me)
end

function JiangHuZhuJiu:UpdateHud()
  if Ui:WindowVisible("UIHud") then
    Ui:GetWindow("UIHud"):UpdateJiangHuZhuJiuHudTime(self.nLeftTime)
  end
  self.nLeftTime = self.nLeftTime - 1
  if self.nLeftTime < 0 then
    return false
  end
  return true
end

function JiangHuZhuJiu:CheckRedPoint(pPlayer)
  for i = 1, 4 do
    if pPlayer.GetUserValue(JiangHuZhuJiu.nSaveGroup, JiangHuZhuJiu.AWARDS[i].nSaveKey) - pPlayer.GetUserValue(JiangHuZhuJiu.nSaveGroup, JiangHuZhuJiu.AWARDS[i].nGotSaveKey) > 0 then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT[string.format("RedPoint_JiangHuZhuJiu_Award%s", i)])
    else
      RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[string.format("RedPoint_JiangHuZhuJiu_Award%s", i)])
    end
  end
end

function JiangHuZhuJiu:IsInGame()
  return (self.nLeftTime or 0) > 0
end
