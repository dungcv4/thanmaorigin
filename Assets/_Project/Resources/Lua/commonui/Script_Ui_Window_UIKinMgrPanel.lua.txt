local tbWnd = Ui:CreateWindow("UIKinMgrPanel")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_KIN_DATA,
      self.onKinSyncData
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  Template = "node/SelectList/ListPanel/ScrollView/Viewport/Content/template",
  btnConfirm = "node/imgBG/Content/btnConfirm",
  SelectList = "node/SelectList",
  btnMaster = "node/imgBG/Content/Master",
  btnViceMaster1 = "node/imgBG/Content/ViceMaster1",
  btnViceMaster2 = "node/imgBG/Content/ViceMaster2",
  btnCommander = "node/imgBG/Content/Commander",
  btnBeauty1 = "node/imgBG/Content/Beauty1",
  btnBeauty2 = "node/imgBG/Content/Beauty2"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnConfirm = function(self)
    self:OnComfirm()
  end,
  btnMaster = function(self)
    self:OnClickCareer(self.btnMaster, Kin.Def.Career_Master)
  end,
  btnViceMaster1 = function(self)
    self:OnClickCareer(self.btnViceMaster1, Kin.Def.Career_ViceMaster, 1)
  end,
  btnViceMaster2 = function(self)
    self:OnClickCareer(self.btnViceMaster2, Kin.Def.Career_ViceMaster, 2)
  end,
  btnCommander = function(self)
    self:OnClickCareer(self.btnCommander, Kin.Def.Career_Command)
  end,
  btnBeauty1 = function(self)
    self:OnClickCareer(self.btnBeauty1, Kin.Def.Career_Mascot, 1)
  end,
  btnBeauty2 = function(self)
    self:OnClickCareer(self.btnBeauty2, Kin.Def.Career_Mascot, 2)
  end
}
tbWnd.tbCareer2Txt = {
  [Kin.Def.Career_Master] = {
    [1] = "node/imgBG/Content/Master/txtName"
  },
  [Kin.Def.Career_ViceMaster] = {
    [1] = "node/imgBG/Content/ViceMaster1/txtName",
    [2] = "node/imgBG/Content/ViceMaster2/txtName"
  },
  [Kin.Def.Career_Command] = {
    [1] = "node/imgBG/Content/Commander/txtName"
  },
  [Kin.Def.Career_Mascot] = {
    [1] = "node/imgBG/Content/Beauty1/txtName",
    [2] = "node/imgBG/Content/Beauty2/txtName"
  }
}
tbWnd.tbCareer2EmptyTxt = {
  [Kin.Def.Career_ViceMaster] = {
    [1] = "node/imgBG/Content/ViceMaster1/txtEmpty",
    [2] = "node/imgBG/Content/ViceMaster2/txtEmpty"
  },
  [Kin.Def.Career_Command] = {
    [1] = "node/imgBG/Content/Commander/txtEmpty"
  },
  [Kin.Def.Career_Mascot] = {
    [1] = "node/imgBG/Content/Beauty1/txtEmpty",
    [2] = "node/imgBG/Content/Beauty2/txtEmpty"
  }
}
tbWnd.tbCareer2Btn = {
  [Kin.Def.Career_Master] = {
    [1] = "node/imgBG/Content/Master"
  },
  [Kin.Def.Career_ViceMaster] = {
    [1] = "node/imgBG/Content/ViceMaster1",
    [2] = "node/imgBG/Content/ViceMaster2"
  },
  [Kin.Def.Career_Command] = {
    [1] = "node/imgBG/Content/Commander"

  [Kin.Def.Career_Mascot] = {

    [2] = "node/imgBG/Content/Beauty2"

function tbWnd:OnCreate()
  self.tbSelectList = Ui:NewScrollView(self, tbControls.Template, function(self, nIndex, tbElement)
    self:OnSelect(nIndex)
  end)
  self:SetClickEmptyToClose(tbControls.SelectList)
end

function tbWnd:OnOpen()
  self:Init()
end

function tbWnd:Init()
  for nCareer, tbPath in pairs(self.tbCareer2Txt) do
    for _, szPath in ipairs(tbPath) do
      self.pPanel:Label_SetText(szPath, "")
      if self.tbCareer2EmptyTxt[nCareer] then
        for _, szEmptyPath in ipairs(self.tbCareer2EmptyTxt[nCareer]) do
          self.pPanel:SetActive(szEmptyPath, true)
        end
      end
    end
  end
  self.pPanel:SetActive(tbControls.SelectList, false)
  local tbMemberList = Kin:GetMemberList() or {}
  self.tbOldCareerInfo = {}
  local tbIndex = {}
  for i = 1, #tbMemberList do
    local nCareer = tbMemberList[i].nCareer
    if self.tbCareer2Txt[nCareer] then
      self.tbOldCareerInfo[nCareer] = self.tbOldCareerInfo[nCareer] or {}
      table.insert(self.tbOldCareerInfo[nCareer], {
        szName = tbMemberList[i].szName,
        nCareer = nCareer,
        nMemberId = tbMemberList[i].nMemberId
      })
      if tbIndex[nCareer] then
        self.pPanel:Label_SetText(self.tbCareer2Txt[nCareer][2], tbMemberList[i].szName)
        tbIndex[nCareer] = 2
      else
        self.pPanel:Label_SetText(self.tbCareer2Txt[nCareer][1], tbMemberList[i].szName)
        tbIndex[nCareer] = 1
      end
      if self.tbCareer2EmptyTxt[nCareer] and self.tbCareer2EmptyTxt[nCareer][tbIndex[nCareer]] then
        self.pPanel:SetActive(self.tbCareer2EmptyTxt[nCareer][tbIndex[nCareer]], false)
      end
    end
  end
  self.tbNewCareerInfo = Lib:CopyTB(self.tbOldCareerInfo)
  self.nSelectCareer = nil
  local nMyCareer = Kin:GetMyCareer()
  for nCareer, tbPath in pairs(self.tbCareer2Btn) do
    if Kin.Def.Career_Authority[nMyCareer][Kin.Def.Career2Authority[nCareer]] then
      for _, szPath in ipairs(tbPath) do
        self.pPanel:Button_SetEnable(szPath, true)
      end
    else
      for _, szPath in ipairs(tbPath) do
        self.pPanel:Button_SetEnable(szPath, false)
      end
    end
  end
end

function tbWnd:OnClickCareer(szPath, nCareer, nIndex)
  nIndex = nIndex or 1
  local tbPosition = self.pPanel:Object_GetPosition(szPath)
  self.pPanel:Object_SetPosition(tbControls.SelectList, Ui.Vector3(tbPosition.x, tbPosition.y, tbPosition.z))
  self.nSelectCareer = nCareer
  self.nSelectIndex = nIndex
  local tbCurrentSelectInfo = self.tbNewCareerInfo[nCareer] and self.tbNewCareerInfo[nCareer][nIndex] or nil
  self:OnUpdateSelectList(tbCurrentSelectInfo)
  self.pPanel:SetActive(tbControls.SelectList, true)
end

function tbWnd:OnUpdateSelectList(tbCurrentSelectInfo)
  local tbMemberList = Kin:GetMemberList() or {}
  self.tbSelectList:Clear()
  self.tbList = {}
  local CurrentChoosePlayerId = tbCurrentSelectInfo and tbCurrentSelectInfo.nMemberId or 0
  local nCounter = 1
  if self.nSelectCareer ~= Kin.Def.Career_Master then
    self.tbSelectList:SetElement(nCounter, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtName", i18n.Get(Kin.Cfg.Msg.CancelAppoint))
    end)
    nCounter = nCounter + 1
    table.insert(self.tbList, {})
  end
  for i = 1, #tbMemberList do
    if tbMemberList[i].nCareer ~= Kin.Def.Career_Master and tbMemberList[i].nMemberId ~= CurrentChoosePlayerId and tbMemberList[i].nCareer ~= Kin.Def.Career_Trainee and tbMemberList[i].nCareer ~= self.nSelectCareer then
      self.tbSelectList:SetElement(nCounter, function(tbElement)
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("txtName", tbMemberList[i].szName)
      end)
      nCounter = nCounter + 1
      table.insert(self.tbList, {
        szName = tbMemberList[i].szName,
        nCareer = tbMemberList[i].nCareer,
        nMemberId = tbMemberList[i].nMemberId,
        nJoinTime = tbMemberList[i].nJoinTime
      })
    end
  end
end

function tbWnd:OnSelect(nIndex)
  if self.tbList[nIndex].nMemberId then
    for nCareer, tbData in pairs(self.tbNewCareerInfo) do
      for nIdx, tbInfo in ipairs(tbData) do
        if self.tbList[nIndex].nMemberId == tbInfo.nMemberId then
          tbData[nIdx] = nil
          self.pPanel:Label_SetText(self.tbCareer2Txt[nCareer][nIdx], "")
          if self.tbCareer2EmptyTxt[nCareer] and self.tbCareer2EmptyTxt[nCareer][nIdx] then
            self.pPanel:SetActive(self.tbCareer2EmptyTxt[nCareer][nIdx], true)
          end
          break
        end
      end
    end
    self.tbNewCareerInfo[self.nSelectCareer] = self.tbNewCareerInfo[self.nSelectCareer] or {}
    self.tbNewCareerInfo[self.nSelectCareer][self.nSelectIndex] = {
      szName = self.tbList[nIndex].szName,
      nCareer = self.tbList[nIndex].nCareer,
      nMemberId = self.tbList[nIndex].nMemberId,
      nJoinTime = self.tbList[nIndex].nJoinTime
    }
    self.pPanel:Label_SetText(self.tbCareer2Txt[self.nSelectCareer][self.nSelectIndex], self.tbList[nIndex].szName)
    if self.tbCareer2EmptyTxt[self.nSelectCareer] and self.tbCareer2EmptyTxt[self.nSelectCareer][self.nSelectIndex] then
      self.pPanel:SetActive(self.tbCareer2EmptyTxt[self.nSelectCareer][self.nSelectIndex], false)
    end
    local nNow = GetTime()
    if nNow - self.tbList[nIndex].nJoinTime < Kin.Def.nCareerFactionTimeLimit then
      me.CenterMsg(i18n.Get(Kin.Cfg.Msg.JoinKinTimeTooShortMsg))
    end
  else
    if self.tbNewCareerInfo[self.nSelectCareer] then
      self.tbNewCareerInfo[self.nSelectCareer][self.nSelectIndex] = nil
    end
    self.pPanel:Label_SetText(self.tbCareer2Txt[self.nSelectCareer][self.nSelectIndex], "")
    if self.tbCareer2EmptyTxt[self.nSelectCareer][self.nSelectIndex] then
      self.pPanel:SetActive(self.tbCareer2EmptyTxt[self.nSelectCareer][self.nSelectIndex], true)
    end
  end
  self.pPanel:SetActive(tbControls.SelectList, false)
end

function tbWnd:onKinSyncData(szDataType)
  if "MemberList" == szDataType then
    self:Init()
  end
end

function tbWnd:OnComfirm()
  if self.tbNewCareerInfo[Kin.Def.Career_Master] and self.tbNewCareerInfo[Kin.Def.Career_Master][1] and self.tbOldCareerInfo[Kin.Def.Career_Master] and self.tbOldCareerInfo[Kin.Def.Career_Master][1] and self.tbNewCareerInfo[Kin.Def.Career_Master][1].nMemberId ~= self.tbOldCareerInfo[Kin.Def.Career_Master][1].nMemberId then
    Kin:RequestChangeMaster(self.tbNewCareerInfo[Kin.Def.Career_Master][1].nMemberId)
    self:Close()
    return
  end
  local tbCancalMember = {}
  local tbSetCareerMember = {}
  for nCareer, tbData in pairs(self.tbCareer2Txt) do
    for nIdx, _ in ipairs(tbData) do
      if self.tbNewCareerInfo[nCareer] and self.tbNewCareerInfo[nCareer][nIdx] and self.tbOldCareerInfo[nCareer] and self.tbOldCareerInfo[nCareer][nIdx] then
        if self.tbNewCareerInfo[nCareer][nIdx].nMemberId ~= self.tbOldCareerInfo[nCareer][nIdx].nMemberId then
          table.insert(tbCancalMember, self.tbOldCareerInfo[nCareer][nIdx].nMemberId)
          table.insert(tbSetCareerMember, {
            nMemberId = self.tbNewCareerInfo[nCareer][nIdx].nMemberId,
            nCareer = nCareer
          })
        end
      elseif self.tbNewCareerInfo[nCareer] and self.tbNewCareerInfo[nCareer][nIdx] and (not self.tbOldCareerInfo[nCareer] or not self.tbOldCareerInfo[nCareer][nIdx]) then
        table.insert(tbSetCareerMember, {
          nMemberId = self.tbNewCareerInfo[nCareer][nIdx].nMemberId,
          nCareer = nCareer
        })
      elseif self.tbOldCareerInfo[nCareer] and self.tbOldCareerInfo[nCareer][nIdx] and (not self.tbNewCareerInfo[nCareer] or not self.tbNewCareerInfo[nCareer][nIdx]) then
        table.insert(tbCancalMember, self.tbOldCareerInfo[nCareer][nIdx].nMemberId)
      end
    end
  end
  if #tbCancalMember > 0 or #tbSetCareerMember > 0 then
    Kin:ChangeCareers(tbCancalMember, tbSetCareerMember)
    self:Close()
  else
    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.NoCareerChange))
    return
  end
end
