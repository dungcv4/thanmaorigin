function DegreeCtrl.GetSalartChannelMaxTimes(nActiveId)
  local bExist = false
  for nIndex, tbSalary in pairs(Kin.Cfg.Salary_Channel) do
    if tbSalary.A_Id == nActiveId then
      bExist = true
      if Kin:CheckSalaryTimeFrameState(nIndex) then
        return tbSalary.MaxTimes
      end
    end
  end
  if not bExist then
    LogErr("DegreeCtrl.GetSalartChannelMaxTimes can not find the nActiveId:", nActiveId, "in Kin.Cfg.Salary_Channel")
  end
end

function DegreeCtrl:GetMaxDegree(szName, pPlayer)
  local nValue = 0
  if tbSetting[szName] then
    nValue = tbSetting[szName][5]
    for _, tbSalary in pairs(Kin.Cfg.Salary_Channel) do
      if tbSalary.A_Id == Kin.Cfg.Salary_Channel_ActiveID[szName] and szName == tbSalary.DegreeName and (tbSalary.szOpenTimeFrame == "" or 1 == GetTimeFrameState(tbSalary.szOpenTimeFrame)) and ("" == tbSalary.szCloseTimeFrame or 0 == GetTimeFrameState(tbSalary.szCloseTimeFrame)) then
        nValue = tbSalary.MaxTimes
        break
      end
    end
    local fnVipExtSetting = self.tbVipExtSetting[szName]
    if fnVipExtSetting then
      if not pPlayer then
        Log(debug.traceback(), szName)
        return
      end
      local nVipLevel = pPlayer.GetVipLevel()
      nValue = nValue + fnVipExtSetting(nVipLevel)
    end
  end
  return nValue
end

function DegreeCtrl:GetNextVipDegree(szName, pPlayer)
  local fnVipExtSetting = self.tbVipExtSetting[szName]
  if not fnVipExtSetting then
    return
  end
  local nCurVipLevel = pPlayer.GetVipLevel()
  local nCurDegree = fnVipExtSetting(nCurVipLevel)
  for i = nCurVipLevel + 1, Vip.nMaxLevel do
    local nDegree = fnVipExtSetting(i)
    if nCurDegree < nDegree then
      return i
    end
  end
end

function DegreeCtrl:GetDegreeSecond(nDataTime, tbUnCountTime)
  local nUnCountBegin, nUnCountEnd = 0, 0
  if tbUnCountTime then
    nUnCountBegin, nUnCountEnd = table.unpack(tbUnCountTime)
  end
  local nSecondPerDay = 86400 - nUnCountEnd + nUnCountBegin
  local nLocalDay = Lib:GetLocalDay(nDataTime)
  local nCurDaySecond = Lib:GetLocalDayTime(nDataTime) % 86400
  if nUnCountBegin < nCurDaySecond and nUnCountEnd > nCurDaySecond then
    nCurDaySecond = nUnCountBegin
  elseif nUnCountBegin < nCurDaySecond then
    nCurDaySecond = nCurDaySecond + nUnCountBegin - nUnCountEnd
  end
  return nLocalDay * nSecondPerDay + nCurDaySecond
end

function DegreeCtrl:GetTimeByDegreeSecond(nDegreeSecond, tbUnCountTime)
  local nUnCountBegin, nUnCountEnd = 0, 0
  if tbUnCountTime then
    nUnCountBegin, nUnCountEnd = table.unpack(tbUnCountTime)
  end
  local nSecondPerDay = 86400 - nUnCountEnd + nUnCountBegin
  local nCurDaySecond = nDegreeSecond % nSecondPerDay
  if nUnCountBegin <= nCurDaySecond then
    nCurDaySecond = nCurDaySecond + nUnCountEnd - nUnCountBegin
  end
  return math.floor(nDegreeSecond / nSecondPerDay) * 24 * 3600 + nCurDaySecond - Lib:GetGMTSec()
end

function DegreeCtrl:GetDegree(pPlayer, szName)
  if tbSetting[szName] then
    local tbInfo = tbSetting[szName]
    local nGroup, nId = tbInfo[1][1], tbInfo[1][2]
    local nTimeGroup, nTimeId = tbInfo[2][1], tbInfo[2][2]
    local nMaxDegree = self:GetMaxDegree(szName, pPlayer)
    local nLastTime = pPlayer.GetUserValue(nTimeGroup, nTimeId)
    if not nLastTime or 0 == nLastTime then
      return nMaxDegree
    end
    local fn = self.tbTimeFunction[tbInfo[3]]
    local nCurTime = GetTime()
    local nClock = tbInfo[4]
    if nClock then
      local nParseTodayTime = Lib:ParseTodayTime(nClock)
      nCurTime = nCurTime - nParseTodayTime
      nLastTime = nLastTime - nParseTodayTime
    end
    local nCurTimeBlock = fn(nCurTime)
    local nLastTimeBlock = fn(nLastTime)
    local nCostedDegree = pPlayer.GetUserValue(nGroup, nId)
    if nCurTimeBlock == nLastTimeBlock then
      return nMaxDegree - nCostedDegree
    else
      return nMaxDegree
    end
  end
end

function DegreeCtrl:SetDegree(pPlayer, szName, nDegree)
  if nDegree < 0 then
    LogErr("Degree can not less then 0. szName:", szName)
    return false
  end
  local tbInfo = tbSetting[szName]
  local nGroup, nId = table.unpack(tbInfo[1])
  local nTimeGroup, nTimeId = table.unpack(tbInfo[2])
  pPlayer.SetUserValue(nGroup, nId, self:GetMaxDegree(szName, pPlayer) - nDegree)
  pPlayer.SetUserValue(nTimeGroup, nTimeId, GetTime())
  Lib:LogFormat("DegreeCtrl:SetDegree pPlayer:%d, name:%s, key:%s, nDegree:%d, nCurTime:%d", pPlayer.dwID, pPlayer.szName, szName, nDegree, GetTime())
  return true
end

function DegreeCtrl:BuyCountCostPrice(pPlayer, szName, nDegree)
  local tbInfo = tbBuyCountSetting[szName]
  local szBuyDegree, szMoneyType, nMoney = table.unpack(tbInfo)
  if type(nMoney) == "number" then
    return szBuyDegree, szMoneyType, nMoney * nDegree, nMoney
  end
  local nHasBuy = self:GetMaxDegree(szBuyDegree, pPlayer) - self:GetDegree(pPlayer, szBuyDegree)
  local nToDegree = nHasBuy + nDegree
  local nTotalPrice = 0
  for i, v in ipairs(nMoney) do
    local nMaxCount, nPrice = table.unpack(v)
    if nHasBuy <= nMaxCount then
      local nBuy = math.min(nMaxCount, nToDegree) - nHasBuy
      nTotalPrice = nTotalPrice + nPrice * nBuy
      nHasBuy = nHasBuy + nBuy
      if nToDegree <= nHasBuy then
        break
      end
    end
  end
  return szBuyDegree, szMoneyType, nTotalPrice
end

function DegreeCtrl:BuyCountCostPriceEx(pPlayer, szName, nDegree)
  local tbInfo = tbBuyCountSetting[szName]
  local szBuyDegree, szMoneyType, nMoney = table.unpack(tbInfo)
  if type(nMoney) == "number" then
    return szBuyDegree, szMoneyType, nMoney, nMoney
  end
  local nHasBuy = self:GetMaxDegree(szBuyDegree, pPlayer) - self:GetDegree(pPlayer, szBuyDegree)
  local nToDegree = nHasBuy + 1
  local nTotalPrice = 0
  for i, v in ipairs(nMoney) do
    local nMaxCount, nPrice = table.unpack(v)
    if nToDegree >= nMaxCount then
      nTotalPrice = math.max(nTotalPrice, nPrice)
    end
  end
  return szBuyDegree, szMoneyType, nTotalPrice
end

function DegreeCtrl:NextCostPrice(pPlayer, szName)
  local tbInfo = tbBuyCountSetting[szName]
  local szBuyDegree, szMoneyType, nMoney = table.unpack(tbInfo)
  if type(nMoney) == "number" then
    return szBuyDegree, szMoneyType, nMoney
  end
  local nMax = self:GetMaxDegree(szBuyDegree, pPlayer)
  local nIndex = 0
  for i, v in ipairs(nMoney) do
    local nMaxCount, nPrice = table.unpack(v)
    if nMax == nMaxCount then
      nIndex = i
    end
  end
  if 0 == nMax then
    return nMoney[1][1] * nMoney[1][2]
  end
  if 0 == nIndex then
    Log("GetNextPriceError", pPlayer.dwID, szName, nMax)
    return 0
  end
  local nCurCount, nCurnPrice = table.unpack(nMoney[nIndex])
  if nMoney[nIndex + 1] then
    local nNextCount, nNextPrice = table.unpack(nMoney[nIndex + 1])
    return (nNextCount - nCurCount) * nNextPrice
  else
    if nMoney[nIndex - 1] then
      local nLastCount, nLastPrice = table.unpack(nMoney[nIndex - 1])
      return (nCurCount - nLastCount) * nCurnPrice
    end
    return nCurCount * nCurnPrice
  end
end
