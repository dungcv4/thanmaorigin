function Player:UseEquip(nItemId)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return false
  end
  if pItem.nUseLevel > me.nTotalLevel then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\232\163\133\229\164\135"))
    return false
  end
  if pItem.nEquipPos and Item:CheckUseEquipCD(me, pItem) then
    me.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\230\147\141\228\189\156\229\164\170\233\162\145\231\185\129\228\186\134\239\188\140\232\175\183\231\168\141\229\144\142\229\134\141\232\175\149\239\188\129"))
    return false
  end
  local nLimitCDEquipPos = Item:LimitOptEquipPosChange(pItem)
  if me.tbUseEquipCD and me.tbUseEquipCD[nLimitCDEquipPos] then
    local tbData = me.tbUseEquipCD[nLimitCDEquipPos]
    local nIndex = tbData.nLastIndex % Item.nUseEquipLimitTime + 1
    tbData.nLastIndex = tbData.nLastIndex + 1
    tbData[nIndex] = GetTime()
  end
  RemoteServer.UseEquip(nItemId)
  return true
end

function Player:OnPkExerciseState(nState, varValue, bWin)
  local AFKFightSetting = AFK:GetSetting():GetFightSetting()
  if nState == self.PK_EXERCISE_READY then
    Ui:OpenWindow("UIReadyGo", Player.nExerciseWaitTime - 1)
  elseif nState == self.PK_EXERCISE_GO then
    me.GetNpc().nExerciseNpcId = varValue
  elseif nState == self.PK_EXERCISE_END then
    if varValue then
      me.CenterMsg(i18n.Parse(varValue))
    end
    local nTargetNpcId = me.GetNpc().nExerciseNpcId
    if nTargetNpcId and nTargetNpcId > 0 then
      local pTargetNpc = KNpc.GetById(nTargetNpcId)
      if pTargetNpc then
        pTargetNpc.nExerciseNpcId = 0
      end
      me.GetNpc().nExerciseNpcId = 0
    end
    AFKFightSetting:ResetFightPos()
  end
end

Player.bLevelUpCache = false

function Player:FlyChar(nNewLevel)
  if Map:GetMapType(me.nMapTemplateId) == Map.emMap_Fuben and Tong.Def.nMainMapTemplateId ~= me.nMapTemplateId then
    if nNewLevel then
      self.bLevelUpCache = true
    end
    return
  end
  if not self.bLevelUpCache and not nNewLevel then
    return
  end
  self.bLevelUpCache = false
end

function Player:ForbidJumpIcon()
  self.bForbinJump = true
end

function Player:IsForbinJumpIcon()
  local bForbinJump = self.bForbinJump
  local bChuanGong = me.bChuanGong
  if nil == bForbinJump then
    bForbinJump = false
  end
  if nil == bChuanGong then
    bChuanGong = false
  end
  if bChuanGong or bForbinJump then
    return true
  end
  return false
end
