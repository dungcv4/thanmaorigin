function ShiYueDao:ClientShowFubenUI(nFubenType, szFunc, Param)
  local tbCfg = tbFubenUI[nFubenType][szFunc]
  if not tbCfg then
    LogErr("ShowFubenUI() tbCfg nil ")
    return
  end
  local tbTimerParam = {}
  local tbContent = {}
  for i = 1, #tbCfg do
    Param[i] = Param[i] or {}
    if tbCfg[i].type == ParamType.Timer then
      local tbTempTimer = {
        table.unpack(Param[i]),
        i18n.Get(tbCfg[i].szTitle)
      }
      table.insert(tbTimerParam, tbTempTimer)
    elseif tbCfg[i].type == ParamType.String then
      local szContent = i18n.Get(tbCfg[i].szTitle, table.unpack(Param[i]))
      table.insert(tbContent, szContent)
    end
  end
  FubenUIController:ShowDefaultInfo(tbTimerParam, tbContent)
end

end

function ShiYueDao:GetFubenBtnAImagePathAndText()

end

function ShiYueDao:IsNeedFubenBtn()
  return self:IsCanPath(), {
    self.AutoPathToEnter,
    self
  }
end

function ShiYueDao:SetBtnPathShow(bShowPathBtn)
  self.bShowPathBtn = bShowPathBtn
  if Ui:WindowVisible("UIHudRightCenter") then
    Ui:GetWindow("UIHudRightCenter"):SetFubenBtnImageAndText(self:GetFubenBtnAImagePathAndText())
    Ui:GetWindow("UIHudRightCenter"):SetFubenBtnShowStatus(self:IsCanPath())
  end
end

function ShiYueDao:IsCanPath()
  return self.bShowPathBtn or false
end

function ShiYueDao:CheckCanJoin()
  if DegreeCtrl:GetDegree(me, "ShiYueDao") <= 0 then
    return false, i18n.EasyGet("\230\130\168\231\154\132\232\154\128\230\156\136\229\178\155\229\143\130\228\184\142\230\172\161\230\149\176\229\183\178\230\187\161")
  end
  return true, ""
end

function ShiYueDao:GoToShiYueDaoNPC()
  local bCanJoin, szMsg = self:CheckCanJoin()
  if not bCanJoin then
    me.CenterMsg(szMsg)
    return
  end
  local nRandomNPCIndex = -1
  for nIndex, nMapID in ipairs(ShiYueDao.LEADER_NPC_MAPID) do
    if me.nMapId == nMapID then
      nRandomNPCIndex = nIndex
      break
    end
  end
  if -1 == nRandomNPCIndex then
    nRandomNPCIndex = math.random(1, 3)
  end
  local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(ShiYueDao.Def.LEADER_NPCID, ShiYueDao.LEADER_NPC_MAPID[nRandomNPCIndex], false)
  
  local function fnCallback()
    local function __timer_Callback()
      local nNpcId = KNpc.GetNpcIdByTemplateId(ShiYueDao.Def.LEADER_NPCID)
      
      if nNpcId then
        Operation:SimpleTap(nNpcId)
      end
    end
    
    Timer:Register(5, __timer_Callback)
  end
  
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  AFK:Stop()
  AutoPath:GotoAndCallEx(ShiYueDao.LEADER_NPC_MAPID[nRandomNPCIndex], nPosX, nPosY, fnCallback, nWalkNearLength, nil, ShiYueDao.Def.LEADER_NPCID)
end

function ShiYueDao:RegisterFubenUIChecker()
  local tbTemplateID = {}
  for i = ShiYueDao.FLOOR.PREPARE, ShiYueDao.FLOOR.HOLE do
    table.insert(tbTemplateID, ShiYueDao.FUBEN_CFG[i].TEMPLATEMAPID)
  end
  FubenUIController:RegisterChecker(tbTemplateID, self)
end

function ShiYueDao:SyncDynMapPoint(nTemplateID, tbPointList)
  self.curTemplateID = nTemplateID
  self.tbEnterPointList = tbPointList
end

function ShiYueDao:AutoPathToEnter()
  if not self.curTemplateID or not self.tbEnterPointList then
    return
  end
  local nTargetX = 0
  local nTargetY = 0
  local nMinDis = -1
  local nMapId, myPosX, myPosY = me.GetWorldPos()
  for _, tbPoint in pairs(self.tbEnterPointList) do
    local nPosX = tbPoint.nPosX
    local nPosY = tbPoint.nPosY
    local nCurDis = Lib:GetDistance(myPosX, myPosY, nPosX, nPosY)
    if -1 == nMinDis or nMinDis > nCurDis then
      nMinDis = nCurDis
      nTargetX = nPosX
      nTargetY = nPosY
    end
  end
  AutoPath:GotoAndCallEx(me.nMapId, nTargetX, nTargetY, nil, 0, nil, ShiYueDao.FUBEN_CFG[ShiYueDao.FLOOR.ISLAND].JINGYINGENTERNPCID)
end

ShiYueDao:RegisterFubenUIChecker()
