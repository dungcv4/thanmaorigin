function Sdk:GetPayUid(nPlayerId)
  local nOrgServerId = Sdk:GetOrgServerIdByPlayerId(nPlayerId)
  Log("Sdk:GetPayUid", nOrgServerId, nPlayerId)
  return string.format("%d_%d", nOrgServerId, nPlayerId)
end

function Sdk:GetOrgServerIdByPlayerId(nPlayerId)
  local nServerId = Sdk:GetServerId()
  local nMainServerId = math.floor(nServerId / 10000)
  local nOrgSubServerId = Lib:LoadBits(nPlayerId, 20, 31)
  local nOrgServerId = nMainServerId * 10000 + nOrgSubServerId
  return nOrgServerId
end

function Sdk:GetOrgServerIdByKinId(nKinId)

end
