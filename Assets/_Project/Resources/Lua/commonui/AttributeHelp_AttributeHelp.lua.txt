AttributeHelp.tbLogPlayerAttrIdList = {}
AttributeHelp.bServerOpenLog = AttributeHelp.bServerOpenLog or false
AttributeHelp.tbPlayerId2AttrInfo = AttributeHelp.tbPlayerId2AttrInfo or {}
AttributeHelp.tbLogAttrTypeList = AttributeHelp.tbLogAttrTypeList or {
  deadlystrike_damage_v = true,
  weaken_deadlystrike_damage_v = true,
  ignore_defense_v = true,
  enhance_final_damage_p = true,
  reduce_final_damage_p = true,
  resist_enhance_final_damage_p = true,
  resist_reduce_final_damage_p = true,
  meleedamage_return_v = true,
  rangedamage_return_v = true,
  steallife_v = true,
  stealmana_v = true
}

function AttributeHelp:LogAttrApply(nPlayerId, szModule, szAttribType, nValue1, nValue2, nValue3)
  self:LogAttrChange(nPlayerId, szModule, szAttribType, nValue1)
end

function AttributeHelp:LogAttrRemove(nPlayerId, szModule, szAttribType, nValue1, nValue2, nValue3)
  self:LogAttrChange(nPlayerId, szModule, szAttribType, -nValue1)
end

function AttributeHelp:OnApplyEquipAttrib(nPlayerId, szAttribType, nValue1)
  self:LogAttrChange(nPlayerId, "Equip", szAttribType, nValue1)
end

function AttributeHelp:OnRemoveEquipAttrib(nPlayerId, szAttribType, nValue1)
  self:LogAttrChange(nPlayerId, "Equip", szAttribType, -nValue1)
end

function AttributeHelp:LogAttrChange(nPlayerId, szModule, szAttribType, nValue1)
  if not nPlayerId or not self.tbLogAttrTypeList[szAttribType] then
    return
  end
  if not self.bServerOpenLog and not self.tbLogPlayerAttrIdList[nPlayerId] then
    return
  end
  self:InitAttrTB(nPlayerId, szModule, szAttribType)
  self.tbPlayerId2AttrInfo[nPlayerId][szModule][szAttribType] = self.tbPlayerId2AttrInfo[nPlayerId][szModule][szAttribType] + nValue1
  self.tbPlayerId2AttrInfo[nPlayerId].allModule[szAttribType] = self.tbPlayerId2AttrInfo[nPlayerId].allModule[szAttribType] + nValue1
  Log("AttributeHelp:LogAttrChange nPlayerId, szModule, szAttribType:", nPlayerId, szModule, szAttribType, string.format(", CurValue:%d, ChangeValue:%d", self.tbPlayerId2AttrInfo[nPlayerId].allModule[szAttribType], nValue1))
end

function AttributeHelp:InitAttrTB(nPlayerId, szModule, szAttribType)
  if not self.tbPlayerId2AttrInfo[nPlayerId] then
    self.tbPlayerId2AttrInfo[nPlayerId] = {}
  end
  if not self.tbPlayerId2AttrInfo[nPlayerId][szModule] then
    self.tbPlayerId2AttrInfo[nPlayerId][szModule] = {}
  end
  if not self.tbPlayerId2AttrInfo[nPlayerId][szModule][szAttribType] then
    self.tbPlayerId2AttrInfo[nPlayerId][szModule][szAttribType] = 0
  end
  if not self.tbPlayerId2AttrInfo[nPlayerId].allModule then
    self.tbPlayerId2AttrInfo[nPlayerId].allModule = {}
  end
  if not self.tbPlayerId2AttrInfo[nPlayerId].allModule[szAttribType] then
    self.tbPlayerId2AttrInfo[nPlayerId].allModule[szAttribType] = 0
  end
end

function AttributeHelp:ShowPlayerAttr(pPlayer)
  if not pPlayer or not self.tbPlayerId2AttrInfo[pPlayer.dwID] then
    return
  end
  Log("AttributeHelp ", string.format("%d %s Show AtttValue Begin", pPlayer.dwID, pPlayer.szName))
  for szModule, tbV in pairs(self.tbPlayerId2AttrInfo[pPlayer.dwID]) do
    Log("AttributeHelp ", string.format("%d %s Show %s AtttValue Begin", pPlayer.dwID, pPlayer.szName, szModule))
    Lib:ShowTB(tbV, string.format("AttributeHelp %d %s", pPlayer.dwID, szModule))
    Log("AttributeHelp ", string.format("%d %s Show %s AtttValue End", pPlayer.dwID, pPlayer.szName, szModule))
  end
  Log("AttributeHelp ", string.format("%d %s Show AtttValue End", pPlayer.dwID, pPlayer.szName))
end

function AttributeHelp:OnLogin(pPlayer)
  if not self.tbPlayerId2AttrInfo[pPlayer.dwID] then
    return
  end
  self:ShowPlayerAttr(pPlayer)
end

function AttributeHelp:OnLogout(pPlayer)
  self:ShowPlayerAttr(pPlayer)
  self.tbPlayerId2AttrInfo[pPlayer.dwID] = nil
end

function AttributeHelp:OnLogoutFromZone(pPlayer)
  self:ShowPlayerAttr(pPlayer)
end
