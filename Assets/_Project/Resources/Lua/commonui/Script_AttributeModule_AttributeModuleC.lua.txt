Require("CommonScript/AttributeModule/AttributeModule.lua")

function AttributeModule:OnSyncEquip()
  if not me.IsMaskPlayerAttr() then
    self:OnEquip(me)
  end
end

function AttributeModule:OnEnterZone()
  me.nMaskPlayerAttrState = 0
  me.tbAttributeModule = {}
end

function AttributeModule:OnLogin()
  if not me.IsMaskPlayerAttr() then
    me.tbAttributeModule = {}
  end
end

function AttributeModule:ApplyPlayerAllAttr()
  local pHorse = me.GetEquipByPos(Item.EQUIPPOS_HORSE)
  if pHorse then
    Horse:ApplyHorseAttr(me, pHorse)
  end
  AttributeModule:OnEquip(me)
  me.nMaskPlayerAttrState = 0
end

function AttributeModule:RemovePlayerAllAttr()
  local pHorse = me.GetEquipByPos(Item.EQUIPPOS_HORSE)
  if pHorse then
    Horse:RemoveHorseAttr(me, pHorse)
  end
  AttributeModule:OnUnEquip(me)
  me.nMaskPlayerAttrState = 1
end

function AttributeModule:S2CMaskPlayerAttrState(nMaskPlayerAttrState)
  local nOldMaskState = me.nMaskPlayerAttrState
  me.nMaskPlayerAttrState = nMaskPlayerAttrState
  if 1 == nMaskPlayerAttrState then
    if 1 ~= nOldMaskState then
      local pHorseItem = me.GetEquipByPos(Item.EQUIPPOS_HORSE)
      if pHorseItem then
        Horse:RemoveHorseAttr(me, pHorseItem)
      end
    end
    if PlayerEvent.bLogin and 1 ~= nOldMaskState then
      Lib:CallBack({
        AttributeModule.OnUnEquip,
        AttributeModule,
        me
      })
    end
  end
end

function AttributeModule:S2CMaskPlayerFactSkillState(nMaskFactSkillState)
  me.nMaskFactSkillState = nMaskFactSkillState
end

function AttributeModule.ApplyPlayerFactSkill()
  me.nMaskFactSkillState = 0
end

function AttributeModule.RemovePlayerFactSkill()
  me.nMaskFactSkillState = 1
end

function AttributeModule.ApplyPlayerPotential()
  me.nMaskPotentialState = 0
end

function AttributeModule.RemovePlayerPotential()
  me.nMaskPotentialState = 1
end
