local tbWnd = Ui:CreateClass("UIwndQiHunInfo", "UIPanel")
tbWnd.tbControls = {
  CurTemplate = "imgBg/CurrentAttr/AttrList/Template",
  txtCurrentLvValue = "imgBg/CurrentAttr/CurrentLv/txtCurrentLvValue",
  txtCurrentLv = "imgBg/CurrentAttr/CurrentLv/txtCurrentLv",
  NextTemplate = "imgBg/NextLvAttr/AttrList/Template",
  txtNextLvValue = "imgBg/NextLvAttr/NextLv/txtNextLvValue",
  txtNextLv = "imgBg/NextLvAttr/NextLv/txtNextLv",
  Empty4Raycast = "Empty4Raycast",
  txtTitle = "imgBg/txtTitle"
}
tbWnd.tbOnClick = {
  Empty4Raycast = function(self)
    self.pPanel:SetActive("Main", false)
  end
}

function tbWnd:OnCreate()
  self.tbCurAttrScrollView = Ui:NewScrollView(self, self.CurTemplate, nil, false, true)
  self.tbNextAttrScrollView = Ui:NewScrollView(self, self.NextTemplate, nil, false, true)
end

function tbWnd:UpdateView(nCurGradeScoreLevel)
  local tbCurCfg = Item.QiHun.tbGradeScoreInfo[nCurGradeScoreLevel]
  local tbNextCfg = Item.QiHun.tbGradeScoreInfo[nCurGradeScoreLevel + 1]
  if tbCurCfg then
    self.pPanel:SetActive(self.txtCurrentLvValue, true)
    local nCurNeedLevel = tbCurCfg[0][1]
    self.pPanel:Label_SetText(self.txtCurrentLvValue, nCurNeedLevel)
  else
    self.pPanel:SetActive(self.txtCurrentLvValue, false)
  end
  if tbNextCfg then
    self.pPanel:SetActive(self.txtNextLvValue, true)
    local nNextNeedNum = tbNextCfg[0][1]
    self.pPanel:Label_SetText(self.txtNextLvValue, nNextNeedNum)
  else
    self.pPanel:SetActive(self.txtNextLvValue, false)
  end
  self:UpdateGradeScoreSV(tbCurCfg, tbNextCfg)
end

function tbWnd:UpdateGradeScoreSV(tbCurCfg, tbNextCfg)
  self.tbCurAttrScrollView:Reset()
  if tbCurCfg then
    for i = 1, #tbCurCfg do
      self.tbCurAttrScrollView:SetElement(i, function(tbElement)
        local szDesc, szValue = FightSkill:GetMagicDescSplit(tbCurCfg[i][1], {
          table.unpack(tbCurCfg[i], 2)
        })
        tbElement.pPanel:Label_SetText("txtAttr", szDesc)
        tbElement.pPanel:Label_SetText("txtAttrValue", szValue)
      end)
    end
  else
    self.tbCurAttrScrollView:SetElement(1, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttr", i18n.EasyGet("\230\156\170\230\191\128\230\180\187"))
      tbElement.pPanel:Label_SetText("txtAttrValue", "")
    end)
  end
  if tbNextCfg then
    self.tbNextAttrScrollView:Reset()
    for i = 1, #tbNextCfg do
      self.tbNextAttrScrollView:SetElement(i, function(tbElement)
        local szDesc, szValue = FightSkill:GetMagicDescSplit(tbNextCfg[i][1], {
          table.unpack(tbNextCfg[i], 2)
        })
        tbElement.pPanel:Label_SetText("txtAttr", szDesc)
        tbElement.pPanel:Label_SetText("txtAttrValue", szValue)
      end)
    end
  else
    self.tbNextAttrScrollView:SetElement(1, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttr", i18n.EasyGet("\229\183\178\230\187\161\231\186\167"))
      tbElement.pPanel:Label_SetText("txtAttrValue", "")
    end)
  end
end
