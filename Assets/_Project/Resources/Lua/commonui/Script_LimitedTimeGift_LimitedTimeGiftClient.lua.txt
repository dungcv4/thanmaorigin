function LimitedTimeGift:OnConnectLost()
  self.tbGiftList = nil
end

function LimitedTimeGift:OnSyncGiftList(tbGiftList, bLogin)
  self.tbGiftList = tbGiftList
  EventNotify.OnNotify(EventNotify.emNOTIFY_LIMITEDTIMEGIFT_SYNC)
  if not bLogin and self:GetAllGiftInfo() and not Ui:WindowVisible("UITimeLimitAwards") then
    Ui:OpenWindow("UITimeLimitAwards")
  end
end

function LimitedTimeGift:GetCurrentGiftInfo()
  if not self.tbGiftList or not next(self.tbGiftList) then
    return
  end
  local tbKey = {}
  for key, _ in pairs(self.tbGiftList) do
    table.insert(tbKey, key)
  end
  table.sort(tbKey, Lib.Comp)
  local tbCurrentGiftInfo
  local nCurTime = GetTime()
  for _, nKey in ipairs(tbKey) do
    local tbInfo = self.tbGiftList[nKey]
    local tbGiftSetting = self:GetGiftSettingByGiftId(tbInfo.nGiftType, tbInfo.nGiftId)
    if tbInfo and tbGiftSetting and tbInfo.nEndTime then
      if nCurTime > tbInfo.nEndTime or tbInfo.nBuyCount >= tbGiftSetting.nMaxBuyCount then
        self.tbGiftList[nKey] = nil
        local tbInfoNext = self.tbGiftList[nKey + 1]
        if tbInfoNext then
          if nCurTime > tbInfo.nEndTime then
            tbInfoNext.nEndTime = tbInfo.nEndTime + tbGiftSetting.nGiftLifeTime
          elseif tbInfo.nBuyCount >= tbGiftSetting.nMaxBuyCount then
            tbInfoNext.nEndTime = nCurTime + tbGiftSetting.nGiftLifeTime
          else
            tbInfoNext.nEndTime = nCurTime + tbGiftSetting.nGiftLifeTime
          end
        end
      else
        tbCurrentGiftInfo = tbInfo
        break
      end
    end
  end
  return tbCurrentGiftInfo
end

function LimitedTimeGift:GetAllGiftInfo()
  if not self.tbGiftList or not next(self.tbGiftList) then
    return
  end
  local tbAllGiftInfo = {}
  local nCurTime = GetTime()
  for nKey, tbInfo in pairs(self.tbGiftList) do
    local tbGiftSetting = self:GetGiftSettingByGiftId(tbInfo.nGiftType, tbInfo.nGiftId)
    if tbInfo and tbGiftSetting and tbInfo.nEndTime then
      if nCurTime > tbInfo.nEndTime or tbInfo.nBuyCount >= tbGiftSetting.nMaxBuyCount then
        self.tbGiftList[nKey] = nil
      else
        tbAllGiftInfo[tbInfo.nGiftType] = tbAllGiftInfo[tbInfo.nGiftType] or {}
        table.insert(tbAllGiftInfo[tbInfo.nGiftType], tbInfo)
      end
    end
  end
  return tbAllGiftInfo
end

function LimitedTimeGift:TimeDesc(nSec)
  if nSec < 3600 then
    return string.format("%d:%02d", math.floor(nSec % 3600 / 60), nSec % 60)
  else
    return string.format("%d:%02d:%02d", math.floor(nSec / 3600), math.floor(nSec % 3600 / 60), nSec % 60)
  end
end
