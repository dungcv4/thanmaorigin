local tbWnd = Ui:CreateWindow("UIKinBuild")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_KIN_DATA,
      self.onKinSyncData
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/txtTitle",
  btnList = "node/imgBG/btnList",
  txtBuildingTitle = "node/imgBG/Content/Palace/PalaceTitle/Title/txtTitle",
  txtBuildingName = "node/imgBG/Content/Palace/PalaceTitle/txtName",
  btnPalace = "node/imgBG/btnList/btnPalace",
  btnTreasure = "node/imgBG/btnList/btnTreasure",
  btnMarket = "node/imgBG/btnList/btnMarket",
  btnSkill = "node/imgBG/btnList/btnSkill",
  txtNb = "node/imgBG/Content/Palace/PalaceTitle/Title/imgTitleBg/txtNb",
  txtLevel = "node/imgBG/Content/Palace/TipsList/Viewport/Content/txtLevelDesc/txtLevel",
  txtLevelDesc = "node/imgBG/Content/Palace/TipsList/Viewport/Content/txtLevelDesc",
  txtTotalFund = "node/imgBG/Content/Palace/txtTips/txtTips1/txtTipsfund",
  txtCostFund = "node/imgBG/Content/Palace/txtTips/txtTips2/txtTipsfund",
  txtTips2 = "node/imgBG/Content/Palace/txtTips/txtTips2",
  txtTips3 = "node/imgBG/Content/Palace/txtTips/txtTips3",
  txtTips4 = "node/imgBG/Content/Palace/txtTips/txtTips4",
  btnUpgrade = "node/imgBG/Content/Palace/btnUpgrade"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnUpgrade = function(self)
    self:UpgradeBuilding()
  end
}

function tbWnd:OnCreate()
  local funcList = {
    [Kin.Def.Building_Main] = "OnClickSelectPalaceBuilding",
    [Kin.Def.Building_Jewelry] = "OnClickSelectTreasureBuilding",
    [Kin.Def.Building_Shop] = "OnClickSelectMarketBuilding",
    [Kin.Def.Building_Skill] = "OnClickSelectSkillBuilding"
  }
  self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\229\174\182\230\151\143\229\141\135\231\186\167"))
  self.tbPanelList = Ui:NewButtonGroup(self, {
    [Kin.Def.Building_Main] = self.btnPalace,
    [Kin.Def.Building_Jewelry] = self.btnTreasure,
    [Kin.Def.Building_Shop] = self.btnMarket,
    [Kin.Def.Building_Skill] = self.btnSkill
  }, true, function(self, nBuildingType)
    self.nBuildingType = nBuildingType
    self[funcList[nBuildingType]](self, nBuildingType)
  end)
  self.tbTipsScrollView = Ui:NewScrollView(self, self.txtLevelDesc)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnPalace, {
    RedPointMgr.EVENT.KinBuilding
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnUpgrade, {
    RedPointMgr.EVENT.KinBuilding
  })
end

function tbWnd:OnOpen()
  self.tbPanelList:Select(Kin.Def.Building_Main)
end

function tbWnd:OnClickSelectPalaceBuilding(nBuildingType)
  self:UpdateBuildingInfo(nBuildingType)
end

function tbWnd:OnClickSelectTreasureBuilding(nBuildingType)
  self:UpdateBuildingInfo(nBuildingType)
end

function tbWnd:OnClickSelectMarketBuilding(nBuildingType)
  self:UpdateBuildingInfo(nBuildingType)
end

function tbWnd:OnClickSelectSkillBuilding(nBuildingType)
  self:UpdateBuildingInfo(nBuildingType)
end

function tbWnd:UpdateMainSpecialInfo(nBuildingType)
  self.pPanel:SetActive(self.txtBuildingName, true)
  local nMaxBuildingLevel = Kin:GetCurMaxBuildingLevel(nBuildingType)
  self.pPanel:Label_SetText(self.txtBuildingName, i18n.EasyGet("\229\189\147\229\137\141\229\187\186\231\173\145\231\173\137\231\186\167\228\184\138\233\153\144{0}\231\186\167", nMaxBuildingLevel))
  self.pPanel:SetActive(self.txtTips3, false)
  local nBuildingLevel = Kin:GetBuildingLevel(nBuildingType)
  if nMaxBuildingLevel <= nBuildingLevel then
    self.pPanel:SetActive(self.txtTips2, false)
    return
  end
  self.pPanel:SetActive(self.txtTips2, true)
  self.pPanel:Label_SetText(self.txtCostFund, Kin:GetBuildingNextLevelCost(nBuildingType))
end

function tbWnd:UpdateCommonSpecialInfo(nBuildingType)
  self.pPanel:SetActive(self.txtBuildingName, false)
  local nBuildingLevel = Kin:GetBuildingLevel(nBuildingType)
  local nFinalMaxBuildingLevel = Kin:GetBuildingMaxLevel(nBuildingType)
  if nBuildingLevel < nFinalMaxBuildingLevel then
    self.pPanel:SetActive(self.txtTips3, true)
    local tbInfo = Kin:GetBuildingInfo(nBuildingType, nBuildingLevel + 1)
    local nMainBuildingLevel = Kin:GetBuildingLevel(Kin.Def.Building_Main)
    self.pPanel:Label_SetText(self.txtTips3, i18n.EasyGet("\233\156\128\228\184\187\230\174\191\232\190\190\229\136\176{0}\231\186\167", tbInfo.nMainBuildingRequire))
    self.pPanel:Label_SetColorByName(self.txtTips3, nMainBuildingLevel >= tbInfo.nMainBuildingRequire and "Enough" or "NotEnough")
    self.pPanel:SetActive(self.txtTips2, true)
    self.pPanel:Label_SetText(self.txtCostFund, Kin:GetBuildingNextLevelCost(nBuildingType, true))
  else
    self.pPanel:SetActive(self.txtTips3, false)
    self.pPanel:SetActive(self.txtTips2, false)
  end
end

function tbWnd:UpdateBuildingInfo(nBuildingType)
  local nBuildingLevel = Kin:GetBuildingLevel(nBuildingType)
  self.pPanel:Label_SetText(self.txtNb, nBuildingLevel)
  local szBuildingName = Kin.Def.BuildingName[nBuildingType]()
  self.pPanel:Label_SetText(self.txtBuildingTitle, szBuildingName)
  if nBuildingType == Kin.Def.Building_Main then
    self:UpdateMainSpecialInfo(nBuildingType)
  else
    self:UpdateCommonSpecialInfo(nBuildingType)
  end
  local tbInfo = Kin:GetBuildingInfo(nBuildingType, nBuildingLevel)
  local tbDesc = Lib:SplitStr(i18n.Parse(tbInfo.szDesc), "\n")
  self.tbTipsScrollView:Clear()
  for nIndex, szDesc in ipairs(tbDesc) do
    self.tbTipsScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("txtLevel", szDesc)
    end)
  end
  local tbBaseInfo = Kin:GetBaseInfo() or {}
  self.pPanel:Label_SetText(self.txtTotalFund, tbBaseInfo.nFund)
  local nCurMaxMainBuildingLevel = Kin:GetCurMaxBuildingLevel(Kin.Def.Building_Main)
  if nBuildingLevel < nCurMaxMainBuildingLevel then
    local nDiscount = Kin:GetBuildingUpGradeDiscount(nBuildingType, nCurMaxMainBuildingLevel - nBuildingLevel)
    if 1 == nDiscount then
      self.pPanel:SetActive(self.txtTips4, false)
    else
      self.pPanel:SetActive(self.txtTips4, true)
      local szMsg = i18n.Get(Kin.Cfg.Msg.KinBuildingUpGradeDiscountMsg, szBuildingName, nDiscount * 100)
      self.pPanel:Label_SetText(self.txtTips4, szMsg)
    end
  else
    self.pPanel:SetActive(self.txtTips4, false)
  end
  self.pPanel:SetActive(self.btnUpgrade .. "/RedPoint", false)
  if nBuildingType == Kin.Def.Building_Main and RedPointMgr:CheckEvent(RedPointMgr.EVENT.KinBuilding) then
    self.pPanel:SetActive(self.btnUpgrade .. "/RedPoint", true)
  end
end

function tbWnd:UpgradeBuilding()
  Kin:BuildingUpgrade(self.nBuildingType)
end

function tbWnd:onKinSyncData(szDataType)
  if "Building" ~= szDataType then
    return
  end
  self:UpdateBuildingInfo(self.nBuildingType)
end
