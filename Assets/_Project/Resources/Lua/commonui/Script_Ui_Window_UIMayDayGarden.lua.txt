local tbWnd = Ui:CreateClass("UIMayDayGarden", "UIPanel")
local tbControls = {
  RightStepTemplate = "Right/List/Viewport/Content/Template",
  btnReward = "Right/btnGift",
  btnHelp = "Right/txtTitle/btnHelp",
  SingleHandInAwardTemplate = "Left/List01/List/Viewport/Content/Template",
  MultiHandInAwardTemplate = "Left/List02/List/Viewport/Content/Template",
  txtTime01 = "Right/txtTime01",
  txtLevel = "Right/txtLV",
  txtSecondTittleUp = "Left/List01/txtSecondTittle",
  txtSecondTittleDown = "Left/List02/txtSecondTittle"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnReward = function(self)
    Ui:OpenWindow("UIReward_Common", self.nAchieveActId)
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.tbRightStepListView = Ui:NewScrollView(self, tbControls.RightStepTemplate)
  self.tbSingleHandInAwardView = Ui:NewItemGridView(self, tbControls.SingleHandInAwardTemplate)
  self.tbMultiHandInAwardView = Ui:NewItemGridView(self, tbControls.MultiHandInAwardTemplate)
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.nIndex = nIndex
  self.nCommonEntryId = tbInfo.nId
  self.szHelpKey = tbInfo.szHelpKey
  local tbSetting = ActivityMgr.CommonEntryLinkData[self.nCommonEntryId].tbSetting
  self.nAchieveActId = tbSetting.nAchieveActId
  self.pPanel:Label_SetText(tbControls.txtSecondTittleUp, i18n.Parse(i18n.Encode(tbSetting.tbszInfo[1])))
  self.pPanel:Label_SetText(tbControls.txtSecondTittleDown, i18n.Parse(i18n.Encode(tbSetting.tbszInfo[2])))
  self:UpdateListView()
  self:UpdateAwardPreview()
  self:UpdateRightInfo()
end

function tbWnd:UpdateRightInfo()
  if not self.nCommonEntryId then
    return
  end
  local pActivityIns = ActivityMgr:GetInstanceById(self.nCommonEntryId)
  if not pActivityIns then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    pActivityIns = ActivityMgr.EntranceSchedule:BuildInstance(self.nCommonEntryId, ActivityMgr.Def.ClassNameList.CommonEntry)
  end
  local pAchieveInstance = ActivityMgr:GetInstanceById(self.nAchieveActId)
  pAchieveInstance = pAchieveInstance or ActivityMgr.EntranceSchedule:BuildInstance(self.nAchieveActId, ActivityMgr.Def.ClassNameList.AchieveAward)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnReward, {
    RedPointMgr.EVENT[pAchieveInstance.tbConfig.szRedPointEvent]
  })
  local nRestTime = pActivityIns.nEndTime - GetTime()
  self.pPanel:Label_SetText(tbControls.txtTime01, i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", i18n.Parse(pActivityIns.tbConfig.szShowTime), Lib:TimeDesc2(nRestTime)))
  self.pPanel:Label_SetText(tbControls.txtLevel, i18n.EasyGet("\231\173\137\231\186\167\233\153\144\229\136\182\239\188\154{0}", pActivityIns.tbConfig.nLevel))
end

function tbWnd:UpdateListView()
  if not self.nCommonEntryId or not ActivityMgr.CommonEntryLinkData[self.nCommonEntryId] then
    return
  end
  local tbStepCfg = ActivityMgr.CommonEntryLinkData[self.nCommonEntryId].tbStep
  if not tbStepCfg or #tbStepCfg <= 0 then
    return
  end
  self.tbRightStepListView:Clear()
  for nStep, tbStepInfo in ipairs(tbStepCfg) do
    self.tbRightStepListView:SetElement(nStep, function(tbElement, nIndex)
      self:UpdateListElement(tbElement, nIndex, tbStepInfo)
    end)
  end
end

function tbWnd:UpdateListElement(tbElement, nIndex, tbStepInfo)
  tbElement.pPanel:Label_SetText("ImgAll/txtNum", nIndex)
  tbElement.pPanel:Label_SetText("ImgAll/txtActivityName", i18n.Parse(i18n.Encode(tbStepInfo.szInfo)))
  tbElement.pPanel:Label_SetText("ImgAll/txtInfo", i18n.Parse(i18n.Encode(tbStepInfo.szTime)))
  tbElement.pPanel:Button_SetText("ImgAll/btnWork", i18n.Parse(i18n.Encode(tbStepInfo.szButtonName)), "Text")
  tbElement.pPanel:Button_BindEvent("ImgAll/btnWork", self.OnStepGoBtnClick, self, tbStepInfo.szCallBack, tbStepInfo.szParam)
  if not Lib:IsEmptyStr(tbStepInfo.szRedPoint) then
    local tbRedPointEvent = Lib:SplitStr(tbStepInfo.szRedPoint, "|")
    if tbRedPointEvent and #tbRedPointEvent > 0 then
      for k, v in ipairs(tbRedPointEvent) do
        tbRedPointEvent[k] = RedPointMgr.EVENT[v]
      end
      RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "ImgAll/btnWork", tbRedPointEvent)
    end
  end
end

function tbWnd:OnStepGoBtnClick(szCallBack, szParam)
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam or "", "|")
  if fn and tbParam and #tbParam > 0 then
    fn(self.tbActivityList[self.nIndex].nId, table.unpack(tbParam))
    if "GotoPos" == szCallBack or "GotoNPC" == szCallBack then
      Ui:CloseWindow("UIMayDayActivity")
    end
  end
end

function tbWnd:UpdateAwardPreview()
  if not self.nCommonEntryId or not ActivityMgr.CommonEntryLinkData[self.nCommonEntryId] then
    return
  end
  local tbSetting = ActivityMgr.CommonEntryLinkData[self.nCommonEntryId].tbSetting
  local nShowAwardId = tbSetting.nShowItemId
  local nShowSingleAwardKey = tbSetting.tbShowAwardKey[1]
  local nShowMultiAwardKey = tbSetting.tbShowAwardKey[2]
  if not Lib:IsInteger(nShowAwardId) then
    return
  end
  if Lib:IsInteger(nShowSingleAwardKey) then
    self.tbSingleHandInAwardView:Clear()
    local tbSingleAwards = Award:GetAward(nShowAwardId, nShowSingleAwardKey, {bNotMerge = true})
    for nIndex, tbAward in ipairs(tbSingleAwards) do
      self.tbSingleHandInAwardView:SetItem(nIndex, tbAward, true)
      local szItemKey = self.tbSingleHandInAwardView:GetItemKey(nIndex)
      self.pPanel:Label_SetText(szItemKey .. "/txtActivityName", i18n.Parse(self:GetAwardName(tbAward, false)))
    end
  end
  if Lib:IsInteger(nShowMultiAwardKey) then
    self.tbMultiHandInAwardView:Clear()
    local tbMultiAwards = Award:GetAward(nShowAwardId, nShowMultiAwardKey, {bNotMerge = true})
    for nIndex, tbAward in ipairs(tbMultiAwards) do
      self.tbMultiHandInAwardView:SetItem(nIndex, tbAward, true)
      local szItemKey = self.tbMultiHandInAwardView:GetItemKey(nIndex)
      self.pPanel:Label_SetText(szItemKey .. "/txtActivityName", i18n.Parse(self:GetAwardName(tbAward, false)))
    end
  end
end

function tbWnd:GetAwardName(tbAward, bNeedCount)
  local nAwardType = Player.AwardType[tbAward[1]]
  local szName = ""
  if 3 == Player.AwardTypeCountIndex[nAwardType] then
    szName = Item:GetName(tbAward[2], me)
  elseif nAwardType == Player.award_type_money then
    szName = Currency:GetName(tbAward[1])
  end
  if Lib:IsEmptyStr(szName) then
    return ""
  end
  if bNeedCount then
    szName = string.format("%sx%s", szName, Player:GetAwardValue(tbAward))
  end
  return szName
end
