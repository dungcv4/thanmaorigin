local tbWnd = Ui:CreateClass("UIDaShiXiongMsg", "UIPanel")
local tbControls = {
  txtElectDaShiXiong = "ElectTitle/txtElectDaShiXiong",
  TemplateDefault = "ScrollView/List/TemplateDefault"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}
local ITEM_GRID_COUNT = 4

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.TemplateDefault)
end

function tbWnd:OnOpen()
  RemoteServer.RequestFactionBigBrother("GetBigBrotherWinnerInfo")
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSV(tbData)
  local tbOtherData = {}
  for nFaction, tbInfo in pairs(tbData) do
    local info = self:GetInfo(tbInfo)
    table.insert(tbOtherData, info)
  end
  
  local function fnShowItem(pItem, nIdx)
    for i = 1, ITEM_GRID_COUNT do
      local nDataIdx = (nIdx - 1) * ITEM_GRID_COUNT + i
      local info = tbOtherData[nDataIdx]
      local szItemKey = string.format("FactionOtherObj/FactionInfo%d", i)
      if info then
        pItem.pPanel:SetActive(szItemKey, true)
        pItem.pPanel:Label_SetText(szItemKey .. "/Faction/txtFaction", info.szFaction)
        pItem.pPanel:Sprite_SetSprite(szItemKey .. "/PortraitObj/imgFactionIcon", info.spFaction)
        pItem.pPanel:Label_SetText(szItemKey .. "/txtFightPowerValue", info.nFightPower)
        pItem.pPanel:Label_SetText(szItemKey .. "/txtPlayerName", info.szName)
        pItem.pPanel:Label_SetText(szItemKey .. "/txtKinName", info.szKinName)
        pItem.pPanel:Label_SetText(szItemKey .. "/PortraitObj/Level/Text", info.nLevel)
        if info.spRoleHead then
          pItem.pPanel:Sprite_SetSprite(szItemKey .. "/PortraitObj/Portrait/Image", info.spRoleHead)
          pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Portrait/Image", true)
        else
          pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Portrait/Image", false)
        end
      else
        pItem.pPanel:SetActive(szItemKey, false)
      end
    end
  end
  
  local nRow = math.ceil(#tbOtherData / ITEM_GRID_COUNT)
  self.tbScrollView:Reset()
  for i = 1, nRow do
    self.tbScrollView:SetElement(i, function(tbElement)
      fnShowItem(tbElement, i)
    end)
  end
end

function tbWnd:GetInfo(info)
  if not info then
    return nil
  end
  local headIcon = PlayerPortrait:GetSprite(info.nPortrait)
  if info.nPlayerId <= 0 then
    headIcon = nil
  end
  return {
    szFaction = Faction:GetName(info.nFaction),
    szKinName = info.szKinName,
    nFightPower = info.nFightPower,
    szName = info.szName,
    spRoleHead = headIcon,
    spFaction = Faction:GetIcon(info.nFaction),
    nLevel = info.nLevel
  }
end
