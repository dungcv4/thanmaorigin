function KeYiMen.C:GetState()
  return self.m.nState
end

function KeYiMen.C:_Init()
  if self.bInited then
    return
  end
  self.m = {}
  self:InitMapClass()
  FubenUIController:RegisterChecker(KeYiMen.Def.tbMapTIDArr, self)
  self.m.nSelectIndex = 1
  self.m.tbRargetDuactionRank = {}
  self.m.nTargetMapTID = nil
  self.bInited = true
end

function KeYiMen.C:_Uninit()
  if not self.bInited then
    return
  end
  self.bInited = false
  self.m.tbRargetDuactionRank = nil
  self.m.nTargetMapTID = nil
  self.m = {}
end

  return self.m.nSelectIndex or 1

function KeYiMen.C:OnRespondData(tbSyncData)
  local szOpenUI = self.m.szOpenUI
  self.m.szOpenUI = nil
  self.m.tbSyncData = tbSyncData
  self.m.nState = tbSyncData.nState
  self.m.nTongJoinCamp = tbSyncData.nTongJoinCamp
  self.m.nTongJoinCampChangeTime = tbSyncData.nTongJoinCampChangeTime
  self.m.bTargetDropped = tbSyncData.bTargetDropped
  if DEBUG_ZJQ then
    Assist.PrintDataStyle(tbSyncData, "----> KeYiMen.C:OnSyncMapData: ")
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_KEYIMEN_ON_RESPOND_STATE, tbSyncData.nState)
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
  if "UIKeYiMenEntrance" == szOpenUI then
    if not KeYiMen.Com:IsOpen() then
      szOpenUI = nil
      me.CenterMsg(i18n.EasyGet("\232\175\165\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
    end
  elseif "UIKeYiMenSelectCamp" == szOpenUI then
  end
  if szOpenUI and not Ui:WindowVisible(szOpenUI) then
    Ui:OpenWindow(szOpenUI)
  end
end

function KeYiMen.C:OnPlayerDead(nMapTID, nX, nY)
  self.m.tbPosWherePlayerDead = {
    nMapTID = nMapTID,
    nX = nX,
    nY = nY
  }
end

function KeYiMen.C:OnMapLoaded(nMapTID)
  if KeYiMen.Com:IsKeYiMenMap(nMapTID) and not AFK:IsFollowFightRunning() then
    local tbPos = self.m.tbPosWherePlayerDead
    if tbPos and KeYiMen.Com:IsEntranceMap(nMapTID) then
      if AFK:IsRunning() and Ui:GetIsAutoBackFightPosAfterRevive() then
        local AFKFightSetting = AFK:GetSetting():GetFightSetting()
        local tbLastFightPos = AFKFightSetting:GetFightPos()
        if tbLastFightPos then
          self:CheckAndGoto(tbPos.nMapTID, tbPos.nX, tbPos.nY)
        end
      end
    elseif AutoPath:GetLastPathReason() == AutoPath.PathReason.emFuben_GoToTargetPos then
      self:NextPos()
    else
      self:ClearPos()
    end
  end
  self.m.tbPosWherePlayerDead = nil
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_MAP_LOADED, KeYiMen.C)
EventNotify:RegistNotify(EventNotify.emNOTIFY_MAP_LOADED, KeYiMen.C.OnMapLoaded, KeYiMen.C)

function KeYiMen.C:GetBossData(nBossTID)
  local tbSyncData = self:GetSyncData()
  if tbSyncData.tbBossDataSet then
    local tbBossData = tbSyncData.tbBossDataSet[nBossTID]
    return tbBossData
  end
end

function KeYiMen.C:IsBossDead(nBossTID)
  local tbSyncData = self:GetSyncData()
  if tbSyncData.tbBossDataSet then
    local tbBoss = tbSyncData.tbBossDataSet[nBossTID]
    if tbBoss then
      return tbBoss.bIsDead == true
    end
  end
  return false
end

function KeYiMen.C:CheckCanEnterFuben(nMapTID)
  if me.nLevel < KeYiMen.Def.nJoinLevel then
    return false, KeYiMen.Text.JoinLevel(KeYiMen.Def.nJoinLevel)
  end
  local nMapLevel = Map:GetEnterLevel(nMapTID)
  if nMapLevel > me.nLevel then
    return false, KeYiMen.Text.MapLevel(nMapLevel)
  end
  
  local function fnEnter()
    RemoteServer.KeYiMenRequestEnterFuben()
  end
  
  local bRet, szMsg = ActivityCheck:CheckCanEnterActivityEx(me, nil, fnEnter, false, false, true, nil, {
    "UIKeYiMenEntrance"
  }, false)
  return bRet, szMsg
end

function KeYiMen.C:RequestEnterFuben()
  local nMapTID = KeYiMen.Def.nEntranceMapTID
  local bRet, szMsg = self:CheckCanEnterFuben(nMapTID)
  if not bRet then
    if not Lib:IsEmptyStr(szMsg) then
      me.CenterMsg(szMsg)
    end
    return false
  end
  return true
end

function KeYiMen.C:_LoadMapNpcInfo(nMapTID)
  local tbNpcInfoSet = {}
  local tbMapCfg = Map:GetMapSetting(nMapTID)
  if tbMapCfg then
    local tbFile = Lib:LoadTabFile(string.format("Setting/Map/%s/info/npc_info.txt", tbMapCfg.LogicFilePath)) or {}
    for _, tbCfg in pairs(tbFile) do
      tbNpcInfoSet[tonumber(tbCfg.NpcTemplateId)] = tbCfg
    end
  end
  return tbNpcInfoSet
end

function KeYiMen.C:GetTrapNpcInfoSet()
  local tbNpcInfoSet = self.m.tbNpcInfoSet
  if not tbNpcInfoSet then
    tbNpcInfoSet = {}
    for _, nMapTID in pairs(KeYiMen.Def.tbMapTIDArr) do
      local tbNpcInfoOneMap = self:_LoadMapNpcInfo(nMapTID)
      for _, tbNpcInfo in pairs(tbNpcInfoOneMap) do
        tbNpcInfoSet[tonumber(tbNpcInfo.NpcTemplateId)] = tbNpcInfo
      end
    end
    self.m.tbNpcInfoSet = tbNpcInfoSet
  end
  return tbNpcInfoSet
end

function KeYiMen.C:GetTrapNpcInfo(nNpcTID)
  local tbNpcInfoSet = self:GetTrapNpcInfoSet()
  for k, tbNpc in pairs(tbNpcInfoSet) do
    if k == nNpcTID then
      return tbNpc
    end
  end
  LogErr("fail to GetTrapNpcInfo:", nNpcTID)
end

function KeYiMen.C:PauseAutoCtrl(bOpenAFK)

end

function KeYiMen.C:ResumeAutoCtrl()

end

function KeYiMen.C:CheckAndGoto(nMapTID, nX, nY)
  if nMapTID and KeYiMen.Com:IsKeYiMenMap(nMapTID) then
    local nLayerIndex = KeYiMen.Com:MapTID2LayerIndex(nMapTID)
    KeYiMen.C:GotoLayer(nLayerIndex, {
      nMapTID = nMapTID,
      nX = nX,
      nY = nY
    })
    return true
  end
  return false
end

function KeYiMen.C:GotoLayer(nEndLayerIndex, tbBossPos)
  local nStartLayerIndex = KeYiMen.Com:MapTID2LayerIndex(me.nMapTemplateId)
  local tbLayerIndexArr = Lib:CreateDijkstraPointArr(KeYiMen.Def.tbReachable, nStartLayerIndex or 1, nEndLayerIndex)
  local tbPosArr = self:CreateGotoPosArr(tbLayerIndexArr)
  self.m.tbPosArr = tbPosArr
  if tbBossPos then
    table.insert(tbPosArr, 1, tbBossPos)
  end
  self:PauseAutoCtrl(nil ~= tbBossPos)
  if not nStartLayerIndex then
    self:RequestEnterFuben()
    AutoPath:SetLastPathReason(AutoPath.PathReason.emFuben_GoToTargetPos)
  else
    self:NextPos()
  end
end

function KeYiMen.C:CreateGotoPosArr(tbLayerIndexArr)
  local tbMapTIDArr = {}
  for _, v in ipairs(tbLayerIndexArr) do
    local nMapTID = KeYiMen.Com:LayerIndex2MapTID(v)
    table.insert(tbMapTIDArr, nMapTID)
  end
  local tbPosArr = {}
  for i = 1, #tbMapTIDArr - 1 do
    local nTo = tbMapTIDArr[i]
    local nFrom = tbMapTIDArr[i + 1]
    local tbTrapPos = KeYiMen.Com:GetTrapPosData(nFrom, nTo)
    local tbNpcInfo = self:GetTrapNpcInfo(tonumber(tbTrapPos.TrapNPC))
    table.insert(tbPosArr, {
      nMapTID = nFrom,
      nX = tonumber(tbNpcInfo.XPos),
      nY = tonumber(tbNpcInfo.YPos)
    })
  end
  return tbPosArr
end

function KeYiMen.C:NextPos()
  if self:IsGoingTo() then
    local tbPos = table.remove(self.m.tbPosArr)
    if tbPos and tbPos.nMapTID == me.nMapTemplateId then
      local nMapID, nX, nY = me.nMapId, tbPos.nX, tbPos.nY
      Timer:Register(10, function()
        self:GoToBossPosition(nMapID, nX, nY)
      end)
    end
  end
end

function KeYiMen.C:ClearPos()
  self.m.tbPosArr = nil
end

function KeYiMen.C:IsGoingTo()
  local tb = self.m.tbPosArr
  return nil ~= tb and #tb > 0
end

function KeYiMen.C:HowToSwitchMapAsSrcMap(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
  local nEndLayerIndex = KeYiMen.Com:MapTID2LayerIndex(nTargetMapTemplateId)
  if nEndLayerIndex then
    local nStartLayerIndex = KeYiMen.Com:MapTID2LayerIndex(nFromTemplateId)
    assert(nStartLayerIndex, "nFromTemplateId = " .. nFromTemplateId)
    local tbLayerIndexArr = Lib:CreateDijkstraPointArr(KeYiMen.Def.tbReachable, nStartLayerIndex, nEndLayerIndex)
    local tbPosArr = self:CreateGotoPosArr(tbLayerIndexArr)
    local tbPos = table.remove(tbPosArr)
    if tbPos then
      return true, {
        nTransPosX = tbPos.nX,
        nTransPosY = tbPos.nY
      }
    end
  else
    Dungeon:ExitDungeon()
    return false, nil
  end
end

function KeYiMen.C:HowToSwitchMapAsDstMap(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
  if ActivityCheck:CheckIsInSafeArea(me, false) then
    self:RequestEnterFuben()
  else
    if TeamMgr:IsFollowTeammate() then
      TeamMgr:StopFollowMember()

  end
end

function KeYiMen.C:InitMapClass()
  for _, nMapTID in ipairs(KeYiMen.Def.tbMapTIDArr) do
    local tbMapObj = Map:GetClass(nMapTID)
    if tbMapObj then
      function tbMapObj.TransmitTo(tbMapObj, nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
        return self:HowToSwitchMapAsSrcMap(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
      end
      
      function tbMapObj.TransmitFrom(tbMapObj, nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
        return self:HowToSwitchMapAsDstMap(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
      end
      
      function tbMapObj.IsNeedTrapPointToSwitchMap(tbMapObj, nFromMapTID, nToMapTID)
        return KeYiMen.Com:MapTID2LayerIndex(nFromMapTID) ~= nil and KeYiMen.Com:MapTID2LayerIndex(nToMapTID) ~= nil
      end
    end
  end
end
