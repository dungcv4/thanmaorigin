PlayerPortrait.tbRedPointIndex = {}
PlayerPortrait.tbPortraitData = {}

function PlayerPortrait:OnSynData(tbData, nNewPortrait)
  self.tbPortraitData = tbData
  if nNewPortrait then
    self.tbRedPointIndex[nNewPortrait] = true
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.NewPortrait)
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_PROTARIT_UPDATE)
end

function PlayerPortrait:OnChangePortrait()
  EventNotify.OnNotify(EventNotify.emNOTIFY_PROTARIT_CHANGE)
end

function PlayerPortrait:ChangePortrait(nPortrait)
  local bRet, szRet = self:CanChangePortrait(nPortrait)
  if not bRet then
    if szRet then
      me.CenterMsg(szRet)
    end
    return
  end
  RemoteServer.PortraitCmd("ChangePortrait", nPortrait)
end

function PlayerPortrait:CanChangePortrait(nPortrait)
  if not self:IsAvaliablePortraits(nPortrait) then
    return false, i18n.EasyGet("\229\164\180\229\131\143\230\156\170\232\167\163\233\148\129\239\188\129")
  end
  return true
end
