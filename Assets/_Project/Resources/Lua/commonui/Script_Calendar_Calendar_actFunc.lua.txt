Calendar.tbCheckComplete = {
  JianXiaCiDian = function()
    return JianXiaCiDian.tbTaskData.nAnsweredQustionNum >= JianXiaCiDian.nTotalQuestionNum
  end,
  YiJunTask = function()
    if not YiJunTask:IsFinishStartTask(me) or not YiJunTask.tbTaskData then
      return false
    end
    return YiJunTask.tbTaskData.nFinishCount >= YiJunTask.DAILY_TASKNUM
  end,
  YiJunFuBen = function()
    return DegreeCtrl:GetDegree(me, "YiJunFuBen") <= 0
  end,
  Battle = function()
    return Battle:GetTodayJoinBattleTimes(me) > 0
  end,
  GuanFuTongJi = function()
    if not GuanFuTongJi:IsFinishBeforeTask(me) or not GuanFuTongJi.tbTaskData then
      return false
    end
    return GuanFuTongJi.tbTaskData.nFinishCount >= GuanFuTongJi.nMaxCount
  end,
  ChuanGong = function()
    return ChuanGong:GetRemainCount(me, ChuanGong.tbRelationType.Recieve) <= 0 and ChuanGong:GetRemainCount(me, ChuanGong.tbRelationType.Send) <= 0
  end,
  XoyoGame = function()
    return DegreeCtrl:GetDegree(me, "XoyoGame") <= 0
  end,
  Barracks = function()
    return DegreeCtrl:GetDegree(me, "Barracks") <= 0
  end,
  Arena = function()
    return DegreeCtrl:GetDegree(me, "ArenaBuyTimes") <= 0 and DegreeCtrl:GetDegree(me, "Arena") <= 0
  end,
  KinFreightTask = function()
    local tbData = KinFreightTask:GetKinFreightData(me)
    tbData.nAcceptCount = tbData.nAcceptCount or 0
    local tbTask = KinFreightTask:GetKinFreightTask()
    return tbData.nAcceptCount > 0 and 0 == Lib:CountTB(tbTask)
  end,
  CommerceTask = function()
    local nDegree = DegreeCtrl:GetDegree(me, "CommerceTask")
    return nDegree < CommerceTask.COUNT_PER_WEEK
  end,
  WuLinGaoShou = function()
    return DegreeCtrl:GetDegree(me, "WuLinGaoShou") >= 1
  end,
  KinLevel = function()
    return ActivePoint:CheckActivityFinish("KinLevel")
  end,
  Loulan = function()
    local nMax = DegreeCtrl:GetMaxDegree("Loulan", me) or 0
    local nDegree = DegreeCtrl:GetDegree(me, "Loulan") or 0
    return nMax > nDegree
  end,
  XiuWeiSit = function()
    return DegreeCtrl:GetDegree(me, "XiuWeiSit") <= 0
  end,
  ZhaiXingLou = function()
    return ActivePoint:CheckFinishByKey(me, "ZhaiXingLou")
  end,
  DaShiXiong = function()
    return not FactionBigBrother:IsBigBrotherStarting()
  end,
  HanWu = function()
    return ActivePoint:CheckFinishByKey(me, "HanWu")
  end,
  CrossTimeRoom = function()
    return DegreeCtrl:GetDegree(me, "CrossTimeRoom") <= 0
  end,
  TieFuCheng = function()
    return ActivePoint:CheckFinishByKey(me, "TieFuCheng")
  end,
  Qiancengta = function()
    return LevelTower:GetRemainFirstPassCount(me) <= 0
  end,
  HanHaiMiCheng = function()
    return ActivePoint:CheckFinishByKey(me, "HanHaiMiCheng")
  end,
  LouLanMiZong = function()
    return DegreeCtrl:GetDegree(me, "LouLanMiZong") <= 0
  end,
  MultiFactionFight = function()
    return ActivePoint:CheckFinishByKey(me, "MultiFactionFight")
  end,
  GlobalDomainBattle = function()
    return ActivePoint:CheckFinishByKey(me, "GlobalDomainBattle")
  end,

  ResourceBattle = function()

function Calendar:GetSimpleActiveDegreeInfoByFinish(bFinish)
  if bFinish then
    return "1/1"
  else
    return "0/1"
  end
end

function Calendar:GetSimpleActiveDegreeInfoByName(szDegree)
  return self:GetSimpleActiveDegreeInfoByFinish(ActivePoint:CheckActivityFinish(szDegree))
end

      local szAddDegree = ""

        szAddDegree = string.format("<color=#%s>+%d</color>", Ui.tbColor:GetColorHexByDefineName(ChuanGong.Cfg.szAddCountColorName), nInnerAddDegree)
      end
      return string.format("%d%s/%d", nInnerDegree, szAddDegree, nInnerMax)
    end
    
    return fnGetDegreeShowText(nMax, nDegree, nAddDegree), fnGetDegreeShowText(nMax2, nDegree2, nAddDegree2), szTitle1, szTitle2
  end,
  KinFreightTask = function()

    local nMax = JiangHuZhuJiu.AWARDS[1].nWeekCount

    local nMaxTimes = DegreeCtrl:GetMaxDegree(szDegreeName, me)

  end,

  end,

  end
