local function fFormatThousands(nNumber)
  local szNumber = tostring(tonumber(nNumber) or 0)
  local nLength = #szNumber
  local nIndex, nBreakTime = nLength, nLength
  while true do
    szNumber, nIndex = string.gsub(szNumber, "^(-?%d+)(%d%d%d)", "%1,%2")
    nBreakTime = nBreakTime - 1
    if 0 == nIndex or 0 == nBreakTime then
      break
    end
  end
  return szNumber
end

function tbWnd:OnCreate()
  for szMemberName, tbTargetInfo in pairs(tbItemGridView) do
    local szTargetPath = table.unpack(tbTargetInfo)
    self[szMemberName] = Ui:NewItemGridView(self, szTargetPath, nil, false, false)
  end
  self.pFreeItemGrid = Ui:NewExistItemGrid(self, self.UIItemGrid)
end

function tbWnd:_MakeDynKey(szPath, pTargetPanel)
  pTargetPanel = pTargetPanel or self.pPanel
  local szKey = string.format("%s|%s", szPath, pTargetPanel)
  return szKey
end

function tbWnd:_IsDynReddotExist(szPath, pTargetPanel)
  return not not self._tbDynReddotList[self:_MakeDynKey(szPath, pTargetPanel)]
end

function tbWnd:_RegDynReddot(szPath, pTargetPanel)
  pTargetPanel = pTargetPanel or self.pPanel
  local szKey = self:_MakeDynKey(szPath, pTargetPanel)
  self._tbDynReddotList[szKey] = pTargetPanel
  RedPointMgr:DynamicRegisteRedPoint(self._tbDynReddotList[szKey], szPath, {
    RedPointMgr.EVENT.DailyGift
  })
end

function tbWnd:_UnRegDynReddot(szKey)
  if not self._tbDynReddotList[szKey] then
    return
  end
  local szPath = (Lib:SplitStr(szKey, "|") or {})[1]
  if not szPath then
    return
  end
  RedPointMgr:DynamicUnRegist(self._tbDynReddotList[szKey], szPath)
  self._tbDynReddotList[szKey] = nil
end

function tbWnd:OnOpen()
  self._tbDynReddotList = {}
  DailyGift.bHaveClick = true
  DailyGift:UpdateReddotState()
  self:UpdateView()
end

function tbWnd:_CanFreeGiftAward()
  return DailyGift:GetDayGiftGetState(me, DailyGift.DailyGiftFree) == DailyGift.nNotGet and DailyGift:GetAccumulateCount(me) > 0
end

function tbWnd:_CheckDailyGiftCanBuy()
  for nDailyGiftType = DailyGift.DailyGiftOne, DailyGift.DailyGiftCount do
    if DailyGift:GetDayGiftRemainTime(me, nDailyGiftType) > Recharge.SHORTCUT_DAY_GIFT_LIMIT then
      return false
    end
  end
  return true
end

function tbWnd:_CheckAnyGiftNotGet()
  local tbGiftTypeList = {
    DailyGift.DailyGiftOne,
    DailyGift.DailyGiftSix,
    DailyGift.DailyGiftTwelve,
    DailyGift.DailyGiftFree
  }
  for _, nGiftType in pairs(tbGiftTypeList) do
    if DailyGift:GetDayGiftGetState(me, nGiftType) == DailyGift.nNotGet then
      return true
    end
  end
  return false
end

function tbWnd:_GetTomorrowFreeGiftKey()
  local nKey = DailyGift:GetAccumulateCount(me)
  if self:_CanFreeGiftAward() then
    return nKey
  end
  nKey = nKey + (DailyGift:GetDayGiftGetState(me, DailyGift.DailyGiftOne) == DailyGift.nHadGet and DailyGift:GetDayGiftGetState(me, DailyGift.DailyGiftSix) == DailyGift.nHadGet and DailyGift:GetDayGiftGetState(me, DailyGift.DailyGiftTwelve) == DailyGift.nHadGet and 1 or -1)
  nKey = math.max(1, math.min(DailyGift.nAccumulateCountMax, nKey))
  return nKey
end

function tbWnd:UpdateRewardScorllView()
  local nUnitMinCount, nUnitMaxCount = table.unpack(tbItemGridShowUnit)
  for szMemberName, tbTargetInfo in pairs(tbItemGridView) do
    local _, nAwardKey, bIsFixAward = table.unpack(tbTargetInfo)
    local pItemGrid = self[szMemberName]
    pItemGrid:Clear()
    local tbAwardInfo = bIsFixAward and Award:GetAward_Fix(DailyGift.Cfg.nAwardId, nAwardKey) or Award:GetAward_Rand(DailyGift.Cfg.nAwardId, nAwardKey, {bNoRandom = true})
    tbAwardInfo = {
      table.unpack(tbAwardInfo or {}, nUnitMinCount, nUnitMaxCount)
    }
    for nIndex, tbAwardInfo in ipairs(tbAwardInfo) do
      pItemGrid:SetItem(nIndex, tbAwardInfo, true)
    end
    pItemGrid:ResetFrom(#tbAwardInfo + 1)
  end
  local tbAwardInfo = Award:GetAward(DailyGift.Cfg.nFreeGiftAwardId, self:_GetTomorrowFreeGiftKey())
  local bEmptyAward = Lib:IsEmptyTB(tbAwardInfo)
  local nFreeGiftState = DailyGift:GetDayGiftGetState(me, DailyGift.DailyGiftFree)
  self.pFreeItemGrid:SetClickEnable(true, not (nFreeGiftState ~= DailyGift.nNotGet or bEmptyAward) and function()
    DailyGift:GetDailyGiftAwardReq(DailyGift.DailyGiftFree)
  end or function()
    Item:ShowItemDetail(self.pFreeItemGrid)
    me.CenterMsg(i18n.EasyGet("\233\162\134\230\187\161\230\137\128\230\156\137\231\164\188\229\140\133\229\144\142\230\152\142\230\151\165\229\143\175\233\162\134\229\143\150"))
  end)
  self.pPanel:SetActive("ItemTemplate", not bEmptyAward)
  if nFreeGiftState == DailyGift.nNotGet and not bEmptyAward then
    self:_RegDynReddot(tbControls.UIItemGrid)
  else
    self:_UnRegDynReddot(self:_MakeDynKey(tbControls.UIItemGrid))
  end
  if bEmptyAward then
    return
  end
  self.pFreeItemGrid:SetGenericItem(tbAwardInfo[1])
  self.pFreeItemGrid.pPanel:SetActive("Higher", self:_CanFreeGiftAward())
end

function tbWnd:GetDayGiftGetState(nType)
  return DailyGift:GetDayGiftGetState(me, nType)
end

function tbWnd:OnClose()
  local tbRegKey = {}
  for szKey, _ in pairs(self._tbDynReddotList) do
    table.insert(tbRegKey, szKey)
  end
  for szKey, _ in pairs(tbRegKey) do
    self:_UnRegDynReddot(szKey)
  end
end
