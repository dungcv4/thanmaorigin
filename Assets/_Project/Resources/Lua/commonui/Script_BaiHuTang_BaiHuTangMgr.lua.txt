function BaiHuTang:QuickEnterHallReq()
  local nMapTemplateId = me.nMapTemplateId
  for _, nMapId in pairs(self.MAP_HALL.ID) do
    if nMapId == nMapTemplateId then
      return
    end
  end
  for _, tbInfoList in pairs(self.MAP_FLOOR) do
    for nMapId, _ in pairs(tbInfoList) do
      if nMapId == nMapTemplateId then
        return
      end
    end
  end
  local nTargetNpcTemplateId = 7015
  local nTargetMapTemplateId = 95
  local nTargetNpcX, nTargetNpcY, nWalkNearLength = Npc:GetNpcPos(nTargetNpcTemplateId, nTargetMapTemplateId, true)
  AutoPath:GotoAndCallEx(nTargetMapTemplateId, nTargetNpcX, nTargetNpcY, function()
    local tbNpcList = KNpc.GetAroundNpcList(me.GetNpc(), GetNpcDialogDistance(nTargetNpcTemplateId)) or {}
    for _, pNpc in pairs(tbNpcList) do
      if pNpc.nTemplateId == nTargetNpcTemplateId then
        Operation:SimpleTap(pNpc.nId)
        return
      end
    end
  end, nWalkNearLength)
end

function BaiHuTang:IsNeedFubenBtn()
  local nMapTid = me.nMapTemplateId
  local bFound = false
  for _, nOtherMapTid in pairs(self.MAP_HALL.ID) do
    if nOtherMapTid == nMapTid then
      bFound = true
    end
  end
  if not bFound then
    return false
  end
  local tbBtnList = {}
  
  local function fnGoto(nNpcTid)
    if nMapTid ~= me.nMapTemplateId then
      return
    end
    local nPosX, nPosY, nNearLength = Npc:GetNpcPos(nNpcTid, nMapTid)
    if not nPosX or not nPosY then
      return
    end
    
    local function fnCallback()
      local nNpcId = KNpc.GetNpcIdByTemplateId(nNpcTid)
      if nNpcId then
        Operation:SimpleTap(nNpcId)
      end
    end
    
    TeamMgr:StopFollowMember()
    AFK:Stop()
    AutoPath:GotoAndCallEx(nMapTid, nPosX, nPosY, fnCallback, nNearLength, nil, nNpcTid)
  end
  
  for nNpcTid, _ in pairs(self.MAP_HALL.TRANSMIT) do
    local tbInfoList = Map:GetMapNpcInfoByNpcTemplate(nMapTid, nNpcTid) or {}
    if tbInfoList[1] then
      table.insert(tbBtnList, {
        szName = i18n.Parse(tbInfoList[1].NpcName),
        fnOnClick = function()
          fnGoto(nNpcTid)
        end
      })
    end
  end
  return true, nil, tbBtnList
end

function BaiHuTang:GetFubenBtnAImagePathAndText()

end

function BaiHuTang:RegisterFubenUIChecker()
  local tbMapList = {}
  for _, nMapId in pairs(self.MAP_HALL.ID) do
    table.insert(tbMapList, nMapId)
  end
  for _, tbInfoList in pairs(self.MAP_FLOOR) do
    for nMapId, _ in pairs(tbInfoList) do
      table.insert(tbMapList, nMapId)
    end
  end
  FubenUIController:RegisterChecker(tbMapList, self)
end

function BaiHuTang:EnterBaiHuTangByCaptainRes(bIsCanEnter)
  if false == bIsCanEnter and TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
end

function BaiHuTang:GetFloor(nTemplateId)
  for nFloor, tbInfoList in pairs(self.MAP_FLOOR) do
    if tbInfoList[nTemplateId] then
      return nFloor
    end
  end
  return nil
end

function BaiHuTang:TransmitTo(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
  local nFloor = self:GetFloor(nFromTemplateId)
  local nTargetFloor = self:GetFloor(nTargetMapTemplateId)
  local tbWayPoint = Lib:RandTable({
    {nTransPosX = 34548, nTransPosY = 32613},
    {nTransPosX = 35657, nTransPosY = 31868}
  })
  if not nTargetFloor then
    local target = self.MAP_FLOOR[nFloor].TARGET
    if "Global" ~= target then
      return true, tbWayPoint
    else
      TeamMgr:S2C_StopFollowMember()
      return true
    end
  elseif nFloor >= nTargetFloor then
    TeamMgr:S2C_StopFollowMember()
    return false
  else
    return true, tbWayPoint
  end
end

function BaiHuTang:TransmitFrom(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
  RemoteServer.RequestBaiHuTang("EnterBaiHuTangByCaptainReq", nTargetMapId)
end

function BaiHuTang:OnLeavePlayer()
  self.tbCacheData = nil
  FubenUIController:CloseFubenUI()
end

function BaiHuTang:Init()
  if not self.bInit then
    self:RegisterFubenUIChecker()
    self.bInit = true
    for i, tbInfoList in pairs(self.MAP_FLOOR) do
      for nMapTemplateId, v in pairs(tbInfoList) do
        local tbBaiHuTangMapClass = Map:GetClass(nMapTemplateId)
        
        function tbBaiHuTangMapClass.TransmitTo(tbBaiHuTangMapClass, nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
          return self:TransmitTo(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
        end
        
        function tbBaiHuTangMapClass.TransmitFrom(tbBaiHuTangMapClass, nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
          return self:TransmitFrom(nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
        end
      end
    end
  end
end

function BaiHuTang:GotoEntrance()
  local bCanJoin, szMsg = BaiHuTang:CheckCanJoin()
  if not bCanJoin then
    me.CenterMsg(szMsg)
    return
  end
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  local bLeadFight = AFK:IsLeadFightRunning()
  if bLeadFight then
    AFK:Stop()
  end
  BaiHuTang:QuickEnterHallReq()
end

function BaiHuTang:CheckCanJoin()
  if DegreeCtrl:GetDegree(me, "BaiHuTang") <= 0 then
    return false, i18n.EasyGet("\230\130\168\231\154\132\231\153\189\232\153\142\229\160\130\229\143\130\228\184\142\230\172\161\230\149\176\229\183\178\230\187\161")
  end
  return true, ""
end
