local tbWnd = Ui:CreateClass("UIAnniversaryMain", "UIFestival_MainBase")
local tbControls = {
  Template = "List/Viewport/Content/Template",
  szTitle = "Tex/txtTitle",
  szTime = "Tex/txtTime01",
  btnHelp = "Tex/txtTitle/btnHelp",
  btnGift = "btnGift",
  wndTip = "node/wndTip"
}
tbWnd.tbControls = tbControls
tbWnd.tbTemplateElement = {
  txtActivityName = "ImgAll/txtActivityName",
  txtTimes = "ImgAll/txtTimes",
  txtInfo = "ImgAll/txtInfo",
  ItemTemplate = "ImgAll/ContList/Viewport/Content/ItemTemplate%d",
  GridTemplate = "ImgAll/ContList/Viewport/Content/ItemTemplate%d/UIItemGrid",
  img_Finish = "ImgAll/img_Finish",
  txtUnOpened = "ImgAll/txtUnOpened",
  btnGo = "ImgAll/btnGo"
}

function tbWnd:OnOpen(nEntranceIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex)
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnGift, {
      RedPointMgr.EVENT.AnniversaryLogin
    })
  end
end

tbWnd.tbOnClick = {
  btnGift = function(self)
    local tbUi = Ui:GetWindow("UIActivity_Anniversary")
    if tbUi then
      tbUi:RefreshSigninGift(true)
    end
  end
}
