Require("CommonScript/HonorLevel/HonorLevelDef.lua")
Require("Script/EventSystem/EventNotify.lua")

function HonorLevel:ActivateReq(nStarLevel)
  local bRet, szErr = self:CanActivate(me, nStarLevel)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestHonorLevel("ActivateReq", nStarLevel)
end

function HonorLevel:ActivateRsp(nOldStarLevel, nNewStarLevel)
  self:RefreshAttrib(me, nOldStarLevel, nNewStarLevel)
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_HONOR_LEVEL, nOldStarLevel, nNewStarLevel)
end

function HonorLevel:SellReq(nItemId)
  local bRet, szErr = self:CanSell(me, nItemId)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  
  local function fnConfirm()
    RemoteServer.RequestHonorLevel("SellReq", nItemId)
  end
  
  local pItem = me.GetItemInBag(nItemId)
  local szName = KItem.GetItemShowInfo(self.SELL_AWARD_ID, me.nFaction, me.nFactionSect, me.nSex)
  local nPrice = self:CalcSellPrice(me, pItem)
  local szItemTypeName = ""
  if pItem.nEquipPos == Item.EQUIPPOS_MANTLE then
    szItemTypeName = i18n.EasyGet("\230\138\171\233\163\142")
  elseif pItem.nEquipPos == Item.EQUIPPOS_GUANYIN then
    szItemTypeName = i18n.EasyGet("\229\174\152\229\141\176")
  end
  me.MsgBox(i18n.Get(HonorLevel.tbText.szMantleRecoverTips, nPrice, szName, szItemTypeName), {
    {
      i18n.szOk,
      fnConfirm
    },
    {
      i18n.szCancel
    }
  })
end

function HonorLevel:ExchangeReq(nItemId, nCount)
  if not Lib:IsInteger(nCount) then
    nCount = 1
  end
  local bRet, szErr = self:CanExchange(me, nItemId, nCount)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  
  local function fnConfirm()
    RemoteServer.RequestHonorLevel("ExchangeReq", nItemId, nCount)
  end
  
  local pOrgItem = me.GetItemInBag(nItemId)
  local nOrgNum = nCount
  local tbSetting = self.EXCHANGE_CFG[pOrgItem.dwTemplateId]
  local szExchItem = KItem.GetItemShowInfo(tbSetting.EXCHANGE_ITEM_ID)
  local nExchNum = tbSetting.EXCHANGE_ITEM_NUM * nOrgNum
  me.MsgBox(i18n.Get(self.tbText.szMaterialExchangeTips, nOrgNum, pOrgItem.szName, nExchNum, szExchItem), {
    {
      i18n.szOk,
      fnConfirm
    },
    {
      i18n.szCancel
    }
  })
end

function HonorLevel:OnLogin()
  self:OnCheckRedPoint()
end

function HonorLevel:OnCheckRedPoint()
  if self:CanActivate(me, self:GetStarLevel(me) + 1) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.HonorLevelUp)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.HonorLevelUp)
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, HonorLevel)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_HONOR_LEVEL, HonorLevel)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_FIGHT_POWER_CHANGE, HonorLevel)
EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, HonorLevel.OnCheckRedPoint, HonorLevel)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_HONOR_LEVEL, HonorLevel.OnCheckRedPoint, HonorLevel)
EventNotify:RegistNotify(EventNotify.emNOTIFY_FIGHT_POWER_CHANGE, HonorLevel.OnCheckRedPoint, HonorLevel)
