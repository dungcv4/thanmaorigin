local UBoxModule = luanet.import_type("UBoxModule")
UBox.ANDROID_APP_KEY = "36b87bd0"
UBox.IOS_APP_KEY = "b99c3720"

function UBox:SetUploadIP(szIp, nPort)
  if not szIp or not nPort then
    return
  end
  UBoxModule.SetUploadIP(szIp, nPort)
end

function UBox:StartRecord(bProfiler, bCapture, bNetwork, bResAnalyzer)
  local szAppKey
  if IOS then
    szAppKey = UBox.IOS_APP_KEY
  else
    szAppKey = UBox.ANDROID_APP_KEY
  end
  local szVersion = Ui.GameEnv.AppVersion
  UBoxModule.StartRecord(szAppKey, szVersion or "", bProfiler or false, bCapture or false, bNetwork or false, bResAnalyzer or false)
  Log("UBox:StartRecord")
end

function UBox:StartLiteBox(szaccountStr, bstartResource, bstartMemory, bstartWwise, bstartStack)
  local szAppKey
  if IOS then
    szAppKey = UBox.IOS_APP_KEY
  else
    szAppKey = UBox.ANDROID_APP_KEY
  end
  local szVersion = Ui.GameEnv.AppVersion
  UBoxModule.StartLiteBox(szAppKey, szVersion or "", szaccountStr or "", bstartResource or false, bstartMemory or false, bstartWwise or false, bstartStack or false)
  Log("UBox:StartLiteBox")
end

function UBox:StopRecordAndUpload(bCompress, bInternet)
  UBoxModule.StopRecordAndUpload(bCompress or false, bInternet)
  Log("UBox:StopRecordAndUpload")
end
