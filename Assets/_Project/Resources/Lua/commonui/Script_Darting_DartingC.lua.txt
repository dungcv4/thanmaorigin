function Darting:OnTryRefreshingDarting(nRefreshNum, nTemplateId, nRefreshTemplateId)
  self.nRefreshNum = nRefreshNum
  self.nCurTemplateId = nTemplateId
  self.nTargetTemplateId = nRefreshTemplateId
  Ui:OpenWindow("UIYunBiao", false, nRefreshNum, nTemplateId, nRefreshTemplateId)
end

function Darting:OnRefreshingDarting(nRefreshNum, nTemplateId, nRefreshTemplateId)
  EventNotify.OnNotify(EventNotify.emNOTIFY_DARTING_BEGIN_REFRESH)
  self.nRefreshNum = nRefreshNum
  self.nCurTemplateId = nTemplateId
  self.nTargetTemplateId = nRefreshTemplateId
  if self.DoDartingTimer then
    Timer:Close(self.DoDartingTimer)
  end
  self.DoDartingTimer = Timer:Register(self.Cfg.nDoDartingDelay * Env.GAME_FPS, function()
    self.DoDartingTimer = nil
    EventNotify.OnNotify(EventNotify.emNOTIFY_DARTING_ON_REFRESH)
    me.CenterMsg(i18n.EasyGet("\229\136\183\230\150\176\229\136\176\228\186\134{0}", Npc:GetNameByTemplateId(nRefreshTemplateId)))
  end)
end

function Darting:IsRefreshing()
  return self.DoDartingTimer ~= nil
end

function Darting:GetDartingCarList()
  local tbAwardInfo
  local tbList = {}
  for _, tbInfo in ipairs(self.Cfg.DartingCarList) do
    tbAwardInfo = self.Cfg.DartingAwardList[tbInfo.nColor]
    table.insert(tbList, {
      nTemplateId = tbInfo.nTemplateId,
      nAwardId = tbAwardInfo.nAwardId,
      nAwardKey = tbAwardInfo.nAwardSuccessKey
    })
  end
  return tbList
end

function Darting:SyncDoDarting(nTemplateId, nTargetTemplateId)
  self.nCurTemplateId = nTemplateId
  self.nTargetTemplateId = nTargetTemplateId
  if self.DoDartingTimer then
    Timer:Close(self.DoDartingTimer)
  end
  self.DoDartingTimer = Timer:Register(self.Cfg.nDoDartingDelay * Env.GAME_FPS, function()
    self.DoDartingTimer = nil
    EventNotify.OnNotify(EventNotify.emNOTIFY_DARTING_DO_DARTING)
  end)
end

    nRefreshNum = self.nRefreshNum,
    bForbidden = self.nCurTemplateId == self.Cfg.DartingCarList[#self.Cfg.DartingCarList].nTemplateId
  }

function Darting:GetDartingStatus()
  local nDartingTime = me.GetUserValue(self.SAVE_KEY, self.SAVE_ID_LIST.DartingTime)
  if nDartingTime > 0 then
    return i18n.EasyGet("\230\156\170\229\188\128\229\167\139")
  end
  local nTeamId = me.GetUserValue(self.SAVE_KEY, self.SAVE_ID_LIST.DartingTeamId)
  if nTeamId > 0 then
    return i18n.EasyGet("\232\191\155\232\161\140\228\184\173")
  end
  return i18n.EasyGet("\229\183\178\229\174\140\230\136\144")
end

function Darting:GotoMap(nMapId)
  if self.nInitNpcTemplateId <= 0 then
    ActivityMgr:GotoMap(nMapId)
    return
  end
  local nPosX, nPosY = self.Cfg.PathList[self.nInitNpcTemplateId].nPosX, self.Cfg.PathList[self.nInitNpcTemplateId].nPosY
  
  local function func()
    local nNpcId = KNpc.GetNpcIdByTemplateId(self.nInitNpcTemplateId)
    if nNpcId then
      Operation:SimpleTap(nNpcId)
    end
  end
  
  ActivityMgr:GotoDynamicNpc(nMapId, nPosX, nPosY, func, GetNpcDialogDistance(self.nInitNpcTemplateId))
end

end

function Darting:Init()
  if self.bHasInitialized then
    return
  end
  self.bHasInitialized = true
  EventNotify:RegistNotify(EventNotify.emNOTIFY_CROSS_DAYS, self.OnCrossDay, self)
end
