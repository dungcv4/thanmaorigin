function DingFengPK:LoadGuessSetting()
  self.tbGuessSetting = Lib:EasyLoadTabFile("Setting/DingFengPK/GuessSetting.tab", {"nIndex"})
  for nIndex, tbGuessSetting in pairs(self.tbGuessSetting) do
    local tbData = tbGuessSetting.tbszItemLimit
    if tbData then
      for nIndex = #tbData, 1, -1 do
        if "" ~= tbData[nIndex] then
          tbData[nIndex] = Lib:SplitStr2Num(tbData[nIndex])
          tbData[tbData[nIndex][1]] = tbData[nIndex][2]
          tbData[nIndex] = nil
        else
          table.remove(tbData, nIndex)
        end
      end
    end
  end
end

function DingFengPK:LoadAwardSetting()
  self.tbSingleAward = Lib:EasyLoadTabFile("Setting/DingFengPK/SingleAward.tab", {"nStage", "nResult"}, {
    "nStage",
    "nResult",
    "nAwardId",
    "nAwardKey"
  })
  self.tbAwardSetting = {}
  self.tbAwardSetting[self.STAGE_AWARD.PRE] = Lib:EasyLoadTabFile("Setting/DingFengPK/PreAward.tab", {"nId"}, {
    "nId",
    "nRank",
    "szDesc",
    "nAwardId",
    "nAwardKey"
  })
  self.tbAwardSetting[self.STAGE_AWARD.FINAL] = Lib:EasyLoadTabFile("Setting/DingFengPK/FinalAward.tab", {"nId"}, {
    "nId",
    "nPhase",
    "szDesc",
    "nAwardId",
    "nAwardKey"
  })
end

function DingFengPK:GetAwardSetting(nType)
  return self.tbAwardSetting[nType]
end
