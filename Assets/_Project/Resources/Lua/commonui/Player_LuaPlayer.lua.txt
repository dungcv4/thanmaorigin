function _LuaPlayer.GetItemCountByTypeInBag(nItemTemplateId)
  if not nItemTemplateId or nItemTemplateId < 0 then
    return 0, {}
  end
  local tbItemList = self.GetItemListInBag(nItemTemplateId)
  if not tbItemList then
    return 0, {}
  end
  local nCount = 0
  for _, pItem in pairs(tbItemList) do
    nCount = nCount + pItem.nCount
  end
  return nCount, tbItemList
end

function _LuaPlayer.GetItemCountInBags(nItemTemplateId, nPos, nTradeLimitType)
  local tbItem = self.GetItemListInBag(nItemTemplateId, nPos)
  local tbRet = {}
  if not tbItem then
    return 0, tbRet
  end
  local nCount = 0
  for _, pItem in pairs(tbItem) do
    if nTradeLimitType then
      if pItem.dwTradeLimitType == nTradeLimitType then
        nCount = nCount + pItem.nCount
        table.insert(tbRet, pItem)
      end
    else
      nCount = nCount + pItem.nCount
      table.insert(tbRet, pItem)
    end
  end
  return nCount, tbRet
end

function _LuaPlayer.GetItemCountInBagsEx(nItemTemplateId, nItemId)
  local tbItem = self.GetItemListInBag(nItemTemplateId)
  if not tbItem then
    return 0, {}
  end
  local nCount = 0
  local tbPItems = {}
  local tbItemIds = {}
  for _, pItem in pairs(tbItem) do
    if pItem.dwId ~= nItemId then
      nCount = nCount + pItem.nCount
      table.insert(tbPItems, pItem)
      table.insert(tbItemIds, pItem.dwId)
    end
  end
  return nCount, tbItem, tbPItems, tbItemIds
end

function _LuaPlayer.GetItemCountInBoxs(nItemTemplateId)

  end
  local tbItem = self.GetItemListInBox(nItemTemplateId)
  if not tbItem then

  end

  for _, pItem in pairs(tbItem) do
    nCount = nCount + pItem.nCount
  end
  return nCount, tbItem
end

function _LuaPlayer.GetCanAddItemCount(nItemTemplateId)
  local tbItemBase = Item:GetItemBaseProp(nItemTemplateId)
  local nMaxCount = tbItemBase.nStackMax
  local nFreeCount = self.GetFreeBagCount()
  if nFreeCount < 0 then
    nFreeCount = 0
  end
  if nMaxCount < 1 then
    nMaxCount = 1
  end
  local nCanAddCount = nMaxCount * nFreeCount
  return nCanAddCount
end

function _LuaPlayer.CanTeamOpt()
  local tbMapSetting = Map:GetMapSetting(self.nMapTemplateId)
  return tbMapSetting.TeamForbidden ~= TeamMgr.forbidden_make_and_operation_team and tbMapSetting.TeamForbidden ~= TeamMgr.forbidden_operation_team_in_client and not self.bForbitTeam
end

function _LuaPlayer.CanTranOpt(bStatus)
  if nil == bStatus then
    bStatus = true
  end
  local nMapTId = self.nMapTemplateId
  if self.nState == Player.emPLAYER_STATE_ALONE and self.nClientMapTemplateId then
    nMapTId = self.nClientMapTemplateId
  end
  if not bStatus then
    local tbMapSetting = Map:GetMapSetting(nMapTId)
    return 1 ~= tbMapSetting.TransForbidden, i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\228\184\141\229\143\175\228\188\160\233\128\129")
  end
  return true
end

function _LuaPlayer.CanTranOptEx(bStatus)
  local bRet, szMsg = self:CanTranOpt(bStatus)
  if not bRet then
    self.CenterMsg(szMsg)
  end
  return bRet
end

function _LuaPlayer.GetCamp()
  local pNpc = self.GetNpc()
  if not pNpc then
    return Npc.CampTypeDef.camp_type_newbie
  end
  return pNpc.nCamp
end

function _LuaPlayer.RestoreCamp(nCamp)
  local pNpc = self.GetNpc()
  if not pNpc then
    return
  end
  if not nCamp then
    if self.nLevel > Player.nNewbieLevel then
      nCamp = Npc.CampTypeDef.camp_type_none
    else
      nCamp = Npc.CampTypeDef.camp_type_newbie
    end
  end
  Log("_LuaPlayer.RestoreCamp", nCamp)
  pNpc.SetCamp(nCamp)
end

function _LuaPlayer.GetVipLevel()

end

function _LuaPlayer.UpdateVipLevel()
