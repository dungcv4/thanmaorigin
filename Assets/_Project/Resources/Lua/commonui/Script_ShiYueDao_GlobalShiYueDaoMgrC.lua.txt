Require("Script/Fuben/FubenUIController.lua")

function GlobalShiYueDao:SetLeaveShow(bShowLeaveBtn)
  self.bShowLeaveButton = bShowLeaveBtn
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function GlobalShiYueDao:GetFubenBtnAImagePathAndText()
  return "UI/atlas/hud/btn_teamfollow.png", i18n.EasyGet("\229\175\187\232\183\175")
end

function GlobalShiYueDao:IsNeedFubenBtn()
  return self:IsCanPath(), {
    self.AutoPathToEnter,
    self
  }
end

function GlobalShiYueDao:SetBtnPathShow(bShowPathBtn)
  self.bShowPathBtn = bShowPathBtn
  if Ui:WindowVisible("UIHudRightCenter") then
    Ui:GetWindow("UIHudRightCenter"):SetFubenBtnImageAndText(self:GetFubenBtnAImagePathAndText())
    Ui:GetWindow("UIHudRightCenter"):SetFubenBtnShowStatus(self:IsCanPath())
  end
end

function GlobalShiYueDao:IsCanPath()
  return self.bShowPathBtn or false
end

function GlobalShiYueDao:CheckCanJoin()
  if DegreeCtrl:GetDegree(me, "ShiYueDao") <= 0 then
    return false, i18n.EasyGet("\230\130\168\231\154\132\232\154\128\230\156\136\229\178\155\229\143\130\228\184\142\230\172\161\230\149\176\229\183\178\230\187\161")
  end
  return true, ""
end

function GlobalShiYueDao:IsNeedExitFubenButton()
  return self.bShowLeaveButton
end

function GlobalShiYueDao:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenDefaultFuBenUI()
end

function GlobalShiYueDao:RegisterFubenUIChecker()
  local tbTemplateID = {
    [1] = GlobalShiYueDao.SYDIslandFubenCfg.nTemplateMapId,
    [2] = GlobalShiYueDao.SYDHoleFubenCfg.nTemplateMapId
  }
  FubenUIController:RegisterChecker(tbTemplateID, self)
end

function GlobalShiYueDao:SyncDynMapPoint(nTemplateID, tbPointList, nTimeOut)
  local tbMapPoints = Lib:CopyTB(tbPointList)
  Map:SyncDynMapPoint(nTemplateID, tbMapPoints, nTimeOut)
  self.curTemplateID = nTemplateID
  self.tbEnterPointList = tbPointList
end

function GlobalShiYueDao:AutoPathToEnter()
  if not self.curTemplateID or not self.tbEnterPointList then
    return
  end
  local nTargetX = 0
  local nTargetY = 0
  local nMinDis = -1
  local nMapId, myPosX, myPosY = me.GetWorldPos()
  for _, tbPoint in pairs(self.tbEnterPointList) do
    local nPosX = tbPoint.nPosX
    local nPosY = tbPoint.nPosY
    local nCurDis = Lib:GetDistance(myPosX, myPosY, nPosX, nPosY)
    if -1 == nMinDis or nMinDis > nCurDis then
      nMinDis = nCurDis
      nTargetX = nPosX
      nTargetY = nPosY
    end
  end
  AutoPath:GotoAndCallEx(me.nMapId, nTargetX, nTargetY, nil, 0, nil, GlobalShiYueDao.SYDIslandFubenCfg.nJingYingEnterNpcId)
end

GlobalShiYueDao:RegisterFubenUIChecker()
