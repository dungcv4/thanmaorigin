local tbWnd = Ui:CreateWindow("UIPlayerModifyName")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  imgRolePortrait = "node/imgBG/Content/PortraitObj/Portrait/Image",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  txtPlayerName = "node/imgBG/Content/txtCurrentName",
  txtCurPlayerName = "node/imgBG/Content/inputNewName/Placeholder",
  txtFree = "node/imgBG/Content/txtFree",
  txtCost = "node/imgBG/Content/Cost",
  btnSave = "node/imgBG/Content/btnSave",
  imgMoney = "node/imgBG/Content/Cost/imgJinbi",
  txtCostNum = "node/imgBG/Content/Cost/txtCostNum",
  inputNewPlayerName = "node/imgBG/Content/inputNewName"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("ChangeNameHelp")
  end,
  btnSave = function(self)
    self:OnClickSave()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nItemId)
  local szIcon = PlayerPortrait:GetIcon(me.nPortrait)
  self.pPanel:Sprite_SetSprite(self.imgRolePortrait, szIcon)
  self.pPanel:Label_SetText(self.txtPlayerName, me.szName)
  self.pPanel:Label_SetText(self.txtCurPlayerName, me.szName)
  self.pPanel:Input_SetText(self.inputNewPlayerName, "")
  local szType, nCost = ChangeName:GetCostByCount(me)
  local szIcon = Currency:GetIcon(ChangeName.szChangeCostType)
  self.pPanel:Sprite_SetSprite(self.imgMoney, szIcon)
  self.pPanel:Label_SetText(self.txtCostNum, nCost)
  local bShowFree = ChangeName:GetFreeChange(me) or ChangeName:GetRemainTimes(me) > 0
  self.pPanel:SetActive(self.txtFree, bShowFree)
  self.pPanel:SetActive(self.txtCost, not bShowFree)
end

function tbWnd:OnClickSave()
  if me.nLevel < ChangeName.OPEN_LEVEL then
    me.CenterMsg(i18n.EasyGet("{0}\231\186\167\229\144\142\229\143\175\228\187\165\230\148\185\229\144\141", ChangeName.OPEN_LEVEL))
    return
  end
  if Ui.bInZone then
    me.CenterMsg(i18n.EasyGet("\232\183\168\230\156\141\228\184\173\230\151\160\230\179\149\232\191\155\232\161\140\229\189\147\229\137\141\230\147\141\228\189\156"))
    return
  end
  local szNewName = self.pPanel:Input_GetText(self.inputNewPlayerName)
  local bRet, szMsg = ChangeName:CheckNameinValid(szNewName)
  if false == bRet and szMsg then
    me.CenterMsg(szMsg)
    return
  end
  local bShowFree = ChangeName:GetFreeChange(me) or ChangeName:GetRemainTimes(me) > 0
  if not bShowFree then
    local szMoneyType, nCost = ChangeName:GetCostByCount(me)
    local nMoney = me.GetMoney(szMoneyType)
    if nCost > nMoney then
      me.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132{0}\228\184\141\232\182\179\239\188\129", Currency:GetName(szMoneyType)))
      return
    end
  end
  RemoteServer.RequestChangeName(szNewName)
end
