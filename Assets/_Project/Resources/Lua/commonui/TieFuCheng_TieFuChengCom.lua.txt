Require("CommonScript/TieFuCheng/TieFuChengCfg.lua")

function TieFuCheng:IsGameMap(nMapTid)
  for _, nSomeMapTid in pairs(self.FIGHT_MAP) do
    if nSomeMapTid == nMapTid then
      return true
    end
  end
  return false
end

function TieFuCheng:IsMeetTimeFrame()
  return self.bMeetTimeFrame
end

function TieFuCheng:IsCloseThisWeek()
  return not self.bMeetTimeFrame
end
