function CardPicker:OnLogin()
  self:CheckRedPoint()
end

function CardPicker:RequestCardPick(szType)
  if not CardPicker:CheckCardPickMoney(szType) then
    return
  end
  RemoteServer.OnCardPickerRequest(szType)
end

function CardPicker:CheckCardPickMoney(szType)
  local nItemCount = me.GetItemCountInBags(CardPicker.Def.nPartnerPickItemId)
  local nSkillBookItemCount = me.GetItemCountInBags(CardPicker.Def.nPartnerSkillBookPickItemId)
  local nLimitedCardPickTicketCount = me.GetItemCountInBags(CardPicker.Def.nPartnerLimitedPickItemId)
  if "CoinTenPick" == szType then
    if me.GetMoney("Coin") < CardPicker.Def.nCoinTenCost then
      self:NoMoney("Coin")
      return false
    end
  elseif "CoinPick" == szType then
    if me.GetMoney("Coin") < CardPicker.Def.nCoinCost then
      self:NoMoney("Coin")
      return false
    end
  elseif "GoldTenPick" == szType and nItemCount > 0 and nItemCount < 10 then
    local nYuanBaoNeed = (10 - nItemCount) * 188
    if nYuanBaoNeed > me.GetMoney("YuanBao") then
      self:NoMoney("YuanBao")
      return false
    else
      self:UsePartnerPickItemButNotEnough(szType)
      return false
    end
  elseif "GoldTenPick" == szType and nItemCount >= 10 then
    return true
  elseif "GoldTenPick" == szType then
    if me.GetMoney("YuanBao") < CardPicker.Def.nGoldTenCost then
      self:NoMoney("YuanBao")
      return false
    end
  elseif "GoldPick" == szType then
    if nItemCount > 0 then
      return true
    elseif me.GetMoney("YuanBao") < CardPicker.Def.nGoldCost then
      self:NoMoney("YuanBao")
      return false
    end
  elseif "SkillBookPick" == szType then
    if nSkillBookItemCount > 0 then
      return true
    elseif me.GetMoney("YuanBao") < CardPicker.Def.nSkillBookCost then
      self:NoMoney("YuanBao")
      return false
    end
  elseif "SkillBookTenPick" == szType and nSkillBookItemCount > 0 and nSkillBookItemCount < 10 then
    if CardPicker.bIsInSkillBookTenPickHalfPriceActivity then
      local nLastTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_HALFPRICE_SKILLBOOK_PICK_TIME)
      if Lib:IsDiffDay(14400, nLastTime) then
        if nSkillBookItemCount >= 5 then
          return true
        else
          local nYuanBaoNeed = (5 - nSkillBookItemCount) * 88
          if nYuanBaoNeed > me.GetMoney("YuanBao") then
            self:NoMoney("YuanBao")
            return false
          else
            self:UsePartnerSkillBookPickItemButNotEnoughAtHalfPrice(szType)
            return false
          end
        end
      else
        local nYuanBaoNeed = (10 - nSkillBookItemCount) * 88
        if nYuanBaoNeed > me.GetMoney("YuanBao") then
          self:NoMoney("YuanBao")
          return false
        else
          self:UsePartnerSkillBookPickItemButNotEnough(szType)
          return false
        end
      end
    else
      local nYuanBaoNeed = (10 - nSkillBookItemCount) * 88
      if nYuanBaoNeed > me.GetMoney("YuanBao") then
        self:NoMoney("YuanBao")
        return false
      else
        self:UsePartnerSkillBookPickItemButNotEnough(szType)
        return false
      end
    end
  elseif "SkillBookTenPick" == szType and nSkillBookItemCount >= 10 then
    return true
  elseif "SkillBookTenPick" == szType then
    if me.GetMoney("YuanBao") < CardPicker.Def.nSkillBookTenCost then
      self:NoMoney("YuanBao")
      return false
    end
  elseif "LimitedPick" == szType then
    if nLimitedCardPickTicketCount < CardPicker.Def.nLimitedCost then
      self:NoMoney("LimitedTicket")
      return false
    end
  elseif "LimitedTenPick" == szType and nLimitedCardPickTicketCount < CardPicker.Def.nLimitedTenCost then
    self:NoMoney("LimitedTicket")
    return false
  end
  return true
end

function CardPicker:SetLimitedPoolActivityState(bOpenActivity, tbUpPartnerInfo, nEndTime)
  CardPicker.bIsInLimitedPoolActivity = bOpenActivity
  CardPicker.tbLimitedPoolUpInfo = tbUpPartnerInfo
  CardPicker.nLimitedPoolActivityEndTime = nEndTime
  self:CheckRedPoint()
  Ui:CloseWindow("UICardPickingNew")
end

function CardPicker:CheckRedPoint()
  local nCoinFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_COIN_PICK_TIME)
  local nGoldFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_GOLD_PICK_TIME)
  local nCurTime = GetTime()
  local nLeftCoinSecond = 28800 - (nCurTime - nCoinFreeTime)
  local nLeftGoldSecond = 86400 - (nCurTime - nGoldFreeTime)
  if nLeftCoinSecond <= 0 then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.CardPickCoinFree)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.CardPickCoinFree)
  end
  if nLeftGoldSecond <= 0 then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.CardPickYuanBaoFree)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.CardPickYuanBaoFree)
  end
  if self.bIsInLimitedPoolActivity then
    local nLimitedFreeTime = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_FREE_LIMITED_PICK_TIME)
    local nLeftLimitedSecond = 86400 - (nCurTime - nLimitedFreeTime)
    if nLeftLimitedSecond <= 0 then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.CardPickLimitedFree)
    end
  end
  self:ResetRedPointCheckTimer()
end

function CardPicker:ResetRedPointCheckTimer()
  Ui:RegTimer("CardPick_RedPointCheck", Env.GAME_FPS * 1800, function()
    CardPicker:CheckRedPoint()
  end)
end
