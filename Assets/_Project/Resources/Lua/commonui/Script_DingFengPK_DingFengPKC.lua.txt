function DingFengPK:OnGetRanBoadData(nOpenType, tbData, nPage, nMaxLength)
  self.tbAllRank[nOpenType] = self.tbAllRank[nOpenType] or {}
  self.tbAllRank[nOpenType][nPage] = tbData
  self.tbUpdateDataTime = self.tbUpdateDataTime or {}
  self.tbUpdateDataTime[nOpenType] = self.tbUpdateDataTime[nOpenType] or {}
  self.tbUpdateDataTime[nOpenType][nPage] = GetTime()
  self.tbMaxLength = self.tbMaxLength or {}
  self.tbMaxLength[nOpenType] = nMaxLength
  EventNotify.OnNotify(EventNotify.emNOTIFY_DINGFENGPK_SYNC_RANK)
end

function DingFengPK:CheckUpdateData(nOpenType, nPage)
  if not self.tbAllRank then
    return
  end
  local tbData = self.tbAllRank[nOpenType] and self.tbAllRank[nOpenType][nPage]
  if not tbData then
    self:RequestOpenRankBoard(nOpenType, nPage)
    return
  end
  local nUpdateTime = self.tbUpdateDataTime[nOpenType] and self.tbUpdateDataTime[nOpenType][nPage]
  if not nUpdateTime or GetTime() - nUpdateTime > 15 then
    self:RequestOpenRankBoard(nOpenType, nPage)
    return
  end
  return tbData
end

function DingFengPK:RequestOpenRankBoard(nOpenType, nPage)
  if self.nRequestTimer then
    Timer:Close(self.nRequestTimer)
    self.nRequestTimer = nil
  end
  
  local function fnCall()
    RemoteServer.RequestDingFengPK("RequestRankData", nOpenType, nPage)
  end
  
  self.nRequestTimer = Timer:Register(9, fnCall)
end

function DingFengPK:ApplyMagicAttrib(nSelFaction, nSelFactionSect)
  local tbAttrListCfg = DingFengPK.tbEnhanceLevelCfg[nSelFaction] and DingFengPK.tbEnhanceLevelCfg[nSelFaction][nSelFactionSect] or nil
  if not tbAttrListCfg then
    return false
  end
  for _, tbAttrInfo in ipairs(tbAttrListCfg) do
    me.ApplyMagicAttrib(tbAttrInfo[1], tbAttrInfo[2], tbAttrInfo[3], tbAttrInfo[4])
  end
end

function DingFengPK:RemoveMagicAttrib(tbAttrList)
  if not tbAttrList then
    return false
  end
  for _, tbAttrInfo in ipairs(tbAttrList) do
    me.RemoveMagicAttrib(tbAttrInfo[1], tbAttrInfo[2], tbAttrInfo[3], tbAttrInfo[4])
  end
end
