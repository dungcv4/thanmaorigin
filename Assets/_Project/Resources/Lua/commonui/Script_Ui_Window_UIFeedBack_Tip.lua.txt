local tbWnd = Ui:CreateWindow("UIFeedBack_Tip")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_COLLECTPOINT_UPDATE,
      self.UpdateLevelList
    }
  }
end

local tbControls = {
  btnGo = "node/imgBG/Content/btnGo",
  btnClose = "node/imgBG/btnClose",
  Template = "node/imgBG/Content/Down/ScrollView/Viewport/Content/Template",
  txtTittle = "node/imgBG/Content/Down/imgTittle/txtActivityTittle",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  txtInfo = "node/imgBG/Content/Up/txtInfo"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnGo = function(self)
    if not self.bOpenExchange then
      return
    end
    self.pIns:ReqExchange(self.tbSelectNums)
  end
}

function tbWnd:OnCreate()
  self.tbExchangeList = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(nActId)
  self.pIns = ActivityMgr:GetInstanceById(nActId)
  if not self.pIns then
    return
  end
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self:UpdateLevelList()
  self.bOpenExchange = GetTime() >= Lib:GetDate2Time(self.pIns.tbConfig.nStartExchangeTime)
  self.nPoint = self.pIns:GetPoint(me)
  local bHavePoint = self.nPoint > 0
  self.pPanel:Button_SetEnable(tbControls.btnGo, self.bOpenExchange and bHavePoint)
  local szExchangeItemName = KItem.GetItemShowInfo(self.pIns.tbConfig.nExchangeItemId)
  self.pPanel:Label_SetText(tbControls.txtInfo, i18n.EasyGet("\230\175\143\228\189\191\231\148\1681\228\184\170{0}\229\143\175\232\142\183\229\190\1511\231\130\185\229\165\150\229\138\177\231\130\185\239\188\140\229\143\175\228\189\191\231\148\168\229\165\150\229\138\177\231\130\185\229\133\145\230\141\162\228\187\165\228\184\139\229\165\150\229\138\177", i18n.Parse(szExchangeItemName)))
end

function tbWnd:UpdateLevelList(nPoint)
  local tbExchange = self.pIns:GetExchangeInfo()
  self.tbExchangeList:Clear()
  self.nPoint = nPoint or self.pIns:GetPoint(me)
  self.pPanel:Label_SetText(tbControls.txtTittle, i18n.EasyGet("\229\189\147\229\137\141\229\165\150\229\138\177\231\130\185\239\188\154{0}", self.nPoint))
  self.tbSelectNums = {}
  for index, tbInfo in ipairs(tbExchange) do
    self.tbSelectNums[index] = 0
    self.tbExchangeList:SetElement(index, function(tbElement)
      local tbAward = Award:GetAward(tbInfo.nAwardId, tbInfo.nAwardKey)
      local tbShowAwardScrollView = Ui:NewItemGridScrollView(tbElement, "rewardList/reward", 1, 1)
      tbShowAwardScrollView:UpdateGenericItem(tbAward, true)
      local tbAddAndSubGroup = Ui:NewExistAddAndSubGroup(tbElement, "UIAddAndSubGroup")
      tbAddAndSubGroup:SetNumRange(0, self.nPoint)
      tbAddAndSubGroup:SetNumPadPos(-600, -50)
      tbAddAndSubGroup:SetInputValue(0)
      tbAddAndSubGroup:BindEvent(function(nValue)
        self.tbSelectNums[index] = nValue
        local nTotal = 0
        for key, value in pairs(self.tbSelectNums) do
          nTotal = nTotal + value
        end
        if nTotal > self.nPoint then
          self.tbSelectNums[index] = nValue - (nTotal - self.nPoint)
          nValue = self.tbSelectNums[index]
        end
        return nValue
      end, nil, nil, nil, {})
    end)
  end
end

function tbWnd:OnClose()
  self.tbExchangeList:Clear()
end
