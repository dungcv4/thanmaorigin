local tbWnd = Ui:CreateWindow("UISearch")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnTemplate = "node/imgBG/Content/ScrollViewSearchTips/Viewport/Content/btnTemplate",
  inputSearch = "node/imgBG/Content/inputSearch",
  DropdownSearch = "node/imgBG/Content/DropdownSearch",
  btnSearch = "node/imgBG/Content/btnSearch"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSearch = function(self)
    self:Close()
    local szSearchText = self.pPanel:Input_GetText(tbControls.inputSearch)
    if Lib:IsEmptyStr(szSearchText) then
      me.CenterMsg(i18n.EasyGet("\232\175\183\232\190\147\229\133\165\229\133\179\233\148\174\229\173\151"))
      return
    end
    if Ui:WindowVisible("UIBag") then
      Ui:GetWindow("UIBag"):OnSearchItem(self.nItemType, szSearchText)
    end
  end
}

function tbWnd:OnCreate()
  self.tbTxtTypeScrollView = Ui:NewScrollView(self, tbControls.btnTemplate, self.OnTxtClick)
  self.pPanel:Dropdown_BindEvent(tbControls.DropdownSearch, function(nIndex)
    self.nItemType = self.tbIndex2ItemType[nIndex + 1]
    self:UpdateView()
  end)
end

function tbWnd:OnDestroy()
end

function tbWnd:OnOpen(nItemType)
  self.nItemType = nItemType
  local tbLevelOptions = {
    [Ui.tbToggle.TogNormal] = i18n.EasyGet("\229\184\184\231\148\168"),
    [Ui.tbToggle.TogEquip] = i18n.EasyGet("\232\163\133\229\164\135"),
    [Ui.tbToggle.TogStone] = i18n.EasyGet("\229\174\157\231\159\179"),
    [Ui.tbToggle.TogPartner] = i18n.EasyGet("\229\144\140\228\188\180")
  }
  self.tbIndex2ItemType = {
    [1] = Ui.tbToggle.TogNormal,
    [2] = Ui.tbToggle.TogEquip,
    [3] = Ui.tbToggle.TogStone,
    [4] = Ui.tbToggle.TogPartner
  }
  if Item.ZhenYuan:IsSystemOpen(me) then
    table.insert(tbLevelOptions, i18n.EasyGet("\231\156\159\229\133\131"))
    table.insert(self.tbIndex2ItemType, Ui.tbToggle.TogZhenYuan)
  end
  if JingMai:IsSystemOpen(me, 1) then
    table.insert(tbLevelOptions, i18n.EasyGet("\231\187\143\232\132\137"))
    table.insert(self.tbIndex2ItemType, Ui.tbToggle.TogJingMai)
  end
  table.insert(tbLevelOptions, i18n.EasyGet("\229\133\168\233\131\168"))
  table.insert(self.tbIndex2ItemType, Ui.tbToggle.TogAll)
  local nSelectIndex = 0
  for index, value in ipairs(self.tbIndex2ItemType) do
    if nItemType == value then
      nSelectIndex = index - 1
    end
  end
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownSearch, tbLevelOptions)
  self.pPanel:Dropdown_SetValue(tbControls.DropdownSearch, nSelectIndex)
  self.pPanel:Input_SetPlaceHolderText(tbControls.inputSearch, i18n.EasyGet("\232\190\147\229\133\165\229\133\179\233\148\174\229\173\151\230\144\156\231\180\162"))
  self.pPanel:Input_SetText(tbControls.inputSearch, "")
  self.pPanel:Input_SetEnable(tbControls.inputSearch, true)
  self:UpdateShowTxtFind()
end

function tbWnd:OnClose()
end

function tbWnd:OnTxtClick(nIndex)
  local szText = self.tbQuickSearchType[nIndex]
  self.pPanel:Input_SetText(tbControls.inputSearch, i18n.Get(szText))
end

function tbWnd:UpdateView()
  self.pPanel:Input_SetText(tbControls.inputSearch, "")
  self:UpdateShowTxtFind()
end

function tbWnd:UpdateShowTxtFind()
  self.tbQuickSearchType = ItemSearch.tbItemType2QuickSearchText[self.nItemType]
  self.tbTxtTypeScrollView:Reset()
  for i, szText in pairs(self.tbQuickSearchType) do
    self.tbTxtTypeScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtTips1", i18n.Get(szText))
    end)
  end
end
