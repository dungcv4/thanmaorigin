function Kin:GetBuildingOpenLevel(nBuildingId)
  if nBuildingId == Kin.Def.Building_Main then
    return Kin:GetMainBuildingOpenLevel()
  else
    local nMaxLevel = Kin:GetBuildingMaxLevel(nBuildingId)
    if 1 == nMaxLevel then
      return 1
    end
    local nMainLevel = Kin:GetLevel()
    local nNextLevel = Kin:GetBuildingLevel(nBuildingId) + 1
    local bCanLevelup, nMainBuildeRequre = Kin:CanLevelUp(nBuildingId, nNextLevel, nMainLevel)
    return bCanLevelup and nNextLevel or nNextLevel - 1
  end
end

function Kin:GetBuildingData(nBuildingId)

  return tbBuilding[nBuildingId]
end

function Kin:GetBuildingNextLevelCost(nBuildingType, bCanUp)
  local nBuildingLevel = self:GetBuildingLevel(nBuildingType)
  local nMaxLevel = not bCanUp and self:GetCurMaxBuildingLevel(nBuildingType) or self:GetBuildingMaxLevel(nBuildingType)
  if nBuildingLevel >= nMaxLevel then
    return 0
  end
  local tbLevelInfo = Kin:GetBuildingInfo(nBuildingType, nBuildingLevel + 1)
  if not tbLevelInfo then
    return 0
  end
  local nMaxMainLevel = self:GetCurMaxBuildingLevel(Kin.Def.Building_Main)
  local nDiscount = self:GetBuildingUpGradeDiscount(nBuildingType, nMaxMainLevel - nBuildingLevel)
  return math.ceil(tbLevelInfo.nLevelUpCost * nDiscount)
end

  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.KinBuilding)

function Kin:CheckBuildingLevelUp()
  local nCareer = self:GetMyCareer()
  if not nCareer or nCareer > Kin.Def.Career_ViceMaster then
    self:FakeKinBuildingRedPoint()
    return
  end
  local nBuildingType = Kin.Def.Building_Main
  local nCurLevel = self:GetBuildingLevel(nBuildingType)
  local nMaxLevel = self:GetCurMaxBuildingLevel(nBuildingType)
  if nCurLevel >= nMaxLevel then
    self:FakeKinBuildingRedPoint()
    return
  end
  local tbLevelInfo = self:GetBuildingInfo(nBuildingType, nCurLevel + 1)
  local nMainLevel = self:GetBuildingLevel(Kin.Def.Building_Main)
  if nMainLevel < tbLevelInfo.nMainBuildingRequire then
    self:FakeKinBuildingRedPoint()
    return
  end
  local nMaxMainLevel = self:GetCurMaxBuildingLevel(Kin.Def.Building_Main)
  local nDiscount = self:GetBuildingUpGradeDiscount(nBuildingType, nMaxMainLevel - nCurLevel)
  local nFund = self:GetFund()
  if nFund < math.ceil(tbLevelInfo.nLevelUpCost * nDiscount) then
    self:FakeKinBuildingRedPoint()
    return
  end
  RedPointMgr:LightRedPoint(RedPointMgr.EVENT.KinBuilding)
end

function Kin:CheckCanChatInCommand()
  local authority = Kin.Def.Authority_CommanderChat
  local nCareer
  if Kin:AmIMaster() then
    nCareer = Kin.Def.Career_Master
  else
    local myMemberData = Kin:GetMyMemberData()
    if myMemberData and myMemberData.tbAuthority and myMemberData.tbAuthority[authority] then
      return true
    end
    nCareer = self:GetMyCareer()
  end
  if not nCareer then
    return false
  end
  local careerAuthority = Kin.Def.Career_Authority[nCareer] or {}
  return careerAuthority[authority]
end

function Kin:CheckMyAuthority(authority)
  local nCareer
  if Kin:AmIMaster() then
    nCareer = Kin.Def.Career_Master
  else
    nCareer = self:GetMyCareer()
  end
  if not nCareer then
    return
  end
  local careerAuthority = Kin.Def.Career_Authority[nCareer] or {}
  if careerAuthority[authority] then
    return true
  end
  return false
end

function Kin:CheckCanKickOut(nCareer)

  end

    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.LevelLimit, Kin.Cfg.nLevelLimit))
    return false
  end

    me.CenterMsg(i18n.EasyGet("\232\175\183\232\190\147\229\133\165\229\174\182\230\151\143\229\144\141\229\173\151"))
    return false
  end

    me.CenterMsg(i18n.EasyGet("\229\174\182\230\151\143\229\144\141\229\189\147\228\184\173\228\184\141\229\143\175\229\144\171\230\156\137\231\169\186\230\160\188"))
    return false

  if ReplaceLimitWords(szKinName) then
    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.IllegalChar))
    return false
  end
  local nNameLen = Lib:StrLen(szKinName)
  if nNameLen > Kin.Cfg.nMaxKinNameLength or nNameLen < Kin.Cfg.nMinKinNameLength then

    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.IllegalChar))
    return false
  end
  local bOk, nMin, nMax = Lib:CheckLength(szAddDeclare, i18n.CurrentLanguageCode(), 0, Kin.Cfg.nMaxAddDeclareLength)

function Kin:GetMemberCareerInfo(nCareer)
  local tbMemberList = self:GetMemberList() or {}
  
  local function fnEqual(tbMember, nCareer)
    return tbMember.nCareer == nCareer
  end
  
  local nCount = 0
  local tbCareerPos = {}
  for _, tbItem in pairs(tbMemberList) do
    if fnEqual(tbItem, nCareer) then
      nCount = nCount + 1
      if tbItem.nCareerPos then
        tbCareerPos[tbItem.nCareerPos] = tbItem.nMemberId
      end
    end
  end
  return nCount, tbCareerPos
end

function Kin:InsertClientMember(tbCacheMember, tbCacheMap, tbMember)
  if tbCacheMap[tbMember.nMemberId] then
    for k, v in pairs(tbCacheMember) do
      if v.nMemberId == tbMember.nMemberId then
        table.remove(tbCacheMember, k)
        break
      end
    end
  end
  table.insert(tbCacheMember, tbMember)
  tbCacheMap[tbMember.nMemberId] = tbMember
end

function Kin:DeleteClientMember(tbCacheMember, tbCacheMap, tbMember)
  if tbCacheMap[tbMember.nMemberId] then
    for k, v in pairs(tbCacheMember) do
      if v.nMemberId == tbMember.nMemberId then
        table.remove(tbCacheMember, k)
        break
      end
    end
  end
  tbCacheMap[tbMember.nMemberId] = nil
end

function Kin:OnSyncMemberList(tbMemberList)
  local tbCacheMember, nVersion, tbCacheMap = self:GetMemberList()
  local tbOnlineMembers = self:GetMemberState() or {}
  tbCacheMember = tbCacheMember or {}
  nVersion = nVersion or 0
  tbCacheMap = tbCacheMap or {}
  for _, tbMember in pairs(tbMemberList) do
    if not tbMember.bDelete then
      self:InsertClientMember(tbCacheMember, tbCacheMap, tbMember)
      tbOnlineMembers[tbMember.nMemberId] = 1 == tbMember.nIsOnLine and true or nil
    else
      self:DeleteClientMember(tbCacheMember, tbCacheMap, tbMember)
      tbOnlineMembers[tbMember.nMemberId] = nil
    end
  end
  Kin:CacheData("MemberList", tbCacheMember, nVersion)
  Kin:CacheData("MemberMap", tbCacheMap, 0)
  self:SetMemberState(tbOnlineMembers)
end

function Kin:SendMail(szContent)

    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.PermissionDenied))
    return false
  end
  local nLeftMailCount = Kin:GetLeftMailCount()
  if nLeftMailCount <= 0 then
    me.CenterMsg(i18n.Get(Kin.Cfg.Msg.MailCountLimit))
    return false
  end
  local bOk, nErr = Kin:CheckMailContent(szContent)
  if not bOk then
    me.CenterMsg(Kin:GetErrorMsg(nErr))
    return false
  end
  RemoteServer.OnKinRequest("SendKinMail", szContent)
  return true
end

function Kin:GetErrorMsg(MsgCode)
  local szMsg = ""
  if type(MsgCode) == "table" then
    MsgCode[1] = tbMsgCode2Msg[MsgCode[1]]
    if MsgCode[1] then
      szMsg = i18n.Get(table.unpack(MsgCode))
    else
      error("Kin:OnServerResult  Unknow MsgCode")
    end
  else
    szMsg = tbMsgCode2Msg[MsgCode]
    if szMsg then
      szMsg = i18n.Get(szMsg)
    else
      error("Kin:OnServerResult  Unknow MsgCode")
    end
  end
  return szMsg
end

function Kin:OnServerResult(szRequestType, bSuccess, MsgCode)
  if MsgCode then
    local szMsg = Kin:GetErrorMsg(MsgCode)
    if szMsg then
      me.CenterMsg(szMsg)
    end
    if "Apply" == szRequestType and bSuccess then
      Timer:Register(Env.GAME_FPS * 1, function()
        Kin:OpenKinWindow()
      end)
    end
  end
end

function Kin:NotifySalaryRedPoint(nGotSalary)
  local nLastWeekSalary = 0
  local myMemberData = self:GetMyMemberData()
  if myMemberData and myMemberData.nLastWeekSalary then
    nLastWeekSalary = myMemberData.nLastWeekSalary
  end
  if nGotSalary then
    nLastWeekSalary = nLastWeekSalary - nGotSalary
  end
  if nLastWeekSalary > 0 then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.KinSalary)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.KinSalary)
  end
end

function Kin:UpdateKinGiftBox()
  local nNow = GetTime()
  local nLeftCount, nNextBuyTime = Kin:GetGiftBoxData()
  nLeftCount = nLeftCount or 0
  nNextBuyTime = nNextBuyTime or nNow
  if self.nGiftBoxCountDownTimer then
    Timer:Close(self.nGiftBoxCountDownTimer)
    self.nGiftBoxCountDownTimer = nil
  end
  self.nGiftBoxCountDownTimer = Timer:Register(Env.GAME_FPS, function()
    local nLeftTime = nNextBuyTime - GetTime()
    if Ui:WindowVisible("UIKin") then
      EventNotify.OnNotify(EventNotify.emNOTIFY_UPDATE_KIN_GIFTBOX, nLeftTime, nLeftCount)
    end
    if nLeftTime < 0 or nLeftCount <= 0 then
      self:UpdateGiftRedPoint()
      self.nGiftBoxCountDownTimer = nil
      return false
    end
    return true
  end)
end

function Kin:OnSyncLeaderQuitTime(nQuitTime)
  self.tbQuitTime = self.tbQuitTime or {}
  self.tbQuitTime[me.dwID] = nQuitTime
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_LEADER_QUIT)
end

function Kin:IsInLeaderQuitTime()
  if not self.tbQuitTime then
    return false
  end
  if not self.tbQuitTime[me.dwID] then
    return false
  end
  if self.tbQuitTime[me.dwID] < GetTime() then
    return false
  end
  return true
end
