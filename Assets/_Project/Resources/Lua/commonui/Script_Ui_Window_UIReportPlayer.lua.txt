local tbWnd = Ui:CreateWindow("UIReportPlayer", "UIPanel")
tbWnd.tbControls = {
  btnClose = "imgBG/Frame/btnClose",
  txtPlayerName = "imgBG/Frame/InnerBg/txtPlayerName",
  Dropdown = "imgBG/Frame/InnerBg/Dropdown",
  InputField = "imgBG/Frame/InnerBg/InputField",
  btnSubmit = "imgBG/Frame/InnerBg/btnSubmit"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSubmit = function(self)
    self:OnClickSubmit()
  end
}

function tbWnd:OnCreate()
  self:InitDropdown()
  self.pPanel:Dropdown_BindEvent(self.Dropdown, function(index)
    self.nReportReasonId = index + 1
  end)
  self.pPanel:Input_OnEndEdit(self.InputField, function(text)
    self.szReportDesc = text
  end)
end

function tbWnd:OnOpen(dwPlayerId, szPlayerName, szChatMsg)
  self.szReportDesc = nil
  self.nReportReasonId = Report.DefaultDropdownIndex + 1
  self.pPanel:Dropdown_SetValue(self.Dropdown, Report.DefaultDropdownIndex)
  self.playerId = dwPlayerId
  self.szChatMsg = szChatMsg
  self.pPanel:Label_SetText(self.txtPlayerName, i18n.EasyGet("{0}", szPlayerName))
end

function tbWnd:OnClose()
end

function tbWnd:InitDropdown()
  local tbReportCfg = Report:GetReportCfg()
  local dropdownOptions = {}
  for reportId, reportCfgData in pairs(tbReportCfg) do
    table.insert(dropdownOptions, i18n.Parse(reportCfgData.ReportDesc))
  end
  self.pPanel:Dropdown_ResetOptions(self.Dropdown, dropdownOptions)
end

function tbWnd:OnClickSubmit()
  Report:EnsureReport(self.playerId, self.nReportReasonId, self.szReportDesc, self.szChatMsg)
end
