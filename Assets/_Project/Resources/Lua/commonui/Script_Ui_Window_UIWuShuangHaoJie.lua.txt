function tbWnd:OnCreate()
  self.tbScrollViewMember = Ui:NewScrollView(self, tbControls.TemplateMember)
  local tbBtnList = {
    tbControls.btnTeam,
    tbControls.btnWuShuang,
    tbControls.btnHaoJie
  }
  self.tbButtonGroup = Ui:NewButtonGroup(self, tbBtnList, true, function(self, nIndex)
    self:UpdatePanel(nIndex)
  end)
  self.tbWuShuangAwardScrollView = Ui:NewItemGridScrollView(self, tbControls.WuShuangItemTemplate, 1, 1)
  self.tbHaoJieAwardScrollView = Ui:NewItemGridScrollView(self, tbControls.HaoJieItemTemplate, 1, 1)
end

function tbWnd:OnClose()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self:UnRegistRedPoint()
end

function tbWnd:OnOpen(nOtherTeamId)
  local nPanel = self.PANEL_TEAM
  if nOtherTeamId then
    WuLinDaHui:RequestTeamInfo(nOtherTeamId)
    WuLinDaHui.tbTeamInfo = nil
    self.tbButtonGroup:SetActive(self.PANEL_TEAM, true)
    self.tbButtonGroup:SetActive(self.PANEL_HAOJIE, false)
    self.tbButtonGroup:SetActive(self.PANEL_WUSHUANG, false)
    self.tbButtonGroup:Select(nPanel)
    return
  end
  local nTeamId = WuLinDaHui:GetBattleTeamId(me.dwID)
  if nTeamId <= 0 and not WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.SIGN) and not WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.PRE) then
    self.tbButtonGroup:SetActive(self.PANEL_TEAM, false)
    if nPanel == self.PANEL_TEAM then
      nPanel = self.PANEL_WUSHUANG
    end
  else
    self.tbButtonGroup:SetActive(self.PANEL_TEAM, true)
  end
  self.tbButtonGroup:SetActive(self.PANEL_WUSHUANG, true)
  self.tbButtonGroup:SetActive(self.PANEL_HAOJIE, false)
  self.pPanel:Label_SetText(tbControls.btnWuShuang .. "/txtWuShuang", i18n.EasyGet("\230\136\152\229\140\186"))
  WuLinDaHui:RequestTeamInfo()
  self.tbButtonGroup:Select(nPanel)
  self:MarkMyWarZone()
  self:RegistRedPoint()
  self:UpdateRedPoint()
end

function tbWnd:MarkMyWarZone()
  self.pPanel:SetActive(tbControls.btnWuShuang .. "/imgMark", false)
  self.pPanel:SetActive(tbControls.btnHaoJie .. "/imgMark", false)
  if 1 == me.GetUserValue(WuLinDaHui.SAVE_GROUP, WuLinDaHui.SAVE_ELIGIBLE) then
    local nTeamWarZone = WuLinDaHui:GetWarZone()
    if nTeamWarZone then
      if nTeamWarZone == WuLinDaHui.WAR_ZONE.WUSHUANG then
        self.pPanel:SetActive(tbControls.btnWuShuang .. "/imgMark", true)
      elseif nTeamWarZone == WuLinDaHui.WAR_ZONE.HAOJIE then
        self.pPanel:SetActive(tbControls.btnHaoJie .. "/imgMark", true)
      end
    end
  end
end

function tbWnd:UpdatePanel(nPanel)
  self.nPanel = nPanel
  for i = 1, #self.tbPanelIndex2Controls do
    if i == nPanel then
      self.pPanel:SetActive(self.tbPanelIndex2Controls[i], true)
    else
      self.pPanel:SetActive(self.tbPanelIndex2Controls[i], false)
    end
  end
  if nPanel == self.PANEL_TEAM then
    self:UpdateTeamPanel()
  elseif nPanel == self.PANEL_WUSHUANG then
    self:UpdateWuShuangPanel()
  elseif nPanel == self.PANEL_HAOJIE then
    self:UpdateHaoJiePanel()
  end
end

function tbWnd:UpdateTeamPanel()
  self.tbTeamInfo = WuLinDaHui.tbTeamInfo
  if not self.tbTeamInfo then
    self.pPanel:Button_SetText(tbControls.btnJoin, i18n.EasyGet("\229\136\155\229\187\186\230\136\152\233\152\159"), "Text")
    self.pPanel:Label_SetText(tbControls.txtTeamName, i18n.EasyGet("\230\136\152\233\152\159\229\144\141\231\167\176"))
    self.pPanel:SetActive(tbControls.btnExit, true)
    self.pPanel:SetActive(tbControls.btnRecruit, true)
    self.pPanel:SetActive(tbControls.btnPartner, false)
    self.pPanel:SetActive(tbControls.btnResult, false)
    self.pPanel:SetActive(tbControls.btnAward, true)
    self.pPanel:SetActive(tbControls.btnJoin, true)
    self.pPanel:Button_SetEnable(tbControls.btnJoin, true)
  else
    self.bShowSelf = self.tbTeamInfo[WldhTeam.TEAM_ID] == WuLinDaHui:GetBattleTeamId(me.dwID)
    self.tbButtonGroup:SetButtonText(self.PANEL_TEAM, self.bShowSelf and i18n.EasyGet("\230\136\145\231\154\132\230\136\152\233\152\159") or i18n.EasyGet("\230\136\152\233\152\159\228\191\161\230\129\175"))
    self.pPanel:Button_SetText(tbControls.btnJoin, i18n.EasyGet("\229\143\130\229\138\160\230\180\187\229\138\168"), "Text")
    self.pPanel:Label_SetText(tbControls.txtTeamName, self.tbTeamInfo[WldhTeam.TEAM_NAME])
    self.pPanel:SetActive(tbControls.btnExit, self.bShowSelf)
    self.pPanel:SetActive(tbControls.btnRecruit, self.bShowSelf)
    self.pPanel:SetActive(tbControls.btnJoin, self.bShowSelf)
    self.pPanel:SetActive(tbControls.btnPartner, self.bShowSelf)
    self.pPanel:SetActive(tbControls.btnAward, self.bShowSelf)
    self.pPanel:SetActive(tbControls.btnResult, true)
    self.pPanel:Button_SetEnable(tbControls.btnJoin, WuLinDaHui.nGameState == WuLinDaHui.GAME_STATE.OPEN)
  end
  self:UpdateTeamInfo()
end

function tbWnd:UpdateTeamInfo()
  local function funSetTeammateData(tbElement, nIndex, tbInfo, bShowSelf)
    local pPanel = tbElement.pPanel
    
    if not tbInfo then
      pPanel:SetActive("imgEmpty", true)
      pPanel:SetActive("txtEmpty", true)
      pPanel:SetActive("imgModel", false)
      pPanel:SetActive("txtName", false)
      pPanel:SetActive("PortraitObj", false)
      pPanel:SetActive("HonorTitle", false)
      pPanel:SetActive("FightPower", false)
      pPanel:SetActive("txtServer", false)
      if bShowSelf then
        Ui:Button_BindEvent(pPanel, "Main", function()
          local nBattleTeamId = me.GetUserValue(WuLinDaHui.SAVE_GROUP, WuLinDaHui.SAVE_TEAMID)
          if nBattleTeamId <= 0 then
            me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\136\155\229\187\186\230\136\152\233\152\159"))
            return
          end
          if WldhTeam:ChkPlayerInMapEx(me) then
            me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\228\184\141\229\143\175\230\147\141\228\189\156\230\136\152\233\152\159"))
            return
          end
          Ui:OpenWindow("UITeamInviteWuLin", "WuLinDaHui")
        end)
      else
        Ui:Button_UnBindEvent(pPanel, "Main")
      end
      return
    end
    pPanel:SetActive("imgEmpty", false)
    pPanel:SetActive("txtEmpty", false)
    pPanel:SetActive("txtName", true)
    pPanel:SetActive("PortraitObj", true)
    pPanel:SetActive("HonorTitle", true)
    pPanel:SetActive("FightPower", true)
    pPanel:SetActive("txtServer", true)
    pPanel:Label_SetText("txtName", tbInfo.szPlayerName)
    local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
    tbPortraitObjPanel:SetRoleIcon(tbInfo.nPortrait)
    tbPortraitObjPanel:SetPortraitFrame(tbInfo.nPortraitFrame)
    tbPortraitObjPanel:SetFactionIcon(tbInfo.nFaction)
    tbPortraitObjPanel:SetLevel(tbInfo.nLevel)
    pPanel:Label_SetText("FightPower/txtFightPowerValue", tbInfo.nFight or 0)
    pPanel:Label_SetText("txtServer", tbInfo.szServerName)
    Ui:Button_UnBindEvent(pPanel, "Main")
    local szHonorName, szHonorColor = HonorLevel:GetHonorName(tbInfo.nHonorLevel, false)
    local obj = pPanel:GetObject("HonorTitle")
    local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
    pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
  end
  
  local tbMember = {}
  if self.tbTeamInfo then
    local szServerName = GetServerName(self.tbTeamInfo[WldhTeam.SERVER])
    for _, tbMemberData in pairs(self.tbTeamInfo[WldhTeam.MEMBER]) do
      tbMemberData.szServerName = szServerName
      tbMember[tbMemberData.nPosition] = tbMemberData
    end
  end
  self.tbScrollViewMember:Reset()
  for i = 1, WuLinDaHui.TeamCfg.MAX_LEAGUE_MEMBER do
    self.tbScrollViewMember:SetElement(i, function(tbElement)
      funSetTeammateData(tbElement, i, tbMember[i], self.bShowSelf)
    end)
  end
end

function tbWnd:ShowResult()
  if not self.tbTeamInfo then
    me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\230\178\161\230\156\137\230\136\152\233\152\159"))
    return
  end
  self.pPanel:SetActive(tbControls.wndResult, true)
  local nWin, nFail, nDraw, nAll = WldhTeam:BattleResultStat(self.tbTeamInfo)
  local nWinRate = 50
  if nAll > 0 then
    nWinRate = math.floor(nWin / nAll * 100)
  end
  local nNextWinRate = math.floor((nWin + 1) / (nAll + 1) * 100)
  local nNextFailRate = math.floor(nWin / (nAll + 1) * 100)
  local nScore = self.tbTeamInfo[WldhTeam.SCORE]
  local szText = i18n.EasyGet("\230\136\152\233\152\159\230\136\152\231\187\169\239\188\154{0}\232\131\156 {1}\229\185\179 {2}\232\180\159\n\230\136\152\233\152\159\232\131\156\231\142\135\239\188\154{3}%\n\228\184\139\229\156\186\232\142\183\232\131\156\229\144\142\232\131\156\231\142\135\239\188\154{4}%\n\228\184\139\229\156\186\229\164\177\232\180\165\232\131\156\231\142\135\239\188\154{5}%\n\230\175\148\232\181\155\231\167\175\229\136\134\239\188\154{6}", nWin, nDraw, nFail, nWinRate, nNextWinRate, nNextFailRate, nScore)
  self.pPanel:Label_SetText(tbControls.txtTemplate, szText)
end

function tbWnd:SendChatRecruit()
  local nBattleTeamId = me.GetUserValue(WuLinDaHui.SAVE_GROUP, WuLinDaHui.SAVE_TEAMID)
  if nBattleTeamId <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\136\155\229\187\186\230\136\152\233\152\159"))
    return
  end
  if WldhTeam:ChkPlayerInMapEx(me) then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\228\184\141\229\143\175\230\147\141\228\189\156\230\136\152\233\152\159"))
    return
  end
  local tbCondition = {
    [WuLinDaHui.emMsgSignTime] = 1,
    [WuLinDaHui.emMsgNeedBattleTeam] = 1
  }
  if not WuLinDaHui:ChkConditionAndSendMsg(me, tbCondition) then
    return false
  end
  local nCurMember = Lib:CountTB(self.tbTeamInfo[WldhTeam.MEMBER])
  if nCurMember >= WuLinDaHui.TeamCfg.MAX_LEAGUE_MEMBER then
    me.CenterMsg(i18n.EasyGet("\230\136\152\233\152\159\228\186\186\230\149\176\229\183\178\232\190\190\229\136\176\230\156\172\232\181\155\229\173\163\231\187\132\233\152\159\228\186\186\230\149\176\233\153\144\229\136\182"))
    return
  end
  Ui:CloseWindow(self.ClassName)
  Ui:OpenWindowWithCallback("UIChat", function(_, tbWnd)
    tbWnd:AddWuLinDaHuiRecruit(self.tbTeamInfo[WldhTeam.TEAM_ID], self.tbTeamInfo[WldhTeam.TEAM_NAME], nCurMember, WuLinDaHui.TeamCfg.MAX_LEAGUE_MEMBER)
  end, nil, ChatMgr.ChannelType.Public)
end

function tbWnd:UpdateSameObj(nWarZone)
  local nTeamWarZone = WuLinDaHui:GetWarZone()
  local szStageShow = WuLinDaHui:GetStageShow()
  local bShowMatching = WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.PRE) or WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.MID) or WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.FINAL)
  local szStatus = WuLinDaHui.nGameState == WuLinDaHui.GAME_STATE.OPEN and i18n.EasyGet("\230\175\148\232\181\155\228\184\173") or i18n.EasyGet("\233\151\180\230\173\135\230\156\159")
  self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/Stage/txtStageValue", szStageShow)
  self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/txtInfoList/Viewport/Content/txtInformation", i18n.Get(WuLinDaHui.UiDesc.txtInformation[nTeamWarZone]))
  self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/txtInfoList/Viewport/Content/txtInformation", i18n.Get(WuLinDaHui.UiDesc.txtInformation[nTeamWarZone]))
  self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/txtStatus", szStatus)
  self.pPanel:Button_SetEnable(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnJoin", WuLinDaHui.nGameState == WuLinDaHui.GAME_STATE.OPEN)
  self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/List/btnLive", false)
  self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/txtStatus", bShowMatching)
  if WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.MID) then
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnReport", true)
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnGuess", false)
  elseif WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.FINAL) then
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnReport", false)
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnGuess", true)
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnJoin", true)
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/List/btnLive", false)
  end
  if self.tbTeamInfo then
    self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnJoin/Text", i18n.EasyGet("\229\143\130\228\184\142"))
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnJoin", true)
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo", true)
    self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo/txtTitle", i18n.Get(WuLinDaHui.UiDesc.szTeamTitleName))
    self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo/txtName", i18n.Get(WuLinDaHui.UiDesc.szTeamName))
    self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo/txtName/txtNameValue", self.tbTeamInfo[WldhTeam.TEAM_NAME])
    self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo/txtFightNum", i18n.Get(WuLinDaHui.UiDesc.szJoinCount))
    self.pPanel:Label_SetText(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo/txtFightNum/txtNameValue", string.format("%d/%d", #self.tbTeamInfo[WldhTeam.WIN_COUNT], WuLinDaHui.PreCfg.TOTAL_COUNT))
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/List/btnPartner", true)
  else
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/Bottom/Right/btnJoin", false)
    self.pPanel:SetActive(self.tbPanelIndex2Controls[self.nPanel] .. "/TeamInfo", false)

function tbWnd:UpdateRedPoint()
  local nCurPartnerId = me.GetUserValue(WuLinDaHui.SAVE_GROUP, WuLinDaHui.SAVE_PARTNER)
  if 0 == nCurPartnerId then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.WuLinDaHuiPartner)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.WuLinDaHuiPartner)
  end
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnPartner, {
    RedPointMgr.EVENT.WuLinDaHuiPartner
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnPartner)
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_WULINDAHUI_SYNC_TEAM,
      self.UpdateTeamPanel,
      self
    },
    {
      EventNotify.emNOTIFY_WULINDAHUI_SYNC_PARTNER,
      self.UpdateRedPoint,
      self
    }
  }
end
