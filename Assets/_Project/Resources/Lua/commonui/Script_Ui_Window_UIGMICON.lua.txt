local tbWnd = Ui:CreateWindow("UIGMICON")
tbWnd.tbControls = {Button = "Button"}
tbWnd.tbOnClick = {
  Button = function(self)
    if Ui:WindowVisible("UIGM") then
      Ui:CloseWindow("UIGM")
    else
      Ui:OpenWindow("UIGM")
    end
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
end
