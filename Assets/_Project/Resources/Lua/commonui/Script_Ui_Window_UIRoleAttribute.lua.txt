Require("Script/Player/AttributeDesc.lua")
local tbWnd = Ui:CreateWindow("UIRoleAttribute")
tbWnd.tbControls = {
  btnAttrTemplate = "imgBG/panelAttrList/AttrList/AttrList/btnAttrTemplate",
  txtDesc = "imgBG/AttributeInfo/imgDesc/Text",
  imgDesc = "imgBG/AttributeInfo/imgDesc",
  AttributeInfo = "imgBG/AttributeInfo",
  btnCloseAttributeInfo = "imgBG/Empty4Raycast"
}
tbWnd.tbOnClick = {
  btnCloseAttributeInfo = function(self)
    self.pPanel:SetActive(self.AttributeInfo, false)
    self.pPanel:SetActive(self.btnCloseAttributeInfo, false)
  end
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.tbScrollView = Ui:NewScrollView(self, self.btnAttrTemplate, function(self, nIndex, tbElement)
    self.pPanel:SetActive(self.btnCloseAttributeInfo, true)
    self.pPanel:SetActive(self.AttributeInfo, true)
    self.pPanel:Label_SetText(self.txtDesc, tbOtherAttribute[nIndex].fnDesc(self.pAsyncRole, self.tbAsyncAttr))
  end, true, true)
  self.pPanel:SetActive(self.AttributeInfo, false)
  self.isMe = false
end

function tbWnd:OnOpen(dwRoleId)
  if 0 == dwRoleId then
    self.isMe = true
    self.tbAsyncAttr = nil
    self.pAsyncRole = nil
  else
    self.isMe = false
    self.pAsyncRole = KPlayer.GetAsyncData(dwRoleId)
    self.tbAsyncAttr = self.pAsyncRole.GetAsyncAttr()
  end
  self:UpdateAttr()
end

function tbWnd:OnClose()
  self.pPanel:SetActive(self.AttributeInfo, false)
end

function tbWnd:UpdateAttr()
  for i = 1, #tbOtherAttribute do
    self.tbScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtKey", tbOtherAttribute[i].fnName())
      pPanel:Label_SetText("txtValue", tbOtherAttribute[i].func(self.pAsyncRole, self.tbAsyncAttr))
      pPanel:SetActive("imgLine", tbOtherAttribute[i].bShowLine)
    end)
  end
end
