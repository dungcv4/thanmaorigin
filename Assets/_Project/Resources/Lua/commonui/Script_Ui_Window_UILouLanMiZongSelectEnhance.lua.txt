local tbWnd = Ui:CreateWindow("UILoulanSkillSelect")
local tbControls = {
  panelSkill = "imgBG/BG/Scroll View/Viewport/Content/Template",
  btnClose = "imgBG/BG/btnClose",
  imgSelect = "imgBG/BG/Scroll View/Viewport/Content/Template/Img_Select",
  btnConfirm = "imgBG/btnChange",
  btnCancel = "imgBG/btnSave",
  txtConfirm = "imgBG/btnChange/Text",
  txtCancel = "imgBG/btnSave/Text"
}
tbWnd.tbControls = tbControls
local nEnhanceNum = LouLanMiZong.CONSTANTS.nBuffCount
tbWnd.tbOnClick = {
  btnConfirm = function(self)
    self:SelEnhanceReq()
  end,
  btnCancel = function(self)
    self:Close()
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_LOULANMIZONG_SYNC_SEL_ENHANCE,
      self.OnSyncEnhanceInfo
    }
  }
end

function tbWnd:OnCreate()
  self.pPanel:Label_SetText(self.txtConfirm, i18n.EasyGet("\231\161\174\229\174\154"))
  self.pPanel:Label_SetText(self.txtCancel, i18n.EasyGet("\229\143\150\230\182\136"))
  self.tbEnhanceList = Ui:NewScrollView(self, self.panelSkill, self.OnClickSelEnhance, false, false)
end

function tbWnd:OnOpen(nNpcId)
  self.nNpcId = nNpcId
  self:ShowEnhanceInfo()
  self.nChoseIndex = 1
end

function tbWnd:OnClose()
  self.nChoseIndex = nil
end

function tbWnd:ShowEnhanceInfo()
  local nNpcId = self.nNpcId
  local tbBuffList = LouLanMiZong.tbBuffList[nNpcId]
  local tbSkillList = tbBuffList.tbSkillList
  local tbEnhanceList = LouLanMiZong.tbEnhanceInfo or {}
  self.tbEnhanceList:Clear()
  for nIndex = 1, nEnhanceNum do
    self.tbEnhanceList:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local nSkillGroup = tbSkillList[nIndex]
      local tbSkillEnhanceInfo = tbEnhanceList[nSkillGroup]
      local nSkillLevel = (tbSkillEnhanceInfo and tbSkillEnhanceInfo.nLevel or 0) + 1
      local nSkillId = LouLanMiZong.PLAYERSKILLGROUP[nSkillGroup][nSkillLevel]
      local tbSkillInfo = FightSkill:GetSkillSetting(nSkillId, nSkillLevel)
      pPanel:Label_SetText("txtName", i18n.Parse(tbSkillInfo.SkillName))
      pPanel:Label_SetText("txtInfo", i18n.Parse(tbSkillInfo.Desc))
      pPanel:Label_SetText("txtLV", i18n.EasyGet("\231\173\137\231\186\167{0}", nSkillLevel))
      pPanel:Sprite_SetSprite("SkillIcon1/btnSkill", tbSkillInfo.IconAtlas)
      pPanel:SetActive("Img_Recommend", self:IsRecommendSkill(nSkillGroup, tbEnhanceList))
      pPanel:SetActive("Img_New", 1 == nSkillLevel)
      pPanel:SetActive("Img_ArrowDown", false)
      pPanel:SetActive("Img_ArrowUp", 1 ~= nSkillLevel)
      pPanel:SetActive("Img_Select", nIndex == self.nChoseIndex)
    end)
  end
end

function tbWnd:SelEnhanceReq()
  if not self.nNpcId then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\175\185\232\177\161\228\184\141\229\173\152\229\156\168"))
    self:Close()
  end
  if not self.nChoseIndex then
    me.CenterMsg(i18n.EasyGet("\232\175\183\233\128\137\230\139\169\230\138\128\232\131\189"))
  end
  LouLanMiZong:SelEnhanceReq(self.nNpcId, self.nChoseIndex)
end

function tbWnd:OnSyncEnhanceInfo(nNpcId)
  self.nNpcId = nNpcId
  self:ShowEnhanceInfo()
end

function tbWnd:OnClickSelEnhance(nIndex)
  self.nChoseIndex = nIndex
  for nIndex = 1, nEnhanceNum do
    local tbElement = self.tbEnhanceList:GetElement(nIndex)
    local pPanel = tbElement.pPanel
    pPanel:SetActive("Img_Select", nIndex == self.nChoseIndex)
  end
end

function tbWnd:IsRecommendSkill(nSkillGroup, tbEnhanceList)
  if tbEnhanceList[nSkillGroup] then
    return false
  end
  for nOwnSkillGroup, _ in pairs(tbEnhanceList) do
    local tbRecommendList = LouLanMiZong.tbRecommendList[nOwnSkillGroup]
    if tbRecommendList and tbRecommendList[nSkillGroup] then
      return true
    end
  end
  return false
end
