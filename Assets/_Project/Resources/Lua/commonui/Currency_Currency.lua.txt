function Currency:GetIcon(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetIcon Failure!", szCurrencyType)
    return ""
  end
  return tbSetting.Icon
end

function Currency:GetName(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetName Failure!" .. szCurrencyType)
    return ""
  end
  return i18n.Parse(tbSetting.Name)
end

function Currency:GetDesc(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetDesc Failure!" .. szCurrencyType)
    return ""
  end
  return i18n.Parse(tbSetting.TipDesc)
end

function Currency:GetValue(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetValue Failure!" .. szCurrencyType)
    return 0
  end
  return tbSetting.Value
end

function Currency:GetCfgMaxValue(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetCfgMaxValue Failure!" .. szCurrencyType)
    return 0
  end
  return tbSetting.MaxValue
end

function Currency:GetSource(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetSource Failure!" .. szCurrencyType)
    return ""
  end
  return i18n.Parse(tbSetting.Source)
end

function Currency:Valid(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  return nil ~= tbSetting
end

function Currency:GetItemId(szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    LogErr("Currency:GetItemId Failure!" .. szCurrencyType)
    return ""
  end
  return tonumber(tbSetting.Param)
end
