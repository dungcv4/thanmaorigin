local tbWnd = Ui:CreateClass("UILegionDomainStat", "UIPanel")
tbWnd.tbControls = {
  goEmpty4Raycast = "Empty4Raycast",
  txtDomainInfo = "imgBg/txtDomainInfo"
}
tbWnd.tbOnClick = {
  goEmpty4Raycast = function(self)
    self:Hide()
  end
}

function tbWnd:Show()
  local tbDomainList = GlobalDomainBattle:GetDomain(me.dwLegionId)
  if 0 == #tbDomainList then
    self.pPanel:Label_SetText(self.txtDomainInfo, i18n.EasyGet("\229\189\147\229\137\141\230\156\170\229\141\160\233\162\134\233\162\134\229\156\159\239\188\140\232\175\183\229\134\155\229\155\162\233\149\191/\229\137\175\229\134\155\229\155\162\233\149\191\229\156\168\228\184\139\228\184\128\229\156\186\233\162\134\229\156\159\230\136\152\229\189\147\229\164\1694:00-20:50\233\128\137\230\139\169\233\162\134\229\156\159\229\174\163\230\136\152"))
  else
    local tbMsg = {}
    table.insert(tbMsg, i18n.EasyGet("\230\156\172\229\134\155\229\155\162\233\162\134\229\156\176\230\149\176\233\135\143\239\188\154{0}", #tbDomainList))
    local nStarLv = 0
    for _, nMapTid in pairs(tbDomainList) do
      nStarLv = nStarLv + GlobalDomainBattle.MAP_FIGHT[nMapTid].STAR_LV
    end
    table.insert(tbMsg, i18n.EasyGet("\230\156\172\229\134\155\229\155\162\233\162\134\229\156\176\230\152\159\231\186\167\239\188\154{0}", nStarLv))
    local tbBaseInfo = Legion:GetBaseInfo() or {}
    local nRank = tbBaseInfo.nDomainBattleRanking
    if nRank then
      table.insert(tbMsg, i18n.EasyGet("\229\134\155\229\155\162\230\142\146\229\144\141\239\188\154{0}", nRank))
    end
    self.pPanel:Label_SetText(self.txtDomainInfo, table.concat(tbMsg, "\n"))
  end
  self.pPanel:SetActive("Main", true)
end

function tbWnd:Hide()
  self.pPanel:SetActive("Main", false)
end
