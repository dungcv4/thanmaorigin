local tbWnd = Ui:CreateWindow("UITakePhoto")
local tbControls = {
  btnClose = "panel1/ops/btnClose",
  btnCloseSetting = "panel3/Empty4Raycast",
  panel1ops = "panel1/ops",
  btnPrepareTake = "panel1/ops/btnPrepareTake",
  btnSetting = "panel1/ops/btnSetting",
  sliderView = "panel1/ops/Slider",
  btnAdd = "panel1/ops/btnAdd",
  btnSub = "panel1/ops/btnSub",
  btnEyeOpen = "panel1/btnEyeOpen",
  btnEyeClose = "panel1/btnEyeClose",
  ToggleStranger = "panel3/imgBg/ToggleStranger",
  ToggleSelf = "panel3/imgBg/ToggleSelf",
  ToggleNpc = "panel3/imgBg/ToggleNpc",
  ToggleHeadText = "panel3/imgBg/ToggleHeadText",
  ToggleBloodBar = "panel3/imgBg/ToggleBloodBar"
}
local nMinView = 1
local nMaxView = 3.6
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnPrepareTake = function(self)
    self.pPanel:SetActive("panel1", false)
    Ui.SceneCameraController.StartCapture()
  end,
  btnSetting = function(self)
    self.pPanel:SetActive("panel3", true)
  end,
  btnEyeOpen = function(self)
    self.pPanel:SetActive(tbControls.btnEyeOpen, false)
    self.pPanel:SetActive(tbControls.btnEyeClose, true)
    self.pPanel:SetActive(tbControls.panel1ops, false)
  end,
  btnEyeClose = function(self)
    self.pPanel:SetActive(tbControls.btnEyeOpen, true)
    self.pPanel:SetActive(tbControls.btnEyeClose, false)
    self.pPanel:SetActive(tbControls.panel1ops, true)
  end,
  btnAdd = function(self)
    self.pPanel:Slider_SetValue(tbControls.sliderView, 0)
    Ui.SceneCameraController.SetSize(nMinView)
  end,
  btnSub = function(self)
    self.pPanel:Slider_SetValue(self.sliderView, 1)
    Ui.SceneCameraController.SetSize(nMaxView)
  end,
  btnCloseSetting = function(self)
    self.pPanel:SetActive("panel3", false)
  end
}
tbWnd.tbOnSliderClick = {
  sliderView = function(self, nValue)
    local fCameraSize = (nMaxView - nMinView) * nValue + 1
    Ui.SceneCameraController.SetSize(fCameraSize)
  end
}
tbWnd.tbOnToggleClick = {
  ToggleStranger = function(self, isOn)
    Client.RepresentModule.SetPlayerVisable(not isOn)
    Client.RepresentModule.SetPlayerBloodBarVisable(not isOn)
    Client.RepresentModule.SetPlayerHeadTextVisable(not isOn)
    Client.RepresentModule.SetPlayerShadowVisable(not isOn)
  end,
  ToggleSelf = function(self, isOn)
    Client.RepresentModule.SetMeVisable(not isOn)
    Client.RepresentModule.SetMeBloodBarVisable(not isOn)
    Client.RepresentModule.SetMeHeadTextVisable(not isOn)
    Client.RepresentModule.SetMeShadowVisable(not isOn)
  end,
  ToggleNpc = function(self, isOn)
    Client.RepresentModule.SetNpcVisable(not isOn)
    Client.RepresentModule.SetNpcBloodBarVisable(not isOn)
    Client.RepresentModule.SetNpcHeadTextVisable(not isOn)
    Client.RepresentModule.SetNpcShadowVisable(not isOn)
  end,
  ToggleHeadText = function(self, isOn)
    Client.RepresentModule.SetNpcHeadTextVisable(not isOn)
    Client.RepresentModule.SetPlayerHeadTextVisable(not isOn)
    Client.RepresentModule.SetMeHeadTextVisable(not isOn)
  end,
  ToggleBloodBar = function(self, isOn)
    Client.RepresentModule.SetNpcBloodBarVisable(not isOn)
    Client.RepresentModule.SetPlayerBloodBarVisable(not isOn)
    Client.RepresentModule.SetMeBloodBarVisable(not isOn)
  end
}

function tbWnd:OnCreate()
  self:SetGroup(2)
end

function tbWnd:OnOpen()
  self.pPanel:SetActive("panel3", false)
  self.pPanel:SetActive("panel1", true)
  Client.UIModule.SetGroupActive(1, false)
  self.pPanel:SetActive(tbControls.btnEyeOpen, true)
  self.pPanel:SetActive(tbControls.btnEyeClose, false)
  local fCameraSize = Ui.SceneCameraController.GetSize()
  local percent = fCameraSize / (nMaxView - nMinView + 1)
  self.pPanel:Slider_SetValue(tbControls.sliderView, percent)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleStranger, true)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleSelf, true)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleNpc, true)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleHeadText, true)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleBloodBar, true)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleStranger, false)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleSelf, false)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleNpc, false)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleHeadText, false)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleBloodBar, false)
end

function tbWnd:ShowPanel()
  self.pPanel:SetActive("panel1", true)
end

function tbWnd:OnClose()
  Client.UIModule.SetGroupActive(1, true)
  Ui:ResetCamera()
  Client.RepresentModule.SetMeVisable(true)
  Client.RepresentModule.SetMeShadowVisable(true)
  Ui:GetClass("UISetting_FightPanel"):ApplyFightSetting()
  EventNotify.OnNotify(EventNotify.emNOTIFY_SUPER_VIEW_REFRESH)
end
