local tbWnd = Ui:CreateWindow("UIAwardTips")
local AWARD_MAX_COUNT = 6
tbWnd.tbControls = {
  AwardTemplate = "imgBG/AwardContainer/UIItemGrid",
  AwardContainer = "imgBG/AwardContainer"
}

function tbWnd:OnCreate()
  self.tbItemGridList = {}
  self.UIAwardTipsFly = self.pPanel:GetObject(self.AwardContainer):GetComponent("UIAwardTipsFly")
end

function tbWnd:OnOpen(tbAward, nLogWay, szMsg, bHint)
  if not Ui.tbMisc:IsAwardTipShow() then
    Ui.tbMisc:AddAwardOpenParam(tbAward, nLogWay)
    return
  end
  local nShowNum = 0
  for _, v in ipairs(tbAward) do
    local nType = Player.AwardType[v[1]]
    if nType ~= Player.award_type_exp and (nType ~= Player.award_type_item or not Item:CheckAutoRandItem(v[2])) then
      nShowNum = nShowNum + 1
      v[4] = nil
      v[5] = nil
      if not self.tbItemGridList[nShowNum] then
        self.tbItemGridList[nShowNum] = Ui:NewExistItemGrid(self, self.AwardTemplate .. nShowNum)
      end
      local tbGrid = self.tbItemGridList[nShowNum]
      tbGrid:SetGenericItem(v)
      tbGrid:SetActive(true)
      if nShowNum >= AWARD_MAX_COUNT then
        break
      end
    end
  end
  if 0 == nShowNum then
    return
  end
  self.UIAwardTipsFly:StartAnimation(nShowNum, function()
    self:Close()
  end)
  if bHint then
    self:SendChatTip(tbAward, nLogWay, szMsg)
  end
end

function tbWnd:SendChatTip(tbAward, nLogWay, szMsg)
end

function tbWnd:OnClose()
  for i = 1, #self.tbItemGridList do
    self.pPanel:SetActive(self.AwardTemplate .. i, false)
  end
end

function tbWnd:CloseByDefault()
end
