function SuperstarWars:OpenScheduleUI(bSchedule)
  local tbCallBackList = {
    {
      SuperstarWars.SendProtocolSignUp,
      SuperstarWars
    },
    {
      SuperstarWars.SendProtocolUpdateMidRankDataS,
      SuperstarWars
    },
    {
      SuperstarWars.SendProtocolUpdateFinalDataS,
      SuperstarWars
    },
    {
      SuperstarWars.SendProtocolUpdateFinalDataS,
      SuperstarWars
    },
    {
      SuperstarWars.SendProtocolUpdateFinalDataS,
      SuperstarWars
    }
  }
  local tbCurSetting
  local tbStateStting = self.UIScheduleCfg
  for i, tbSetting in ipairs(tbStateStting) do
    tbSetting.szStateName = i18n.Get(tbSetting.szStateName)
    tbSetting.szBtnName = i18n.Get(tbSetting.szBtnName)
    if tbCallBackList[i] then
      tbSetting.tbCallBack = tbCallBackList[i]
    end
    if GetTime() >= Lib:GetConfigDate2Time(tbSetting.nStartTime) and GetTime() <= Lib:GetConfigDate2Time(tbSetting.nEndTime) then
      tbCurSetting = tbSetting
    end
  end
  if bSchedule or not tbCurSetting then
    Ui:OpenWindow("UIActivity_WuLinDaHuiSchedule", tbStateStting, nil, nil)
  else
    Lib:CallBack(tbCurSetting.tbCallBack)
  end
end

function SuperstarWars:OpenRuleUI(nBigType)
  local tbSetting = {}
  tbSetting.szTitle = i18n.Get(SuperstarWars.tbRuleAwardSetting.szUITitle)
  tbSetting.tbBigData = SuperstarWars.tbRuleAwardSetting.tbBigData
  tbSetting.tbSmallData = {}
  for _, tbRuleCfg in ipairs(SuperstarWars.tbRuleAwardSetting.tbRuleSetting) do
    tbRuleCfg.nBigType = 1
    table.insert(tbSetting.tbSmallData, tbRuleCfg)
  end
  for nIndex, tbAwardCfg in ipairs(SuperstarWars.tbRuleAwardSetting.tbAwardSetting) do
    tbAwardCfg.nBigType = 2
    local tbSingleSetting = SuperstarWars.tbRuleAwardSetting.tbAwardSingleSetting[nIndex]
    if tbSingleSetting then
      tbAwardCfg.tbWinAwardData = {
        nAwardId = tbSingleSetting[1],
        nAwardKey = tbSingleSetting[2]
      }
      tbAwardCfg.tbDefeatAwardData = {
        nAwardId = tbSingleSetting[3],
        nAwardKey = tbSingleSetting[4]
      }
    end
    tbAwardCfg.tbRankAwards = SuperstarWars.tbRuleAwardSetting.tbAwardShowSetting[nIndex]
    table.insert(tbSetting.tbSmallData, tbAwardCfg)
  end
  Ui:OpenWindow("UIActivity_WuLinDaHuiRule", tbSetting, nBigType)
end

function SuperstarWars:UpdateFinalDataC(tbFinalMatchData, tbPlayerDataList, tbIndexMapId)
  local tbDataList = {}
  for _, tbPlayerData in pairs(tbPlayerDataList) do
    local tbData = {}
    tbData.nId = tbPlayerData[self.PLAYER_DATA.PLAYER_ID]
    tbData.szName = tbPlayerData[self.PLAYER_DATA.PLAYER_NAME]
    tbData.tbMember = {
      {
        nId = tbPlayerData[self.PLAYER_DATA.PLAYER_ID],
        szPlayerName = "",
        nLevel = tbPlayerData[self.PLAYER_DATA.LEVEL],
        nFaction = tbPlayerData[self.PLAYER_DATA.FACTION],
        nPortrait = tbPlayerData[self.PLAYER_DATA.PORTRAIT],
        nFight = tbPlayerData[self.PLAYER_DATA.FIGHT]
      }
    }
    table.insert(tbDataList, tbData)
  end
  local tbSetting = {
    nGuessStartTime = SuperstarWars.GuessCfg.GUESS_START_TIME,
    nGuessEndTime = SuperstarWars.GuessCfg.GUESS_END_TIME,
    funcGetGuessId = function()
      return me.GetUserValue(SuperstarWars.SAVE_GROUP, SuperstarWars.SAVE_GUESS)
    end
  }
  local tbGuessCallBack = {
    function(nId)
      local tbSetting = {
        szTitle = i18n.Get(SuperstarWars.tbText.szGuessTitle),
        szTitle1 = i18n.EasyGet("\229\133\165\229\155\180\231\142\169\229\174\182"),
        szTitle2 = i18n.EasyGet("\231\142\169\229\174\182\228\191\161\230\129\175"),
        szTitle3 = i18n.EasyGet("\230\136\152\230\150\151\229\138\155")
      }
      local tbCallBack = {
        SuperstarWars.SendProtocolApplyGuess,
        SuperstarWars
      }
      Ui:OpenWindow("UITeamBattleGuessing", tbSetting, tbDataList, tbCallBack, nId)
    end
  }
  local tbWatchCallBack = {
    SuperstarWars.C2SWatchBattle,
    SuperstarWars
  }
  local tbId2Name = {}
  for _, tbData in ipairs(tbFinalMatchData[self.PHASE_INIT] or {}) do
    tbId2Name[tbData.nPlayerId] = tbData.szName
  end
  Ui:OpenWindow("UISuperStarCompetitionReport", tbSetting, tbGuessCallBack, tbWatchCallBack, tbFinalMatchData, tbId2Name, tbIndexMapId)
end

function SuperstarWars:OnLeave()
  if me.nTemplateMapId ~= SuperstarWars.Cfg.READY_MAP_ID and me.nTemplateMapId ~= SuperstarWars.Cfg.FIGHT_MAP_ID then
    FubenUIController:CloseFubenUI()
  end
end

function SuperstarWars:FuncExitFuben()
  if self:IsShowTipWhenQuit() then
    local szTip = i18n.EasyGet("\230\130\168\231\161\174\229\174\154\233\128\128\229\135\186\229\189\147\229\137\141\229\156\176\229\155\190\229\144\151\239\188\159")
    me.MsgBox(szTip, {
      {
        i18n.szOk,
        function()
          Dungeon:ExitDungeon()
        end
      },
      {
        i18n.szCancel
      }
    })
  else
    Dungeon:ExitDungeon()
  end
end

function SuperstarWars:IsNeedExitFubenButton()
  return self.bShowLeaveButton, {
    SuperstarWars.FuncExitFuben,
    SuperstarWars
  }
end

function SuperstarWars:IsNeedFuncScoreButton()
  return self.bShowScoreButton, {
    SuperstarWars.SendProtocolUpdateFinalDataS,
    SuperstarWars
  }
end

end
