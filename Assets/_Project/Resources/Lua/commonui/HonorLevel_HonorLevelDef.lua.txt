HonorLevel.IS_SYS_OPEN = true
HonorLevel.SELL_AWARD_ID = 901
HonorLevel.EXCHANGE_ONCE_NUM = 1
HonorLevel.nSaveGroup = GetGroupKey("HonorLevel")
HonorLevel.nSaveKeyLevel = 1

function HonorLevel:LoadSetting()
  self.LEVEL_CFG = Lib:LoadTabFileEx("Setting/HonorLevel/LevelCfg.tab", "ddsddddd", "REAL_LEVEL", {
    "REAL_LEVEL",
    "HONOR_LEVEL",
    "TIMEFRAME_REQ",
    "FIGHTPOWER_REQ",
    "ITEM_ID",
    "ITEM_NUM",
    "ATTRIB_GROUP",
    "ATTRIB_LEVEL"
  })
  self.AWARD_CFG = Lib:LoadTabFileEx("Setting/HonorLevel/AwardCfg.tab", "dsddddddddddddd", "HONOR_LEVEL", {
    "HONOR_LEVEL",
    "HONOR_NAME",
    "ITEM_ID",
    "ITEM_TIMEOUT",
    "FONT_COLOR",
    "FONT_OUTLINE_COLOR",
    "GLOW_COLOR",
    "OUTLINE_COLOR",
    "STAR_COLOR",
    "STAR_GLOW_COLOR",
    "GRADIENT1",
    "GRADIENT2",
    "STAR_GRADIENT1",
    "STAR_GRADIENT2",
    "STAR_OUTLINE_COLOR"
  })
  local tbGroupList = {}
  for nStarLevel, tbSetting in ipairs(self.LEVEL_CFG) do
    local nHonorLevel = tbSetting.HONOR_LEVEL
    if not tbGroupList[nHonorLevel] then
      tbGroupList[nHonorLevel] = {}
    end
    table.insert(tbGroupList[nHonorLevel], nStarLevel)
  end
  self.LEVEL_GROUP_CFG = SetReadOnly(tbGroupList)
  self.EXCHANGE_CFG = Lib:LoadTabFileEx("Setting/HonorLevel/ExchangeCfg.tab", "ddd", "ORIGIN_ITEM_ID", {
    "ORIGIN_ITEM_ID",
    "EXCHANGE_ITEM_ID",
    "EXCHANGE_ITEM_NUM"
  })
end

HonorLevel:LoadSetting()
