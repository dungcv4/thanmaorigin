Ui.tbMisc = {}
local tb = Ui.tbMisc

function tb:IsAwardTipShow()
  if self.bShowAwardTip == nil then
    return true
  end
  return self.bShowAwardTip
end

function tb:SetAwardTipShow(bShow)
  self.bShowAwardTip = bShow
  if bShow and self.tbShowAwardTipParam ~= nil and #self.tbShowAwardTipParam > 0 then
    Ui:OpenWindow("UIAwardTips", table.unpack(self.tbShowAwardTipParam[#self.tbShowAwardTipParam]))
    self.tbShowAwardTipParam = {}
  end
end

function tb:AddAwardOpenParam(...)
  self.tbShowAwardTipParam = self.tbShowAwardTipParam or {}
  table.insert(self.tbShowAwardTipParam, {
    ...
  })
end

function tb:GetNextAwardParam()
  if not self.tbShowAwardTipParam or 0 == Lib:CountTB(self.tbShowAwardTipParam) then
    return nil
  end
  local tbAward = table.remove(self.tbShowAwardTipParam, 1)
  return tbAward
end

function tb:ClearAwardParam()
  self.tbShowAwardTipParam = {}
end

function tb:SwitchMap(nMapTemplateId, bOpenMiniMap)
  if me.nMapTemplateId == nMapTemplateId then
    if bOpenMiniMap then
      Ui:OpenWindow("MiniMap", me.nMapTemplateId)
    end
  else
    local bOk, szMsg = Map:CanSwitchMap(me.nTotalLevel, nMapTemplateId)
    if not bOk then
      me.CenterMsg(szMsg)
      return
    end
    Map:SwitchMap(nMapTemplateId)
    return true
  end
end

function tb:DelayOnlyOnce(tbSelf, szTimerName, fnCallback, nInterval)
  tbSelf.__DelayTimer__UniqueName__ = tbSelf.__DelayTimer__UniqueName__ or {}
  if tbSelf.__DelayTimer__UniqueName__[szTimerName] ~= nil then
    return
  end
  tbSelf.__DelayTimer__UniqueName__[szTimerName] = Timer:Register(nInterval or 1, function()
    fnCallback()
    tbSelf.__DelayTimer__UniqueName__[szTimerName] = nil
  end)
end

function tb:CloseDelay(tbSelf)
  if tbSelf.__DelayTimer__UniqueName__ == nil then
    return
  end
  for k, v in pairs(tbSelf.__DelayTimer__UniqueName__) do
    Timer:Close(v)
  end
  tbSelf.__DelayTimer__UniqueName__ = nil
end
