function WuLinDaHui:RequestGuessMatch(nGuessTeamId, nGuessNum)
  local tbNowGuessCfg = WuLinDaHui:GetNowGuessCfg(WuLinDaHui.tbFinalMatchList)
  if not tbNowGuessCfg then
    return
  end
  if nGuessNum > tbNowGuessCfg.MATCH_GUESS_MAX then
    me.CenterMsg(i18n.EasyGet("\230\175\143\229\156\186\230\175\148\232\181\155\230\156\128\229\164\154\228\184\139\230\179\168{0}\228\184\170", tbNowGuessCfg.MATCH_GUESS_MAX))
    return
  end
  local nItemID = self.GuessCfg.nMatchGuessID
  local szItemName = Item:GetName(nItemID, me)
  local nNum = me.GetItemCountByTypeInBag(nItemID)
  if nGuessNum > nNum then
    me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
    return
  end
  
  local function fnAgree()
    RemoteServer.RequestWuLinDaHui("ApplyGuessMatch", nGuessTeamId, nGuessNum)
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\232\175\165\229\156\186\228\189\160\229\143\170\232\131\189\231\171\158\231\140\156\228\184\128\230\172\161\232\142\183\232\131\156\232\128\133\239\188\140\231\161\174\232\174\164\229\144\142\229\176\134\228\184\141\232\131\189\230\155\180\230\148\185\239\188\140\228\189\160\230\152\175\229\144\166\231\161\174\232\174\164\239\188\159"), {

function WuLinDaHui:UpdateHudFubenUI(tbUiData)
  if not tbUiData.nType then
    return
  end
  if not tbUiData.nCurSeconds then
    tbUiData.nCurSeconds = 0
  end
  local tbTimerParam
  local szContent = ""
  local bShowLeave = true
  if tbUiData.nType == WuLinDaHui.FUBEN_UI_TYPE.FIGHT then
    if 1 == tbUiData.nEvent then
      tbTimerParam = {
        {
          tbUiData.nCurSeconds,
          i18n.EasyGet("\231\173\137\229\190\133\233\152\182\230\174\181\n\230\175\148\232\181\155\229\141\179\229\176\134\229\188\128\229\144\175\n\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154%s")
        }
      }
    elseif 2 == tbUiData.nEvent then
      tbTimerParam = {
        {
          tbUiData.nCurSeconds,
          i18n.EasyGet("\230\136\152\230\150\151\233\152\182\230\174\181\n\230\175\148\232\181\155\228\184\173\n\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154%s")
        }
      }
      szContent = i18n.EasyGet("\229\175\185\230\150\185\233\128\160\230\136\144\228\188\164\229\174\179\233\135\143\239\188\154{0}\n\229\183\177\230\150\185\233\128\160\230\136\144\228\188\164\229\174\179\233\135\143\239\188\154{1}", tbUiData.tbDamage[2], tbUiData.tbDamage[1])
    elseif 3 == tbUiData.nEvent then
      tbTimerParam = {
        {
          tbUiData.nCurSeconds,
          i18n.EasyGet("\231\187\147\231\174\151\233\152\182\230\174\181\n\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154%s")
        }
      }
    end
    if 1 ~= me.GetUserValue(WuLinDaHui.SAVE_GROUP, WuLinDaHui.SAVE_WATCH) then
      bShowLeave = false
    end
  elseif tbUiData.nType == WuLinDaHui.FUBEN_UI_TYPE.READY then
    local szTittle = ""
    if WuLinDaHui.MISSION_READY_RULE[tbUiData.nStage][tbUiData.nEvent].DESC then
      szTittle = WuLinDaHui.MISSION_READY_RULE[tbUiData.nStage][tbUiData.nEvent].DESC()
    end
    if tbUiData.nStage == self.MATCH_STAGE.PRE then
      local tbTeamInfo = WuLinDaHui.tbTeamInfo
      if tbTeamInfo then
        szContent = i18n.Get(WuLinDaHui.CommonMsg.TIP_PRE_MAP, tbUiData.nMatchIndex, WuLinDaHui.PRE_DAILY_COUNT, tbTeamInfo[WldhTeam.DAY_JOIN], WuLinDaHui.PreCfg.DAY_COUNT, tbTeamInfo[WldhTeam.TOTAL_JOIN], WuLinDaHui.PreCfg.TOTAL_COUNT)
      end
    elseif tbUiData.nStage == self.MATCH_STAGE.MID then
      local nTeamId = self:GetBattleTeamId(me.dwID)
      if nTeamId and nTeamId > 0 then
        for nArea, tbMatchTeams in pairs(WuLinDaHui.tbMidMatchList) do
          for nAreaIdx, tbMatchTeam in pairs(tbMatchTeams) do
            if tbMatchTeam.nTeamId == nTeamId then
              szContent = i18n.EasyGet("\230\156\172\232\189\174\229\143\175\229\143\130\228\184\142\230\172\161\230\149\176\239\188\154<color=#54AB4D>{0}/{1}</color>\n\229\189\147\229\137\141\232\131\156\229\156\186\230\149\176\239\188\154<color=#54AB4D>{2}</color>\n", tbUiData.nMatchIndex, WuLinDaHui.MID_DAILY_COUNT, tbMatchTeam.nWin)
              break
            end
          end
        end
      end
    else
      local nRule = WuLinDaHui.tbWinRule[tbUiData.nPhase]
      local szRule = WuLinDaHui.tbWinRuleDesc[nRule]()
      szContent = i18n.EasyGet("\232\181\155\229\136\182\239\188\154<color=#54AB4D>{0}</color>", szRule)
      local nTeamId = self:GetBattleTeamId(me.dwID)
      if nTeamId and nTeamId > 0 then
        for i, tbData in pairs(WuLinDaHui.tbFinalMatchList[tbUiData.nPhase] or {}) do
          if tbData.nTeamId == nTeamId then
            local nEnemy = tbUiData.nPhase - i + 1
            local tbEnemyData = WuLinDaHui.tbFinalMatchList[tbUiData.nPhase][nEnemy]
            if tbEnemyData then
              szContent = i18n.EasyGet("{0}\n\229\189\147\229\137\141\229\175\185\230\137\139\239\188\154<color=#54AB4D>{1}</color>\n\230\136\145\230\150\185\228\184\142\229\175\185\230\150\185\230\175\148\229\136\134\239\188\154<color=#54AB4D>{2}:{3}</color>\n", szContent, WuLinDaHui.tbFinalMatchList[tbUiData.nPhase][nEnemy].szTeamName, tbData.nWin, WuLinDaHui.tbFinalMatchList[tbUiData.nPhase][nEnemy].nWin)
            end
          end
        end
      end
    end
    tbTimerParam = {
      {
        tbUiData.nCurSeconds,
        i18n.EasyGet("{0}\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154%s", szTittle)
      }
    }
  end
  FubenUIController:OpenDefaultFuBenUI("WuLinDaHuiHelp", tbTimerParam, {szContent})
  WuLinDaHui:SetLeaveShow(bShowLeave)
end

function WuLinDaHui:OnGetRanBoadData(tbData, nPage, nMaxLength)
  self.tbAllPreRank[nPage] = tbData
  self.tbUpdateDataTime = self.tbUpdateDataTime or {}
  self.tbUpdateDataTime[nPage] = GetTime()
  self.nMaxLength = nMaxLength
  EventNotify.OnNotify(EventNotify.emNOTIFY_WULINDAHUI_SYNC_PRERANK)
end

function WuLinDaHui:CheckUpdateData(nPage)
  if not self.tbAllPreRank or not self.tbAllPreRank then
    return
  end
  local tbData = WuLinDaHui.tbAllPreRank[nPage]
  if not tbData then
    self:RequestOpenRankBoard(nPage)
    return
  end
  local nUpdateTime = self.tbUpdateDataTime and self.tbUpdateDataTime[nPage]
  if not nUpdateTime or GetTime() - nUpdateTime > 15 then
    self:RequestOpenRankBoard(nPage)
    return
  end
  return tbData
end

function WuLinDaHui:RequestOpenRankBoard(nPage)
  if self.nRequestTimer then
    Timer:Close(self.nRequestTimer)
    self.nRequestTimer = nil
  end
  
  local function fnCall()
    RemoteServer.RequestWuLinDaHui("RequestPreRankData", nPage)
  end
  
  self.nRequestTimer = Timer:Register(9, fnCall)
end
