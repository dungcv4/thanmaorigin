local UIWindow = Ui:CreateClass("UIWindow", "UIPanel")

function UIWindow:OnAwake(view)
  self.view = view
  self:SetGroup(1)
  Ui.tbUi[self.ClassName] = self
  self:InitWindow(self.view.transform, self.ClassName)
  self:InitNotifyEvent()
end

function UIWindow:OnDestroyed()
end

function UIWindow:OnPreDestroy()
  self:UnInitNotifyEvent()
  self:UnInitPanel()
  Ui.tbUi[self.ClassName] = nil
  self.view = nil
  self.tbRegEvents = nil
end

function UIWindow:DestroyWindow(uiName)
  Ui:DestroyWindow(uiName)
end

function UIWindow:SetGroup(nGroup)
  self.view:SetGroup(nGroup)
end

function UIWindow:InitNotifyEvent()
  self.tbRegEvents = {}
  if self.NotifyEvents then
    self.tbRegEvents = self:NotifyEvents()
  end
  if self.tbRegEvents then
    for _, tbEvent in pairs(self.tbRegEvents) do
      EventNotify:RegistNotify(tbEvent[1], tbEvent[2], tbEvent[3] or self)
    end
  end
end

function UIWindow:UnInitNotifyEvent()
  if self.tbRegEvents then
    for _, tbEvent in pairs(self.tbRegEvents) do
      EventNotify:UnRegistNotify(tbEvent[1], tbEvent[3] or self)
    end
  end
  self.tbRegEvents = {}
end

function UIWindow:Visible()
  return Ui:WindowVisible(self.ClassName)
end

function UIWindow:Close()
  Ui:CloseWindow(self.ClassName)
end

function UIWindow:CloseByDefault()
  self:Close()
end
