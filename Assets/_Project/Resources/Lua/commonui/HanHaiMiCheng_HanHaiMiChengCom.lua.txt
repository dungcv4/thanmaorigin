Require("CommonScript/HanHaiMiCheng/HanHaiMiChengCfg.lua")
HanHaiMiCheng.IS_SYS_OPEN = true
HanHaiMiCheng.GAME_STATE = {
  CLOSED = 0,
  READY = 1,
  START = 2
}
HanHaiMiCheng.tbTeamState = {
  BossFight = 1,
  KeyFight = 2,
  BossCompete = 3,
  BossCompeteSucc = 4
}
HanHaiMiCheng.nProbFraction = 10000

function HanHaiMiCheng:IsBossFightMap(nMapTid)
  if self.MAPSETTING.BOSS_FIGHT_MAP.nMapId == nMapTid then
    return true
  end
  return false
end

function HanHaiMiCheng:IsKeyFightMap(nMapTid)
  if self.MAPSETTING.KEY_FIGHT_MAP.nMapId == nMapTid then
    return true
  end
  return false
end

function HanHaiMiCheng:IsGameMap(nMapTid)
  for szType, tbInfo in pairs(self.MAPSETTING) do
    if tbInfo.nMapId == nMapTid then
      return true
    end
  end
  return false
end
