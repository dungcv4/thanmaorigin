CanaryRelease.tbCalendarSetting = Lib:EasyLoadTabFile("Setting/CanaryRelease/CanaryCalendar.tab")
CanaryRelease.tbSysNotifySetting = Lib:EasyLoadTabFile("Setting/CanaryRelease/CanarySysNotify.tab")

function CanaryRelease:IsCalendarActive(nActivityID, nWeekDay)
  local szName = Calendar:GetActivityDetail(nActivityID)
  if not szName then
    return false
  end
  local nId, bMaster
  for _, tbSetting in ipairs(self.tbCalendarSetting) do
    if szName == tbSetting.szMasterName then
      nId = tbSetting.nId
      bMaster = true
      break
    elseif szName == tbSetting.szSlaveName then
      nId = tbSetting.nId
      bMaster = false
      break
    end
  end
  local bOpen = self:CheckSwitch(nId, nWeekDay)
  if bMaster and not bOpen then
    return false
  end
  if not bMaster and bOpen then
    return false
  end
  return true
end

function CanaryRelease:IsSysNotifyActive(szKey)
  local nId, bMaster
  for _, tbSetting in ipairs(self.tbSysNotifySetting) do
    if szKey == tbSetting.szKey then
      nId = tbSetting.nId
      bMaster = true
      break
    end
  end
  if bMaster and not self:CheckSwitch(nId) then
    return false
  end
  return true
end
