function Validator:Register(fn, ...)
  if not EDITOR and not MODULE_GAMESERVER then
    return
  end
  self.tbEvent = self.tbEvent or {}
  self.tbEvent[fn] = {
    fn,
    {
      ...
    }
  }
end

function Validator:CheckAll()
  if not EDITOR and not MODULE_GAMESERVER then
    return 1
  end
  local tbEventCalls = self.tbEvent
  for _, tbEventCall in pairs(tbEventCalls or {}) do
    local bSuccess, bResult = Lib:CallBack({
      tbEventCall[1],
      table.unpack(tbEventCall[2])
    })
    if not bSuccess or not bResult then
      self.bNotPass = true
    end
  end
  if self.bNotPass then
    return false
  end
  self.tbEvent = {}
  return 1
end
