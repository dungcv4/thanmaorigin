if not Npc.tbClassBase then
  Npc.tbClassBase = {}
end

  Npc.tbClass = {
    default = Npc.tbClassBase,
    [""] = Npc.tbClassBase
  }
end

  Npc.tbOnDialog = {}

function Npc:Init()
  Npc:LoadNpcFace()
end

function Npc:LoadNpcFace()
  self.tbNpcFace = Lib:LoadTabFileEx("Setting/Npc/NpcFace.tab", "dss", "FaceId", {
    "FaceId",
    "Atlas",
    "Sprite"
  })
  if not self.tbNpcFace then
    Log("Load Npc Face Failed!!!!")
    return
  end
end

function Npc:GetFace(nFaceId)
  local tbFace = self.tbNpcFace[nFaceId] or self.tbNpcFace[999]
  if not tbFace then
    return
  end
  return tbFace.Atlas, tbFace.Sprite
end

function Npc:CanProcessAI(pNpc)
  return pNpc.bCanAI or 0
end

function Npc:DragPerson(pNpcA, pNpcB)
  if not pNpcA or not pNpcB then
    return
  end
  local pPlayerB = pNpcB.GetPlayer()
  if not pPlayerB then
    return
  end
  local nMapId, nPosX, nPosY = pNpcA.GetWorldPos()
  pPlayerB.SwitchMap(nMapId, nPosX, nPosY)
end

function Npc:IsDead(nId)
  local pNpc = KNpc.GetById(nId)
  if not pNpc then
    return true
  end
  local doing = pNpc.GetAction()
  if doing == Npc.Doing.death or doing == Npc.Doing.revive then
    return true
  end
  return false
end

function Npc:IsMe(pNpc)
  if MODULE_GAMECLIENT and me and me.GetNpc() then
    return pNpc.nId == me.GetNpc().nId
  end
end

function Npc:IsAlone(pNpc)
  if MODULE_GAMESERVER then
    local pPlayer = pNpc.GetPlayer()
    if nil ~= pPlayer and pPlayer.nState == Player.emPLAYER_STATE_ALONE then
      return true
    end
  elseif 1 == IsAlone() then
    return true
  end
  return false
end

function Npc:GetClass(szClassName, bNotCreate)
  local tbClass = self.tbClass[szClassName]
  if not tbClass and 1 ~= bNotCreate then
    tbClass = Lib:NewClass(self.tbClassBase)
    self.tbClass[szClassName] = tbClass
  end
  return tbClass
end

function Npc:CallClassFunc(pNpc, szFunc, ...)
  local tbClass = self.tbClass[pNpc.szClass]
  if not tbClass then
    return
  end
  if not tbClass[szFunc] then
    return
  end
  local func = tbClass[szFunc]
  if not func then
    return
  end
  return func(tbClass, pNpc, ...)
end

function Npc:OnCreate(szClassName, szParam, pNpc)
  local tbClass = self.tbClass[szClassName]
  if not tbClass then
    return
  end
  if tbClass.OnCreate then
    GameSetting:SetGlobalObj(me, pNpc, it)
    Lib:CallBack({
      tbClass.OnCreate,
      tbClass,
      szParam
    })
    GameSetting:RestoreGlobalObj()
  end
end

Npc.tbDialogTitle = {}

local function AddOpt(tbDialogInfo, tbOpt)
  if tbOpt then
    table.insert(tbDialogInfo.OptList, tbOpt)
    return true

end

local function AddOptList(tbDialogInfo, tbOptList)

function Npc:OnDialog(szClassName, szParam)
  if not MODULE_GAMESERVER then
    return self:ShowDefaultDialog(szClassName, szParam)
  end
  local nTemplateId = him.nTemplateId
  local szDefaultText = him.szDefaultDialogInfo
  local tbDialogInfo = {
    Text = szDefaultText,
    NpcTemplateId = nTemplateId,
    OptList = {}
  }
  local bTaskFlag = self:AddTaskDialogOption(tbDialogInfo, nTemplateId)
  local nSameTaskOptionIndex
  if self:AddExtraDialogOption(tbDialogInfo, me, him) then
    bTaskFlag = true
  end
  if bTaskFlag then
    local szText = self.tbDialogTitle[him.szClass] or i18n.EasyGet("\229\133\182\229\174\131\228\186\139\230\131\133")
    tbDialogInfo.OptList[#tbDialogInfo.OptList + 1] = {
      Text = szText,
      Type = "Script",
      Callback = self.ShowDefaultDialog,
      Param = {
        self,
        szClassName,
        szParam,
        bTaskFlag
      }
    }
    Dialog:Show(tbDialogInfo, me, him, nSameTaskOptionIndex)
  elseif not self:ShowDefaultDialog(szClassName, szParam) then
    tbDialogInfo.OptList[#tbDialogInfo.OptList + 1] = {
      Text = i18n.EasyGet("\229\133\182\229\174\131\228\186\139\230\131\133")
    }
    Dialog:Show(tbDialogInfo, me, him, nSameTaskOptionIndex)
  end
end

function Npc:AddTaskDialogOption(tbDialogInfo, nTemplateId)

    bTaskFlag = true
  end

function Npc:AddExtraDialogOption(tbDialogInfo, pPlayer, pNpc)
  local tbList = self:GetExtraDialogList(pPlayer, pNpc)
  local bExtra = false
  for _, tbInfo in pairs(tbList) do
    local tbParam = {
      table.unpack(tbInfo, 4)
    }
    table.insert(tbParam, pNpc.nId)
    table.insert(tbDialogInfo.OptList, {
      Text = tbInfo[2],
      Callback = tbInfo[3],
      Param = tbParam
    })
    bExtra = true
  end
  return bExtra
end

function Npc:OnDeath(szClassName, szParam, pKiller)
  local tbClass = self.tbClass[szClassName]
  local funcCalcNpcDeath = tbClass and tbClass.CalcNpcDeath or Npc.CalcNpcDeath
  if funcCalcNpcDeath then
    local bOk, pMainNpc = Lib:CallBack({
      funcCalcNpcDeath,
      tbClass or Npc,
      him,
      pKiller
    })
    if pMainNpc then
      pKiller = pMainNpc
    end
  end
  local tbOnDeath = him.tbOnDeath
  if tbOnDeath then
    local tbCall = {
      table.unpack(tbOnDeath)
    }
    local tbArg = {pKiller}
    Lib:MergeTable(tbCall, tbArg)
    local bOK, nRet = Lib:CallBack(tbCall)
    if not bOK or 1 ~= nRet then
      him.tbOnDeath = nil
    end
  end
  if tbClass and tbClass.OnDeath then
    tbClass:OnDeath(pKiller)
  end
  if MODULE_GAMESERVER and SERVER_RUNNING_MODE ~= SERVER_MODE.ZONE_SERVER then
    Lib:CallBack({
      YiJunTask.OnKillNpc,
      YiJunTask,
      him,
      pKiller
    })
    Lib:CallBack({
      GuanFuTongJi.OnKillNpc,
      GuanFuTongJi,
      him,
      pKiller
    })
    Lib:CallBack({
      Task.OnKillNpc,
      Task,
      him,
      pKiller
    })
    Lib:CallBack({
      Task.tbLogicFieldShareNpc.OnKillNpc,
      Task.tbLogicFieldShareNpc,
      him.nId,
      pKiller
    })
  end
end

function Npc:DropAiBuffer(szBufferParams, pNpc, nMapId, nPosX, nPosY)
  local tbBufferParams = Lib:SplitStr(szBufferParams, "|")
  if not tbBufferParams then
    Log("[ERROR] Npc:DropAiBuffer error: szBufferParams not exist", szBufferParams)
    return
  end
  local szBuffId, szRate, szDis, szAngle, szDelay = table.unpack(tbBufferParams)
  if not szBuffId or not szRate then
    Log("[ERROR] Npc:DropAiBuffer error: szBufferParams error, not exist buffId", szBufferParams)
    return
  end
  local nBuffId = tonumber(szBuffId)
  local nRate = tonumber(szRate)
  local nDis = szDis and tonumber(szDis) or 0
  local nAngle = szAngle and tonumber(szAngle) or 0
  local nDelayFrame = szDelay and tonumber(szDelay)
  if nDis > 0 then
    local nRad = math.rad(nAngle)
    local nDis = Lib:Map2Logic(nDis)
    nPosX = nPosX + math.cos(nRad) * nDis
    nPosY = nPosY + math.sin(nRad) * nDis
  end
  
  local function OnTimerOut()
    Item.Obj:CallBuffer(nBuffId, nRate, nMapId, nPosX, nPosY)
  end
  
  if nDelayFrame and nDelayFrame > 0 then
    Timer:Register(nDelayFrame, OnTimerOut)
  else
    OnTimerOut()
  end
end

function Npc:CreateAiNpc(szNpcParams, pNpc, nMapId, nPosX, nPosY)
  local tbNpcParams = Lib:SplitStr(szNpcParams, "|")
  if not tbNpcParams then
    Log("[ERROR] Npc:CreateAiNpc error: szNpcParams not exist", szNpcParams, pNpc.nTemplateId)
    return
  end
  local szNpcTemplateId, szLevel, szSeries, szX, szY, szZ, szDir = table.unpack(tbNpcParams)
  local nNpcTemplateId = szNpcTemplateId and tonumber(szNpcTemplateId)
  if not nNpcTemplateId then
    Log("[ERROR] AiEvent:CallNpc error: NpcTemplateId not exist ", szNpcParams, pNpc.nTemplateId)
    return
  end
  local nLevel = szLevel and tonumber(szLevel) or -1
  local nSeries = szSeries and tonumber(szSeries) or 0
  local nX = szX and tonumber(szX) or nPosX
  local nY = szY and tonumber(szY) or nPosY
  local nDir = szDir and tonumber(szDir) or 0
  if nLevel < 0 then
    nLevel = pNpc.nLevel
  end
  return KNpc.Add(nNpcTemplateId, nLevel, nSeries, nMapId, nX, nY, 0, 0, nDir)
end

function Npc:CreateAiRandNpc(szNpcParams, pNpc, nMapId, nPosX, nPosY, nDir)
  local tbNpcParams = Lib:SplitStr(szNpcParams, "|")
  if not tbNpcParams then
    Log("[ERROR] Npc:CreateAiNpc error: szNpcParams not exist", szNpcParams, pNpc.nTemplateId)
    return
  end
  local szNpcTemplateId, szLevel, szSeries, szDis, szAngle, szDelay, szDir = table.unpack(tbNpcParams)
  local nNpcTemplateId = szNpcTemplateId and tonumber(szNpcTemplateId)
  if not nNpcTemplateId then
    Log("[ERROR] AiEvent:CallNpc error: NpcTemplateId not exist ", szNpcParams, pNpc.nTemplateId)
    return
  end
  local nLevel = szLevel and tonumber(szLevel) or -1
  local nSeries = szSeries and tonumber(szSeries) or 0
  local nDis = szDis and tonumber(szDis) or 0
  local nAngle = szAngle and tonumber(szAngle) or 0
  local nDir = szDir and tonumber(szDir) or nDir
  local nDelayFrame = szDelay and tonumber(szDelay)
  if nDis > 0 then
    local nRad = math.rad(nAngle + nDir / Env.LOGIC_MAX_DIR * 360)
    local nDis = Lib:Map2Logic(nDis)
    nPosX = nPosX + math.cos(nRad) * nDis
    nPosY = nPosY + math.sin(nRad) * nDis
  end
  if nLevel < 0 then
    nLevel = pNpc.nLevel
  end
  
  local function OnTimerOut()
    KNpc.Add(nNpcTemplateId, nLevel, nSeries, nMapId, nPosX, nPosY, 0, 0, nDir)
  end
  
  if nDelayFrame and nDelayFrame > 0 then
    Timer:Register(nDelayFrame, OnTimerOut)
  else
    OnTimerOut()
  end
end

function Npc:RegisterNpcHpPercent(pNpc, nPercent, fnCallBack, ...)
  pNpc.tbHpChangeEvent = pNpc.tbHpChangeEvent or {}
  local tbHpChangeEvent = pNpc.tbHpChangeEvent
  local nId = tbHpChangeEvent.nCurId or 1
  tbHpChangeEvent.nCurId = nId + 1
  tbHpChangeEvent.tbPercentEvent = tbHpChangeEvent.tbPercentEvent or {}
  local nIdx = 1
  for i, tbInfo in ipairs(tbHpChangeEvent.tbPercentEvent) do
    if nPercent > tbInfo.nPercent then
      break
    end
    nIdx = i + 1
  end
  table.insert(tbHpChangeEvent.tbPercentEvent, nIdx, {
    nId = nId,
    nPercent = nPercent,
    tbCallBack = {
      fnCallBack,
      ...
    }
  })
  pNpc.SetNotifyHpInfo(1)
  return nId
end

function Npc:RegisterNpcHpChange(pNpc, fnCallBack)
  pNpc.tbHpChangeEvent = pNpc.tbHpChangeEvent or {}
  local tbHpChangeEvent = pNpc.tbHpChangeEvent
  local nId = tbHpChangeEvent.nCurId or 1
  tbHpChangeEvent.nCurId = nId + 1
  tbHpChangeEvent.tbEvent = tbHpChangeEvent.tbEvent or {}
  pNpc.SetNotifyHpInfo(1)
  table.insert(tbHpChangeEvent.tbEvent, {fnCallBack = fnCallBack, nId = nId})
  return nId
end

function Npc:UnRegisterNpcHpEvent(nNpcId, nId)
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc then
    return
  end
  local tbEvent = (pNpc.tbHpChangeEvent or {}).tbEvent or {}
  local tbPercentEvent = (pNpc.tbHpChangeEvent or {}).tbPercentEvent or {}
  for nIdx, tbInfo in pairs(tbEvent) do
    if tbInfo.nId == nId then
      table.remove(tbEvent, nIdx)
      break
    end
  end
  for nIdx, tbInfo in pairs(tbPercentEvent) do
    if tbInfo.nId == nId then
      table.remove(tbPercentEvent, nIdx)
      break
    end
  end
  if 0 == #tbEvent and 0 == #tbPercentEvent then
    pNpc.SetNotifyHpInfo(0)
  end
end

function Npc:OnNpcHpChange(pNpc, nOldHp, nNewHp, nMaxHp)
  local tbEvent = (pNpc.tbHpChangeEvent or {}).tbEvent or {}
  local tbPercentEvent = (pNpc.tbHpChangeEvent or {}).tbPercentEvent or {}
  GameSetting:SetGlobalObj(me, pNpc, it)
  local nCurPercent = math.ceil(100 * nNewHp / nMaxHp)
  local tbnIdxToRemove = {}
  for nIdx, tbInfo in ipairs(tbPercentEvent or {}) do
    if nCurPercent > tbInfo.nPercent then
      break
    end
    table.insert(tbInfo.tbCallBack, tbInfo.nPercent)
    table.insert(tbInfo.tbCallBack, nCurPercent)
    Lib:CallBack(tbInfo.tbCallBack)
    table.insert(tbnIdxToRemove, 1, nIdx)
  end
  for _, nIdx in ipairs(tbnIdxToRemove) do
    table.remove(tbPercentEvent, nIdx)
  end
  for _, tbInfo in pairs(tbEvent) do
    Lib:CallBack({
      tbInfo.fnCallBack,
      nOldHp,
      nNewHp,
      nMaxHp
    })
  end
  GameSetting:RestoreGlobalObj()
end

function Npc:OnTimer(tbLisenter, func)
  local bOK, bContinue = func()
  if not bContinue then
    local npcDelay = KNpc.GetById(tbLisenter.nNpcId)
    if npcDelay then
      self:RemoveTimer(npcDelay, tbLisenter.nRegisterId)
    end
  end
  return bOK, bContinue
end

function Npc:AddTimer(pNpc, period, func)
  if not pNpc.TimerComs then
    pNpc.TimerComs = {}
  end
  local lisenter = {}
  lisenter.nNpcId = pNpc.nId
  lisenter.nRegisterId = Timer:Register(period, self.OnTimer, self, lisenter, func)
  pNpc.TimerComs[lisenter.nRegisterId] = lisenter.nRegisterId
end

function Npc:RemoveTimer(pNpc, nTimer)
  if not pNpc.TimerComs then
    return
  end
  pNpc.TimerComs[nTimer] = nil
  Timer:Close(nTimer)
end

function Npc:RemoveAllTimer(pNpc)
  if not pNpc.TimerComs then
    return
  end
  for k, v in pairs(pNpc.TimerComs) do
    Timer:Close(v)
  end
  pNpc.TimerComs = nil
end

function Npc:OnNpcDying(szClassName, pNpc)
  local tbClass = self.tbClass[szClassName]
  if not tbClass then
    return
  end
  if not tbClass.OnNpcDying then
    return
  end
  GameSetting:SetGlobalObj(me, pNpc, it)
  tbClass:OnNpcDying(pNpc)
  GameSetting:RestoreGlobalObj()
end

function Npc:IsOutOfCircle(nNpcId, nRadius, nCenterX, nCenterY)
  if not nNpcId or nNpcId <= 0 then
    return true
  end
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc then
    return true
  end
  local doing = pNpc.GetAction()
  if doing == Npc.Doing.death or doing == Npc.Doing.revive then
    return true
  end
  if not nRadius or nRadius <= 0 then
    return true
  end
  local _, nNpcX, nNpcY = pNpc.GetWorldPos()
  if not (nNpcX and not (nNpcX < 0) and nNpcY) or nNpcY < 0 then
    return true
  end
  local _, nMeX, nMeY = me.GetWorldPos()
  nCenterX = nCenterX or nMeX
  nCenterY = nCenterY or nMeY
  return nRadius < Lib:GetDistance(nNpcX, nNpcY, nCenterX, nCenterY)
end

function Npc:GetNpcPos(nNpcTemplateId, nMapTemplateId, bRandom)
  local tbMapNpcInfo = Map:GetMapNpcInfoByNpcTemplate(nMapTemplateId, nNpcTemplateId)
  if not tbMapNpcInfo then
    return
  end
  local tbNpcInfo = tbMapNpcInfo[1]
  if bRandom and #tbMapNpcInfo > 1 then
    tbNpcInfo = tbMapNpcInfo[MathRandom(#tbMapNpcInfo)]
  end
  tbNpcInfo.WalkNearLength = GetNpcDialogDistance(nNpcTemplateId)
  return tbNpcInfo.XPos, tbNpcInfo.YPos, tbNpcInfo.WalkNearLength
end
