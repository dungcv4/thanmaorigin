ExpBook.tbExpBookData = {}

function ExpBook:GiveItemExpBook(dwFriendId, nTemplateId, nGiveCount)
  local nLeftTimes = ExpBook:GetLeftSendTimes(dwFriendId)
  if nGiveCount <= nLeftTimes then
    RemoteServer.GiveItemExpBook(dwFriendId, nTemplateId, nGiveCount)
  else
    me.CenterMsg(i18n.EasyGet("\229\137\169\228\189\153\232\181\160\233\128\129\230\172\161\230\149\176\228\184\141\232\182\179!"))
  end
end

function ExpBook:OnSynExpBookData(tbData, nUpdatePlayerID)
  self.tbExpBookData = tbData or {}
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYN_EXPBOOK_DATA_SINGGLE, nUpdatePlayerID)
end

function ExpBook:GetLeftSendTimes(nPlayerId)
  return self.tbExpBookData[nPlayerId] or ExpBook.Cfg.MAXSENDTIMES
end

function ExpBook:OnSynAllExpBookData(tbData)
  self.tbExpBookData = tbData or {}
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYN_EXPBOOK_DATA_FINISH)
end

function ExpBook:RequestUpdateExpBook()
  RemoteServer.UpdateExpBook()
end

function ExpBook:C2SExpBookExpInfo(dwId)
  RemoteServer.C2SExpBookExpInfo(dwId)
end
