local tbAllGroups = {
  "NoAdd",
  "AddTwo",
  "AddFour",
  "AddSix"
}
tbWnd.tbControls = {
  ButtonListView = "MainPanel/btnListScrollView",
  ButtonTemplate = "MainPanel/btnListScrollView/Viewport/Content/btnTemplate",
  mainPanel = "MainPanel"
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.mainPanelTransform = self.pPanel:GetRectTransform(self.mainPanel)
  self.tbScrollView = Ui:NewScrollView(self, self.ButtonTemplate)
end

function tbWnd:OnOpen(szOpenType, tbCustomSetting)
  if not szOpenType and (not tbCustomSetting or Lib:IsEmptyTB(tbCustomSetting)) then
    return
  end
  self.tbScrollView:Clear()
  if tbCustomSetting then
    self:UpdateView(tbCustomSetting)
  else
    self.szOpenType = szOpenType
    for i = 1, Lib:CountTB(tbAllGroups) do
      self.tbScrollView:SetElement(i, function(tbElement)
        local setting = tbButtonSetting[tbAllGroups[i]]
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("Btn/Text", setting.fnName())
        Ui:Button_BindEvent(pPanel, "Btn", setting.fnOnClick, self)
      end)
    end
  end
  self:SetAnchoredPos(750, -303)
end

function tbWnd:SetPos(x, y)
  local transform = self.pPanel:GetTransform(self.mainPanel)
  transform.position = Ui.Vector3(x, y, transform.position.z)
end

function tbWnd:SetAnchoredPos(anchoredPosX, anchoredPosY)
  self.mainPanelTransform.anchoredPosition3D = Ui.Vector3(anchoredPosX, anchoredPosY, 0)
end

function tbWnd:OnClose()
end

function tbWnd:SendParam(nMultiNum)
  if self.szOpenType == "Tong" then
    local UITong = Ui:GetWindow("UITong")
    if UITong then
      UITong:SetYuanbaoMuitl(nMultiNum)
    end
    self:Close()
  else
    local UIKin = Ui:GetWindow("UIKin")
    if UIKin then
      UIKin:SetYuanbaoMuitl(nMultiNum)
    end
    self:Close()
  end
end

function tbWnd:UpdateView(tbCustomSetting)
  for nIndex, tbSetting in ipairs(tbCustomSetting) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("Btn/Text", tbSetting.szName)
      if tbSetting.bGrey then
        pPanel:Button_SetEnable("Btn", false)
        pPanel:Sprite_SetSprite("Btn", "UI/atlas/common_btn/btn_common1_1.png")
      elseif tbSetting.bSelected then
        pPanel:Button_SetEnable("Btn", true)
        pPanel:Sprite_SetSprite("Btn", "UI/atlas/common_btn/btn_common1_3.png")
      else
        pPanel:Button_SetEnable("Btn", true)
        pPanel:Sprite_SetSprite("Btn", "UI/atlas/common_btn/btn_common1_1.png")
      end
      Ui:Button_BindEvent(pPanel, "Btn", function()
        tbSetting.fnOnClick(table.unpack(tbSetting.tbParam or {}))
        if not tbSetting.bNotClose then
          self:Close()
        end
      end)
    end)
  end
end
