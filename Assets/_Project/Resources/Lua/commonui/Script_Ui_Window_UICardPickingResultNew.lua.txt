local tbWnd = Ui:CreateWindow("UICardPickingResultNew")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.UpdateBottomUI
    }
  }
end

  btnShare = "imgBG/bottom/btnShare"

tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnBG = function(self)
    if self.bFirstClick then
      self:ShowAllResult()
      self.bFirstClick = false
      return
    end
    self:Close()
    if self.szType ~= "PartnerXinWu" then
      if self.cardPickType == "Coin" then
        Ui:OpenWindow(Partner.szCardPickingUIClassName, 1)
      elseif self.cardPickType == "Gold" then
        Ui:OpenWindow(Partner.szCardPickingUIClassName, 2)
      elseif self.cardPickType == "Limited" then
        Ui:OpenWindow(Partner.szCardPickingUIClassName, 3)
      end
    end
  end,
  btnPick = function(self)
    if self.cardPickType == "Coin" then
      CardPicker:RequestCardPick("CoinPick")
    elseif self.cardPickType == "Gold" then
      CardPicker:RequestCardPick("GoldPick")
    elseif self.cardPickType == "Limited" then
      CardPicker:RequestCardPick("LimitedPick")
    end
  end,
  btnTenPick = function(self)
    if self.cardPickType == "Coin" then
      if not CardPicker:CheckCardPickMoney("CoinTenPick") then
        return
      end
      local nLastTime = LocalData:GetRoleNumber("Partner_CloseCoinCardPickerNoticeTime", 0)
      local bIsDiffDay = Lib:IsDiffDay(14400, nLastTime)
      if not bIsDiffDay then
        CardPicker:RequestCardPick("CoinTenPick")
      else
        local function fnCardPick()
          CardPicker:RequestCardPick("CoinTenPick")
        end
        
        Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\231\161\174\232\174\164\229\141\129\232\191\158\230\139\155\229\139\159\229\144\140\228\188\180\239\188\159"), {
          {fnCardPick},
          {}
        }, {
          i18n.szOk,
          i18n.szCancel
        }, "CoinCardPicker", nil, i18n.EasyGet("\228\187\138\230\151\165\228\184\141\229\134\141\230\143\144\233\134\146"))
      end
    elseif self.cardPickType == "Gold" then
      if not CardPicker:CheckCardPickMoney("GoldTenPick") then
        return
      end
      local nLastTime = LocalData:GetRoleNumber("Partner_CloseGoldCardPickerNoticeTime", 0)
      local bIsDiffDay = Lib:IsDiffDay(14400, nLastTime)
      if not bIsDiffDay then
        CardPicker:RequestCardPick("GoldTenPick")
      else
        local function fnCardPick()
          CardPicker:RequestCardPick("GoldTenPick")
        end
        
        Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\231\161\174\232\174\164\229\141\129\232\191\158\230\139\155\229\139\159\229\144\140\228\188\180\239\188\159"), {
          {fnCardPick},
          {}
        }, {
          i18n.szOk,
          i18n.szCancel
        }, "GoldCardPicker", nil, i18n.EasyGet("\228\187\138\230\151\165\228\184\141\229\134\141\230\143\144\233\134\146"))
      end
    elseif self.cardPickType == "Limited" then
      if not CardPicker:CheckCardPickMoney("LimitedTenPick") then
        return
      end
      local nLastTime = LocalData:GetRoleNumber("Partner_CloseLimitedCardPickerNoticeTime", 0)
      local bIsDiffDay = Lib:IsDiffDay(14400, nLastTime)
      if not bIsDiffDay then
        CardPicker:RequestCardPick("LimitedTenPick")
      else
        local function fnCardPick()
          CardPicker:RequestCardPick("LimitedTenPick")
        end
        
        Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\231\161\174\232\174\164\229\141\129\232\191\158\230\139\155\229\139\159\229\144\140\228\188\180\239\188\159"), {
          {fnCardPick},
          {}
        }, {
          i18n.szOk,
          i18n.szCancel
        }, "LimitedCardPicker", nil, i18n.EasyGet("\228\187\138\230\151\165\228\184\141\229\134\141\230\143\144\233\134\146"))
      end
    end
  end
}

function tbWnd:OnCreate()
  self.tbOneNpcModel = nil
  self.tbTenNpcModel = {}
  self.tbOneItemGrid = nil
  self.tbTenItemGrid = {}
end

local tbQuality2Grade = {
  [1] = "SSS",
  [2] = "SS",
  [3] = "S",
  [4] = "A",
  [5] = "B"
}

function tbWnd:OnOpen(tbItems, szType)
  local szPickItemIconPath = "UI/icon/item/partner/partnerpick.png"
  local szCoinIconPath = "UI/icon/money/icon_coin.png"
  local szYuanbaoIconPath = "UI/icon/money/icon_yuanbao.png"
  local szLimitedPickItemIconPath = "UI/icon/item/partner/partnerpick_Gold.png"
  self.pPanel:SetActive(tbControls.btnShare, false)
  for _, szQuality in pairs(tbQuality2Grade) do
    self.pPanel:SetActive(string.format(tbControls.OneCardGrade, szQuality), false)
    self.pPanel:SetActive(string.format(tbControls.TenCardGrade, 1, szQuality), false)
  end
  self.tbItems = tbItems
  self.szType = szType
  self.cardPickType = nil
  if 1 == Lib:CountTB(tbItems) then
    for _, szQuality in pairs(tbQuality2Grade) do
      self.pPanel:SetActive(string.format(tbControls.OneCardGrade, szQuality), false)
    end
    local obj = self.pPanel:GetObject(tbControls.NpcOneCardModelPath)
    self.tbOneNpcModel = obj:GetComponent(Ui.UINpcModel)
    self.tbOneItemGrid = Ui:NewExistItemGrid(self, tbControls.OneCardItemGrid)
    self.pPanel:SetActive(tbControls.goOneCard, true)
    self.pPanel:SetActive(tbControls.goTenCard, false)
    local value = 5
    if tbItems[1].szItemType == "Partner" then
      value = Partner.tbAllPartnerInfo[tbItems[1].nTemplateId].nQualityLevel
    end
    local szPartnerQuailty = tbQuality2Grade[value] or "SSS"
    local pMaterial = self.pPanel:GetObject(tbCardMaterialList[value]):GetComponent(Ui.MeshRenderer).materials
    self.pPanel:GetObject(tbControls.OneCardItem .. "/Card"):GetComponent(Ui.MeshRenderer).materials = pMaterial
    local szPath = string.format(tbControls.txtOneCardName, szPartnerQuailty)
    local szGradePath = string.format(tbControls.OneCardGrade, szPartnerQuailty)
    for key, value in pairs(tbQuality2Grade) do
      self.pPanel:SetActive(string.format(tbControls.OneCardGrade, value), false)
    end
    self.pPanel:SetActive(tbControls.OneCardBook, false)
    self.pPanel:SetActive(szGradePath, true)
    self.pPanel:Label_SetText(szPath, i18n.Parse(tbItems[1].szName))
    if tbItems[1].szItemType == "Partner" then
      self.pPanel:SetActive(tbControls.OneCardItemGrid, false)
      local nNpcResId = KNpc.GetNpcTemplateInfo(tbItems[1].nItemId).nNpcResID
      self.tbOneNpcModel:Release()
      self.tbOneNpcModel:ShowNpc(nNpcResId)
      if Partner.tbPartnerScaleInfo[nNpcResId].Scale ~= "" then
        self.tbOneNpcModel.fModelScale = 125 * Partner.tbPartnerScaleInfo[nNpcResId].Scale
      else
        self.tbOneNpcModel.fModelScale = 125
      end
    else
      self.pPanel:SetActive(tbControls.OneCardItemGrid, true)
      self.tbOneNpcModel:Release()
      self.tbOneNpcModel:ShowNpc(1)
      self.pPanel:SetActive(tbControls.OneCardBook, true)
      self.tbOneItemGrid:SetItemByTemplate(tbItems[1].nItemId, tbItems[1].nItemNum)
      self.tbOneItemGrid:SetClickEnable(true)
    end
  elseif 10 == Lib:CountTB(tbItems) then
    self.pPanel:SetActive(tbControls.goOneCard, false)
    self.pPanel:SetActive(tbControls.goTenCard, true)
    self:StartTimer()
  end
  if "CoinPick" == szType or "CoinTenPick" == szType then
    self.cardPickType = "Coin"
    self.pPanel:SetActive(tbControls.goBottom, true)
    self.pPanel:Label_SetText(tbControls.txtTip2, "")
    self.pPanel:Label_SetText(tbControls.txtTenPickValue, CardPicker.Def.nCoinTenCost)
    self.pPanel:Label_SetText(tbControls.txtPickValue, CardPicker.Def.nCoinCost)
    self.pPanel:Sprite_SetSprite(tbControls.imgIconTenPick, szCoinIconPath)
    self.pPanel:Sprite_SetSprite(tbControls.imgIconPick, szCoinIconPath)
  elseif "GoldPick" == szType or "GoldTenPick" == szType then
    self.cardPickType = "Gold"
    self.pPanel:SetActive(tbControls.goBottom, true)
    self.nSSCount = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_CALC_SUBSIDY_SS_COUNT)
    local nSSCount = CardPicker.Def.nCardPickMustSSCount - self.nSSCount
    self.pPanel:Label_SetText(tbControls.txtTip2, i18n.EasyGet("\229\134\141\230\139\155\229\139\159<color=#C5B505>{0}</color>\230\172\161\229\191\133\229\190\151<color=#C56D05>\230\158\129\229\147\129</color>", nSSCount))
    local nItemCount = me.GetItemCountInBags(CardPicker.Def.nPartnerPickItemId)
    if nItemCount > 0 then
      self.pPanel:Label_SetText(tbControls.txtPickValue, 1)
      self.pPanel:Sprite_SetSprite(tbControls.imgIconTenPick, szPickItemIconPath)
      self.pPanel:Sprite_SetSprite(tbControls.imgIconPick, szPickItemIconPath)
      if nItemCount > 0 and nItemCount < 10 then
        self.pPanel:Label_SetText(tbControls.txtTenPickValue, string.format("%d/10", nItemCount))
      else
        self.pPanel:Label_SetText(tbControls.txtTenPickValue, 10)
      end
    else
      self.pPanel:Label_SetText(tbControls.txtTenPickValue, CardPicker.Def.nGoldTenCost)
      self.pPanel:Label_SetText(tbControls.txtPickValue, CardPicker.Def.nGoldCost)
      self.pPanel:Sprite_SetSprite(tbControls.imgIconTenPick, szYuanbaoIconPath)
      self.pPanel:Sprite_SetSprite(tbControls.imgIconPick, szYuanbaoIconPath)
    end
  elseif "LimitedPick" == szType or "LimitedTenPick" == szType then
    self.cardPickType = "Limited"
    self.pPanel:SetActive(tbControls.goBottom, true)
    self.pPanel:Label_SetText(tbControls.txtTip2, "")
    self.pPanel:Label_SetText(tbControls.txtTenPickValue, CardPicker.Def.nLimitedTenCost)
    self.pPanel:Label_SetText(tbControls.txtPickValue, CardPicker.Def.nLimitedCost)
    self.pPanel:Sprite_SetSprite(tbControls.imgIconTenPick, szLimitedPickItemIconPath)
    self.pPanel:Sprite_SetSprite(tbControls.imgIconPick, szLimitedPickItemIconPath)
    self.nLimitedSSSCount = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_CALC_SUBSIDY_SSS_COUNT)
    self.nLimitedSSCount = me.GetUserValue(CardPicker.Def.CARD_SAVE_GROUP, CardPicker.Def.SAVE_CALC_SUBSIDY_LIMITED_SS_COUNT)
    local nSSSCount = CardPicker.Def.nCardPickMustSSSCount - self.nLimitedSSSCount
    local nSSCount = CardPicker.Def.nCardPickMustSSCount - self.nLimitedSSCount
    self.pPanel:Label_SetText(tbControls.txtTip2, i18n.EasyGet("\229\134\141\230\139\155\229\139\159<color=#C5B505>{0}</color>\230\172\161\229\191\133\229\190\151<color=#C5B505>\231\187\157\229\147\129</color>", nSSSCount))
  elseif "PartnerXinWu" == szType then
    self.cardPickType = ""
    self.pPanel:SetActive(tbControls.goBottom, false)
  end
end

function tbWnd:UpdateBottomUI()
  local szPickItemIconPath = "UI/icon/item/partner/partnerpick.png"
  local szCoinIconPath = "UI/icon/money/icon_coin.png"
  local szYuanbaoIconPath = "UI/icon/money/icon_yuanbao.png"
  if self.szType == "GoldPick" or self.szType == "GoldTenPick" then
    local nItemCount = me.GetItemCountInBags(CardPicker.Def.nPartnerPickItemId)
    if nItemCount > 0 then
      self.pPanel:Label_SetText(tbControls.txtPickValue, 1)
      self.pPanel:Sprite_SetSprite(tbControls.imgIconTenPick, szPickItemIconPath)
      self.pPanel:Sprite_SetSprite(tbControls.imgIconPick, szPickItemIconPath)
      if nItemCount > 0 and nItemCount < 10 then
        self.pPanel:Label_SetText(tbControls.txtTenPickValue, string.format("%d/10", nItemCount))
      else
        self.pPanel:Label_SetText(tbControls.txtTenPickValue, 10)
      end
    end
  end
end

function tbWnd:OnClose()
  self.bFirstClick = false
  Partner:OnSyncPartnerUpdate(nil, "All")
  self:ReleaseAllNpcModel()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.nIndex = nil
end

function tbWnd:OnDestroy()
end

function tbWnd:ReleaseAllNpcModel()
  if self.tbOneNpcModel ~= nil and self.tbOneNpcModel ~= {} then
    self.tbOneNpcModel:Release()
  end
  for k, v in pairs(self.tbTenNpcModel) do
    v:Release()
  end
end

local nDelayTimeFrame = 5

function tbWnd:StartTimer()
  for i = 1, 10 do
    local szActKey = "Item" .. i
    local szNewKey = self.pPanel:IsObjectExist(string.format(tbControls.TenCardItem, i)) and string.format(tbControls.TenCardItem, i) or self.pPanel:CloneObject("imgBG/TenCard/Item1", szActKey)
    self.pPanel:SetActive(string.format(tbControls.TenCardItem, i), true)
    self.pPanel:SetActive(string.format(tbControls.TenCardBook, i), false)
    self.pPanel:Animator_SetEnable(string.format(tbControls.animTenCardAnimation, i), true)
    local Animator = self.pPanel:GetAnimator(string.format(tbControls.TenCardItem, i))
    Animator:Play("cardOuto", 0, 0)
    local obj = self.pPanel:GetObject(string.format(tbControls.NpcTenCardModelPath, i))
    local NpcModel = obj:GetComponent(Ui.UINpcModel)
    table.insert(self.tbTenNpcModel, NpcModel)
    local grid = Ui:NewExistItemGrid(self, string.format(tbControls.TenCardItemGrid, i))
    table.insert(self.tbTenItemGrid, grid)
    for _, szQuality in pairs(tbQuality2Grade) do
      self.pPanel:SetActive(string.format(tbControls.TenCardGrade, i, szQuality), false)
    end
  end
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self:ShowOneResult(1, self.tbItems[1])
  self.nIndex = 2
  self.bFirstClick = true
  self.nTimer = Timer:Register(nDelayTimeFrame, self.OnTimer, self)
end

function tbWnd:ShowOneResult(nIndex, tbInfo)
  self.pPanel:SetActive(string.format(tbControls.TenCardItem, nIndex), true)
  self.pPanel:Animator_SetEnable(string.format(tbControls.animTenCardAnimation, nIndex), true)
  local Animator = self.pPanel:GetAnimator(string.format(tbControls.TenCardItem, nIndex))
  Animator:Play("cardInto", 0, 0)
  local value = 5
  if tbInfo.szItemType == "Partner" then
    value = Partner.tbAllPartnerInfo[tbInfo.nTemplateId].nQualityLevel
  end
  local szPartnerQuailty = tbQuality2Grade[value] or "SSS"
  local pMaterial = self.pPanel:GetObject(tbCardMaterialList[value]):GetComponent(Ui.MeshRenderer).materials
  self.pPanel:GetObject(string.format(tbControls.TenCardItem, nIndex) .. "/Card"):GetComponent(Ui.MeshRenderer).materials = pMaterial
  local szPath = string.format(tbControls.txtTenCardName, nIndex, szPartnerQuailty)
  local szGradePath = string.format(tbControls.TenCardGrade, nIndex, szPartnerQuailty)
  for key, value in pairs(tbQuality2Grade) do
    self.pPanel:SetActive(string.format(tbControls.TenCardGrade, nIndex, value), false)
  end
  self.pPanel:SetActive(szGradePath, true)
  self.pPanel:Label_SetText(szPath, i18n.Parse(tbInfo.szName))
  if tbInfo.szItemType == "Partner" then
    self.pPanel:SetActive(string.format(tbControls.TenCardItemGrid, nIndex), false)
    local nNpcResId = KNpc.GetNpcTemplateInfo(tbInfo.nItemId).nNpcResID
    self.tbTenNpcModel[nIndex]:Release()
    self.tbTenNpcModel[nIndex]:ShowNpc(nNpcResId)
    if Partner.tbPartnerScaleInfo[nNpcResId].Scale ~= "" then
      self.tbTenNpcModel[nIndex].fModelScale = 125 * Partner.tbPartnerScaleInfo[nNpcResId].Scale
    else
      self.tbTenNpcModel[nIndex].fModelScale = 125
    end
  else
    self.pPanel:SetActive(string.format(tbControls.TenCardItemGrid, nIndex), true)
    self.tbTenNpcModel[nIndex]:Release()
    self.tbTenNpcModel[nIndex]:ShowNpc(1)
    self.pPanel:SetActive(string.format(tbControls.TenCardBook, nIndex), true)
    self.tbTenItemGrid[nIndex]:SetItemByTemplate(tbInfo.nItemId, tbInfo.nItemNum)
    self.tbTenItemGrid[nIndex]:SetClickEnable(true)
  end
end

function tbWnd:OnTimer()
  if self.nIndex > 10 then
    self.nIndex = 2
    self.nTimer = nil
    self.bFirstClick = false
    return false
  end
  self:ShowOneResult(self.nIndex, self.tbItems[self.nIndex])
  self.nIndex = self.nIndex + 1
  return true
end

function tbWnd:ShowAllResult()
  if not self.nTimer then
    return
  end
  Timer:Close(self.nTimer)
  if self.nIndex > 10 then
    return
  end
  for i = self.nIndex, 10 do
    self:ShowOneResult(i, self.tbItems[i])
  end
end
