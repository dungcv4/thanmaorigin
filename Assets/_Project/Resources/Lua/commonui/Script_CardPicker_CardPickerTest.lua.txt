function CardPicker:RequestCardPickTenThousandTime(nTime, szPickType)
  if self.nTestCardPickerTimer ~= nil then
    Timer:Close(self.nTestCardPickerTimer)
    
    self.nTestCardPickerTimer = nil
  end
  self.nTestPickCount = 0
  self.nTestCardPickerTimer = Timer:Register(Env.GAME_FPS * 3, self.RequestCardPickTenThousandTime2, self, nTime, szPickType)
end

function CardPicker:RequestCardPickTenThousandTime2(nTime, szPickType)
  RemoteServer.OnCardPickerRequest(szPickType)
  self.nTestPickCount = self.nTestPickCount + 1
  if nTime <= self.nTestPickCount then
    return false
  else
    return true
  end
end
