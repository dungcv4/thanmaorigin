function Legion:LoadLegionLevelSetting()
  local szLegionCountSettingPath = "Setting/Legion/LevelSetting.tab"
  
  local tbSetting = tbSetting or Lib:LoadTabFileEx(szLegionCountSettingPath, "dddd", "Level", {
    "Level",
    "MaxMember",
    "MaxMaster",
    "MaxViceMaster"
  })
  self.tbLegionLevelSetting = tbSetting
end

Legion:LoadLegionLevelSetting()
local CareerSettingNameMap = {
  [Legion.Def.Career_Master] = "MaxMaster",
  [Legion.Def.Career_ViceMaster] = "MaxViceMaster",
  [Legion.Def.Career_Normal] = "MaxMember",
  MaxMember = "MaxMember"
}

function Legion:GetMaxMember(nLevel, itemKey)
  local szItemName = assert(CareerSettingNameMap[itemKey])
  return self.tbLegionLevelSetting[nLevel][szItemName]
end

function Legion:IsMaster()
  local tbMember = Legion:GetMemberData(me.dwID)
  if tbMember and tbMember.nCareer == Legion.Def.Career_Master then
    return true
  end
  return false
end

function Legion:IsViceMaster()
  local tbMember = Legion:GetMemberData(me.dwID)
  if tbMember and tbMember.nCareer == Legion.Def.Career_ViceMaster then
    return true
  end
  return false
end

function Legion:IsMasterOrViceMaster()
  return Legion:IsMaster() or Legion:IsViceMaster()
end

function Legion:CheckMailContent(szContent)
  if not szContent or "" == szContent then
    return false, Legion.Def.tbMsgCode.PleaseEnterContent
  end
  if "" ~= szContent and ReplaceLimitWords(szContent) then
    return false, Legion.Def.tbMsgCode.IllegalChar
  end
  if Lib:StrLen(szContent) > Legion.Cfg.nMaxMailLength then
    return false, Legion.Def.tbMsgCode.OutOfLengthLimit
  end
  return true
end
