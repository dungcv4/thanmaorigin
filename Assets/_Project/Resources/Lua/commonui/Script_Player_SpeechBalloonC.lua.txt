SpeechBalloon.tbRedPoint = {}

function SpeechBalloon:ChangeReq(nSpeechBalloonId)
  local bRet, szRet = self:CanChange(nSpeechBalloonId)
  if not bRet then
    if szRet then
      me.CenterMsg(szRet)
    end
    return
  end
  RemoteServer.RequestSpeechBalloon("ChangeReq", nSpeechBalloonId)
end

function SpeechBalloon:CanChange(nSpeechBalloonId)
  if not self:CheckId(nSpeechBalloonId) then
    return false, i18n.EasyGet("\232\129\138\229\164\169\230\176\148\230\179\161\228\184\141\229\173\152\229\156\168\239\188\129")
  end
  if not self:CheckValid(me, nSpeechBalloonId, self.tbData) then
    return false, i18n.EasyGet("\232\129\138\229\164\169\230\176\148\230\179\161\230\156\170\232\167\163\233\148\129\239\188\129")
  end
  return true
end

function SpeechBalloon:SyncDataReq()
  RemoteServer.RequestSpeechBalloon("SyncDataReq")
end

function SpeechBalloon:SyncData(tbData, nId, bNewSpeechBalloon)
  self.tbData = tbData
  if bNewSpeechBalloon then
    SpeechBalloon.tbRedPoint[nId] = true
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.NewSpeechBalloon)
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_SPEECH_BALLOON_CHANGE)
end

function SpeechBalloon:SetSpeechBalloon(pPanel, szPath, nIndex)
  self:SetDefaultSpeechBalloon(pPanel, szPath)
  if not nIndex then
    return
  end
  local tbInfo = self:GetInfoById(nIndex)
  if not tbInfo then
    return
  end
  pPanel:Sprite_SetSprite(szPath, "UI/atlas/UIChatBox/" .. tbInfo.Bg)
  if tbInfo.Center and tbInfo.Center ~= "" then
    pPanel:SetActive(szPath .. "/Img_Moon", true)
    pPanel:Sprite_SetSprite(szPath .. "/Img_Moon", "UI/atlas/UIChatBox/" .. tbInfo.Center)
  end
  if tbInfo.UL and "" ~= tbInfo.UL then
    pPanel:SetActive(szPath .. "/Img_UpLeft", true)
    pPanel:Sprite_SetSprite(szPath .. "/Img_UpLeft", "UI/atlas/UIChatBox/" .. tbInfo.UL)
  end
  if tbInfo.UR and "" ~= tbInfo.UR then
    pPanel:SetActive(szPath .. "/Img_UpRight", true)
    pPanel:Sprite_SetSprite(szPath .. "/Img_UpRight", "UI/atlas/UIChatBox/" .. tbInfo.UR)
  end
  if tbInfo.DL and "" ~= tbInfo.DL then
    pPanel:SetActive(szPath .. "/Img_DownLeft", true)
    pPanel:Sprite_SetSprite(szPath .. "/Img_DownLeft", "UI/atlas/UIChatBox/" .. tbInfo.DL)
  end
  if tbInfo.DR and "" ~= tbInfo.DR then
    pPanel:SetActive(szPath .. "/Img_DownRight", true)
    pPanel:Sprite_SetSprite(szPath .. "/Img_DownRight", "UI/atlas/UIChatBox/" .. tbInfo.DR)
  end
end

function SpeechBalloon:SetDefaultSpeechBalloon(pPanel, szPath)
  pPanel:Sprite_SetSprite(szPath, "UI/atlas/chat/img_chatbg_R.png")
  pPanel:SetActive(szPath .. "/Img_Moon", false)
  pPanel:SetActive(szPath .. "/Img_UpLeft", false)
  pPanel:SetActive(szPath .. "/Img_UpRight", false)
  pPanel:SetActive(szPath .. "/Img_DownLeft", false)
  pPanel:SetActive(szPath .. "/Img_DownRight", false)
end

function SpeechBalloon:ChangeSpeechBalloon()
  EventNotify.OnNotify(EventNotify.emNOTIFY_SPEECH_BALLOON_CHANGE)
end
