local tbWnd = Ui:CreateClass("UIDuanWuOutputInfo", "UIPanel")
local tbControls = {
  txtOutputTarget = "Info/txtOutputTarget",
  OutputTemplate = "Info/OutputGroup/Viewport/Content/OutputTemplate"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbOutputGroup = {}
  self.tbScrollView = Ui:NewScrollView(self, tbControls.OutputTemplate, nil, false, true)
end

function tbWnd:OnActivate()
  self.nLastTime = 0
  self.pPanel:Label_SetText(tbControls.txtOutputTarget, i18n.EasyGet("\231\173\137\229\190\133\230\175\148\232\181\155\229\188\128\229\167\139"))
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_DRAGONBOATRACE_DATA_SYNC, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_DRAGONBOATRACE_DATA_SYNC, self.OnSyncData, self)
  self:OnSyncData()
end

function tbWnd:OnDeactivate()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_DRAGONBOATRACE_DATA_SYNC, self)
end

function tbWnd:OnSyncData()
  if self.nLastTime < DragonBoatRace.nReceiveTime then
    if self.nTimer then
      Timer:Close(self.nTimer)
    end
    self.nLastTime = DragonBoatRace.nReceiveTime
    self:UpdateTimerText(DragonBoatRace.nShowTime)
    self.nTimer = Timer:Register(Env.GAME_FPS, self.UpdateTimer, self)
  end
  if not DragonBoatRace.tbRankList or #DragonBoatRace.tbRankList < 1 then
    return
  end
  self.tbScrollView:Clear()
  for i = 1, #DragonBoatRace.tbRankList do
    self.tbScrollView:SetElement(i, function(tbElement)
      self:SetRank(tbElement, DragonBoatRace.tbRankList[i], i)
    end)
  end
end

function tbWnd:SetRank(tbElement, tbData, i)
  local pPanel = tbElement.pPanel
  pPanel:Label_SetText("img", i)
  local szName = tbData.szName
  if szName == me.szName then
    szName = string.format("<color=#c5b505>%s</color>", szName)
  end
  pPanel:Label_SetText("Name", szName)
  local szWin = i18n.EasyGet("<color=#54ab4d>\229\183\178\229\174\140\230\136\144</color>")
  local szFail = i18n.EasyGet("\230\156\170\229\174\140\230\136\144")
  local szProcess = tbData.bFinish and szWin or szFail
  pPanel:Label_SetText("OutputDamage", szProcess)
end

function tbWnd:UpdateTimer()
  local nEndTime = DragonBoatRace.nShowTime + DragonBoatRace.nReceiveTime
  local nLeftTime = nEndTime - GetTime()
  if nLeftTime > 0 then
    self:UpdateTimerText(nLeftTime)
    return true
  elseif DragonBoatRace.bEnd then
    self.pPanel:Label_SetText(tbControls.txtOutputTarget, i18n.EasyGet("\230\175\148\232\181\155\231\187\147\230\157\159"))
    return false
  end
end

function tbWnd:UpdateTimerText(nLeftTime)
  local szColor = Ui.tbColor:GetColorByType("ClassifyParameter")
  local szDesc = string.format("<color=#%s>", szColor) .. Lib:TimeDesc(nLeftTime) .. "</color>"
  local szTime = i18n.EasyGet("<color=#{1}>\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154</color>{0}", szDesc, Ui.tbColor:GetColorByType("Classify"))
  self.pPanel:Label_SetText(tbControls.txtOutputTarget, szTime)
end
