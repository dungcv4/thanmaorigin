local tbWnd = Ui:CreateWindow("UISelectSeries")
local tbSeriesXiangSheng = {
  [1] = "Water",
  [2] = "Fire",
  [3] = "Wood",
  [4] = "Earth",
  [5] = "Metal"
}
local tbSeriesXiangShengText = {
  [1] = function()
    return i18n.EasyGet("\233\135\145\231\179\187")
  end,
  [2] = function()
    return i18n.EasyGet("\230\156\168\231\179\187")
  end,
  [3] = function()
    return i18n.EasyGet("\230\176\180\231\179\187")
  end,
  [4] = function()
    return i18n.EasyGet("\231\129\171\231\179\187")
  end,
  [5] = function()
    return i18n.EasyGet("\229\156\159\231\179\187")
  end,
  [6] = function()
    return i18n.EasyGet("\230\151\160")
  end
}
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  toggleMetal = "node/imgBG/Content/ToggleSeriesList/Metal",
  toggleWood = "node/imgBG/Content/ToggleSeriesList/Wood",
  toggleWater = "node/imgBG/Content/ToggleSeriesList/Water",
  toggleFire = "node/imgBG/Content/ToggleSeriesList/Fire",
  toggleEarth = "node/imgBG/Content/ToggleSeriesList/Earth",
  btnOK = "node/imgBG/Content/btnAuthenticate",
  txtPrice = "node/imgBG/Content/Price/txtSale",
  txtSetNumValue = "node/imgBG/Content/SetNum/Setting/txtSetNumValue",
  txtNumStart = "node/imgBG/Content/SetNum/txtNumStart",
  txtNumEnd = "node/imgBG/Content/SetNum/txtNumEnd",
  txtTip = "node/imgBG/Content/Tips/txtTip2",
  goSlider = "node/imgBG/Content/SetNum/Slider",
  imgSeriesActive = "node/imgBG/Content/Series/%s/imgActive",
  imgSeriesUnActive = "node/imgBG/Content/Series/%s/imgUnActive",
  imgSeriesActiveArrow = "node/imgBG/Content/Series/%s/imgActiveArrow",
  imgSeriesUnActiveArrow = "node/imgBG/Content/Series/%s/imgUnActiveArrow"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    self:OnClickOK()
  end
}
tbWnd.TOGGLE_GROUP = {
  tbWnd.tbControls.toggleMetal,
  tbWnd.tbControls.toggleWood,
  tbWnd.tbControls.toggleWater,
  tbWnd.tbControls.toggleFire,
  tbWnd.tbControls.toggleEarth
}
tbWnd.tbOnSliderClick = {
  goSlider = function(self, nValue)
    self.nSelectCount = nValue
    self.pPanel:Label_SetText(self.txtSetNumValue, self.nSelectCount)
    self.pPanel:Label_SetText(self.txtPrice, self.nEquipCost * self.nSelectCount)
  end
}

function tbWnd:OnCreate()
  for nId, szControl in ipairs(self.TOGGLE_GROUP) do
    Ui:Toggle_BindEvent(self.pPanel, szControl, function(self, isOn)
      if isOn then
        self:OnToggleSeries(nId)
      else
        self.nSeries = nil
      end
    end, self)
  end
end

function tbWnd:OnOpen(nItemId, fnCallback)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    self:Close()
    return
  end
  local pItem = me.GetItemInBag(nItemId)
  if not pItem then
    return
  end
  local nEquipCount = pItem.nCount
  self.nEquipCost = 0
  self.nMaxCount = nEquipCount
  local nBagFreeCount = me.GetFreeBagCount()
  if nEquipCount > nBagFreeCount then
    self.nMaxCount = nBagFreeCount
  end
  self.nSelectCount = self.nMaxCount
  self.pPanel:SetActive(self.goSlider, true)
  self.pPanel:SetActive(self.txtNumStart, true)
  self.pPanel:SetActive(self.txtNumEnd, true)
  self.pPanel:Label_SetText(self.txtSetNumValue, self.nMaxCount)
  self.pPanel:Label_SetText(self.txtNumStart, 1)
  self.pPanel:Label_SetText(self.txtNumEnd, self.nMaxCount)
  self.pPanel:Slider_SetEnable(self.goSlider, true)
  self.pPanel:Slider_SetRange(self.goSlider, 1, self.nMaxCount)
  self.pPanel:Slider_SetValue(self.goSlider, self.nMaxCount)
  if 1 == self.nMaxCount then
    self.pPanel:Slider_SetRange(self.goSlider, 0, self.nMaxCount)
    self.pPanel:Slider_SetValue(self.goSlider, self.nMaxCount)
    self.pPanel:Slider_SetEnable(self.goSlider, false)
    self.pPanel:SetActive(self.goSlider, false)
    self.pPanel:SetActive(self.txtNumStart, false)
    self.pPanel:SetActive(self.txtNumEnd, false)
  end
  if self.nMaxCount <= 0 then
    me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\231\154\132\232\131\140\229\140\133\230\187\161\228\186\134\239\188\140\230\149\180\231\144\134\228\184\128\228\184\139\229\144\167"))
    self:Close()
    return
  end
  local nItemTemplateId = pItem.dwTemplateId
  local nEquipTemplateId = KItem.GetItemExtParam(nItemTemplateId, Item.EquipForge.EXT_PARAM_SEED_TO_EQUIP)
  local nPos = KItem.GetEquipPos(nEquipTemplateId)
  local tbEquip = me.GetEquips()
  if tbEquip and tbEquip[nPos] then
    local pEquip = KItem.GetItemObj(tbEquip[nPos])
    self.nSeries = pEquip.GetSeries()
  else
    self.nSeries = Item.EquipForge.ATTR_ACTIVATE_LIST[nPos][KPlayer.GetPlayerInitInfo(me.nFaction).nSeries]
  end
  self.fnCallback = fnCallback
  self.pPanel:Toggle_SetChecked(self.TOGGLE_GROUP[self.nSeries], true)
  local nQuality = Item.EquipForge:GetQuality(pItem)
  self.nEquipCost = Item.EquipForge.IDENTIFY_COST[nPos] and Item.EquipForge.IDENTIFY_COST[nPos][nQuality] and Item.EquipForge.IDENTIFY_COST[nPos][nQuality][pItem.nLevel] or 0
  self.pPanel:Label_SetText(self.txtPrice, self.nEquipCost * self.nSelectCount)
  self:UpdateSeriesActive(nPos)
end

function tbWnd:OnClose()
  if self.nSeries then
    self.pPanel:Toggle_SetChecked(self.TOGGLE_GROUP[self.nSeries], false)
  end
end

function tbWnd:OnToggleSeries(nSeries)
  self.nSeries = nSeries
end

function tbWnd:OnClickOK()
  local nCost = self.nEquipCost * self.nSelectCount
  if nCost > me.GetMoney("Coin") then
    GainCurrency:ReadyForJump("Coin")
    return
  end
  if not self.nSeries then
    return me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\233\128\137\230\139\169\228\186\148\232\161\140\229\177\158\230\128\167\239\188\129"))
  end
  if self.nSeries <= Item.emSERIES_TYPE.SERIES_NONE or self.nSeries >= Item.emSERIES_TYPE.SERIES_NUM then
    return
  end
  if self.fnCallback then
    self.fnCallback(self.nSeries, self.nSelectCount)
  end
  self:Close()
end

function tbWnd:UpdateSeriesActive(nPos)
  local tbSeriesList = Item.EquipSeriesActive:StatSeries(me)
  for nSeries = Item.emSERIES_TYPE.SERIES_NONE + 1, Item.emSERIES_TYPE.SERIES_NUM - 1 do
    local bActiveAttrib = false
    if tbSeriesList[nSeries] >= 2 then
      bActiveAttrib = true
    end
    self.pPanel:SetActive(string.format(self.imgSeriesUnActive, tbSeriesXiangSheng[nSeries]), true)
    self.pPanel:SetActive(string.format(self.imgSeriesUnActiveArrow, tbSeriesXiangSheng[nSeries]), true)
    self.pPanel:SetActive(string.format(self.imgSeriesActive, tbSeriesXiangSheng[nSeries]), bActiveAttrib)
    self.pPanel:SetActive(string.format(self.imgSeriesActiveArrow, tbSeriesXiangSheng[nSeries]), bActiveAttrib)
  end
  local pEquip = me.GetEquipByPos(nPos)
  if pEquip then
    local nSeries = pEquip.GetSeries()
    self.pPanel:Label_SetText(self.txtTip, tbSeriesXiangShengText[nSeries]())
  else
    self.pPanel:Label_SetText(self.txtTip, tbSeriesXiangShengText[6]())
  end
end
