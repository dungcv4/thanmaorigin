PlayerPortraitFrame.CUR_PFID_TASK = 1
PlayerPortraitFrame.SAVE_GROUP = GetGroupKey("PlayerPortraitFrame")
PlayerPortraitFrame.MessageType = {RedPacket = true}

function PlayerPortraitFrame:LoadSetting()
  local tbSetting = Lib:LoadTabFileEx("Setting/Player/PortraitFrame.tab", "ddssssd", "Id", {
    "Id",
    "nDefault",
    "szName",
    "Icon",
    "PlayInterval",
    "SourceDesc",
    "nBackup"
  })
  assert(tbSetting, "[PlayerPortraitFrame LoadSetting Fail]")
  self.tbPortraitFrameSetting = {}
  for _, v in pairs(tbSetting) do
    assert(not self.tbPortraitFrameSetting[v.Id], "[PlayerPortraitFrame LoadSetting] Error, Key Repeat")
    self.tbPortraitFrameSetting[v.Id] = {
      nId = v.Id,
      nDefault = v.nDefault,
      szName = v.szName,
      szIcon = v.Icon,
      nPlayInterval = tonumber(v.PlayInterval),
      szSourceDesc = v.SourceDesc,
      nBackup = v.nBackup
    }
  end
  self.tbPortraitFrameSetting = SetReadOnly(self.tbPortraitFrameSetting)
end

PlayerPortraitFrame:LoadSetting()

function PlayerPortraitFrame:GetPortraitFrameInfoById(nId)
  assert(nId, "[PlayerPortraitFrame GetPortraitFrameInfoById Id Null]")
  local tInfo = self.tbPortraitFrameSetting[nId]
  assert(tInfo, "[PlayerPortraitFrame GetPortraitFrameInfoById Id Error]")
  return tInfo
end

function PlayerPortraitFrame:CheckPFIdExist(nId)
  assert(nId, "[PlayerPortraitFrame CheckPFIdExist Id Null]")
  local bIsExist = self.tbPortraitFrameSetting[nId]
  return bIsExist
end

function PlayerPortraitFrame:GetMeCurPortraitFrameId(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.CUR_PFID_TASK)
end

function PlayerPortraitFrame:GetIcon(nPortraitFrameID)
  local tbInfo = self.tbPortraitFrameSetting[nPortraitFrameID]
  if not tbInfo then
    return ""
  end
  return tbInfo.szIcon
end

function PlayerPortraitFrame:GetPlayInterval(nPortraitFrameID)
  local tbInfo = self.tbPortraitFrameSetting[nPortraitFrameID]
  if not tbInfo then
    return nil
  end
  return tbInfo.nPlayInterval
end

function PlayerPortraitFrame:IsDefaultProtrait(nPortraitFrameID)
  local tbInfo = self:GetPortraitFrameInfoById(nPortraitFrameID)
  if tbInfo and 1 == tbInfo.nDefault then
    return true
  end
  return false
end

function PlayerPortraitFrame:IsPermanentPortrait(pPlayer, nPortraitFrameID, tbPlayerData)
  if MODULE_GAMESERVER then
    tbPlayerData = self:GetPlayerData(pPlayer)
  end
  if tbPlayerData[nPortraitFrameID] and tbPlayerData[nPortraitFrameID].nGetTime == tbPlayerData[nPortraitFrameID].nLimitTime then
    return true
  end
  return false
end

function PlayerPortraitFrame:CheckIsInValidTimeAndInSameDay(pPlayer, nPortraitFrameID, tbPlayerData)
  if PlayerPortraitFrame:IsDefaultProtrait(nPortraitFrameID) then
    return true
  end
  if PlayerPortraitFrame:IsPermanentPortrait(pPlayer, nPortraitFrameID, tbPlayerData) then
    return true
  end
  if MODULE_GAMESERVER then
    tbPlayerData = self:GetPlayerData(pPlayer)
  end
  if not tbPlayerData[nPortraitFrameID] then
    return false
  end
  local nTime = GetTime()
  if nTime < tbPlayerData[nPortraitFrameID].nLimitTime then
    if tbPlayerData[nPortraitFrameID].nLimitTime - nTime < 86400 then
      return true, true, tbPlayerData[nPortraitFrameID].nLimitTime - nTime
    end
    return true, false
  elseif nTime > tbPlayerData[nPortraitFrameID].nLimitTime then
    return false, false
  end
end
