local tbWnd = Ui:CreateWindow("UIWuShuangHaoJieGuessing")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SERVER_TIME_ZONE_UPDATE,
      self.UpdateSchedule
    },
    {
      EventNotify.emNOTIFY_WULINDAHUI_SYNC_FINAL,
      self.UpdateShow
    },
    {
      EventNotify.emNOTIFY_WULINDAHUI_SYNC_GUESS,
      self.UpdateShow
    }
  }
end

function tbWnd:OnCreate()
  for nPhase, tbIndex in pairs(self.tbViewPhase2Idx) do
    for nIndex, nBtnIdx in ipairs(tbIndex) do
      Ui:Button_BindEvent(self.pPanel, tbControls.btnView .. nBtnIdx, function()
        WuLinDaHui:C2SWatchBattle(nPhase, nIndex)
      end)
    end
  end
end

function tbWnd:OnClose()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:OnOpen(nWarZone)
  self.nWarZone = nWarZone
  WuLinDaHui:RequestFinalMatch()
  self:UpdateSchedule()
  self:UpdateShow()
  self:UpdateGuessState()
end

function tbWnd:UpdateSchedule()
  local function getShowTime(nDate)
    return os.date("%m.%d %H:%M", Lib:GetConfigDate2Time(nDate))
  end
  
  self.pPanel:Label_SetText(self.txtTime .. 1, getShowTime(WuLinDaHui.FinalCfg.SHOW_TIME_1))
  self.pPanel:Label_SetText(self.txtTime .. 7, getShowTime(WuLinDaHui.FinalCfg.SHOW_TIME_1))
  self.pPanel:Label_SetText(self.txtTime .. 2, getShowTime(WuLinDaHui.FinalCfg.SHOW_TIME_2))
  self.pPanel:Label_SetText(self.txtTime .. 6, getShowTime(WuLinDaHui.FinalCfg.SHOW_TIME_2))
  self.pPanel:Label_SetText(self.txtTime .. 3, getShowTime(WuLinDaHui.FinalCfg.SHOW_TIME_3))
  self.pPanel:Label_SetText(self.txtTime .. 5, getShowTime(WuLinDaHui.FinalCfg.SHOW_TIME_3))
  self.pPanel:Label_SetText(self.txtTime .. 4, getShowTime(WuLinDaHui.FinalCfg["SHOW_TIME_4_ZONE" .. WuLinDaHui.nWarZone]))
end

function tbWnd:UpdateGuessState()
  self.pPanel:SetActive(tbControls.Countdown, false)
  self.pPanel:Button_SetEnable(tbControls.btnGuess, true)
end

function tbWnd:UpdateShow()
  if not WuLinDaHui.tbFinalMatchList then
    return
  end
  self.tbFinalMatchList = WuLinDaHui.tbFinalMatchList
  self.tbMyGuessData = WuLinDaHui.tbMyGuessData
  self.tbIndexMapId = WuLinDaHui.tbIndexMapId
  for i = 1, 30 do
    self.pPanel:Label_SetText(tbControls.Team .. i .. "/txtName", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
  end
  self.pPanel:Label_SetText(tbControls.Champion .. "/txtName", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
  for nPhase, tbPhaseData in pairs(self.tbFinalMatchList) do
    if self.tbPhase2Idx[nPhase] then
      local tbGuessTeam = {}
      if self.tbMyGuessData and self.tbMyGuessData[nPhase] then
        do
          local tbGuessData = self.tbMyGuessData[nPhase]
          for _, nTeamId in ipairs(tbGuessData.tbGuess) do
            tbGuessTeam[nTeamId] = true
          end
        end
      end
      for nIdx, tbData in pairs(tbPhaseData) do
        if self.tbPhase2Idx[nPhase][nIdx] then
          local tbTeam = WuLinDaHui.tbFinalTeamData[tbData.nTeamId]
          local szTitle = ""
          if tbTeam.nServerSubId then
            szTitle = i18n.EasyGet("[{0}\230\156\141]", tbTeam.nServerSubId) .. "-"
          end
          if nPhase == WuLinDaHui.FINAL_PAHSE.PAHSE1 then
            self.pPanel:Label_SetText(tbControls.Champion .. "/txtName", tbData.szTeamName and szTitle .. tbData.szTeamName or i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
          else
            local nNameIdx = self.tbPhase2Idx[nPhase][nIdx]
            self.pPanel:Label_SetText(tbControls.Team .. nNameIdx .. "/txtName", tbData.szTeamName and szTitle .. tbData.szTeamName or i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
            self.pPanel:SetActive(tbControls.Team .. nNameIdx .. "/imgWin", tbData.nResult == WldhTeam.Win)
            self.pPanel:SetActive(tbControls.Team .. nNameIdx .. "/imgSelect", tbGuessTeam[tbData.nTeamId])
          end
        end
      end
    end
  end
  for nPhase, tbIndex in pairs(self.tbViewPhase2Idx) do
    for nIndex, nBtnIdx in ipairs(tbIndex) do
      local bShowWatch = WldhTeam:ChkPlayerInMapEx(me) and not WuLinDaHui:CheckContestant(me.dwID, nPhase)
      local bHaveMatch = false
      if self.tbIndexMapId and self.tbIndexMapId[nPhase] and self.tbIndexMapId[nPhase][nIndex] then
        bHaveMatch = true
      end
      bShowWatch = bHaveMatch and bShowWatch or false
      self.pPanel:SetActive(tbControls.btnView .. nBtnIdx, bShowWatch)
    end
  end
end
