local tbWnd = Ui:CreateWindow("UIDaShiXiong")
tbWnd.tbControls = {
  txtPollNow = "node/imgBG/txtPollNow",
  txtElectDaShiXiong = "node/imgBG/ElectTitle/txtElectDaShiXiong",
  goVoteCentTemplate = "node/imgBG/ScrollView/Viewport/electList/voteCent",
  btnClose = "node/imgBG/btnClose"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbDaShiXiongScrollView = Ui:NewScrollView(self, self.goVoteCentTemplate, nil, false, true)
end

function tbWnd:OnOpen()
  local nMyFaction = me.nFaction
  local bOpenMale = Faction:IsOpenSex(nMyFaction, Npc.SexDef.MALE)
  local bOpenFemale = Faction:IsOpenSex(nMyFaction, Npc.SexDef.FEMALE)
  if not bOpenMale and bOpenFemale then
    self.pPanel:Label_SetText(self.txtElectDaShiXiong, i18n.EasyGet("\232\175\132\233\128\137\233\151\168\230\180\190\229\164\167\229\184\136\229\167\144"))
  else
    self.pPanel:Label_SetText(self.txtElectDaShiXiong, i18n.EasyGet("\232\175\132\233\128\137\233\151\168\230\180\190\229\164\167\229\184\136\229\133\132"))
  end
  FactionBigBrother:SynData()
end

function tbWnd:UpdateDaShiXiongData()
  self.tbDaShiXiongScrollView:Clear()
  local nMyFaction = me.nFaction
  local tbHouXuanRenInfo = FactionBigBrother.tbBigBrotherData
  if not tbHouXuanRenInfo.tbAll[nMyFaction] then
    return
  end
  for nIndex, tbPlayerInfo in pairs(tbHouXuanRenInfo.tbAll[nMyFaction]) do
    self.tbDaShiXiongScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("txtPlayerName", tbPlayerInfo.szName)
      local szKinName = tbPlayerInfo.szKinName
      local szTongName = tbPlayerInfo.szTongName
      if "" == szKinName then
        szKinName = i18n.EasyGet("\230\154\130\230\151\160")
      end
      if "" == szTongName then
        szTongName = i18n.EasyGet("\230\154\130\230\151\160")
      end
      tbElement.pPanel:Label_SetText("PortraitObj/Level/Text", tbPlayerInfo.nLevel)
      tbElement.pPanel:Label_SetText("RankInfo/Family/txtFamilyName", szKinName)
      tbElement.pPanel:Label_SetText("RankInfo/BangHui/txtBangHuiName", szTongName)
      tbElement.pPanel:Label_SetText("RankInfo/FightingValue/txtFightingValueNum", tbPlayerInfo.nFightPower)
      tbElement.pPanel:Label_SetText("RankInfo/Poll/txtPollNum", tbPlayerInfo.nScore)
      tbElement.pPanel:Button_BindEvent("btnVote", function()
        local bRet, szReason = FactionBigBrother:CheckVote()
        if bRet then
          RemoteServer.RequestFactionBigBrother("VoteBigBrother", tbPlayerInfo.nPlayerId)
        else
          me.CenterMsg(szReason)
        end
      end)
      local szPortraitPath = PlayerPortrait:GetIcon(tbPlayerInfo.nPortrait)
      tbElement.pPanel:Sprite_SetSprite("PortraitObj/Portrait/Image", szPortraitPath)
      PlayerPortraitFrame:SetPortraitFrame(tbElement.pPanel, "PortraitObj/PortraitFrame", tbPlayerInfo.nPortraitFrame)
    end)
  end
  self.pPanel:Label_SetText(self.txtPollNow, i18n.EasyGet("\229\189\147\229\137\141\229\143\175\230\138\149\231\165\168\230\149\176\239\188\154<color=#138446>{0}</color>", FactionBigBrother.tbBigBrotherData.nVoteNumLeft))
end
