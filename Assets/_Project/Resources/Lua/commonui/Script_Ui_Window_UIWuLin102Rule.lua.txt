function tbWnd:OnCreate()
  self.tbTree = Ui:NewTree(self, tbControls.uiTree, self.OnClickTreeBig, self.OnClickTreeSmall)
  self.tbBigData = {}
  self.tbSmallData = {}
  for nIndex, tbSetting in ipairs(self.PAGE_SETTING) do
    table.insert(self.tbBigData, {
      nType = nIndex,
      szText = tbSetting.fnTitle(),
      tbInfo = {nBigIndex = nIndex}
    })
    for nSubIndex, tbSubSetting in ipairs(tbSetting.tbChild) do
      table.insert(self.tbSmallData, {
        nType = nIndex,
        szText = tbSubSetting.fnTitle(),
        tbInfo = {nSmallIndex = nSubIndex}
      })
    end
  end
  self.tbTree:SetData(self.tbBigData, self.tbSmallData)
  self.tbRuleScroll = Ui:NewScrollView(self, tbControls.txtRuleTemplate)
  self.tbGetWayScroll = Ui:NewScrollView(self, tbControls.goGetWayTemplate)
  self:SetClickEmptyCallback(tbControls.wndItemGet)
  self.tbGetVoteRankScroll = Ui:NewScrollView(self, tbControls.goRewardTemplate)
  self.tbUseVoteRankScroll = Ui:NewScrollView(self, tbControls.goRewardRankTemplate)
end

function tbWnd:OnOpen(szBigType, szSmallType)
  if self.nRuleTimer then
    Timer:Close(self.nRuleTimer)
    self.nRuleTimer = nil
  end
  self.nRuleTimer = Timer:Register(Env.GAME_FPS, self.UpdateRuleTime, self)
  self.nBigIndex = 1
  self.nSmallIndex = 1
  for _, tbData in ipairs(self.tbBigData) do
    if szBigType == self.PAGE_SETTING[tbData.tbInfo.nBigIndex].szType then
      self.nBigIndex = tbData.tbInfo.nBigIndex
      break
    end
  end
  for _, tbData in ipairs(self.tbSmallData) do
    if szSmallType == self.PAGE_SETTING[self.nBigIndex].tbChild[tbData.tbInfo.nSmallIndex].szType then
      self.nSmallIndex = tbData.tbInfo.nSmallIndex
      break
    end
  end
  self.tbTree:SelectNode(self.nBigIndex, self.nSmallIndex, true)
end

function tbWnd:OnClose()
  if self.nRuleTimer then
    Timer:Close(self.nRuleTimer)
    self.nRuleTimer = nil
  end
end

function tbWnd:OnClickTreeBig(tbInfo)
  self.nBigIndex = tbInfo.nBigIndex
  for nIndex, tbSetting in ipairs(self.PAGE_SETTING) do
    self.pPanel:SetActive(tbSetting.szPanel, nIndex == self.nBigIndex)
  end
  self.tbTree:AutoSelectSmall(self.nBigIndex)
end

function tbWnd:OnClickTreeSmall(tbInfo)
  self.nSmallIndex = tbInfo.nSmallIndex
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex]
  if tbPageSetting then
    if "Rule" == tbPageSetting.szType then
      self:UpdateRuleIntro()
      self:UpdateRuleTime()
    elseif tbPageSetting.szType == "GetVoteRank" then
      self:UpdateGetVoteRankAward()
      self:UpdateGetVoteRankTime()
    elseif tbPageSetting.szType == "UseVoteRank" then
      self:UpdateUseVoteRankAward()
      self:UpdateUseVoteRankTime()
    end
  end
end

function tbWnd:UpdateRuleIntro()
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex] and self.PAGE_SETTING[self.nBigIndex].tbChild[self.nSmallIndex] or nil
  local tbRuleSetting = tbPageSetting and AvenueOfStars.tbRuleIntro[tbPageSetting.szType] or nil
  if not tbRuleSetting then
    return
  end
  self.pPanel:Label_SetText(tbControls.txtRuleName, tbRuleSetting.fnTitle())
  self.tbRuleScroll:Reset()
  for nIndex, tbIntro in ipairs(tbRuleSetting.tbIntro) do
    self.tbRuleScroll:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("Main", tbIntro.szSeq .. tbIntro.fnText())
    end)
  end
end

function tbWnd:UpdateRuleTime()
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex]
  if not tbPageSetting or "Rule" ~= tbPageSetting.szType then
    return true
  end
  tbPageSetting = tbPageSetting.tbChild[self.nSmallIndex] or nil
  if not tbPageSetting then
    return true
  end
  local tbDate, tbDateYMD, szApplyLabel
  if tbPageSetting.szType == "Enroll" then
    tbDate = AvenueOfStars.ENROLL_DATE
    tbDateYMD = AvenueOfStars.ENROLL_DATE_YMD
    szApplyLabel = i18n.EasyGet("\230\138\165\229\144\141")
  else
    tbDate = AvenueOfStars.VOTE_DATE[tbPageSetting.nPhase]
    tbDateYMD = AvenueOfStars.VOTE_DATE_YMD[tbPageSetting.nPhase]
    szApplyLabel = i18n.EasyGet("\230\138\149\231\165\168")
  end
  if not tbDate then
    return true
  end
  local nGameTime = GetTime()
  local nStartTime = Lib:ParseConfigDateTime(tbDate[1])
  local nEndTime = Lib:ParseConfigDateTime(tbDate[2])
  if nGameTime < nStartTime then
    self.pPanel:Label_SetText(tbControls.txtRuleTime, string.format("%s-%s", tbDateYMD[1], tbDateYMD[2]))
    self.pPanel:Button_SetEnable(tbControls.btnApply, false)
  elseif nGameTime < nEndTime then
    self.pPanel:Label_SetText(tbControls.txtRuleTime, string.format("%s-%s %s%s", tbDateYMD[1], tbDateYMD[2], i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154"), Lib:TimeDesc2(nEndTime - nGameTime)))
    self.pPanel:Button_SetEnable(tbControls.btnApply, true)
  else
    self.pPanel:Label_SetText(tbControls.txtRuleTime, string.format("%s-%s %s", tbDateYMD[1], tbDateYMD[2], i18n.EasyGet("\229\183\178\231\187\147\230\157\159")))
    self.pPanel:Button_SetEnable(tbControls.btnApply, false)
  end
  self.pPanel:Button_SetText(tbControls.btnApply, szApplyLabel, "Text")
  if tbPageSetting.szType == "Enroll" then
    self.pPanel:SetActive(tbControls.btnGetWay, false)
  else
    local szGetName
    local tbRuleSetting = AvenueOfStars.tbRuleIntro[tbPageSetting.szType]
    if tbRuleSetting and tbRuleSetting.nItemId then
      szGetName = Item:GetName(AvenueOfStars.VOTE_ITEM_ID[tbRuleSetting.nItemId])
    end
    if not Lib:IsEmptyStr(szGetName) then
      self.pPanel:Button_SetText(tbControls.btnGetWay, i18n.EasyGet("{0}\232\142\183\229\143\150\230\150\185\229\188\143\227\128\139", i18n.Get(szGetName)), "Text")
      self.pPanel:SetActive(tbControls.btnGetWay, true)
    else
      self.pPanel:SetActive(tbControls.btnGetWay, false)
    end
  end
  return true
end

function tbWnd:UpdateGetVoteRankAward()
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex] and self.PAGE_SETTING[self.nBigIndex].tbChild[self.nSmallIndex] or nil
  local tbAwardSetting = tbPageSetting and AvenueOfStars.tbGetVoteRankAward[tbPageSetting.szType] or nil
  if not tbAwardSetting then
    return
  end
  self.tbGetVoteRankScroll:Reset()
  for nIndex, tbAward in ipairs(tbAwardSetting) do
    self.tbGetVoteRankScroll:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtRank", tbAward.fnTitle())
      if not tbElement.tbItemGridView then
        tbElement.tbItemGridView = Ui:NewItemGridView(self, tbElement.szKey .. "/rewardList/reward")
      end
      local tbItemGridView = tbElement.tbItemGridView
      local tbItemList = Award:GetAward(tbAward.tbAward[1], tbAward.tbAward[2])
      for nItemIndex, tbItemInfo in ipairs(tbItemList) do
        tbItemGridView:SetItem(nItemIndex, tbItemInfo, true)
      end
      local nItemNum = #tbItemList
      tbItemGridView:ShowRange(1, nItemNum)
      tbItemGridView:HideRange(nItemNum + 1)
    end)
  end
end

function tbWnd:UpdateGetVoteRankTime()

function tbWnd:OnClickGetWay()
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex]
  if not tbPageSetting or "Rule" ~= tbPageSetting.szType then
    return
  end
  tbPageSetting = tbPageSetting.tbChild[self.nSmallIndex] or nil
  if not tbPageSetting or tbPageSetting.szType == "Enroll" then
    return
  end
  local tbRuleSetting = AvenueOfStars.tbRuleIntro[tbPageSetting.szType]
  if not tbRuleSetting or not tbRuleSetting.nItemId then
    return
  end
  local tbWaySetting = AvenueOfStars.tbItemGetWay[tbRuleSetting.nItemId]
  if not tbWaySetting then
    return
  end
  self.tbGetWayScroll:Reset()
  for nIndex, tbWay in ipairs(tbWaySetting) do
    self.tbGetWayScroll:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtWay", tbWay.fnDesc())
      if tbWay.fnRand then
        pPanel:Label_SetText("txtNumber", tostring(tbWay.nNum) .. tbWay.fnRand())
      else
        pPanel:Label_SetText("txtNumber", tostring(tbWay.nNum))
      end
    end)
  end
  self.pPanel:SetActive(tbControls.wndItemGet, true)
end

function tbWnd:OnClickEmptyClose()
  self.pPanel:SetActive(tbControls.wndItemGet, false)
end

function tbWnd:OnClickApply()
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex]
  if not tbPageSetting or "Rule" ~= tbPageSetting.szType then
    return
  end
  tbPageSetting = tbPageSetting.tbChild[self.nSmallIndex] or nil
  if not tbPageSetting then
    return
  end
  if tbPageSetting.szType == "Enroll" then
    AvenueOfStars:GotoOutgameEnroll()
  else
    AvenueOfStars:GotoOutgameWeb()
  end
end

function tbWnd:OnClickHelp()
  local tbPageSetting = self.PAGE_SETTING[self.nBigIndex] and self.PAGE_SETTING[self.nBigIndex].tbChild[self.nSmallIndex] or nil
  if not tbPageSetting or not tbPageSetting.szHelp then
    return
  end
  Ui:OnHelpClicked(tbPageSetting.szHelp)
end
