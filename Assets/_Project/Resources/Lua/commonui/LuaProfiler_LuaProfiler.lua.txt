LuaProfiler.bBootOpenProfiler = false
LuaProfiler.bOpenProfiler = true
LuaProfiler.szServerOutFile = "LuaProfiler/profiler_server.txt"
LuaProfiler.szClientOutFile = "LuaProfiler/profiler_client.txt"

function LuaProfiler:BootStart(bServer)
  if not self.bBootOpenProfiler then
    return
  end
  self:Start(bServer)
end

function LuaProfiler:Start(bServer)
  if not self.bOpenProfiler then
    return
  end
  if self.bStart then
    return
  end
  local szPath = bServer and self.szServerOutFile or self.szClientOutFile
  profiler_start(szPath, 1)
  self.bStart = true
end

function LuaProfiler:Stop()
  if not self.bStart then
    return
  end
  profiler_stop()
  self.bStart = nil
end
