local tbWnd = Ui:CreateWindow("UIReward_Newyear")
local tbControls = {
  txtTitle = "node/imgBG/txtTitle",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  btnAdd = "node/imgBG/btnAdd",
  btnClose = "node/imgBG/btnClose",
  btnGo = "node/imgBG/btnGo",
  szTongNumber = "node/imgBG/Cont_TxtAll/Cont_Txt01/txtNumber",
  szMineNumber = "node/imgBG/Cont_TxtAll/Cont_Txt01/txtMineNumber",
  txtInfo = "node/imgBG/Cont_TxtAll/Cont_Txt02/txtInfo",
  szItemName = "node/imgBG/Content/Template/txtItemCount",
  szItemNums = "node/imgBG/Content/Template/txtItemCountValue",
  szItemGrid = "node/imgBG/Content/Template/UIItemGrid",
  szTemplate = "node/imgBG/Content/Scroll View/Viewport/Content/Template",
  Template = "node/imgBG/Content/Template",
  Cont_TxtAll = "node/imgBG/Cont_TxtAll",
  txtTitleNumber = "node/imgBG/Content/imgTitleBg/txtTitleNumber",
  txtTitleReward = "node/imgBG/Content/imgTitleBg/txtTitleReward",
  txtTitlePrg = "node/imgBG/Content/imgTitleBg/txtTitlePrg"
}
tbWnd.tbControls = tbControls
tbWnd.tbTemplateElement = {
  reward = "rewardList/reward",
  txtPrg = "prg/txtPrg",
  imgBar = "prg/imgBar",
  imgComplete = "imgComplete",
  CompleteText = "imgComplete/Text",
  txtNumber = "txtNumber",
  Prg = "prg",
  btnGet = "btnGet"
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UPDATE_EXTRA_AWARD,
      self.UpdateReward,
      self
    }
  }
end

tbWnd.tbOnClick = {
  btnAdd = function(self)
    self.pIns:OnUiEvent(self, "OnClick", "btnAdd")
  end,
  btnGo = function(self)
    self.pIns:OnUiEvent(self, "OnClick", "btnGo")
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbLevelList = Ui:NewScrollView(self, self.szTemplate)
end

function tbWnd:OnOpen(value, ...)
  if type(value) == "number" then
    self.pIns = ActivityMgr:GetInstanceById(value)
    if not self.pIns then
      return
    end
    self.nActId = value
  elseif type(value) == "table" then
    if not value.OnUiEvent then
      return
    end
    self.pIns = value
  else
    return
  end
  self.pIns:OnUiEvent(self, "OnOpen", self.nActId, ...)
  local szTitle = self.pIns.GetUiTitle and self.pIns:GetUiTitle() or ""
  if not Lib:IsEmptyStr(szTitle) then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.Parse(szTitle))
  end
end

function tbWnd:UpdateReward()
  self.pIns:OnUiEvent(self, "UpdateReward", self.nActId)
end

function tbWnd:OnClose()
  self.tbLevelList:Clear()
end
