function GlobalHangUp:LoadPosConfig()
  self.tbRevivePos = {}
  local szFile, tbPosList
  for nMapTemplateId, szName in pairs(self.Cfg.NpcPosList) do
    szFile = Map:GetMapPosFile(nMapTemplateId, szName)
    tbPosList = Lib:LoadTabFileEx(szFile, "ddd", nil, {
      "NpcTemplateId",
      "PosX",
      "PosY"
    })
    self.tbRevivePos[nMapTemplateId] = tbPosList
  end
end

GlobalHangUp:LoadPosConfig()

function GlobalHangUp:IsGlobalHangUpMap(nMapTemplateId)
  return Lib:IIF(self.Cfg.MapList[nMapTemplateId] ~= nil, true, false)
end

function GlobalHangUp:CheckCanKongQueHeJoin()
  return me.nTotalLevel >= self.Cfg.MapList[self.KongQueHeMapTemplateID].Level
end

function GlobalHangUp:CheckCanLeaveKongQueHeJoin()
  return true
end

function GlobalHangUp:CheckKongQueHeEnter()
  if MODULE_GAMESERVER then
    return self:CheckCanEnter(me)
  end
  return true
end

function GlobalHangUp:CheckLeaveKongQueHeEnter()
  if MODULE_GAMESERVER then
    return self:CheckCanLeave(me)
  end
  return true
end

function GlobalHangUp:GetGlobalHangUpAddExpP(nMapTemplateId)
  local nServerMaxLevel = self:GetMaxLevel()
  return self.Cfg.AddExpPList[nServerMaxLevel] or 0
end

function GlobalHangUp:CanSwitchMap(nCurTemplateId, nMapId)
  if not self.Cfg.MapList[nMapId] then
    return true
  end
  if self.Cfg.CanSwitchMapList[nCurTemplateId] then
    return true
  end
  return false, i18n.Get(self.Cfg.szCanNotSwitchMapMsg)
end
