local tbWnd = Ui:CreateWindow("UINewInfoTimeline")
local tbControls = {
  InfoTemplate = "InfoList/Viewport/Content/InfoTemplate",
  txtTitle = "txtTitle"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.NewTimeFrameReminder)
  if not pInstance then
    return 0
  end
  local tbInfo = pInstance:GetCurReminderInfo()
  if not tbInfo then
    LogErr("failed to get cur reminder info")
    return
  end
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.Parse(tbInfo.szShowTitle))
  self.pPanel:Label_SetText(tbControls.InfoTemplate, i18n.Parse(tbInfo.szDesc))
  pInstance:CheckRedPoint(true)
end
