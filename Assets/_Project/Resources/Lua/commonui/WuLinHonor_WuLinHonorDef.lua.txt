  "nOnceTriggerItem",
  "nActivityId",
  "nGrade"
})
WuLinHonor.tbTriggerType = {RecordTimes = 1, Once = 2}
WuLinHonor.tbRecordTimesTriggerCfg = Lib:EasyLoadTabFile("Setting/WuLinHonor/RecordTimesTriggerCondition.tab", {
  "nRecordTimesTriggerItem",
  "nActivityId",
  "nGrade"
})
WuLinHonor.tbRankInfo = {}
for nRecordTimesTriggerItem, tbItemList in pairs(WuLinHonor.tbRecordTimesTriggerCfg) do
  for nActivityId, tbActList in pairs(tbItemList) do
    WuLinHonor.tbRankInfo[nActivityId] = WuLinHonor.tbRankInfo[nActivityId] or {}
    for nGrade, _ in pairs(tbActList) do
      WuLinHonor.tbRankInfo[nActivityId][nGrade] = WuLinHonor.tbRankInfo[nActivityId][nGrade] or {}
      table.insert(WuLinHonor.tbRankInfo[nActivityId][nGrade], {
        nTriggerId = nRecordTimesTriggerItem,
        nType = WuLinHonor.tbTriggerType.RecordTimes
      })
    end
  end
end
for nOnceTriggerItem, tbItemList in pairs(WuLinHonor.tbOnceTriggerCfg) do
  for nActivityId, tbActList in pairs(tbItemList) do
    WuLinHonor.tbRankInfo[nActivityId] = WuLinHonor.tbRankInfo[nActivityId] or {}
    for nGrade, _ in pairs(tbActList) do
      WuLinHonor.tbRankInfo[nActivityId][nGrade] = WuLinHonor.tbRankInfo[nActivityId][nGrade] or {}
      table.insert(WuLinHonor.tbRankInfo[nActivityId][nGrade], {
        nTriggerId = nOnceTriggerItem,
        nType = WuLinHonor.tbTriggerType.Once
      })
    end
  end
end
WuLinHonor.tbQualificationTriggerCfg = Lib:EasyLoadTabFile("Setting/WuLinHonor/QualificationCondition.tab", {
  "nOnceTriggerItem",
  "nQualificationId"
})
WuLinHonor.tbQualification = {}
for nOnceTriggerItem, tbItemList in pairs(WuLinHonor.tbQualificationTriggerCfg) do
  for nQualificationId, tbInfo in pairs(tbItemList) do
    WuLinHonor.tbQualification[tbInfo.nActivityId] = WuLinHonor.tbQualification[tbInfo.nActivityId] or {}
    table.insert(WuLinHonor.tbQualification[tbInfo.nActivityId], {
      nQualificationId = nQualificationId,
      nOnceTriggerItem = nOnceTriggerItem,
      nMonth = tbInfo.nMonth
    })
  end
end
