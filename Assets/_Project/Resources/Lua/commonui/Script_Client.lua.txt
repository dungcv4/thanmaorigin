function Client:LoadHotFix()
  local Application = CS.UnityEngine.Application
  local File = CS.System.IO.File
  local path = Application.persistentDataPath .. "/hotfix"
  local hotFixFile = path .. "/hotfix"
  if File.Exists(hotFixFile) then
    local content = File.ReadAllBytes(hotFixFile)
    local func, err = load(content, hotFixFile, "b")
    if err then
      return
    end
    if not func then
      return
    end
    xpcall(func, Lib.ShowStack)
    Log("Client:LoadHotFix:Finish")
  end
end

function Client:SetPlayerDir(nDir, nMapTemplateId)
  if not nDir or not nMapTemplateId then
    return
  end
  me.nSetDirMapTId = nMapTemplateId
  me.nSetDirDir = nDir
  self:DoSetPlayerDir()
end

function Client:DoSetPlayerDir(bConfirm)
  if not me.nSetDirMapTId or not me.nSetDirDir then
    return
  end
  if me.nMapTemplateId ~= me.nSetDirMapTId then
    return
  end
  if not bConfirm then
    Timer:Register(5, self.DoSetPlayerDir, self, true)
    return
  end
  me.GetNpc().SetDir(me.nSetDirDir)
  me.nSetDirMapTId = nil
end

end

function Client:IsOnlyUseIPv4()
  local bResult = true
  local szIP, tbResolved = socket.dns.toip(socket.dns.gethostname())
  for k, v in ipairs(tbResolved.ip or {}) do
    local tbChunks = {
      v:match("(%d+)%.(%d+)%.(%d+)%.(%d+)")
    }
    if 4 ~= #tbChunks then
      bResult = false
    end
    Log("Client:IsOnlyUseIPv4", #tbChunks, v)
  end
  return bResult
end

function Client:StartResourceMonitor()
  if Client.nMonitorTimer then
    Timer:Close(Client.nMonitorTimer)
    Client.nMonitorTimer = nil
  end
  Client.nMonitorTimer = Timer:Register(Env.GAME_FPS, function()
    local nResourceCache = Client.ResourceModule.GetResourceCacheCount()
    local nRuningCount = Client.ResourceModule.GetResourceRuningCount()
    local nAssetCache = Client.AssetResourceModule.GetCacheCount()
    local nLoadingCount = Client.AssetResourceModule.GetLoadingCount()
    local nAudioCache = Client.AssetResourceModule.GetAudioCacheCount()
    local tbWnd = Ui:GetWindow("UIHud")
    if tbWnd then
      tbWnd.pPanel:SetActive("top/Countdown/Text", true)
      tbWnd.pPanel:Label_SetText("top/Countdown/Text", string.format("Cache %d Audio %d Loading %d Resource %d Runing %d Total %d", nAssetCache, nAudioCache, nLoadingCount, nResourceCache, nRuningCount, nAudioCache + nAssetCache + nLoadingCount + nResourceCache + nRuningCount))
    end
    return true
  end)
end

function Client:AutoGC(nGCTimeInterval)
  if Client.nGCTimer then
    Timer:Close(Client.nGCTimer)
    Client.nGCTimer = nil
  end
  Client.nGCTimer = Timer:Register(nGCTimeInterval * Env.GAME_FPS, function()
    local nNow = GetTime()
    if not Client.nNextDeepGCTime then
      Client.nNextDeepGCTime = nNow + nGCTimeInterval - 1
    end
    if nNow > Client.nNextDeepGCTime then
      Client:DeepGC()
      Client.AssetResourceModule.ReleaseCacheLoader(100)
      Client.AssetResourceModule.ReleaseAudioCacheLoader(100)
      Client.nNextDeepGCTime = nNow + 2 * nGCTimeInterval - 1
    else
      Client.ResourceModule.UnLoadResourceCache(false)
    end
    return true
  end)
end

function Client:RegistServerZoneCheckTimer()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  
  local function fnCheck()
    local nZoneTime = GetZoneTimeSecDiff()
    if Lib:IsInteger(nZoneTime) and 0 ~= nZoneTime and nZoneTime ~= Client.nServerTimeZoneSec then
      Client.nServerTimeZoneSec = nZoneTime
      EventNotify.OnNotify(EventNotify.emNOTIFY_SERVER_TIME_ZONE_UPDATE)
      Log("[Client] ZoneTimeDiff has diff! diff=", nZoneTime)
    end
  end
  
  fnCheck()
  self.nTimer = Timer:Register(Env.GAME_FPS, function()
    fnCheck()
  end)
end

function Client:OnReturnToLogin()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.nServerTimeZoneSec = 0
end
