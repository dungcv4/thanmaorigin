Require("CommonScript/BlissSharing/BlissSharingDef.lua")

function BlissSharing:IsSysOpen(pPlayer)
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\231\179\187\231\187\159\230\156\170\229\188\128\230\148\190")
  end
  if Lib:GetServerOpenDay_4AM() > self.CONDITION.SERVER_OPEN_DAY_LIMIT then
    return false, i18n.EasyGet("\230\180\187\229\138\168\229\183\178\231\187\147\230\157\159")
  end
  if pPlayer.nLevel < self.CONDITION.PLAYER_LEVEL_REQ then
    return false, i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179")
  end
  return true
end

function BlissSharing:CanGetAward(pPlayer, szType, nIndex)
  local bRet, szErr = self:IsSysOpen(pPlayer)
  if not bRet then
    return false, szErr
  end
  if not self.AWARD_TYPE[szType] then
    return false, i18n.EasyGet("\229\165\150\229\138\177\233\133\141\231\189\174\233\148\153\232\175\175")
  end
  local tbSetting = self.SELF_AWARD[szType].AWARD[nIndex]
  if not tbSetting then
    return false, i18n.EasyGet("\229\165\150\229\138\177\233\133\141\231\189\174\233\148\153\232\175\175")
  end
  if 0 == pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_ID_INVITER) then
    return false, i18n.EasyGet("\230\156\170\230\142\165\229\143\151\233\130\128\232\175\183\239\188\140\230\151\160\230\179\149\233\162\134\229\165\150")
  end
  if not self.AWARD_CHECKER[szType](pPlayer, tbSetting.COND) then
    return false, i18n.EasyGet("\230\156\170\232\190\190\229\136\176\233\162\134\229\143\150\230\157\161\228\187\182")
  end
  local nValue = pPlayer.GetUserValue(self.SAVE_GROUP, self.AWARD_SAVE_ID[szType])
  local nFlag = Lib:LoadBits(nValue, nIndex - 1, nIndex - 1)
  if 0 ~= nFlag then
    return false, i18n.EasyGet("\229\165\150\229\138\177\229\183\178\233\162\134\229\143\150\232\191\135")
  end
  return true
end

function BlissSharing:CanGenInviteCode(pPlayer)
  local bRet, szErr = self:IsSysOpen(pPlayer)
  if not bRet then
    return false, szErr
  end
  if pPlayer.nLevel < self.INVITE_AWARD.INVITER_LEVEL_REQ then
    return false, i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}\231\186\167", self.INVITE_AWARD.INVITER_LEVEL_REQ)
  end
  if MODULE_GAMESERVER then
    if pPlayer.tbBlissSharingCache and pPlayer.tbBlissSharingCache.szMyInviteCode then
      return false, i18n.EasyGet("\228\189\160\229\183\178\230\156\137\233\130\128\232\175\183\231\160\129")
    end
  elseif self.tbInviteInfo and self.tbInviteInfo.szMyInviteCode then
    return false, i18n.EasyGet("\228\189\160\229\183\178\230\156\137\233\130\128\232\175\183\231\160\129")
  end
  if 0 ~= pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_ID_INVITER) then
    return false, i18n.EasyGet("\228\189\160\229\183\178\231\187\143\230\142\165\229\143\151\232\191\135\233\130\128\232\175\183")
  end
  return true
end

function BlissSharing:CanPreUseInviteCode(pPlayer, szInviteCode)
  local bRet, szErr = self:IsSysOpen(pPlayer)
  if not bRet then
    return false, szErr
  end
  if pPlayer.nLevel >= self.INVITE_AWARD.INVITER_LEVEL_REQ then
    return false, i18n.EasyGet("\231\173\137\231\186\167\233\156\128\229\176\143\228\186\142{0}\231\186\167", self.INVITE_AWARD.INVITER_LEVEL_REQ)
  end
  return true
end
