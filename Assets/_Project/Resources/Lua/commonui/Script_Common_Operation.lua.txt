function Operation:GoDirection(nDir)
  if not nDir then
    return
  end
  if AFK:IsRunning() then
    AFK:InterruptStep_GoToFightPos()
  end
  AFK:Suspend()
  local nDoing = me.GetDoing()
  if self.nLastGoDir == nDir and nDoing ~= Npc.Doing.do_hover and nDoing ~= Npc.Doing.do_qingkung and nDoing ~= Npc.Doing.stand then
    return
  end
  if TeamMgr:IsFollowTeammateWithMsg(i18n.EasyGet("\229\189\147\229\137\141\230\173\163\229\156\168\232\183\159\230\136\152\239\188\140\228\184\141\229\143\175\232\135\170\231\148\177\231\167\187\229\138\168")) then
    return
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_MOVE_BY_STICK)
  local nPlayerDoing = me.GetDoing()
  if nPlayerDoing == Npc.Doing.dospecialmidmove or nPlayerDoing == Npc.Doing.dospecialpremove then
    return
  end
  local pNpc = me.GetNpc()
  if not pNpc then
    return
  end
  if pNpc.HaveRefFlag(Npc.RefFlagType.npc_forbid_move) then
    return
  end
  if nPlayerDoing == Npc.Doing.common then
    me.StopDirection()
  end
  if nPlayerDoing == Npc.Doing.dospecialmove or nPlayerDoing == Npc.Doing.dospecialstand then
    if nDir <= 0 then
      nDir = 0
    else
      nDir = 1
    end
    RemoteServer.DoSpecialMoveStep(nDir)
    return
  end
  if self.bBlock then
    return
  end
  if not self.bOnJoyStick then
    local nDoing = me.GetDoing()
    if Npc.Doing.do_hover ~= nDoing and Npc.Doing.do_qingkung ~= nDoing then
      Operation:SetPositionEffect(false)
    end
  end
  me.GoDirection(nDir, Env.GAME_FPS * 10)
  me.StartDirection(nDir)
  self.nLastGoDir = nDir
  self.bOnJoyStick = true
  Operation.nKickFrame = 0
  AFKFightSetting:CloseAutoFightRange(false)
end

function Operation:StopGoDir()
  if not me then
    return
  end
  local nPlayerDoing = me.GetDoing()
  if nPlayerDoing == Npc.Doing.dospecialstand or nPlayerDoing == Npc.Doing.dospecialmove or nPlayerDoing == Npc.Doing.dospecialmidmove or nPlayerDoing == Npc.Doing.dospecialpremove then
    return
  end
  local pMeNpc = me.GetNpc()
  if not pMeNpc then
    return
  end
  if pMeNpc.HaveRefFlag(Npc.RefFlagType.npc_forbid_move) then
    return
  end
  if self.bBlock then
    return
  end
  if self.nLastGoDir then
    me.GoDirection(self.nLastGoDir, 2)
  end
  me.StopDirection()
  self:ClearLastDir()
end

function Operation:ClearLastDir()
  self.nLastGoDir = nil
  self.bOnJoyStick = false
end

function Operation:SimpleTap(nNpcID, isFriend)
  local pNpc = KNpc.GetById(nNpcID)
  if not pNpc then
    return
  end
  if pNpc.nPlayerID == me.dwID then
    return
  end
  if pNpc.nKind == Npc.KIND.player then
    if 0 == pNpc.nPlayerID then
      Operation:SetNpcSelected(nNpcID, false)
      if not isFriend then
        Operation:SimpleTapEnemy(pNpc)
      end
    elseif 0 == pNpc.nMirriorPlayerId then
      local tbData = {
        dwRoleId = pNpc.nPlayerID,
        nNpcID = nNpcID,
        szName = pNpc.szName,
        dwKinId = pNpc.dwKinId,
        dwTongId = pNpc.dwTongId,
        szKinOrTongName = Kin:GetKinNameByNpcID(nNpcID),
        nLevel = pNpc.nLevel,
        nZongShiLevel = pNpc.nZongShiLevel,
        nTotalLevel = pNpc.nLevel + pNpc.nZongShiLevel,
        nVipLevel = 0,
        nFaction = pNpc.nFaction,
        nHonorLevel = pNpc.nHonorLevel,
        nBodySex = pNpc.GetBodySex(),
        nPortrait = pNpc.nPortrait
      }
      local tbMember = Kin:GetMemberData(tbData.dwRoleId)
      if tbMember then
        tbData.tbChuanGong = tbMember.tbChuanGong
      end
      local hudWnd = Ui:GetWindow("UIHud")
      if hudWnd then
        hudWnd:ShowRoleHeadPop(tbData)
      end
      Operation:SetNpcSelected(nNpcID, false)
      if not isFriend then
        Operation:SimpleTapEnemy(pNpc)
      end
    end
  elseif pNpc.nKind == Npc.KIND.dialoger then
    Operation:OnDialogerClicked(pNpc.nId)
    Operation:SetNpcSelected(nNpcID, false)
  elseif pNpc.nKind == Npc.KIND.normal and not isFriend then
    Operation:SimpleTapEnemy(pNpc)
  end
end

function Operation:SimpleTapEnemy(pNpc)
  if not pNpc then
    return
  end
  local nTargetId = self:GetLockTargetNpcId()
  if nil == nTargetId then
    nTargetId = pNpc.nId
  end
  if AFK:IsLeadFightRunning() then
    if not AFK:IsLeadFightInFightStep() then
      return
    end
    AFKAutoFight:SetIntentTargetNpc(nTargetId)
    me.GetNpc().ClearSkillCommandCache()
  else
    self:NotLeadFightRunningAction(nTargetId, FightSkill:GetSkillIdByBtnName(me.nFaction, me.nFactionSect))
  end
end

function Operation:SwitchAttackTargetManual()
  local nSelectType = AFKFightSetting.nSelectType
  local nTargetId
  nSelectType = math.max(nSelectType, AFKFightSetting.SelectType.AnyOne)
  nSelectType = math.min(nSelectType, AFKFightSetting.SelectType.Boss)
  if AFK:IsLeadFightRunning() then
    if not AFK:IsLeadFightInFightStep() then
      return
    end
    local nCenterX, nCenterY = AFKAutoFight:ChooseFightCenterPoint()
    if nCenterX < 0 or nCenterY < 0 then
      return
    end
    local tbBlackboard = AFKAutoFight.tbAutoFightAI:GetBlackboard()
    local nCurSkillId = FightSkill:GetSkillIdByBtnName(me.nFaction, me.nFactionSect)
    if not nCurSkillId or nCurSkillId < 0 then
      nCurSkillId = AFKSkillMgr:GetOneCanCastSkill2()
      if not nCurSkillId or nCurSkillId < 0 then
        return
      end
    end
    tbBlackboard.nCurSkillId = nCurSkillId
    nTargetId = AFKSkillMgr:GetNearestNpc(nCurSkillId, AFKFightSetting.nSearchRadius, nSelectType, nCenterX, nCenterY, tbBlackboard.tbTargetNpcBlackList, tbBlackboard.nTargetNpcId)
    if not nTargetId or nTargetId <= 0 then
      tbBlackboard.nTargetNpcId = nil
      return
    end
    AFKAutoFight:SetIntentTargetNpc(nTargetId)
    me.GetNpc().ClearSkillCommandCache()
  else
    local nCurSkillId = FightSkill:GetSkillIdByBtnName(me.nFaction, me.nFactionSect)
    if nCurSkillId and nCurSkillId > 0 then
      if not AFKSkillMgr:IsSkillCanCast(nCurSkillId) then
        nCurSkillId = AFKSkillMgr:GetOneCanCastSkill2()
      end
    else
      nCurSkillId = AFKSkillMgr:GetOneCanCastSkill2()
    end
    if not nCurSkillId or nCurSkillId < 0 then
      return
    end
    nTargetId = self:GetLockTargetNpcId()
    if nil == nTargetId then
      local pNpc = me.GetNpc()
      nTargetId = nTargetId or pNpc.GetNearestNpcIdBySkillRelation(nCurSkillId, nSelectType, false, AFKFightSetting.nSearchRadius, 0, 0, self.nLastTargetId)
      nTargetId = nTargetId or pNpc.GetNearestNpcIdBySkillRelation(nCurSkillId, 0, false, AFKFightSetting.nSearchRadius, 0, 0, self.nLastTargetId)
    end
    if not nTargetId or nTargetId < 0 then
      return
    end
    self:NotLeadFightRunningAction(nTargetId, nCurSkillId)
  end
end

end

    local _, nMeX, nMeY = me.GetWorldPos()

function Operation:OnDialogerClicked(nNpcID, nTaskId)
  local pNpc = KNpc.GetById(nNpcID)
  if not pNpc then
    return
  end
  local nDistance = me.GetNpc().GetDistance(pNpc.nId)
  local nDialogDistance = GetNpcDialogDistance(pNpc.nTemplateId)
  local nMapId, nX, nY = pNpc.GetWorldPos()
  if nDistance < 0 then
    return
  end
  if nDistance > nDialogDistance then
    if AutoPath:IsPathing() then
      return
    end
    AutoPath:GotoAndCallEx(nMapId, nX, nY, function()
      Operation:OnDialogerClicked(nNpcID)
    end, nDialogDistance - Npc.MIN_DISTANCE, nil, pNpc.nTemplateId)
    return
  end
  AFK:WaitTime(2)
  if pNpc.IsAlone() then
    GameSetting:SetGlobalObj(me, pNpc, it)
    if pNpc.szClass and pNpc.szClass ~= "" and not pNpc.bUseDefaultDialog then
      Npc:OnDialog(him.szClass, him.szScriptParam)
    else
      Dialog:ShowDefaultDialog(me, him)
    end
    GameSetting:RestoreGlobalObj()
  else
    RemoteServer.OnSimpleTapNpc(pNpc.nId)
  end
  if pNpc.szTag ~= "NoTurn" then
  end
end

function Operation:Turn2Player(pNpc)
  local _, nX1, nY1 = pNpc.GetWorldPos()
  local _, nX2, nY2 = me.GetWorldPos()
  local nHudu = math.atan(nX1 - nX2, nY1 - nY2)

function Operation:GetSkillHorseLimited(nSkillId)
  if not nSkillId or nSkillId < 0 then
    return false
  end
  local tbSkillInfo = KFightSkill.GetSkillInfo(nSkillId, 1)
  if not tbSkillInfo then
    return false
  end
  if 1 ~= tbSkillInfo.RideLimited then
    return false
  end
  return true
end

function Operation:GetDownHorse(nSkillId)
  self.tbRequestDownHorseTime = self.tbRequestDownHorseTime or {}
  local nLastSendTime = self.tbRequestDownHorseTime[nSkillId] or 0
  local nCurTime = GetTime()
  if nCurTime >= nLastSendTime + self.nRideDownOperationInterval then
    self.tbRequestDownHorseTime[nSkillId] = nCurTime
    SetRideState(0)
  end
end

function Operation:SkillCheckHorseLimited(nSkillID)
  if not nSkillID or nSkillID < 0 then
    return false
  end
  self.tbRequestDownHorseTime = self.tbRequestDownHorseTime or {}
  local tbSkillInfo = KFightSkill.GetSkillInfo(nSkillID, 1)
  if 1 == tbSkillInfo.RideLimited and 1 == me.nRideState then
    local nLastSendTime = self.tbRequestDownHorseTime[nSkillID] or 0
    local nCurTime = GetTime()
    if nCurTime >= nLastSendTime + self.nRideDownOperationInterval then
      self.tbRequestDownHorseTime[nSkillID] = nCurTime
      SetRideState(0)
    end
    return false
  end
  return true
end

function Operation:IsTargetOutLockRange(nSkillId)
  local target = KNpc.GetById(self.nLastTargetId)
  if not target then
    return true
  end
  if not nSkillId or nSkillId <= 0 then
    return true
  end
  local _, nEnemyX, nEnemyY = target.GetWorldPos()
  local _, meX, meY = me.GetWorldPos()
  local nLockRange = FightSkill:GetSkillLockRange(nSkillId)
  return nLockRange <= Lib:GetDistance(meX, meY, nEnemyX, nEnemyY)
end

function Operation:IsNeedChangeTarget(pNpc, nSkillID, bCheckLockRange, nSelectType)
  if nil == bCheckLockRange then
    bCheckLockRange = true
  end
  if not self.nLastTargetId or self.nLastTargetId <= 0 then
    return true
  end
  local bNeedChangeTarget = false
  local nCurFrame = GetFrame()
  if not bNeedChangeTarget then
    local target = KNpc.GetById(self.nLastTargetId)
    bNeedChangeTarget = not target
  end
  bNeedChangeTarget = bNeedChangeTarget or Npc:IsDead(self.nLastTargetId)
  if not bNeedChangeTarget then
    local pTarget = KNpc.GetById(self.nLastTargetId)
    if pTarget and nSelectType == AFKFightSetting.SelectType.Player and 0 == pTarget.nPlayerID then
      bNeedChangeTarget = true
    end
  end
  bNeedChangeTarget = bNeedChangeTarget or not me.CheckSkillAvailable2Npc(nSkillID, self.nLastTargetId)
  bNeedChangeTarget = bNeedChangeTarget or nCurFrame >= self.nLastTargetLockTime + self.nLockTargetFrameTime
  if bCheckLockRange and not bNeedChangeTarget then
    bNeedChangeTarget = self:IsTargetOutLockRange(nSkillID)
    local nTargetId = self:GetLockTargetNpcId()
    if nil ~= nTargetId then
      bNeedChangeTarget = false
    end
  end
  return bNeedChangeTarget
end

function Operation:ShowSkillLockRange(nSkillID, bNeedCheckOnJoyStick)
  if not AFKFightSetting.bOpenAttackRadius then
    return false
  end
  if bNeedCheckOnJoyStick and Operation.bOnJoyStick then

    return false

  if not nSkillID or nSkillID <= 0 then
    return false
  end
  local nLockRange = FightSkill:GetSkillLockRange(nSkillID)
  local nScale = nLockRange * 2 / 800
  Client.RepresentModule.SetSkillLockRangeEffect(true, nScale)
  return true
end

function Operation:IsInAttackRadius(nTargetNpcId, nSkillId)
  if not nTargetNpcId or nTargetNpcId <= 0 then
    return false
  end
  if not nSkillId or nSkillId <= 0 then
    return false
  end
  local nMeNpcId = me.GetNpc().nId
  if nTargetNpcId == nMeNpcId then
    return true
  end
  local pTargetNpc = KNpc.GetById(nTargetNpcId)
  if not pTargetNpc then
    return false
  end
  local nSkillAttackRadius = FightSkill:GetAttackRadius(nSkillId)
  if not nSkillAttackRadius or nSkillAttackRadius < 0 then
    return false
  end
  local _, nSelfX, nSelfY = me.GetWorldPos()
  local _, nTargetNpcX, nTargetNpcY = pTargetNpc.GetWorldPos()
  local nDisRequired = nSkillAttackRadius ^ 2
  local nDisSquare = Lib:GetDistsSquare(nSelfX, nSelfY, nTargetNpcX, nTargetNpcY)
  if nDisRequired >= nDisSquare then
    return true
  end
  return false
end

function Operation:GetTargetBySkill(nSkillID)
  if Operation.nTimerCheckRandomMove then
    Timer:Close(Operation.nTimerCheckRandomMove)
    Operation.nTimerCheckRandomMove = nil
  end
  Operation.nKickFrame = 0
  if self.bStopTarget then
    return 0
  end
  self:SetSkillLockRange(nSkillID)
  local pNpc = me.GetNpc()
  local nDir = self.nLastGoDir or pNpc.GetDir()
  local nTargetId
  local nSelectType = AFKFightSetting.nSelectType
  local bNeedChangeTarget = false
  self.nLastTargetId = self.nLastTargetId or 0
  self.nLastTargetLockTime = self.nLastTargetLockTime or 0
  self.nLastAttackSkillId = nSkillID
  if AFKFightSetting.bOpenAttackRadius then
    self.nLastAttackTime = GetFrame()
  end
  bNeedChangeTarget = self:IsNeedChangeTarget(pNpc, nSkillID, nil, nSelectType)
  local nLockRange = FightSkill:GetSkillLockRange(nSkillID)
  nSelectType = math.max(nSelectType, AFKFightSetting.SelectType.AnyOne)
  nSelectType = math.min(nSelectType, AFKFightSetting.SelectType.Boss)
  if bNeedChangeTarget then
    if self.bOnJoyStick and not nTargetId then
      nTargetId = pNpc.GetNearestNpcByDirAndSkillRelation2(nDir, self.nJoyStickFindTargetDirDiff, nLockRange, nSkillID, nSelectType, false, 0, 0, 0)
    end
    nTargetId = nTargetId or pNpc.GetNearestNpcIdBySkillRelation(nSkillID, nSelectType, false, nLockRange, 0, 0)
    if self.bOnJoyStick and not nTargetId then
      nTargetId = pNpc.GetNearestNpcByDirAndSkillRelation2(nDir, self.nJoyStickFindTargetDirDiff, nLockRange, nSkillID, 0, false, 0, 0, 0)
    end
    nTargetId = nTargetId or pNpc.GetNearestNpcIdBySkillRelation(nSkillID, 0, false, nLockRange, 0, 0)
    self:SetNpcAttackNoneSelected(self.nLastTargetId, 1)
    self:SetNpcAttackNoneSelected(self.nLastTargetId, 2)
    self.nLastTargetId = nTargetId
    self.nLastTargetLockTime = GetFrame()
  end
end

function Operation:GetAttackTarget(nSkillID)
  self:GetTargetBySkill(nSkillID)
  if not self.nLastTargetId or self.nLastTargetId <= 0 then
    return 0
  end
  if not self:IsInAttackRadius(self.nLastTargetId, nSkillID) then
    return 0
  end
  Operation:SetNpcAttackSelectedByIndex(self.nLastTargetId, 1)
  Operation:SetNpcAttackNoneSelected(self.nLastTargetId, 2)
  return self.nLastTargetId
end

function Operation:JoyStickAttack(nSkillID)
  self:GetTargetBySkill(nSkillID)
  local nDir = self.nLastGoDir or me.GetNpc().GetDir()
  if not self.nLastTargetId or self.nLastTargetId <= 0 then
    local tbSkillInfo = FightSkill:GetFactionSkillInfo(me.nFaction, me.nFactionSect, nSkillID)
    if tbSkillInfo and tbSkillInfo.CastType == FightSkill.SkillCasterType.target then
      return
    end
    Operation:UseSkillToDir(nSkillID, nDir, 1)
    return
  end
  if not self:IsInAttackRadius(self.nLastTargetId, nSkillID) then
    if not self.bOnJoyStick then
      local target = KNpc.GetById(self.nLastTargetId)
      if not target then
        Operation:UseSkillToDir(nSkillID, nDir)
        return
      end
      AFKSkillMgr:SetIntentTargetNpc(self.nLastTargetId)
      local _, nX, nY = me.GetWorldPos()
      local nEnemyMapId, nEnemyX, nEnemyY = target.GetWorldPos()
      if nEnemyX and nEnemyY then
        me.bIsForbidUpHorseFlag = true
        local nSkillAttackRadius = FightSkill:GetAttackRadius(nSkillID)
        AutoPath:GotoAndCallEx(nEnemyMapId, nEnemyX, nEnemyY, nil, nSkillAttackRadius * 2 / 3)
      end
      return
    end
    self:StopGoto()
    Operation:UseSkillToDir(nSkillID, nDir, 1)
    return
  end
  Operation:SetNpcAttackSelectedByIndex(self.nLastTargetId, 1)
  Operation:SetNpcAttackNoneSelected(self.nLastTargetId, 2)
  local pNpc = me.GetNpc()
  if not pNpc.IsTargetInDirRange(self.nLastTargetId, self.nSelectLastTargetDirRange) then
    if not self.bOnJoyStick then
      Operation:UseSkillToNpc(nSkillID, self.nLastTargetId)
      self.nLastTargetLockTime = GetFrame()
      return
    end
    local tbSkillInfo = FightSkill:GetSkillSetting(nSkillID, 1)
    if tbSkillInfo and tbSkillInfo.SkillType == FightSkill.SkillTypeDef.skill_type_melee then
      Operation:UseSkillToDir(nSkillID, nDir, 1)
    else
      Operation:UseSkillToNpc(nSkillID, self.nLastTargetId)
      self.nLastTargetLockTime = GetFrame()
    end
    return
  end
  Operation:UseSkillToNpc(nSkillID, self.nLastTargetId)
  self.nLastTargetLockTime = GetFrame()
end

function Operation:SetNpcSelected(nNpcID, bIsAttack)
  if nNpcID == me.GetNpc().nId then
    return
  end
  local eDoing = me.GetDoing()
  if eDoing == Npc.Doing.stand then
    me.GetNpc().SetDirToNpc(nNpcID or 0)
  end
  local nCurSelectNpcId = Client.NpcManager.GetTargetSelectNpcID(0)
  if not nNpcID or 0 == nNpcID or nNpcID == nCurSelectNpcId then
    return
  end
  local pNpc = KNpc.GetById(nNpcID)
  if not pNpc then
    return
  end
  if pNpc.nKind == Npc.KIND.god or pNpc.nKind == Npc.KIND.sporter or pNpc.nKind == Npc.KIND.novertical then
    return
  end
  local npcRep = Client.NpcManager.GetNpc(nNpcID)
  if not npcRep then
    return
  end
  me.nAttackSkillId = me.nAttackSkillId or FightSkill:GetSkillIdByBtnName(me.nFaction, me.nFactionSect)
  local bCanAttack = me.CheckSkillAvailable2Npc(me.nAttackSkillId, nNpcID) and true or false
  if bIsAttack then
    if me.nAttackSkillId > 0 and bCanAttack then
      npcRep:SetSelectedEffect(true, true, 0)
    end
    return
  end
  if not bCanAttack then
    npcRep:SetSelectedEffect(true, false, 0)
    return
  end
end

function Operation:UseSkillToNpc(nSkillID, nNpcID)
  if not nNpcID or 0 == nNpcID then
    nNpcID = me.GetNpc().nId
  end
  if Operation.tbPublicFrame and Operation.tbPublicFrame[nSkillID] and Operation.tbPublicFrame[nSkillID] > 0 then
    return false
  end
  local bRet = me.UseSkill(nSkillID, -1, nNpcID)
  self:OnUseSkill(nNpcID, nSkillID)
  if self.bOnJoyStick then
    self:GoDirection(self.nLastGoDir)
    self.nLastGoDir = nil
  end
  return bRet
end

function Operation:OnUseSkill(nNpcID, nSkillID)
  local tbCheck = self.tbCheckSkill[nSkillID]
  if nil ~= tbCheck then
    local nCheckNpcId = tbCheck.nCheckNpcId
    local OnUseSkillCallback = tbCheck.OnUseSkillCallback
    OnUseSkillCallback()
    self.tbCheckSkill[nSkillID] = nil
  elseif nil ~= nNpcID then
    local tbCheck = self.tbCheckNpc[nNpcID]
    if nil ~= tbCheck then
      local nCheckNpcId = tbCheck.nCheckNpcId
      local nCheckSkillID = tbCheck.nCheckSkillID
      if nCheckSkillID == nSkillID then
        local OnUseSkillCallback = tbCheck.OnUseSkillCallback
        OnUseSkillCallback()
        self.tbCheckNpc[nNpcID] = nil
      end
    end
  end
end

  else
    RemoteServer.OnSimpleTapNpc(pNpc.nId)
  end
end

function Operation:XHKeyOn(...)
  if not Login.bEnterGame then
    return
  end
  local tbHKOnState = {}
  for _, szKey in ipairs({
    ...
  }) do
    local szKeyOnType = CustomHotkey.HotkeyBinding[szKey].KeyOnType
    tbHKOnState[szKeyOnType] = tbHKOnState[szKeyOnType] or {}
    tbHKOnState[szKeyOnType][szKey] = 1
  end
  for szKeyOnType, v in pairs(tbKeyOnTypeFun) do
    local tbLast = self.tbLastHKOnState and self.tbLastHKOnState[szKeyOnType]
    local tbNow = tbHKOnState[szKeyOnType]
    if tbLast and not tbNow then
      if v.Leave then
        v.Leave(self)
      end
    elseif not tbLast and tbNow and v.Enter ~= nil then
      v.Enter(self, tbNow)
    end
    if tbNow and v.On then
      v.On(self, tbNow)
    end
  end
  self.tbLastHKOnState = tbHKOnState
end

  for i = 1, #tbArg, 2 do

end

function Operation:XHStateValueProcess(tbNew, szLast, tbOnProcess)
  local tbCache = {}
  local tbLast = self[szLast] or {}
  for k, v in pairs(tbLast) do
    tbCache[k] = -1
  end
  for k, v in pairs(tbNew) do
    tbCache[k] = (tbCache[k] or 0) + 1
  end
  for k, v in pairs(tbCache) do
    local tbFunc = tbOnProcess[k]
    if nil ~= tbFunc then
      if 1 == v then
        if tbFunc.Enter then
          tbFunc.Enter(self, tbNew[k])
        end
      elseif 0 == v then
        if tbFunc.On then
          tbFunc.On(self, tbNew[k])
        end
      elseif tbFunc.Leave then
        tbFunc.Leave(self)
      end
    end
  end
  self[szLast] = tbNew
end

  local bResult = false
  local nLogicFrame = GetFrame()

  local nPublicFrame = 0
  if nil ~= self.tbPublicFrame and nil ~= self.tbPublicFrame[nSkillID] and self.tbPublicFrame[nSkillID] > 0 then
