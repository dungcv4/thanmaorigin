BubbleTalk:LoadSetting()

function BubbleTalk:Talk(nBubbleId)
  local tbConfig = BubbleTalk.tbSetting[nBubbleId]
  if tbConfig then
    local szContent
    local nDuration = 5
    local nTryCount = 0
    while nil == szContent or "" == szContent do
      local index = math.random(1, 10)
      szContent = tbConfig["Talk" .. index]
      nDuration = tbConfig["Time" .. index] or 5
      nTryCount = nTryCount + 1
      if 20 == nTryCount then
        szContent = tbConfig.Talk1
        nDuration = tbConfig.Time1 or 5
      end
    end
    him.BubbleTalk(i18n.Parse(szContent), tostring(nDuration))
  else
    Log("don't exist Bubble id", nBubbleId)
  end
end
