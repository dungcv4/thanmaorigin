function PlayerTitle:LoadSetting()
  self.tbTitleCategory = Lib:LoadTabFileEx("Setting/Player/TitleCategory.tab", "ds", "nCategoryId", {
    "nCategoryId",
    
    "szCategoryName"
  })
  self.tbTitleTemplate = Lib:LoadTabFileEx("Setting/Player/Title.tab", "dsdddddssdddddddddddddd", "TitleID", {
    "TitleID",
    "Name",
    "nCategoryId",
    "IsCustomName",
    "Faction",
    "Quality",
    "Timeout",
    "Desc",
    "Source",
    "bShow",
    "bShowNoGet",
    "bShowGetTips",
    "AttrGroup",
    "AttrLevel",
    "AttributePower",
    "FightPower",
    "bAutoUse",
    "ColorID",
    "GTopColorID",
    "GBottomColorID",
    "OutlineColorID",
    "OutlineAlpha",
    "BGEffectResID"
  })
end

PlayerTitle:LoadSetting()

function PlayerTitle:GetTitleTemplate(nTitleID)
  return self.tbTitleTemplate[nTitleID]
end

function PlayerTitle:GetAllTitle()
  return self.tbTitleTemplate
end

function PlayerTitle:GetCategoryName(nCategoryId)
  return i18n.Parse(self.tbTitleCategory[nCategoryId].szCategoryName)
end

function PlayerTitle:GetTitleFightPower(nTitleID)
  return self.tbTitleTemplate[nTitleID] and self.tbTitleTemplate[nTitleID].FightPower
end

function PlayerTitle:GetTitleAttributePower(nTitleID)
  return self.tbTitleTemplate[nTitleID] and self.tbTitleTemplate[nTitleID].AttributePower
end

function PlayerTitle:IsTitleHasAttr(nTitleID)
  local tbInfo = self.tbTitleTemplate[nTitleID]
  if not tbInfo then
    return
  end
  return tbInfo.AttrGroup > 0 and tbInfo.AttrLevel > 0
end

function PlayerTitle:IsCanAdd(pPlayer, nTitleID)
  local tbInfo = self:GetTitleTemplate(nTitleID)
  if not tbInfo then
    return
  end
  if tbInfo.Faction > 0 then
    local nFaction = pPlayer.nFaction
    if nFaction ~= tbInfo.Faction then
      return
    end
  end
  return true
end

function PlayerTitle:IsCanShow(nTitleID)
  local tbInfo = self.tbTitleTemplate[nTitleID]
  if not tbInfo then
    return
  end
  return tbInfo.bShow > 0
end

function PlayerTitle:GetShowTitleCount()
  local nCount = 0
  for _, tbInfo in pairs(self.tbTitleTemplate) do
    if tbInfo.bShow > 0 then
      nCount = nCount + 1
    end
  end
  return nCount
end

function PlayerTitle:GetTitleAnimation(nTitleID)
  local tbInfo = self:GetTitleTemplate(nTitleID)
  return tbInfo and tbInfo.ImgPrefix
end

function PlayerTitle:IsSpecailTitle(nTitleID)
  return self:GetTitleAnimation(nTitleID)
end

function PlayerTitle:IsReplace(pPlayer, nTitleID, nTimeout)
  local tbData = self:GetPlayerTitleData(pPlayer)
  if not (tbData and tbData.tbAllTitle) or not tbData.tbAllTitle[nTitleID] then
    return false, false, 0
  end
  if not nTimeout then
    return true, true, nTitleID
  end
  if -1 == tbData.tbAllTitle[nTitleID].nEndTime then
    return true, false, 0
  end
  local tbTitleTemp = self:GetTitleTemplate(nTitleID)
  local nEndTime
  if nTimeout and nTimeout > 0 then
    nEndTime = nTimeout
  elseif tbTitleTemp.Timeout and tbTitleTemp.Timeout > 0 then
    nEndTime = tbTitleTemp.Timeout + GetTime()
  end
  if nEndTime > tbData.tbAllTitle[nTitleID].nEndTime then
    return true, true, nTitleID
  end
  return true, false, 0
end
