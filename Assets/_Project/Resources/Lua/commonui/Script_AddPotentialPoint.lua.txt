Require("Script/EventSystem/EventNotify.lua")

function PotentialPoint:OnCheckRedPoint()
  if HanWuYiJi:CheckInHanWuYiJi() then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.PotentialPointLeft)
    return
  end
  if me.nLeftPotentialPoint > 0 then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.PotentialPointLeft)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.PotentialPointLeft)
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, PotentialPoint)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_POTENTIAL_INFO, PotentialPoint)
EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, PotentialPoint.OnCheckRedPoint, PotentialPoint)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_POTENTIAL_INFO, PotentialPoint.OnCheckRedPoint, PotentialPoint)
