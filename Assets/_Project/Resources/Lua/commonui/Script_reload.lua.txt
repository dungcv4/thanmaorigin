function Reload:Update(from, to)
  if not from or not to then
    return false
  end
  local funName, func
  for i = from, to - 1 do
    funName = string.format("Update%dTo%d", from, i + 1)
    func = self[funName]
    if func then
      Lib:CallBack({func, self})
    else
      Log("Reload:Update error, function not exist: ", funName)
    end
  end
  return true
end

function Reload.Reinstall()
end

function Reload.Restart()
end

function Reload.Quit()
end

function Reload:ShowReinstallMsgBox(szMsg, szBtnOk, szBtnCancel)
  Ui:OpenWindow("RestartMessageBox", szMsg, {
    {
      Reload.Reinstall
    },
    {
      Reload.Quit
    }
  }, {szBtnOk, szBtnCancel})
end

function Reload:ShowRestartMsgBox(szMsg, szBtnOk, szBtnCancel)
  if szBtnCancel then
    Ui:OpenWindow("RestartMessageBox", szMsg, {
      {
        Reload.Restart
      },
      {
        Reload.Quit
      }
    }, {szBtnOk, szBtnCancel})
  else
    Ui:OpenWindow("RestartMessageBox", szMsg, {
      {
        Reload.Quit
      }
    }, {szBtnOk})
  end
end

function Reload:ShowQuitMsgBox(szMsg, szBtnOk, szBtnCancel)
  Ui:OpenWindow("RestartMessageBox", szMsg, {
    {
      Reload.Quit
    },
    {
      Reload.Quit
    }
  }, {szBtnOk, szBtnCancel})
end
