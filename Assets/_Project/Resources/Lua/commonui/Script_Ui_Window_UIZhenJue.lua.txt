tbWnd.tbOnClick = {
  btnAddNormal = function(self)
    self.pPanel:SetActive(tbControls.wndJumpTo, true)
  end,
  btnAddPerfect = function(self)
    self.pPanel:SetActive(tbControls.wndJumpTo, true)
  end,
  btnClose = function(self)
    Ui:OpenWindow("UIZhenFa_New")
    self:Close()
  end,
  btnLevelUp = function(self)
    if not self.nLWType then
      return me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\139\190\233\128\137\233\162\134\230\130\159\228\184\185"))
    end
    ZhenFa:ZFZhenJueLingWuReqC(self.nZJSelect, self.nLWType)

tbWnd.tbOnToggleClick = {
  LWDToggle = function(self, isOn)
    if isOn then
      self.pPanel:Toggle_SetChecked(self.LWDPreToggle, false)
      self.nLWType = ZhenFa.nNormalLingWu
    else
      self.nLWType = nil
    end
  end,
  LWDPreToggle = function(self, isOn)
    if isOn then
      self.pPanel:Toggle_SetChecked(self.LWDToggle, false)
      self.nLWType = ZhenFa.nPerfectLingWu
    else
      self.nLWType = nil
    end
  end
}

function tbWnd:OnCreate()
  self.tbZhenJueSV = Ui:NewScrollView(self, tbControls.ZhenJueTemplate, self.OnZhenJueClick)
  self.tbZhenJueItemGrid = Ui:NewExistItemGrid(self, tbControls.ZhenJueItemGrid)
  self.tbLWDItemGrid = Ui:NewExistItemGrid(self, tbControls.LWDItemGrid)
  self.tbLWDItemGrid:SetClickEnable(true)
  self.tbLWDPreItemGrid = Ui:NewExistItemGrid(self, tbControls.LWDPreItemGrid)
  self.tbLWDPreItemGrid:SetClickEnable(true)
  self.pPanel:Toggle_SetChecked(tbControls.LWDToggle, false)
  self.pPanel:Toggle_SetChecked(tbControls.LWDPreToggle, false)
  self.tbWayScrollView = Ui:NewScrollView(self, tbControls.WayTemplate)
  self:SetClickEmptyToClose(tbControls.wndJumpTo)
end

function tbWnd:OnOpen()
  self.pPanel:Toggle_SetChecked(tbControls.LWDPreToggle, true)
  self:UpdateZhenJueSV()
  self.tbZhenJueSV:SelectElement(1)
  local tbEventList = {}
  for nRedPoint = RedPointMgr.EVENT.ZhenFaZJLevelUpBegin, RedPointMgr.EVENT.ZhenFaZJLevelUpEnd do
    table.insert(tbEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUp, tbEventList)
  self:UpdateGettingWay()
end

function tbWnd:UpdateView()
  self:UpdateZhenJueSV()
  self:UpdateDetailInfo()
end

function tbWnd:OnClose()
  if self.nZJSelect then
    local tbOldElement = self.tbZhenJueSV:GetElement(self.nZJSelect)
    tbOldElement.pPanel:SetActive("imgSelected", false)
  end
  self.nZJSelect = nil
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUp)
  LocalData:SetRoleNumber(ZhenFa.szZhenJueClick, GetTime())
  for nHoleIndex = 1, ZhenFa.nMaxZhenJue do
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZhenFaZJLevelUpBegin + nHoleIndex - 1)
  end
end

function tbWnd:UpdateZhenJueSV()
  local nCurPutOnPos, _ = ZhenFa:GetZhenFaCurPutOnPos(me)
  self.tbShowZJId = ZhenFa:GetZhenJueAttrShowItemIdList(me, nCurPutOnPos)
  self.tbZhenJueSV:Reset()
  for i = 1, #self.tbShowZJId do
    self.tbZhenJueSV:SetElement(i, function(tbElement, nIndex)
      local pPanel = tbElement.pPanel
      local nItemId = self.tbShowZJId[nIndex]
      local nLevel = ZhenFa:GetZhenJueHoleLevel(me, nIndex)
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(nItemId)
      itemGrid:SetLabelSuffixText("+" .. tostring(nLevel))
      itemGrid:SetClickEnable(true, function()
        self:OnZhenJueClick(nIndex, tbElement)
      end)
      RedPointMgr:DynamicRegisteRedPoint(pPanel, "Main", {
        RedPointMgr.EVENT.ZhenFaZJLevelUpBegin + i - 1
      })
    end)
  end
end

function tbWnd:OnZhenJueClick(nIndex, tbElement)
  if self.nZJSelect then
    local tbOldElement = self.tbZhenJueSV:GetElement(self.nZJSelect)
    tbOldElement.pPanel:SetActive("imgSelected", false)
  end
  self.nZJSelect = nIndex
  tbElement.pPanel:SetActive("imgSelected", true)
  self:UpdateDetailInfo()
end

function tbWnd:UpdateDetailInfo()
  local nCurPutOnPos, _ = ZhenFa:GetZhenFaCurPutOnPos(me)
  local nItemId = self.tbShowZJId[self.nZJSelect]
  local szName = Item:GetName(nItemId, me)
  self.pPanel:Label_SetText(tbControls.txtZhenJueName, szName)
  self.tbZhenJueItemGrid:SetItemByTemplate(nItemId)
  self.tbZhenJueItemGrid:SetClickEnable(true, function()
    Item:ShowItemDetail({
      nTemplate = nItemId,
      tbItemExtParams = {
        nPos = nCurPutOnPos,
        nHole = self.nZJSelect
      }
    })
  end)
  local nLevel = ZhenFa:GetZhenJueHoleLevel(me, self.nZJSelect)
  local nExp = ZhenFa:GetZFZhenJueExp(me, self.nZJSelect)
  local nMaxLevel, nMaxTimeFrameLevel, nMaxQualityLevel = ZhenFa:GetZFZhenJueMaxLevel()
  local nNextLevel
  if nLevel >= nMaxLevel then
    self.pPanel:SetActive(tbControls.btnLevelUp, false)
    self.pPanel:SetActive(tbControls.txtMaxLevel, true)
    nNextLevel = nLevel
    self.pPanel:Sprite_SetFill(tbControls.ProgressBar, 1)
    if nMaxTimeFrameLevel <= nMaxLevel then
      local szTimeFrame = ZhenFa:GetNextLevelTimeframe(2, nMaxLevel)
      local nNextWorldLevel = Calendar:GetWorldLevelByTimeframe(szTimeFrame)
      if nNextWorldLevel > 0 then
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\230\156\128\233\171\152\231\173\137\231\186\167\239\188\140\229\188\128\230\148\190{0}\231\173\137\231\186\167\228\184\138\233\153\144\229\144\142\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\239\188\129", Lib:GetLevelText(nNextWorldLevel)))
      else
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\231\186\167\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
      end
    elseif nMaxQualityLevel <= nMaxLevel then
      local nTipGrade = ZhenFa:GetNextLevelTipsGrade(2, nMaxLevel)
      if nTipGrade > 0 then
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\230\156\128\233\171\152\231\173\137\231\186\167\239\188\140\232\163\133\229\164\135\228\187\187\230\132\143\231\137\185\231\186\167{0}\233\152\182\233\152\181\230\179\149\229\144\142\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\239\188\129", nTipGrade))
      else
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\231\186\167\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
      end
    else
      self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\231\186\167\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
    end
  else
    self.pPanel:SetActive(tbControls.txtMaxLevel, false)
    self.pPanel:SetActive(tbControls.btnLevelUp, true)
    nNextLevel = nLevel + 1
    local nNextLevelExp = ZhenFa.ZhenJueLevelInfo[nNextLevel][6]
    local nValue = nExp / nNextLevelExp
    self.pPanel:Sprite_SetFill(tbControls.ProgressBar, nValue)
  end
  self.pPanel:Label_SetText(tbControls.txtLevelNow, i18n.EasyGet("{0}\231\186\167", nLevel))
  self.pPanel:Label_SetText(tbControls.txtLevelNext, i18n.EasyGet("{0}\231\186\167", nNextLevel))
  local nAttrId = me.GetUserValue(ZhenFa.tbSaveGroupKey[nCurPutOnPos], self.nZJSelect + ZhenFa.ZHENJUE_ATTR_BEGIN)
  local tbAttrCur = ZhenFa.ZhenJueAttrInfo[nAttrId][nLevel][1]
  local tbAttrNext = ZhenFa.ZhenJueAttrInfo[nAttrId][nNextLevel][1]
  local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCur[1], {
    table.unpack(tbAttrCur, 2)
  })
  local _, szValueNext = FightSkill:GetMagicDescSplit(tbAttrNext[1], {
    table.unpack(tbAttrNext, 2)
  })
  self.pPanel:Label_SetText(tbControls.txtAttrTip, szAttribMsg)
  local szSubValue = FightSkill:StringValueSubResult(szValue, szValueNext)
  self.pPanel:Label_SetText(tbControls.txtAttrValueNow, szValue)
  self.pPanel:Label_SetText(tbControls.txtAttrValueNext, szValueNext)
  if "" == szSubValue then
    self.pPanel:Label_SetText(tbControls.txtAttrValueUp, "")
    self.pPanel:SetActive(tbControls.imgArrow2, false)
  else
    self.pPanel:Label_SetText(tbControls.txtAttrValueUp, szSubValue)
    self.pPanel:SetActive(tbControls.imgArrow2, true)
  end
  local nLWDItemTemplateId, nLWDItemNum = ZhenFa.ZhenJueLevelInfo[nNextLevel][2], ZhenFa.ZhenJueLevelInfo[nNextLevel][3]
  self.tbLWDItemGrid:SetItemByTemplate(nLWDItemTemplateId)
  local nLWDCurNum = me.GetItemCountByTypeInBag(nLWDItemTemplateId)
  szName = Item:GetName(nLWDItemTemplateId, me)
  self.pPanel:Label_SetText(tbControls.txtLWDName, szName)
  self.pPanel:Label_SetText(tbControls.txtLWDNum, string.format("%d/%d", nLWDCurNum, nLWDItemNum))
  if nLWDItemNum <= nLWDCurNum then
    self.pPanel:Label_SetColorByName(tbControls.txtLWDNum, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtLWDNum, "NotEnough")
  end
  local nLWDPreItemTemplateId, nLWDPreItemNum = ZhenFa.ZhenJueLevelInfo[nNextLevel][4], ZhenFa.ZhenJueLevelInfo[nNextLevel][5]
  self.tbLWDPreItemGrid:SetItemByTemplate(nLWDPreItemTemplateId)
  szName = Item:GetName(nLWDPreItemTemplateId, me)
  self.pPanel:Label_SetText(tbControls.txtLWDPreName, szName)
  local nLWDPreCurNum = me.GetItemCountByTypeInBag(nLWDPreItemTemplateId)
  self.pPanel:Label_SetText(tbControls.txtLWDPreNum, string.format("%d/%d", nLWDPreCurNum, nLWDPreItemNum))
  if nLWDPreItemNum <= nLWDPreCurNum then
    self.pPanel:Label_SetColorByName(tbControls.txtLWDPreNum, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtLWDPreNum, "NotEnough")
  end
end

function tbWnd:UpdateGettingWay()
  self.tbWayScrollView:Clear()
  for i, tbWayInfo in ipairs(self.GETTING_WAY) do
    self.tbWayScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtName", tbWayInfo.szTitle())
      tbElement.pPanel:Label_SetText("txtDesc", tbWayInfo.szDesc())
      if tbWayInfo.func then
        tbElement.pPanel:SetActive("imgArrow", true)
        Ui:Button_BindEvent(tbElement.pPanel, "", function()
          tbWayInfo.func()
          self.pPanel:SetActive(tbControls.wndJumpTo, false)
        end)
      else
        tbElement.pPanel:SetActive("imgArrow", false)
      end
    end)
  end
end
