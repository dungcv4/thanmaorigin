local tbWnd = Ui:CreateWindow("UIXiaoYaoLu")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtUnlockedValue = "node/imgBG/Content/Unlocked/txtUnlockedValue",
  txtRankingValue = "node/imgBG/Content/Ranking/txtRankingValue",
  Template = "node/imgBG/Content/List/Viewport/Content/Template",
  btnRanking = "node/imgBG/Content/btnRanking"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnRanking = function(self)
    Ui:OpenWindow("UIRanking", "XoyoGameRecord")
  end
}

function tbWnd:OnCreate()
  self.tbCardSV = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(pItem)
  self.pPanel:Label_SetText(tbControls.txtRankingValue, "")
  local tbCardId = XoyoGame.XoyoGameRecordCfg.tbCardId
  local nMaxCount = #tbCardId
  local nCount = 0
  for i = 1, nMaxCount do
    local nItemTemplateId = pItem.GetIntValue(i)
    local szName = Item:GetName(tbCardId[i], me)
    self.tbCardSV:SetElement(i, function(tbElement, nIndex)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("Locked/txtNameLocked", szName)
      pPanel:Label_SetText("Unlocked/txtNameUnlocked", szName)
      if nItemTemplateId == tbCardId[i] then
        pPanel:SetActive("Locked", false)
        pPanel:SetActive("Unlocked", true)
        nCount = nCount + 1
      else
        pPanel:SetActive("Unlocked", false)
        pPanel:SetActive("Locked", true)
      end
    end)
  end
  self.pPanel:Label_SetText(tbControls.txtUnlockedValue, string.format("%d/%d", nCount, nMaxCount))
  XoyoGame:XYGRecordRankReqC()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateMyRank(nPosition)
  if not nPosition then
    self.pPanel:Label_SetText(tbControls.txtRankingValue, i18n.EasyGet("\230\156\170\228\184\138\230\166\156"))
  else
    self.pPanel:Label_SetText(tbControls.txtRankingValue, nPosition)
  end
end
