function TreasureChest:LoadFile()
  self.tbTypeCfg = Lib:LoadTabFileEx("Setting/BaiBaoXiang/BBXAwardType.tab", "dsddd", "AwardType", {
    "AwardType",
    "Name",
    "ShowTemplateId1",
    "ShowTemplateId2",
    "ShowTemplateId3"
  })
  self.tbTypeCfg = SetReadOnly(self.tbTypeCfg)
  local tbAwardValue = Lib:LoadTabFileEx("Setting/BaiBaoXiang/BBXAward.tab", "dddddd", nil, {
    "AwardType",
    "Level",
    "AwardId",
    "BetTypeKey1",
    "BetTypeKey2",
    "BetTypeKey3"
  })
  self.tbAwardValue = {}
  for _, tbInfo in pairs(tbAwardValue or {}) do
    self.tbAwardValue[tbInfo.AwardType] = self.tbAwardValue[tbInfo.AwardType] or {}
    local tbTypeInfo = self.tbAwardValue[tbInfo.AwardType]
    tbTypeInfo[tbInfo.Level] = {
      nAwardId = tbInfo.AwardId,
      tbBetTypeKey = {
        tbInfo.BetTypeKey1,
        tbInfo.BetTypeKey2,
        tbInfo.BetTypeKey3
      }
    }
  end
  self.tbAwardValue = SetReadOnly(self.tbAwardValue)
end

function TreasureChest:GetLevelColor(nLevel)
  local nQuality = self.tbLvQuality[nLevel]
  local _, szColor = Item:GetQualityColor(nQuality)
  return szColor or ""
end

function TreasureChest:GetMultColor(nAwardNum)
  local tbBetTypeSetting = self.tbBetTypeSetting
  local nQuality = 0
  if nAwardNum == tbBetTypeSetting[1][1] then
    nQuality = 2
  elseif nAwardNum == tbBetTypeSetting[2][1] then
    nQuality = 3
  elseif nAwardNum == tbBetTypeSetting[3][1] then
    nQuality = 4
  end
  local _, szColor = Item:GetQualityColor(nQuality)
  return szColor or ""
end

function TreasureChest:CheckAllowGetResult(pPlayer, nBetType)
  if not (pPlayer and nBetType) or self:CheckState() == false then
    return 0
  end
  local nInterval = pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_INTERVAL)
  if GetTime() - nInterval < self.NEED_TIME_INTERVAL then
    return 0
  end
  local nUseCount = (self.tbBetTypeSetting[nBetType] or {})[2]
  if not nUseCount then
    return 0
  end
  local bOverflow = pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_OVERFLOW)
  if 1 == bOverflow then
    return 0
  end
  local nTimes = pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_TIMES)
  local bContinue = pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_CONTINUE)
  if nTimes > 0 and 0 == bContinue then
    return 0
  end
  local nHaveBetType = pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_BET_TYPE)
  if nHaveBetType <= 0 then
    local nCurNum = pPlayer.GetItemCountByTypeInBag(self.nShellTemplateId)
    if nUseCount > nCurNum then
      return 0
    end
  end
  return 1
end

function TreasureChest:CheckState()
  if 1 ~= GetTimeFrameState("OpenDay2B") then
    return false
  end
  if MODULE_GAMECLIENT then
    return OnOffController:Check(OnOffController.KEY.TreasureChest)
  end
  return self.bOpenTreasureChest
end
