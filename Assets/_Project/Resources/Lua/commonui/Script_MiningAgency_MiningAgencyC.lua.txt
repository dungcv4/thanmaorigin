function MiningAgency:GetErrorMsg(nErrorCode, ...)
  local szMsg = self.ErrorMsg[nErrorCode]
  if szMsg then
    return i18n.Get(szMsg, ...)
  else
    assert(szMsg, nErrorCode)
  end
end

function MiningAgency:ReqSetMoney(nMoney)
  local nLackOfMoney = nMoney - Currency:GetMoney(me, "Gold")
  if nLackOfMoney > 0 then
    me.CenterMsg(i18n.EasyGet("\232\190\147\229\133\165\229\164\177\232\180\165\239\188\140\229\189\147\229\137\141\231\188\186\229\176\145{0}{1}", nLackOfMoney, Currency:GetName("Gold")))
    GainCurrency:ReadyForJump("Gold")
    return
  end
  RemoteServer.RequestMiningAgency("C2S_SetMoney", nMoney)
  return true
end

function MiningAgency:OnStartMining(nProcessorId, tbOtherInfo, nMeType)
  local nOwnerId, nHelperId
  if nMeType == self.OWNER then
    nOwnerId = me.dwID
    nHelperId = tbOtherInfo.nPlayerId
  else
    nOwnerId = tbOtherInfo.nPlayerId
    nHelperId = me.dwID
  end
  self.pProcessor = self:CreateNewProcessor(nOwnerId, nHelperId)
  self.pProcessor:SetProcessorId(nProcessorId)
  Ui:OpenWindow("UIMiningAgency", tbOtherInfo, nMeType)
end

function MiningAgency:OnCancel(nCancelPlayerId, szCancelPlayerName)
  if nCancelPlayerId ~= me.dwID or szCancelPlayerName ~= me.szName then
    me.CenterMsg(i18n.Get(self.MSG.OTHER_SIDE_CANCEL, szCancelPlayerName), true)
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_MININGAGENCY_ONCANCEL, nCancelPlayerId, szCancelPlayerName)
  self:Clear()
end

function MiningAgency:OnChangeMoney(nPlayerId, nMoney)
  if not self.pProcessor then
    return
  end
  self.pProcessor:SetPlayerMiningMoney(nPlayerId, nMoney)
  EventNotify.OnNotify(EventNotify.emNOTIFY_MININGAGENCY_ONCHANGEMONEY, nPlayerId, nMoney)
end

function MiningAgency:OnStateLock(nPlayerId, szPlayerName)
  if not self.pProcessor then
    return
  end
  self.pProcessor:SetPlayerMiningState(nPlayerId, self.LOCK_STATE)
  EventNotify.OnNotify(EventNotify.emNOTIFY_MININGAGENCY_ONCHANGESTATE, self.LOCK_STATE, nPlayerId, szPlayerName)
end

function MiningAgency:OnStateConfirm(nPlayerId, szPlayerName)
  if not self.pProcessor then
    return

  self.pProcessor:SetPlayerMiningState(nPlayerId, self.CONFIRM_STATE)
  EventNotify.OnNotify(EventNotify.emNOTIFY_MININGAGENCY_ONCHANGESTATE, self.CONFIRM_STATE, nPlayerId, szPlayerName)

function MiningAgency:OnResetState()
  if not self.pProcessor then
    return
  end
  local nPlayerId1, nPlayerId2 = self.pProcessor:GetPlayerIds()
  self.pProcessor:SetPlayerMiningState(nPlayerId1, self.UNLOCK_STATE)
  self.pProcessor:SetPlayerMiningState(nPlayerId2, self.UNLOCK_STATE)
  EventNotify.OnNotify(EventNotify.emNOTIFY_MININGAGENCY_ONCHANGESTATE, self.UNLOCK_STATE)
end

  return self.pProcessor ~= nil
end

  self.tbRecordAwardList = nil
  self.tbTimeStamp = nil

function MiningAgency:OnMiningSuccess(tbAwardList, nTimeStamp)
  if not self.pProcessor then
    return
  end
  self.tbRecordAwardList = self.tbRecordAwardList or {}
  self.tbTimeStamp = self.tbTimeStamp or {}
  table.insert(self.tbRecordAwardList, Award:MergetSameTypeAward(tbAwardList))
  table.insert(self.tbTimeStamp, nTimeStamp)
  self.pProcessor:Reset()
  EventNotify.OnNotify(EventNotify.emNOTIFY_MININGAGENCY_ONMININGSUCCESS)
end

function MiningAgency:OnServerResult(nErrorCode, tbParam)
  local szMsg = self:GetErrorMsg(nErrorCode, table.unpack(tbParam or {}))
  if szMsg then
    me.CenterMsg(szMsg)
  end
end

function MiningAgency:GetChosenItemList()
  if not self.pProcessor then
    return
  end
  local tbChosenItems = self.pProcessor:GetChosenItems()
  if not tbChosenItems or Lib:CountTB(tbChosenItems) <= 0 then
    return {}
  end
  local tbChosenItemList = {}
  for nTemplateId, nCount in pairs(tbChosenItems) do
    local tbItem = {nTemplateId = nTemplateId, nCount = nCount}
    table.insert(tbChosenItemList, tbItem)
  end
  table.sort(tbChosenItemList, function(tbItemA, tbItemB)
    return tbItemA.nTemplateId > tbItemB.nTemplateId
  end)
  return tbChosenItemList
end

function MiningAgency:GetRecordTextList()
  if not self.tbRecordAwardList then
    return
  end
  local tbRecordText = {}
  for nIdx, tbAwardList in ipairs(self.tbRecordAwardList) do
    local szText = ""
    local szTime = Lib:TimeDesc11(self.tbTimeStamp[nIdx])
    szText = string.format("%d. %s ", nIdx, szTime)
    if Lib:IsEmptyTB(tbAwardList) then
      local szDesc = i18n.EasyGet("\230\151\160")
      szText = string.format("%s %s", szText, szDesc)
    else
      for i, tbAward in ipairs(tbAwardList) do
        local szDesc = Award:GetAwardDescCount(me, tbAward, true, true)
        szDesc = Lib:StrFilterChars(szDesc, {"\195\151"}, "*")
        local nAwardType = Player.AwardType[tbAward[1]]
        if (nAwardType == Player.award_type_item or nAwardType == Player.award_type_item_Unsafe or nAwardType == Player.award_type_itemshow) and not Item:CheckAutoRandItem(tbAward[2]) then
          local nFaction = me.nFaction
          local nFactionSect = me.nFactionSect
          local nSex = Player:GetPlayerSex(me)
          local szItemName, _, _, nQuality = KItem.GetItemShowInfo(tbAward[2], nFaction, nFactionSect, nSex)
          local _, szTxtColor = Item:GetQualityColor(nQuality)
          szDesc = string.format("<color=#%s>%s</color>", szTxtColor, szDesc)
        end
        if i == #tbAwardList then
          szText = string.format("%s %s", szText, szDesc)
        else
          szText = string.format("%s %s, ", szText, szDesc)
        end
      end
    end
    table.insert(tbRecordText, szText)
  end
  return tbRecordText
end

function MiningAgency:OnTimer()
  if not self.pProcessor then
    return true
  end
  local dwOtherId = self.pProcessor:GetOtherPlayerId(me.dwID)
  local pNpc = KNpc.GetByPlayerId(dwOtherId)
  if not pNpc then
    self:ReqCancel()
    return true
  end
  if me.nMapTemplateId ~= pNpc.nMapTemplateId then
    self:ReqCancel()
    return true
  end
  local nDistance = me.GetNpc().GetDistance(pNpc.nId)
  if nDistance < 0 or nDistance >= MiningAgency.nDistance then
    self:ReqCancel()
    return true
  end
  return true
end

function MiningAgency:StartTimer()
  if self.nTimerId then
    Timer:Close(self.nTimerId)
    self.nTimerId = nil
  end
  self.nTimerId = Timer:Register(Env.GAME_FPS, self.OnTimer, self)
end

MiningAgency:StartTimer()
