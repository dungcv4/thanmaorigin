AdaptiveFPS.nInterval = 5

function AdaptiveFPS:Start()
  Client.QualityModule.RestoreTargetFrameRate()
end

function AdaptiveFPS:Stop()
  if self.nGuardTimer then
    Timer:Close(self.nGuardTimer)
    self.nGuardTimer = nil
  end
end

function AdaptiveFPS:OnTime()
  if me then
    local nDoing = me.GetDoing()
    if nDoing == Npc.Doing.run or nDoing == Npc.Doing.jump or Ui:WindowVisible("UIChat") then
      if self.bMoving then
        Client.QualityModule.RestoreTargetFrameRate()
      else
        self.bMoving = true
      end
    elseif not self.bMoving then
      if Client.QualityModule._Setting and Client.QualityModule._Setting.RenderFPS >= 45 then
        Client.QualityModule.SetTargetFrameRate(45)
      else
        Client.QualityModule.SetTargetFrameRate(30)
      end
    else
      self.bMoving = false
    end
  end
  return true
end
