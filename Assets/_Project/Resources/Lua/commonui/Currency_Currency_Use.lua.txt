function Currency:CheckOverMaxValue_JingHuo(pPlayer, nAdd)
  local nValue = self:GetMoney(pPlayer, Currency.TYPE.JingHuo)
  local nMaxValue = self:GetMaxValue_JingHuo(pPlayer)
  if nMaxValue < nValue + nAdd then
    return false
  end
  return true
end

function Currency:GetMaxValue_JingHuo(pPlayer)
  local nLevel = pPlayer.nLevel
  local nMaxValue = 0
  for key, value in pairs(self.tbMaxValue_JingHuo) do
    if key <= nLevel and value > nMaxValue then
      nMaxValue = value
    end
  end
  return nMaxValue
end

function Currency:GetUserMaxValue(pPlayer, szCurrencyType)
  local tbSetting = self.tbSpecialCurrencyType[szCurrencyType]
  if tbSetting then
    if tbSetting.GetMaxValueFunc then
      if tbSetting.GetMaxValueFunc == "default" then
        return Currency:GetCfgMaxValue(szCurrencyType)
      else
        return self[tbSetting.GetMaxValueFunc](self, pPlayer, self.tbSetting[szCurrencyType])
      end
    else
      return 0
    end
  else
    return Currency:GetCfgMaxValue(szCurrencyType)
  end
end

function Currency:GetMoney(pPlayer, szCurrencyType)
  local tbSetting = self.tbSpecialCurrencyType[szCurrencyType]
  if tbSetting then
    if tbSetting.GetFunc then
      if tbSetting.GetFunc == "default" then
        return self:GetUserMoney(pPlayer, szCurrencyType)
      else
        return self[tbSetting.GetFunc](self, pPlayer, self.tbSetting[szCurrencyType])
      end
    else
      return 0
    end
  else
    return self:GetUserMoney(pPlayer, szCurrencyType)
  end
end

function Currency:GetUserMoney(pPlayer, szCurrencyType)
  local tbSetting = self.tbSetting[szCurrencyType]
  if not tbSetting then
    return 0
  end
  return pPlayer.GetUserValue(self.nSaveGroupId, tbSetting.SaveKey)
end

function Currency:GetTongFund(pPlayer)
  if MODULE_GAMESERVER then
    local tbTong = Tong:GetTongById(pPlayer.dwTongId)
    if not tbTong then
      return 0
    end
    return tbTong:GetBuildFund()
  else
    return Tong:GetBuildFund()
  end
end

function Currency:GetItemCurrency(pPlayer, tbSetting)
  if not tbSetting then
    return 0
  end
  local nItemTemplateId = tonumber(tbSetting.Param)
  if not nItemTemplateId then
    return 0
  end
  return pPlayer.GetItemCountInBags(nItemTemplateId)
end

function Currency:GetZhenQi(pPlayer, tbSetting)
  if not tbSetting then
    return 0
  end
  local bRet, _ = Item.ZhenYuan:IsSystemOpen(pPlayer)
  if not bRet then
    return 0
  end
  local nLastCountTime = Item.ZhenYuan:GetLastCountTime(pPlayer)
  local nTimeNow = GetTime()
  if nLastCountTime <= 0 then
    nLastCountTime = nTimeNow
    if MODULE_GAMESERVER then
      Item.ZhenYuan:SetLastCountTime(pPlayer, nLastCountTime)
    end
  end
  local nCalcCnt = 0
  if nTimeNow - nLastCountTime < 3600 then
    return pPlayer.GetUserValue(self.nSaveGroupId, tbSetting.SaveKey)
  end
  nCalcCnt = math.floor((nTimeNow - nLastCountTime) / 3600)
  local nTotal = self:CalcZhenQi(pPlayer, nCalcCnt)
  if nTotal <= 0 then
    return pPlayer.GetUserValue(self.nSaveGroupId, tbSetting.SaveKey)
  end
  if MODULE_GAMESERVER then
    Item.ZhenYuan:SetLastCountTime(pPlayer, nLastCountTime + nCalcCnt * 3600)
    pPlayer.AddMoney(Currency.TYPE.ZhenQi, nTotal, Env.LogWay_ZhenYuanAutoZhenQi)
    return pPlayer.GetUserValue(self.nSaveGroupId, tbSetting.SaveKey)
  elseif MODULE_GAMECLIENT then
    return math.min(pPlayer.GetUserValue(self.nSaveGroupId, tbSetting.SaveKey) + nTotal, Currency:GetUserMaxValue(pPlayer, Currency.TYPE.ZhenQi))
  end
  return 0
end

function Currency:CalcZhenQi(pPlayer, nCalcCnt)
  local tbPosInfo = pPlayer.GetPartnerPosInfo()
  local nTotal = 0
  for _, nPartnerId in pairs(tbPosInfo) do
    if nPartnerId > 0 then
      local pPartner = pPlayer.GetPartnerObj(nPartnerId)
      local nZhenQi = Item.ZhenYuan.tbPartnerZhenQiList[Partner.tbAllPartnerInfo[pPartner.nTemplateId].nQualityLevel] or 0
      nTotal = nTotal + nZhenQi * nCalcCnt
    end
  end
  return nTotal
end
