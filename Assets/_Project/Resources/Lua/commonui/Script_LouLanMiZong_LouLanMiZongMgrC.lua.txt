end

end

function LouLanMiZong:SelEnhanceRsp(tbEnhanceInfo, tbBuffList, nNpcId, bOpenUI)
  self.tbEnhanceInfo = tbEnhanceInfo
  self.tbBuffList = self.tbBuffList or {}
  self.tbBuffList[nNpcId] = tbBuffList
  if bOpenUI then
    Ui:OpenWindow("UILoulanSkillSelect", nNpcId)
  else
    Ui:CloseWindow("UILoulanSkillSelect")
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_LOULANMIZONG_SYNC_SEL_ENHANCE, nNpcId)
  EventNotify.OnNotify(EventNotify.emNOTIFY_HRS_SUBTYPE_MODIFICATION)
end

function LouLanMiZong:SyncInfo(bShowPathBtn, tbPos)
  self:SetBtnPathShow(bShowPathBtn, tbPos)
end

function LouLanMiZong:SetBtnPathShow(bShowPathBtn, tbPos)
  self.bShowPathBtn = bShowPathBtn
  self.tbPos = tbPos
  if Ui:WindowVisible("UIHudRightCenter") then
    Ui:GetWindow("UIHudRightCenter"):SetFubenBtnImageAndText(self:GetFubenBtnAImagePathAndText())
    Ui:GetWindow("UIHudRightCenter"):SetFubenBtnShowStatus(self:CanPath())
  end
end

function LouLanMiZong:CanPath()
  return self.bShowPathBtn or false
end

function LouLanMiZong:AutoPath()
  if not self.tbPos then
    return
  end
  local nTargetX = self.tbPos.nPosX
  local nTargetY = self.tbPos.nPosY
  AFK:Stop()
  local bAutoFight
  
  local function fnCallback()
    local function __timer_Callback()
      AFK:Start()
    end
    
    AFK:Stop()
    Timer:Register(5, __timer_Callback)
  end
  
  AutoPath:GotoAndCallEx(me.nMapId, nTargetX, nTargetY, fnCallback)
end

function LouLanMiZong:IsNeedFubenBtn()
  return self:CanPath(), {
    self.AutoPath,
    self
  }
end

  end
end

function LouLanMiZong:OpenResultWindow(tbDungeonData, tbTeamData, tbAwards, tbScoreInfo)

function LouLanMiZong:OnBossStart(nTime)
  Ui:OpenWindow("UIReadyGo", nTime)
  EventNotify.OnNotify(EventNotify.emNOTIFY_LOULANMIZONG_SYNC_BOSSSTART, true)
  self.nBossStartTimer = Timer:Register(nTime * Env.GAME_FPS, function()
    EventNotify.OnNotify(EventNotify.emNOTIFY_LOULANMIZONG_SYNC_BOSSSTART, false)
    if Ui:WindowVisible("UIProgressLoulan") then
      Ui:CloseWindow("UIProgressLoulan")
    end
    Timer:Close(self.nBossStartTimer)
    self.nBossStartTimer = nil
  end)
end

end

function LouLanMiZong:OpenRankBoard()
  local tbData = {
    szTitle = i18n.EasyGet("\230\165\188\229\133\176\232\191\183\232\184\170"),
    tbRankLayerIdList = {14}
  }
  local tbPermanantRankData = {
    szRankType = "LouLanMiZong",
    szScoreTitle = i18n.EasyGet("\233\128\154\229\133\179\230\151\182\233\151\180:"),
    tbRankingAward = self.tbRankingAward,
    nAwardId = 1231,
    fnProcessHighValue = function(nHighValue)
      nHighValue = LouLanMiZong.TIMESETTING.nTotalTime - nHighValue
      local nMin = math.floor(nHighValue / 60)
      local nSec = math.floor(nHighValue % 60)
      local szHighValue = string.format("%02d:%02d", nMin, nSec)
      return szHighValue
    end
  }
  Ui:OpenWindow("UITemporaryActivityRank", tbData, 1, tbPermanantRankData)
end
