Require("CommonScript/GlobalOtherDiffDefine.lua")
local tbWnd = Ui:CreateClass("UICheapMonthGift", "UIPanel")
local tbControls = {
  TemplateTeammate = "pnlMask/viewport/pnlContant/TemplateTeammate"
}
tbWnd.tbControls = tbControls
tbWnd.tbTempCfg = {
  [1] = {
    Title = function()
      return i18n.EasyGet("\229\176\138\228\186\171\232\177\170\229\141\142\230\156\136\229\141\161\229\164\167\231\164\188\229\140\133")
    end,
    Desc = function()
      return i18n.EasyGet("\229\140\133\229\144\171\230\156\136\229\141\161\227\128\1294\229\145\168\229\145\168\229\141\161\227\128\12928\230\151\165\230\175\143\230\151\165\231\164\188\229\140\133")
    end,
    nDiscount = Vip.VipDefine.nCheapMonthGiftDiscount,
    fnGetBuyState = function()
      return tbWnd:CheapMonthGiftBuyState()
    end,
    fnOnClickHelp = function()
      Ui:OnHelpClicked("CheapMonthGiftHelp")
    end,
    fnOnClickBuy = function()
      CheapMonthGift:OnChargeReq()
    end
  }
}
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.pScorllView = Ui:NewScrollView(self, self.TemplateTeammate)
end

function tbWnd:OnOpen()
  CheapMonthGift.bHaveClick = true
  CheapMonthGift:UpdateReddotState()
  self:UpdateView()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateView()
  self.pScorllView:Clear()
  for i, tbConfig in ipairs(self.tbTempCfg) do
    self.pScorllView:SetElement(i, function(tbElement)
      self:SetElementInfo(tbElement, tbConfig)
    end)
  end
end

function tbWnd:SetElementInfo(tbElement, tbConfig)
  tbElement.pPanel:Label_SetText("Full/txttitle", tbConfig.Title())
  tbElement.pPanel:Label_SetText("Full/imgTitle/txtTitle", tbConfig.Desc())
  if tbConfig.nDiscount then
    tbElement.pPanel:SetActive("Full/imgZK", true)
    tbElement.pPanel:Label_SetText("Full/imgZK/txtZK", i18n.GetDiscountText(tbConfig.nDiscount))
  else
    tbElement.pPanel:SetActive("Full/imgZK", false)
  end
  Ui:Button_BindEvent(tbElement.pPanel, "Full/BtnHelp", function()
    tbConfig.fnOnClickHelp()
  end)
  local bCanClick, szButton = tbConfig.fnGetBuyState()
  tbElement.pPanel:Button_SetEnable("Full/BtnBuy", bCanClick)
  tbElement.pPanel:Button_SetText("Full/BtnBuy", szButton, "Text")
  Ui:Button_BindEvent(tbElement.pPanel, "Full/BtnBuy", function()
    tbConfig.fnOnClickBuy()
  end)
end

function tbWnd:CheapMonthGiftBuyState()
  local nLeftDay = CheapMonthGift:GetBuyCDEndLeftDay(me)
  local szButton = ""
  if nLeftDay > 0 then
    szButton = i18n.EasyGet("\229\134\183\229\141\180{0}\230\151\165", nLeftDay)
  else
    local tbProductSetting = Recharge:GetCheapMonthGiftSetting()
    local szShowPrice = Recharge:GetShowPrice(tbProductSetting)
    szButton = szShowPrice
  end
  return nLeftDay <= 0, szButton
end
