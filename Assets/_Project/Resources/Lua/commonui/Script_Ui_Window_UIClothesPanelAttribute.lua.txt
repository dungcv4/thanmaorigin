local tbWnd = Ui:CreateClass("UIClothesPanelAttribute", "UIPanel")
local tbPartID2Name = {
  [Npc.NpcResPartsDef.npc_part_body] = function()
    return i18n.EasyGet("\229\164\150\232\161\163\239\188\154")
  end,
  [Npc.NpcResPartsDef.npc_part_weapon] = function()
    return i18n.EasyGet("\230\173\166\229\153\168\239\188\154")
  end,
  [Npc.NpcResPartsDef.npc_part_horse] = function()
    return i18n.EasyGet("\229\157\144\233\170\145\239\188\154")
  end,
  [Npc.NpcResPartsDef.npc_part_head] = function()
    return i18n.EasyGet("\229\164\150\229\184\189\239\188\154")
  end
}
local tbControls = {
  Template1 = "imgBg/Scroll View/Viewport/Content/Template1",
  txtTitle = "imgBg/txtTitle",
  empty = "imgBg/empty"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.tbAttribScrollView = Ui:NewScrollView(self, self.Template1, nil, false, true)
end

function tbWnd:UpdateView()
  self.tbAttribScrollView:Clear()
  local tbAttributes = {}
  local tbHaveFashion = Fashion:GetHaveFashions()
  for nFashionId, _ in pairs(tbHaveFashion) do
    local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
    if bCanUse and tbSetting and Fashion.AttrCfg[nFashionId] then
      local szFinalName, szFinalValue = FightSkill:GetMagicDescSplit(Fashion.AttrCfg[nFashionId][1], {
        table.unpack(Fashion.AttrCfg[nFashionId], 2)
      })
      if not tbAttributes[szFinalName] then
        tbAttributes[szFinalName] = {
          nIndex = nIndex,
          szFinalValue = szFinalValue,
          tbSetting = tbSetting
        }
      elseif FightSkill:StringValueSubNumberResult(tbAttributes[szFinalName].szFinalValue, szFinalValue) < 0 then
        tbAttributes[szFinalName].szFinalValue = szFinalValue
      end
    end
  end
  local nIndex = 0
  for szFinalName, tbAttribute in pairs(tbAttributes) do
    nIndex = nIndex + 1
    local szDesc = self:GetPrettyAttribute(szFinalName, tbAttribute.szFinalValue)
    if szDesc then
      self.tbAttribScrollView:SetElement(nIndex, function(tbElement)
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("txtEquip", tbPartID2Name[tbAttribute.tbSetting.nPart]())
        pPanel:Label_SetText("txtAttribute", szDesc)
      end)
    end
  end
  self.pPanel:SetActive(self.empty, nIndex > 0)
  local nPrettyValue = Fashion:GetPrettyValue(me)
  self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\230\128\187\233\173\133\229\138\155\229\128\188\239\188\154{0}", tostring(nPrettyValue)))
end

function tbWnd:GetPrettyAttribute(szFinalName, szFinalValue)
  if nil == szFinalName then
    return nil
  end
  if nil == szFinalValue then
    return szFinalName
  end
  return string.format("%s%s", szFinalName, szFinalValue)
end
