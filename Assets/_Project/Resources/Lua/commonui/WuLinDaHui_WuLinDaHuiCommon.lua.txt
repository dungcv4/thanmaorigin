function WuLinDaHui:IsSysOpen()
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local nNowTime = GetTime()
  if nNowTime < Lib:GetConfigDate2Time(self.Cfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.Cfg.END_DATE) then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  return true
end

function WuLinDaHui:IsStageOpen(nStage)
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local nNowTime = GetTime()
  if nStage == self.STAGE.YUGAO and (nNowTime < Lib:GetConfigDate2Time(self.Cfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.SignCfg.START_DATE)) then
    return false, i18n.EasyGet("\228\184\141\229\156\168\233\162\132\229\145\138\230\156\159\229\134\133")
  elseif nStage == self.STAGE.SIGN and (nNowTime < Lib:GetConfigDate2Time(self.SignCfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.SignCfg.END_DATE)) then
    return false, i18n.EasyGet("\228\184\141\229\156\168\230\138\165\229\144\141\230\156\159\229\134\133")
  elseif nStage == self.STAGE.PRE and (nNowTime < Lib:GetConfigDate2Time(self.PreCfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.PreCfg.END_DATE)) then
    return false, i18n.EasyGet("\228\184\141\229\156\168\233\162\132\233\128\137\232\181\155\230\151\182\233\151\180")
  elseif nStage == self.STAGE.MID and (nNowTime < Lib:GetConfigDate2Time(self.MidCfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.MidCfg.END_DATE)) then
    return false, i18n.EasyGet("\228\184\141\229\156\168\230\183\152\230\177\176\232\181\155\230\151\182\233\151\180")
  elseif nStage == self.STAGE.FINAL and (nNowTime < Lib:GetConfigDate2Time(self.FinalCfg.START_DATE) or nNowTime > Lib:GetConfigDate2Time(self.FinalCfg.END_DATE)) then
    return false, i18n.EasyGet("\228\184\141\229\156\168\229\134\179\232\181\155\230\151\182\233\151\180")
  else
    return true
  end
end

function WuLinDaHui:GetStageShow()
  if not self.IS_SYS_OPEN then
    return i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
  local nDate = tonumber(os.date("%Y%m%d", GetTime()))
  local nNowTime = GetTime()
  if nNowTime < Lib:GetConfigDate2Time(self.Cfg.START_DATE) then
    return i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  elseif nNowTime < Lib:GetConfigDate2Time(self.SignCfg.START_DATE) then
    return i18n.EasyGet("\233\162\132\229\145\138\230\156\159")
  elseif nNowTime >= Lib:GetConfigDate2Time(self.PreCfg.START_DATE) and nNowTime <= Lib:GetConfigDate2Time(self.PreCfg.END_DATE) then
    return i18n.EasyGet("\233\162\132\233\128\137\232\181\155")
  elseif nNowTime <= Lib:GetConfigDate2Time(self.SignCfg.END_DATE) then
    return i18n.EasyGet("\230\138\165\229\144\141\230\156\159")
  elseif nNowTime <= Lib:GetConfigDate2Time(self.MidCfg.END_DATE) then
    return i18n.EasyGet("\230\183\152\230\177\176\232\181\155")
  else
    for nIndex, tbDateInfo in ipairs(WuLinDaHui.FinalMatchDateCfg) do
      if nDate <= tbDateInfo.DATE then
        return i18n.Get(tbDateInfo.DESC)
      end
    end
    return i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175")
  end
end

function WuLinDaHui:GetNowGuessCfg(tbFinalMatchList)
  local tbReturnCfg
  for _, tbGuessCfg in ipairs(self.GuessSettingCfg) do
    if tbFinalMatchList[tbGuessCfg.PHASE] then
      tbReturnCfg = tbGuessCfg
    end
  end
  return tbReturnCfg
end

function WuLinDaHui:ChkCondition(pPlayer, tbCondition)
  if tbCondition[WuLinDaHui.emMsgNotBegin] and not WuLinDaHui:IsSysOpen() then
    return false, WuLinDaHui.emMsgNotBegin
  end
  if tbCondition[WuLinDaHui.emMsgSignTime] and not WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.SIGN) and not WuLinDaHui:IsStageOpen(WuLinDaHui.STAGE.PRE) then
    return false, WuLinDaHui.emMsgSignTime
  end
  if tbCondition[WuLinDaHui.emMsgMinLevel] and pPlayer.nLevel < WuLinDaHui.Cfg.MIN_LEVEL then
    return false, WuLinDaHui.emMsgMinLevel, {
      WuLinDaHui.Cfg.MIN_LEVEL
    }
  end
  if tbCondition[WuLinDaHui.emMsgMinLevel] and pPlayer.nLevel < WuLinDaHui.Cfg.MIN_LEVEL then
    return false, WuLinDaHui.emMsgMinLevel, {
      WuLinDaHui.Cfg.MIN_LEVEL
    }
  end
  if tbCondition[WuLinDaHui.emMsgNotTeamCaptain] and pPlayer.dwTeamID > 0 and not TeamMgr:IsCaptain(pPlayer) then
    return false, WuLinDaHui.emMsgNotTeamCaptain
  end
  if tbCondition[WuLinDaHui.emMsgTeamMemeberErr] and pPlayer.dwTeamID > 0 then
    local tbMember = TeamMgr:GetMembers(pPlayer.dwTeamID)
    if tbCondition[WuLinDaHui.emMsgTeamMemeberErr] and #tbMember > WuLinDaHui.TeamCfg.MAX_LEAGUE_MEMBER then
      return false, WuLinDaHui.emMsgTeamMemeberErr, {
        WuLinDaHui.TeamCfg.MAX_LEAGUE_MEMBER
      }
    end
  end
  if tbCondition[WuLinDaHui.emMsgNeedBattleTeam] and self:GetBattleTeamId(pPlayer.dwID) <= 0 then
    return false, WuLinDaHui.emMsgNeedBattleTeam, {
      pPlayer.szName
    }
  end
  if tbCondition[WuLinDaHui.emMsgBattleTeamCaptain] and not WldhTeamMgr:OpearateTeamByPlayer(pPlayer.dwID, WldhTeam.IsCaptain) then
    return false, WuLinDaHui.emMsgBattleTeamCaptain, {
      pPlayer.szName
    }
  end
  if MODULE_GAMESERVER and pPlayer.dwTeamID > 0 then
    local tbMember = TeamMgr:GetMembers(pPlayer.dwTeamID)
    if tbMember then
      for _, nPlayerId in pairs(tbMember) do
        if nPlayerId ~= pPlayer.dwID then
          local pMember = KPlayer.GetPlayerObjById(nPlayerId)
          if not pMember then
            if tbCondition[WuLinDaHui.emMsgMemberOffline] then
              return false, WuLinDaHui.emMsgMemberOffline
            else
              break
            end
          end
          if tbCondition[WuLinDaHui.emMsgMemberMinLevel] and pMember.nLevel < WuLinDaHui.Cfg.MIN_LEVEL then
            return false, WuLinDaHui.emMsgMemberMinLevel, {
              pMember.szName,
              WuLinDaHui.Cfg.MIN_LEVEL
            }
          end
          if tbCondition[WuLinDaHui.emMsgMemberSafeMap] and not Map:IsCityMap(pMember.nMapTemplateId) and not Map:IsFieldFightMap(pMember.nMapTemplateId) then
            return false, WuLinDaHui.emMsgMemberSafeMap, {
              pMember.szName
            }
          end
          if tbCondition[WuLinDaHui.emMsgMemberNotSafePoint] and Map:IsFieldFightMap(pMember.nMapTemplateId) and pMember.nFightMode ~= Npc.FIGHT_MODE.emFightMode_None then
            return false, WuLinDaHui.emMsgMemberNotSafePoint, {
              pMember.szName
            }
          end
          if tbCondition[WuLinDaHui.emMsgMemberAloneState] and not pMember.nState == Player.emPLAYER_STATE_ALONE then
            return false, WuLinDaHui.emMsgMemberAloneState, {
              pMember.szName
            }
          end
          if tbCondition[WuLinDaHui.emMsgMemberHaveTeam] and self:ResetTeamGroup(pMember.dwID) and self:GetBattleTeamId(pMember.dwID) > 0 then
            return false, WuLinDaHui.emMsgMemberHaveTeam, {
              pMember.szName
            }
          end
        end
      end
    end
  end
  if tbCondition[WuLinDaHui.emMsgSafeMap] and not Map:IsCityMap(pPlayer.nMapTemplateId) and not Map:IsFieldFightMap(pPlayer.nMapTemplateId) then
    return false, WuLinDaHui.emMsgSafeMap
  end
  if tbCondition[WuLinDaHui.emMsgNotSafePoint] and Map:IsFieldFightMap(pPlayer.nMapTemplateId) and pPlayer.nFightMode ~= Npc.FIGHT_MODE.emFightMode_None then
    return false, WuLinDaHui.emMsgNotSafePoint
  end
  if tbCondition[WuLinDaHui.emMsgAloneState] and not pPlayer.nState == Player.emPLAYER_STATE_ALONE then
    return false, WuLinDaHui.emMsgAloneState
  end
  if tbCondition[WuLinDaHui.emMsgHaveTeam] and self:ResetTeamGroup(pPlayer.dwID) and self:GetBattleTeamId(pPlayer.dwID) > 0 then
    return false, WuLinDaHui.emMsgHaveTeam
  end
  if tbCondition[WuLinDaHui.emMsgWeekOverCount] and self:ChkWeekJoinEnough(pPlayer.dwID) then
    return false, WuLinDaHui.emMsgWeekOverCount
  end
  if tbCondition[WuLinDaHui.emMsgTotalOverCount] and self:ChkTotalJoinEnough(pPlayer.dwID) then
    return false, WuLinDaHui.emMsgTotalOverCount
  end
  if tbCondition[WuLinDaHui.emMsgJoinOverCount] and self:ChkDailyJoinEnough(pPlayer.dwID) then
    return false, WuLinDaHui.emMsgJoinOverCount
  end
  return true
end

function WuLinDaHui:ChkConditionAndSendMsg(pPlayer, tbCondition)
  if not pPlayer then
    LogErr("WuLinDaHui:ChkConditionAndSendMsg pPlayer is nil ")
    return false
  end
  local bRet = true
  local nCode, tbParam
  bRet, nCode, tbParam = self:ChkCondition(pPlayer, tbCondition)
  if not bRet then
    if MODULE_GAMECLIENT then
      self:MsgCode(pPlayer.dwID, nCode, table.unpack(tbParam or {}))
    end
    if MODULE_GAMESERVER then
      if not pPlayer.dwTeamID or pPlayer.dwTeamID <= 0 then
        self:MsgCode(pPlayer.dwID, nCode, table.unpack(tbParam or {}))
      else
        local tbMember = TeamMgr:GetMembers(pPlayer.dwTeamID) or {}
        if #tbMember <= 0 then
          self:MsgCode(pPlayer.dwID, nCode, table.unpack(tbParam or {}))
        end
        for _, nMemberId in pairs(tbMember) do
          self:MsgCode(nMemberId, nCode, table.unpack(tbParam or {}))
        end
      end
    end
  end
  return bRet
end

function WuLinDaHui:MsgCode(nPlayerId, nCode, ...)
  if MODULE_GAMESERVER then
    local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
    if not pPlayer then
      return
    end
    pPlayer.CallClientScript("WuLinDaHui:MsgCode", nPlayerId, nCode, ...)
  elseif MODULE_GAMECLIENT then
    local szMsg = ""
    if (...) then
      szMsg = i18n.Get(WuLinDaHui.tbMsg[nCode](), ...)
    else
      szMsg = WuLinDaHui.tbMsg[nCode]()
    end
    Ui:OpenWindow("UIMessageBox", szMsg)
  end
end

function WuLinDaHui:GetBattleTeamId(nPlayerId)
  local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
  if not pPlayer then
    return 0
  end
  if MODULE_GAMESERVER then
    WuLinDaHui:ResetTeamGroup(nPlayerId)
  end
  return pPlayer.GetUserValue(WuLinDaHui.SAVE_GROUP, WuLinDaHui.SAVE_TEAMID)
end

function WuLinDaHui:GetPreRankAward(nPlayerRank)
  local tbAward
  local tbAllAward = self:GetAwardSetting(self.TYPE_AWARD_PRERANK)
  if not tbAllAward then
    return
  end
  for _, tbData in ipairs(tbAllAward) do
    if nPlayerRank <= tbData.nRank then
      tbAward = {
        tbData.nAwardId,
        tbData.nAwardKey
      }
      break
    end
  end
  return tbAward
end

function WuLinDaHui:CheckContestant(nPlayerId, nPhase)
  local nTeamId = self:GetBattleTeamId(nPlayerId)
  if not nTeamId or nTeamId <= 0 then
    return false
  end
  if not self.tbFinalMatchList then
    return false
  end
  local tbMatchList = self.tbFinalMatchList[nPhase]
  if not tbMatchList then
    return false
  end
  for nIndex, tbData in pairs(tbMatchList) do
    if tbData.nTeamId == nTeamId then
      return true
    end
  end
  return false
end

if MODULE_GAMESERVER then
  function WuLinDaHui:SyncPlayerTeamInfoToPlayer(pPlayer, nTeamId)
    if not pPlayer then
      return
    end
    nTeamId = nTeamId or WuLinDaHui:GetBattleTeamId(pPlayer.dwID)
    local tbTeamInfo = WldhTeamMgr:GetTeamData(nTeamId)
    pPlayer.CallClientScript("WuLinDaHui:SyncTeamInfoC", tbTeamInfo, self.nGameState)
  end
  
  function WuLinDaHui:ChkWeekJoinEnough(nPlayerId)
    local tbTeamData = WldhTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    if not tbTeamData then
      return
    end
    return WldhTeam:IsWeekOverCount(tbTeamData)
  end
  
  function WuLinDaHui:ChkTotalJoinEnough(nPlayerId)
    local tbTeamData = WldhTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    if not tbTeamData then
      return true
    end
    return WldhTeam:IsTotalJoinEnough(tbTeamData)
  end
  
  function WuLinDaHui:ChkDailyJoinEnough(nPlayerId)
    local tbTeamData = WldhTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    if not tbTeamData then
      return false
    end
    return WldhTeam:IsDailyJoinEnough(tbTeamData)
  end
end

function WuLinDaHui:CheckGmWarch(pPlayer)
  local nGmCard = pPlayer.GetItemCountByTypeInBag(self.Cfg.GM_CARD_ID)
  return nGmCard > 0
end
