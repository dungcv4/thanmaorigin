local tbWnd = Ui:CreateWindow("UIKinFlagDivination")
tbWnd.tbControls = {
  BtnClose = "imgBG/BtnClose",
  btnQian = "imgBG/btnList/btnQian",
  btnKan = "imgBG/btnList/btnKan",
  btnGen = "imgBG/btnList/btnGen",
  btnZhen = "imgBG/btnList/btnZhen",
  btnXun = "imgBG/btnList/btnXun",
  btnLi = "imgBG/btnList/btnLi",
  btnKun = "imgBG/btnList/btnKun",
  btnDui = "imgBG/btnList/btnDui",
  txtSelectValue = "imgBG/txtSelect/txtSelectValue",
  txtCountdownValue = "imgBG/txtCountdown/txtCountdownValue",
  btnOK = "imgBG/btnOK"
}
tbWnd.tbOnClick = {
  BtnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    if Lib:CountTB(self.tbSelect) < self.nCount then
      me.CenterMsg(i18n.EasyGet("\230\156\170\233\128\137\230\139\169\232\182\179\229\164\159\231\154\132\229\141\166\232\177\161"))
      return
    end
    local tbData = {}
    for k, v in pairs(self.tbSelect) do
      table.insert(tbData, k)
    end
    TongParty:CSDivinationReq(tbData)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbTongPosBtnGroup = Ui:NewMultipleButtonGroup(self, {
    self.btnQian,
    self.btnXun,
    self.btnKan,
    self.btnGen,
    self.btnKun,
    self.btnZhen,
    self.btnLi,
    self.btnDui
  }, function(self, nIndex)
    self:OnDivinationClick(nIndex)
  end, function(self, nIndex)
    if self.tbSelect[nIndex] then
      return true
    end
    if Lib:CountTB(self.tbSelect) >= self.nCount then
      return false
    else
      return true
    end
  end)
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(self.BtnClose, false)
  self.tbSelect = {}
  self.nCount = TongParty:GetAnswerRightNum()
  self.pPanel:Label_SetText(self.txtSelectValue, "0/" .. self.nCount)
  self:SetRemainTime()
end

function tbWnd:OnClose()
  if self.nRemainTimer then
    Timer:Close(self.nRemainTimer)
    self.nRemainTimer = nil
  end
end

function tbWnd:OnDivinationClick(nIndex)
  if self.tbSelect[nIndex] then
    self.tbSelect[nIndex] = nil
  elseif Lib:CountTB(self.tbSelect) >= self.nCount then
    return
  else
    self.tbSelect[nIndex] = true
  end
  local nSelect = Lib:CountTB(self.tbSelect)
  self.pPanel:Label_SetText(self.txtSelectValue, nSelect .. "/" .. self.nCount)
end

function tbWnd:SetRemainTime()
  if self.nRemainTimer then
    Timer:Close(self.nRemainTimer)
    self.nRemainTimer = nil
  end
  self.nRemainTimer = Timer:Register(Env.GAME_FPS, function(self)
    local nLeftTime = TongParty.nDivinationTime - GetTime()
    if nLeftTime >= 0 then
      self.pPanel:Label_SetText(self.txtCountdownValue, nLeftTime)
      return true
    else
      self:Close()
      return false
    end
  end, self)
end
