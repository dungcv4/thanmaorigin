local tbWnd = Ui:CreateWindow("UIUpgradeEffect")
local tbControls = {
  txtSuccess = "imgBG/Success_Effect/txtSuccess",
  imgBG = "imgBG"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(szText, tbPos)
  if tbPos then
    self.pPanel:Object_SetLocalPosition(tbControls.imgBG, Ui.Vector3(tbPos.x or 0, tbPos.y or 220, tbPos.z or 0))
  end
  self.pPanel:Label_SetText(tbControls.txtSuccess, szText)
  self.nTimer = Timer:Register(Env.GAME_FPS, function(self)
    self:Close()
  end, self)
end

function tbWnd:OnClose()
  self.pPanel:Object_SetLocalPosition(tbControls.imgBG, Ui.Vector3(0, 220, 0))
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
  self.nTimer = nil
end
