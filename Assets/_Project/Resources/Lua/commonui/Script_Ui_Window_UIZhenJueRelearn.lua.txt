function tbWnd:OnCreate()
  self:SetClickEmptyToClose(tbControls.wndZhenJueGallery)
  self:SetClickEmptyToClose(tbControls.wndJumpTo)
  self.tbAttrIdSVNow = Ui:NewScrollView(self, tbControls.AttrTempNow)
  self.tbAttrIdSVNext = Ui:NewScrollView(self, tbControls.AttrTempNext)
  self.tbAttrIdSVLib = Ui:NewScrollView(self, tbControls.AttrTempLib)
  self.tbCostItemGrid = Ui:NewExistItemGrid(self, tbControls.CostItemGrid)
  self.tbCostItemGrid:SetClickEnable(true)
  self.tbWayScrollView = Ui:NewScrollView(self, tbControls.WayTemplate)
end

function tbWnd:OnOpen()
  self.bHadReRandom = false
  self.pPanel:SetActive(tbControls.Empty, true)
  self.pPanel:SetActive(tbControls.btnRelearn, true)
  self.pPanel:SetActive(tbControls.btnReplace, false)
  self.pPanel:SetActive(tbControls.btnRelearnContinue, false)
  self:UpAttrIdNowSV()
  self.tbAttrIdSVNext:Reset()
  self:UpAttrIdLibSV()
  self:UpdateCost()
  self:UpdateGettingWay()
end

function tbWnd:UpdateCost()
  self.tbCostItemGrid:SetItemByTemplate(ZhenFa.nHuiXinDanId)
  local nCurNum = me.GetItemCountByTypeInBag(ZhenFa.nHuiXinDanId)
  self.pPanel:Label_SetText(tbControls.txtCost, string.format("%d/%d", nCurNum, ZhenFa.nHuiXinDanCostNum))
  if nCurNum >= ZhenFa.nHuiXinDanCostNum then
    self.pPanel:Label_SetColorByName(tbControls.txtCost, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtCost, "NotEnough")
  end
end

function tbWnd:OnClose()
end

function tbWnd:UpAttrIdNowSV()
  local nCurPutOnPos, _ = ZhenFa:GetZhenFaCurPutOnPos(me)
  local tbZJAttrId = ZhenFa:GetZhenJueAttrIdList(me, nCurPutOnPos)
  local tbShowZJId = ZhenFa:GetZhenJueAttrShowItemIdList(me, nCurPutOnPos)
  self.tbAttrIdSVNow:Reset()
  for i = 1, #tbZJAttrId do
    if tbShowZJId[i] then
      self.tbAttrIdSVNow:SetElement(i, function(tbElement, nIndex)
        local pPanel = tbElement.pPanel
        local szName = Item:GetName(tbShowZJId[nIndex], me)
        pPanel:Label_SetText("txtName", szName)
        local nLevel = ZhenFa:GetZhenJueHoleLevel(me, nIndex)
        local tbAttrCur = ZhenFa.ZhenJueAttrInfo[tbZJAttrId[nIndex]][nLevel][1]
        local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCur[1], {
          table.unpack(tbAttrCur, 2)
        })
        pPanel:Label_SetText("txtAttr", szAttribMsg)
        pPanel:Label_SetText("txtLV", nLevel)
        if ZhenFa.tbAttactZJCfg[tbZJAttrId[nIndex]] then
          pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeAttact)
        else
          pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeDefance)
        end
        local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        itemGrid:SetItemByTemplate(tbShowZJId[nIndex])
        itemGrid:SetLabelSuffixText("+" .. tostring(nLevel))
      end)
    end
  end
end

function tbWnd:UpAttrIdNextSV(tbZJAttrId)

function tbWnd:UpAttrIdLibSV()
  local nCurPutOnPos, nCurPutOnSubPos = ZhenFa:GetZhenFaCurPutOnPos(me)
  local tbCfg = ZhenFa.ZhenFaInfo[nCurPutOnPos][nCurPutOnSubPos]
  local tbAttackZJLib = ZhenFa.ZhenJueLibrary[tbCfg[6]]
  local tbDefenseZJLib = ZhenFa.ZhenJueLibrary[tbCfg[8]]
  local tbZJAttrId = {}
  Lib:MergeTable(tbZJAttrId, tbAttackZJLib)
  Lib:MergeTable(tbZJAttrId, tbDefenseZJLib)
  local tbShowZJId = {}
  for nHole, nAttrId in ipairs(tbZJAttrId) do
    local nItemId = ZhenFa.ZhenJueAttrInfo[nAttrId][1][2]
    tbShowZJId[nHole] = nItemId
  end
  self.tbAttrIdSVLib:Reset()
  for i = 1, #tbZJAttrId do
    if tbShowZJId[i] then
      self.tbAttrIdSVLib:SetElement(i, function(tbElement, nIndex)
        local pPanel = tbElement.pPanel
        local szName = Item:GetName(tbShowZJId[nIndex], me)
        pPanel:Label_SetText("txtName", szName)
        local nLevel = ZhenFa:GetZhenJueHoleLevel(me, nIndex)
        local tbAttrCur = ZhenFa.ZhenJueAttrInfo[tbZJAttrId[nIndex]][nLevel][1]
        local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCur[1], {
          table.unpack(tbAttrCur, 2)
        })
        pPanel:Label_SetText("txtAttr", szAttribMsg)
        if ZhenFa.tbAttactZJCfg[tbZJAttrId[i]] then
          pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeAttact)
        else
          pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeDefance)

        local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        itemGrid:SetItemByTemplate(tbShowZJId[nIndex])
      end)
    end
  end
end

function tbWnd:ReplaceRsp()
  self:OnOpen()
end

function tbWnd:UpdateGettingWay()
  self.tbWayScrollView:Clear()
  for i, tbWayInfo in ipairs(self.GETTING_WAY) do
    self.tbWayScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtName", tbWayInfo.szTitle())
      tbElement.pPanel:Label_SetText("txtDesc", tbWayInfo.szDesc())
      if tbWayInfo.func then
        tbElement.pPanel:SetActive("imgArrow", true)
        Ui:Button_BindEvent(tbElement.pPanel, "", function()
          tbWayInfo.func()
          self.pPanel:SetActive(tbControls.wndJumpTo, false)
        end)
      else
        tbElement.pPanel:SetActive("imgArrow", false)
      end
    end)
  end
end
