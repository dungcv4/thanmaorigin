local tbWnd = Ui:CreateWindow("UIYueNanWomenDayReward")
local tbControls = {
  Template = "node/imgBG/Content/Scroll View/Viewport/Content/Template",
  txtTitleNumber = "node/imgBG/Content/imgTitleBg/txtTitleNumber",
  txtTitleReward = "node/imgBG/Content/imgTitleBg/txtTitleReward",
  txtTitlePrg = "node/imgBG/Content/imgTitleBg/txtTitlePrg",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  btnClose = "node/imgBG/btnClose"
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UPDATE_EXTRA_AWARD,
      self.UpdateListView,
      self
    },
    {
      EventNotify.emNOTIFY_MAP_ENTER,
      self.OnEnterMap,
      self
    }
  }
end

tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnEnterMap()
  self:Close()
end

function tbWnd:OnCreate()
  self.tbAchieveListView = Ui:NewScrollView(self, tbControls.Template, nil)
end

function tbWnd:OnOpen(nAchieveActId)
  if not Lib:IsInteger(nAchieveActId) then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return 0
  end
  self.pAchieveActInstance = ActivityMgr:GetInstanceById(nAchieveActId)
  if not self.pAchieveActInstance or not self.pAchieveActInstance:IsOpen() then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return 0
  end
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self:UpdateTxtInfo()
  self:UpdateListView()
end

function tbWnd:UpdateTxtInfo()
  self.pPanel:Label_SetText(tbControls.txtTitleNumber, i18n.EasyGet("\229\175\187\231\188\152\230\172\161\230\149\176"))
  self.pPanel:Label_SetText(tbControls.txtTitleReward, i18n.EasyGet("\229\165\150\229\138\177"))
  self.pPanel:Label_SetText(tbControls.txtTitlePrg, i18n.EasyGet("\232\191\155\229\186\166"))
end

function tbWnd:UpdateListView()
  local tbAchieveDataList = self.pAchieveActInstance:GetExtraAwardList()
  for nIndex, tbAchieveData in ipairs(tbAchieveDataList) do
    self.tbAchieveListView:SetElement(nIndex, function(tbElement)
      self:__SetViewElement(tbElement, tbAchieveData)
    end)
  end
end

function tbWnd:__SetViewElement(tbElement, tbAchieveData)
  local pPanel = tbElement.pPanel
  pPanel:Label_SetText("txtNumber", i18n.Parse(tbAchieveData.szInfo))
  if not tbElement.tbItemGridView then
    tbElement.tbItemGridView = Ui:NewItemGridView(tbElement, "rewardList/reward")
  else
    tbElement.tbItemGridView:Clear()
  end
  for nIndex, tbAward in ipairs(tbAchieveData.tbAward) do
    tbElement.tbItemGridView:SetItem(nIndex, tbAward, true)
  end
  if tbAchieveData.bSuccess and not tbAchieveData.bHasAward then
    pPanel:SetActive("btnGet", true)
    pPanel:SetActive("imgComplete", false)
    pPanel:SetActive("prg", false)
    pPanel:Button_BindEvent("btnGet", function(nAchieveId)
      self.pAchieveActInstance:GetExtraAward(nAchieveId)
    end, tbAchieveData.nAchieveId)
  elseif tbAchieveData.bSuccess and tbAchieveData.bHasAward then
    pPanel:SetActive("imgComplete", true)
    pPanel:SetActive("prg", false)
    pPanel:SetActive("btnGet", false)
  else
    pPanel:Label_SetText("prg/txtPrg", string.format("%s/%s", tbAchieveData.nFinishCount, tbAchieveData.nNeedCount))
    pPanel:Sprite_SetFillAmount("prg/imgBar", 0, tbAchieveData.nFinishCount / tbAchieveData.nNeedCount, 1, true)
    pPanel:SetActive("prg", true)
    pPanel:SetActive("imgComplete", false)
    pPanel:SetActive("btnGet", false)
  end
end
