  UIEquipPanel = "Equip/UIEquipPanel",
  NpcModelPath = "Equip/NpcModel",
  HitZone = "Equip/NpcModel/HitZone",
  imgRolePortrait = "Attrib/Info/PortraitObj/Portrait/Image",

  txtNoteName = "Attrib/Info/nameCell/txtNoteName",
  txtHonorTip = "Attrib/Info/txtHonorTip",
  txtHonor = "Attrib/Info/txtHonorTip/HonorTitle",
  txtFightPowerTip = "Equip/UIEquipPanel/EquipPanel/txtFightPowerTip",
  txtFightPower = "Equip/UIEquipPanel/EquipPanel/txtFightPowerTip/txtFightPower",
  btnSeriesText = "Equip/UIEquipPanel/btnSeriesText",
  QiHunValue = "Equip/UIEquipPanel/grid/btnQiHunInfo/TextGroup/Text",
  txtRoleIDTip = "Attrib/Info/txtRoleIDTip",
  txtRoleID = "Attrib/Info/txtRoleIDTip/txtRoleID",

  txtEnergy = "Attrib/Info/txtEnergyTip/txtEnergy",

tbWnd.tbControls = tbControls
local tbSeriesNumberToTextName = {
  [1] = "txtMetal",
  [2] = "txtWood",
  [3] = "txtWater",
  [4] = "txtFire",
  [5] = "txtEarth"
}
tbWnd.tbOnClick = {
  btnOtherAttrib = function(self)
    Ui:OpenWindow("UIRoleAttribute", self.tbOwner.dwRoleId)
  end,
  AttributeInfo = function(self)
    self.pPanel:SetActive(self.AttributeInfo, false)
  end,
  btnCloseAttributeInfo = function(self)
    self.pPanel:SetActive(self.AttributeInfo, false)
    self.pPanel:SetActive(self.btnCloseAttributeInfo, false)
  end
}
tbWnd.tbOnSwipe = {
  HitZone = function(self, pos)
    Fashion:SwipeModel(pos.x)
  end
}

function tbWnd:OnCreate()
  self.tbEquipPanel = Ui:CreatePanel("UIEquipPanel", tbControls.UIEquipPanel, self, self.tbOwner.nSex)
  self.tbEquipPanel:UnRegistRedPoint()
  self.pPanel:SetActive(self.AttributeInfo, false)
  self.tbBaseAttribScrollView = Ui:NewScrollView(self, tbControls.goBaseAttribTemplate, function(self, nIndex, tbElement)
    self.pPanel:SetActive(self.btnCloseAttributeInfo, true)
    self.pPanel:SetActive(tbControls.AttributeInfo, true)
    local szText = tbBaseAttribute[nIndex].fnDesc(self.pAsyncRole, self.tbAsyncAttr)
    self.pPanel:Label_SetText(tbControls.AttributeInfo .. "/imgBg/txtInfo", szText)
    local tbPosition = tbElement.pPanel:GetSelfPosition()
    self.pPanel:Object_SetPosition(tbControls.AttributeInfo, Ui.Vector3(tbPosition.x - 3.5, tbPosition.y, tbPosition.z))
    self.tbBaseAttribScrollView:ClearSelected()
  end)
  local obj = self.pPanel:GetObject(tbControls.txtHonor)
  self.pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
end

function tbWnd:OnOpen()
  local obj = self.pPanel:GetObject(tbControls.NpcModelPath)
  self.NpcModel = obj:GetComponent(Ui.UINpcModel)
  Fashion:Preview_Init(obj)
  Fashion:SetResDir(0)
  self.pAsyncRole = self.tbOwner.pAsyncRole
  self.tbAsyncAttr = self.pAsyncRole.GetAsyncAttr()
  self.tbEquipPanel:OnOpen()
  self:ShowRoleInfo()
  self:UpdateAttributeScrollView()
  self:ShowModel()
  self.pPanel:SetActive(tbControls.btnOtherAttrib, false)
end

function tbWnd:OnClose()
  self.NpcModel:Release()
  self.tbEquipPanel:OnClose()
end

function tbWnd:ShowRoleInfo()
  local pAsyncRole = self.pAsyncRole
  local tbRole = ViewRole:GetRoleInfoById(pAsyncRole.dwID, true)
  if not tbRole then
    Log("[VIEW ROLE]==pAsyncRole, pAsyncRole.dwID, self.nPlayerId==", pAsyncRole, pAsyncRole.dwID, self.tbOwner.nPlayerId)
    return
  end
  local nSumLevel = 0
  for i = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
    local nLevel = pAsyncRole.GetQiHun(i + 1)
    if nLevel then
      nSumLevel = nSumLevel + nLevel
    end
  end
  self.pPanel:Label_SetText(tbControls.txtQiHunValue, nSumLevel)
  self.pPanel:Label_SetText(tbControls.txtFightPowerTip, i18n.EasyGet("\230\136\152\229\138\155\229\128\188"))
  local szIcon = PlayerPortrait:GetIcon(tbRole.nPortrait)
  self.pPanel:Sprite_SetSprite(tbControls.imgRolePortrait, szIcon)
  PlayerPortraitFrame:SetPortraitFrame(self.pPanel, tbControls.imgRolePortraitFrame, tbRole.nPortraitFrame)
  self.pPanel:Label_SetText(tbControls.txtRoleName, pAsyncRole.szName)
  local szRemark = FriendShip:GetPlayerRemarkName(pAsyncRole.dwID)
  local nlen = Lib:StrLen(szRemark)
  if nlen > 0 then
    local szName = i18n.EasyGet("({0})", szRemark)
    self.pPanel:Label_SetText(tbControls.txtNoteName, szName)
    self.pPanel:SetActive(tbControls.txtNoteName, true)
  else
    self.pPanel:SetActive(tbControls.txtNoteName, false)
  end
  self.pPanel:Label_SetText(tbControls.QiHunValue, nSumLevel)
  self.pPanel:SetActive(tbControls.txtRoleIDTip, false)
  self.pPanel:Sprite_SetSprite(tbControls.imgFactionIcon, Faction:GetIcon(self.tbOwner.nFaction))
  if pAsyncRole.GetLevel() >= GameSetting.MAX_PLAYER_LEVEL then
    self.pPanel:Label_SetText(tbControls.txtLevelTitle, i18n.EasyGet("\229\174\151\229\184\136"))
    self.pPanel:Label_SetText(tbControls.txtLevel, pAsyncRole.GetZongShiLevel())
  else
    self.pPanel:Label_SetText(tbControls.txtLevelTitle, i18n.EasyGet("\231\173\137\231\186\167"))
    self.pPanel:Label_SetText(tbControls.txtLevel, pAsyncRole.GetLevel())
  end
  local nSeries = KPlayer.GetPlayerInitInfo(self.tbOwner.nFaction).nSeries
  local szSeries = Npc.Series[nSeries]()
  self.pPanel:Label_SetText(tbControls.txtSeries, szSeries)
  for k, v in pairs(tbSeriesNumberToTextName) do
    if k == nSeries then
      self.pPanel:SetActive(tbControls.btnSeriesText .. "/" .. v, true)
    else
      self.pPanel:SetActive(tbControls.btnSeriesText .. "/" .. v, false)
    end
  end
  self.tbEquipPanel:SetFightPower(pAsyncRole.GetFightPower())
  self.tbEquipPanel:UpdateJZGradeScore(pAsyncRole.GetJingZhuGradeScore())
  self.tbEquipPanel:UpdateStoneGradeScore(pAsyncRole.GetStoneGradeScore())
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    self.tbEquipPanel:SetEquip(i, self.tbOwner.tbEquips[i - 1], pAsyncRole)
  end
  self.tbEquipPanel:SetEquip(Item.EQUIPPOS_MANTLE + 1, self.tbOwner.tbEquips[Item.EQUIPPOS_MANTLE], pAsyncRole)
  self.tbEquipPanel:SetEquip(Item.EQUIPPOS_GUANYIN + 1, self.tbOwner.tbEquips[Item.EQUIPPOS_GUANYIN], pAsyncRole)
  self.tbEquipPanel:SetEquip(Item.EQUIPPOS_WUXINGYIN + 1, self.tbOwner.tbEquips[Item.EQUIPPOS_WUXINGYIN], pAsyncRole)
  local tbAsyncAttr = self.tbAsyncAttr
  self.pPanel:Label_SetText(tbControls.txtHP, tbAsyncAttr[7])
  self.pPanel:Label_SetText(tbControls.txtMP, tbAsyncAttr[37])
  self.pPanel:Label_SetText(tbControls.txtAttackValue, tbAsyncAttr[8])
  local nGV = GoodwillValue.Cfg.nMaxValue - pAsyncRole.GetGoodwillValue()
  self.pPanel:Label_SetText(tbControls.txtPKValue, nGV)
  self:UpdateHonor(pAsyncRole.GetHonorStarLevel())
  self.pPanel:Label_SetText(tbControls.txtFightPower, pAsyncRole.GetFightPower())
end

function tbWnd:UpdateHonor(nHonorStarLevel)
  if nHonorStarLevel > 0 then
    self.pPanel:Label_SetText(tbControls.txtHonorTip, i18n.EasyGet("\229\164\180\232\161\148"))
  else
    self.pPanel:Label_SetText(tbControls.txtHonorTip, i18n.EasyGet("\230\154\130\230\151\160\229\164\180\232\161\148"))
  end
  local szName, szColor = HonorLevel:GetHonorNameByStarLevel(nHonorStarLevel, false)
  self.pHonorTitle:SetHonorTitle(i18n.Parse(szName), szColor)
end

function tbWnd:UpdateAttributeScrollView()
  local pAsyncRole = self.tbOwner.pAsyncRole
  local tbAsyncAttr = self.tbAsyncAttr
  for i = 1, #tbBaseAttribute do
    self.tbBaseAttribScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtAttributeName", string.format("%s:", tbBaseAttribute[i].fnName()))
      pPanel:Label_SetText("txtAttributeValue", tbBaseAttribute[i].func(pAsyncRole, tbAsyncAttr))
    end)
  end
end

function tbWnd:ShowModel()
  local tbFeature = {}
  local pAsyncRole = self.pAsyncRole
  for nEquipPos, nIndex in pairs(Item.EQUIPPOS_TO_NPC_RES_PART) do
    local nResId
    local nFashionId = pAsyncRole.GetFashion(nIndex + 1)
    if nFashionId > 0 then
      local tbFashionSetting = Fashion:GetFashionSettingNotPlayer(nFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
      if nil ~= tbFashionSetting then
        nResId = tbFashionSetting.nResId
      end
    end
    if nil == nResId then
      local nItemId = self.tbOwner.tbEquips[nEquipPos]
      if nItemId and nItemId > 0 then
        local pEquip = KItem.GetItemObj(nItemId)
        local _, _, _, _, nInnerResId = KItem.GetItemShowInfo(pEquip.dwTemplateId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
        nResId = nInnerResId
      end
    end
    if nil ~= nResId then
      tbFeature[nIndex] = nResId
    else
      tbFeature[nIndex] = 0
    end
  end
  if tbFeature[Npc.NpcResPartsDef.npc_part_body] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_body] = 1
  end
  if tbFeature[Npc.NpcResPartsDef.npc_part_head] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_head] = 1
  end
  local nWeaponEffectResId = 0
  if tbFeature[Npc.NpcResPartsDef.npc_part_weapon] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_weapon] = 1
  elseif tbFeature[Npc.NpcResPartsDef.npc_part_weapon] > 0 then
    local nWeaponEffectFashionId = pAsyncRole.GetFashionEffect(Npc.NpcResPartsDef.npc_part_weapon + 1) or 0
    if nWeaponEffectFashionId > 0 then
      local tbSetting = Fashion:GetFashionSettingNotPlayer(nWeaponEffectFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
      if tbSetting then
        nWeaponEffectResId = tbSetting.nResId
      end
    end
    if not nWeaponEffectResId or nWeaponEffectResId <= 0 then
      local nWeaponId = self.tbOwner.tbEquips[Item.EQUIPPOS_WEAPON] or 0
      if nWeaponId > 0 then
        local pWeapon = KItem.GetItemObj(nWeaponId)
        local tbPlayer = {
          nFaction = self.pAsyncRole.GetFaction(),
          nFactionSect = self.pAsyncRole.GetFactionSect(),
          nSex = self.pAsyncRole.GetSex()
        }
        nWeaponEffectResId = Item.EquipEnhance:GetEffectResId(tbPlayer, self.pAsyncRole.GetEnhance(Item.EQUIPPOS_WEAPON + 1))
      end
    end
  end
  local nHorseEffectResId = 0
  if tbFeature[Npc.NpcResPartsDef.npc_part_horse] > 0 then
    local tbPlayer = {
      nFaction = self.pAsyncRole.GetFaction(),
      nFactionSect = self.pAsyncRole.GetFactionSect(),
      nSex = self.pAsyncRole.GetSex()
    }
    local nHorseFashionId = pAsyncRole.GetFashionEffect(Npc.NpcResPartsDef.npc_part_horse + 1) or 0
    if nHorseFashionId > 0 then
      local tbSetting = Fashion:GetFashionSettingNotPlayer(nHorseFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
      if tbSetting then
        nHorseEffectResId = tbSetting.nResId
      end
    end
  end
  self.NpcModel:ShowModel(self.pAsyncRole.GetSex(), 1, tbFeature[Npc.NpcResPartsDef.npc_part_body], tbFeature[Npc.NpcResPartsDef.npc_part_weapon], tbFeature[Npc.NpcResPartsDef.npc_part_mantle], tbFeature[Npc.NpcResPartsDef.npc_part_horse], tbFeature[Npc.NpcResPartsDef.npc_part_head], tbFeature[Npc.NpcResPartsDef.npc_part_wind], nWeaponEffectResId, nHorseEffectResId)
end
