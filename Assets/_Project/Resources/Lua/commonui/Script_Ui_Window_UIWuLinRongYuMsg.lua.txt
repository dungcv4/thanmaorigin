local tbWnd = Ui:CreateClass("UIWuLinRongYuMsg", "UIPanel")
local tbControls = {
  Template = "imgList/List/Template"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewSuperListView(self, tbControls.Template, self.SetRankInfo, nil, nil)
end

function tbWnd:OnOpen(bOnSync)
  self.tbScrollView:SetFuncSetItem(self.SetRankInfo)
  local tbRankList = WuLinHonor:GetRankList(bOnSync)
  self.tbScrollView:SetAllElement(tbRankList)
  self.tbScrollView:MovePanelToFirstItem()
end

function tbWnd:SetRankInfo(tbElement, tbInfo, nIndex)
  tbElement.pPanel:Label_SetText("txtName", tbInfo.szName)
  tbElement.pPanel:Label_SetText("txtTong", tbInfo.szKinOrTongName or "")
  local SpFaction = Faction:GetIcon(tbInfo.nFaction)
  tbElement.pPanel:Sprite_SetSprite("imgFaction", SpFaction)
  if not tbElement.tbIconScrollView then
    tbElement.tbIconScrollView = Ui:NewScrollView(tbElement, "Icon/IconTemp")
  else
    tbElement.tbIconScrollView:Reset()
  end
  for i = 1, tbInfo.nPlatinaNum do
    tbElement.tbIconScrollView:SetElement(i, function(tbEle)
      tbEle.pPanel:Sprite_SetSprite("Main", WuLinHonor.szPlatinaIcon, true)
    end)
  end
end

function tbWnd:UpdateSV()
  self:OnOpen(true)
end
