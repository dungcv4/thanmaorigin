local tbWnd = Ui:CreateClass("UIChildrensMain", "UIActivityMainBase")
local tbControls = {
  Template = "List/Viewport/Content/Template",
  szTitle = "Tex/txtTitle",
  szTime = "Tex/txtTime01",
  btnHelp = "Tex/txtTitle/btnHelp",
  txtTimeTittle01 = "Tex/txtTime01/txtTimeTittle01"
}
tbWnd.tbControls = tbControls
tbWnd.tbTemplateElement = {
  txtActivityName = "ImgAll/txtActivityName",
  txtTimes = "ImgAll/txtTimes",
  txtInfo = "ImgAll/txtInfo",
  ItemTemplate = "ImgAll/ContList/Viewport/Content/ItemTemplate%d",
  GridTemplate = "ImgAll/ContList/Viewport/Content/ItemTemplate%d/UIItemGrid",
  img_Finish = "ImgAll/img_Finish",
  txtUnOpened = "ImgAll/txtUnOpened",
  btnGo = "ImgAll/btnGo"
}

function tbWnd:UpdateInfo(nEntranceIndex, _, szWindowName)
  self.szWindowName = szWindowName
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local szShowTime = ActivityMgr.EntranceSchedule:GetSmallTitle(nEntranceIndex)
  self.pPanel:Label_SetText(self.szTitle, ActivityMgr.EntranceSchedule:GetTitle(nEntranceIndex))
  self.pPanel:Label_SetText(self.szTime, ActivityMgr.EntranceSchedule:GetSmallTitle(nEntranceIndex))
  local nEndTime = Lib:GetConfigDate2Time(ActivityMgr.EntranceScheduleCfg[nEntranceIndex].szEndTime)
  if nEndTime > GetTime() then
    local nDiff = nEndTime - GetTime()
    self.pPanel:Label_SetText(tbControls.szTime, i18n.EasyGet("<color=#1FFB24>{0}</color>", szShowTime))
    self.pPanel:Label_SetText(tbControls.txtTimeTittle01, i18n.EasyGet("{0}\229\144\142\231\187\147\230\157\159", Lib:TimeDesc2(nDiff)))
  end
  self:UpdateActivityList(nEntranceIndex)
  self.pPanel:SetActive(self.btnHelp, false)
end
