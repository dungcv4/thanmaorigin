SnowFight = SnowFight or {}
SnowFight.RaceMap = SnowFight.RaceMap or {}
local RaceMap = SnowFight.RaceMap

function SnowFight:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    SnowFight.Cfg.nPrepareMapTemplateId
  }, self)
end

function SnowFight:OnFubenMapLoaded(pPlayer)
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if not tbWnd then
    return
  end
  local subPanel = tbWnd:GetSubPanel("UIHudLeftPanelFuben")
  if subPanel.szFubenType == "SnowFight" then
    return
  else
    tbWnd:SetPanelState(tbWnd.panelTaskTeamFuBen, tbWnd.TP_FUBEN)
    Ui:OpenHudFubenCustomWithTeam("UISnowFight", "SnowFight")
  end
end

function SnowFight:OnLeave()
  FubenUIController:CloseFubenUI()
end

function SnowFight:IsNeedExitFubenButton()
  return true
end

SnowFight:RegisterFubenUIChecker()

function RaceMap:RegisterRaceMapUIChecker()
  FubenUIController:RegisterChecker({
    SnowFight.Cfg.nFightMapTemplateId
  }, self)
end

function RaceMap:OnFubenMapLoaded()
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if not tbWnd then
    return
  end
  local subPanel = tbWnd:GetSubPanel("UIHudLeftPanelFuben")
  if subPanel.szFubenType == "SnowFight" then
    return
  else
    tbWnd:SetPanelState(tbWnd.panelTaskTeamFuBen, tbWnd.TP_FUBEN)
    Ui:OpenHudFubenCustomWithTeam("UISnowFight", "SnowFight")
  end
end

function RaceMap:OnLeave()
  FubenUIController:CloseFubenUI()
end

RaceMap:RegisterRaceMapUIChecker()

function SnowFight:UpdatePreMapCount(nSignUpPlayerNum, nTotalTime)
  local nNow = GetTime()
  self:ShowHudInfo(SnowFight.RaceStatus.Prepare, nSignUpPlayerNum, nTotalTime + nNow)
end

function SnowFight:UpdateFightMapUI(nTotalTime, nBossTime, nFightRound)
  local nNow = GetTime()
  self:ShowHudInfo(SnowFight.RaceStatus.Fight, nTotalTime + nNow, nBossTime + nNow, nFightRound)
end

function SnowFight:ShowHudInfo(szStatus, ...)
  self.szStatus = szStatus
  self.tbEndTime = {
    ...
  }
  if SnowFight.RaceStatus[szStatus] then
    Ui:OpenHudFubenCustomWithTeam("UISnowFight", "SnowFight", szStatus, ...)
  end
end

function SnowFight:UpdateRestMapUI(nTotalTime, nFightRound)
  local nNow = GetTime()
  self:ShowHudInfo(SnowFight.RaceStatus.Rest, nTotalTime + nNow, nFightRound)
end

function SnowFight:UpdateSettlementMapUI(nTotalTime)
  local nNow = GetTime()
  self:ShowHudInfo(SnowFight.RaceStatus.Settlement, nTotalTime + nNow)
end

function SnowFight:OnSyncRankData(tbRankInfo)
  self.tbRankData = tbRankInfo or self.tbRankData
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_SNOWFIGHT_RANK, self.tbRankData)
end

function SnowFight:OnUpdatePrepareInfo(nJoinPlayerNum, nSignUpTimer)
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_BATTLEENTRANCE_COMMON_UPDATE, nil, nJoinPlayerNum, nSignUpTimer)
end
