local tbWnd = Ui:CreateWindow("UIFightPowerTips")
local nShowTime = 100
local nDelayCloseTime = 1.5
tbWnd.tbControls = {
  txtValue = "FightPower/Txt/txtValue"
}

function tbWnd:OnCreate()
  local txtFightPowerGo = self.pPanel:GetObject(self.txtValue)
  self.rollAnimator = txtFightPowerGo:GetComponent(typeof(CS.UITweenRollNumber))
  self.txtFightPower = self.pPanel:GetText(self.txtValue)
  self.bShow = false
  self.nStep = 1
end

function tbWnd:OnOpen(nEnd, nStart)
  if self.nCloseTimer and -1 ~= Timer:GetRestTime(self.nCloseTimer) then
    Timer:Close(self.nCloseTimer)
  end
  Sound:PlaySound("WealthAscension", _)
  self.rollAnimator:Play(nStart, nEnd, function()
    self.nCloseTimer = Timer:Register(Env.GAME_FPS * nDelayCloseTime, function()
      self:Close()
      return false
    end)
  end)
end
