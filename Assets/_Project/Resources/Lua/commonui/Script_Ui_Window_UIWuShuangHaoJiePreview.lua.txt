local tbWnd = Ui:CreateWindow("UIWuShuangHaoJiePreview")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SERVER_TIME_ZONE_UPDATE,
      self.UpdateLeftTime
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  ItemTemplate = "node/imgBG/Content/Preview/AwardList/Viewport/Content/ItemTemplate",
  txtInformation = "node/imgBG/Content/Preview/txtInfoList/Viewport/Content/txtInformation",
  txtCountdownValue = "node/imgBG/Content/Preview/Countdown/txtCountdownValue"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbAwardScrollView = Ui:NewItemGridScrollView(self, tbControls.ItemTemplate, 1, 1)
end

function tbWnd:OnClose()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:OnOpen()
  self:UpdateLeftTime()
  self:UpdatePreview()
end

function tbWnd:UpdateLeftTime()
  local szTime = ""
  local nEndTime = Lib:GetConfigDate2Time(WuLinDaHui.SignCfg.START_DATE)
  local nTimeRemaining = nEndTime - GetTime()
  if nTimeRemaining > 0 then
    szTime = Lib:TimeDesc2(nTimeRemaining)
  end
  self.pPanel:Label_SetText(tbControls.txtCountdownValue, szTime)
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nCurTime = tonumber(os.date("%Y%m%d%H%M", GetTime()))
  local nNextTime = 1
  if nTimeRemaining > 3600 then
    nNextTime = Lib:GetDate2Time(nCurTime + 1) - GetTime() + 1
  end
  if nTimeRemaining > 0 then
    self.nTimer = Timer:Register(nNextTime * Env.GAME_FPS, self.UpdateLeftTime, self)
  end
  return false
end

function tbWnd:UpdatePreview()
  self.pPanel:Label_SetText(tbControls.txtInformation, i18n.Get(WuLinDaHui.UiDesc.szPreViewInfo))
  local tbAwards = Award:GetAward(WuLinDaHui.PreviewShowAward.nAwardId, WuLinDaHui.PreviewShowAward.nAwardKey)
  self.tbAwardScrollView:UpdateGenericItem(tbAwards, true)
end
