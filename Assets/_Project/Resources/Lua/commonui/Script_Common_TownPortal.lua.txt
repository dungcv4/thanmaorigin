function TownPortal:CanDoTownPortal(bNoMsg)
  local pNpc = me.GetNpc()
  
  if not pNpc then
    return false
  end
  local nResult = pNpc.CanDoTownPortal()
  if 1 ~= nResult then
    if 4 == nResult and not bNoMsg then
      me.CenterMsg(i18n.EasyGet("\229\155\158\229\159\142\232\191\152\229\156\168CD\228\184\173"))
    end
    return false
  end
  local nTemplateId = me.nMapTemplateId
  local bIsMapForbid = Map:IsTransForbid(nTemplateId)
  if bIsMapForbid then
    if not bNoMsg then
      me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\231\166\129\230\173\162\228\188\160\233\128\129"))
    end
    return false
  end
  if me.nFightMode == Npc.FIGHT_MODE.emFightMode_None then
    if not bNoMsg then
      me.CenterMsg(i18n.EasyGet("\229\174\137\229\133\168\229\140\186\229\159\159\229\134\133\230\151\160\230\179\149\228\189\191\231\148\168\229\155\158\229\159\142"))
    end
    return false
  end
  local nDoing = me.GetDoing()
  if nDoing == Npc.Doing.death or nDoing == Npc.Doing.revive then
    if not bNoMsg then
      me.CenterMsg(i18n.EasyGet("\230\173\187\228\186\161\231\138\182\230\128\129\230\151\160\230\179\149\228\189\191\231\148\168\229\155\158\229\159\142"))
    end
    return false
  end
  return true
end

function TownPortal:DoTownPortal()
  local pNpc = me.GetNpc()
  if not pNpc then
    return
  end
  local bCanDoTownPortal = self:CanDoTownPortal()
  if not bCanDoTownPortal then
    return
  end
  AutoPath:StopGoto()
  RemoteServer.TownPortal()
end

function TownPortal:TownPortalHandle()
  local pNpc = me.GetNpc()
  if not pNpc then
    return
  end
  pNpc.TownPortalHandle()
end

function TownPortal:DoTownPortalResult()
  local szMsg = i18n.EasyGet("\229\183\178\229\155\158\229\136\176\229\174\137\229\133\168\229\140\186")
  me.SendBlackBoardMsg(szMsg, true, nil)
end
