local tbWnd = Ui:CreateWindow("UIWuLinRongYuAward")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/txtTitle",
  Template = "node/imgBG/Content/AwardList/ScrollView/Viewport/Content/Template",
  UIItemGrid = "ItemList/Item/UIItemGrid"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewSuperListView(self, tbControls.Template, self.SetAwardList, nil, nil)
end

function tbWnd:OnOpen(nActivityId)
  if not nActivityId then
    return false
  end
  self.tbScrollView:SetFuncSetItem(self.SetAwardList)
  self.nActivityId = nActivityId
  self.pPanel:Label_SetText(tbControls.txtTitle, WuLinHonor:GetNameByActivityType(self.nActivityId))
  local tbAwardList = WuLinHonor:GetAwardShowList(self.nActivityId)
  self.tbScrollView:SetAllElement(tbAwardList)
end

function tbWnd:SetAwardList(tbElement, tbInfo, nIndex)
  tbElement.pPanel:Label_SetText("txtRank", tbInfo.szText)
  tbElement.pPanel:Label_SetColorByName("txtRank", tbInfo.szTextColor)
  tbElement.pPanel:Sprite_SetSprite("IconTemp", tbInfo.szIconPath)
  if not tbElement.tbItemGridScrollView then
    tbElement.tbItemGridScrollView = Ui:NewItemGridScrollView(tbElement, "ItemList/Item", 2, 1)
  else
    tbElement.tbItemGridScrollView:Reset()
  end
  tbElement.tbItemGridScrollView:UpdateGenericItem(tbInfo.tbAward, true)
end
