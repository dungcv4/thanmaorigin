function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_VIEWROLE_DATA,
      self.OnGetAsyncData,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbwndJingZhuInfo = Ui:CreatePanel("UIwndJingZhuInfo", tbControls.wndJingZhuInfo, self)
  self.tbwndSeriesActive = Ui:CreatePanel("UISeriesActive", tbControls.wndSeriesActive, self)
  self.tbwndQiHunInfo = Ui:CreatePanel("UIwndQiHunInfo", tbControls.wndQiHunInfo, self)
  self.tbwndEquipSuitInfo = Ui:CreatePanel("UIEquipSuitInfoSmall", tbControls.wndEquipSuit, self)
  self.tbwndHorsePassiveSkill = Ui:CreatePanel("UIwndHorsePassiveSkills", tbControls.wndHorsePassiveSkills, self)
  self.tbwndDivineSuit = Ui:CreatePanel("UIDivineSuit", tbControls.wndDivineSuit, self)
  self.tbwndDivineHonor = Ui:CreatePanel("UIDivineHonor", tbControls.wndDivineHonor, self)
  self:SetClickEmptyCallback(tbControls.wndEquipSuit)
  self:SetClickEmptyCallback(tbControls.wndDivineHonor)
  self:SetClickEmptyCallback(tbControls.wndDivineSuit)
  self:SetClickEmptyToClose(tbControls.wndHorsePassiveSkills)
  self.pPanel:SetActive(tbControls.wndHorsePassiveSkills, false)
end

function tbWnd:OnOpen(tbEquip, tbExtEquip, tbPartnerInfo, dwRoleId)
  local pAsyncRole = KPlayer.GetAsyncData(dwRoleId)
  self.dwRoleId = dwRoleId
  self.pAsyncRole = pAsyncRole
  self.tbEquips = tbEquip
  self.tbExtEquips = tbExtEquip
  self.tbPartners = tbPartnerInfo
  self.nFaction = pAsyncRole.GetFaction()
  self.nFactionSect = pAsyncRole.GetFactionSect()
  self.nSex = self.pAsyncRole.GetSex()
  self.nPlayerId = pAsyncRole.dwID
  self.nVipLevel = pAsyncRole.GetVipLevel()
  self.nPanelType = self.PANEL_TYPE_PLAYER_INFO
  if not self.tbTabControl:OnOpen(self.nPanelType, tbEquip, tbExtEquip, tbPartnerInfo, dwRoleId) then
    return 0
  end
end

function tbWnd:OnClickEmptyClose()
  self.pPanel:SetActive(tbControls.wndEquipSuit, false)
  self.pPanel:SetActive(tbControls.wndDivineHonor, false)
  self.pPanel:SetActive(tbControls.wndDivineSuit, false)
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  self.nPanelType = self.PANEL_TYPE_PLAYER_INFO
end

function tbWnd:HasZongShiSkill()
  local pAsyncRole = self.pAsyncRole
  if pAsyncRole.GetLevel() >= GameSetting.MAX_PLAYER_LEVEL then
    return true
  end
  return false
end

function tbWnd:IsHadPartner()
  if self.tbPartners and Lib:CountTB(self.tbPartners) > 0 then
    return true
  end
  return false
end

function tbWnd:HasHorse()
  local pAsyncRole = self.pAsyncRole
  local nCurGrade = pAsyncRole.GetHorseGrade()
  if not Horse.tbGradeInfoCfg[nCurGrade] then
    return false
  end
  return true
end

function tbWnd:HasMiji()
  local SecretBook = Item.SecretBook
  local nPos, nItemId
  local pAsyncRole = self.pAsyncRole
  for _, nRank in pairs(SecretBook.tbRank) do
    nPos = SecretBook.tbRankToSlot[nRank]
    nItemId = self.tbExtEquips[nPos]
    if nItemId and nItemId > 0 then
      return true
    end
  end
  return false
end

function tbWnd:SwitchPanel(nNewPanelIndex)
  self.nPanelType = nNewPanelIndex
  if self.dwRoleId ~= ViewRole.dwRoleId then
    local tbEquip, tbExtEquip, tbPartnerInfo = ViewRole:GetCurResInfo(self.dwRoleId)
    if not (tbEquip and tbExtEquip) or not tbPartnerInfo then
      return
    end
    self.tbEquips = tbEquip
    self.tbExtEquips = tbExtEquip
    self.tbPartners = tbPartnerInfo
  end
  self.tbTabControl:OnOpen(self.nPanelType, self.tbEquip, self.tbExtEquip, self.tbPartnerInfo, self.dwRoleId)
end

function tbWnd:OnGetAsyncData(dwID)
  if dwID ~= self.dwRoleId then
    return
  end
  local tbEquip, tbExtEquip, tbPartnerInfo = ViewRole:GetCurResInfo(self.dwRoleId)
  if not (tbEquip and tbExtEquip) or not tbPartnerInfo then
    return
  end
  self.tbEquips = tbEquip
  self.tbExtEquips = tbExtEquip
  self.tbPartners = tbPartnerInfo
  self.tbTabControl:OnOpen(self.nPanelType, tbEquip, tbExtEquip, tbPartnerInfo, self.dwRoleId)
end
