GeneralHelp = GeneralHelp or {}

function GeneralHelp:Init()
  local tbTemp = {}
  local tbHelpContent = Lib:LoadTabFile("Setting/Help/GeneralHelp.tab", {nIndex = 1})
  assert(tbHelpContent, "[GeneralHelp] LoadSetting GeneralHelp Fail")
  for _, tbInfo in pairs(tbHelpContent) do
    local szKey = tbInfo.szHelpKey
    tbTemp[szKey] = tbTemp[szKey] or {
      szBigTitle = tbInfo.szBigTitle,
      tbContent = {}
    }
    if tbInfo.nIndex > 0 then
      tbTemp[szKey].tbContent[tbInfo.nIndex] = {
        szContent = tbInfo.szContent,
        szTitle = tbInfo.szTitle
      }
    end
  end
  self.tbHelpInfo = tbTemp
  self.tbHelpInfo = SetReadOnly(self.tbHelpInfo)
end

if not GeneralHelp.tbHelpInfo then
  GeneralHelp:Init()
end
