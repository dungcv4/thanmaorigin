Player.tbPlayerFeature = Player.tbPlayerFeature or {}
local tbPlayerFeature = Player.tbPlayerFeature
if MODULE_GAMESERVER then
  function tbPlayerFeature:OnLogin(pPlayer)
  end
  
  function tbPlayerFeature:SetMantleDisableState(pPlayer, nState)
    self:UpdateMantleFeature(pPlayer)
    pPlayer.CallClientScript("Player.tbPlayerFeature:UpdateMantleFeature")
  end
end

function tbPlayerFeature:UpdateMantleFeature(pPlayer)
  pPlayer = pPlayer or me
  local pNpc = pPlayer.GetNpc()
  if not pNpc then
    return
  end
  local pMantle = pPlayer.GetEquipByPos(Item.EQUIPPOS_MANTLE)
  if not pMantle then
    return
  end
  local _, _, _, _, nResId = KItem.GetItemShowInfo(pMantle.dwTemplateId, pPlayer.nFaction, pPlayer.nFactionSect, pPlayer.nSex)
  if 0 ~= nResId then
    pNpc.ModifyPartFeatureEquip(Npc.NpcResPartsDef.npc_part_mantle, nResId, Npc.NpcPartLayerDef.npc_part_layer_base)
  end
end
