local tbWnd = Ui:CreateWindow("UIQuickUseItemOB")
local tbControls = {
  btnClose = "imgBG/QuickUsePanel/btnClose",
  btnUse = "imgBG/QuickUsePanel/btnUse",
  ItemGrid = "imgBG/QuickUsePanel/Item/ItemGrid"
}
tbWnd.tbControls = tbControls

function tbWnd.NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      tbWnd.OnItemUpdate
    },
    {
      EventNotify.emNOTIFY_DEL_ITEM,
      tbWnd.OnItemDelete
    }
  }
end

tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnUse = function(self)
    if self.nItemId > 0 then
      RemoteServer.UseItem(self.nItemId)
    end
  end
}

function tbWnd:OnCreate()
  self.itemGrid = Ui:NewExistItemGrid(self, self.ItemGrid)
end

function tbWnd:OnOpen(nItemId, szBtnName)
  self.nItemId = nItemId
  self.szBtnName = szBtnName
  self:Update()
end

function tbWnd:Update(nItemId, szBtnName)
  local pItem = KItem.GetItemObj(self.nItemId or -1)
  if not pItem then
    Timer:Register(1, Ui.CloseWindow, Ui, "UIQuickUseItemOB")
    return
  end
  self.pPanel:Button_SetText(self.btnUse, i18n.Parse(self.szBtnName), "Text")
  local nItemCount = me.GetItemCountByTypeInBag(pItem.dwTemplateId)
  self.itemGrid:SetItemByTemplate(pItem.dwTemplateId, nItemCount, nil, nil, {bShowTip = true})
  self.itemGrid:SetClickEnable(true)
end

function tbWnd:OnClose()
  self.nItemId = 0
  self.szBtnName = i18n.EasyGet("\228\189\191\231\148\168")
end

function tbWnd:GetItemId()
  return self.nItemId
end

function tbWnd:OnItemUpdate(nItemId)
  if nItemId == self.nItemId then
    self:Update()
  end
end

function tbWnd:OnItemDelete(nItemId)
  if nItemId == self.nItemId then
    self:Update()
  end
end
