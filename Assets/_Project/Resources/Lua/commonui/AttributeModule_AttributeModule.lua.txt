AttributeModule.tbModule = AttributeModule.tbModule or {}
AttributeModule.bEnableDebugLog = false
AttributeModule.bEnable = {
  EquipJingZhu = true,
  EquipLianHua = true,
  QiHun = true,
  Stone = true,
  ZhenFa = true,
  KinSkill = true,
  Partner = true,
  HonorLevel = true,
  Fashion = true,
  JingMai = true,
  PartnerEquip = true,
  Blessing = true,
  HorseSkill = true,
  ShenBingPu = true,
  ZongShiSkill = true,
  PartnerZhenFa = true,
  EquipCuiLing = true,
  WXHuShi = true
}

function AttributeModule:OnEquip(pPlayer)
  if MODULE_GAMECLIENT and not pPlayer then
    pPlayer = me
  end
  if not self.bEnable or not pPlayer then
    return
  end
  if not pPlayer.tbAttributeModule then
    pPlayer.tbAttributeModule = {}
  end
  for szClass, tbClass in pairs(self.tbModule) do
    if self.bEnableDebugLog then
      print("AttributeModule:OnEquip", szClass, pPlayer.szName)
    end
    if self.bEnable[szClass] and not pPlayer.tbAttributeModule[szClass] then
      assert(pPlayer.tbAttributeModule[szClass] == nil)
      if tbClass.OnEquip then
        tbClass:OnEquip(pPlayer)
        pPlayer.tbAttributeModule[szClass] = true
      else
        LogErr(string.format("%s need define OnEquip function", szClass))
      end
    end
  end
end

function AttributeModule:OnUnEquip(pPlayer)
  if MODULE_GAMECLIENT and not pPlayer then
    pPlayer = me
  end
  if not self.bEnable or not pPlayer then
    return
  end
  if not pPlayer.tbAttributeModule then
    pPlayer.tbAttributeModule = {}
  end
  for szClass, tbClass in pairs(self.tbModule) do
    if self.bEnableDebugLog then
      print("AttributeModule:OnUnEquip", szClass, pPlayer.szName)
    end
    if self.bEnable[szClass] and pPlayer.tbAttributeModule[szClass] then
      if tbClass.OnUnEquip then
        tbClass:OnUnEquip(pPlayer)
        pPlayer.tbAttributeModule[szClass] = nil
      else
        LogErr(string.format("%s need define OnUnEquip function", szClass))
      end
    end
  end
end

function AttributeModule:OnCreatePlayerRobotNpc(pPlayerAsync, pNpc)
  if not self.bEnable or not pNpc then
    return
  end
  for szClass, tbClass in pairs(self.tbModule) do
    if self.bEnableDebugLog then
      print("AttributeModule:OnCreatePlayerRobotNpc", szClass, pNpc.szName)
    end
    if self.bEnable[szClass] then
      if tbClass.OnCreatePlayerRobotNpc then
        tbClass:OnCreatePlayerRobotNpc(pPlayerAsync, pNpc)
      else
        LogErr(string.format("%s need define OnCreatePlayerRobotNpc function", szClass))
      end
    end
  end
end

function AttributeModule:SetAttrActive(pPlayer, bEnable, szClass)
  if not self.bEnable or not pPlayer then
    return
  end
  if not pPlayer.tbAttributeModule then
    pPlayer.tbAttributeModule = {}
  end
  if not self.bEnable[szClass] then
    return
  end
  local tbClass = self.tbModule[szClass]
  if not tbClass then
    return
  end
  if bEnable and not pPlayer.tbAttributeModule[szClass] and tbClass.OnEquip then
    tbClass:OnEquip(pPlayer)
    pPlayer.tbAttributeModule[szClass] = true
  elseif not bEnable and pPlayer.tbAttributeModule[szClass] and tbClass.OnUnEquip then
    tbClass:OnUnEquip(pPlayer)
    pPlayer.tbAttributeModule[szClass] = nil
  end
end

function AttributeModule:Register(szClass, tbClass)
  self.tbModule[szClass] = tbClass
end

function AttributeModule:CheckSystemIsHadFunc()
  for szClass, tbModule in pairs(self.tbModule) do
    if not tbModule.OnEquip then
      LogErr("AttributeModule:CheckSystemIsHadFunc error!", string.format("szClass:%s not had OnEquip function", szClass))
      return false
    end
    if not tbModule.OnUnEquip then
      LogErr("AttributeModule:CheckSystemIsHadFunc error!", string.format("szClass:%s not had OnUnEquip function", szClass))
      return false
    end
    if not tbModule.OnCreatePlayerRobotNpc then
      LogErr("AttributeModule:CheckSystemIsHadFunc error!", string.format("szClass:%s not had OnCreatePlayerRobotNpc function", szClass))
      return false
    end
  end
  return true
end

Validator:Register(AttributeModule.CheckSystemIsHadFunc, AttributeModule)
