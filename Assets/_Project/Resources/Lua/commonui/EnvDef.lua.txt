Env.GAME_FPS = 18
Env.LOGIC_MAX_DIR = 256
Env.INT_MAX = 2147483647
Env.UINT_MAX = 4294967295
Env.CAMERA_DEFAULT_X = 30
Env.DAILY_BIT_OFFSET = 2048
Env.NEW_DAY_OFFSET_TIME = 14400
local tbLogWayDesc = {}
local tbLogWayMainWays = {}
local tbLogWaySubWays = {}
local tbSettingPaths = {
  "Setting/LogWay.tab",
  "Setting/LogWayMBI.tab"
}
for _, szPath in pairs(tbSettingPaths) do
  local tbLogWaySet = Lib:LoadTabFileEx(szPath, "ssssds", nil, {
    "MsgType",
    "EventId",
    "MainWayName",
    "SubWayName",
    "Value",
    "Desc"
  }, nil, 2)
  for i, v in ipairs(tbLogWaySet) do
    if Env[v.SubWayName] and v.SubWayName ~= "" then
      LogErr(string.format("repeat key %s in %s!!!", v.SubWayName, szPath))
    end
    if v.SubWayName ~= "" then
      Env[v.SubWayName] = v.Value
      tbLogWayDesc[v.Value] = v.Desc
      tbLogWayMainWays[v.Value] = v.MainWayName
      tbLogWaySubWays[v.Value] = v.SubWayName
    end
  end
end
Env.tbLogWayDesc = tbLogWayDesc
Env.tbLogWayMainWays = tbLogWayMainWays
Env.tbLogWaySubWays = tbLogWaySubWays
local tbLogWayGameDesc = {}
local tbLogWaySet = Lib:LoadTabFileEx("Setting/LogWayGame.tab", "ds", nil, {"Value", "Desc"})
for _, tbInfo in pairs(tbLogWaySet) do
  if tbLogWayGameDesc[tbInfo.Value] then
    LogErr(string.format("repeat key %d in LogWayGame.tab !!!", tbInfo.Value))
  else
    tbLogWayGameDesc[tbInfo.Value] = tbInfo.Desc
  end
end
Env.tbLogWayGameDesc = tbLogWayGameDesc
local tbLogWaySet = Lib:LoadTabFileEx("Setting/LogWayTracking.tab", "dssss", nil, {
  "EventType",
  "MainWayName",
  "SubWayName",
  "Value",
  "ExtParams"
})
for _, tbInfo in pairs(tbLogWaySet) do
  if Env[tbInfo.Value] then
    LogErr(string.format("repeat key %s in LogWayTracking.tab !!!", tbInfo.Value))
  else
    Env[tbInfo.Value] = {
      tbInfo.EventType,
      tbInfo.MainWayName,
      tbInfo.SubWayName,
      Lib:SplitStr(tbInfo.ExtParams)
    }
  end
end
