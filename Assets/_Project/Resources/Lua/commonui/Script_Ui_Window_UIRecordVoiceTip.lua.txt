local tbWnd = Ui:CreateWindow("UIRecordVoiceTip")
tbWnd.tbControls = {
  speakTips = "imgBG/SpeakTips",
  micLevel1 = "imgBG/SpeakTips/MicLevel/1",
  micLevel2 = "imgBG/SpeakTips/MicLevel/2",
  micLevel3 = "imgBG/SpeakTips/MicLevel/3",
  micLevel4 = "imgBG/SpeakTips/MicLevel/4"
}
tbWnd.tbOnClick = {}

function tbWnd:OnOpen(nLevel)
  self:ShowMicTips(nLevel)
end

function tbWnd:ShowMicTips(nLevel)
  self.pPanel:SetActive(self.speakTips, true)
  self.pPanel:SetActive(self.micLevel1, nLevel > 0)
  self.pPanel:SetActive(self.micLevel2, nLevel > 1)
  self.pPanel:SetActive(self.micLevel3, nLevel > 2)
  self.pPanel:SetActive(self.micLevel4, nLevel > 3)
end

function tbWnd:HideMicTips()
  self.pPanel:SetActive(self.speakTips, false)
end
