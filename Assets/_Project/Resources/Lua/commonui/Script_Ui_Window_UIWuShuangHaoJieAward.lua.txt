local tbWnd = Ui:CreateWindow("UIWuShuangHaoJieAward")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnPreliminary = "node/imgBG/Content/btnList/btnPreliminary",
  btnWuShuangFinal = "node/imgBG/Content/btnList/btnWuShuangFinal",
  btnHaoJieFinal = "node/imgBG/Content/btnList/btnHaoJieFinal",
  txtTips = "node/imgBG/Content/txtTips",
  Template = "node/imgBG/Content/AwardList/ScrollView/Viewport/Content/Template",
  txtRank = "node/imgBG/Content/TitleList/txtRank"
}
tbWnd.tbControls = tbControls
tbWnd.PAGE_PRE = 1
tbWnd.PAGE_FINAL_WUSHUANG = 2
tbWnd.PAGE_FINAL_HAOJIE = 3
local tbPageBtnIndex = {
  [1] = tbControls.btnPreliminary,
  [2] = tbControls.btnWuShuangFinal,
  [3] = tbControls.btnHaoJieFinal
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbPageGroup = Ui:NewButtonGroup(self, tbPageBtnIndex, true, function(self, nIndex)
    self:OnSelect(nIndex)
  end)
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen()
  local nDefaultType = self.PAGE_PRE
  self.tbPageGroup:Select(nDefaultType)
  self.tbPageGroup:SetActive(self.PAGE_FINAL_HAOJIE, false)
  self.pPanel:Label_SetText(tbControls.btnWuShuangFinal .. "/Text", i18n.EasyGet("\229\134\179\232\181\155"))
end

function tbWnd:OnSelect(nIndex)
  if nIndex == self.PAGE_PRE then
    self.pPanel:Label_SetText(tbControls.txtTips, i18n.Get(WuLinDaHui.AwardTips.szPreTips))
  elseif nIndex == self.PAGE_FINAL_WUSHUANG then
    self.pPanel:Label_SetText(tbControls.txtTips, i18n.Get(WuLinDaHui.AwardTips.szFinalTips))
  end
  self.tbAwards = WuLinDaHui:GetAwardSetting(nIndex)
  self:UpdateView()
end

function tbWnd:UpdateView()
  self:SetTemplate()
end

function tbWnd:SetTemplate()
  self.ScrollView:Reset()
  for i, tbAward in ipairs(self.tbAwards) do
    self.ScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtRank", i18n.Parse(tbAward.szDesc))
      local tbAwards = Award:GetAward(tbAward.nAwardId, tbAward.nAwardKey)
      tbElement.tbAwardScrollView = tbElement.tbAwardScrollView or Ui:NewItemGridScrollView(tbElement, "Scroll ViewItem/Viewport/Content/Template", 1, 1)
      tbElement.tbAwardScrollView:UpdateGenericItem(tbAwards, true)
    end)
  end
end
