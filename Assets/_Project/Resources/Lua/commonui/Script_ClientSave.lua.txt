function Client:GetNumber(key, defaultV)
  local value = self:GetValue(key, defaultV)
  
  if nil == value or type(value) ~= "number" then
    assert(false)
    return 0
  end
  return value
end

function Client:SetNumber(key, value, bNoSave)
  if nil == value or type(value) ~= "number" then
    assert(false)
    return false
  end
  return self:SetValue(key, value, bNoSave)
end

function Client:GetValue(key, defaultV)
  if not key then
    assert(false)
    return nil
  end
  local tbFlag = Client:GetUserInfo("LocalFlags")
  local value = tbFlag[key]
  if nil == value then
    value = defaultV
  end
  return value
end

function Client:SetValue(key, value, bNoSave)
  if not key then
    assert(false)
    return false
  end
  if nil == value then
    assert(false)
    return false
  end
  local tbFlag = Client:GetUserInfo("LocalFlags")
  tbFlag[key] = value
  if not bNoSave then
    self:SaveUserInfo()
  end
  return true
end

function Client:ClearValue(key, bNoSave)
  local tbFlag = Client:GetUserInfo("LocalFlags")
  tbFlag[key] = nil
  if not bNoSave then
    self:SaveUserInfo()
  end
end

function Client:ResetUserInfo()
  self.tbUserInfo = nil
end

function Client:GetUserInfo(szType, nId, bUseNewTable)
  if not self.tbUserInfo then
    self:InitUserInfo()
  end
  if not nId then
    nId = -1
    if me and me.dwID then
      nId = me.dwID
    end
  end
  self.tbUserInfo[nId] = self.tbUserInfo[nId] or {}
  if bUseNewTable then
    self.tbUserInfo[nId][szType] = {}
  else
    self.tbUserInfo[nId][szType] = self.tbUserInfo[nId][szType] or {}
  end
  return self.tbUserInfo[nId][szType]
end

function Client:SaveUserInfo()
  local szValue = Lib:Val2Str(self.tbUserInfo)
  local szPath = g_szUserPath .. "User/UserInfo.lua"
  local szLuaValue = "return" .. szValue
  local bResult = Lib:WriteFile_W(szPath, "w+", szLuaValue)
  if not bResult then
    Log("Client:SaveUserInfo ERR ?? file is nil !!" .. szPath)
    return
  end
end

function Client:TransPrivateMsg()
  local szPath = g_szUserPath .. "User/UserPrivateMsg.lua"
  local File = CS.System.IO.File
  if not File.Exists(szPath) then
    return
  end
  local szFileContent, nLen = Lib:ReadFile_W(szPath, "r")
  if 0 == nLen then
    return
  end
  local tbTmpEnv = {}
  szFileContent = "local fnLoad = function () " .. szFileContent .. " end tbPrivateMsg = fnLoad();"
  local fnFile = load(szFileContent, nil, "bt", tbTmpEnv)
  if fnFile then
    fnFile()
  end
  local szLocalDataRolePath = "%sUser/Role/%s.json"
  if tbTmpEnv.tbPrivateMsg then
    for nId, tbData in pairs(tbTmpEnv.tbPrivateMsg) do
      local tb = {}
      tb.PrivateMsg = tbData
      local szFilePath = string.format(szLocalDataRolePath, g_szUserPath, nId)
      Lib:WriteFile_W(szFilePath, "w+", Lib:EncodeJson(tb))
    end
  end
  CS.System.IO.File.Delete(szPath)
end

function Client:InitUserInfo()
  local szFileContent, nLen = Lib:ReadFile_W(g_szUserPath .. "User/UserInfo.lua", "r")
  if 0 == nLen then
    szFileContent = "return {};"
  end
  self.tbTmpEnv = {}
  szFileContent = "local fnLoad = function () " .. szFileContent .. " end tbUserInfo = fnLoad();"
  local fnFile = load(szFileContent, nil, "bt", self.tbTmpEnv)
  if fnFile then
    fnFile()
  end
  self.tbUserInfo = self.tbTmpEnv.tbUserInfo or {}
end

function Client:TransAfkSetting()
  local szKey = "AutoFight"
  if not self.tbUserInfo then
    self:InitUserInfo()
  end
  local szLocalDataRolePath = "%sUser/Role/%s.json"
  for nId, tbInfo in pairs(self.tbUserInfo) do
    local tbValue = tbInfo[szKey]
    if tbValue then
      local nPlayerId = math.floor(nId / 10000)
      local nFaction = nId % 10000 / 100
      local nFactionSect = nId % 10
      local szFilePath = string.format(szLocalDataRolePath, g_szUserPath, nPlayerId)
      local szFileContent, nLen = Lib:ReadFile_W(szFilePath, "r")
      if 0 == nLen then
        szFileContent = "{}"
      end
      local tbRoleData = Lib:DecodeJson(szFileContent)
      tbRoleData[szKey] = tbRoleData[szKey] or {}
      tbRoleData[szKey][tostring(nFaction)] = tbRoleData[szKey][tostring(nFaction)] or {}
      tbRoleData[szKey][tostring(nFaction)][tostring(nFactionSect)] = tbValue
      local bResult = Lib:WriteFile_W(szFilePath, "w+", Lib:EncodeJson(tbRoleData))
      if not bResult then
        LogErr("LocalData:Flush ERR ?? file is nil !!" .. szFilePath)
      end
      tbInfo[szKey] = nil
    end
  end
  Client:SaveUserInfo()
end

function Client:TransLocalFlagsUserInfo(tbKeys)
  if not self.tbUserInfo then
    self:InitUserInfo()
  end
  local szParentKey = "LocalFlags"
  local szLocalDataRolePath = "%sUser/Role/%s.json"
  for nId, tbInfo in pairs(self.tbUserInfo) do
    local tbValue = tbInfo[szParentKey]
    if tbValue then
      local szFilePath = string.format(szLocalDataRolePath, g_szUserPath, nId)
      local szFileContent, nLen = Lib:ReadFile_W(szFilePath, "r")
      if 0 == nLen then
        szFileContent = "{}"
      end
      local tbRoleData = Lib:DecodeJson(szFileContent)
      local bWrite = false
      for szKey, _ in pairs(tbKeys) do
        if nil ~= tbValue[szKey] then
          tbRoleData[szKey] = tbValue[szKey]
          tbInfo[szParentKey][szKey] = nil
          bWrite = true
        end
      end
      if bWrite then
        local bResult = Lib:WriteFile_W(szFilePath, "w+", Lib:EncodeJson(tbRoleData))
        if not bResult then
          LogErr("LocalData:Flush ERR ?? file is nil !!" .. szFilePath)
        end
      end
    end
  end
  Client:SaveUserInfo()
end
