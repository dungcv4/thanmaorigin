AvenueOfStars.bOpen = true

function AvenueOfStars:IsOpenDate()
  if not self.bOpen then
    return false
  end
  local nStartTime = Lib:ParseDateTime(self.ACT_DATE[1])
  local nEndTime = Lib:ParseDateTime(self.ACT_DATE[2])
  local nGameTime = GetTime()
  if nStartTime <= nGameTime and nEndTime > nGameTime then
    return true
  end
  return false
end

function AvenueOfStars:IsEnrollDate(nGameTime)
  nGameTime = nGameTime or GetTime()
  local nStartTime = Lib:ParseDateTime(self.ENROLL_DATE[1])
  local nEndTime = Lib:ParseDateTime(self.ENROLL_DATE[2])
  if nGameTime >= nStartTime and nGameTime < nEndTime then
    return true
  end
  return false
end

function AvenueOfStars:IsVoteDate(nGameTime)
  nGameTime = nGameTime or GetTime()
  for nPhase, tbDate in ipairs(self.VOTE_DATE) do
    local nStartTime = Lib:ParseDateTime(tbDate[1])
    local nEndTime = Lib:ParseDateTime(tbDate[2])
    if nGameTime >= nStartTime and nGameTime < nEndTime then
      return true, nPhase
    end
  end
  return false
end

function AvenueOfStars:IsSettleDate(nGameTime)
  nGameTime = nGameTime or GetTime()
  for nPhase, tbDate in ipairs(self.SETTLE_DATE) do
    local nStartTime = Lib:ParseDateTime(tbDate[1])
    local nEndTime = Lib:ParseDateTime(tbDate[2])
    if nGameTime >= nStartTime and nGameTime < nEndTime then
      return true, nPhase
    end
  end
  return false
end

function AvenueOfStars:IsUseVoteLevelSettleDate(nGameTime, nPhase)
  nGameTime = nGameTime or GetTime()
  local tbDate = self.USE_VOTE_LEVEL_SETTLE_DATE[nPhase]
  if not tbDate then
    return false
  end
  local nStartTime = Lib:ParseDateTime(tbDate[1])
  local nEndTime = Lib:ParseDateTime(tbDate[2])
  if nGameTime < nStartTime or nGameTime >= nEndTime then
    return false
  end
  return true
end
