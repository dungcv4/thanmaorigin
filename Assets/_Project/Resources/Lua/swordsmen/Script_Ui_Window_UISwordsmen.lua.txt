function tbWnd:OnCreate()
  self.bIsVoice = true
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbChatMsgScrollView = Ui:NewSuperListView(self, tbControls.ChatScrollItem, self.SetChatItem2Scroll)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
end

function tbWnd:OnOpen(nOpen)
  nOpen = nOpen or 1
  self.nCurIndex = nOpen or 1
  self.tbTabControl:OnOpen(nOpen)
  self.pPanel:SetActive(tbControls.btnNewMsg, false)
  self:UpdateNewMsg()
  self.tbMoneyPanel:OnOpen()
  self.tbMoneyPanel:SetMoneyValue()
  self:SwitchInputMode()
  self.pPanel:SetActive(tbControls.wndTips, false)
end

function tbWnd:CheckCanSwitch(nIndex)
  local szCheckValue = self.tbPanelCheckValue[nIndex]
  if not TeamMgr:IsCaptain(me.dwID) then
    me.CenterMsg(i18n.Get(Marry.Cfg.Msg.CAPTAIN_LIMIT))
    self.tbTabControl:SelectPanel(self.nCurIndex)
    return false
  end
  if szCheckValue and 0 == Marry[szCheckValue] then
    local szMsg = self.tbNoOpenMsg[nIndex]()
    me.CenterMsg(szMsg)
    self.tbTabControl:SelectPanel(self.nCurIndex)
    return false
  end
  self.nCurIndex = nIndex
  return true
end

function tbWnd:SwitchInputMode()
  if self.nLeftTime and self.nLeftTime > 0 and not self.bIsVoice then
    me.CenterMsg(i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
  end
  self.bIsVoice = not self.bIsVoice
  self.pPanel:SetActive(tbControls.InputVoice, self.bIsVoice)
  self.pPanel:SetActive(tbControls.btnKeyboard, self.bIsVoice)
  self.pPanel:SetActive(tbControls.InputChat, not self.bIsVoice)
  self.pPanel:SetActive(tbControls.btnVoice, not self.bIsVoice)
  if not self.bIsVoice then
    self:UpdateCdExpression()
  end
end

function tbWnd:UpdateCdExpression()
  if self.nCdTimer then
    Timer:Close(self.nCdTimer)
    self.nCdTimer = nil
  end
  local nCd = ChatMgr:GetCd(ChatMgr.ChannelType.Team, nil)
  if nCd <= 0 then
    self.nLeftTime = 0
    self.pPanel:Input_SetEnable(tbControls.InputChat, true)
    self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputPlaceHolder, ChatMgr.nMaxMsgLengh))
    return
  end
  self.nLeftTime = nCd
  self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
  self.pPanel:Input_SetEnable(tbControls.InputChat, false)
  self.nCdTimer = Timer:Register(Env.GAME_FPS, function()
    self.nLeftTime = self.nLeftTime - 1
    if self.nCdTimer and self.nLeftTime > 0 then
      self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
      self.pPanel:Input_SetEnable(tbControls.InputChat, false)
      return true
    else
      self.pPanel:Input_SetEnable(tbControls.InputChat, true)
      self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputPlaceHolder, ChatMgr.nMaxMsgLengh))
      self.nCdTimer = nil
      self.nLeftTime = nil
      return false
    end
  end)
end

function tbWnd:PlusClickFunc()
  if self.bIsVoice then
    self:SwitchInputMode()
  end
  if not Ui:WindowVisible("UIChatPlus") then
    Ui:OpenWindow("UIChatPlus", function(...)
      self:DealOuterInput(...)
    end)
  end
end

function tbWnd:DealOuterInput(szTab, szMsg)
  local szContent = self.pPanel:Input_GetText(tbControls.InputChat) .. szMsg
  if self.nLeftTime and self.nLeftTime > 0 then
    me.CenterMsg(i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
    return
  end
  if not self.pPanel:IsActive(tbControls.InputChat) then
    return
  end
  self.pPanel:Input_SetText(tbControls.InputChat, szContent)
end

function tbWnd:SendClickFunc()
  local szMsg = self.pPanel:Input_GetText(tbControls.InputChat)
  local bSucceed = false
  bSucceed = ChatMgr:SendMsg(ChatMgr.ChannelType.Team, nil, szMsg)
  if bSucceed then
    self.pPanel:Input_SetText(tbControls.InputChat, "")
    self:UpdateCdExpression()
    self.tbChatMsgScrollView:MovePanelToLastItem()
    local _, bEmojiExist = ChatMgr:FilterPlayerMsg(ChatMgr.ChannelType.Team, szMsg)
    if bEmojiExist then
      EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_EMOJI)
    end
  end
end

function tbWnd:_NewChatMsg(nChannelId)
  if not self:Visible() then
    return
  end
  if nChannelId ~= ChatMgr.ChannelType.Team then
    return
  end
  self:UpdateNewMsg()
  ChatMgr:SetUnReadChannelMsg(ChatMgr.ChannelType.Team)
end

function tbWnd:UpdateNewMsg()
  local tbItemData = ChatMgr:GetChannelChatData(ChatMgr.ChannelType.Team) or {}
  for i = #tbItemData, 1, -1 do
    if tbItemData[i].nChannelType == ChatMgr.ChannelType.System then
      table.remove(tbItemData, i)
    end
  end
  self.tbChatMsgScrollView:SetAllElement(tbItemData)
  self.tbChatMsgScrollView:MovePanelToLastItem()
end

function tbWnd:OnChatDataCahnged()
  self:UpdateNewMsg()
end

function tbWnd:SetChatItem2Scroll(tbElement, tbMsg, nIndex)
  local pPanel = tbElement.pPanel
  local bSelf = tbMsg.nSenderId == me.dwID
  local bHasVoice = tbMsg.szMsgVoiceFileId ~= nil
  tbMsg.nPortrait = bSelf and me.nPortrait or tbMsg.nPortrait
  tbMsg.nPortraitFrame = bSelf and PlayerPortraitFrame:GetMeCurPortraitFrameId(me) or tbMsg.nPortraitFrame
  tbMsg.nLevel = bSelf and me.nTotalLevel or tbMsg.nLevel
  pPanel:SetActive("txtChatTime", false)
  pPanel:SetActive("SelfMsg", bSelf)
  pPanel:SetActive("OtherMsg", not bSelf)
  if bSelf then
    pPanel:SetActive("SelfMsg", true)
    pPanel:SetActive("SelfMsg/TxtNode", not bHasVoice)
    pPanel:SetActive("SelfMsg/VoiceNode", bHasVoice)
  else
    pPanel:SetActive("OtherMsg/TxtNode", not bHasVoice)
    pPanel:SetActive("OtherMsg/VoiceNode", bHasVoice)
  end
  
  local function funcSetEmoji(szKey)
    local txtEmoji = pPanel:GetEmojiText(szKey)
    local tbLinkInfo = tbMsg and tbMsg.tbLinkInfo or {}
    txtEmoji.onClickLink = nil
    txtEmoji.raycastTarget = false
    if not tbLinkInfo.nLinkType or not ChatMgr.tbLinkClickFns[tbLinkInfo.nLinkType] then
      txtEmoji.onClickLink = nil
      txtEmoji.raycastTarget = false
    else
      txtEmoji.raycastTarget = true
      
      function txtEmoji.onClickLink(param)
        tbMsg.bClickLink = true
        if tbLinkInfo.nLinkType and ChatMgr.tbLinkClickFns[tbLinkInfo.nLinkType] then
          ChatMgr.tbLinkClickFns[tbLinkInfo.nLinkType](tbLinkInfo, tonumber(param), tbMsg)
        end
      end
      
      local bSuccess, szShowMsg = Lib:CallBack({
        ChatMgr.DealMsgWithLinkColor,
        ChatMgr,
        tbMsg.szMsg,
        tbLinkInfo,
        tbMsg,
        true
      })
      if bSuccess then
        tbMsg.szShowMsg = szShowMsg
      else
        tbMsg.szShowMsg = nil
      end
    end
    txtEmoji.text = tbMsg.szShowMsg or tbMsg.szMsg
  end
  
  local szPath = bSelf and "SelfMsg" or "OtherMsg"
  local szMsg = ""
  local szSendName = tbMsg.szSenderName
  local szRemarkName = FriendShip:GetPlayerRemarkName(tbMsg.nSenderId)
  if "" ~= szRemarkName then
    szSendName = szRemarkName
  end
  szMsg = szSendName
  local szBg = ChatMgr.ChannelBg[tbMsg.nChannelType] or ChatMgr.ChannelBg[ChatMgr.ChannelType.Team]
  local szColor = ChatMgr.ChannelTagColor[szBg]
  szBg = i18n.Get(szBg)
  if szColor then
    szBg = string.format("<color=%s>%s</color>", szColor, szBg)
  end
  local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, szPath .. "/HeadFrame/PortraitObj")
  tbPortraitObjPanel:SetRoleIcon(tbMsg.nPortrait)
  tbPortraitObjPanel:SetPortraitFrame(tbMsg.nPortraitFrame)
  tbPortraitObjPanel:SetFactionIcon(tbMsg.nFaction)
  tbPortraitObjPanel:SetLevel(tbMsg.nLevel)
  pPanel:Label_SetText(szPath .. "/InfoNode/txtName", szMsg)
  pPanel:Label_SetText(szPath .. "/InfoNode/txtChannel", szBg)
  local hf_tf = tbElement.pPanel:GetObject(szPath .. "/HeadFrame"):GetComponent(Ui.RectTransform)
  local in_tf = tbElement.pPanel:GetObject(szPath .. "/InfoNode"):GetComponent(Ui.RectTransform)
  if not bHasVoice then
    funcSetEmoji(szPath .. "/TxtNode/TxtBG/txtChatMsg")
    local txtNodeKey = szPath .. "/TxtNode"
    self:AdjustItemHeight(tbElement, txtNodeKey, hf_tf.sizeDelta.y, in_tf.sizeDelta.y + 2)
  else
    if not tbMsg.szMsg or "" == tbMsg.szMsg then
      pPanel:SetActive(szPath .. "/VoiceNode/VoiceBG/txtChatMsg", false)
    else
      pPanel:Label_SetText(szPath .. "/VoiceNode/VoiceBG/txtChatMsg", tbMsg.szMsg)
    end
    self:AdjustItemHeight(tbElement, szPath .. "/VoiceNode", hf_tf.sizeDelta.y, in_tf.sizeDelta.y + 2)
    pPanel:Label_SetText(szPath .. "/VoiceNode/VoiceBG/VoiceMsg/Text", math.ceil(tbMsg.uVoiceTime / 1000) .. "\"")
    Ui:Button_BindEvent(pPanel, szPath .. "/VoiceNode/VoiceBG/VoiceMsg", function()
      ChatMgr:PlayVoice(tbMsg.szMsgVoiceFileId)
    end, self)
  end
end

function tbWnd:AdjustItemHeight(tbElement, szKey, minHeight, offset)
  minHeight = minHeight or 0
  offset = offset or 0
  local childObj = tbElement.pPanel:GetObject(szKey)
  local childCsfComp = childObj:GetComponent(Ui.ContentSizeFitter)
  childCsfComp:SetLayoutVertical()
  tbElement.pPanel:ContentSizeFitter_Refresh(szKey)
  local childTf = childObj:GetComponent(Ui.RectTransform)
  local height = math.max(minHeight, childTf.sizeDelta.y + offset)
  local szKeyTemp = szKey
  while nil ~= szKeyTemp do
    szKeyTemp = self:SplitLast(szKeyTemp, "/")
    if szKeyTemp then
      local obj = tbElement.pPanel:GetObject(szKeyTemp)
      local tf = obj:GetComponent(Ui.RectTransform)
      tf.sizeDelta = Ui.Vector2(tf.sizeDelta.x, height)
    end
  end
  local rootObj = self.pPanel:GetObject(tbElement.szKey)
  local rootTf = rootObj:GetComponent(Ui.RectTransform)
  rootTf.sizeDelta = Ui.Vector2(rootTf.sizeDelta.x, height)
end

function tbWnd:SplitLast(szPath, sep)
  local szReverse = string.reverse(szPath)
  local i = string.find(szReverse, sep)
  if not i then
    return nil, nil
  end
  local l = string.len(szReverse)
  local m = l - i
  return string.sub(szPath, 1, m), string.sub(szPath, m + 2, l)
end

end

function tbWnd:OnClose()

end
