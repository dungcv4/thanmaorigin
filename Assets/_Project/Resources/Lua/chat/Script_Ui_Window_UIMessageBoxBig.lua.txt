local tbWnd = Ui:CreateWindow("UIMessageBoxBig")
tbWnd.tbControls = {
  txtTitle = "imgBG/imgTitle/txtTitle",
  txtMsg = "imgBG/imgMsgBG/txtMsg",
  btnOk = "imgBG/btnOK",
  btnCancel = "imgBG/btnCancel",
  btnCenter = "imgBG/btnCenter",
  CheckTips = "imgBG/CheckTips",
  togCheckTips = "imgBG/CheckTips/ToggleCheck"
}
tbWnd.tbOnClick = {
  btnOk = function(self)
    self:onClickOK()
  end,
  btnCenter = function(self)
    self:onClickOK()
  end,
  btnCancel = function(self)
    if self.tbCancelCallback then
      self.tbCancelCallback()
    end
    self:Close()
  end
}

function tbWnd:OnClose()
  if self.nTimerId then
    Timer:Close(self.nTimerId)
    self.nTimerId = nil
  end
  self.bSkipConnect = false
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_MESSAGEBOX_CLOSE)
end

function tbWnd:OnOpen(szText, tbProc, tbBtnTxt, szNotTipsType, nTime, bSkipConnect)
  local szInfo = string.format(szText, tonumber(nTime) or 0)
  self.pPanel:Label_SetText(self.txtMsg, szText)
  if nTime and nTime > 0 then
    self.nEndTime = GetTime() + nTime
    self.nTimerId = Timer:Register(Env.GAME_FPS, function()
      if self.nEndTime - GetTime() <= 0 then
        self.nTimerId = nil
        self:Close()
        return
      end
      local nCurTime = math.max(self.nEndTime - GetTime(), 0)
      self.pPanel:Label_SetText(self.txtMsg, string.format("%s %s", szText, nCurTime))
      return true
    end)
  end
  if tbProc then
    self.tbOKCallback = tbProc[1]
    self.tbCancelCallback = tbProc[2]
  else
    self.tbOKCallback = nil
    self.tbCancelCallback = nil
  end
  tbBtnTxt = tbBtnTxt or {
    i18n.szOk,
    i18n.szCancel
  }
  self.szNotTipsType = szNotTipsType
  self.pPanel:Toggle_SetChecked(self.togCheckTips, false)
  self.pPanel:SetActive(self.CheckTips, self.szNotTipsType and true or false)
  self.pPanel:Button_SetText(self.btnOk, tbBtnTxt[1] or i18n.szOk, "Text")
  self.pPanel:Button_SetText(self.btnCancel, tbBtnTxt[2] or i18n.szCancel, "Text")
  self.pPanel:Button_SetText(self.btnCenter, tbBtnTxt[1] or i18n.szOk, "Text")
  local isSingle = tbProc and 1 == #tbProc or false
  if not tbProc then
    isSingle = true
  end
  self.bSkipConnect = bSkipConnect
  self.pPanel:SetActive(self.btnCenter, isSingle)
  self.pPanel:SetActive(self.btnOk, not isSingle)
  self.pPanel:SetActive(self.btnCancel, not isSingle)
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_MESSAGEBOX_OPEN)
end

function tbWnd:onClickOK()
  local bRemainOpen, _
  if self.tbOKCallback then
    _, bRemainOpen = self.tbOKCallback()
  end
  if not bRemainOpen then
    self:Close()
  end
  if self.szNotTipsType and self.pPanel:Toggle_GetChecked(self.togCheckTips) == true then
    Ui:SetNotShowTips(self.szNotTipsType)
  end
end
