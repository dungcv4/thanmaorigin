local tbWnd = Ui:CreateWindow("UIEmoji")
tbWnd.tbControls = {
  emojiTemplate = "imgBG/Panel/Viewport/Grid/Item",
  btnClose = "imgBG/closeBG"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MAP_LEAVE,
      self.Close
    }
  }
end

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, self.emojiTemplate, function(self, nIndex, tbElement)
    local uiChat = Ui:GetWindow("UIChat")
    if nil ~= uiChat then
      uiChat:AddEmoji(tbElement.id)
    end
  end, true, true)
  local nIndex = 1
  local emojiConfig = CS.EmojiText:GetConfig()
  local iter = emojiConfig:GetEnumerator()
  while iter:MoveNext() do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      tbElement.id = iter.Current.Key
      tbElement.pPanel:Label_SetText("Emoji", tbElement.id)
    end)
    nIndex = nIndex + 1
  end
end
