local tbWnd = Ui:CreateWindow("UIMessageBoxFudai")
tbWnd.tbControls = {
  txtTitle = "imgBG/imgTitle/txtTitle",
  txtMsg = "imgBG/imgMsgBG/txtMsg",
  btnClose = "imgBG/btnClose",
  btnKeepOpen = "imgBG/btnKeepOpen",
  btnDredgeMonthCard = "imgBG/btnDredgeMonthCard"
}
tbWnd.tbOnClick = {
  btnKeepOpen = function(self)
    self:OnClickOK()
  end,
  btnDredgeMonthCard = function(self)
    self:OnClickCancel()
  end,
  btnClose = function(self)
    self:CloseByDefault()
  end
}

function tbWnd:OnClose()
  self.nItemId = nil
  self.nCurValue = nil
  self.nCurTime = nil
end

function tbWnd:OnOpen(nItemId, nCount, nCurTime, nDifferenceValue)
  self.nItemId = nItemId
  self.nCurValue = nCount
  self.nCurTime = nCurTime
  self.nDifferenceValue = nDifferenceValue
  local szText = i18n.EasyGet("\228\187\138\230\151\165\233\171\152\231\186\167\229\165\150\229\138\177\230\172\161\230\149\176\229\183\178\231\148\168\229\174\140\239\188\140\231\187\167\231\187\173\228\189\191\231\148\168\229\176\134\232\142\183\229\190\151\228\191\157\229\186\149\229\165\150\229\138\177\239\188\140\230\152\175\229\144\166\229\188\128\229\144\175\239\188\159\239\188\136\229\188\128\233\128\154\230\156\136\229\141\161\229\144\142\230\175\143\230\151\165\233\171\152\231\186\167\229\165\150\229\138\177\230\172\161\230\149\176+{0}\239\188\137", self.nDifferenceValue)
  self.pPanel:Label_SetText(self.txtMsg, szText)
  local szBtnOK = i18n.EasyGet("\231\187\167\231\187\173\229\188\128\229\144\175")
  local szbtnCancel = i18n.EasyGet("\229\188\128\233\128\154\230\156\136\229\141\161")
  self.pPanel:Button_SetText(self.btnKeepOpen, szBtnOK or i18n.szOk, "Text")
  self.pPanel:Button_SetText(self.btnDredgeMonthCard, szbtnCancel or i18n.szCancel, "Text")
end

function tbWnd:OnClickOK()
  Item:ClientUseManyItem(self.nItemId, self.nCurValue)
  Ui.PlayerPrefs.SetInt(string.format("NoticeGoldFudai_%s", me.dwID), self.nCurTime)
  self:Close()
end

function tbWnd:OnClickCancel()
  Ui:OpenWindow("UIWelfare", "PANEL_TYPE_WEEK_MONTH_CARD")
  self:Close()
end

function tbWnd:CloseByDefault()
  self:Close()
end
