local tbWnd = Ui:CreateWindow("UIMessageBoxItem")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MAP_ENTER,
      self.OnEnterMap
    }
  }
end

tbWnd.tbControls = {
  txtTips1 = "imgBG/Content/txtTips1",
  txtTips2 = "imgBG/Content/txtTips2",
  txtTips3 = "imgBG/Content/txtTips3",
  ItemGrid = "imgBG/Content/UIItemGrid",
  btnOK = "imgBG/Content/btnOK",
  btnCancel = "imgBG/Content/btnCancel",
  btnGo = "imgBG/Content/btnGo"
}
local tbControls = tbWnd.tbControls
tbWnd.tbOnClick = {
  btnOK = function(self)
    self:OnClickOK()
  end,
  btnCancel = function(self)
    self:OnClickCancel()
  end,
  btnGo = function(self)
    self:OnClickGo()
  end
}

function tbWnd:OnCreate()
  self.tbItemGrid = Ui:NewExistItemGrid(self, tbControls.ItemGrid)
end

function tbWnd:OnClose()
  self.tbOKCallback = nil
  self.tbCancelCallback = nil
end

function tbWnd:OnOpen(nItemId, nCount, szText1, szText2, szText3, tbProc, szBtnGoText, tbGoProc)
  self.tbItemGrid:SetItemByTemplate(nItemId, 1)
  self.tbItemGrid:SetClickEnable(true)
  self.tbItemGrid:LabelSuffix_SetText(nCount .. "/1")
  self.tbItemGrid:LabelSuffix_SetActive(true)
  if szText1 then
    self.pPanel:Label_SetText(tbControls.txtTips1, szText1)
  end
  if szText2 then
    self.pPanel:Label_SetText(tbControls.txtTips2, szText2)
  end
  if szText3 then
    self.pPanel:Label_SetText(tbControls.txtTips3, szText3)
  end
  if tbProc then
    self.tbOKCallback = tbProc[1]
    self.tbCancelCallback = tbProc[2]
  else
    self.tbOKCallback = nil
    self.tbCancelCallback = nil
  end
  if szBtnGoText and tbGoProc then
    self.pPanel:SetActive(tbControls.btnGo, true)
    self.tbGoCallBack = tbGoProc
  else
    self.pPanel:SetActive(tbControls.btnGo, false)
    self.tbGoCallBack = nil
  end
end

function tbWnd:OnClickOK()
  if self.tbOKCallback then
    Lib:CallBack(self.tbOKCallback)
  end
  self:Close()
end

function tbWnd:OnClickCancel()
  if self.tbCancelCallback then
    Lib:CallBack(self.tbCancelCallback)
  end
  self:Close()
end

function tbWnd:OnEnterMap()
  self:Close()
end

function tbWnd:OnClickGo()
  if self.tbGoCallBack then
    Lib:CallBack(self.tbGoCallBack)
  end
  self:Close()
end
