local tbWnd = Ui:CreateWindow("UISelectChatTitlePrefix")
local tbControls = {
  ChatTitlePrefixTemplate = "Scroll View/Viewport/Content/Template",
  togShowInHud = "Template/Toggle",
  ChatTitlePrefixRightDesc = "ChatTitleRightDesc",
  objRightStaticChatTitlePrefix = "ChatTitleRightDesc/ChatTitle",
  txtRightNormalChatTitlePrefix = "ChatTitleRightDesc/txtChatTitle",
  txtDesc = "ChatTitleRightDesc/Scroll View/Viewport/Content/txtDesc",
  txtGainWay = "ChatTitleRightDesc/Scroll View/Viewport/Content/txtGetWay",
  txtTime = "ChatTitleRightDesc/Scroll View/Viewport/Content/txtTime",
  btnOk = "ChatTitleRightDesc/btnOk",
  txtOk = "ChatTitleRightDesc/btnOk/Text",
  objToggle = "Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbSmallType = {Normal = 1, Static = 2}
tbWnd.tbOnClick = {
  btnOk = function(self)
    local nCurChatTitlePrefixId = ChatMgr:GetCurChatTitlePrefixTempId(me)
    if self.nCurIndex and self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex] then
      if nCurChatTitlePrefixId == self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex].nTemplateId then
        self:Unuse()
      else
        self:Use()
      end
    end
  end
}
tbWnd.tbOnToggleClick = {
  togShowInHud = function(self, bIsOn)
    local nCurChatTitlePrefixId = ChatMgr:GetCurChatTitlePrefixTempId(me)
    if self.nCurIndex and self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex] and nCurChatTitlePrefixId == self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex].nTemplateId then
      self:Use()
    end
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.ChatTitlePrefixTemplate, function(self, nIndex, tbElement)
    self:OnSelectChatTitlePrefix(nIndex)
  end, true, true)
end

function tbWnd:OnOpen()
  self:Update()
end

function tbWnd:Update()
  local bShowLocked = true
  local tbWnd = Ui:GetWindow("UISelectPortrait")
  if tbWnd then
    bShowLocked = tbWnd:GetLockedStatus()
    self.nSmallType = tbWnd:GetCurSmallType()
  end
  self:InitAllChatTitlePrefix(bShowLocked)
  if not self.nCurIndex or 0 == self.nCurIndex then
    self:AutoSelectChatTitlePrefix()
  else
    self:OnSelectChatTitlePrefix(self.nCurIndex)
  end
end

function tbWnd:OnClose()
  if self.nCurIndex and 0 ~= self.nCurIndex then
    local pCurPanel = self.tbScrollView:GetElement(self.nCurIndex).pPanel
    pCurPanel:SetActive("imgSelected", false)
  end
  self.nCurIndex = nil
  self.nSmallType = nil
  self.tbIndex2ChatTitlePrefixTempId = nil
  self.tbRedPointIndex = {}
  self.tbScrollView:Clear()
end

function tbWnd:OnSelectChatTitlePrefix(nIndex)
  if not nIndex then
    Log("ChatTitlePrefix nIndex is Nil")
    self.pPanel:SetActive(self.ChatTitlePrefixRightDesc, false)
    return
  end
  self.pPanel:SetActive(self.ChatTitlePrefixRightDesc, true)
  local pPanel = self.tbScrollView:GetElement(nIndex).pPanel
  local nTemplateId = self.tbIndex2ChatTitlePrefixTempId[nIndex].nTemplateId
  local bLocked = self.tbIndex2ChatTitlePrefixTempId[nIndex].bLocked
  if self.nCurIndex and 0 ~= self.nCurIndex then
    local pCurPanel = self.tbScrollView:GetElement(self.nCurIndex).pPanel
    pCurPanel:SetActive("imgSelected", false)
  end
  pPanel:SetActive("imgSelected", true)
  self.nCurIndex = nIndex
  if not bLocked then
    pPanel:SetActive("imgRedPoint", false)
    ChatMgr:RemoveChatTitlePrefixRedPoint(nTemplateId)
  end
  if ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Normal" then
    self.pPanel:SetActive(self.objRightStaticChatTitlePrefix, false)
    self.pPanel:SetActive(self.txtRightNormalChatTitlePrefix, true)
    local tbCfg = ChatMgr:GetNormalChatTitlePrefixConfig(nTemplateId)
    local szName = ChatMgr:GetNormalChatTitlePrefixName(tbCfg.tbPrefixType.szType, table.unpack(tbCfg.tbPrefixType.tbParam))
    self.pPanel:Label_SetText(self.txtRightNormalChatTitlePrefix, i18n.EasyGet(i18n.Parse(tbCfg.szName)))
    self.pPanel:Label_SetText(self.txtDesc, i18n.EasyGet("\229\137\141\231\188\128\230\143\143\232\191\176\239\188\154{0}", i18n.Parse(tbCfg.szDesc)))
    self.pPanel:Label_SetText(self.txtGainWay, i18n.EasyGet("\232\142\183\229\190\151\230\150\185\229\188\143\239\188\154{0}", i18n.Parse(tbCfg.szGainWay)))
    self.pPanel:SetActive(self.txtTime, false)
    self.pPanel:SetActive(self.objToggle, false)
    self.pPanel:SetActive(self.btnOk, false)
  elseif ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Static" then
    self.pPanel:SetActive(self.objRightStaticChatTitlePrefix, true)
    self.pPanel:SetActive(self.txtRightNormalChatTitlePrefix, false)
    local tbCfg = ChatMgr:GetStaticChatTitlePrefixConfig(nTemplateId)
    local obj = self.pPanel:GetObject(self.objRightStaticChatTitlePrefix)
    if obj then
      local pChatTitle = obj:GetComponent(Ui.UIChatTitle)
      if pChatTitle then
        pChatTitle:SetChatTitlePrefix(tbCfg.nTemplateId)
      end
    end
    self.pPanel:Label_SetText(self.txtDesc, i18n.EasyGet("\229\137\141\231\188\128\230\143\143\232\191\176\239\188\154{0}", i18n.Parse(tbCfg.szDesc)))
    self.pPanel:Label_SetText(self.txtGainWay, i18n.EasyGet("\232\142\183\229\190\151\230\150\185\229\188\143\239\188\154{0}", i18n.Parse(tbCfg.szGainWay)))
    self.pPanel:SetActive(self.txtTime, not bLocked)
    self.pPanel:SetActive(self.objToggle, not bLocked)
    self.pPanel:SetActive(self.btnOk, true)
    local bShowInHud = Lib:IIF(1 == ChatMgr:GetCurChatTitlePrefixShowInHudState(me), true, false)
    self.pPanel:Toggle_SetChecked(self.togShowInHud, bShowInHud)
    if not bLocked then
      local nLeftTime = ChatMgr:GetChatTitlePrefixLeftTime(nTemplateId)
      self.pPanel:Label_SetText(self.txtTime, i18n.EasyGet("\229\137\141\231\188\128\230\151\182\230\149\136\239\188\154{0}", Lib:TimeDesc5(nLeftTime)))
      local nCurChatTitlePrefixId = ChatMgr:GetCurChatTitlePrefixTempId(me)
      if self.nCurIndex and self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex] then
        if nCurChatTitlePrefixId == self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex].nTemplateId then
          self.pPanel:Label_SetText(self.txtOk, i18n.EasyGet("\229\141\184\228\184\139"))
        else
          self.pPanel:Label_SetText(self.txtOk, i18n.EasyGet("\228\189\191\231\148\168"))
        end
      end
    else
      self.pPanel:Label_SetText(self.txtOk, i18n.EasyGet("\230\156\170\230\139\165\230\156\137"))
    end
  end
end

function tbWnd:InitAllChatTitlePrefix(bShowLocked)
  local tbResult = {}
  local tbLockedResult = {}
  local tbChatTitlePrefixConfig
  if ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Normal" then
    tbChatTitlePrefixConfig = ChatMgr:GetAllNormalChatTitlePrefixConfig()
    for nId, tbInfo in pairs(tbChatTitlePrefixConfig) do
      if ChatMgr:CheckHaveChatTitlePrefix(me, nId) then
        table.insert(tbResult, {
          nTemplateId = nId,
          bLocked = false,
          nSortOrder = tbInfo.nSortOrder
        })
      elseif bShowLocked then
        table.insert(tbLockedResult, {
          nTemplateId = nId,
          bLocked = true,
          nSortOrder = tbInfo.nSortOrder
        })
      end
    end
  elseif ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Static" then
    tbChatTitlePrefixConfig = ChatMgr:GetAllStaticChatTitlePrefixConfig()
    for nId, tbInfo in pairs(tbChatTitlePrefixConfig) do
      if true == tbInfo.bShow then
        if ChatMgr:CheckHaveChatTitlePrefix(me, nId) then
          table.insert(tbResult, {
            nTemplateId = nId,
            bLocked = false,
            nSortOrder = tbInfo.nSortOrder
          })
        elseif bShowLocked then
          table.insert(tbLockedResult, {
            nTemplateId = nId,
            bLocked = true,
            nSortOrder = tbInfo.nSortOrder
          })
        end
      end
    end
  end
  
  local function sortfun(a, b)
    return a.nSortOrder < b.nSortOrder
  end
  
  table.sort(tbResult, sortfun)
  if bShowLocked then
    table.sort(tbLockedResult, sortfun)
    for i = 1, #tbLockedResult do
      table.insert(tbResult, tbLockedResult[i])
    end
  end
  self.tbIndex2ChatTitlePrefixTempId = tbResult
  self:_SetElement(tbResult)
end

function tbWnd:_SetElement(tbShowList)
  self.tbScrollView:Clear()
  for index, tbInfo in ipairs(tbShowList) do
    self.tbScrollView:SetElement(index, function(tbElement)
      local pPanel = tbElement.pPanel
      local bLocked = tbInfo.bLocked
      pPanel:SetActive("imgLocked", bLocked)
      if ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Normal" then
        local tbCfg = ChatMgr:GetNormalChatTitlePrefixConfig(tbInfo.nTemplateId)
        pPanel:SetActive("ChatTitle", false)
        pPanel:SetActive("txtChatTitle", true)
        local szName = ChatMgr:GetNormalChatTitlePrefixName(tbCfg.tbPrefixType.szType, table.unpack(tbCfg.tbPrefixType.tbParam))
        pPanel:Label_SetText("txtChatTitle", i18n.EasyGet(i18n.Parse(tbCfg.szName)))
        pPanel:SetActive("imgLocked/ChatTitleLocked", false)
      elseif ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Static" then
        local tbCfg = ChatMgr:GetStaticChatTitlePrefixConfig(tbInfo.nTemplateId)
        pPanel:SetActive("ChatTitle", true)
        pPanel:SetActive("txtChatTitle", false)
        local obj = pPanel:GetObject("ChatTitle")
        if obj then
          local pChatTitle = obj:GetComponent(Ui.UIChatTitle)
          if pChatTitle then
            pChatTitle:SetChatTitlePrefix(tbCfg.nTemplateId)
          end
        end
        pPanel:SetActive("imgLocked/ChatTitleLocked", true)
        pPanel:Sprite_SetSprite("imgLocked/ChatTitleLocked/imgTitleBg", tbCfg.szSpritePath)
      end
      pPanel:SetActive("imgSelected", false)
      pPanel:SetActive("imgRedPoint", false)
      if ChatMgr:GetChatTitlePrefixRedPointState(tbInfo.nTemplateId) then
        pPanel:SetActive("imgRedPoint", true)
      end
    end)
  end
  if self.nCurIndex and self.nCurIndex > #tbShowList then
    self.nCurIndex = nil
  end
end

function tbWnd:AutoSelectChatTitlePrefix()
  local nSelectIndex
  if ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Static" then
    local nCurChatTitlePrefixId = ChatMgr:GetCurChatTitlePrefixTempId(me)
    for k, v in ipairs(self.tbIndex2ChatTitlePrefixTempId) do
      if v.nTemplateId == nCurChatTitlePrefixId then
        nSelectIndex = k
      end
    end
  elseif ChatMgr:GetChatTitlePrefixTypeByTreeSmallType(self.nSmallType) == "Normal" then
    nSelectIndex = 1
  end
  self:OnSelectChatTitlePrefix(nSelectIndex)
end

function tbWnd:ReqChange()
end

function tbWnd:Use()
  local nShowInHud = Lib:IIF(self.pPanel:Toggle_GetChecked(self.togShowInHud), 1, 0)
  ChatMgr:ReqUseChatTitlePrefix(self.tbIndex2ChatTitlePrefixTempId[self.nCurIndex].nTemplateId, nShowInHud)
end

function tbWnd:Unuse()
  ChatMgr:ReqUnuseChatTitlePrefix()
end
