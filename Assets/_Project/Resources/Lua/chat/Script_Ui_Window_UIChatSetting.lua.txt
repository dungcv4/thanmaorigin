function tbWnd:CloseByDefault()
  ChatMgr:SaveSetting(self.tbChatSetting)
  self:Close()
end

tbWnd.tbOnClick = {
  btnClose = function(self)
    self:CloseByDefault()
  end
}
tbWnd.tbOnToggleClick = {
  CheckPublic = function(self, isOn)
    self:onClickToggleSetting(isOn, "CheckPublic")
  end,
  CheckNearby = function(self, isOn)
    self:onClickToggleSetting(isOn, "CheckNearby")
  end,
  CheckKin = function(self, isOn)
    self:onClickToggleSetting(isOn, "CheckKin")
  end,
  CheckTong = function(self, isOn)
    self:onClickToggleSetting(isOn, "CheckTong")
  end,
  CheckTeam = function(self, isOn)
    self:onClickToggleSetting(isOn, "CheckTeam")
  end,
  CheckFriend = function(self, isOn)
    self:onClickToggleSetting(isOn, "CheckFriend")

  CheckSystem = function(self, isOn)

  CheckZone = function(self, isOn)

  end,

    self:onClickToggleSetting(isOn, "CheckPublicVoice")
  end,

  end,
  CheckKinVoice = function(self, isOn)

    self:onClickToggleSetting(isOn, "CheckTextVoice")
  end
}

function tbWnd:OnCreate()
  self.tbChatSetting = ChatMgr:GetSetting()
end

function tbWnd:OnOpen()
  self:InitCheckState()
  self.pPanel:SetActive(self.TextToVoiceObj, Client.EnableVoiceToText)
end

function tbWnd:InitCheckState()
  for szCheckName, bCheck in pairs(self.tbChatSetting) do
    if self.tbOnToggleClick[szCheckName] then
      self.pPanel:Toggle_SetChecked(self[szCheckName], bCheck)
    end
  end
end

function tbWnd:onClickToggleSetting(isOn, szCheckName)
  self.tbChatSetting[szCheckName] = isOn
end
