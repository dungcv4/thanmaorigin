local tbWnd = Ui:CreateWindow("UIMessageBox_Xiakedao")
tbWnd.tbControls = {
  txtTitle = "imgBG/imgTitle/txtTitle",
  txtMsg = "imgBG/imgMsgBG/txtMsg",
  btnOk = "imgBG/btnOK",
  btnCancel = "imgBG/btnCancel",
  btnCenter = "imgBG/btnCenter",
  CheckTips = "imgBG/CheckTips",
  togCheckTips = "imgBG/CheckTips/ToggleCheck",
  togCheckTipsText = "imgBG/CheckTips/ToggleCheck/Label",
  txtCoinValue = "imgBG/txtCoinValue"
}

function tbWnd:OnCreate()
end

function tbWnd:CloseByDefault()
  XiaKeDao:OnXiaKeDaoClientLeaveMap()
  self:Close()
end

tbWnd.tbOnClick = {
  btnOk = function(self)
    self:onClickOK()
  end,
  btnCancel = function(self)
    self:CloseByDefault()
  end
}

function tbWnd:OnClose()
  if self.nTimerId then
    Timer:Close(self.nTimerId)
    self.nTimerId = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_MESSAGEBOX_CLOSE)
end

function tbWnd:OnOpen(szText, nTime, nNeedCoin)
  szText = i18n.Parse(szText)
  local szInfo = szText
  if nTime and nTime > 0 then
    szInfo = string.format(szText, tonumber(nTime) or 0)
  end
  self.pPanel:Label_SetText(self.txtMsg, szInfo)
  self.pPanel:Button_SetText(self.btnOk, i18n.EasyGet("\231\171\139\229\141\179\229\164\141\230\180\187"), "Text")
  self.pPanel:Button_SetText(self.btnCancel, i18n.EasyGet("\231\166\187\229\188\128({0})", nTime), "Text")
  self.nEndTime = GetTime() + nTime
  self.nTimerId = Timer:Register(Env.GAME_FPS, function()
    if self.nEndTime - GetTime() <= 0 then
      self.nTimerId = nil
      self:Close()
      return
    end
    local nCurTime = math.max(self.nEndTime - GetTime(), 0)
    self.pPanel:Button_SetText(self.btnCancel, i18n.EasyGet("\231\166\187\229\188\128({0})", nCurTime), "Text")
    return true
  end)
  self.nNeedCoin = nNeedCoin
  if self.nNeedCoin <= me.GetMoney(Currency.TYPE.Coin) then
    self.pPanel:Label_SetText(self.txtCoinValue, i18n.EasyGet("{0}", self.nNeedCoin))
  else
    self.pPanel:Label_SetText(self.txtCoinValue, i18n.EasyGet("<color=#c50505>{0}</color>", self.nNeedCoin))
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_MESSAGEBOX_OPEN)
end

function tbWnd:onClickOK()
  if self.nNeedCoin > me.GetMoney(Currency.TYPE.Coin) then
    me.CenterMsg(i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179"))
    return
  end
  XiaKeDao:OnXiaKeDaoClientRevive()
  self:Close()
end
