local tbWnd = Ui:CreateWindow("UIMessageBoxWuLinLianSai")
local tbControls = {
  txtTitle = "imgBG/imgTitle/txtTitle",
  txtMsg = "imgBG/imgMsgBG/txtMsg",
  btnOk = "imgBG/btnOK",
  btnCancel = "imgBG/btnCancel",
  btnCenter = "imgBG/btnCenter",
  MemberCheck = "imgBG/MemberCheckTips/MemberCheck"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnOk = function(self)
    self:OnClickOK()
  end,
  btnCenter = function(self)
    self:OnClickOK()
  end,
  btnCancel = function(self)
    self:CloseByDefault()
  end
}

function tbWnd:OnCreate()
  self.ScrollView = Ui:NewScrollView(self, tbControls.MemberCheck)
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen(szText, tbMemberState)
  szText = i18n.Parse(szText)
  self.pPanel:Label_SetText(tbControls.txtMsg, szText)
  self.pPanel:SetActive(tbControls.btnCenter, false)
  self.pPanel:SetActive(tbControls.btnOk, true)
  self.pPanel:SetActive(tbControls.btnCancel, true)
  self:UpdateMemberState(tbMemberState)
end

function tbWnd:UpdateMemberState(tbMemberState)
  local bAllAgree = true
  self.ScrollView:Reset()
  for i, tbState in pairs(tbMemberState) do
    if not tbState.bAgree then
      bAllAgree = false
    end
    local szMsg = i18n.EasyGet("{0}    {1}\231\186\167{2}", tbState.szName, tbState.nLevel, Faction:GetName(tbState.nFaction))
    self.ScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtName", szMsg)
      tbElement.pPanel:SetActive("txtName/imgCheck", tbState.bAgree)
    end)
  end
  if bAllAgree then
    self.pPanel:Button_SetEnable(tbControls.btnOk, true)
    self.pPanel:Button_SetText(tbControls.btnOk, i18n.EasyGet("\231\161\174\229\174\154"), "Text")
  else
    self.pPanel:Button_SetEnable(tbControls.btnOk, false)
    self.pPanel:Button_SetText(tbControls.btnOk, i18n.EasyGet("\231\173\137\229\190\133\231\161\174\229\174\154"), "Text")
  end
end

function tbWnd:OnClickOK()
  self:Close()
end

function tbWnd:CloseByDefault()
  self:Close()
end
