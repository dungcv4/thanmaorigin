function tbWnd:OnCreate()
  self.tbInputList = {
    panelInput = tbControls.panelInput,
    panelInputNoTeam = tbControls.panelInputNoTeam,
    panelInputNoKin = tbControls.panelInputNoKin,
    panelInputNoTong = tbControls.panelInputNoTong,
    panelInputFriend = tbControls.panelInputFriend,
    panelInputSystem = tbControls.panelInputSystem,
    panelNotAllowChat = tbControls.panelNotAllowChat,
    panelInputNoFriend = tbControls.panelInputNoFriend
  }
  self.pPanel:SetDealEmojiInputLimit(tbControls.InputChat, ChatMgr.nMaxMsgLengh)
  self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputPlaceHolder, ChatMgr.nMaxMsgLengh))
  self:InitChannel2InputTab()
  self.curInput = nil
  self.nCurChannelId = nil
  self.bIsVoice = false
  self.ChatContainerPanel = Ui:GetPanelByName("UIChatContainer")
  self.LargePanel = Ui:GetWindow("UIChat")
  self:DisableAllInput()
end

function tbWnd:DisableAllInput()
  for _, v in pairs(self.tbInputList) do
    self.pPanel:SetActive(v, false)
  end
end

function tbWnd:SwitchInput(nChannelId)
  assert(nChannelId)
  if not self.tbChannel2Input[nChannelId] then
    return
  end
  self.nCurChannelId = nChannelId
  if self.curInput then
    self.pPanel:SetActive(self.curInput, false)
  end
  local fnSet = self.tbChannel2Input[nChannelId]
  fnSet()
  ChatMgr:SetCurVoiceInfo(self.nCurChannelId, self.tbChatTargetInfo)
  self:UpdateCdExpression()
end

function tbWnd:SwitchPrivateInput(bCanInput)
  if Ui.bInZone and HanWuYiJi:IsInFubenMap(me) then
    return
  end
  self.pPanel:SetActive(tbControls.panelInput, bCanInput)
  self.pPanel:SetActive(tbControls.panelInputFriend, not bCanInput)
  self.curInput = bCanInput and tbControls.panelInput or tbControls.panelInputFriend
end

function tbWnd:SwitchInputMode()
  local cdTime = self.nLeftTime or 0
  if cdTime > 0 and not self.bIsVoice then
    me.CenterMsg(i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
  end
  self.bIsVoice = not self.bIsVoice
  self.pPanel:SetActive(tbControls.InputVoice, self.bIsVoice)
  self.pPanel:SetActive(tbControls.btnKeyboard, self.bIsVoice)
  self.pPanel:SetActive(tbControls.InputChat, not self.bIsVoice)
  self.pPanel:SetActive(tbControls.btnVoice, not self.bIsVoice)
  if not self.bIsVoice then
    self:UpdateCdExpression()
  end
end

function tbWnd:UpdateTargetInfo(tbChatTargetInfo)
  ChatMgr:SetCurVoiceInfo(self.nCurChannelId, tbChatTargetInfo)
  self.tbChatTargetInfo = tbChatTargetInfo
end

function tbWnd:UpdateCdExpression()
  if self.nCdTimer then
    Timer:Close(self.nCdTimer)
    self.nCdTimer = nil
  end
  local nCd = ChatMgr:GetCd(self.nCurChannelId, nil)
  if nCd <= 0 then
    self.nLeftTime = 0
    self.pPanel:Input_SetEnable(tbControls.InputChat, true)
    self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputPlaceHolder, ChatMgr.nMaxMsgLengh))
    return
  end
  self.nLeftTime = nCd
  self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
  self.pPanel:Input_SetEnable(tbControls.InputChat, false)
  self.nCdTimer = Timer:Register(Env.GAME_FPS, function()
    self.nLeftTime = self.nLeftTime - 1
    if self.nCdTimer and self.nLeftTime > 0 then
      self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
      self.pPanel:Input_SetEnable(tbControls.InputChat, false)
      return true
    else
      self.pPanel:Input_SetEnable(tbControls.InputChat, true)
      self.pPanel:Input_SetPlaceHolderText(tbControls.InputChat, i18n.Get(ChatMgr.Terms.InputPlaceHolder, ChatMgr.nMaxMsgLengh))
      self.nCdTimer = nil
      self.nLeftTime = nil
      return false
    end
  end)
end

function tbWnd:AddItem(nId)
  local itemObj = KItem.GetItemObj(nId)
  if not itemObj then
    return
  end
  local szName = i18n.Parse(itemObj.GetItemShowInfo(me.nFaction, me.nFactionSect))
  print(szName)
  local szMsg = self.pPanel:Input_GetText(tbControls.InputChat) or ""
  local szItem = string.format("<%s>", szName)
  szMsg = szItem .. string.gsub(szMsg, "^(%<.+%>)(.*)$", "%2")
  if Lib:StrLen(szMsg) > ChatMgr.nMaxMsgLengh then
    me.CenterMsg(i18n.Get(ChatMgr.Terms.ContentTooLong))
    return
  end
  local tbChatLink = {
    nLinkType = ChatMgr.LinkType.Item,
    linkParam = nId
  }
  self:TryUpdateInputContent(szMsg, tbChatLink)
end

function tbWnd:AddHawk(nId)
  local itemObj = KItem.GetItemObj(nId)
  if not itemObj then
    return
  end
  local szName = itemObj.GetItemShowInfo(me.nFaction, me.nFactionSect)
  local szMsg = i18n.EasyGet("\230\136\145\232\166\129\229\135\186\229\148\174<{0}>\239\188\140\232\176\129\232\166\129\228\185\176\232\175\183\232\129\148\231\179\187", szName)
  if Lib:StrLen(szMsg) > ChatMgr.nMaxMsgLengh then
    me.CenterMsg(i18n.Get(ChatMgr.Terms.ContentTooLong))
    return
  end
  ChatMgr:SetChatLink(ChatMgr.LinkType.Item, nId)
  ChatMgr:SendMsg(ChatMgr.ChannelType.Public, ChatMgr.ChannelType.Hawk, szMsg)
end

function tbWnd:DealOuterInput(szCurTab, szMsg, tbChatLink)
  local szContent = self.pPanel:Input_GetText(tbControls.InputChat)
  if "Position" == szCurTab and string.match(szMsg, "^(%<.+%>)(.*)$") then
    szContent = szMsg .. string.gsub(szContent, "^(%<.+%>)(.*)$", "%2")
  elseif "Emoji" == szCurTab then
    szContent = szContent .. szMsg
  elseif "History" == szCurTab then
    szContent = szMsg
  else
    szContent = szContent .. szMsg
  end
  self:TryUpdateInputContent(szContent, tbChatLink)
end

function tbWnd:TryUpdateInputContent(szMsg, tbChatLink)
  if self.nLeftTime and self.nLeftTime > 0 then
    me.CenterMsg(i18n.Get(ChatMgr.Terms.InputCD, self.nLeftTime))
    return
  end
  if not self.pPanel:IsActive(tbControls.InputChat) then
    return
  end
  self.pPanel:Input_SetText(tbControls.InputChat, szMsg)
  if tbChatLink then
    ChatMgr:SetChatLink(tbChatLink.nLinkType, tbChatLink.linkParam)
  end
end

function tbWnd:CloseAllSubPanel()

    Ui:CloseWindow("UIChatPlus")
  end
end

function tbWnd:SendClickFunc()
  local szMsg = self.pPanel:Input_GetText(tbControls.InputChat)
  local bSucceed = false
  local bPrivate = self.nCurChannelId == ChatMgr.ChannelType.Private
  if bPrivate then
    bSucceed = ChatMgr:SendPrivateMessage(self.tbChatTargetInfo.dwID, szMsg)
  else
    bSucceed = ChatMgr:SendMsg(self.nCurChannelId, nil, szMsg)
  end
  if bSucceed then
    self.pPanel:Input_SetText(tbControls.InputChat, "")
    self:UpdateCdExpression()
    self.tbOwner:LockChat(false)
    local _, bEmojiExist = ChatMgr:FilterPlayerMsg(self.nCurChannelId, szMsg)
    if bEmojiExist then
      EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_EMOJI)
    end
  end
end

function tbWnd:ColorClickFunc()

function tbWnd:PlusClickFunc()

    self:SwitchInputMode()
  end
  if not Ui:WindowVisible("UIChatPlus") then
    Ui:OpenWindow("UIChatPlus", function(...)
      self:DealOuterInput(...)
    end)
  end
end

function tbWnd:LocationClickFunc()
  if self.bIsVoice then
    self:SwitchInputMode()
  end
  local nMapId, nPosX, nPosY = me.GetWorldPos()
  local szLocation = string.format("<%s(%.0f,%.0f)>", i18n.Parse(me.szMapName), nPosX * Map.nShowPosScale, nPosY * Map.nShowPosScale)
  if me.nMapTemplateId == Tong.Def.nMainMapTemplateId then
    nMapId = Tong.Def.nMainMapTemplateId
  end
  self:DealOuterInput("Position", szLocation, {
    nLinkType = ChatMgr.LinkType.Position,
    linkParam = {
      nMapId,
      nPosX,
      nPosY,
      me.nMapTemplateId,
      me.nMapTemplateId
    }
  })
end
