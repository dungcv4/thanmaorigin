ChatMgr.CHAT_SAVE_GROUP = GetGroupKey("Chat")
ChatMgr.CHAT_SAVE_KEY_ZONE_COUNT = 3
ChatMgr.CHAT_SAVE_KEY_PUBLIC_COUNT_BUY = 4

function ChatMgr:GetPublicChatLeftCount(pPlayer)
  local nLeftCount = pPlayer.GetUserValue(ChatMgr.CHAT_SAVE_GROUP, ChatMgr.CHAT_SAVE_KEY_PUBLIC_COUNT_BUY)
  return DegreeCtrl:GetDegree(pPlayer, "PublicChatCount") + nLeftCount
end

function ChatMgr:GetColorChatLeftCount(pPlayer)
  local nColorTimes = pPlayer.GetUserValue(ChatMgr.COLOR_MSG_USER_VALUE_GROUP, ChatMgr.COLOR_MSG_USER_VALUE_KEY)
  return nColorTimes
end

function ChatMgr:GetZoneChatLeftCount(pPlayer)
  return pPlayer.GetUserValue(ChatMgr.CHAT_SAVE_GROUP, ChatMgr.CHAT_SAVE_KEY_ZONE_COUNT)
end

function ChatMgr:GetNamePrefix(nNamePrefix)
  local tbInfo = self:GetChatPrefixConfig(nNamePrefix)
  if tbInfo then
    return tbInfo.Name
  end
  return nil
end

function ChatMgr:GetNamePrefixColor(nNamePrefix)
  local tbInfo = self:GetChatPrefixConfig(nNamePrefix)
  if tbInfo then
    return tbInfo.NameColor
  end
  return nil
end

function ChatMgr:LoadChatIconConfig()
  self.tbChatIcon = {}
  self.tbChatIcon = Lib:LoadTabFileEx("Setting/Chat/ChatIcon.tab", "dss", "TemplateId", {
    "TemplateId",
    "Altlas",
    "SpriteName"
  })
end

function ChatMgr:LoadChatIconSetting()
  self.ChatIconSetting = {}
  self.ChatIconSetting = Lib:LoadTabFileEx("Setting/Chat/ChatIconSetting.tab", "dddsdsssd", "TemplateId", {
    "TemplateId",
    "HeadIconId",
    "BubbleIocnId",
    "Condition",
    "FixedTiem",
    "IconName",
    "GetDesc",
    "RuleDesc",
    "IsIosShenhe"
  })
end

function ChatMgr:LoadChatPrefixConfig()
  self.tbChatPrefix = {}
  self.tbChatPrefix = Lib:LoadTabFileEx("Setting/Chat/ChatPrefix.tab", "dsdds", nil, {
    "TemplateId",
    "Name",
    "NameColor",
    "Priority",
    "Condition"
  }, nil, nil, false)
  
  local function sortfun(a, b)
    return a.Priority > b.Priority
  end
  
  table.sort(self.tbChatPrefix, sortfun)
  self.tbChatPrefix = SetReadOnly(self.tbChatPrefix)
end

function ChatMgr:LoadConfig()
  self:LoadChatIconConfig()
  self:LoadChatIconSetting()
  self:LoadChatPrefixConfig()
end

ChatMgr:LoadConfig()

function ChatMgr:GetAllChatIconConfig()
  return self.ChatIconSetting
end

function ChatMgr:GetChatIconConfig(nId)
  return self.ChatIconSetting and self.ChatIconSetting[nId]
end

function ChatMgr:GetChatPrefixConfig(nId)
  for _, tbInfo in pairs(self.tbChatPrefix) do
    if tbInfo.TemplateId == nId then
      return tbInfo
    end
  end
  return nil
end
