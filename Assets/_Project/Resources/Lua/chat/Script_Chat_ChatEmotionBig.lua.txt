ChatMgr.EmotionBig = {}
local EmotionBig = ChatMgr.EmotionBig

function EmotionBig:LoadSetting()
  self.tbEmotionBig = Lib:LoadTabFileEx("Setting/Chat/EmotionBig.tab", "dsss", "Id", {
    "Id",
    "Name",
    "Atlas",
    "AniPrefix"
  })
  self.tbSortedKey = {}
  for k, v in pairs(self.tbEmotionBig) do
    table.insert(self.tbSortedKey, k)
  end
  table.sort(self.tbSortedKey)
  self.tbSortedKey = SetReadOnly(self.tbSortedKey)
end

EmotionBig:LoadSetting()

function EmotionBig:GetEmotionBig(nId)
  return nId and self.tbEmotionBig[nId]
end

function EmotionBig:GetSortedKeys()
  return self.tbSortedKey
end
