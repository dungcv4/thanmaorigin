ChatGroup.tbGroupData = ChatGroup.tbGroupData or {}
ChatGroup.tbPlayerInfoInGroup = ChatGroup.tbPlayerInfoInGroup or {}
local Oper2Notify = {
  [ChatGroup.S2COper.Create] = EventNotify.emChatGroup_Create,
  [ChatGroup.S2COper.Disband] = EventNotify.emChatGroup_Disband,
  [ChatGroup.S2COper.Join] = EventNotify.emChatGroup_Join,
  [ChatGroup.S2COper.ChangeName] = EventNotify.emChatGroup_ChangeName,
  [ChatGroup.S2COper.ChangeAnnouncement] = EventNotify.emChatGroup_ChangeAnnouncement,
  [ChatGroup.S2COper.Kickout] = EventNotify.emChatGroup_Kickout,
  [ChatGroup.S2COper.Leave] = EventNotify.emChatGroup_Leave
}

function ChatGroup:OnChatGroupOperSync(nChatGroupId, nOper, bSuccess_uselessnow, tbParam)
  if nOper == self.S2COper.Create then
    self.tbGroupData[nChatGroupId] = tbParam[1]
  elseif nOper == self.S2COper.Disband then
    self.tbGroupData[nChatGroupId] = nil
  elseif nOper == self.S2COper.Join then
    if self.tbGroupData[nChatGroupId] ~= nil then
      self.tbGroupData[nChatGroupId].tbMember[tbParam[1]] = tbParam[2]
    else
      self:ReqChatGroupData(nChatGroupId)
    end
    self.tbPlayerInfoInGroup[tbParam[1]] = tbParam[3]
  elseif nOper == self.S2COper.ChangeName then
    self.tbGroupData[nChatGroupId].szName = tbParam[1]
  elseif nOper == self.S2COper.ChangeAnnouncement then
    self.tbGroupData[nChatGroupId].szAnnouncement = tbParam[1]
  elseif nOper == self.S2COper.Kickout or nOper == self.S2COper.Leave then
    local nKickoutPlayerId = tbParam[1]
    if nKickoutPlayerId == me.dwID then
      self.tbGroupData[nChatGroupId] = nil
    else
      self.tbGroupData[nChatGroupId].tbMember[nKickoutPlayerId] = nil
    end
  end
  if Oper2Notify[nOper] then
    EventNotify.OnNotify(Oper2Notify[nOper], nChatGroupId, table.unpack(tbParam))
  end
end

function ChatGroup:ReqChatGroupData(nChatGroupId)
  RemoteServer.ReqChatGroupData(nChatGroupId)
end

function ChatGroup:OnSyncOneChatGroupData(tbChatGroupData)
  self.tbGroupData[tbChatGroupData.nChatGroupId] = tbChatGroupData
end

function ChatGroup:OnSyncChatGroupData(tbAllGroup)
  self.tbGroupData = {}
  for k, v in pairs(tbAllGroup.tbGroupData) do
    self.tbGroupData[k] = v
  end
end

function ChatGroup:OnSyncChatGroupMemberInfo(tbGroupMemberData)
  for k, v in pairs(tbGroupMemberData) do
    self.tbPlayerInfoInGroup[k] = v
  end
end

function ChatGroup:GetChatGroupId()
  local tbRet = {}
  for nChatGroupId, _ in pairs(self.tbGroupData) do
    table.insert(tbRet, nChatGroupId)
  end
  return tbRet
end

function ChatGroup:GetPlayerInfo(dwPlayerID)
  if dwPlayerID == me.dwID then
    return me
  end
  return self.tbPlayerInfoInGroup[dwPlayerID]
end

function ChatGroup:GetChatGroup(nChatGroupId)
  return nChatGroupId and self.tbGroupData[nChatGroupId]
end
