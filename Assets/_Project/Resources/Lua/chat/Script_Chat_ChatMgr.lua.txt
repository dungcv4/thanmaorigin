local function CheckCanDoSwitchMap(nSrcMapId, nDstMapId, nSrcMapTemplateId, nDstMapTemplateId)
  local bResult = true
  local szErrMsg
  if nil == nSrcMapTemplateId then
    LogErr("Illegal param 'nSrcMapTemplateId'.")
    bResult = false
    szErrMsg = nil
  end
  local tbSrcMapSetting = Map:GetMapSetting(nSrcMapTemplateId)
  if nil == tbSrcMapSetting then
    LogErr("Failed finding corresponding mapsetting. 'nSrcMapTemplateId' : ", nSrcMapTemplateId)
    bResult = false
    szErrMsg = nil
  elseif nil == nDstMapTemplateId then
    LogErr("Illegal param 'nDstMapTemplateId'.")
    bResult = false
    szErrMsg = nil
  elseif not IsSameMapId(nSrcMapId, nDstMapId) and nil ~= tbSrcMapSetting.TransForbidden and 1 == tonumber(tbSrcMapSetting.TransForbidden) then
    bResult = false
    szErrMsg = i18n.Get(ChatMgr.Terms.TransForbidden)
  end
  return bResult, szErrMsg
end

function ChatMgr._AddColor2Link(sz)
  if "</%w->" == sz then
    return sz
  end
  if "[]" == sz then
    return sz
  end
  if string.match(sz, "<%w-=#?%w->") ~= nil then
    return sz
  end
  return ChatMgr._AddColor2Link2(sz)
end

function ChatMgr._AddColor2Link2(sz)
  local tbLinkInfo = tbLinkColorValueCache.links[tbLinkColorValueCache.i]
  local tbMsg = tbLinkColorValueCache.tbMsg
  local bLarge = tbLinkColorValueCache.bLarge
  tbLinkColorValueCache.i = tbLinkColorValueCache.i + 1
  if nil == tbLinkInfo then
    return sz
  end
  local szLinkColor = ChatMgr:_GetLinkTypeColor(tbLinkInfo, tbMsg, bLarge)
  return szLinkColor and string.format("<color=#%sff><a href=%d>%s</a></color>", szLinkColor, tbLinkColorValueCache.i, sz) or sz
end

function ChatMgr:_GetLinkTypeColor(tbLinkInfo, tbMsg, bLarge)
  if not tbLinkInfo or tbLinkInfo.nLinkType == ChatMgr.LinkType.None then
    return nil
  end
  local szLinkColor = tbLinkColor[tbLinkInfo.nLinkType]
  if tbLinkInfo.nLinkType == ChatMgr.LinkType.XiaKeDao then
    if tbLinkInfo.linkParam.bFailure then
      szLinkColor = "939392"
    elseif DegreeCtrl:GetDegree(me, "XiaKeDaoAssistNum") <= 0 then
      szLinkColor = "939392"
    end
  end
  if tbSpecColor[tbLinkInfo.nLinkType] then
    szLinkColor = tbSpecColor[tbLinkInfo.nLinkType][bLarge and "Spec" or "Norl"][tbMsg.bClickLink and "Click" or "Norl"]
  end
  if tbLinkInfo.nLinkType == ChatMgr.LinkType.Item then
    local nQuality
    if tbLinkInfo.nTemplateId ~= nil then
      if tbLinkInfo.bUseMeInfo then
        tbLinkInfo.nFaction = me.nFaction
        tbLinkInfo.nFactionSect = me.nFactionSect
        tbLinkInfo.nBodySex = me.GetNpc().GetBodySex()
      end
      local szName, nIconImage, nViewImage, nQuality2, nResId = KItem.GetItemShowInfo(tbLinkInfo.nTemplateId, tbLinkInfo.nFaction, tbLinkInfo.nFactionSect, tbLinkInfo.nBodySex)
      nQuality = nQuality2
    end
    if tbLinkInfo.linkParam then
      local pItem = KItem.GetItemObj(tbLinkInfo.linkParam[1])
      if pItem then
        nQuality = pItem.nQuality
      end
    end
    if tbLinkInfo.bIsEquip then
      local tbInfo = Item:GetItemBaseProp(tbLinkInfo.nTemplateId)
      nQuality = tbInfo and tbInfo.nQuality
    elseif tbLinkInfo.nTemplateId then
      local tbInfo = Item:GetItemBaseProp(tbLinkInfo.nTemplateId)
      nQuality = tbInfo and tbInfo.nQuality
    end
    local szLargeColor, szSmallColor = Item:GetChatColor(nQuality or 1)
    szLinkColor = bLarge and szLargeColor or szSmallColor
  end
  return szLinkColor
end

function ChatMgr:InitPrivateList()
  self.RecentPrivateList = {}
  self.PrivateChatUnReadCache = {}
  self.PrivateChatReadCache = {}
  self.PrivateChatRedPointNum = 0
  self.RecentWantToChatList = {
    tbRoleInfo = {},
    tbCheckMap = {}
  }
  local tbSaveData = Client:GetPrivateMsgData()
  local tbData = tbSaveData.tbData
  if not tbData then
    return
  end
  for i, v in ipairs(tbData) do
    table.insert(self.RecentPrivateList, v[1])
    self.PrivateChatUnReadCache[v[1].dwID] = v[2]
    self.PrivateChatReadCache[v[1].dwID] = v[3]
  end
  self.nInitPrivateList = GetTime()
end

function ChatMgr:AddRecentWantToChatList(tbRoleInfo)
  if nil == tbRoleInfo then
    return
  end
  local dwID = tbRoleInfo.dwID
  local tbRecentRoleInfo = self.RecentWantToChatList.tbRoleInfo
  local tbCheckMap = self.RecentWantToChatList.tbCheckMap
  if tbCheckMap[dwID] then
    for i = 1, #tbRecentRoleInfo do
      if tbRecentRoleInfo[i].dwID == dwID then
        table.remove(tbRecentRoleInfo, i)
        table.insert(tbRecentRoleInfo, 1, tbRoleInfo)
        break
      end
    end
  else
    table.insert(tbRecentRoleInfo, 1, tbRoleInfo)
    if #tbRecentRoleInfo > self.MAX_WANTTOCHAT_COUNT then
      tbCheckMap[tbRecentRoleInfo[self.MAX_WANTTOCHAT_COUNT + 1].dwID] = nil
      table.remove(tbRecentRoleInfo, self.MAX_WANTTOCHAT_COUNT + 1)
    end
  end
  tbCheckMap[dwID] = true
end

function ChatMgr:RemoveRecentWantToChatList(dwID)
  local tbRecentRoleInfo = self.RecentWantToChatList.tbRoleInfo
  local tbCheckMap = self.RecentWantToChatList.tbCheckMap
  for i = 1, #tbRecentRoleInfo do
    if tbRecentRoleInfo[i].dwID == dwID then
      table.remove(tbRecentRoleInfo, i)
      break
    end
  end
  tbCheckMap[dwID] = nil
end

function ChatMgr:CheckUpdateStrangerState()
  local nNow = GetTime()
  if not self.nInitPrivateList or nNow - self.nInitPrivateList < 3600 then
    return
  end
  local tbCheckRoleIDs = {}
  for i, v in ipairs(self.RecentPrivateList) do
    local tbData = FriendShip:GetFriendDataInfo(v.dwID)
    if not tbData then
      table.insert(tbCheckRoleIDs, v.dwID)
    end
  end
  if next(tbCheckRoleIDs) then
    RemoteServer.RequestChatRoleBaseInfo(tbCheckRoleIDs)
  end
  self.nInitPrivateList = nNow
end

function ChatMgr:SavePriveMsg()
  if not Login.bEnterGame then
    return
  end
  local tbSaveData = Client:GetPrivateMsgData()
  tbSaveData.tbData = nil
  local tbSavePrivateMsgs = {}
  local PrivateChatUnReadCache = self.PrivateChatUnReadCache
  local PrivateChatReadCache = self.PrivateChatReadCache
  for i, v in ipairs(self.RecentPrivateList) do
    local tbUnRead = PrivateChatUnReadCache[v.dwID] or {}
    local tbReaded = PrivateChatReadCache[v.dwID] or {}
    local nToTal = #tbUnRead + #tbReaded
    local nNeedDel = nToTal - self.MAX_PRIVATE_MSG_NUM
    if nNeedDel > 0 then
      for i2 = 1, #tbReaded do
        table.remove(tbReaded, 1)
        nNeedDel = nNeedDel - 1
        if 0 == nNeedDel then
          break
        end
      end
      if nNeedDel > 0 then
        for i2 = 1, #tbUnRead do
          table.remove(tbUnRead, 1)
          nNeedDel = nNeedDel - 1
          if 0 == nNeedDel then
            break
          end
        end
      end
    end
    table.insert(tbSavePrivateMsgs, {
      v,
      tbUnRead,
      tbReaded
    })
  end
  tbSaveData.tbData = tbSavePrivateMsgs
  Client:SavePrivateMsgData()
end

  self.nTimerSavePrivate = Timer:Register(Env.GAME_FPS * 30, function()
    ChatMgr:SavePriveMsg()
    self.nTimerSavePrivate = nil
  end)
end

function ChatMgr:IsValidVoiceMsg(szVoiceFileId)
  return nil ~= szVoiceFileId and "" ~= szVoiceFileId

function ChatMgr:Filter4CharString(szMsg)
  local tbWords = Lib:GetUft8Chars(szMsg)
  local tbValidWords = {}
  for _, c in pairs(tbWords) do
    if string.len(c) <= 3 then
      table.insert(tbValidWords, c)
    end
  end
  return table.concat(tbValidWords)
end

function ChatMgr:FilterUnityTag(szMsg)
  szMsg = string.gsub(szMsg, "</?color.->", "")
  szMsg = string.gsub(szMsg, "</?size.->", "")
  szMsg = string.gsub(szMsg, "</?b.->", "")
  szMsg = string.gsub(szMsg, "</?i.->", "")
  szMsg = string.gsub(szMsg, "</?material.->", "")
  szMsg = string.gsub(szMsg, "</?quad.->", "")
  szMsg = string.gsub(szMsg, "</?a.->", "")
  szMsg = string.gsub(szMsg, "</?i18n.->", "")
  szMsg = string.gsub(szMsg, "</?fmt.->", "")
  return szMsg
end

function ChatMgr:SendMsg(nChannelType, nExtra, szMsg, bVoice, szVoiceFileId, nVoiceTime)
  if string.len(szMsg) > 2 and string.sub(szMsg, 1, 2) == "//" then
    return false
  end
  szMsg = ChatMgr:Filter4CharString(szMsg)
  szMsg = ChatMgr:FilterUnityTag(szMsg)
  if not ChatMgr:CheckSendMsg(nChannelType, nExtra, szMsg, false, szVoiceFileId, nVoiceTime) then
    return false
  end
  if nChannelType == ChatMgr.ChannelType.Kin then
    nExtra = ChatMgr.KinCommand:GetExtra()
  end
  local tbChatLink = self.tbChatLink or {}
  local nLinkType = tbChatLink.nLinkType
  local linkParam = tbChatLink.linkParam
  if nLinkType == ChatMgr.LinkType.Team and linkParam[4] and nChannelType == ChatMgr.ChannelType.Public then
    nExtra = ChatMgr.ChannelType.Recruit
  end
  self.tbChatLink = nil
  if not ChatMgr:CheckLinkAvailable(szMsg, nLinkType, linkParam) then
    nLinkType = 0
    linkParam = 0
  end
  nExtra = nExtra or 0
  if ChatMgr:IsValidVoiceMsg(szVoiceFileId) then
    SendChannelMessage(nChannelType, nExtra, szMsg, nLinkType, linkParam, szVoiceFileId, nVoiceTime)
  else
    SendChannelMessage(nChannelType, nExtra, szMsg, nLinkType, linkParam)
  end
  ChatMgr:DealAchievement(nChannelType, szMsg, bVoice)
  ChatMgr:InsertRecentSendMsg(szMsg, tbChatLink)
  return true
end

function ChatMgr:DealAchievement(nChannelId, szMsg, bVoice)
  if bVoice then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_VOICE)
  end
  if nChannelId == ChatMgr.ChannelType.Public then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_WORLD)
  elseif nChannelId == ChatMgr.ChannelType.Private then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_PRIVATEMSG)
  elseif nChannelId == ChatMgr.ChannelType.Kin then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_KIN)
  elseif nChannelId == ChatMgr.ChannelType.Tong then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_TONG)
  elseif nChannelId == ChatMgr.ChannelType.Color then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_COLOR)
  end
end

function ChatMgr:InsertRecentSendMsg(szMsg, tbChatLink)
  if not szMsg and string.len(szMsg) <= 1 then
    return
  end
  local tbRecentMsg = LocalData:GetRoleTable("ChatMsgHistory", {})
  for nIdx, tbOldMsg in ipairs(tbRecentMsg) do
    if tbOldMsg[1] == szMsg then
      table.remove(tbRecentMsg, nIdx)
      break
    end
  end
  if tbChatLink and Lib:IsEmptyTB(tbChatLink) then
    tbChatLink = nil
  end
  table.insert(tbRecentMsg, 1, {szMsg, tbChatLink})
  if #tbRecentMsg > ChatMgr.nMaxMsgHistoryCount then
    table.remove(tbRecentMsg)
  end
  LocalData:SetRoleTable("ChatMsgHistory", tbRecentMsg)
end

function ChatMgr:ClearColorFlag(szMsg)

  return szMsg

local function IsPlayerEmotion(nEmotionId)
  if nEmotionId <= ChatMgr.MAX_CHAT_EMOTION_LEN then
    return true
  end
  return ChatMgr.tbSpcialUserEmotion[nEmotionId]
end

function ChatMgr:FilterPlayerMsg(nChannelId, szMsg)
  if nChannelId == ChatMgr.ChannelType.System then
    return szMsg
  end
  szMsg = ChatMgr:ClearColorFlag(szMsg)
  local tbMsg = Lib:SplitStr(szMsg, "#")
  local bEmojiExist = false
  for i = 2, #tbMsg do
    local szStr = tbMsg[i]
    local szEmoId = string.match(szStr, "^%d%d?%d?")
    if szEmoId and not IsPlayerEmotion(tonumber(szEmoId)) then
      tbMsg[i] = " " .. szStr
    elseif szEmoId and IsPlayerEmotion(tonumber(szEmoId)) then
      bEmojiExist = true
    end
  end
  szMsg = table.concat(tbMsg, "#")
  return szMsg, bEmojiExist
end

function ChatMgr:_OnChannelMessage(nChannelType, nExtra, nSenderId, szSenderName, nFaction, nPortrait, nLevel, szMsg, szVoiceFileId, uVoiceTime, nNamePrefix, tbLinkInfo, tbExtra)
  if ChatMgr:InBlackList(nChannelType, nSenderId) then
    return
  end
  if ChatMgr:SpecialMsg(nChannelType, nSenderId, szSenderName, szMsg) then
    return
  end
  szMsg = ChatMgr:FilterPlayerMsg(nChannelType, szMsg) or szMsg
  if tbLinkInfo and (tbLinkInfo.nLinkType == ChatMgr.LinkType.LuaPacker or tbLinkInfo.nLinkType == ChatMgr.LinkType.LuaCustom) then
    tbLinkInfo = tbLinkInfo.tbData
  end
  if "" == szVoiceFileId then
    szVoiceFileId = nil
  end
  if nChannelType == ChatMgr.ChannelType.Public and nExtra and nExtra == ChatMgr.ChannelType.Hawk then
    nChannelType = ChatMgr.ChannelType.Hawk
    nExtra = 0
  end
  local nNow = GetTime()
  local tbMsg = {}
  tbMsg.nChannelType = nChannelType
  tbMsg.nSenderId = nSenderId
  tbMsg.szSenderName = szSenderName
  tbMsg.nNamePrefix = nNamePrefix
  tbMsg.nPortrait = nPortrait
  tbMsg.nFaction = nFaction
  tbMsg.nLevel = nLevel
  tbMsg.szMsg = szMsg
  tbMsg.nTime = nNow
  tbMsg.tbLinkInfo = tbLinkInfo
  tbMsg.szMsgVoiceFileId = szVoiceFileId
  tbMsg.uVoiceTime = uVoiceTime
  tbMsg.tbExtra = tbExtra and tbExtra.tbExtra
  tbMsg.nExtra = nExtra
  if tbMsg.tbExtra then
    tbMsg.nSpeechBalloon = tbMsg.tbExtra.nPlayerSpeechBalloon
  end
  local bAllHit, nSession, tbRoleInfoList = Player.RoleInfoCache:GetCache({nSenderId})
  if not bAllHit then
    tbMsg.nSession = nSession
    tbMsg.nPortraitFrame = 0
    self._GetCacheSessionMap[nSession] = {nChannelType = nChannelType, nExtra = nExtra}
  else
    tbMsg.nPortraitFrame = tbRoleInfoList[nSenderId].nPortraitFrame
  end
  if tbLinkInfo and tbLinkInfo.nLinkType == ChatMgr.LinkType.Item and tbLinkInfo.nTemplateId then
    local szReplaceName = KItem.GetItemShowInfo(tbLinkInfo.nTemplateId, tbLinkInfo.nFaction or me.nFaction, tbLinkInfo.nFactionSect or me.nFactionSect, tbLinkInfo.nBodySex or me.GetNpc().GetBodySex())
    szReplaceName = string.format("<%s>", i18n.Parse(szReplaceName))
    for w in szMsg:gmatch("%<.-%>") do
      if not w:match("<%w-=#?%w->") and not w:match("<%w- ?%w-=%w->") and not w:match("<?/%w->") then
        szMsg = szMsg:gsub(w, szReplaceName)
        break
      end
    end
    tbMsg.szMsg = szMsg
  end
  local nRealChannel = nChannelType
  if nChannelType == ChatMgr.ChannelType.System then
    nRealChannel, nExtra = ChatMgr:GetSystemMsgChannel(nSenderId)
    nExtra = nExtra or 0
  end
  if nChannelType == ChatMgr.ChannelType.Color then
    self:AddChannelMsg2Channel(nChannelType, nExtra, tbMsg)
    nRealChannel = ChatMgr.ChannelType.Public
  end
  self:AddChannelMsg2Channel(nRealChannel, nExtra, tbMsg)
  if ChatMgr:CheckShowInSmall(nChannelType, nExtra, nRealChannel) then
    self:AddChannelMsg2Channel(ChatMgr.chatSmall, nil, tbMsg)
  end
  if nChannelType == ChatMgr.ChannelType.Color and not self.nColorUpdateTimer then
    self.nColorUpdateTimer = Timer:Register(Env.GAME_FPS * ChatMgr.nColorMsgFreshTime, self.ColorUpdate, self)
    self:ColorUpdate()
  end
  if szVoiceFileId and tbMsg.nSenderId ~= me.dwID and ChatMgr:CheckVoiceAutoEnable(nChannelType) then
    ChatMgr:AddAutoPlayVoice(szVoiceFileId)
  end
  local szMsgBubble = szMsg
  if szVoiceFileId then
    szMsgBubble = string.format("#v7u2ic5e%s", szMsgBubble)
  end
  ChatMgr:DealChatBubbleTalk(nChannelType, nSenderId, szMsgBubble)
  ChatMgr:SetUnReadChannelMsg(nRealChannel, true)
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_NEW_MSG, nRealChannel, tbMsg, nExtra)
  ChatMgr:OnReceiveMsgTip(tbMsg)
  if tbLinkInfo then
    self:InitLinkUpdate()
  end
end

function ChatMgr:SendPrivateMessage(dwReceive, szMsg, szVoiceFileId, nVoiceTime)
  if not ChatMgr:CheckSendMsg(ChatMgr.ChannelType.Private, nil, szMsg, false, szVoiceFileId, nVoiceTime) then
    return
  end
  if FriendShip:IsHeInMyBlack(dwReceive) then
    me.CenterMsg(i18n.Get(self.Terms.BlackListLimited))
    return
  end
  szMsg = ChatMgr:Filter4CharString(szMsg)
  szMsg = ChatMgr:FilterUnityTag(szMsg)
  szMsg = ReplaceLimitWords(szMsg) or szMsg
  local tbChatLink = self.tbChatLink or {}
  local nLinkType = tbChatLink.nLinkType
  local linkParam = tbChatLink.linkParam
  if not ChatMgr:CheckLinkAvailable(szMsg, nLinkType, linkParam) then
    nLinkType = 0
    linkParam = 0
  end
  if ChatMgr:IsValidVoiceMsg(szVoiceFileId) then
    SendPrivateMessage(dwReceive, szMsg, nLinkType, linkParam, szVoiceFileId, nVoiceTime)
  else
    szVoiceFileId = nil
    nVoiceTime = 0
    SendPrivateMessage(dwReceive, szMsg, nLinkType, linkParam)
  end
  self:DealAchievement(ChatMgr.ChannelType.Private, szMsg, ChatMgr:IsValidVoiceMsg(szVoiceFileId))
  return true
end

function ChatMgr:SortRecentPrivateList(dwRoleId, bOnline)
  if not dwRoleId then
    return false
  end
  local RecentPrivateList = self.RecentPrivateList
  for i, v in ipairs(RecentPrivateList) do
    if v.dwID == dwRoleId then
      RecentPrivateList[1], RecentPrivateList[i] = RecentPrivateList[i], RecentPrivateList[1]
      if bOnline then
        v.nState = 2
      end
      local nTime = GetTime()
      if not v.nSyncTime or nTime - v.nSyncTime > ChatMgr.PrivateRequestInterval then
        v.nSyncTime = nTime
        RemoteServer.RequestChatRoleBaseInfo({dwRoleId})
      end
      return v
    end
  end
  if #RecentPrivateList >= self.MAX_PRIVATE_LIST_NUM then
    table.remove(RecentPrivateList)
  end
  local tbData = FriendShip:GetFriendDataInfo(dwRoleId)
  if tbData then
    table.insert(RecentPrivateList, 1, Lib:CopyTB1(tbData))
    return tbData
  else
    table.insert(RecentPrivateList, 1, {
      dwID = dwRoleId,
      szName = i18n.EasyGet("\233\153\140\231\148\159\228\186\186"),
      nPortrait = 1,
      nPortraitFrame = 0,
      nLevel = 1,
      nHonorLevel = 0,
      nFaction = 1,
      nVipLevel = 0,
      nState = 2,
      nSyncTime = GetTime(),
      bClientEmpty = true
    })
    RemoteServer.RequestChatRoleBaseInfo({dwRoleId})
    return false
  end
end

  local tbReaded = ChatMgr.PrivateChatReadCache[dwReceive] or {}
  local lastV = next(tbReaded) and tbReaded[#tbReaded]
  CheckInSertPrivateMsgTime(lastV, tbReaded, nNow)
  if "" == szVoiceFileId then
    szVoiceFileId = nil
  end
  table.insert(tbReaded, {
    nSenderId = me.dwID,
    szSenderName = me.szName,
    nFaction = me.nFaction,
    szMsg = szMsg,
    tbLinkInfo = tbLinkInfo,
    tbExtra = tbExtra and tbExtra.tbExtra,

    uVoiceTime = uVoiceTime,

  })

  self:CheckSavePrivateMsg()
  ChatMgr:InsertRecentSendMsg(szMsg)
  EventNotify.OnNotify(EventNotify.emNoTIFY_NEW_PRIVATE_MSG, dwReceive)

function ChatMgr:OnPrivateMessage(dwSender, nTime, szMsg, szVoiceFileId, uVoiceTime, tbLinkInfo, tbExtra)
  if FriendShip:IsHeInMyBlack(dwSender) then
    return
  end
  if "" == szVoiceFileId then
    szVoiceFileId = nil
  end
  if tbLinkInfo and tbLinkInfo.nLinkType == ChatMgr.LinkType.LuaPacker then
    tbLinkInfo.nLinkType = tbLinkInfo.tbData and tbLinkInfo.tbData.nLinkType
  end
  if tbLinkInfo and tbLinkInfo.nLinkType == ChatMgr.LinkType.LuaCustom then
    tbLinkInfo = tbLinkInfo.tbData
  end
  local tbPersonCache = ChatMgr.PrivateChatUnReadCache[dwSender] or {}
  ChatMgr.PrivateChatUnReadCache[dwSender] = tbPersonCache
  local tbPersonReadedCache = ChatMgr.PrivateChatReadCache[dwSender] or {}
  local tbData = self:SortRecentPrivateList(dwSender, true)
  local szSenderName = i18n.EasyGet("\233\153\140\231\148\159\228\186\186")
  local nFaction
  if tbData then
    szSenderName = tbData.szName or szSenderName
    nFaction = tbData.nFaction or tbData.nFactionSect
  end
  local tbMsg = {
    nChannelType = ChatMgr.ChannelType.Private,
    nSenderId = dwSender,
    nTime = nTime,
    szMsg = szMsg,
    tbLinkInfo = tbLinkInfo,
    szMsgVoiceFileId = szVoiceFileId,
    uVoiceTime = uVoiceTime,
    tbExtra = tbExtra and tbExtra.tbExtra,
    szSenderName = szSenderName,
    nFaction = nFaction
  }
  if tbMsg.tbExtra then
    tbMsg.nSpeechBalloon = tbMsg.tbExtra.nPlayerSpeechBalloon
  end
  if nTime <= GetTime() - 5 then
    table.insert(tbPersonCache, 1, tbMsg)
  else
    table.insert(tbPersonCache, tbMsg)
    self:AddChannelMsg2Channel(ChatMgr.chatSmall, nil, tbMsg)
  end
  if #tbPersonReadedCache + #tbPersonCache > FriendShip.nMaxPrivateMessages then
    if 0 == #tbPersonReadedCache then
      table.remove(tbPersonCache, 1)
    else
      table.remove(tbPersonReadedCache, 1)
    end
  end
  if not Ui:WindowVisible("UIChat") then
    ChatMgr.PrivateChatRedPointNum = ChatMgr.PrivateChatRedPointNum + 1
  end
  EventNotify.OnNotify(EventNotify.emNoTIFY_NEW_PRIVATE_MSG, dwSender)
  self:CheckSavePrivateMsg()
end

function ChatMgr:GetPrivateMsg(dwSender)
  if FriendShip:IsHeInMyBlack(dwSender) then
    ChatMgr.PrivateChatUnReadCache[dwSender] = nil
    ChatMgr.PrivateChatReadCache[dwSender] = nil
    return
  end
  local tbUnRead = ChatMgr.PrivateChatUnReadCache[dwSender] or {}
  local tbReaded = ChatMgr.PrivateChatReadCache[dwSender] or {}
  local tbRoleInfo = self:SortRecentPrivateList(dwSender)
  if not tbRoleInfo then
    return
  end
  for i, v in ipairs(tbReaded) do
    if v.nChannelType == ChatMgr.ChannelType.Private then
      v.szSenderName = tbRoleInfo.szName
      v.nPortrait = tbRoleInfo.nPortrait
      v.nPortraitFrame = tbRoleInfo.nPortraitFrame
      v.nLevel = tbRoleInfo.nLevel
      v.nFaction = tbRoleInfo.nFaction
    end
  end
  local lastV = next(tbReaded) and tbReaded[#tbReaded]
  for i, v in ipairs(tbUnRead) do
    v.szSenderName = tbRoleInfo.szName
    v.nPortrait = tbRoleInfo.nPortrait
    v.nPortraitFrame = tbRoleInfo.nPortraitFrame
    v.nLevel = tbRoleInfo.nLevel
    v.nFaction = tbRoleInfo.nFaction
    CheckInSertPrivateMsgTime(lastV, tbReaded, v.nTime)
    table.insert(tbReaded, v)
    lastV = v
  end
  ChatMgr.PrivateChatReadCache[dwSender] = tbReaded
  ChatMgr.PrivateChatUnReadCache[dwSender] = nil
  EventNotify.OnNotify(EventNotify.emNOTIFY_PRIVATE_MSG_NUM_CHANGE)
  if next(tbUnRead) then
    self:CheckSavePrivateMsg()
  end
  return tbReaded
end

function ChatMgr:GetPrivateMsgWithoutModified(dwSender)
  local tbUnRead = ChatMgr.PrivateChatUnReadCache[dwSender] or {}
  local tbReaded = ChatMgr.PrivateChatReadCache[dwSender] or {}

  for _, v in ipairs(tbUnRead) do
    table.insert(tbResult, v)
  end
  for _, v in ipairs(tbReaded) do
    table.insert(tbResult, v)
  end
  return tbResult
end

function ChatMgr:OnSyncEnterChatRoom(dwRoomHighId, dwRoomLowId, dwRoomHighKey, dwRoomLowKey, uBusinessId, nIsLarge, nPrivilege, uMemberId, tbUrl)
  if self.ChatRoom.dwRoomHighId == dwRoomHighId and self.ChatRoom.dwRoomLowId == dwRoomLowId then
    return
  end
  self:LeaveCurChatRoom()
  self.ChatRoom.dwRoomHighId = dwRoomHighId
  self.ChatRoom.dwRoomLowId = dwRoomLowId
  self.ChatRoom.dwRoomHighKey = dwRoomHighKey
  self.ChatRoom.dwRoomLowKey = dwRoomLowKey
  self.ChatRoom.uBusinessId = uBusinessId
  self.ChatRoom.nIsLarge = nIsLarge
  self.ChatRoom.nPrivilege = nPrivilege
  self.ChatRoom.uMemberId = uMemberId
  self.ChatRoom.isEnterRoom = false
  self.ChatRoom.bSpeakerState = false
  self.ChatRoom.bMicState = false
  for idx, szUrl in pairs(tbUrl) do
    if 1 ~= nIsLarge and string.sub(tbUrl[idx], 1, string.len("udp://")) ~= "udp://" then
      tbUrl[idx] = "udp://" .. szUrl
    end
  end
  self.ChatRoom.tbUrl = tbUrl
  self:StartEnterRoom()
end

function ChatMgr:StartCheckEnterTimer()
  if self.ChatRoom.nCheckTimer then
    return
  end
  local nRet = ChatRoomMgr.GetEnterRoomResult(self.ChatRoom.nIsLarge)
  if nRet == ChatMgr.ApolloVoiceErr.APOLLO_VOICE_JOIN_SUCC then
    ChatMgr:OnEnterChatRoom()
    return
  end
  local nCheckCount = 0
  self.ChatRoom.nCheckTimer = Timer:Register(Env.GAME_FPS * 1, function()
    local nRet = ChatRoomMgr.GetEnterRoomResult(self.ChatRoom.nIsLarge)
    if nRet == ChatMgr.ApolloVoiceErr.APOLLO_VOICE_JOIN_SUCC then
      ChatMgr:OnEnterChatRoom()
      return false
    end
    if nCheckCount >= 30 then
      ChatRoomMgr.LeaveRoom(self.ChatRoom.dwRoomHighId, self.ChatRoom.dwRoomLowId, self.ChatRoom.uMemberId, self.ChatRoom.nIsLarge)
      ChatMgr:StartEnterRoom()
      return false
    end
    nCheckCount = nCheckCount + 1
    return true
  end)
end

function ChatMgr:CloseCheckEnterTimer()
  if not self.ChatRoom.nCheckTimer then
    return
  end
  Timer:Close(self.ChatRoom.nCheckTimer)
  self.ChatRoom.nCheckTimer = nil
end

function ChatMgr:OnUploadFileComplete(nCode, szFilePath, nFileId)
  local szLocalPath = CS.VoiceModule.GetLocalVoiceFile(nFileId)
  local ret, err = os.rename(szFilePath, szLocalPath)
  if not ret then
    LogErr("OnUploadFileComplete rename", err, szFilePath, szLocalPath, nFileId)
    ret = CS.VoiceModule.ReNameLocalVoiceFile(szFilePath, szLocalPath)
    if not ret then
      LogErr("OnUploadFileComplete c# rename", szFilePath, szLocalPath, nFileId)
    end
  end
  ChatMgr.tbVoiceId2Channel[nFileId] = ChatMgr.tbVoicePath2Channel[szFilePath]
  if Client.EnableVoiceToText then
    local language = "cmn-Hans-CN"
    CS.VoiceModule.SpeechToText(nFileId, language, language)
  else
    local nVoiceLength = CS.VoiceModule.GetVoiceDurationByFileId(nFileId)
    self:SendVoiceMsg(false, nFileId, "", nVoiceLength)
  end
end

function ChatMgr:OnSpeechToTextComplete(code, fileId, result)
  local bOnlySendText = self:OnlySendVoiceText()
  local szMsg
  local nVoiceLength = CS.VoiceModule.GetVoiceDurationByFileId(fileId)
  if 0 ~= code and nil ~= result then
    szMsg = ""
  else
    szMsg = result
  end
  self:SendVoiceMsg(bOnlySendText, fileId, szMsg, nVoiceLength)
end

function ChatMgr:SendVoiceMsg(bOnlySendText, szFileId, szMsg, nVoiceLength)
  local nChannel = ChatMgr.tbVoiceId2Channel[szFileId].nChannelId
  local tbCurTargetInfo = ChatMgr.tbVoiceId2Channel[szFileId].tbChatTargetInfo
  local bPrivate
  local bSucceed = false
  if 0 == self.nSpecifyVoiceChannel then
    bPrivate = nChannel == ChatMgr.ChannelType.Private
  else
    bPrivate = self.nSpecifyVoiceChannel == ChatMgr.ChannelType.Private
    nChannel = self.nSpecifyVoiceChannel
  end
  if bPrivate then
    if bOnlySendText then
      bSucceed = self:SendPrivateMessage(tbCurTargetInfo.dwID, szMsg)
    else
      bSucceed = self:SendPrivateMessage(tbCurTargetInfo.dwID, szMsg, szFileId, nVoiceLength)
    end
  elseif bOnlySendText then
    bSucceed = self:SendMsg(nChannel, nil, szMsg)
  else
    bSucceed = self:SendMsg(nChannel, nil, szMsg, true, szFileId, nVoiceLength)
  end
end

function ChatMgr:PlayVoice(szVoiceFileId)
  self:UpdateMusicAndSound(true)
  self.szCurPlayingVoice = szVoiceFileId
  CS.VoiceModule.PlayVoice(szVoiceFileId)
  local nIndex = self:GetAutoPlayIndex(szVoiceFileId)
  if nIndex >= 0 then
    table.remove(self.tbAutoVoiceList, nIndex)
  end
end

function ChatMgr:GetAutoPlayIndex(szVoiceFileId)
  if self.tbAutoVoiceList == nil then
    return -1
  end
  for nIndex, szFileId in ipairs(self.tbAutoVoiceList) do
    if self.tbAutoVoiceList[nIndex] == szVoiceFileId then
      return nIndex
    end
  end
  return -1
end

function ChatMgr:OnPlayVoiceComplete(nCode, szFilePath)
  self.szCurPlayingVoice = nil
  if nil ~= self.tbAutoVoiceList and #self.tbAutoVoiceList > 0 then
    self.nStartPlayVoiceTimer = Timer:Register(Env.GAME_FPS * 2, function()
      if not self.tbAutoVoiceList and Lib:CountTB(self.tbAutoVoiceList) > 0 then

      else
        self:UpdateMusicAndSound(false)
      end
      self.nStartPlayVoiceTimer = -1
    end)
  else
    self:UpdateMusicAndSound(false)
  end
end

function ChatMgr:UpdateChatRoomByMap(nMapTemplateId)
  local bInRoom = self:IsInChatRoom()
  local bInLegionChatMap = ChatMgr:InChatMap(me.nMapTemplateId, ChatMgr.tbChatRoomForLegionMap)
  local bInKinAndTongChatMap = self:InChatMap(nMapTemplateId, self.tbChatRoomForKinAndTongMap)
  local bInKinChatMap = self:InChatMap(nMapTemplateId, self.tbChatRoomForKinMap)
  local bInTongChatMap = self:InChatMap(nMapTemplateId, self.tbChatRoomForTongMap)
  local bInTeamChatMap = self:InChatMap(nMapTemplateId, self.tbChatRoomForTeamMap)
  local bHasKin = Kin:HasKin()
  local bHasTong = Tong:HasTong()
  local bHasTeam = TeamMgr:HasTeam()
  local bHasLegion = Legion:HasLegion()
  Log("[ChatMgr] update chat room by map ", nMapTemplateId, bInRoom, bInLegionChatMap, bInKinAndTongChatMap, bInKinChatMap, bInTongChatMap, bInTeamChatMap, bHasKin, bHasTong, bHasTeam, bHasLegion)
  if bInKinAndTongChatMap then
    if bHasTong then
      local szTongChatRoomId = self:GetChatRoomId(self.nRoomTypeTong)
      if not self.tbChatRoom.bSpeakerOn and (not self.tbChatRoom.bMicOn or not Tong:IsTongVoiceManager()) then
        self:ExitChatRoom()
      elseif bInRoom and self.szChatRoomId ~= nil and self.szChatRoomId ~= szTongChatRoomId then
        self:ExitChatRoom()
        self:EnterChatRoom(szTongChatRoomId)
      elseif not bInRoom then
        self:EnterChatRoom(szTongChatRoomId)
      end
    elseif bHasKin then
      local szKinChatRoomId = self:GetChatRoomId(self.nRoomTypeKin)
      if not self.tbChatRoom.bSpeakerOn and (not self.tbChatRoom.bMicOn or not Kin:IsKinVoiceManager()) then
        self:ExitChatRoom()
      elseif bInRoom and self.szChatRoomId ~= nil and self.szChatRoomId ~= szKinChatRoomId then
        self:ExitChatRoom()
        self:EnterChatRoom(szKinChatRoomId)
      elseif not bInRoom then
        self:EnterChatRoom(szKinChatRoomId)
      end
    elseif bInRoom then
      self:ExitChatRoom()
    end
  elseif bInLegionChatMap then
    local szLegionChatRoomId = self:GetChatRoomId(self.nRoomTypeLegion)
    if not self.tbChatRoom.bSpeakerOn and (not self.tbChatRoom.bMicOn or not Legion:IsLegionVoiceManager()) then
      self:ExitChatRoom()
    elseif not bHasLegion and bInRoom then
      self:ExitChatRoom()
    elseif bHasLegion and bInRoom and self.szChatRoomId ~= nil and self.szChatRoomId ~= szLegionChatRoomId then
      self:ExitChatRoom()
      self:EnterChatRoom(szLegionChatRoomId)
    elseif bHasLegion and not bInRoom then
      self:EnterChatRoom(szLegionChatRoomId)
    end
  elseif bInTongChatMap then
    local szTongChatRoomId = self:GetChatRoomId(self.nRoomTypeTong)
    if not self.tbChatRoom.bSpeakerOn and (not self.tbChatRoom.bMicOn or not Tong:IsTongVoiceManager()) then
      self:ExitChatRoom()
    elseif not bHasTong and bInRoom then
      self:ExitChatRoom()
    elseif bHasTong and bInRoom and self.szChatRoomId ~= nil and self.szChatRoomId ~= szTongChatRoomId then
      self:ExitChatRoom()
      self:EnterChatRoom(szTongChatRoomId)
    elseif bHasTong and not bInRoom then
      self:EnterChatRoom(szTongChatRoomId)
    end
  elseif bInKinChatMap then
    local szKinChatRoomId = self:GetChatRoomId(self.nRoomTypeKin)
    if not self.tbChatRoom.bSpeakerOn and (not self.tbChatRoom.bMicOn or not Kin:IsKinVoiceManager()) then
      self:ExitChatRoom()
    elseif not bHasKin and bInRoom then
      self:ExitChatRoom()
    elseif bHasKin and bInRoom and self.szChatRoomId ~= nil and self.szChatRoomId ~= szKinChatRoomId then
      self:ExitChatRoom()
      self:EnterChatRoom(szKinChatRoomId)
    elseif bHasKin and not bInRoom then
      self:EnterChatRoom(szKinChatRoomId)
    end
  elseif bInTeamChatMap then
    local szTeamChatRoomId = self:GetChatRoomId(self.nRoomTypeTeam)
    if not self.tbChatRoom.bMicOn and not self.tbChatRoom.bSpeakerOn then
      self:ExitChatRoom()
    elseif not bHasTeam and bInRoom then
      self:ExitChatRoom()
    elseif bHasTeam and bInRoom and self.szChatRoomId ~= nil and self.szChatRoomId ~= szTeamChatRoomId then
      self:ExitChatRoom()
      self:EnterChatRoom(szTeamChatRoomId)
    elseif bHasTeam and not bInRoom then
      self:EnterChatRoom(szTeamChatRoomId)
    end
  else
    self:ExitChatRoom()
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_ROOM_STATUS)
end

function ChatMgr:InChatMap(nMapTemplateId, tbRoomMapList)
  for _, v in pairs(tbRoomMapList) do
    if v == nMapTemplateId then
      if nMapTemplateId == Tong.Def.nMainMapTemplateId then
        local isTongChallenge = Calendar:IsActivityInOpenState("TongChallenge")
        local isTongPart = Calendar:IsActivityInOpenState("TongParty")
        return isTongChallenge or isTongPart
      elseif WuLinGaoShou:IsWuLinGaoShouMap(nMapTemplateId) then
        if WuLinGaoShou:IsCurTimeFrameWuLinGaoShouMap(nMapTemplateId) then
          return Calendar:IsActivityInOpenState("WuLinGaoShou")
        else
          return false
        end
      else
        return true
      end
    end
  end
  return false
end

function ChatMgr:SwitchMicState(bOn)
  if Sdk:GetOsType() == Sdk.eOSType_Android and bOn then
    local nMicPermissionState = CS.VoiceModule.CheckMicPermissionState()
    if 1 ~= nMicPermissionState then
      self:ShowRequestMicUi()
      local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
      if tbWnd then
        tbWnd:CloseMic()
      end
      return
    end
  end
  local bInLegionChatMap = ChatMgr:InChatMap(me.nMapTemplateId, ChatMgr.tbChatRoomForLegionMap)
  local bInKinAndTongChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForKinAndTongMap)
  local bInKinChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForKinMap)
  local bInTongChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForTongMap)
  local bInTeamChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForTeamMap)
  local bInRoom = self:IsInChatRoom()
  local bHasKin = Kin:HasKin()
  local bHasTong = Tong:HasTong()
  local bHasTeam = TeamMgr:HasTeam()
  local bHasLegion = Legion:HasLegion()
  self.tbChatRoom.bMicOn = bOn
  if bInKinAndTongChatMap then
    if not bHasKin and not bHasTong then
      LogErr("[ChatMgr] SwitchMicState : should not call this api when player in kin and Tong activity and does not has kin and Tong")
      return
    end
    if bHasTong then
      local bHasPermission = Tong:IsTongVoiceManager()
      self.tbChatRoom.bMicOn = bHasPermission and bOn
    elseif bHasKin then
      local bHasPermission = Kin:IsKinVoiceManager()
      self.tbChatRoom.bMicOn = bHasPermission and bOn
    end
  elseif bInLegionChatMap then
    if not bHasLegion then
      LogErr("[ChatMgr] SwitchMicState : should not call this api when player in Legion activity and does not has Legion")
      return
    end
    local bHasPermission = Legion:IsLegionVoiceManager()
    self.tbChatRoom.bMicOn = bHasPermission and bOn
  elseif bInTongChatMap then
    if not bHasTong then
      LogErr("[ChatMgr] SwitchMicState : should not call this api when player in Tong activity and does not has Tong")
      return
    end
    local bHasPermission = Tong:IsTongVoiceManager()
    self.tbChatRoom.bMicOn = bHasPermission and bOn
  end
  if bInKinChatMap then
    if not bHasKin then
      LogErr("[ChatMgr] SwitchMicState : should not call this api when player in kin activity and does not has kin")
      return
    end
    local bHasPermission = Kin:IsKinVoiceManager()
    self.tbChatRoom.bMicOn = bHasPermission and bOn
  end
  if bInTeamChatMap and not bHasTeam then
    LogErr("[ChatMgr] SwitchMicState:cannot call this api in team activity map when player not in team.")
    return
  end
  if self.tbChatRoom.bMicOn then
    if bInRoom then
      CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn)
    end
  else
    CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn)
  end
  self:UpdateChatRoomByMap(me.nMapTemplateId)
end

  local bInKinChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForKinMap)
  local bInTongChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForTongMap)
  local bInTeamChatMap = self:InChatMap(me.nMapTemplateId, self.tbChatRoomForTeamMap)
  local bHasPermission = false
  local bHasKin = Kin:HasKin()
  local bHasTong = Tong:HasTong()
  local bHasTeam = TeamMgr:HasTeam()
  local bHasLegion = Legion:HasLegion()
  if bInKinAndTongChatMap then
    if not bHasKin and not bHasTong then
      CS.VoiceModule.EnableMic(false)
      return
    end
    if bHasTong then
      bHasPermission = Tong:IsTongVoiceManager()
      CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn and bHasPermission)
    elseif bHasKin then
      bHasPermission = Kin:IsKinVoiceManager()
      CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn and bHasPermission)
    end
  elseif bInLegionChatMap then
    if not bHasLegion then
      CS.VoiceModule.EnableMic(false)
      return
    end
    bHasPermission = Legion:IsLegionVoiceManager()
    CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn and bHasPermission)
  elseif bInTongChatMap then
    if not bHasTong then
      CS.VoiceModule.EnableMic(false)
      return
    end
    bHasPermission = Tong:IsTongVoiceManager()
    CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn and bHasPermission)
  elseif bInKinChatMap then
    if not bHasKin then
      CS.VoiceModule.EnableMic(false)
      return
    end
    bHasPermission = Kin:IsKinVoiceManager()
    CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn and bHasPermission)
  elseif bInTeamChatMap then
    if not bHasTeam then
      CS.VoiceModule.EnableMic(false)
      return
    end
    CS.VoiceModule.EnableMic(self.tbChatRoom.bMicOn)
  end
end

function ChatMgr:CloseLinkUpdate()
  if self.nLinkUpdateTimer then
    Timer:Close(self.nLinkUpdateTimer)
    self.nLinkUpdateTimer = nil
  end
  self.tbLinkInfoCache = {}
  self.tbLinkModUpdate = {}
end

function ChatMgr:InitLinkUpdate()
  if not self.bLinkUpdateSwitch then
    return
  end
  if not self.nLinkUpdateTimer then
    self.nLinkUpdateTimer = Timer:Register(Env.GAME_FPS * self.LinkUpdateInterval, self.OnLinkUpdateTimer, self)
  end
end

function ChatMgr:OnLinkUpdateTimer()
  if not self.bLinkUpdateSwitch then
    self.nLinkUpdateTimer = nil
    return
  end
  local tbQuerySet = {}
  local tbLocalSet = {}
  for _, nChannelType in pairs(self.LinkUpdateChannel) do
    for _, tbChatData in ipairs(self:GetChannelChatData(nChannelType)) do
      local tbLinkInfo = tbChatData.tbLinkInfo
      if tbLinkInfo then
        local nLinkType = tbLinkInfo.nLinkType
        local fnQuery = self.tbLinkQueryFns[nLinkType]
        if fnQuery then
          if not tbQuerySet[nLinkType] then
            tbQuerySet[nLinkType] = {}
          end
          if not tbLocalSet[nLinkType] then
            tbLocalSet[nLinkType] = {}
          end
          fnQuery(tbLinkInfo, tbQuerySet[nLinkType], self.tbLinkInfoCache[nLinkType], tbLocalSet[nLinkType], self.tbLinkModUpdate[nLinkType])
        end
      end
    end
  end
  for nLinkType, tbLinkQuerySet in pairs(tbQuerySet) do
    if Lib:IsEmptyTB(tbLinkQuerySet) then
      tbQuerySet[nLinkType] = nil
    end
  end
  if not Lib:IsEmptyTB(tbQuerySet) then
    RemoteServer.RequestChatLinkInfo(tbQuerySet)
  end
  for nLinkType, tbLinkLocalSet in pairs(tbLocalSet) do
    if Lib:IsEmptyTB(tbLinkLocalSet) then
      tbLocalSet[nLinkType] = nil
    end
  end
  if not Lib:IsEmptyTB(tbLocalSet) then
    EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_LINK_UPDATE)
  end
  self.tbLinkModUpdate = {}
  return true
end

function ChatMgr:RespondChatLinkInfo(tbResultSet)
  for nLinkType, tbLinkResultSet in pairs(tbResultSet) do
    local fnUpdate = self.tbLinkUpdateFns[nLinkType]
    if fnUpdate then
      if not self.tbLinkInfoCache[nLinkType] then
        self.tbLinkInfoCache[nLinkType] = {}
      end
      fnUpdate(tbLinkResultSet, self.tbLinkInfoCache[nLinkType])
    end
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHAT_LINK_UPDATE)
end
