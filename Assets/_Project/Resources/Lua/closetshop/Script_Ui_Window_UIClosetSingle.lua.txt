local tbModelState = {
  All = 1,
  OnlyPeople = 2,
  OnlyHorse = 3
}
tbWnd.tbOnSwipe = {
  HitZone = function(self, pos)
    Fashion:SwipeModel(pos.x)
  end
}
tbWnd.nHorsePartId = 4
tbWnd.tbShopType = {All = 1, Type = 2}
tbWnd.tbShowType = {
  [1] = function()
    return i18n.EasyGet("\229\141\179\229\176\134\229\188\128\230\148\190")
  end,
  [2] = function()
    return i18n.EasyGet("\233\166\150\229\133\133\232\142\183\229\190\151")
  end,
  [3] = function()
    return i18n.EasyGet("\229\165\151\232\163\133\232\180\173\228\185\176")
  end
}
tbWnd.tbShowTypeDefine = {
  nNormalType = 0,
  nObtainedFirstTime = 2,
  nGotoShopSuitShowType = 3
}
tbWnd.tbOnClick = {
  btnLeft = function(self)
    Fashion:Preview_TurnLeft()
  end,
  btnRight = function(self)
    Fashion:Preview_TurnRight()
  end,
  btnAll = function(self)
    self.bClickBtnAll = true
    self:OnSelectAllDropdown(0)
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnAll, true)
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnType, false)
    if #self.tbShopStyle > 1 then
      self.pPanel:SetActive(tbControls.btnTypeHide, true)
    end
    self.bClickBtnAll = false
  end,
  btnType = function(self)
    self.bClickBtnType = true
    self:OnSelectTypeDropdown(0)
    self.pPanel:SetActive(tbControls.btnAllHide, true)
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnAll, false)
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnType, true)
    self.bClickBtnType = false
  end,
  btnAllHide = function(self)
    self.bClickBtnAll = true
    self:OnSelectAllDropdown(self.nLastAllDropDownSelectIndex or 0)
    self.pPanel:SetActive(tbControls.btnAllHide, false)
    self.pPanel:SetActive(tbControls.btnTypeHide, true)
    self.bClickBtnAll = false
  end,
  btnTypeHide = function(self)
    self.bClickBtnType = true
    self:OnSelectTypeDropdown(self.nLastTypeDropDownSelectIndex or 0)
    self.pPanel:SetActive(tbControls.btnAllHide, true)
    self.pPanel:SetActive(tbControls.btnTypeHide, false)
    self.bClickBtnType = false

    Ui:CloseWindow("UIClosetShop")
    Ui:CloseWindow("UIShop")
    Ui:OpenWindow("UIBag", 5)
  end,
  btnModelSwitchPeopleShop = function(self)
    if self.nModelState == tbModelState.All then
      self.nModelState = tbModelState.OnlyPeople
      self:RefreshModelSwitchStateShow()
    elseif self.nModelState == tbModelState.OnlyPeople then
      self.nModelState = tbModelState.All
      self:RefreshModelSwitchStateShow()
    end
  end,
  btnModelSwitchHorseShop = function(self)
    if self.nModelState == tbModelState.All then
      self.nModelState = tbModelState.OnlyHorse
      self:RefreshModelSwitchStateShow()
    elseif self.nModelState == tbModelState.OnlyHorse then
      self.nModelState = tbModelState.All
      self:RefreshModelSwitchStateShow()
    end

  btnBuy = function(self)
    if self.bIsBuying then
      return
    end

      local tbShopItem = Shop:GetFashionItem(self.tbSelectPart2ShopItemId[self.nSelectPartIndex])

          return
        end

      Ui:OpenWindow("UIClosetBuy", {
        self.tbSelectPart2ShopItemId[self.nSelectPartIndex]
      })
    end

tbWnd.tbOnToggleClick = {
  toggleGender = function(self, isOn)
    if isOn then
      Fashion.nTemporarySex = 1
    else
      Fashion.nTemporarySex = 0
    end
    if Fashion.bShieldHorse then
      Fashion:Preview_RefreshPreview()
    else
      Fashion:Preview_RefreshPreview_AllPart()
    end
    self:RefreshSingleScrollView(self.nSelectShopId)
  end
}

function tbWnd:OnCreate()
  self.pPanel:Dropdown_BindEvent(tbControls.AllDropdown, function(value)
    self:OnSelectAllDropdown(value)
  end)
  self.pPanel:Dropdown_BindEvent(tbControls.TypeDropdown, function(value)
    self:OnSelectTypeDropdown(value)
  end)
  self.tbPartConfig = Fashion:GetFashionPartConfig()
  self.tbSingleScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.tbLeftPartScrollView = Ui:NewScrollView(self, tbControls.leftItemTemplate)
end

function tbWnd:OnOpen(nSelectPartIndex)
  self.pPanel:SetActive(tbControls.btnBuy, false)
  self.pPanel:SetActive(tbControls.btnGift, false)
  self.pPanel:SetActive(tbControls.shopCost, false)
  self.pPanel:SetActive(tbControls.BtnWishlistMark, false)
  self.pPanel:SetActive(tbControls.btnWishlistEtry, false)
  self.pPanel:ButtonGroup_SetSelect(tbControls.btnAll, false)
  self.pPanel:ButtonGroup_SetSelect(tbControls.btnType, false)
  self.bIsBuying = false
  self.nSelectShopId = nil
  self.nFixSelectShopId = nil
  self.nSelectPartIndex = nil
  self.nOpenShopType = tbWnd.tbShopType.All
  self.nModelState = tbModelState.OnlyPeople
  self.tbSelectPart2ShopItemId = {}
  for nIndex, _ in ipairs(self.tbPartConfig) do
    self.tbSelectPart2ShopItemId[nIndex] = 0
  end
  self:ResetPreview()
  self:SetDefaultSelectPartIndex(nSelectPartIndex)
  self:UpdateView()
end

function tbWnd:OnClose()
  Fashion.nTemporarySex = me.nSex
  Fashion:Preview_Destroy()
  self.nSelectShopId = nil
  self.tbSelectPart2ShopItemId = {}
  if self.nEffect2FashionTimerId then
    Timer:Close(self.nEffect2FashionTimerId)
    self.nEffect2FashionTimerId = nil
  end
end

function tbWnd:UpdateView()
  if self.nSelectShopId then
    self:RefreshSingleScrollView(self.nSelectShopId)
  else
    self:RefreshLeftPartScrollView()
    if not self.nDefaultPartIndex then
      self:RefreshShopStyle()
    end
  end
end

function tbWnd:RefreshFixSelectShop(nSelectPartId, nSelectLayer, nSelectShopId, nSelectSingleId)
  if nSelectPartId == self.nHorsePartId then
    if not self:IsOpenHorseShop() then
      return
    else
      Fashion:SetResDir(1)
    end
  end
  for nIndex, tbPartSetting in ipairs(self.tbPartConfig) do
    if tbPartSetting.nPartId == nSelectPartId and tbPartSetting.nLayer == nSelectLayer then
      self.nSelectPartIndex = nil
      self.nFixSelectShopId = nSelectShopId
      self.tbSelectPart2ShopItemId[nIndex] = nSelectSingleId
      self:SetDefaultSelectPartIndex(nIndex)
      self:RefreshLeftPartScrollView()
      break
    end
  end
end

function tbWnd:RefreshShopStyle()
  if not self.nSelectPartIndex then
    LogErr("RefreshShopStyle self.nSelectPartIndex is nil.", debug.traceback())
    return
  end
  self.tbShopStyle = Shop:GetFashiopShopStyleData(self.nSelectPartIndex, tbWnd.tbShopType.All)
  if not self.tbShopStyle or Lib:IsEmptyTB(self.tbShopStyle) then
    LogErr("RefreshShopStyle GetFashiopShopStyleData is nil or empty table.", self.nSelectPartIndex)
    return
  end
  local nLength = #self.tbShopStyle
  if 1 == nLength then
    self:RefreshAllStyleShow()
    self.pPanel:SetActive(tbControls.btnType, false)
    self.pPanel:SetActive(tbControls.TypeDropdown, false)
  elseif nLength > 1 then
    self:RefreshAllStyleShow()
    self:RefreshTypeStyleShow()
  end
end

function tbWnd:RefreshAllStyleShow()
  local tbType = self.tbShopStyle[tbWnd.tbShopType.All]
  local tbStyles = tbType.tbShopList
  if 1 == #tbStyles then
    if 1 == #self.tbShopStyle then
      self.pPanel:SetActive(tbControls.typeCell, false)
    else
      self.pPanel:SetActive(tbControls.typeCell, true)
      self.pPanel:SetActive(tbControls.btnAll, true)
      self.pPanel:SetActive(tbControls.AllDropdown, false)
      self.pPanel:Label_SetText(tbControls.AllLabel, i18n.Parse(tbType.szTypeName))
    end
    if self.nOpenShopType == tbWnd.tbShopType.All then
      if self.nFixSelectShopId then
        local nShopId = self.nFixSelectShopId
        self.nFixSelectShopId = nil
        self:RefreshSingleScrollView(nShopId)
      else
        self:RefreshSingleScrollView(tbType.nShopId)
      end
      self.pPanel:ButtonGroup_SetSelect(tbControls.btnAll, true)
    else
      self.pPanel:ButtonGroup_SetSelect(tbControls.btnAll, false)
    end
  elseif #tbStyles > 1 then
    self.pPanel:SetActive(tbControls.typeCell, true)
    self.pPanel:SetActive(tbControls.btnAll, false)
    self.pPanel:SetActive(tbControls.AllDropdown, true)
    local Optations = {}
    for _, tbStyle in ipairs(tbStyles) do
      table.insert(Optations, i18n.Parse(tbStyle.szStyle))
    end
    self.pPanel:SetActive(tbControls.btnAllHide, false)
    self.pPanel:Label_SetText(tbControls.txtAllHideName, i18n.Parse(tbType.szTypeName))
    self.pPanel:Dropdown_ResetOptions(tbControls.AllDropdown, Optations)
    if self.nOpenShopType == tbWnd.tbShopType.All then
      if self.nFixSelectShopId then
        local nShopId = self.nFixSelectShopId
        self.nFixSelectShopId = nil
        self:RefreshSingleScrollView(nShopId)
      else
        self:RefreshSingleScrollView(tbStyles[1].nShopId)
      end
    end
  end
end

function tbWnd:RefreshTypeStyleShow()
  local tbType = self.tbShopStyle[tbWnd.tbShopType.Type]
  local tbStyles = tbType.tbShopList
  if 1 == #tbStyles then
    self.pPanel:SetActive(tbControls.btnType, true)
    self.pPanel:SetActive(tbControls.TypeDropdown, false)
    self.pPanel:Label_SetText(tbControls.TypeLabel, i18n.Parse(tbType.szTypeName))
    if self.nOpenShopType == tbWnd.tbShopType.Type then
      if self.nFixSelectShopId then
        local nShopId = self.nFixSelectShopId
        self.nFixSelectShopId = nil
        self:RefreshSingleScrollView(nShopId)
      else
        self:RefreshSingleScrollView(tbType.nShopId)
      end
      self.pPanel:ButtonGroup_SetSelect(tbControls.btnType, true)
    else
      self.pPanel:ButtonGroup_SetSelect(tbControls.btnType, false)
    end
  elseif #tbStyles > 1 then
    self.pPanel:SetActive(tbControls.btnType, false)
    self.pPanel:SetActive(tbControls.TypeDropdown, true)
    local Optations = {}
    for _, tbStyle in ipairs(tbStyles) do
      table.insert(Optations, i18n.Parse(tbStyle.szStyle))
    end
    self.pPanel:SetActive(tbControls.btnTypeHide, true)
    self.pPanel:Label_SetText(tbControls.txtTypeHideName, i18n.Parse(tbType.szTypeName))
    self.pPanel:Dropdown_ResetOptions(tbControls.TypeDropdown, Optations)
    if self.nOpenShopType == tbWnd.tbShopType.Type then
      if self.nFixSelectShopId then
        local nShopId = self.nFixSelectShopId
        self.nFixSelectShopId = nil
        self:RefreshSingleScrollView(nShopId)
      else
        self:RefreshSingleScrollView(tbType.nShopId)
      end
    end
  end
end

function tbWnd:OnSelectAllDropdown(value)
  if value == self.nLastAllDropDownSelectIndex and not self.bClickBtnAll then
    return
  end
  self.nLastAllDropDownSelectIndex = value
  local nIndex = value + 1
  local tbTypes = self.tbShopStyle[1]
  if not tbTypes or nIndex > #tbTypes.tbShopList then
    LogErr("UIClosetSuit:OnSelectAllDropdown the length of tbShopList ", #tbTypes.tbShopList, " less than value + 1", nIndex)
    return
  end
  local tbStyle = tbTypes.tbShopList[nIndex]
  local nShopId = tbStyle.nShopId
  self:RefreshSingleScrollView(nShopId)
end

function tbWnd:OnSelectTypeDropdown(value)
  if value == self.nLastTypeDropDownSelectIndex and not self.bClickBtnType then
    return
  end
  self.nLastTypeDropDownSelectIndex = value
  local nIndex = value + 1
  local tbTypes = self.tbShopStyle[2]
  if not tbTypes or nIndex > #tbTypes.tbShopList then
    LogErr("UIClosetSuit:OnSelectTypeDropdown the length of tbShopList ", #tbTypes.tbShopList, " less than value  + 1", nIndex)
    return
  end
  local tbStyle = tbTypes.tbShopList[nIndex]
  local nShopId = tbStyle.nShopId
  self:RefreshSingleScrollView(nShopId)
end

function tbWnd:FashionBuyResult(bSuccess)
  self.bIsBuying = false
  if bSuccess then
    self:RefreshSingleScrollView(self.nSelectShopId)
  end
end

function tbWnd:SetDefaultSelectPartIndex(nSelectPartIndex)
  if nSelectPartIndex and nSelectPartIndex > 0 and nSelectPartIndex <= #self.tbPartConfig then
    if self.tbPartConfig[nSelectPartIndex].nPartId == self.nHorsePartId and not self:IsOpenHorseShop() then
      LogErr("UIClosetSingle SetDefaultSelectPartIndex horse shop is not open yet. nSelectPartIndex is illegal.", nSelectPartIndex)
      return
    end
    self.nDefaultPartIndex = nSelectPartIndex
  else
    self.nDefaultPartIndex = 1
  end
end

function tbWnd:RefreshLeftPartScrollView()
  self.tbLeftPartScrollView:Reset()
  for nIndex, tbPart in ipairs(self.tbPartConfig) do
    while true do
      do
        local nPartId = tbPart.nPartId
        if nPartId ~= self.nHorsePartId or self:IsOpenHorseShop() then
          self.tbLeftPartScrollView:SetElement(nIndex, function(tbElement)
            local pPanel = tbElement.pPanel
            
            local function fnClick()
              if self.nSelectPartIndex ~= nIndex then
                local lastElement = self.tbLeftPartScrollView:GetElement(self.nSelectPartIndex)
                if lastElement then
                  lastElement.pPanel:SetActive("imgSelected", false)
                end
                pPanel:SetActive("imgSelected", true)
                self.nSelectPartIndex = nIndex
                self:RefreshShopStyle()
                if nPartId == self.nHorsePartId then
                  Fashion.bHorse = true
                  self.pPanel:SetActive(tbControls.btnModelSwitchHorseShop, true)
                  self.pPanel:SetActive(tbControls.btnModelSwitchPeopleShop, false)
                  if not self.nModelState or self.nModelState == tbModelState.OnlyPeople then
                    self.nModelState = tbModelState.OnlyHorse
                  end
                  self:RefreshModelSwitchStateShow()
                else
                  Fashion.bHorse = false
                  self.pPanel:SetActive(tbControls.btnModelSwitchHorseShop, false)
                  self.pPanel:SetActive(tbControls.btnModelSwitchPeopleShop, true)
                  if not self.nModelState or self.nModelState == tbModelState.OnlyHorse then
                    self.nModelState = tbModelState.OnlyPeople
                  end
                  self:RefreshModelSwitchStateShow()
                end
              end
            end
            
            if nIndex == self.nDefaultPartIndex then
              self.nDefaultPartIndex = nil
              fnClick()
            else
              pPanel:SetActive("imgSelected", false)
            end
            pPanel:Button_BindEvent("Button", fnClick)
            pPanel:Label_SetText("txtClothes", i18n.Parse(tbPart.szName))
          end)
          self:SetSelectPartIcon(nIndex, self.tbSelectPart2ShopItemId[nIndex])
        end
      end
      break
    end
  end
end

function tbWnd:SetSelectPartIcon(nIndex, nFashionId)
  if not nIndex or nIndex <= 0 then
    return
  end
  local tbElement = self.tbLeftPartScrollView:GetElement(nIndex)
  if tbElement then
    if nFashionId and nFashionId > 0 then
      local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
      if tbSetting then
        tbElement.pPanel:SetActive("Empty", false)
        tbElement.pPanel:SetActive("UIItemGrid", true)
        tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        Fashion:InitFashionGrid(tbElement.pGrid, tbSetting)
      end
    else
      tbElement.pPanel:SetActive("Empty", true)
      tbElement.pPanel:SetActive("UIItemGrid", false)
    end
  end
end

function tbWnd:IsOpenHorseShop()
  local pHorseItem = me.GetEquipByPos(Item.EQUIPPOS_HORSE)
  if not pHorseItem then
    return false
  end
  return true
end

function tbWnd:RefreshSingleScrollView(nShopId)
  if not nShopId then
    LogWarn("RefreshSingleScrollView nShopId is nil")
    return
  end
  local tbItems = Shop:GetShopItemsToShow(nShopId)
  if not tbItems or Lib:IsEmptyTB(tbItems) then
    LogWarn("RefreshSingleScrollView GetShopItemsToShow tbItems is nil or empty table")
    return
  end
  self.nSelectShopId = nShopId
  self:ClearSelectFashionInfoShow()
  if not self.nSelectPartIndex or 0 == self.nSelectPartIndex then
    LogErr("RefreshSingleScrollView nSelectPartIndex is nil or 0.", self.nSelectPartIndex)
    return
  end
  local tbUsefulItems = {}
  local tbPartSetting = self.tbPartConfig[self.nSelectPartIndex]
  for _, tbItem in ipairs(tbItems) do
    local nFashionId = tbItem.nFashionID
    local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
    if tbSetting and tbSetting.nLayer == tbPartSetting.nLayer then
      tbItem.tbSetting = tbSetting
      table.insert(tbUsefulItems, tbItem)
    end
  end
  local tbHighlightPinned = {}
  local nLength = #tbUsefulItems
  local nPreviewFashionPartIndex, nPreviewFashionId = self:GetPreviewFashionPartInfo(tbPartSetting.nPartId, tbPartSetting.nLayer)
  if nPreviewFashionId and tbPartSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
    for nIndex = nLength, 1, -1 do
      local tbItem = tbUsefulItems[nIndex]
      if Lib:Is_include(nPreviewFashionId, tbItem.tbSetting.tbEffect2FashionIds) then
        local tbTemp = table.remove(tbUsefulItems, nIndex)
        table.insert(tbHighlightPinned, tbTemp)
      end
    end
  elseif nPreviewFashionId then
    for nIndex = nLength, 1, -1 do
      local tbItem = tbUsefulItems[nIndex]
      if Lib:Is_include(nPreviewFashionId, tbItem.tbSetting.tbFashion2EffectIds) then
        local tbTemp = table.remove(tbUsefulItems, nIndex)
        table.insert(tbHighlightPinned, tbTemp)
      end
    end
  end
  local nHaveHightlightCount = #tbHighlightPinned
  if nHaveHightlightCount > 0 then
    tbUsefulItems = Lib:MergeTable(tbHighlightPinned, tbUsefulItems)
  end
  self.tbSingleScrollView:Reset()
  if Lib:IsEmptyTB(tbUsefulItems) then
    self.pPanel:SetActive(tbControls.EmptyShop, true)
    self.pPanel:Label_SetText(tbControls.txtEmptyDesc, i18n.EasyGet("\230\154\130\230\151\160\229\143\175\229\135\186\229\148\174\231\154\132\229\164\150\232\167\130"))
    return
  else
    self.pPanel:SetActive(tbControls.EmptyShop, false)
  end
  self.nSelectIndex = nil
  local tbHave = Fashion:GetHaveFashions()
  for nIndex, tbItem in ipairs(tbUsefulItems) do
    self.tbSingleScrollView:SetElement(nIndex, function(tbElement)
      local tbSetting = tbItem.tbSetting
      tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      Fashion:InitFashionGrid(tbElement.pGrid, tbSetting)
      local pPanel = tbElement.pPanel
      if tbHave[tbItem.nFashionID] then
        pPanel:SetActive("imgHaveBg", true)
      else
        pPanel:SetActive("imgHaveBg", false)
      end
      local szMoneyIcon = Currency:GetIcon(tbItem.szMoneyType)
      if 0 == tbItem.nShowType then
        pPanel:SetActive("Cost", true)
        pPanel:SetActive("txtComingSoon", false)
        pPanel:Sprite_SetSprite("Cost/imgMoney", szMoneyIcon)
        if tbItem.nDiscountPrice > 0 then
          pPanel:SetActive("MarkDiscounts", true)
          pPanel:SetActive("Cost/CostDiscounts", true)
          pPanel:Label_SetText("Cost/txtCost", tbItem.nDiscountPrice)
          pPanel:Label_SetText("Cost/CostDiscounts/txtCost", tbItem.nPrice)
          pPanel:Label_SetText("MarkDiscounts/txt", i18n.GetDiscountText(tbItem.nDiscount))
        else
          pPanel:SetActive("MarkDiscounts", false)
          pPanel:SetActive("Cost/CostDiscounts", false)
          pPanel:Label_SetText("Cost/txtCost", tbItem.nPrice)
        end
      else
        pPanel:SetActive("Cost", false)
        pPanel:SetActive("txtComingSoon", true)
        pPanel:SetActive("MarkDiscounts", false)
        pPanel:Label_SetText("txtComingSoon", tbWnd.tbShowType[tbItem.nShowType]())
      end
      
      local function fnSingleClick(tbClickItem)
        if self.nSelectIndex ~= nIndex then
          if 0 == tbClickItem.nShowType or tbClickItem.nShowType == tbWnd.tbShowTypeDefine.nGotoShopSuitShowType then
            self.pPanel:SetActive(tbControls.btnBuy, true)
            self.pPanel:SetActive(tbControls.shopCost, true)
          else
            self.pPanel:SetActive(tbControls.btnBuy, false)
            self.pPanel:SetActive(tbControls.shopCost, false)
          end
          pPanel:SetActive("imgSelected", true)
          if self.nSelectIndex then
            local pLastElement = self.tbSingleScrollView:GetElement(self.nSelectIndex)
            pLastElement.pPanel:SetActive("imgSelected", false)
          end
          self.nSelectIndex = nIndex
          self.tbSelectPart2ShopItemId[self.nSelectPartIndex] = tbClickItem.nId
          self:PreviewWear(tbClickItem.nFashionID)
          self:SetSelectPartIcon(self.nSelectPartIndex, tbClickItem.nFashionID)
          self:_HandleClickOnOtherFashion(tbClickItem.nFashionID)
          if nIndex > nHaveHightlightCount and nHaveHightlightCount > 0 and tbSetting.nLayer ~= Npc.NpcPartLayerDef.npc_part_layer_effect then
            nHaveHightlightCount = 0
            self:HightLightAllSingleElement()
            self:SetSelectPartIcon(nPreviewFashionPartIndex)
            me.CenterMsg(i18n.EasyGet("\229\164\150\232\167\130\228\184\141\229\140\185\233\133\141\239\188\140\229\183\178\233\135\141\230\150\176\229\143\150\230\182\136\231\137\185\230\149\136\233\162\132\232\167\136\233\128\137\229\174\154"))
          end
          self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(tbSetting.szName))
          self.pPanel:Label_SetText(tbControls.txtDesc, i18n.Parse(tbSetting.szDesc))
          self.pPanel:Label_SetText(tbControls.txtPretty, i18n.EasyGet("\233\173\133\229\138\155\229\128\188+{0}", tbSetting.nPrettyValue))
          if tbClickItem.nFashionID and Fashion.AttrCfg[tbClickItem.nFashionID] then
            local szMagicName = Fashion.AttrCfg[tbClickItem.nFashionID][1]
            local tbMagic = {
              table.unpack(Fashion.AttrCfg[tbClickItem.nFashionID], 2)
            }
            local szFinalName, szFinalValue = FightSkill:GetMagicDescSplit(szMagicName, tbMagic)
            local szAttribute = string.format("%s : %s", szFinalName, szFinalValue)
            self.pPanel:SetActive(tbControls.txtAttribute, true)
            self.pPanel:Label_SetText(tbControls.txtAttribute, i18n.EasyGet(szAttribute))
          else
            self.pPanel:SetActive(tbControls.txtAttribute, false)
          end
          Fashion:IsNeedPopBoxPartShield(tbSetting.nPart)
          self:_HandleShopItemShowType(tbClickItem)
          self.pPanel:Sprite_SetSprite(tbControls.imgMoney, szMoneyIcon)
        else
          pPanel:SetActive("imgSelected", false)
          self:_HandleRepeatClickFashion(tbClickItem.nFashionID)
        end
      end
      
      Ui:Button_BindEvent(pPanel, "Button", fnSingleClick, tbItem)
      if self.tbSelectPart2ShopItemId[self.nSelectPartIndex] == tbItem.nId then
        fnSingleClick(tbItem)
      else
        pPanel:SetActive("imgSelected", false)
      end
      if tbPartSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
        if not nPreviewFashionId or nHaveHightlightCount > 0 and nIndex <= nHaveHightlightCount then
          pPanel:Sprite_SetColor("Button", 48, 48, 48, 0)
        else
          pPanel:Sprite_SetColor("Button", 48, 48, 48, 153)
        end
      elseif not nPreviewFashionId or 0 == nHaveHightlightCount or nHaveHightlightCount > 0 and nIndex <= nHaveHightlightCount then
        pPanel:Sprite_SetColor("Button", 48, 48, 48, 0)
      else
        pPanel:Sprite_SetColor("Button", 48, 48, 48, 153)
      end
      pPanel:SetActive("MarkNew", tbItem.bShowNewTip)
    end)
  end
  if self.nSelectIndex and self.nSelectIndex > 0 and self.tbSingleScrollView:GetCount() >= self.nSelectIndex then
    self.tbSingleScrollView:ScrollToIndex(self.nSelectIndex, false)
  end
end

function tbWnd:_HandleClickOnOtherFashion(nFashionId)
  local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
  local nPreviewFashionPartIndex, nPreviewFashionId = self:GetPreviewFashionPartInfo(tbSetting.nPart + 1, tbSetting.nLayer)
  if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
    if not nPreviewFashionId or not Lib:Is_include(nPreviewFashionId, tbSetting.tbEffect2FashionIds) then
      local nSpecifyShopItemId = Shop:GetShopItemIdByFashionId(nPreviewFashionPartIndex, tbWnd.tbShopType.All, nFashionId)
      if nSpecifyShopItemId > 0 then
        self.nEffect2FashionTimerId = Timer:Register(3, function()
          if self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex] > 0 then
            me.CenterMsg(i18n.EasyGet("\229\164\150\232\167\130\228\184\142\231\137\185\230\149\136\228\184\141\229\140\185\233\133\141\239\188\140\229\183\178\233\135\141\230\150\176\233\128\137\230\139\169\229\140\185\233\133\141\229\164\150\232\167\130"))
          else
            me.CenterMsg(i18n.EasyGet("\229\183\178\230\160\185\230\141\174\231\137\185\230\149\136\233\128\137\229\174\154\229\140\185\233\133\141\229\164\150\232\167\130"))
          end
          self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex] = nSpecifyShopItemId
          self:SetDefaultSelectPartIndex(nPreviewFashionPartIndex)
          self:RefreshLeftPartScrollView()
        end)
      else
        LogErr("UIClosetSingle Can not find a usefully nSpecifyShopItemId by Shop:GetShopItemIdByFashionId", nFashionId, nPreviewFashionPartIndex, nSpecifyShopItemId)
      end
      return
    end
  elseif nPreviewFashionId and Lib:Is_include(nPreviewFashionId, tbSetting.tbFashion2EffectIds) then
    self:PreviewWear(nPreviewFashionId)
  else
    local tbPreviewEffectDict = Fashion:GetPreviewEffectList()
    local nPreviewEffectResId = tbPreviewEffectDict[tbSetting.nPart].nResId
    local nPreviewEffectFashionId = tbPreviewEffectDict[tbSetting.nPart].nFashionId
    if nPreviewEffectFashionId then
      local nWearingEffectFashionId = Fashion:GetWearingEffectFashionIdByPart(me, tbSetting.nPart)
      if 0 == nPreviewEffectFashionId then
        if 0 == nWearingEffectFashionId or not Lib:Is_include(nWearingEffectFashionId, tbSetting.tbFashion2EffectIds) then
          Fashion:ClearPartEffectPreview(tbSetting.nPart)
          self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex] = 0
        end
      elseif not Lib:Is_include(nPreviewEffectFashionId, tbSetting.tbFashion2EffectIds) then
        Fashion:ClearPartEffectPreview(tbSetting.nPart)
        self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex] = 0
      end
    elseif tbSetting.nAddEnhanceEffect > 0 and nPreviewEffectResId ~= Item.EquipEnhance:GetEffectResId(me, tbSetting.nAddEnhanceEffect) then
      Fashion:ClearPartEffectPreview(tbSetting.nPart)
      self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex] = 0
      LogErr("UIClosetSingle _HandleClickOnOtherFashion onClick the fashion nAddEnhanceEffect > 0 but preview effect is not enhance effect")
    end
  end
end

function tbWnd:_HandleRepeatClickFashion(nFashionId)
  self.nSelectIndex = nil
  self:ClearSelectFashionInfoShow()
  self.tbSelectPart2ShopItemId[self.nSelectPartIndex] = 0
  self:SetSelectPartIcon(self.nSelectPartIndex, nil)
  local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
  if tbSetting.nLayer ~= Npc.NpcPartLayerDef.npc_part_layer_effect then
    Fashion:ResetPartPreview(tbSetting.nPart)
    local nPreviewFashionPartIndex, nPreviewFashionId = self:GetPreviewFashionPartInfo(tbSetting.nPart + 1, tbSetting.nLayer)
    if nPreviewFashionPartIndex and nPreviewFashionId then
      self:HightLightAllSingleElement()
      self:SetSelectPartIcon(nPreviewFashionPartIndex)
      self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex] = 0
      me.CenterMsg(i18n.EasyGet("\229\164\150\232\167\130\228\184\141\229\140\185\233\133\141\239\188\140\229\183\178\233\135\141\230\150\176\229\143\150\230\182\136\231\137\185\230\149\136\233\162\132\232\167\136\233\128\137\229\174\154"))
    end
  end
  Fashion:ResetPartEffectPreview(tbSetting.nPart)
end

function tbWnd:_HandleShopItemShowType(tbShopItem)
  local szMoneyIcon = Currency:GetIcon(tbShopItem.szMoneyType)
  if tbShopItem.nShowType == tbWnd.tbShowTypeDefine.nGotoShopSuitShowType then
    local nSuitId = tbShopItem.tbSuitIds[1]
    local tbSuitInfo = Shop:GetFashionSuit(nSuitId)
    if 2 ~= #tbSuitInfo then
      LogErr("UIClosetSingle. Single shopItem click error. gotoShopSuitShowType can not find correct suitInfo", tbSuitInfo, " suitId:", nSuitId, "Length is ", #tbSuitInfo)
      return
    end
    local tbSuitDiscountInfo = Shop:GetFashionSuitDiscountInfo()
    if Lib:IsEmptyTB(tbSuitDiscountInfo) then
      LogErr("UIClosetSingle _HandleShopItemShowType get FashionSuitDiscountInfo is empty table")
      return
    end
    local tbSuitSetting = Shop:GetFashionSuitSetting(nSuitId)
    if not tbSuitSetting then
      LogErr("UIClosetSingle. Single shopItem click error. can not find suitSetting by suitId:", nSuitId)
      return
    end
    local nDiscountPrice = tbSuitDiscountInfo[nSuitId] or 0
    if nDiscountPrice > 0 then
      self.pPanel:Label_SetText(tbControls.txtCost, nDiscountPrice)
      self.pPanel:Label_SetText(tbControls.txtOrgCost, tbSuitSetting.nPrice)
    else
      self.pPanel:Label_SetText(tbControls.txtOrgCost, tbSuitInfo[1].nPrice + tbSuitInfo[2].nPrice)
      self.pPanel:Label_SetText(tbControls.txtCost, tbSuitSetting.nPrice)
    end
    self.pPanel:SetActive(tbControls.CostDiscounts, true)
    self.pPanel:Sprite_SetSprite(tbControls.imgOrgMoney, szMoneyIcon)
  elseif tbShopItem.nDiscountPrice > 0 then
    self.pPanel:Label_SetText(tbControls.txtOrgCost, tbShopItem.nPrice)
    self.pPanel:Label_SetText(tbControls.txtCost, tbShopItem.nDiscountPrice)
    self.pPanel:Sprite_SetSprite(tbControls.imgOrgMoney, szMoneyIcon)
    self.pPanel:SetActive(tbControls.CostDiscounts, true)
  else
    self.pPanel:SetActive(tbControls.CostDiscounts, false)
    self.pPanel:Label_SetText(tbControls.txtCost, tbShopItem.nPrice)
  end
end

    if tbElement then

function tbWnd:GetPreviewFashionPartInfo(nPartId, nLayer)
  local nPreviewFashionPartIndex = 0
  if nPartId and nPartId > 0 then
    for nIndex, tbPart in pairs(self.tbPartConfig) do
      if tbPart.nPartId == nPartId and tbPart.nLayer ~= nLayer then
        nPreviewFashionPartIndex = nIndex
      end
    end
  end
  local nPreviewShopItemId = self.tbSelectPart2ShopItemId[nPreviewFashionPartIndex]
  if nPreviewShopItemId then
    local tbPriviewShopItem = Shop:GetFashionItem(nPreviewShopItemId)
    if tbPriviewShopItem then
      return nPreviewFashionPartIndex, tbPriviewShopItem.nFashionID
    end
  end
  if nPreviewFashionPartIndex then
    return nPreviewFashionPartIndex, nil
  end
end

function tbWnd:PreviewWear(nFashionId)
  local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
  if not tbSetting or Lib:IsEmptyTB(tbSetting) then
    return
  end
  if not bCanUse then
    return
  end
  if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
    local tbPreviewIdList = Fashion:GetPreviewEffectList()
    if not tbPreviewIdList or not tbPreviewIdList[tbSetting.nPart] then
      return
    end
    if tbPreviewIdList[tbSetting.nPart].nFashionId == nFashionId then
      return
    end
    Fashion:PreviewTryOnEffect(tbSetting.nPart, nFashionId)
  else
    local tbPreviewIdList = Fashion:Preview_GetPreviewList()
    if not tbPreviewIdList or not tbPreviewIdList[tbSetting.nPart] then
      return
    end
    if tbPreviewIdList[tbSetting.nPart].nFashionId == nFashionId then
      return
    end
    Fashion:Preview_TryOnAFashion(tbSetting.nPart, nFashionId)
  end
end

function tbWnd:RefreshModelSwitchStateShow()
  local bState = false
  local bStateAll = false
  if self.nModelState == tbModelState.All then
    Fashion.bShieldHorse = false
    Fashion:Preview_RefreshPreview_AllPart()
    bState = true
  elseif self.nModelState == tbModelState.OnlyPeople then
    Fashion.bShieldHorse = true
    Fashion:Preview_RefreshPreview()
    bStateAll = true
  elseif self.nModelState == tbModelState.OnlyHorse then
    Fashion.bShieldHorse = true
    Fashion:Preview_RefreshPreview()
    bStateAll = true
  end
  if Fashion.bHorse then
    self.pPanel:SetActive(tbControls.stateHorse, bState)
    self.pPanel:SetActive(tbControls.HorseStateAll, bStateAll)
  else
    self.pPanel:SetActive(tbControls.statePeople, bState)
    self.pPanel:SetActive(tbControls.PeopleStateAll, bStateAll)
  end
end

function tbWnd:GetSuitShopIdbySuitId(nSuitId)
  local tbSuit = Shop:GetFashionSuit(nSuitId)
  if tbSuit and not Lib:IsEmptyTB(tbSuit) then
    local nId1 = tbSuit[1].nId
    local nId2 = tbSuit[2].nId
    local tbShopItemExcel1 = Shop:GetFashionShopItemExcelInfo(nId1)
    local tbShopItemExcel2 = Shop:GetFashionShopItemExcelInfo(nId2)
    local tbShopIds1 = tbShopItemExcel1.tbShopIds
    local tbShopIds2 = tbShopItemExcel2.tbShopIds
    for _, nShopId1 in ipairs(tbShopIds1) do
      for _, nShopId2 in ipairs(tbShopIds2) do
        if nShopId1 == nShopId2 then
          return nShopId1
        end
      end
    end
  end
  return 0
end
