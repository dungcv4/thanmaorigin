    if self.nHorseRidingState == tbHorseRidingState.OnlyPeople then

      self:RefreshModelSwitchStateShow()
    elseif self.nHorseRidingState == tbHorseRidingState.HorseAndPeople then
      self.nHorseRidingState = tbHorseRidingState.OnlyPeople
      self:RefreshModelSwitchStateShow()
    end
  end,
  btnModelSwitchHorse = function(self)
    if self.nHorseRidingState == tbHorseRidingState.OnlyHorse then
      self.nHorseRidingState = tbHorseRidingState.HorseAndPeople
      self:RefreshModelSwitchStateShow()
    elseif self.nHorseRidingState == tbHorseRidingState.HorseAndPeople then
      self.nHorseRidingState = tbHorseRidingState.OnlyHorse
      self:RefreshModelSwitchStateShow()
    end
  end,
  btnShop = function(self)
    self:OnClose()
    Ui:CloseWindow("UIBag")
    Ui:OpenWindow("UIClosetShop", 3, false, self.nSelectPartIndex)
  end,
  btnWear = function(self)
    local nSelectFashionId = self.tbPart2SelectFashion[self.nSelectPartIndex] or 0
    local tbFashionSetting, bCanUse = Fashion:GetFashionSetting(me, nSelectFashionId)
    if bCanUse and tbFashionSetting then
      local nPartId = tbFashionSetting.nPart
      if tbFashionSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
        local nWearFashionId = Fashion:GetWearingFashionIdByPart(nPartId)
        if not tbFashionSetting.tbEffect2FashionIds or Lib:IsEmptyTB(tbFashionSetting.tbEffect2FashionIds) then
          LogErr("UIAppearancePanel2 click btnWear error. Can not find tbEffect2FashionIds by effect fashion id:", nSelectFashionId)
          return

        if not nWearFashionId or not Lib:Is_include(nWearFashionId, tbFashionSetting.tbEffect2FashionIds) then
          local bCanWear = false
          local tbHaveFashions = Fashion:GetHaveFashions()
          for _, nFashionId in ipairs(tbFashionSetting.tbEffect2FashionIds) do
            if tbHaveFashions[nFashionId] then
              bCanWear = true
              Fashion:WearFashion(nFashionId)
              break
            end
          end
          if not bCanWear then
            LogErr("UIAppearancePanel2 click btnWear error. Can not find have effect2fashion to wear by effect fashion id:", nSelectFashionId)
            return
          end
        end
      else
        local nWearEffectFashionId = Fashion:GetWearingEffectFashionIdByPart(me, nPartId)
        if Lib:IsEmptyTB(tbFashionSetting.tbFashion2EffectIds) or not Lib:Is_include(nWearEffectFashionId, tbFashionSetting.tbFashion2EffectIds) then
          Fashion:TakeOffEffectFashion(nPartId)
        end
      end
      Fashion:WearFashion(nSelectFashionId)
    else
      me.CenterMsg(i18n.EasyGet("\230\173\164\232\161\163\232\163\179\228\184\141\229\143\175\231\148\168"), true)
    end

      local tbPartInfo = self.tbPartSetting[self.nSelectPartIndex]
      local nPartId = tbPartInfo.nPartId - 1
      local tbFashionSetting = Fashion:GetFashionSetting(me, self.tbPart2SelectFashion[self.nSelectPartIndex] or 0)
      if tbFashionSetting and tbFashionSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_base then
        Fashion:TakeOffFashion(nPartId)
      end
      Fashion:TakeOffEffectFashion(nPartId)
    end

function tbWnd:OnCreate()
  self.tbLeftButtonScrollView = Ui:NewScrollView(self, tbControls.leftItemTemplate)
  self.tbMyClothesScrollView = Ui:NewScrollView(self, tbControls.myClothesTemplate)
  self.tbPromoteScrollView = Ui:NewScrollView(self, tbControls.PromoteTemplate)
  self.objResPreview = self.pPanel:GetObject(tbControls.NpcModelPath)
  self.NpcModel = self.objResPreview:GetComponent(Ui.UINpcModel)
end

function tbWnd:OnOpen()
  Fashion.nTemporarySex = me.nSex
  self:ResetPreview()
  self.bIsInOpen = true
  self.tbPart2SelectFashion = {}
  self.tbPartSetting = Fashion:GetFashionPartConfig()
  self:SetDefaultSelectPartIndex()
  self:RefreshLeftButtonShow()
  self:RefreshTotalCharm()
  self.pPanel:SetActive(tbControls.btnShop, true)
  self.pPanel:SetActive(tbControls.btnAll, false)
  self.pPanel:SetActive(tbControls.btnType, false)
  self.pPanel:SetActive(tbControls.AllDropdown, false)
  self.pPanel:SetActive(tbControls.TypeDropdown, false)
  self.pPanel:SetActive(tbControls.txtCharm, true)
  self.pPanel:SetActive(tbControls.btnCharmValue, true)
  self.pPanel:SetActive(tbControls.btnAttributeLook, true)
  self.pPanel:SetActive(tbControls.ToggleShieldMount, false)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnCharmValue, {
    RedPointMgr.EVENT.FashionPrettyStageAward
  })
end

function tbWnd:OnClose()
  self.bIsInOpen = false
  Fashion:Preview_Destroy()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnCharmValue)
end

function tbWnd:ResetPreview()
  Fashion:Preview_Init(self.objResPreview)
  self:Redirection()
end

function tbWnd:Redirection()
  if Fashion.bHorse then
    Fashion:SetResDir(1)
  else
    Fashion:SetResDir(0)
  end
end

function tbWnd:GotoClosetShop(tbShopItem)
  self:OnClose()
  Ui:CloseWindow("UIBag")
  if tbShopItem then
    local tbItemExcel = Shop:GetFashionShopItemExcelInfo(tbShopItem.nId)
    local nLayer = 0
    if self.nSelectPartIndex and self.nSelectPartIndex > 0 then
      local tbPartSetting = self.tbPartSetting[self.nSelectPartIndex]
      nLayer = tbPartSetting.nLayer
    end
    Ui:OpenWindow("UIClosetShop", 3, true, tbShopItem.tbSetting.nPart + 1, nLayer, tbItemExcel.tbShopIds[1], tbShopItem.nId)
  end
end

function tbWnd:OnFashionDataUpdate()
  if Ui:WindowVisible("UIBag") and self.bIsInOpen then
    self:ResetPreview()
    self:RefreshLeftButtonShow()
    self:RefreshTotalCharm()
  end
end

function tbWnd:SetDefaultSelectPartIndex()
  self.nSelectPartIndex = nil
  for nIndex, tbPart in ipairs(self.tbPartSetting) do
    local nPartId = tbPart.nPartId - 1
    local nFashionId = Fashion:GetWearingFashionIdByPart(nPartId)
    if nFashionId and nFashionId > 0 then
      self.nSelectPartIndex = nIndex
    end
  end
  if not self.nSelectPartIndex then
    self.nSelectPartIndex = 1
  end
end

function tbWnd:RefreshLeftButtonShow()
  self.tbLeftButtonScrollView:Clear()
  for nIndex, tbSetting in ipairs(self.tbPartSetting) do
    while true do
      do
        local nPartId = tbSetting.nPartId - 1
        if nPartId == Npc.NpcResPartsDef.npc_part_horse then
          local tbHorseItem = me.GetEquipByPos(Item.EQUIPPOS_HORSE)
          if not tbHorseItem then
            break
          end
        end
        self.tbLeftButtonScrollView:SetElement(nIndex, function(tbElement)
          local tbPanel = tbElement.pPanel
          tbPanel:Label_SetText("txtClothes", i18n.Parse(tbSetting.szName))
          local nFashionId = 0
          if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
            nFashionId = Fashion:GetWearingEffectFashionIdByPart(me, nPartId)
          else
            nFashionId = Fashion:GetWearingFashionIdByPart(nPartId) or 0
          end
          if nFashionId and nFashionId > 0 then
            tbPanel:SetActive("Empty", false)
            tbPanel:SetActive("UIItemGrid", true)
            local tbFashionSetting = Fashion:GetFashionSetting(me, nFashionId)
            local pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
            Fashion:InitFashionGrid(pGrid, tbFashionSetting)
          else
            tbPanel:SetActive("Empty", true)
            tbPanel:SetActive("UIItemGrid", false)
          end
          
          local function fnPartButtonClick()
            if self.nSelectPartIndex ~= nIndex then
              local bIsHorse = nPartId == Npc.NpcResPartsDef.npc_part_horse
              Fashion.bHorse = bIsHorse
              self.pPanel:SetActive(tbControls.btnModelSwitchHorse, bIsHorse)
              self.pPanel:SetActive(tbControls.btnModelSwitchPeople, not bIsHorse)
              if bIsHorse and (not self.nHorseRidingState or self.nHorseRidingState == tbHorseRidingState.OnlyPeople) then
                self.nHorseRidingState = tbHorseRidingState.OnlyHorse
                self:RefreshModelSwitchStateShow()
              elseif not bIsHorse and (not self.nHorseRidingState or self.nHorseRidingState == tbHorseRidingState.OnlyHorse) then
                self.nHorseRidingState = tbHorseRidingState.OnlyPeople
                self:RefreshModelSwitchStateShow()
              end
              if self.nSelectPartIndex then
                local tbLastElement = self.tbLeftButtonScrollView:GetElement(self.nSelectPartIndex)
                if tbLastElement then
                  tbLastElement.pPanel:SetActive("imgSelected", false)
                end
              end
              self.nSelectPartIndex = nIndex
              self:Redirection()
              self:RefreshMyClothesShow()
              tbPanel:SetActive("imgSelected", true)
            end
          end
          
          Ui:Button_BindEvent(tbPanel, "Button", fnPartButtonClick)
          if nIndex == self.nSelectPartIndex then
            self.nSelectPartIndex = nil
            fnPartButtonClick()
          else
            tbPanel:SetActive("imgSelected", false)
          end
        end)
      end
      break
    end
  end
end

function tbWnd:RefreshMyClothesShow()
  if not self.nSelectPartIndex then
    return
  end
  local tbPart = self.tbPartSetting[self.nSelectPartIndex]
  local nPartId = tbPart.nPartId - 1
  local tbHaveFashions = Fashion:GetHaveFashions()
  for nFashionId, tbFashion in pairs(tbHaveFashions) do
    local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
    if tbSetting and bCanUse then
      tbFashion.tbSetting = tbSetting
    else
      LogErr("UIAppearancePanel2 RefreshMyClothesShow", "Can not find fashion setting by nFashionId", nFashionId)
      return
    end
  end
  local nWearFashionId = Fashion:GetWearingFashionIdByPart(nPartId) or 0
  local nWearEffectFashionId = Fashion:GetWearingEffectFashionIdByPart(me, nPartId)
  local bInEffectPart = tbPart.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect
  local nPartFashionId = 0
  if bInEffectPart then
    local tbPartSetting = self.tbPartSetting[self.nSelectPartIndex]
    local nSelectFashionPartIndex, _ = Fashion:GetPartConfigByPartId(tbPartSetting.nPartId, 0)
    if nSelectFashionPartIndex and nSelectFashionPartIndex > 0 then
      nPartFashionId = self.tbPart2SelectFashion[nSelectFashionPartIndex] or 0
      if 0 == nPartFashionId then
        nPartFashionId = nWearFashionId
      end
    end
  end
  local nFristFashionId = 0
  local nCanTryOnCount = 0
  local tbShowFashion = {}
  for nFashionId, tbFashion in pairs(tbHaveFashions) do
    if tbFashion.tbSetting.nPart == nPartId and tbFashion.tbSetting.nLayer == tbPart.nLayer then
      if bInEffectPart then
        if nPartFashionId > 0 and Lib:Is_include(nPartFashionId, tbFashion.tbSetting.tbEffect2FashionIds) then
          nCanTryOnCount = nCanTryOnCount + 1
          if nWearEffectFashionId == nFashionId then
            nFristFashionId = nFashionId
          else
            table.insert(tbShowFashion, 1, tbFashion)
          end
        else
          table.insert(tbShowFashion, tbFashion)
        end
      elseif not bInEffectPart and nWearFashionId == nFashionId then
        table.insert(tbShowFashion, 1, tbFashion)
      else
        table.insert(tbShowFashion, tbFashion)
      end
    end
  end
  if nFristFashionId > 0 then
    table.insert(tbShowFashion, 1, tbHaveFashions[nFristFashionId])
  end
  local nSelectFashionId = self.tbPart2SelectFashion[self.nSelectPartIndex] or 0
  if 0 == nSelectFashionId then
    if bInEffectPart and nCanTryOnCount > 0 and nFristFashionId == nWearEffectFashionId then
      nSelectFashionId = nWearEffectFashionId
    elseif not bInEffectPart then
      nSelectFashionId = nWearFashionId
    end
  end
  local bEmptyFashions = Lib:IsEmptyTB(tbShowFashion)
  self.pPanel:SetActive(tbControls.empty, bEmptyFashions)
  self.pPanel:SetActive(tbControls.ClothesList, not bEmptyFashions)
  if bEmptyFashions then
    self:RefreshPromoteShopItem()
    self.pPanel:SetActive(tbControls.btnOff, false)
    self.pPanel:SetActive(tbControls.btnWear, false)
    return
  end
  if not nSelectFashionId or nSelectFashionId <= 0 then
    self:ClearSelectFashionInfoShow()
    self.pPanel:SetActive(tbControls.btnOff, false)
    self.pPanel:SetActive(tbControls.btnWear, false)
  elseif nSelectFashionId > 0 then
    if nSelectFashionId == nWearEffectFashionId or nSelectFashionId == nWearFashionId then
      self.pPanel:SetActive(tbControls.btnOff, true)
      self.pPanel:SetActive(tbControls.btnWear, false)
    else
      self.pPanel:SetActive(tbControls.btnOff, false)
      self.pPanel:SetActive(tbControls.btnWear, true)
    end
  end
  self.tbMyClothesScrollView:Reset()
  for nIndex, tbFashion in ipairs(tbShowFashion) do
    local tbSetting = tbFashion.tbSetting
    local nFashionId = tbSetting.nFashionId
    self.tbMyClothesScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      Fashion:InitFashionGrid(tbElement.pGrid, tbSetting)
      local bIsWearFashion = nFashionId == nWearFashionId or nFashionId == nWearEffectFashionId
      if nSelectFashionId == nFashionId then
        self:PreviewWear(nFashionId)
        pPanel:SetActive("imgSelected", true)
        self.nLastSelectMyClothesIndex = nIndex
        if bIsWearFashion then
          self.pPanel:SetActive(tbControls.btnOff, true)
          self.pPanel:SetActive(tbControls.btnWear, false)
        else
          self.pPanel:SetActive(tbControls.btnOff, false)
          self.pPanel:SetActive(tbControls.btnWear, true)
        end
        self.tbPart2SelectFashion[self.nSelectPartIndex] = nFashionId
        self:RefreshClothesInfoShow(tbSetting.szName, tbSetting.nPrettyValue, tbSetting.szDesc, tbFashion.nExpiredTime, nFashionId)
      else
        pPanel:SetActive("imgSelected", false)
      end
      if bIsWearFashion then
        pPanel:SetActive("imgEquipedBg", true)
      else
        pPanel:SetActive("imgEquipedBg", false)
      end
      if tbFashion.nExpiredTime and tbFashion.nExpiredTime > 0 then
        pPanel:SetActive("imgLimitBg", true)
      else
        pPanel:SetActive("imgLimitBg", false)
      end
      if tbPart.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
        if nCanTryOnCount >= nIndex then
          pPanel:Sprite_SetColor("Button", 48, 48, 48, 0)
        else
          pPanel:Sprite_SetColor("Button", 48, 48, 48, 153)
        end
      else
        pPanel:Sprite_SetColor("Button", 48, 48, 48, 0)
      end
      Ui:Button_BindEvent(pPanel, "Button", function()
        if bInEffectPart and nCanTryOnCount < nIndex then
          me.CenterMsg(i18n.EasyGet("\231\137\185\230\149\136\228\184\142\229\164\150\232\167\130\228\184\141\229\140\185\233\133\141"))
          return
        end
        if self.tbPart2SelectFashion[self.nSelectPartIndex] ~= nFashionId then
          if self.nLastSelectMyClothesIndex and self.nLastSelectMyClothesIndex ~= nIndex then
            local lastSelectElement = self.tbMyClothesScrollView:GetElement(self.nLastSelectMyClothesIndex)
            lastSelectElement.pPanel:SetActive("imgSelected", false)
          end
          self.nLastSelectMyClothesIndex = nIndex
          self:PreviewWear(nFashionId)
          self.tbPart2SelectFashion[self.nSelectPartIndex] = nFashionId
          if not bInEffectPart then
            self:_HandleFashionEffectMatch(nFashionId)
          end
          Fashion:IsNeedPopBoxPartShield(tbSetting.nPart)
          pPanel:SetActive("imgSelected", true)
          self.pPanel:SetActive(tbControls.btnOff, bIsWearFashion)
          self.pPanel:SetActive(tbControls.btnWear, not bIsWearFashion)
          self:RefreshClothesInfoShow(tbSetting.szName, tbSetting.nPrettyValue, tbSetting.szDesc, tbFashion.nExpiredTime, nFashionId)
        else
          self.tbPart2SelectFashion[self.nSelectPartIndex] = nil
          self.nLastSelectMyClothesIndex = nil
          self:ClearSelectFashionInfoShow()
          pPanel:SetActive("imgSelected", false)
          self.pPanel:SetActive(tbControls.btnOff, false)
          self.pPanel:SetActive(tbControls.btnWear, false)
          if tbSetting.nLayer ~= Npc.NpcPartLayerDef.npc_part_layer_effect then
            Fashion:ResetPartPreview(tbSetting.nPart)
          end
          Fashion:ResetPartEffectPreview(tbSetting.nPart)
        end
      end)
    end)
  end
end

function tbWnd:_HandleFashionEffectMatch(nFashionId)
  local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
  local tbPreviewEffect = Fashion:GetPreviewEffectList()
  local nEffectPartIndex = Fashion:GetPartConfigByPartId(tbSetting.nPart + 1, Npc.NpcPartLayerDef.npc_part_layer_effect)
  nEffectPartIndex = nEffectPartIndex or 0
  local nWearEffectFashionId = Fashion:GetWearingEffectFashionIdByPart(me, tbSetting.nPart)
  local nPreviewEffectId = self.tbPart2SelectFashion[nEffectPartIndex] or nWearEffectFashionId
  if nPreviewEffectId and not Lib:IsEmptyTB(tbSetting.tbFashion2EffectIds) and Lib:Is_include(nPreviewEffectId, tbSetting.tbFashion2EffectIds) then
    self:PreviewWear(nPreviewEffectId)
  elseif tbSetting.nAddEnhanceEffect > 0 then
    if tbPreviewEffect[tbSetting.nPart].nResId ~= Item.EquipEnhance:GetEffectResId(me, tbSetting.nAddEnhanceEffect) then
      Fashion:ClearPartEffectPreview(tbSetting.nPart)
      self.tbPart2SelectFashion[nEffectPartIndex] = nil
      LogErr("UIAppearancePanel2 RefreshMyClothesShow onClick the fashion nAddEnhanceEffect > 0 but preview effect is not enhance effect")
    end
  else
    Fashion:ClearPartEffectPreview(tbSetting.nPart)
    self.tbPart2SelectFashion[nEffectPartIndex] = nil
  end
end

function tbWnd:RefreshPromoteShopItem()
  local tbPart = self.tbPartSetting[self.nSelectPartIndex]
  local tbStyle = Shop:GetFashiopShopStyleData(self.nSelectPartIndex, 1)
  if not tbStyle or Lib:IsEmptyTB(tbStyle) then
    LogErr("RefreshPromoteShopItem GetShopStyle is nil or empty table.")
    return
  end
  tbStyle = tbStyle[1]
  local nShopId = tbStyle.nShopId
  if tbStyle.tbShopList[1] then
    nShopId = tbStyle.tbShopList[1].nShopId
  end
  local tbRecordIdDict = {}
  local tbPromoteItems = {}
  local tbShopItems = Shop:GetShopItemsToShow(nShopId)
  if not tbShopItems or Lib:IsEmptyTB(tbShopItems) then
    LogErr("UIAppearancePanel2 RefreshPromoteShopItem.", "Get tbShopItem by shopId", nShopId, " is nil")
    return
  end
  for nIndex = #tbShopItems, 1, -1 do
    local tbItem = tbShopItems[nIndex]
    local tbSetting, _ = Fashion:GetFashionSetting(me, tbItem.nFashionID)
    if tbSetting then
      tbItem.tbSetting = tbSetting
      if tbSetting.nLayer ~= tbPart.nLayer then
        table.remove(tbShopItems, nIndex)
      end
    end
  end
  if Lib:IsEmptyTB(tbShopItems) then
    LogWarn("UIAppearancePanel2 RefreshPromoteShopItem.", "The tbShopItems is empty table after filtering the part effects. nShopId", nShopId)
    return
  end
  if tbShopItems and not Lib:IsEmptyTB(tbShopItems) then
    for _, tbItem in ipairs(tbShopItems) do
      if (tbItem.nDiscountPrice and 0 ~= tbItem.nDiscountPrice or tbItem.bShowNewTip) and not tbRecordIdDict[tbItem.nId] then
        tbRecordIdDict[tbItem.nId] = true
        table.insert(tbPromoteItems, tbItem)
        if not (#tbPromoteItems >= 3) then
        end
      end
    end
  end
  if #tbPromoteItems < 3 then
    local tbPromoteSetting = Lib:SplitStr(tbPart.szDefRmdShopItem, "|")
    for _, szId in ipairs(tbPromoteSetting) do
      local nId = tonumber(szId)
      local tbShopItem = Shop:GetFashionItem(nId)
      if tbShopItem and not tbRecordIdDict[nId] then
        tbRecordIdDict[nId] = true
        table.insert(tbPromoteItems, tbShopItem)
        if #tbPromoteItems >= 3 then
          break
        end
      end
    end
  end
  self.tbPromoteScrollView:Clear()
  self.tbPromoteScrollView:Reset()
  for nIndex, tbPromote in ipairs(tbPromoteItems) do
    if nIndex > 3 then
      return
    end
    local tbSetting = tbPromote.tbSetting
    self.tbPromoteScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      Fashion:InitFashionGrid(tbElement.pGrid, tbSetting)
      local pPanel = tbElement.pPanel
      local szMoneyIcon = Currency:GetIcon(tbPromote.szMoneyType)
      if 0 == tbPromote.nShowType then
        pPanel:SetActive("Cost", true)
        pPanel:SetActive("txtComingSoon", false)
        pPanel:Sprite_SetSprite("Cost/imgMoney", szMoneyIcon)
        if tbPromote.nDiscountPrice and tbPromote.nDiscountPrice > 0 then
          pPanel:SetActive("MarkDiscounts", true)
          pPanel:SetActive("Cost/CostDiscounts", true)
          pPanel:Label_SetText("Cost/txtCost", tbPromote.nDiscountPrice)
          pPanel:Label_SetText("Cost/CostDiscounts/txtCost", tbPromote.nPrice)
          pPanel:Label_SetText("MarkDiscounts/txt", i18n.EasyGet("{0}\230\138\152", tbPromote.nDiscount))
        else
          pPanel:SetActive("MarkDiscounts", false)
          pPanel:Label_SetText("Cost/txtCost", tbPromote.nPrice)
        end
      else
        pPanel:SetActive("Cost", false)
        pPanel:SetActive("txtComingSoon", true)
        pPanel:SetActive("MarkDiscounts", false)
        pPanel:Label_SetText("txtComingSoon", tbWnd.tbShowType[tbPromote.nShowType]())
      end
      pPanel:SetActive("MarkNew", tbPromote.bShowNewTip)
      Ui:Button_BindEvent(pPanel, "Button", function()
        self:GotoClosetShop(tbPromote)
      end)
    end)
  end
end

function tbWnd:RefreshClothesInfoShow(szName, nPrettyValue, szDesc, nExpiredTime, nFashionId)
  self.pPanel:Label_SetText(tbControls.txtDesc, i18n.Parse(szDesc))
  self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(szName))
  if nFashionId and Fashion.AttrCfg[nFashionId] then
    local szMagicName = Fashion.AttrCfg[nFashionId][1]
    local tbMagic = {
      table.unpack(Fashion.AttrCfg[nFashionId], 2)
    }
    local szFinalName, szFinalValue = FightSkill:GetMagicDescSplit(szMagicName, tbMagic)
    local szAttribute = string.format("%s : %s", szFinalName, szFinalValue)
    self.pPanel:SetActive(tbControls.txtAttribute, true)
    self.pPanel:Label_SetText(tbControls.txtAttribute, i18n.EasyGet(szAttribute))
  else
    self.pPanel:SetActive(tbControls.txtAttribute, false)
  end
  self.pPanel:SetActive(tbControls.txtTime, false)
  if nExpiredTime and nExpiredTime > 0 then
    local szShowTime = nExpiredTime - GetTime()
    if szShowTime > 0 then
      self.pPanel:SetActive(tbControls.txtTime, true)
      self.pPanel:Label_SetText(tbControls.txtTime, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180:{0}", Lib:TimeDesc2(szShowTime)))
    end
    self.pPanel:SetActive(tbControls.txtPretty, false)
  else
    local szPretty = nPrettyValue and i18n.EasyGet("\233\173\133\229\138\155\229\128\188+{0}", nPrettyValue) or ""
    if nPrettyValue and nPrettyValue > 0 then
      self.pPanel:SetActive(tbControls.txtPretty, true)
      self.pPanel:Label_SetText(tbControls.txtPretty, szPretty)
    else
      self.pPanel:SetActive(tbControls.txtPretty, false)
    end
  end
end

function tbWnd:PreviewWear(nFashionId)
  local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
  if not tbSetting or Lib:IsEmptyTB(tbSetting) then
    return
  end
  if not bCanUse then
    return
  end
  if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
    local tbPreviewIdList = Fashion:GetPreviewEffectList()
    if not tbPreviewIdList or not tbPreviewIdList[tbSetting.nPart] then
      return
    end
    if tbPreviewIdList[tbSetting.nPart].nFashionId == nFashionId then
      return
    end
    Fashion:PreviewTryOnEffect(tbSetting.nPart, nFashionId)
  else
    local tbPreviewIdList = Fashion:Preview_GetPreviewList()
    if not tbPreviewIdList or not tbPreviewIdList[tbSetting.nPart] then
      return
    end
    if tbPreviewIdList[tbSetting.nPart].nFashionId == nFashionId then
      return
    end
    Fashion:Preview_TryOnAFashion(tbSetting.nPart, nFashionId)
  end
end

function tbWnd:RefreshModelSwitchStateShow()
  local bState = false
  local bStateAll = false
  if self.nHorseRidingState == tbHorseRidingState.HorseAndPeople then
    Fashion.bShieldHorse = false
    Fashion:Preview_RefreshPreview_AllPart()
    bState = true
  elseif self.nHorseRidingState == tbHorseRidingState.OnlyPeople then
    Fashion.bShieldHorse = true
    Fashion:Preview_RefreshPreview()
    bStateAll = true
  elseif self.nHorseRidingState == tbHorseRidingState.OnlyHorse then
    Fashion.bShieldHorse = true
    Fashion:Preview_RefreshPreview()
    bStateAll = true
  end
  if Fashion.bHorse then
    self.pPanel:SetActive(tbControls.stateHorse, bState)
    self.pPanel:SetActive(tbControls.HorseStateAll, bStateAll)
  else
    self.pPanel:SetActive(tbControls.statePeople, bState)
    self.pPanel:SetActive(tbControls.PeopleStateAll, bStateAll)
  end
end
