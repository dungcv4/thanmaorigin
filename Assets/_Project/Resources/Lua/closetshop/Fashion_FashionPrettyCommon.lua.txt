FashionPretty.SaveGroup = GetGroupKey("ReceiveFashionPrettyStageAward")

function FashionPretty:LoadPrettyConfig()
  local szStageAwardPath = "Setting/Appearance/AppearancePrettyRank/StageAward.tab"
  local szRankAwardPath = "Setting/Appearance/AppearancePrettyRank/RankingAward.tab"
  self.tbRankAward = Lib:EasyLoadTabFile(szRankAwardPath, nil, nil, nil, false)
  self.tbStageAward = Lib:EasyLoadTabFile(szStageAwardPath, nil, nil, nil, false)
  table.sort(self.tbRankAward, function(a, b)
    return a.nRank < b.nRank
  end)
  table.sort(self.tbStageAward, function(a, b)
    return a.nStage < b.nStage
  end)
  self:InitAwardKey()
  Validator:Register(self.CheckRankAward, self)
  Validator:Register(self.CheckStageAward, self)
end

function FashionPretty:InitAwardKey()
  for _, tbInfo in ipairs(self.tbRankAward) do
    tbInfo.nAwardKey = tbInfo.nAwardKey or tbInfo.nShowAwardKey
    tbInfo.nShowAwardKey = tbInfo.nShowAwardKey or tbInfo.nAwardKey
  end
  for _, tbInfo in ipairs(self.tbStageAward) do
    tbInfo.nAwardKey = tbInfo.nAwardKey or tbInfo.nShowAwardKey
    tbInfo.nShowAwardKey = tbInfo.nShowAwardKey or tbInfo.nAwardKey
  end
end

function FashionPretty:CheckRankAward()
  local tbRecord = {}
  for _, tbAward in ipairs(self.tbRankAward) do
    if tbRecord[tbAward.nRank] then
      LogErr("FashionPretty:CheckRankAward error. RankingAward have duplicate ranking config", tbAward.nRank)
      return false
    end
    tbRecord[tbAward.nRank] = true
  end
  return true
end

function FashionPretty:CheckStageAward()
  local tbRecord = {}
  for _, tbAward in ipairs(self.tbStageAward) do
    if tbRecord[tbAward.nStage] then
      LogErr("FashionPretty:CheckStageAward error. StageAward have duplicate stage config", tbAward.nStage)
      return false
    end
    tbRecord[tbAward.nStage] = true
  end
  return true
end

FashionPretty:LoadPrettyConfig()

function FashionPretty:CheckCanReceiveStageAward(pPlayer, nStage, bCenterMsg)
  if nStage <= 0 or nStage > #self.tbStageAward then
    LogErr("FashionPretty:CheckCanReceiveStageAward error. nStage out of the border", nStage, #self.tbStageAward)
    return false
  end
  local tbStageAward = self.tbStageAward[nStage]
  if not tbStageAward or Lib:IsEmptyTB(tbStageAward) then
    LogErr("FashionPretty:CheckCanReceiveStageAward error. Get tbStageAward by nStage is nil.", nStage)
    return false
  end
  local nReceive = pPlayer.GetUserValue(self.SaveGroup, nStage)
  if 1 == nReceive then
    if bCenterMsg then
      pPlayer.CenterMsg(i18n.EasyGet("\230\130\168\229\183\178\233\162\134\229\143\150\229\165\150\229\138\177"))
    end
    return false
  end
  local nTotalPretty = Fashion:GetPrettyValue(pPlayer)
  if nTotalPretty < tbStageAward.nStage then
    if bCenterMsg then
      pPlayer.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\233\173\133\229\138\155\229\128\188\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\233\162\134\229\143\150"))
    end
    return false
  end
  return true
end
