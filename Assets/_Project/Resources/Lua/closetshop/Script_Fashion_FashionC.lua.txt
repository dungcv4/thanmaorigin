Fashion.tbTimeOutIndexs = {}

function Fashion:OnSyncFashionData(tbData)
  self:ClearData()
  self.tbFashionClientData = tbData
  local tbData = self:GetPlayerFashionData(me)
  local nCurTime = GetTime()
  for nFashionId, tbFashionInfo in pairs(tbData.tbFashionHaveList) do
    if tbFashionInfo.nExpiredTime and tbFashionInfo.nExpiredTime > 0 then
      if nCurTime >= tbFashionInfo.nExpiredTime then
        self:NotifyFashionTimeout(nFashionId)
      else
        self.tbTimeOutIndexs[nFashionId] = Timer:Register((tbFashionInfo.nExpiredTime - nCurTime) * Env.GAME_FPS, self.NotifyFashionTimeout, self, nFashionId)
      end
    elseif tbFashionInfo.szExpiredDate and not Lib:IsEmptyStr(tbFashionInfo.szExpiredDate) then
      local nEndTime = Lib:ParseDateTime(tbFashionInfo.szExpiredDate)
      if nCurTime >= nEndTime then
        self:NotifyFashionTimeout(nFashionId)
      else
        self.tbTimeOutIndexs[nFashionId] = Timer:Register((nEndTime - nCurTime) * Env.GAME_FPS, self.NotifyFashionTimeout, self, nFashionId)
      end
    end
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_FASHION_DATA_CHANGE, true)
end

function Fashion:OnFashionTimeout(nFashionId)
  local tbSetting = self:GetFashionSetting(me, nFashionId)
  if not tbSetting then
    me.CenterMsg(i18n.EasyGet("\230\173\164\229\164\150\232\163\133\228\184\141\229\173\152\229\156\168"), true)
    return
  end
  local tbData = self:GetPlayerFashionData(me)
  if not tbData.tbFashionHaveList[nFashionId] then
    me.CenterMsg(i18n.EasyGet("\228\190\160\229\163\171\230\178\161\230\156\137\232\191\153\228\187\182\229\164\150\232\163\133"), true)
    return
  end
  tbData.tbFashionHaveList[nFashionId] = nil
  if self.tbTimeOutIndexs[nFashionId] then
    Timer:Close(self.tbTimeOutIndexs[nFashionId])
    self.tbTimeOutIndexs[nFashionId] = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_FASHION_DATA_CHANGE)
  me.CenterMsg(i18n.EasyGet("{0} \229\183\178\232\191\135\230\156\159", i18n.Parse(tbSetting.szName)))

function Fashion:Preview_Init(objResPreview)
  self.objResPreview = objResPreview
  self.comResPreview = objResPreview:GetComponent(Ui.UINpcModel)
  self.comResPreview:Release()
  self.nCurDir = 0
  self.tbPickingEffects = {}
  self.tbPickingFashions = {}
  self.tbPickingHideFashions = {}
  for nPart = Npc.NpcResPartsDef.npc_part_body, Npc.NpcResPartsDef.npc_part_wind do
    self.tbPickingEffects[nPart] = {}
    self.tbPickingFashions[nPart] = {}
    local nWearFashionId = self:GetWearingFashionIdByPart(nPart) or 0
    local bIsHide = self:IsFeatureHide(me, nPart)
    if bIsHide then
      local nResId = self:GetWearingResIdByPart(nPart) or self:GetEquipResIdByPart(nPart) or 0
      self.tbPickingHideFashions[nPart] = {
        nFashionId = nWearFashionId,
        nResId = nResId > 0 and nResId or self:GetNpcDefaultFeature(nPart),
        nEffectFashionId = self:GetWearingEffectFashionIdByPart(me, nPart) or 0,
        nEffectResId = self:GetEffectResId(nPart, nWearFashionId)
      }
      self.tbPickingEffects[nPart].nResId = 0
      self.tbPickingEffects[nPart].nFashionId = 0
      self.tbPickingFashions[nPart].nResId = 0
      self.tbPickingFashions[nPart].nFashionId = 0
    else
      self.tbPickingHideFashions[nPart] = {}
      self.tbPickingFashions[nPart].nFashionId = nWearFashionId
      local nResId = self:GetWearingResIdByPart(nPart) or 0
      self.tbPickingFashions[nPart].nResId = nResId > 0 and nResId or self:GetNpcDefaultFeature(nPart)
      self.tbPickingEffects[nPart].nFashionId = self:GetWearingEffectFashionIdByPart(me, nPart) or 0
      self.tbPickingEffects[nPart].nResId = self:GetEffectResId(nPart, nWearFashionId)
    end
  end
  if self.bShieldHorse then
    self:Preview_RefreshPreview()
  else
    self:Preview_RefreshPreview_AllPart()
  end
end

function Fashion:Preview_TryOnAFashion(nPart, nFashionId)
  local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
  if not tbSetting or not bCanUse then
    me.CenterMsg(i18n.EasyGet("\230\173\164\229\164\150\232\163\133\228\184\141\229\143\175\231\148\168"), true)
    return
  end
  if Lib:IsEmptyTB(self.tbPickingHideFashions[nPart]) then
    self.tbPickingFashions[nPart].nFashionId = nFashionId
    self.tbPickingFashions[nPart].nResId = tbSetting.nResId
  else
    self.tbPickingHideFashions[nPart].nFashionId = nFashionId
    self.tbPickingHideFashions[nPart].nResId = tbSetting.nResId
  end
  local bCanShowBindEffect = false
  if nPart == Npc.NpcResPartsDef.npc_part_weapon then
    local nPreviewEffectId = self.tbPickingEffects[nPart] and self.tbPickingEffects[nPart].nFashionId or 0
    if nPreviewEffectId > 0 then
      if not Lib:Is_include(nPreviewEffectId, tbSetting.tbFashion2EffectIds) then
        bCanShowBindEffect = true
      end
    else
      bCanShowBindEffect = true
    end
  end
  if bCanShowBindEffect and tbSetting.nAddEnhanceEffect > 0 then
    if Lib:IsEmptyTB(self.tbPickingHideFashions[nPart]) then
      self.tbPickingEffects[nPart].nFashionId = 0
      self.tbPickingEffects[nPart].nResId = Item.EquipEnhance:GetEffectResId(me, tbSetting.nAddEnhanceEffect)
    else
      self.tbPickingHideFashions[nPart].nEffectFashionId = 0
      self.tbPickingHideFashions[nPart].nEffectResId = Item.EquipEnhance:GetEffectResId(me, tbSetting.nAddEnhanceEffect)
    end
  end
  if self.bShieldHorse then
    self:Preview_RefreshPreview()
  else
    self:Preview_RefreshPreview_AllPart()
  end
  return true
end

function Fashion:GetEffectResId(nPart, nFashionId)
  local nEffectResId = 0
  local tbSetting = self:GetFashionSetting(me, nFashionId)
  if tbSetting then
    local nEffectFashionId = self:GetWearingEffectFashionIdByPart(me, nPart) or 0
    if nEffectFashionId > 0 and Lib:Is_include(nEffectFashionId, tbSetting.tbFashion2EffectIds) then
      nEffectResId = self:GetWearingEffectResIdByPart(nPart)
    elseif nPart == Npc.NpcResPartsDef.npc_part_weapon and tbSetting.nAddEnhanceEffect > 0 then
      nEffectResId = Item.EquipEnhance:GetEffectResId(me, tbSetting.nAddEnhanceEffect)
    end
  end
  return nEffectResId

function Fashion:ResetPartPreview(nPart)
  if not nPart then
    return
  end
  if not self.tbPickingFashions[nPart] then
    return
  end
  local nFashionId = self:GetWearingFashionIdByPart(nPart)
  if self.tbPickingFashions[nPart].nFashionId > 0 and self.tbPickingFashions[nPart].nFashionId == nFashionId then
    return
  end
  local nResId = self:GetWearingResIdByPart(nPart) or 0
  if self:IsFeatureHide(me, nPart) then
    self.tbPickingFashions[nPart].nResId = 0
    self.tbPickingFashions[nPart].nFashionId = 0
    self.tbPickingHideFashions[nPart].nFashionId = nFashionId or 0
    self.tbPickingHideFashions[nPart].nResId = nResId > 0 and nResId or self:GetNpcDefaultFeature(nPart)
  else
    self.tbPickingFashions[nPart].nFashionId = nFashionId or 0
    self.tbPickingFashions[nPart].nResId = nResId > 0 and nResId or self:GetNpcDefaultFeature(nPart)
  end
  if self.bShieldHorse then
    self:Preview_RefreshPreview()
  else
    self:Preview_RefreshPreview_AllPart()
  end
end

function Fashion:ResetPartEffectPreview(nPart)
  if not nPart then
    return
  end
  if not self.tbPickingEffects[nPart] then

    self:Preview_RefreshPreview()

function Fashion:HidePartFashionPreview(nPart, bHide)

    return
  end
  if not self.tbPickingFashions[nPart] then
    return
  end
  if bHide then
    if not Lib:IsEmptyTB(self.tbPickingHideFashions[nPart]) then
      return
    end
    local tbHideFashionInfo = {
      nResId = self.tbPickingFashions[nPart].nResId or 0,
      nFashionId = self.tbPickingFashions[nPart].nFashionId or 0,
      nEffectResId = self.tbPickingEffects[nPart].nResId or 0,
      nEffectFashionId = self.tbPickingEffects[nPart].nFashionId or 0
    }
    self.tbPickingHideFashions[nPart] = tbHideFashionInfo
    self.tbPickingEffects[nPart].nResId = 0
    self.tbPickingEffects[nPart].nFashionId = 0
    self.tbPickingFashions[nPart].nResId = 0
    self.tbPickingFashions[nPart].nFashionId = 0
  else
    local tbHideFashionInfo = self.tbPickingHideFashions[nPart] or {}
    if not Lib:IsEmptyTB(tbHideFashionInfo) then
      self.tbPickingFashions[nPart].nResId = tbHideFashionInfo.nResId
      self.tbPickingFashions[nPart].nFashionId = tbHideFashionInfo.nFashionId
      self.tbPickingEffects[nPart].nResId = tbHideFashionInfo.nEffectResId
      self.tbPickingEffects[nPart].nFashionId = tbHideFashionInfo.nEffectFashionId
    end
    self.tbPickingHideFashions[nPart] = {}
  end
  if self.bShieldHorse then
    self:Preview_RefreshPreview()
  else
    self:Preview_RefreshPreview_AllPart()
  end
end

function Fashion:IsMatchEffect(nFashionId, nEffectFashionId)
  local tbSetting = self:GetFashionSetting(me, nFashionId)
  if not tbSetting then
    return false
  end
  if not tbSetting.tbFashion2EffectIds or #tbSetting.tbFashion2EffectIds <= 0 then
    return false
  end
  local tbEffectSetting = self:GetFashionSetting(me, nEffectFashionId)
  if not tbEffectSetting then
    return false
  end
  if not tbEffectSetting.tbEffect2FashionIds or #tbEffectSetting.tbEffect2FashionIds <= 0 then
    return false
  end
  if Lib:Is_include(nEffectFashionId, tbSetting.tbFashion2EffectIds) then
    return true
  end
  return false
end

function Fashion:GetWearingResIdList()
  local tbEffectResId = {}
  local tbFashionResId = {}
  for nPart = Npc.NpcResPartsDef.npc_part_body, Npc.NpcResPartsDef.npc_res_part_count - 1 do
    local nResId = self:GetWearingResIdByPart(nPart)
    if nResId then
      tbFashionResId[nPart] = nResId
    else
      tbFashionResId[nPart] = self:GetNpcDefaultFeature(nPart)
    end
    local nEffectResId = self:GetWearingEffectResIdByPart(nPart)
    tbEffectResId[nPart] = nEffectResId or 0
  end
  return tbFashionResId, tbEffectResId
end
