function FashionPretty:GetStageAward()
  return self.tbStageAward
end

function FashionPretty:GetRankAward()
  return self.tbRankAward
end

function FashionPretty:HaveReceived(nStage)
  if 1 == me.GetUserValue(self.SaveGroup, nStage) then
    return true
  else
    return false
  end
end

function FashionPretty:GetRankAwardInfoByRank(nRank)
  if not nRank then
    return
  end
  for _, tbInfo in ipairs(self.tbRankAward) do
    if nRank <= tbInfo.nRank then
      return tbInfo
    end
  end
  return
end

function FashionPretty:ReceiveStageRewardsReq(nStage)
  if not self:CheckCanReceiveStageAward(me, nStage, true) then
    return
  end
  RemoteServer.FashionPrettyStageRewards(nStage)
end

function FashionPretty:ReceiveStageRewardsResp(bSuccess)
  if bSuccess then
    self:TryFadeRedPoint()
    EventNotify.OnNotify(EventNotify.emNOTIFY_FASHIONPRETTY_STAGEREWARD)
  end
end

function FashionPretty:TryLightRedPoint()
  for nStage, _ in ipairs(self.tbStageAward) do
    if self:CheckCanReceiveStageAward(me, nStage) and not self:HaveReceived(nStage) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.FashionPrettyStageAward)
      return
    end
  end
  self:TryFadeRedPoint()
end

function FashionPretty:TryFadeRedPoint()
  for nStage, _ in ipairs(self.tbStageAward) do
    if self:CheckCanReceiveStageAward(me, nStage) and not self:HaveReceived(nStage) then
      return
    end
  end
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.FashionPrettyStageAward)
end

function FashionPretty:OnFashionDataChange(bAllData)
  if bAllData then
    if self.nTryLightRedPointTimer then
      Timer:Close(self.nTryLightRedPointTimer)
      self.nTryLightRedPointTimer = nil
    end
    self.nTryLightRedPointTimer = Timer:Register(Env.GAME_FPS * 3, self.TryLightRedPoint, self)
  else
    self:TryLightRedPoint()
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_FASHION_DATA_CHANGE, FashionPretty)
EventNotify:RegistNotify(EventNotify.emNOTIFY_FASHION_DATA_CHANGE, FashionPretty.OnFashionDataChange, FashionPretty)
