tbWnd.tbOnToggleClick = {
  toggleGender = function(self, isOn)
    if isOn then
      Fashion.nTemporarySex = 1
    else
      Fashion.nTemporarySex = 0
    end
    if Fashion.bShieldHorse then
      Fashion:Preview_RefreshPreview()
    else
      Fashion:Preview_RefreshPreview_AllPart()
    end
    self:RefreshSuitScrollView(self.nSelectShopId)
  end

function tbWnd:OnCreate()
  self.pPanel:Dropdown_BindEvent(tbControls.AllDropdown, function(value)
    self:OnSelectAllDropdown(value)
  end)
  self.pPanel:Dropdown_BindEvent(tbControls.TypeDropdown, function(value)
    self:OnSelectTypeDropdown(value)
  end)
  self.tbSuitScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(tbControls.btnGift, false)
  Fashion.bHorse = false
  Fashion.bShieldHorse = TreasureShop
  self.bIsBuying = false
  self.tbSelectIds = {}
  self.tbShopStyle = Shop:GetFashiopShopStyleData(0, self.nShopType)
  self:ResetPreview()
  self.nModelState = self.tbModelState.OnlyPeople
  self:RefreshStyleData()
  self.pPanel:SetActive(tbControls.btnGift, false)
end

function tbWnd:OnClose()
  Fashion.nTemporarySex = me.nSex
  Fashion:Preview_Destroy()
  self.nLastAllDropDownSelectIndex = nil
  self.nLastTypeDropDownSelectIndex = nil
end

function tbWnd:OnDestroy()
  self.tbSuitScrollView:Clear()
end

function tbWnd:UpdateView()
  if self.nSelectShopId then
    self:RefreshSuitScrollView()
  else
    self:RefreshStyleData()
  end
end

function tbWnd:RefreshFixSelectShop(nSelectShopId, nSelectSuitId)
  self.nFixShopId = nSelectShopId
  self.nSelectSuitId = nSelectSuitId
  self:RefreshStyleData()
end

function tbWnd:FashionBuyResult(bSuccess)
  self.bIsBuying = false
  if bSuccess and self.nSelectShopId then
    self:RefreshSuitScrollView(self.nSelectShopId)
  end
end

function tbWnd:RefreshStyleData()
  local nLength = #self.tbShopStyle
  if 1 == nLength then
    self:RefreshAllStyleShow()
    self.pPanel:SetActive(tbControls.btnType, false)
    self.pPanel:SetActive(tbControls.TypeDropdown, false)
    self.pPanel:SetActive(tbControls.btnTypeHide, false)
  elseif 2 == nLength then
    self:RefreshAllStyleShow()
    self:RefreshTypeStyleShow()
  end
end

function tbWnd:RefreshAllStyleShow()
  local tbType = self.tbShopStyle[1]
  local tbStyles = tbType.tbShopList
  if 1 == #tbStyles then
    self.pPanel:SetActive(tbControls.btnAll, true)
    self.pPanel:SetActive(tbControls.AllDropdown, false)
    self.pPanel:Label_SetText(tbControls.AllLabel, i18n.Parse(tbType.szTypeName))
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnAll, true)
    if self.nFixShopId then
      self:RefreshSuitScrollView(self.nFixShopId)
      self.nFixShopId = nil
    else
      self:RefreshSuitScrollView(tbType.nShopId)
    end
  elseif #tbStyles > 1 then
    self.pPanel:SetActive(tbControls.btnAll, false)
    self.pPanel:SetActive(tbControls.AllDropdown, true)
    local Optations = {}
    for _, tbStyle in ipairs(tbStyles) do
      table.insert(Optations, i18n.Parse(tbStyle.szStyle))
    end
    self.pPanel:SetActive(tbControls.TypeimgSelect, true)
    self.pPanel:Label_SetText(tbControls.txtAllHideName, i18n.Parse(tbType.szTypeName))
    self.pPanel:Dropdown_ResetOptions(tbControls.AllDropdown, Optations)
    if self.nFixShopId then
      for nIndex, tbStyle in ipairs(tbStyles) do
        if tbStyle.nShopId == self.nFixShopId then
          if 1 == nIndex then
            self:RefreshSuitScrollView(self.nFixShopId)
            break
          end
          self.pPanel:Dropdown_SetValue(tbControls.AllDropdown, nIndex - 1)
          break
        end
      end
      self.nFixShopId = nil
    else
      self:RefreshSuitScrollView(tbStyles[1].nShopId)
    end
  end
  self.pPanel:SetActive(tbControls.btnAllHide, false)
end

function tbWnd:RefreshTypeStyleShow()
  local tbType = self.tbShopStyle[2]
  local tbStyles = tbType.tbShopList
  if 1 == #tbStyles then
    self.pPanel:SetActive(tbControls.btnType, true)
    self.pPanel:SetActive(tbControls.TypeDropdown, false)
    self.pPanel:Label_SetText(tbControls.TypeLabel, i18n.Parse(tbType.szTypeName))
  elseif #tbStyles > 1 then
    self.pPanel:SetActive(tbControls.btnType, false)
    self.pPanel:SetActive(tbControls.TypeDropdown, true)
    local Optations = {}
    for _, tbStyle in ipairs(tbStyles) do
      table.insert(Optations, i18n.Parse(tbStyle.szStyle))
    end
    self.pPanel:SetActive(tbControls.btnTypeHide, true)
    self.pPanel:Label_SetText(tbControls.txtTypeHideName, i18n.Parse(tbType.szTypeName))
    self.pPanel:Dropdown_ResetOptions(tbControls.TypeDropdown, Optations)
  end
  self.pPanel:ButtonGroup_SetSelect(tbControls.btnType, false)
end

function tbWnd:RefreshSuitScrollView(nShopId)
  if not nShopId then
    return
  end
  local tbSuits = Shop:GetFashionShopSuits(nShopId)
  if not tbSuits or 0 == #tbSuits then
    LogErr("RefreshSuitScrollView GetFashionShopSuits is empty")
    return
  end
  self:RefreshModelSwitchShow(nShopId)
  self.tbSelectIds = {}
  self.nSelectIndex = nil
  self.nSelectShopId = nShopId
  self:ClearSelectFashionInfoShow()
  self.tbSuitScrollView:Reset()
  local tbHave = Fashion:GetHaveFashions()
  for nIndex, tbSuit in ipairs(tbSuits) do
    self.tbSuitScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local bHaveEffect = false
      local tbFashions = tbSuit.tbItemList
      for nId, tbItem in pairs(tbFashions) do
        local nFashionId = tbItem.nFashionID
        local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
        if tbSetting then
          tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid" .. nId)
          Fashion:InitFashionGrid(tbElement.pGrid, tbSetting)
          if tbHave[nFashionId] then
            pPanel:SetActive("imgHaveBg" .. nId, true)
          else
            pPanel:SetActive("imgHaveBg" .. nId, false)
          end
          tbItem.tbSetting = tbSetting
          if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
            bHaveEffect = true
          end
        end
      end
      if tbSuit.nShowType and self.tbShowType[tbSuit.nShowType] then
        pPanel:SetActive("Cost", false)
        pPanel:SetActive("MarkDiscounts", false)
        pPanel:SetActive("txtComingSoon", true)
        pPanel:Label_SetText("txtComingSoon", self.tbShowType[tbSuit.nShowType]())
      else
        local nAllPrice = tbFashions[1].nPrice + tbFashions[2].nPrice
        local szMoneyIcon = Currency:GetIcon(tbSuit.szMoneyType)
        pPanel:Sprite_SetSprite("Cost/imgMoney", szMoneyIcon)
        pPanel:SetActive("txtComingSoon", false)
        if tbSuit.nDiscountPrice > 0 then
          pPanel:SetActive("MarkDiscounts", true)
          pPanel:Label_SetText("Cost/txtCost", tbSuit.nDiscountPrice)
          pPanel:Label_SetText("Cost/CostDiscounts/txtCost", tbSuit.nPrice)
          pPanel:Label_SetText("MarkDiscounts/txt", i18n.GetDiscountText(tbSuit.nSuitDiscount))
        else
          pPanel:SetActive("MarkDiscounts", false)
          pPanel:Label_SetText("Cost/txtCost", tbSuit.nPrice)
          pPanel:Label_SetText("Cost/CostDiscounts/txtCost", nAllPrice)
        end
      end
      pPanel:SetActive("Cost/CostDiscounts", true)
      pPanel:SetActive("MarkNew", tbSuit.bShowNewTip)
      
      local function fnClickFunc()
        if self.nSelectIndex ~= nIndex then
          if tbSuit.nShowType and self.tbShowType[tbSuit.nShowType] then
            self.pPanel:SetActive(tbControls.btnBuy, false)
            self.pPanel:SetActive(tbControls.shopCost, false)
          else
            self.pPanel:SetActive(tbControls.btnBuy, true)
            self.pPanel:SetActive(tbControls.shopCost, true)
          end
          pPanel:SetActive("imgSelected", true)
          if self.nSelectIndex then
            local pLastElement = self.tbSuitScrollView:GetElement(self.nSelectIndex)
            pLastElement.pPanel:SetActive("imgSelected", false)
          end
          self.nSelectIndex = nIndex
          self.tbSelectIds = {
            tbFashions[1].nId,
            tbFashions[2].nId
          }
          self.pPanel:SetActive(tbControls.suitAttribute, true)
          self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(tbSuit.szSuitName))
          self.pPanel:Label_SetText(tbControls.txtDesc, i18n.Parse(tbSuit.szSuitDesc))
          for nId, tbFashion in ipairs(tbFashions) do
            local nFashionId = tbFashion.nFashionID
            if nFashionId and Fashion.AttrCfg[nFashionId] then
              local szMagicName = Fashion.AttrCfg[nFashionId][1]
              local tbMagic = {
                table.unpack(Fashion.AttrCfg[nFashionId], 2)
              }
              local szFinalName, szFinalValue = FightSkill:GetMagicDescSplit(szMagicName, tbMagic)
              local szAttribute = string.format("%s%s", szFinalName, szFinalValue)
              self.pPanel:SetActive(tbControls["txtAttribute" .. nId], true)
              self.pPanel:Label_SetText(tbControls["txtAttribute" .. nId], i18n.EasyGet(szAttribute))
            else
              self.pPanel:SetActive(tbControls["txtAttribute" .. nId], false)
            end
            local tbSetting = tbFashion.tbSetting
            self.pPanel:Label_SetText(tbControls["txtPretty" .. nId], i18n.EasyGet("\233\173\133\229\138\155\229\128\188+{0}", tbSetting.nPrettyValue))
            local _, tbPartSetting = Fashion:GetPartConfigByPartId(tbSetting.nPart + 1, tbSetting.nLayer)
            if tbPartSetting and not Lib:IsEmptyTB(tbPartSetting) then
              if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
                self.pPanel:Label_SetText(tbControls["txtAttributeTitle" .. nId], i18n.EasyGet("\231\137\185\230\149\136 : "))
              else
                self.pPanel:Label_SetText(tbControls["txtAttributeTitle" .. nId], i18n.EasyGet("{0} : ", i18n.Parse(tbPartSetting.szName)))
              end
            end
            self:PreviewWear(nFashionId)
          end
          if tbSuit.nDiscountPrice > 0 then
            self.pPanel:Label_SetText(tbControls.txtOrgCost, tbSuit.nPrice)
            self.pPanel:Label_SetText(tbControls.txtCost, tbSuit.nDiscountPrice)
          else
            self.pPanel:Label_SetText(tbControls.txtOrgCost, tbFashions[1].nPrice + tbFashions[2].nPrice)
            self.pPanel:Label_SetText(tbControls.txtCost, tbSuit.nPrice)
          end
          local szMoneyIcon = Currency:GetIcon(tbSuit.szMoneyType)
          self.pPanel:Sprite_SetSprite(tbControls.imgMoney, szMoneyIcon)
          self.pPanel:Sprite_SetSprite(tbControls.imgOrgMoney, szMoneyIcon)
        else
          self.nSelectIndex = nil
          self:ClearSelectFashionInfoShow()
          pPanel:SetActive("imgSelected", false)
          if bHaveEffect then
            Fashion:ResetPartPreview(Npc.NpcResPartsDef.npc_part_weapon)
            Fashion:ResetPartEffectPreview(Npc.NpcResPartsDef.npc_part_weapon)
          else
            Fashion:ResetPartPreview(Npc.NpcResPartsDef.npc_part_head)
            Fashion:ResetPartPreview(Npc.NpcResPartsDef.npc_part_body)
          end
        end
      end
      
      if tbSuit.nSuitId == self.nSelectSuitId then
        fnClickFunc()
        self.nSelectSuitId = nil
      else
        pPanel:SetActive("imgSelected", false)
      end
      Ui:Button_BindEvent(pPanel, "Button", fnClickFunc)
    end)
  end
  if self.nSelectIndex and self.tbSuitScrollView:GetCount() >= self.nSelectIndex then
    self.tbSuitScrollView:ScrollToIndex(self.nSelectIndex)
  end
end

function tbWnd:ClearSelectFashionInfoShow()
  self.pPanel:Label_SetText(tbControls.txtName, "")
  self.pPanel:Label_SetText(tbControls.txtDesc, i18n.EasyGet("\232\175\183\233\128\137\229\143\150\229\164\150\232\167\130\230\159\165\231\156\139\230\143\143\232\191\176"))
  self.pPanel:Label_SetText(tbControls.txtPretty1, "")
  self.pPanel:Label_SetText(tbControls.txtPretty2, "")
  self.pPanel:SetActive(tbControls.txtAttribute1, false)
  self.pPanel:SetActive(tbControls.txtAttribute2, false)
  self.pPanel:SetActive(tbControls.suitAttribute, false)
  self.pPanel:SetActive(tbControls.btnBuy, false)
  self.pPanel:SetActive(tbControls.shopCost, false)
end

function tbWnd:OnSelectAllDropdown(value)
  if value == self.nLastAllDropDownSelectIndex and not self.bClickBtnAll then
    return
  end
  self.nLastAllDropDownSelectIndex = value
  local nIndex = value + 1
  local tbTypes = self.tbShopStyle[1]
  if not tbTypes or nIndex > #tbTypes.tbShopList then
    LogErr("UIClosetSuit:OnSelectAllDropdown the length of tbShopList ", #tbTypes.tbShopList, " less than value + 1", nIndex)
    return
  end
  local tbStyle = tbTypes.tbShopList[nIndex]
  local nShopId = tbStyle.nShopId
  self:RefreshSuitScrollView(nShopId)
end

function tbWnd:OnSelectTypeDropdown(value)
  if value == self.nLastTypeDropDownSelectIndex and not self.bClickBtnType then
    return
  end
  self.nLastTypeDropDownSelectIndex = value
  local nIndex = value + 1
  local tbTypes = self.tbShopStyle[2]
  if not tbTypes or nIndex > #tbTypes.tbShopList then
    LogErr("UIClosetSuit:OnSelectTypeDropdown the length of tbShopList ", #tbTypes.tbShopList, " less than value  + 1", nIndex)
    return
  end
  local tbStyle = tbTypes.tbShopList[nIndex]
  local nShopId = tbStyle.nShopId
  self:RefreshSuitScrollView(nShopId)
end

function tbWnd:ResetPreview()
  self.objResPreview = self.pPanel:GetObject(tbControls.NpcModel)
  Fashion:Preview_Init(self.objResPreview)
  self:Redirection()
  Fashion.nTemporarySex = me.nSex
  if 0 == me.nSex then
    self.pPanel:Toggle_SetChecked(tbControls.toggleGender, false)
  else
    self.pPanel:Toggle_SetChecked(tbControls.toggleGender, true)
  end
  Fashion:Preview_RefreshPreview()
end

function tbWnd:Redirection()
  if Fashion.bHorse then
    Fashion:SetResDir(1)
  else
    Fashion:SetResDir(0)
  end
end

function tbWnd:PreviewWear(nFashionId)
  local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
  if not tbSetting or Lib:IsEmptyTB(tbSetting) then
    return
  end
  if not bCanUse then
    return
  end
  if tbSetting.nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
    local tbPreviewIdList = Fashion:GetPreviewEffectList()
    if not tbPreviewIdList or not tbPreviewIdList[tbSetting.nPart] then
      return
    end
    if tbPreviewIdList[tbSetting.nPart].nFashionId == nFashionId then
      return
    end
    Fashion:PreviewTryOnEffect(tbSetting.nPart, nFashionId)
  else
    local tbPreviewIdList = Fashion:Preview_GetPreviewList()
    if not tbPreviewIdList or not tbPreviewIdList[tbSetting.nPart] then
      return
    end
    if tbPreviewIdList[tbSetting.nPart].nFashionId == nFashionId then
      return
    end
    Fashion:Preview_TryOnAFashion(tbSetting.nPart, nFashionId)
  end
end

function tbWnd:RefreshModelSwitchShow(nShopId)
  if not nShopId and nShopId < 0 then
    LogErr("UIClosetSuit. RefreshModelSwitchShow nShopId is vaild")
    return
  end
  local tbShopSetting = Shop:GetFashionShopConfigByShopId(nShopId)
  local bIsHorsePart = tbShopSetting.nPart == self.nHorsePartId
  Fashion.bHorse = bIsHorsePart
  self.pPanel:SetActive(tbControls.btnModelSwitch, not bIsHorsePart)
  self.pPanel:SetActive(tbControls.btnModelSwitchHorseShop, bIsHorsePart)
  if nShopId == self.nAllSuitShopId then
    self.nModelState = self.tbModelState.All
  end
  if self.nModelState == self.tbModelState.All then
    Fashion.bShieldHorse = false
    if bIsHorsePart then
      self.pPanel:SetActive(tbControls.HorseStateAll, false)
      self.pPanel:SetActive(tbControls.HorseStateHorse, true)
    else
      self.pPanel:SetActive(tbControls.PeopleStateAll, false)
      self.pPanel:SetActive(tbControls.PeopleStatePeople, true)
    end
    Fashion:Preview_RefreshPreview_AllPart()
  else
    Fashion.bShieldHorse = true
    if bIsHorsePart and (self.nModelState == self.tbModelState.OnlyPeople or self.nModelState == self.tbModelState.OnlyHorse) then
      self.nModelState = self.tbModelState.OnlyHorse
      self.pPanel:SetActive(tbControls.HorseStateAll, true)
      self.pPanel:SetActive(tbControls.HorseStateHorse, false)
    elseif not bIsHorsePart and (self.nModelState == self.tbModelState.OnlyPeople or self.nModelState == self.tbModelState.OnlyHorse) then
      self.nModelState = self.tbModelState.OnlyPeople
      self.pPanel:SetActive(tbControls.PeopleStateAll, true)
      self.pPanel:SetActive(tbControls.PeopleStatePeople, false)
    end
