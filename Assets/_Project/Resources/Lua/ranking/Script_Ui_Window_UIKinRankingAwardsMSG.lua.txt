local tbWnd = Ui:CreateClass("UIKinRankingAwardsMSG", "UIPanel")
local tbControls = {
  Info = "Awards/Info",
  txtActEndTime = "Awards/Info/txtInfo1",
  txtKinTitle = "Awards/Info/txtInfo2",
  txtSituation = "Awards/Info/txtInfo3",
  txtMyRankValue = "MyRank/txtMyRankValue",
  txtKinRankingInfo1 = "KinRanking/txtInfo1",
  DomainListTitle = "KinRanking/DomainListTitle",
  AwardTemplate = "Awards/imgList/List/Template",
  RankTemplate = "KinRanking/imgList/List/Template",
  Awards = "Awards",
  KinRanking = "KinRanking",
  imgList = "Awards/imgList/List",
  MyRank = "MyRank",
  btnCheck = "btnCheck"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCheck = function(self)
    self:OpenKinRanking()
  end
}

function tbWnd:OpenKinRanking()
  self.tbOwner:Close()
  Ui:OpenWindow("UIRanking")
end

function tbWnd:DoUpdateRank()
  self.pInstance:UpdateTop10RankView()
end

function tbWnd:DoUpdateMyRank()
  self.pInstance:UpdateMyRankInfo()
end

function tbWnd:OnUpdateTop10Rank(tbRankInfo)
  self.KinRankScrollView:Clear()
  if 0 == #tbRankInfo then
    me.CenterMsg(i18n.EasyGet("\230\154\130\230\151\160\229\174\182\230\151\143\228\184\138\230\166\156"))
  else
    for i = 1, #tbRankInfo do
      local tbKinDetail = tbRankInfo[i]
      if tbKinDetail then
        self.KinRankScrollView:SetElement(i, function(tbElement)
          self:SetRankElement(tbElement, i, tbKinDetail)
        end)
      end
    end
  end
end

function tbWnd:OnUpdateMyRank(szRankInfo)
  self.pPanel:SetActive(tbControls.txtMyRankValue, true)
  self.pPanel:Label_SetText(tbControls.txtMyRankValue, i18n.Parse(szRankInfo))
end

function tbWnd:SetRankElement(tbElement, nIndex, tbKinDetail)
  if 1 == nIndex then
    tbElement.pPanel:SetActive("imgRank", true)
    tbElement.pPanel:SetActive("txtRank", false)
    tbElement.pPanel:Label_SetText("txtKin", i18n.EasyGet(tbKinDetail.szKinName))
    tbElement.pPanel:Label_SetText("txtMasterName", i18n.EasyGet(tbKinDetail.szMasterName))
  else
    tbElement.pPanel:SetActive("imgRank", false)
    tbElement.pPanel:SetActive("txtRank", true)
    tbElement.pPanel:Label_SetText("txtKin", i18n.EasyGet(tbKinDetail.szKinName))
    tbElement.pPanel:Label_SetText("txtRank", i18n.EasyGet("{0}", nIndex))
    tbElement.pPanel:Label_SetText("txtMasterName", i18n.EasyGet(tbKinDetail.szMasterName))
  end
end

function tbWnd:OnCreate()
  self.pInstance = ActivityMgr:GetKinRankingActivityInstance()
end

function tbWnd:OnOpen()
  Event:RegisterEvent(Event.emEVENT_KINRANKINGAWARD_UPDATE_RANK, self, "OnUpdateTop10Rank")
  Event:RegisterEvent(Event.emEVENT_KINRANKINGAWARD_UPDATE_MYRANK, self, "OnUpdateMyRank")
  if RedPointMgr:CheckEvent(RedPointMgr.EVENT.KinRankingAward) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.KinRankingAward)
  end
  local bIsLock = self.pInstance:IsLockRank()
  if not bIsLock then
    self.tbAwardsInfo = self.pInstance:GetAwardConfig()
    self.pPanel:SetActive(tbControls.KinRanking, false)
    self.pPanel:SetActive(tbControls.Awards, true)
    self.pPanel:SetActive(tbControls.btnCheck, true)
    self.pPanel:SetActive(tbControls.MyRank, true)
    self:ShowTxtInfo()
    self.AwardsScrollView = Ui:NewScrollView(self, tbControls.AwardTemplate, nil, false, false)
    self:UpdateAwardsView(self.tbAwardsInfo)
    self:DoUpdateMyRank()
  else
    self.pPanel:SetActive(tbControls.Awards, false)
    self.pPanel:SetActive(tbControls.MyRank, false)
    self.pPanel:SetActive(tbControls.KinRanking, true)
    self.pPanel:SetActive(tbControls.btnCheck, false)
    self.pPanel:Label_SetText(tbControls.txtKinRankingInfo1, i18n.EasyGet("\229\174\182\230\151\143\232\175\132\233\128\137\230\180\187\229\138\168\239\188\140\232\175\132\233\128\137\231\187\147\230\158\156:"))
    self.KinRankScrollView = Ui:NewScrollView(self, tbControls.RankTemplate, nil, false, false)
    self:DoUpdateRank()
  end
end

function tbWnd:OnClose()
  Event:UnRegisterEvent(Event.emEVENT_KINRANKINGAWARD_UPDATE_RANK, self)
  Event:UnRegisterEvent(Event.emEVENT_KINRANKINGAWARD_UPDATE_MYRANK, self)
  if self.AwardsScrollView then
    self.AwardsScrollView:Clear()
  end
  if self.KinRankScrollView then
    self.KinRankScrollView:Clear()
  end
end

function tbWnd:UpdateAwardsView(tbAwardsInfo)
  if tbAwardsInfo then
    for i = 1, #tbAwardsInfo do
      self.AwardsScrollView:SetElement(i, function(tbElement)
        self:SetAwardElement(tbElement, tbAwardsInfo[i])
      end)
    end
  end
end

function tbWnd:SetAwardElement(tbElement, tbAward)
  local tbAllAwards = self:GetAwardInfo(tbAward)
  local ret = string.find(tbAward.szRank, "-")
  if not ret then
    tbElement.pPanel:Label_SetText("txtRank", i18n.EasyGet("\231\172\172{0}\229\144\141", tbAward.szRank))
  else
    tbElement.pPanel:Label_SetText("txtRank", i18n.EasyGet("{0}\229\144\141", tbAward.szRank))
  end
  tbElement.itemGrid1 = Ui:NewExistItemGrid(tbElement, "MasterAward/UIItemGrid")
  tbElement.itemGrid2 = Ui:NewExistItemGrid(tbElement, "ManagerAward/UIItemGrid")
  tbElement.itemGrid3 = Ui:NewExistItemGrid(tbElement, "MemberAward/UIItemGrid")
  for i = 1, 3 do
    tbElement["itemGrid" .. i]:SetGenericItem(tbAllAwards[i][1])
    tbElement["itemGrid" .. i]:SetClickEnable(true)
    local szItemName = Item:GetName(tbAllAwards[i][1][2], me)
    if 1 == i then
      tbElement.pPanel:Label_SetText("MasterAward/txtTips", i18n.EasyGet(szItemName))
    elseif 2 == i then
      tbElement.pPanel:Label_SetText("ManagerAward/txtTips", i18n.EasyGet(szItemName))
    elseif 3 == i then
      tbElement.pPanel:Label_SetText("MemberAward/txtTips", i18n.EasyGet(szItemName))
    end
  end
end

function tbWnd:GetAwardInfo(tbAward)
  local tbAllAwards = {}
  for i = 1, 3 do
    local tbAwards = Award:GetAward(tbAward.nAwardId, tbAward["nAwardKey" .. tostring(i)], {})
    table.insert(tbAllAwards, tbAwards)
  end
  return tbAllAwards
end

function tbWnd:ShowTxtInfo()
  self.pPanel:SetActive(tbControls.Info, true)
  local szEndDay, nRemainDay, szSituationDay = self.pInstance:GetTimesInfo()
  self.pPanel:Label_SetText(tbControls.txtActEndTime, i18n.EasyGet(string.format("%s(%s)", szEndDay, nRemainDay)))
  self.pPanel:Label_SetText(tbControls.txtKinTitle, i18n.EasyGet("\231\153\190\229\164\167\229\174\182\230\151\143\232\175\132\233\128\137\239\188\154<color=#efaa02>\227\128\144\231\172\172\228\184\128\229\174\182\230\151\143\227\128\145</color>\227\128\129<color=#bd5385>\227\128\144\229\141\129\229\164\167\229\174\182\230\151\143\227\128\145</color>\227\128\129<color=#734bc7>\227\128\144\229\141\147\232\182\138\229\174\182\230\151\143\227\128\145</color>"))
  self.pPanel:Label_SetText(tbControls.txtSituation, i18n.EasyGet("{0} \228\185\139\229\137\141\232\191\155\229\133\165\229\174\182\230\151\143\231\154\132\230\136\144\229\145\152\229\143\175\233\162\134\229\165\150", szSituationDay))
  self.pPanel:SetActive(tbControls.txtActEndTime, true)
  self.pPanel:SetActive(tbControls.txtKinTitle, true)
  self.pPanel:SetActive(tbControls.txtSituation, true)
end
