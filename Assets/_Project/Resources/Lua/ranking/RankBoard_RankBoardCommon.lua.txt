function RankBoard:Init()
  self.tbSetting = Lib:LoadTabFileEx("Setting/rankboard.tab", "sdsdddsdddddsss", "Key", {
    "Key",
    "ID",
    "Name",
    "Type",
    "Refresh",
    "MaxNum",
    "Sub",
    "NotSave",
    "bShow",
    "bAutoUpdate",
    "SortId",
    "ShowLevel",
    "szTimeFrameOpen",
    "szTimeFrameClose",
    "szShowCallBack"
  })
  self.tbId2Key = {}
  for k, v in pairs(self.tbSetting) do
    self.tbId2Key[v.ID] = k
  end
end

function RankBoard:GetMaxShowNum(szKey)
  local tbInfo = self.tbSetting[szKey]
  return tbInfo and tbInfo.MaxNum or 1
end

function RankBoard:GetKeyById(id)
  return self.tbId2Key[id]
end

RankBoard:Init()
