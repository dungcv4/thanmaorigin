local tbWnd = Ui:CreateWindow("UIRankPromotion")
local tbControls = {
  imgGradeEffect = "node/Succeed_Effect/HongYunDangTou_Effect/00",
  txtAwardValue = "node/Succeed_Effect/HongYunDangTou_Effect/AwardTips/txtAwardValue"
}

function tbWnd:OnCreate()
  self:SetClickEmptyCallback()
end

function tbWnd:OnOpen(nActivityId, nCurGrade)
  self.bShowAll = false
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * 5, self.Close, self)
  self:InitTips(nActivityId, nCurGrade)
end

function tbWnd:InitTips(nActivityId, nCurGrade)
  for nGrade = 1, Lib:CountTB(WuLinHonor.tbGradeGroup) do
    if nGrade == nCurGrade then
      self.pPanel:SetActive(tbControls.imgGradeEffect .. nGrade, true)
    else
      self.pPanel:SetActive(tbControls.imgGradeEffect .. nGrade, false)
    end
  end
  self.pPanel:Label_SetText(tbControls.txtAwardValue, i18n.EasyGet("{0}\230\174\181\228\189\141\230\136\144\229\138\159\230\153\139\229\141\135\232\135\179<color=#{1}>{2}</color>", i18n.Get(WuLinHonor.tbText.tbActivityText[nActivityId]), Ui.tbColor:GetColorHexByDefineName(WuLinHonor.tbEachGradeCfg[nCurGrade].szTextColor), i18n.Get(WuLinHonor.tbEachGradeCfg[nCurGrade].szText)))
end

function tbWnd:OnClose()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
end

function tbWnd:OnClickEmptyClose()
  if self.bShowAll then
    self:Close()
    return
  end
  self.bShowAll = true
end
