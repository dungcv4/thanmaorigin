function tbDisplayFuns_Default:SetTitle()

function tbDisplayFuns_Default:OnCreate()
  self.tbScrollView = Ui:NewScrollView(tbWnd, tbWnd.RankTemplateDefault)
end

function tbDisplayFuns_Default:Reset()
  self.tbScrollView:Clear()
end

function tbDisplayFuns_Default:SetElement(tbElement, tbRecord)
  local pPanel = tbElement.pPanel
  local szKinName = "-"
  local szTongName = ""
  if tbRecord and not Lib:IsEmptyStr(tbRecord.szKinName) then
    szKinName = tbRecord.szKinName
  end
  if tbRecord and not Lib:IsEmptyStr(tbRecord.szTongName) then
    szTongName = tbRecord.szTongName
  end
  local szFaction = Faction:GetName(tbRecord.nFaction)
  if tbRecord.nPosition <= 3 then
    pPanel:SetActive("txtRank", false)
    pPanel:SetActive("UIRankNo", true)
    pPanel:SetActive("UIRankNo/imgNO1", 1 == tbRecord.nPosition)
    pPanel:SetActive("UIRankNo/imgNO2", 2 == tbRecord.nPosition)
    pPanel:SetActive("UIRankNo/imgNO3", 3 == tbRecord.nPosition)
  else
    pPanel:SetActive("UIRankNo", false)
    pPanel:SetActive("txtRank", true)
    pPanel:Label_SetText("txtRank", tbRecord.nPosition)
  end
  Ui:Button_BindEvent(pPanel, "btnName", function()
    if tbRecord.dwUnitID ~= me.dwID then
      if tbRecord.dwTongId and tbRecord.dwTongId > 0 then
        tbRecord.szKinOrTongName = szTongName
      elseif tbRecord.dwKinId and tbRecord.dwKinId > 0 then
        tbRecord.szKinOrTongName = szKinName
      else
        tbRecord.szKinOrTongName = ""
      end
      Ui:OpenWindow("UIRightPopUpMenu", "roleselectFromUI", tbRecord, self, 450, -160)
    end
  end)
  if tbRecord.dwUnitID == me.dwID then
    pPanel:Sprite_SetColor("Main", 255, 255, 255, 255)
  else
    pPanel:Sprite_SetColor("Main", 0, 0, 0, 20)
  end
  pPanel:Label_SetText("txtPlayerName", tbRecord.szName)
  Ui:SetVipLevelEffect(pPanel, tbRecord.nVipLevel, "txtPlayerName")
  pPanel:Label_SetText("txtKinName", szKinName)
  pPanel:Label_SetText("txtValue", tbRecord.nValue)
  local SpFaction = Faction:GetIcon(tbRecord.nFaction)
  pPanel:Sprite_SetSprite("imgPlayerFaction", SpFaction)
end

function tbDisplayFuns_PrettyValue:SetElement(tbElement, tbRecord)
  local pPanel = tbElement.pPanel
  local szKinName = "-"
  local szTongName = ""
  if tbRecord and not Lib:IsEmptyStr(tbRecord.szKinName) then
    szKinName = tbRecord.szKinName
  end
  if tbRecord and not Lib:IsEmptyStr(tbRecord.szTongName) then
    szTongName = tbRecord.szTongName
  end
  if tbRecord.nPosition <= 3 then
    pPanel:SetActive("txtRank", false)
    pPanel:SetActive("UIRankNo", true)
    pPanel:SetActive("UIRankNo/imgNO1", 1 == tbRecord.nPosition)
    pPanel:SetActive("UIRankNo/imgNO2", 2 == tbRecord.nPosition)
    pPanel:SetActive("UIRankNo/imgNO3", 3 == tbRecord.nPosition)
  else
    pPanel:SetActive("UIRankNo", false)
    pPanel:SetActive("txtRank", true)
    pPanel:Label_SetText("txtRank", tbRecord.nPosition)
  end
  Ui:Button_BindEvent(pPanel, "btnName", function()
    if tbRecord.dwUnitID ~= me.dwID then
      if tbRecord.dwTongId and tbRecord.dwTongId > 0 then
        tbRecord.szKinOrTongName = szTongName
      elseif tbRecord.dwKinId and tbRecord.dwKinId > 0 then
        tbRecord.szKinOrTongName = szKinName
      else
        tbRecord.szKinOrTongName = ""
      end
      Ui:OpenWindow("UIRightPopUpMenu", "roleselectFromUI", tbRecord, self, 450, -160)
    end
  end)
  if tbRecord.dwUnitID == me.dwID then
    pPanel:Sprite_SetColor("Main", 255, 255, 255, 255)
  else
    pPanel:Sprite_SetColor("Main", 0, 0, 0, 20)
  end
  pPanel:Label_SetText("txtPlayerName", tbRecord.szName)
  Ui:SetVipLevelEffect(pPanel, tbRecord.nVipLevel, "txtPlayerName")
  pPanel:Label_SetText("txtKinName", szKinName)
  pPanel:Label_SetText("txtValue", tbRecord.nValue)
  local SpFaction = Faction:GetIcon(tbRecord.nFaction)
  pPanel:Sprite_SetSprite("imgPlayerFaction", SpFaction)
  local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(tbRecord.nHonorStarLevel, false)
  local obj = pPanel:GetObject("txtPlayerName/HonorTitle")
  local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
  pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
end

  return true

local function ChkTimeFrame(szTimeFrameOpen, szTimeFrameClose)
  local nNowServerTime = GetTime()
  if not Lib:IsEmptyStr(szTimeFrameOpen) and not Lib:IsEmptyStr(szTimeFrameClose) then
    local nCalcOpenTime = Lib:CalcTimeFrameOpenTime(szTimeFrameOpen)
    local nCalcEndTime = Lib:CalcTimeFrameOpenTime(szTimeFrameClose)
    if nNowServerTime >= nCalcOpenTime and nNowServerTime < nCalcEndTime then
      return true
    end
  elseif not Lib:IsEmptyStr(szTimeFrameOpen) and Lib:IsEmptyStr(szTimeFrameClose) then
    local nCalcOpenTime = Lib:CalcTimeFrameOpenTime(szTimeFrameOpen)
    if nNowServerTime >= nCalcOpenTime then
      return true
    end
  elseif Lib:IsEmptyStr(szTimeFrameOpen) and not Lib:IsEmptyStr(szTimeFrameClose) then
    local nCalcEndTime = Lib:CalcTimeFrameOpenTime(szTimeFrameClose)
    if nNowServerTime < nCalcEndTime then
      return true
    end
  else
    return true
  end
end

local function ChkIsShowCallBack(szFuncName)
  local bSuccess, bResult = false, true
  if not Lib:IsEmptyStr(szFuncName) and RankBoard.FnIsShowCallBack[szFuncName] then
    bSuccess, bResult = Lib:CallBack({
      RankBoard.FnIsShowCallBack[szFuncName],
      RankBoard
    })
  end
  if bSuccess then
    return bResult
  else
    return true
  end
end

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  local tbOriShowKeys = {}
  for k, v in pairs(RankBoard.tbSetting) do
    if v.bShow > 0 and ChkTimeFrame(v.szTimeFrameOpen, v.szTimeFrameClose) and ChkIsShowCallBack(v.szShowCallBack) and ChkSpecialShow(v.Key) then
      local v = Lib:CopyTB(v)
      if Lib:IsEmptyStr(v.Sub) then
        table.insert(tbOriShowKeys, v)
      else
        local tbMainKey
        for i2, v2 in ipairs(tbOriShowKeys) do
          if v2.Key == v.Sub then
            tbMainKey = v2
            break
          end
        end
        if not tbMainKey then
          table.insert(tbOriShowKeys, {
            Name = v.Sub,
            Key = v.Sub,
            SortId = v.SortId,
            tbLeaves = {},
            ShowLevel = v.ShowLevel
          })
          tbMainKey = tbOriShowKeys[#tbOriShowKeys]
        elseif v.ShowLevel < tbMainKey.ShowLevel then
          tbMainKey.ShowLevel = v.ShowLevel
        end
        v.bSub = true
        table.insert(tbMainKey.tbLeaves, v)
      end
    end
  end
  table.sort(tbOriShowKeys, function(a, b)
    return a.SortId < b.SortId
  end)
  for _, v in pairs(tbOriShowKeys) do
    if v.tbLeaves and next(v.tbLeaves) then
      table.sort(v.tbLeaves, function(left, right)
        return left.SortId < right.SortId
      end)
    end
  end
  self.tbOriShowKeys = tbOriShowKeys
  local tbBigData = {}
  local tbSmallData = {}
  for k, v in ipairs(tbOriShowKeys) do
    if not v.ShowLevel or v.ShowLevel and me.nLevel >= v.ShowLevel then
      local tbLeaves = v.tbLeaves
      local nLeavesCount
      if nil ~= tbLeaves then
        nLeavesCount = #tbLeaves
      else
        nLeavesCount = 0
      end
      table.insert(tbBigData, {
        nType = v.Key,
        szText = i18n.Parse(v.Name),
        tbInfo = v,
        nLeavesCount = nLeavesCount
      })
      if tbLeaves and Lib:CountTB(tbLeaves) > 0 then
        for ksmall, vsmall in ipairs(tbLeaves) do
          table.insert(tbSmallData, {
            nType = v.Key,
            szText = i18n.Parse(vsmall.Name),
            nRealIndex = ksmall,
            tbInfo = vsmall
          })
        end
      end
    end
  end
  self.uiTree = Ui:NewTree(self, self.TreeRoot, self.onClickTreeBigBtn, self.onClickTreeSmallBtn)
  self.uiTree:SetData(tbBigData, tbSmallData)
  self.tbTreeBigData = tbBigData
  self.tbTreeSmallData = tbSmallData
  for _, tbFuns in pairs(self.tbDisplayFuns) do
    if tbFuns.OnCreate then
      tbFuns:OnCreate()
    end
  end
  self.bNeedUpdate = false
end

function tbWnd:OnOpen(szKey)
  self.szCurKey = nil
  if szKey and RankBoard.tbSetting[szKey] then
    self.szCurKey = szKey
  end
  local bDefault = false
  if not self.szCurKey then
    self.szCurKey = "FightPower"
    bDefault = true
  end
  if self.szCurKey then
    local nBigIndex, nSmallIndex, nRealIndex
    for nIndex, tbBigData in pairs(self.tbTreeBigData) do
      if tbBigData.nType == self.szCurKey then
        nBigIndex = nIndex
        break
      end
    end
    if not nBigIndex then
      for nIndex, tbSmallData in pairs(self.tbTreeSmallData) do
        if tbSmallData.tbInfo.Key == self.szCurKey then
          nSmallIndex = nIndex
          nRealIndex = tbSmallData.nRealIndex
          break
        end
      end
      if nSmallIndex then
        local szBigKey = self.tbTreeSmallData[nSmallIndex].nType
        for nIndex, tbBigData in pairs(self.tbTreeBigData) do
          if tbBigData.nType == szBigKey then
            nBigIndex = nIndex
            break
          end
        end
      end
    end
    if nBigIndex then
      if not bDefault then
        self.uiTree:SelectNode(nBigIndex, nRealIndex)
      else
        self.uiTree:SelectNode(nBigIndex, nil, false)
      end
    end
  end
  self.nCurPage = 1
  self.pPanel:SetActive(tbWnd.NotInfo, false)
  self:OnChangeServerRankingOnOff()
  self:DoUpdateRank()
end

function tbWnd:onClickTreeBigBtn(tbInfo, nBigIndex)
  if tbInfo.Sub and Lib:IsEmptyStr(tbInfo.Sub) then
    self.szCurKey = tbInfo.Key
    self.nCurPage = 1
    self:UpdateRank()
  end
  local tbLeaves = tbInfo.tbLeaves
  if nil ~= tbLeaves and #tbLeaves >= 1 then
    local index = 1
    for i = 1, nBigIndex - 1 do
      index = index + self.tbTreeBigData[i].nLeavesCount
    end
    self.uiTree:OnClickSmallButton(index, tbLeaves[1])
  end
end

function tbWnd:onClickTreeSmallBtn(tbInfo)

  self.nCurPage = 1
  self:UpdateRank()
end

      tbFuns:Reset()
    end

  if not tbData or Lib:IsEmptyTB(tbData) then
    self.pPanel:SetActive(tbWnd.NotInfo, true)
    return
  end
  for k, tbFuns in pairs(self.tbDisplayFuns) do
    if k == szKey then
      for nIndex, tbRecord in ipairs(tbData) do
        local tbScrollView = tbFuns.tbScrollView and tbFuns.tbScrollView or tbDisplayFuns_Default.tbScrollView
        local tbFunSet = tbFuns.SetElement and tbFuns or tbDisplayFuns_Default
        tbScrollView:SetElement(nIndex, function(tbElement)
          tbFunSet:SetElement(tbElement, tbRecord)
        end)
      end
    end
  end
  self.pPanel:SetActive(tbWnd.NotInfo, false)
end

function tbWnd:OnUpdate()

    self:DoUpdateRank()
    self.bNeedUpdate = false
  end
end

function tbWnd:UpdateRank()
  self.bNeedUpdate = true
end

function tbWnd:DoUpdateRank()
  local szKey = self.szCurKey
  local nPage = self.nCurPage
  local tbSplitItem = Lib:SplitStr(szKey, "_")
  local MainKey = tbSplitItem[1]
  self.bHideFactionLabel = "FightPower" == MainKey and Lib:CountTB(tbSplitItem) > 1
  self:SetTitle(MainKey)
  local tbData = RankBoard:CheckUpdateData(szKey, nPage)
  self:UpdateMyRankInfo(MainKey)
  if tbData then
    self:SetRankData(MainKey, tbData)
  else
    self:ResetAll()
  end
end

function tbWnd:OnSynRankData(szKey, nPage)
  if self.szCurKey ~= szKey or self.nCurPage ~= nPage then
    return
  end
  self:UpdateRank()
end

function tbWnd:OnChangeServerRankingOnOff()
  self.pPanel:SetActive(self.btnRankingServer, OnOffController:Check(OnOffController.KEY.RankingServer))
end

      self.OnSynRankData
    },
    {
      EventNotify.emNOTIFY_RANKING_SERVER_ONOFF,
      self.OnChangeServerRankingOnOff
    }
  }
