local tbWnd = Ui:CreateWindow("UITongBossRank")

function tbWnd:NotifyEvents()
  return {}
end

tbWnd.nKinRank = 1
tbWnd.nPersonalRank = 2
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnTong = "node/imgBG/Content/btnList/btnTong",
  btnPersonage = "node/imgBG/Content/btnList/btnPersonage",
  txtFaction = "node/imgBG/Content/TitleList/txtFaction",
  txtName = "node/imgBG/Content/TitleList/txtName",
  Template = "node/imgBG/Content/ScrollView/Viewport/Content/Template",
  txtMyRank = "node/imgBG/Content/MyRank/txtRank",
  imgFaction = "node/imgBG/Content/MyRank/imgFaction",
  txtMyName = "node/imgBG/Content/MyRank/txtName",
  txtMyScore = "node/imgBG/Content/MyRank/txtScore"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    return self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbRankTypeGroup = Ui:NewButtonGroup(self, {
    [1] = tbControls.btnTong,
    [2] = tbControls.btnPersonage
  }, true, function(self, nIndex)
    self:OnRankTypeChange(nIndex)
  end)
  self.tbRankScroll = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  self.tbRankScroll:Clear()
  self.nRankType = self.nKinRank
  self.tbKinRankInfo = {}
  self.tbMyKinRankInfo = {}
  self.tbMyPersonalRankInfo = {}
  self.tbPersonalRankInfo = {}
  self:ReqRankInfo(self.nPersonalRank)
  self:ReqRankInfo(self.nKinRank)
  self.tbRankTypeGroup:Select(self.nKinRank)
end

function tbWnd:OnClose()
end

function tbWnd:OnRankTypeChange(nRankType)
  if nRankType == self.nKinRank then
    self.nRankType = self.nKinRank
    self:ReqRankInfo(self.nKinRank)
    self:UpdateSV(self.tbKinRankInfo)
    self:UpdateMyRank(self.tbMyKinRankInfo)
  else
    self.nRankType = self.nPersonalRank
    self:ReqRankInfo(self.nPersonalRank)
    self:UpdateSV(self.tbPersonalRankInfo)
    self:UpdateMyRank(self.tbMyPersonalRankInfo)
  end
end

function tbWnd:ReqRankInfo(nRankType)
  TongChallenge.ReqKCRankInfo(nRankType)
end

function tbWnd:UpdateRank(nRankType, tbRankInfo)
  if nRankType ~= self.nRankType then
    return
  end
  if nRankType == self.nKinRank then
    self.tbKinRankInfo = tbRankInfo.tbRankList
    self.tbMyKinRankInfo = tbRankInfo.tbOwnRank
    self.pPanel:SetActive(tbControls.txtFaction, false)
  else
    self.tbPersonalRankInfo = tbRankInfo.tbRankList
    self.tbMyPersonalRankInfo = tbRankInfo.tbOwnRank
    self.pPanel:SetActive(tbControls.txtFaction, true)
  end
  self:UpdateSV(tbRankInfo.tbRankList)
  self:UpdateMyRank(tbRankInfo.tbOwnRank)
end

function tbWnd:UpdateSV(tbRankList)
  tbRankList = tbRankList or {}
  self.tbRankScroll:Clear()
  for i, v in pairs(tbRankList) do
    self.tbRankScroll:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtRank", i)
      pPanel:Label_SetText("txtName", tbRankList[i].szName)
      pPanel:Label_SetText("txtScore", tbRankList[i].nScore)
      if self.nRankType == self.nPersonalRank and tbRankList[i].nFaction then
        pPanel:Sprite_SetSprite("imgFaction", Faction:GetIcon(tbRankList[i].nFaction))
        pPanel:SetActive("imgFaction", true)
      else
        pPanel:SetActive("imgFaction", false)
      end
    end)
  end
end

function tbWnd:UpdateMyRank(tbOwnRank)
  tbOwnRank = tbOwnRank or {}
  self.pPanel:Label_SetText(tbControls.txtMyRank, tbOwnRank.nIndex or "")
  self.pPanel:Label_SetText(tbControls.txtMyName, tbOwnRank.szName or "")
  self.pPanel:Label_SetText(tbControls.txtMyScore, tbOwnRank.nScore or "")
  if self.nRankType == self.nPersonalRank then
    self.pPanel:Sprite_SetSprite(tbControls.imgFaction, Faction:GetIcon(me.nFaction))
    self.pPanel:SetActive(tbControls.imgFaction, true)
  else
    self.pPanel:SetActive(tbControls.imgFaction, false)
  end
end
