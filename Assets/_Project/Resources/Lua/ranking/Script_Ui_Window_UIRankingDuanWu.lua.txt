local DragonBoatRaceCfg = ActivityCfg.DragonBoatRaceCfg

  Ranking = "node/imgBG/Content/Ranking",

  RankNotInfo = "node/imgBG/Content/Ranking/Right/RankList/NotInfo",
  RankContent = "node/imgBG/Content/Ranking/Right/RankList/Content",
  RankTemplatePlayer = "node/imgBG/Content/Ranking/Right/RankList/Content/TemplatePlayer",
  RankTemplateKin = "node/imgBG/Content/Ranking/Right/RankList/Content/TemplateKin",
  txtPage = "node/imgBG/Content/Ranking/Right/txtPage",
  btnPrePage = "node/imgBG/Content/Ranking/Right/btnPrePage",
  btnNextPage = "node/imgBG/Content/Ranking/Right/btnNextPage",
  btnFirstPage = "node/imgBG/Content/Ranking/Right/btnFirstPage",
  btnLastPage = "node/imgBG/Content/Ranking/Right/btnLastPage",
  txtRank = "node/imgBG/Content/Ranking/Bottom/txtRank",
  txtRankValue = "node/imgBG/Content/Ranking/Bottom/txtRankValue",

local tbControls = tbWnd.tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnPrePage = function(self)
    self:OnClickBtnPrePage()
  end,
  btnNextPage = function(self)
    self:OnClickBtnNextPage()
  end,
  btnFirstPage = function(self)
    self:OnClickBtnFirstPage()
  end,
  btnLastPage = function(self)
    self:OnClickBtnLastPage()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MAP_ENTER,
      self.OnEnterMap
    }
  }
end

local tbFnUiTreeText = {
  fnRank = function()
    return i18n.EasyGet("\230\142\146\229\144\141")
  end,
  fnPersonalRank = function()
    return i18n.EasyGet("\228\184\170\228\186\186\230\142\146\229\144\141")
  end,
  fnKinRank = function()
    return i18n.EasyGet("\229\174\182\230\151\143\230\142\146\229\144\141")
  end,
  fnAwardPreview = function()
    return i18n.EasyGet("\229\165\150\229\138\177\233\162\132\232\167\136")
  end,
  fnAwardPreviewPersonal = function()
    return i18n.EasyGet("\228\184\170\228\186\186\229\165\150\229\138\177")
  end,
  fnAwardPreviewKin = function()
    return i18n.EasyGet("\229\174\182\230\151\143\229\165\150\229\138\177")
  end
}
local tbUiTreeBigIndex = {nRank = 1, nAwardPreview = 2}
local tbPanelIndex = {
  nPersonalRank = 1,
  nKinRank = 2,
  nPersonalAward = 3,
  nKinAward = 4
}

function tbWnd:OnCreate()
  self.uiTree = Ui:NewTree(self, tbControls.TreeRoot, self.OnClickTreeButton, self.OnClickTreeButton)
  local tbUITreeBigData = {
    {
      nType = tbUiTreeBigIndex.nRank,
      szText = tbFnUiTreeText.fnRank(),
      tbInfo = {nOnClickIndex = nil}
    },
    {
      nType = tbUiTreeBigIndex.nAwardPreview,
      szText = tbFnUiTreeText.fnAwardPreview(),
      tbInfo = {nOnClickIndex = nil}
    }
  }
  local tbUITreeSmallData = {
    {
      nType = tbUiTreeBigIndex.nRank,
      szText = tbFnUiTreeText.fnPersonalRank(),
      tbInfo = {
        nOnClickIndex = tbPanelIndex.nPersonalRank
      }
    },
    {
      nType = tbUiTreeBigIndex.nRank,
      szText = tbFnUiTreeText.fnKinRank(),
      tbInfo = {
        nOnClickIndex = tbPanelIndex.nKinRank
      }
    },
    {
      nType = tbUiTreeBigIndex.nAwardPreview,
      szText = tbFnUiTreeText.fnAwardPreviewPersonal(),
      tbInfo = {
        nOnClickIndex = tbPanelIndex.nPersonalAward
      }
    },
    {
      nType = tbUiTreeBigIndex.nAwardPreview,
      szText = tbFnUiTreeText.fnAwardPreviewKin(),
      tbInfo = {
        nOnClickIndex = tbPanelIndex.nKinAward
      }
    }
  }
  self.uiTree:SetData(tbUITreeBigData, tbUITreeSmallData)
  self.tbPersonalRankScrollView = Ui:NewScrollView(self, tbControls.RankTemplatePlayer)
  self.tbKinRankScrollView = Ui:NewScrollView(self, tbControls.RankTemplateKin)
  self.tbPersonalAwardScrollView = Ui:NewScrollView(self, tbControls.AwardTemplatePlayer)
  self.tbKinAwardScrollView = Ui:NewScrollView(self, tbControls.AwardTemplateKin)
end

function tbWnd:OnOpen()
  self.uiTree:AutoSelect()
  self:UpdatePanel(tbPanelIndex.nPersonalRank)
end

function tbWnd:OnClickTreeButton(tbInfo)
  local nOnClickIndex = tbInfo.nOnClickIndex
  if nil ~= nOnClickIndex and self.nPanelIndex ~= nOnClickIndex then
    self:UpdatePanel(nOnClickIndex)
  end
end

function tbWnd:UpdatePanel(nPanelIndex)
  self.nPanelIndex = nPanelIndex
  local bShowRankPanel = nPanelIndex == tbPanelIndex.nPersonalRank or nPanelIndex == tbPanelIndex.nKinRank
  local bShowAwardPanel = nPanelIndex == tbPanelIndex.nPersonalAward or nPanelIndex == tbPanelIndex.nKinAward
  self.pPanel:SetActive(tbControls.Ranking, bShowRankPanel)
  self.pPanel:SetActive(tbControls.Awards, bShowAwardPanel)
  if bShowRankPanel then
    self.pPanel:SetActive(tbControls.TitlePlayer, nPanelIndex == tbPanelIndex.nPersonalRank)
    self.pPanel:SetActive(tbControls.TitleKin, nPanelIndex == tbPanelIndex.nKinRank)
  end
  if bShowAwardPanel then
    self.pPanel:SetActive(tbControls.AwardsListPlayer, nPanelIndex == tbPanelIndex.nPersonalAward)
    self.pPanel:SetActive(tbControls.AwardsListKin, nPanelIndex == tbPanelIndex.nKinAward)
  end
  if nPanelIndex == tbPanelIndex.nPersonalRank then
    self:ShowPersonalRankPanel()
  elseif nPanelIndex == tbPanelIndex.nKinRank then
    self:ShowKinRankPanel()
  elseif nPanelIndex == tbPanelIndex.nPersonalAward then
    self:ShowPersonalAwardPanel()
  elseif nPanelIndex == tbPanelIndex.nKinAward then
    self:ShowKinAwardPanel()
  end
end

function tbWnd:ShowPersonalAwardPanel()
  for nIndex, tbPersonalAwardCfg in ipairs(DragonBoatRaceCfg.tbActivityEndPersonalAward) do
    self.tbPersonalAwardScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("txtRank", i18n.Get(tbPersonalAwardCfg.szRankUIShowText))
      if tbElement.tbLevelAwardShowScrollView == nil then
        tbElement.tbLevelAwardShowScrollView = Ui:NewItemGridScrollView(tbElement, "Awards/ItemTemplate", 1, 1)
      end
      local tbAward = Award:GetAward(DragonBoatRaceCfg.nActivityEndPersonalAwardId, tbPersonalAwardCfg.nAwardKey)
      tbElement.tbLevelAwardShowScrollView:UpdateGenericItem(tbAward, true)
    end)
  end
end

function tbWnd:ShowKinAwardPanel()
  for nIndex, tbKinAwardCfg in ipairs(DragonBoatRaceCfg.tbActivityEndKinAward) do
    self.tbKinAwardScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("txtRank", i18n.Get(tbKinAwardCfg.szRankUIShowText))
      if tbElement.tbLevelAwardShowScrollView == nil then
        tbElement.tbLevelAwardShowScrollView = Ui:NewItemGridScrollView(tbElement, "Awards/ItemTemplate", 1, 1)
      end
      local tbAward = Award:GetAward(DragonBoatRaceCfg.nActivityEndKinAwardId, tbKinAwardCfg.nAwardKey)
      tbElement.tbLevelAwardShowScrollView:UpdateGenericItem(tbAward, true)
    end)
  end
end

function tbWnd:ShowPersonalRankPanel()
  self:PersonalSwitchPage(1)
end

function tbWnd:PersonalSwitchPage(nPageIndex)
  self.nPageIndex = nPageIndex
  self.tbPersonalRankScrollView:Clear()
  self.tbKinRankScrollView:Clear()
  self:SetPersonalBottomRankAndScoreText()
  self:SetPersonalNowPageAndSumPageText()
  self:RequestPersonalRankData()
end

function tbWnd:SetPersonalNowPageAndSumPageText()
  self:SetNowPageAndSumPage(self.nPageIndex, self.nPersonalSumPage or 0)
end

function tbWnd:SetPersonalBottomRankAndScoreText()
  if self.nMyPersonalRank ~= nil and nil ~= self.nMyPersonalScore then
    self:SetMyRankAndScoreShow(i18n.EasyGet("\230\136\145\231\154\132\230\142\146\229\144\141\239\188\154"), self.nMyPersonalRank, i18n.EasyGet("\230\128\187\231\167\175\229\136\134\239\188\154"), self.nMyPersonalScore)
  else
    self:SetNotOnRank()
  end
end

function tbWnd:RequestPersonalRankData()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.DragonBoatRace)
  if nil ~= pInstance and pInstance:IsOpen() then
    pInstance:C2SRequestPersonalRankData(self.nPageIndex)
  end
end

function tbWnd:OnSyncPersonalRank(tbData)
  self.nMyPersonalRank = tbData.nMyRank
  self.nMyPersonalScore = tbData.nMySumScore
  self.nPersonalSumPage = math.ceil(tbData.nRankLength / DragonBoatRaceCfg.nShowNumAPage)
  if self.nPanelIndex == tbPanelIndex.nPersonalRank then
    self:SetPersonalBottomRankAndScoreText()
    self:SetPersonalNowPageAndSumPageText()
    if self.nPageIndex == tbData.nPage then
      local bRankDataNotEmpty = #tbData.tbPagePlayerData > 0
      self.pPanel:SetActive(tbControls.RankContent, bRankDataNotEmpty)
      self.pPanel:SetActive(tbControls.RankNotInfo, not bRankDataNotEmpty)
      if bRankDataNotEmpty then
        for nIndex, tbPlayerData in ipairs(tbData.tbPagePlayerData) do
          self.tbPersonalRankScrollView:SetElement(nIndex, function(tbElement)
            local pPanel = tbElement.pPanel
            local nRank = (self.nPageIndex - 1) * DragonBoatRaceCfg.nShowNumAPage + nIndex
            self:RankElement_SetRank(tbElement, "UIRankNo", "txtRank", nRank)
            pPanel:Label_SetText("txtPlayerName", tbPlayerData.szName)
            Ui:SetVipLevelEffect(pPanel, tbPlayerData.nVipLevel, "txtPlayerName")
            self:RankElement_SetHonor(tbElement, "txtPlayerName/HonorTitle", tbPlayerData.nHonorStarLevel)
            pPanel:Label_SetText("txtKinName", tbPlayerData.szKinName)
            pPanel:Label_SetText("txtScore", tbPlayerData.nSumScore)
            local nHour, nMinute, nSecond = Lib:TransferSecond2NormalTime(tbPlayerData.nSumFinishTime)
            pPanel:Label_SetText("txtTime", string.format("%02d:%02d:%02d", nHour, nMinute, nSecond))
            self:RankElement_SetBackGroundColor(tbElement, me.dwID == tbPlayerData.nPlayerId)
          end)
        end
      end
    end
  end
end

function tbWnd:ShowKinRankPanel()

end

function tbWnd:OnSyncKinRank(tbData)
  self.nMyKinRank = tbData.nMyKinRank
  self.nMyKinScore = tbData.nMyKinScore
  self.nKinSumPage = math.ceil(tbData.nRankLength / DragonBoatRaceCfg.nShowNumAPage)
  if self.nPanelIndex == tbPanelIndex.nKinRank then
    self:SetKinBottomRankAndScoreText()

    if self.nPageIndex == tbData.nPage then
      local bRankDataNotEmpty = #tbData.tbPageKinData > 0
      self.pPanel:SetActive(tbControls.RankContent, bRankDataNotEmpty)
      self.pPanel:SetActive(tbControls.RankNotInfo, not bRankDataNotEmpty)
      if bRankDataNotEmpty then
        for nIndex, tbKinData in ipairs(tbData.tbPageKinData) do
          self.tbKinRankScrollView:SetElement(nIndex, function(tbElement)
            local pPanel = tbElement.pPanel
            local nRank = (self.nPageIndex - 1) * DragonBoatRaceCfg.nShowNumAPage + nIndex
            self:RankElement_SetRank(tbElement, "UIRankNo", "txtKinRank", nRank)
            pPanel:Label_SetText("txtKinName", tbKinData.szKinName)

            pPanel:Label_SetText("txtKinScore", tbKinData.nSumScore)
            self:RankElement_SetBackGroundColor(tbElement, me.dwKinId == tbKinData.nKinId)
          end)
        end
      end
    end
  end
end

function tbWnd:SetMyRankAndScoreShow(szRankName, nRank, szScoreName, nScore)
  self:SetBottomActive(true)
  self.pPanel:Label_SetText(tbControls.txtRank, szRankName)
  self.pPanel:Label_SetText(tbControls.txtRankValue, nRank)
  self.pPanel:Label_SetText(tbControls.txtScore, szScoreName)
  self.pPanel:Label_SetText(tbControls.txtScoreValue, nScore)
end

function tbWnd:SetNotOnRank()
  self:SetBottomActive(false)
end

function tbWnd:SetBottomActive(bIsOnRank)
  self.pPanel:SetActive(tbControls.txtRank, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtRankValue, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtScore, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtScoreValue, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtMyInfoLabelNoData, not bIsOnRank)
end

function tbWnd:SetNowPageAndSumPage(nNowPage, nSumPage)
  self.pPanel:Label_SetText(tbControls.txtPage, string.format("%d/%d", nNowPage, nSumPage))
end

function tbWnd:RankElement_SetRank(tbElement, szTop3RankNoKey, szTxtRankKey, nRank)
  local pPanel = tbElement.pPanel
  local bTop3 = nRank <= 3
  pPanel:SetActive(szTxtRankKey, not bTop3)
  pPanel:SetActive(szTop3RankNoKey, bTop3)
  if bTop3 then
    pPanel:SetActive(szTop3RankNoKey .. "/imgNO1", 1 == nRank)
    pPanel:SetActive(szTop3RankNoKey .. "/imgNO2", 2 == nRank)
    pPanel:SetActive(szTop3RankNoKey .. "/imgNO3", 3 == nRank)
  else
    pPanel:Label_SetText(szTxtRankKey, nRank)
  end
end

function tbWnd:RankElement_SetHonor(tbElement, szHonorKey, nHonorStarLevel)
  local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(nHonorStarLevel, false)
  local obj = tbElement.pPanel:GetObject(szHonorKey)
  local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
  pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
end

function tbWnd:RankElement_SetBackGroundColor(tbElement, bIsSelf)
  if bIsSelf then
    tbElement.pPanel:Sprite_SetColor("Main", 255, 255, 255, 255)
  else
    tbElement.pPanel:Sprite_SetColor("Main", 0, 0, 0, 20)
  end
end

function tbWnd:GetNowShowRankSumPage()
  if self.nPanelIndex == tbPanelIndex.nPersonalRank then
    return self.nPersonalSumPage
  elseif self.nPanelIndex == tbPanelIndex.nKinRank then
    return self.nKinSumPage
  else
    return nil
  end
end

function tbWnd:OnClickBtnPrePage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 and nil ~= self.nPageIndex and self.nPageIndex - 1 >= 1 then
    self:OnClickSwitchRankPage(self.nPageIndex - 1)
  end
end

function tbWnd:OnClickBtnNextPage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 and nil ~= self.nPageIndex and nSumPage >= self.nPageIndex + 1 then
    self:OnClickSwitchRankPage(self.nPageIndex + 1)
  end
end

function tbWnd:OnClickBtnFirstPage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 then
    self:OnClickSwitchRankPage(1)
  end
end

function tbWnd:OnClickBtnLastPage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 then
    self:OnClickSwitchRankPage(nSumPage)
  end
end

function tbWnd:OnClickSwitchRankPage(nPageIndex)
  if self.nPageIndex ~= nPageIndex then
    if self.nPanelIndex == tbPanelIndex.nPersonalRank then
      self:PersonalSwitchPage(nPageIndex)
    elseif self.nPanelIndex == tbPanelIndex.nKinRank then
      self:KinSwitchPage(nPageIndex)
    end
  end
end

function tbWnd:OnEnterMap()
  self:Close()
end
