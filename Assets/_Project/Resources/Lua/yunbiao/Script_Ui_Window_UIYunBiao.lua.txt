    {
      EventNotify.emNOTIFY_DARTING_DO_DARTING,
      self.OnDoDarting

    {
      EventNotify.emNOTIFY_DARTING_ON_REFRESH,
      self.OnRefreshDarting
    },
    {
      EventNotify.emNOTIFY_DARTING_BEGIN_REFRESH,
      self.OnBeginRefreshingDarting
    },
    {
      EventNotify.emNOTIFY_DARTING_ON_LOGIN,
      self.OnLogin
    }
  }
end

function tbWnd:OnCreate()
  self.tbNpcModelList = {}
  self.tbItemGridList = {}
  self.tbAwardsList = {}
  local obj
  for nIndex, tbInfo in ipairs(tbDartingCarList) do
    obj = self.pPanel:GetObject(tbInfo.NpcModel)
    self.tbNpcModelList[nIndex] = obj:GetComponent(Ui.UINpcModel)
    self.tbItemGridList[nIndex] = Ui:NewExistItemGrid(self, tbInfo.UIItemGrid)
    self.tbAwardsList[nIndex] = Ui:NewItemGridView(self, tbInfo.Template)
  end
end

function tbWnd:OnOpen(bInit, nRefreshNum, nTemplateId, nRefreshTemplateId)
  self.pPanel:Label_SetText(tbControls.txtRuleContent, i18n.Get(Darting.Cfg.szDartingDesc))
  self:RefreshInfo(true, bInit)
  self:ShowModel()
  self.pPanel:Button_SetEnable(tbControls.btnRefresh, true)
  self.pPanel:SetActive(tbControls.ReceiveEffect, false)
  self.pPanel:SetActive(tbControls.btnReceive, true)
  if not bInit then
    self:RefreshCostInfo()
    self.pPanel:Label_SetText(tbControls.btnTransport .. "/Text", i18n.EasyGet("\232\191\144\233\128\129{0}", Npc:GetNameByTemplateId(nTemplateId)))
  end
end

function tbWnd:RefreshCostInfo()
  local tbCacheData = Darting:GetCacheData()
  if tbCacheData.nRefreshNum < 0 then
    self.pPanel:SetActive(tbControls.coinNumber, false)
    self.pPanel:SetActive(tbControls.txtFree, true)
    self.pPanel:Label_SetText(tbControls.txtFree, i18n.EasyGet("\229\133\141\232\180\185"))
    self.pPanel:Label_SetText(tbControls.btnRefresh .. "/Text", i18n.EasyGet("\229\136\183\230\150\176\233\149\150\232\189\166"))
  else
    self.pPanel:SetActive(tbControls.coinNumber, true)
    self.pPanel:SetActive(tbControls.txtFree, false)
    self.pPanel:Label_SetText(tbControls.btnRefresh .. "/Text", i18n.EasyGet("\229\136\183\230\150\176\233\149\150\232\189\166({0}/{1})", tbCacheData.nRefreshNum, Darting.Cfg.nMaxRefreshDartingTime))
  end
  self.pPanel:Button_SetEnable(tbControls.btnRefresh, true)
  self.pPanel:SetActive(tbControls.coinNumber .. "/Coin", true)
  self.pPanel:SetActive(tbControls.coinNumber .. "/txtCost", true)
  self.pPanel:Label_SetText(tbControls.coinNumber .. "/txtCost", Darting.Cfg.tbRefreshCost[2])
  self.pPanel:Label_SetText(tbControls.txtCostTips, i18n.EasyGet("\232\138\177\232\180\185"))
  if tbCacheData.nRefreshNum >= Darting.Cfg.nMaxRefreshDartingTime or tbCacheData.bForbidden then
    self.pPanel:SetActive(tbControls.coinNumber .. "/Coin", false)
    self.pPanel:SetActive(tbControls.coinNumber .. "/txtCost", false)
    self.pPanel:Label_SetText(tbControls.txtCostTips, Lib:IIF(tbCacheData.bForbidden, i18n.EasyGet("\229\183\178\230\152\175\230\156\128\233\171\152\231\186\167\233\149\150\232\189\166"), i18n.EasyGet("\230\151\160\229\137\169\228\189\153\229\136\183\230\150\176\230\172\161\230\149\176")))
    self.pPanel:Button_SetEnable(tbControls.btnRefresh, false)
    if tbCacheData.bForbidden then
      self.pPanel:Label_SetText(tbControls.txtFree, i18n.EasyGet("\229\183\178\230\152\175\230\156\128\233\171\152\231\186\167\233\149\150\232\189\166"))
    end
  end
end

function tbWnd:RefreshInfo(isHightLight, bInit)
  local tbList = Darting:GetDartingCarList()
  local tbAward
  local tbCacheData = Darting:GetCacheData()
  local bCur
  local bSuccess = false
  local bImgHLLight
  for nIndex, tbInfo in ipairs(tbDartingCarList) do
    self.pPanel:SetActive(tbInfo.imgLabel, false)
    if tbList[nIndex] then
      tbAward = Award:GetAward(tbList[nIndex].nAwardId, tbList[nIndex].nAwardKey)
      for i = 1, #tbAward do
        self.tbAwardsList[nIndex]:SetItem(i, tbAward[i], true)
      end
      bCur = tbCacheData.nCurTemplateId == tbList[nIndex].nTemplateId
      if bCur then
        self.pPanel:SetActive(tbInfo.imgLabel, bCur)
        bSuccess = true
      end
      bImgHLLight = false
      if 0 ~= tbCacheData.nTargetTemplateId and tbList[nIndex].nTemplateId == tbCacheData.nTargetTemplateId or 0 == tbCacheData.nTargetTemplateId and tbList[nIndex].nTemplateId == tbCacheData.nCurTemplateId then
        bImgHLLight = true
      end
      if not isHightLight then
        bImgHLLight = false
      end
      self.pPanel:SetActive(tbInfo.imgHL, bImgHLLight)
      if tbCacheData.nCurTemplateId > 0 then
        self.pPanel:Label_SetText(tbControls.btnTransport .. "/Text", i18n.EasyGet("\232\191\144\233\128\129{0}", Npc:GetNameByTemplateId(tbCacheData.nCurTemplateId)))
      end
    end
  end
  self.pPanel:SetActive(tbControls.btnReceive, not bSuccess)
  self.pPanel:SetActive(tbControls.btns1, bInit)
  self.pPanel:SetActive(tbControls.btns2, not bInit)
end

function tbWnd:ShowModel()
  local tbList = Darting:GetDartingCarList()
  local nNpcResId
  for nIndex, tbNpcMode in ipairs(self.tbNpcModelList) do
    if tbList[nIndex] then
      nNpcResId = KNpc.GetNpcTemplateInfo(tbList[nIndex].nTemplateId).nNpcResID
      tbNpcMode:ShowNpc(nNpcResId)
    end
  end
end

function tbWnd:OnClose()
  if self.nCloseUiTimer then
    Timer:Close(self.nCloseUiTimer)
    self.nCloseUiTimer = nil
  end
  for _, tbNpcMode in ipairs(self.tbNpcModelList) do
    tbNpcMode:Release()
  end
  for _, tbAwards in ipairs(self.tbAwardsList) do
    tbAwards:Clear()
  end
end

function tbWnd:OnDoDarting()
  self.pPanel:SetActive(tbControls.ReceiveEffect, false)
  self:RefreshInfo(true, true)
  self.nCloseUiTimer = Timer:Register((Darting.Cfg.nDartingMoveDelay - Darting.Cfg.nDoDartingDelay) * Env.GAME_FPS, function()
    self.nCloseUiTimer = nil
    self:Close()
  end)
end

function tbWnd:OnClickReceive()
  Darting:ReqBeginDartingC()
  self.pPanel:SetActive(tbControls.ReceiveEffect, true)
  self.pPanel:SetActive(tbControls.btnReceive, false)
end

function tbWnd:OnClickRefresh()
  if Darting:IsRefreshing() then
    me.CenterMsg(i18n.EasyGet("\229\136\183\230\150\176\228\184\173"))
    return
  end
  self.pPanel:Button_SetEnable(tbControls.btnRefresh, false)
  Darting:ReqRefreshDarting()
end

function tbWnd:OnRefreshDarting()
  self.pPanel:Button_SetEnable(tbControls.btnRefresh, true)
  self.pPanel:SetActive(tbControls.ReceiveEffect, false)
  self:RefreshInfo(true)
  self:RefreshCostInfo()
end

function tbWnd:OnBeginRefreshingDarting()
  self:RefreshInfo(false)
  self.pPanel:SetActive(tbControls.ReceiveEffect, true)
end

function tbWnd:OnLogin()
  self:RefreshInfo(true)
end
