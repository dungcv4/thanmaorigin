  for _, tbSetting in pairs(tbFile) do
    local nRelateId = tbSetting.RelateId
    self.tbRelateItem[nRelateId] = self.tbRelateItem[nRelateId] or {}

      end

function Item:LoadStarLevel()
  local szSetting = "Setting/Item/StarLevel.tab"
  self.tbStarLevelValue = {}
  local tbFile = Lib:LoadTabFile(szSetting, {ItemType = 1})
  if not tbFile then
    print("load " .. szSetting .. " failed!")
    return
  end
  for _, tbRow in pairs(tbFile) do
    if tbRow.ItemType then
      self.tbStarLevelValue[tbRow.ItemType] = {}
      for i = 1, self.MAX_STAR_LEVEL do
        self.tbStarLevelValue[tbRow.ItemType][i] = tonumber(tbRow["StartLv" .. i]) or 0
      end
    end
  end
  self.tbStarLevelValue = SetReadOnly(self.tbStarLevelValue)
end

function Item:GetStarLevel(nItemType, nValue)
  local nLevel = 0
  if not self.tbStarLevelValue then
    return 0
  end
  for i, nStar in ipairs(self.tbStarLevelValue[nItemType] or {}) do
    if nStar <= nValue then
      nLevel = i
    else
      break
    end
  end
  return nLevel
end

function Item:UseManyItem(nId, nCount, ...)
  if not Lib:IsInteger(nCount) or nCount < 0 then
    return
  end
  if 1 == nCount then
    Item:UseItem(nId, ...)
    return
  end
  local pItem, nPos = me.GetItemInAllSpace(nId)
  if not pItem then
    return
  end
  if not self:IsInBag(nPos) then
    return
  end
  if nCount > pItem.nCount then
    return
  end
  local szClassName = pItem.szClass
  local nUseLimit = Item:GetItemUseManyLimit(szClassName)
  if not nUseLimit then
    me.CenterMsg(i18n.EasyGet("\228\184\141\229\143\175\230\137\185\233\135\143\228\189\191\231\148\168"))
    return
  end
  if "RandItem" == szClassName or "GoldFuDai" == szClassName then
    nUseLimit = Item.PROTOCOL_USEITEM
  end
  if nUseLimit > 0 and nCount > nUseLimit then
    me.CenterMsg(i18n.EasyGet("\232\175\165\233\129\147\229\133\183\228\184\128\230\172\161\230\156\128\229\164\154\228\189\191\231\148\168{0}\228\184\170", nUseLimit))
    return
  end
  local bRet, nOpenRetCode, bIgnoreLog = xpcall(self.OnUseMany, Lib.ShowStack, self, pItem, nCount, ...)
  if not bRet then
    Log(nOpenRetCode)
    Log(debug.traceback())
  end
end

  if not nUseLimit then
    me.CenterMsg(i18n.EasyGet("\228\184\141\229\143\175\230\137\185\233\135\143\228\189\191\231\148\168"))
    return

    me.CenterMsg(i18n.EasyGet("\232\175\165\233\129\147\229\133\183\228\184\128\230\172\161\230\156\128\229\164\154\228\189\191\231\148\168{0}\228\184\170", nUseLimit))
    return
  end
  local bRet, nOpenRetCode, bIgnoreLog = xpcall(self.OnUseManyByTemplated, Lib.ShowStack, self, pPlayer, dwTemplateId, nCount, ...)
  if not bRet then
    Log(nOpenRetCode)
    Log(debug.traceback())
  end
end

function Item:OnUseManyByTemplated(pPlayer, dwTemplateId, nCount, ...)
  local tbInfo = Item:GetItemBaseProp(dwTemplateId)
  local szClassName = tbInfo.szClass
  local tbClass = self.tbClass[szClassName] or self.tbClass.default
  local dwPlayerId, szPlayerName, szItemName = pPlayer.dwID, pPlayer.szName, tbInfo.szName
  if tbClass and tbClass.OnUseManyByTemplated then
    local bOk, nRetCode, bIgnoreLog = xpcall(tbClass.OnUseManyByTemplated, Lib.ShowStack, tbClass, dwTemplateId, nCount, ...)
    if bOk then
      self:DoUseManyByTemplated(pPlayer, dwTemplateId, nRetCode, bIgnoreLog)
    else
      self:DoUseManyByTemplated(pPlayer, dwTemplateId, nCount, false)
      LogErr("Item:OnUseManyByTemplated Err", dwPlayerId, szPlayerName, dwTemplateId, i18n.Parse(szItemName), debug.traceback())
    end
  else
    LogErr("Item:OnUseManyByTemplated error: Item can't use many By Templated, TemplateId and ClassName = ", dwTemplateId, szClassName)
  end
end

function Item:DoUseManyByTemplated(pPlayer, dwTemplateId, nRetCode, bIgnoreLog)
  local nCount = nRetCode
  if nRetCode and nRetCode > 0 then
    local tbpItem = pPlayer.GetItemListInBag(dwTemplateId)
    local tbpCanTradeItem, tbpLimitTradeItem = {}, {}
    for index, pItem in ipairs(tbpItem) do
      if 1 == pItem.dwTradeLimitType then
        table.insert(tbpCanTradeItem, pItem)
      else
        table.insert(tbpLimitTradeItem, pItem)
      end
    end
    for index, pItem in ipairs(tbpLimitTradeItem) do
      if nRetCode > 0 then
        local nCost = nRetCode < pItem.nCount and nRetCode or pItem.nCount
        local nRealCostCount = pPlayer.ConsumeItem(pItem, nCost, Env.LogWay_UseItem, bIgnoreLog)
        nRetCode = nRetCode - nRealCostCount
        if nRealCostCount ~= nCost then
          LogErr("DoUseManyByTemplated BindType UseCount Error", dwTemplateId, pPlayer.dwID)
          Lib:LogFormat("DoUseManyByTemplated BindType Error dwTemplateId:%d, nRet:%s, player:%d, nBindCost:%d", dwTemplateId, nCount or "nil", pPlayer.dwID, nCount - nRetCode)
        end
      else
        break
      end
    end
    local nLimitTradeCost = nCount - nRetCode
    for index, pItem in ipairs(tbpCanTradeItem) do
      if nRetCode > 0 then
        local nCost = nRetCode < pItem.nCount and nRetCode or pItem.nCount
        local nRealCostCount = pPlayer.ConsumeItem(pItem, nCost, Env.LogWay_UseItem, bIgnoreLog)
        nRetCode = nRetCode - nRealCostCount
        if nRealCostCount ~= nCost then
          LogErr("DoUseManyByTemplated not BindType UseCount Error", dwTemplateId, pPlayer.dwID)
          Lib:LogFormat("DoUseManyByTemplated not BindType dwTemplateId:%d, nRet:%s, player:%d, nBindCost:%d", dwTemplateId, nCount or "nil", pPlayer.dwID, nLimitTradeCost)
        end
      else
        break
      end
    end
    if 0 == nRetCode then
      Lib:LogFormat("Item:OnUseMany dwTemplateId:%d, nRet:%s, player:%d, nBindCost:%d", dwTemplateId, nCount or "nil", pPlayer.dwID, nLimitTradeCost)
      return true
    else
      LogErr("DoUseManyByTemplated BindType UseCount Error", dwTemplateId, pPlayer.dwID, nRetCode, nCount)
      Lib:LogFormat("Item:OnUseMany Count Error dwTemplateId:%d, nRet:%s, player:%d, nBindCost:%d", dwTemplateId, nCount or "nil", pPlayer.dwID, nLimitTradeCost)
      return false
    end
  end
end

function Item:OnUse(pItem, ...)
  local szClassName = pItem.szClass
  local nRet, szMsg = self:CheckUsable(pItem, szClassName)
  if 1 ~= nRet then
    if szMsg then
      me.CenterMsg(szMsg)
    end
    return 0
  end
  local tbClass = self.tbClass[szClassName]
  if tbClass and tbClass.OnUse then
    local bCount = self:CheckUseCount(pItem)
    if not bCount then
      return 0
    end
    local bOk, nRetCode, bIgnoreLog = xpcall(tbClass.OnUse, Lib.ShowStack, tbClass, pItem, ...)
    if not bOk then
      nRetCode = pItem.nCount
      bIgnoreLog = false
      self:LogUseErr(pItem, me)
    end
    if MODULE_GAMESERVER and nRetCode and nRetCode > 0 then
      if GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
        Lib:CallBack({
          FieldPractice.OnUseItem,
          FieldPractice,
          me,
          pItem.dwTemplateId,
          nRetCode
        })
      end
      assert(me.ConsumeItem(pItem, nRetCode, Env.LogWay_UseItem, bIgnoreLog) == nRetCode)
    end
  end
end

function Item:OnUseMany(pItem, nCount, ...)
  local szClassName = pItem.szClass
  local nRet, szMsg = self:CheckUsable(pItem, szClassName)
  if 1 ~= nRet then
    if szMsg then
      me.CenterMsg(szMsg)
    end
    return 0
  end
  local tbClass = self.tbClass[szClassName]
  if tbClass and tbClass.OnUseMany then
    local bCount = self:CheckUseCount(pItem)
    if not bCount then
      return 0
    end
    local bOk, nRetCode, bIgnoreLog = xpcall(tbClass.OnUseMany, Lib.ShowStack, tbClass, pItem, nCount, ...)
    if not bOk then
      nRetCode = pItem.nCount
      bIgnoreLog = false
      self:LogUseErr(pItem, me)
    end
    if nRetCode and nRetCode > 0 then

function Item:OnUseAll(pItem)
  local szClassName = pItem.szClass
  local nRet, szMsg = self:CheckUsable(pItem, szClassName)
  if 1 ~= nRet then
    if szMsg then
      me.CenterMsg(szMsg)
    end
    return 0
  end
  local tbClass = self.tbClass[szClassName]
  if tbClass and tbClass.OnUseAll then
    local bCount = self:CheckUseCount(pItem)
    if not bCount then
      return 0
    end
    local bOk, nRetCode, bIgnoreLog = xpcall(tbClass.OnUseAll, Lib.ShowStack, tbClass, pItem)
    if not bOk then
      nRetCode = pItem.nCount
      bIgnoreLog = false
      self:LogUseErr(pItem, me)
    end
    if nRetCode and nRetCode > 0 then
      assert(me.ConsumeItem(pItem, nRetCode, Env.LogWay_UseItem, bIgnoreLog) == nRetCode)
    end
  end
end

function Item:ResetUseEquipCD(pPlayer, pEquip)
  if not pPlayer.tbUseEquipCD then
    return
  end
  local nEquipPos = Item:LimitOptEquipPosChange(pEquip)
  pPlayer.tbUseEquipCD[nEquipPos] = nil
end

function Item:CheckUseEquipCD(pPlayer, pEquip)
  local nEquipPos = Item:LimitOptEquipPosChange(pEquip)
  if not pPlayer.tbUseEquipCD then
    pPlayer.tbUseEquipCD = {}
  end
  local nCurTime = GetTime()
  if not pPlayer.tbUseEquipCD[nEquipPos] then
    pPlayer.tbUseEquipCD[nEquipPos] = {
      [1] = nCurTime,
      [2] = nCurTime,
      [3] = nCurTime,
      nFirstIndex = 1,
      nLastIndex = 0
    }
  end
  local tbData = pPlayer.tbUseEquipCD[nEquipPos]
  local nIndex = (tbData.nFirstIndex - 1) % self.nUseEquipLimitTime + 1
  if tbData[nIndex] < nCurTime - self.nUseEquipLimitCD then
    tbData.nFirstIndex = tbData.nFirstIndex + 1
  end
  if tbData.nLastIndex - tbData.nFirstIndex > 1 then
    return true
  end
  return false
end

function Item:LimitOptEquipPosChange(pEquip)
  local nEquipPos = pEquip.nEquipPos
  if nEquipPos == Item.EQUIPPOS_EXTEND then
    local nSlot = Item:GetExtEquipSlot(pEquip)
    nEquipPos = nEquipPos + 100 + nSlot
  end
  return nEquipPos
end

function Item:UseEquip(pPlayer, nId, bSlient, nEquipPos, bBreakMask)
  nId = math.floor(nId)
  local pEquip = pPlayer.GetItemInAllSpace(nId)
  if not pEquip then
    print("Unexsit Equip " .. nId)
    return
  end
  if pEquip.nUseLevel > pPlayer.nTotalLevel then
    return
  end
  local nEquipType = pEquip.nItemType
  if nEquipType < Item.EQUIP_HELM or nEquipType > Item.EQUIP_EXTEND then
    return
  end
  if nEquipType == Item.EQUIP_HORSE and pPlayer.GetDoing() == Npc.Doing.sit then
    pPlayer.CenterMsg(i18n.EasyGet("\230\137\147\229\157\144\230\151\182\228\184\141\232\131\189\232\163\133\229\164\135\233\169\172\229\140\185"))
    return
  end
  if not Lib:IsEmptyStr(pEquip.szClass) then
    local tbEquipClass = Item:GetClass(pEquip.szClass)
    if tbEquipClass.CheckUseEquip then
      local bRet, szMsg = tbEquipClass:CheckUseEquip(pPlayer, pEquip)
      if not bRet then
        pPlayer.CenterMsg(szMsg)
        return
      end
    else
      nEquipPos = -1
    end
  else
    nEquipPos = -1
  end
  if not bBreakMask and pPlayer.IsMaskPlayerAttr() then
    return pPlayer.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\230\151\160\230\179\149\231\169\191\232\132\177\232\163\133\229\164\135"))
  end
  if pEquip.nEquipPos and self:CheckUseEquipCD(pPlayer, pEquip) then
    return pPlayer.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\230\147\141\228\189\156\229\164\170\233\162\145\231\185\129\228\186\134\239\188\140\232\175\183\231\168\141\229\144\142\229\134\141\232\175\149\239\188\129"))
  end
  local nLimitCDEquipPos = Item:LimitOptEquipPosChange(pEquip)
  if pPlayer.tbUseEquipCD and pPlayer.tbUseEquipCD[nLimitCDEquipPos] then
    local tbData = pPlayer.tbUseEquipCD[nLimitCDEquipPos]
    local nIndex = tbData.nLastIndex % self.nUseEquipLimitTime + 1
    tbData.nLastIndex = tbData.nLastIndex + 1
    tbData[nIndex] = GetTime()
  end
  local nResult = pPlayer.UseEquip(nId, nEquipPos or -1)
  if 1 == nResult then
    Event:OnEvent(Event.emEVENT_EQUIP_USE, pPlayer.dwID, pEquip.nEquipPos, pEquip.dwId)
    pPlayer.CallClientScript("Item:OnUseEquipSuccessRsp", pEquip.nEquipPos, pEquip.dwId)
    if pEquip.nEquipPos == Item.EQUIPPOS_HORSE then
      local szEquipName = KItem.GetItemShowInfo(pEquip.dwTemplateId, pPlayer.nFaction, pPlayer.nFactionSect)
      pPlayer.CenterMsg(i18n.EasyGet("{0}\228\184\138\233\152\181\230\136\144\229\138\159", szEquipName))
      Horse:UpdataAsyncHorseGrade(pPlayer.dwID, Horse:GetHorseGrade(pPlayer))
    end
    if SERVER_RUNNING_MODE ~= SERVER_MODE.ZONE_SERVER then
      self:DealEquipAchievement(pPlayer)
    end
    if nEquipType == Item.EQUIP_MANTLE then
      Event:OnEvent(Event.emEVENT_EQUIP_MANTLE_USE, pPlayer.dwID)
    elseif nEquipType == Item.EQUIP_HORSE then
      Event:OnEvent(Event.emEVENT_EQUIP_HORSE_USE, pPlayer.dwID)
    end
  end
  Lib:LogFormat("Item:UseEquip pPlayer:%d, dwTemplateId:%d, nId:%d, nPos:%d, nResult:%d", pPlayer.dwID, pEquip.dwTemplateId, nId, pEquip.nEquipPos, nResult)
end

      end
    end

  me.CallClientScript("Ui.QuickUse:UIQuickUseRemove")

function Item:GetItemTradeLimitDescBySettings(dwTemplateId, dwTradeLimitType)
  local szTradeWay = ""

    tbTradeDesc = self:GetItemTradeWayDesc(nTradeLimitWay)
  end
  for i = Item.eTRADELIMITWAY.P2P, Item.eTRADELIMITWAY.OTHER do
    if tbTradeDesc[i] then
      if "" == szTradeWay then
        szTradeWay = tbTradeDesc[i]

    end
  end
  if "" == szTradeWay then
    return false, i18n.EasyGet("\228\184\141\229\143\175\228\186\164\230\152\147")
  end
  return true, i18n.EasyGet("\228\186\164\230\152\147\230\184\160\233\129\147\239\188\154{0}", szTradeWay)
end

  if weekCardLeft > 0 or bMonthCardState then

  for _, pNpc in pairs(tbNpcList) do

function Item:GetEquipTypeName(nTemplateId)
  local nEquipPos = KItem.GetEquipPos(nTemplateId)
  if nEquipPos ~= Item.EQUIPPOS_EXTEND then
    return self.EQUIPTYPE_NAME[Item:GetItemBaseProp(nTemplateId).nItemType]
  else
    return Item.ExtEquip.EXT_EQUIPTYPE_NAME[Item.ExtEquip:GetSlotByTemplateId(nTemplateId)]
  end
end

function Item:GetDetailTypeClassify(nItemType, nDetailType)
  if nItemType == Item.EQUIP_EXTEND then
    if nDetailType >= Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1 and nDetailType <= Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE5 then
      return {bInNormal = 1, bInHorseSkillStone = 1}
    elseif nDetailType >= Item.ExtEquip.CFG.SLOT.PARTNER_EQUIP_WEAPON and nDetailType <= Item.ExtEquip.CFG.SLOT.PARTNER_EQUIP_BOOTS then
      return {bInNormal = 1, bInPartnerEquip = 1}
    elseif Item.ShenBingPu:CheckIsValidSlot(nDetailType) then
      return {bInNormal = 1, bInShenBingPu = 1}
    elseif Item.SecretBook.tbSlotToRank[nDetailType] then
      return {bInNormal = 1}
    else
      return {
        bInNormal = 1,
        bInEquip = 1,
        bInZhenYuanEquip = 1
      }
    end
  else
    return Item.tbDetailTypeClassify[nDetailType]
  end
end

function Item:ItemTableCheck()
  self.tbCheckTable = Lib:LoadTabFileEx("Setting/Item/Item.tab", "dsdddd", "TemplateId", {
    "TemplateId",
    "ClassName",
    "ExtParam1",
    "ExtParam2",
    "ExtParam3",
    "ExtParam4"
  })
  Validator:Register(function()
    for _, tbAItemInfo in pairs(self.tbCheckTable) do
      if tbAItemInfo.ClassName ~= "" and self.tbClass[tbAItemInfo.ClassName] == nil then
        LogErr("Item ClassName: ", tbAItemInfo.ClassName, "is not exsit, TemplateId: ", tbAItemInfo.TemplateId)
        return false
      end
      if tbAItemInfo.ClassName == "RandItem" then
        local nAwardId = tbAItemInfo.ExtParam1
        local nAwardKey = tbAItemInfo.ExtParam2
        if nil == Award.tbAllAwards[nAwardId] or nil == Award.tbAllAwards[nAwardId].tbRandAwards[nAwardKey] then
          LogErr("RandItem award is not exsit. nAwardId, nAwardKey: ", nAwardId, nAwardKey, ". TemplateId: ", tbAItemInfo.TemplateId)
          return false
        end
      end
      if tbAItemInfo.ClassName == "BuffItem" then
        local nBuffId = tbAItemInfo.ExtParam1
        if nil == FightSkill.tbAllSkillSetting[nBuffId] then
          LogErr("BuffItem buff ID is not exsit. TemplateId: ", tbAItemInfo.TemplateId)
          return false
        end
        local nBuffTime = tbAItemInfo.ExtParam3
        local nTimeTooLongJudgeValue = 1800
        if nBuffTime <= 0 then
          LogErr("BuffItem buff time is less then or equal 0. TemplateId: ", tbAItemInfo.TemplateId)
          return false
        elseif nBuffTime > nTimeTooLongJudgeValue then
          LogErr("BuffItem buff time is too long, more than ", nTimeTooLongJudgeValue, " second. TemplateId: ", tbAItemInfo.TemplateId)
          return false
        end
      end
      if tbAItemInfo.ClassName == "LuckyBox" then
        local nAwardId = tbAItemInfo.ExtParam1
        local nAwardKey = tbAItemInfo.ExtParam2
        if nil == Award.tbAllAwards[nAwardId] or nil == Award.tbAllAwards[nAwardId].tbRandAwards[nAwardKey] then
          LogErr("LuckyBox award is not exsit. nAwardId, nAwardKey: ", nAwardId, nAwardKey, ". TemplateId: ", tbAItemInfo.TemplateId)
          return false
        end
        if not self.tbCheckTable[tbAItemInfo.ExtParam4] then
          LogErr("LuckyBox Guaranteed is not exsit. GuaranteedTemplateId: ", tbAItemInfo.ExtParam4)
          return false
        end
        local bHaveGuaranteed = false
        local tbAllAward = Award:GetAllAwardNoRand(nAwardId, nAwardKey)
        for _, tbAward in ipairs(tbAllAward) do
          if tbAward[2] == tbAItemInfo.ExtParam4 then
            bHaveGuaranteed = true
          end
        end
        if not bHaveGuaranteed then
          LogErr("LuckyBox Random rewards do not include guaranteed rewards. nAwardId, nAwardKey: ", nAwardId, nAwardKey, ". guaranteedTemplateId: ", tbAItemInfo.ExtParam4)
          return false
        end
      end
    end
    return true
  end)
end

Item:ItemTableCheck()
