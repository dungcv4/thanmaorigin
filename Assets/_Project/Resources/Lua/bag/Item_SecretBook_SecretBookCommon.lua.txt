Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
Require("CommonScript/Faction/Faction.lua")
Require("CommonScript/Player/FightPowerCommom.lua")
Require("CommonScript/Item/ItemForEquip.lua")
local SecretBook = Item.SecretBook

function SecretBook:GetLevel(pEquip)
  local nLevel = pEquip.GetIntValue(self.tbItemSaveKey.Level)
  if 0 == nLevel then
    nLevel = 1
  end
  return nLevel
end

function SecretBook:GetAttributeValue(pEquip, nAttribute)
  local nValue = pEquip.GetIntValue(self.tbAttributeToEquipSaveKey[nAttribute])
  if 0 == nValue then
    local nInitValue = self.tbEachRankCfg[self:GetRank(pEquip)].nInitAttributeValue
    nValue = nInitValue
  end
  return nValue
end

function SecretBook:GetAllAttributeValue(pEquip)
  local tbRet = {}
  for _, nAttribute in pairs(self.tbAttribute) do
    tbRet[nAttribute] = self:GetAttributeValue(pEquip, nAttribute)
  end
  return tbRet
end

function SecretBook:GetHuishouXiuweiNum(pItem)
  if not pItem then
    return 0
  end
  local nRank = self:GetRank(pItem)
  if not nRank then
    return 0
  end
  local nLevelUpCostXiuWeiSum = 0
  for nUpgradeLevel = 2, self:GetLevel(pItem) do
    nLevelUpCostXiuWeiSum = nLevelUpCostXiuWeiSum + self:GetLevelUpCostXiuWei(nRank, nUpgradeLevel)
  end
  local nReturnXiuWei = Lib:MathRound(XiuWei.Cfg.fSellReturnXiuWeiRate * nLevelUpCostXiuWeiSum)
  return nReturnXiuWei
end

SecretBook.tbFactionSectTypeToAttribute = {
  [Faction.tbSectType.Inner] = function()
    return "dexterity_v", i18n.EasyGet("\229\134\133\229\138\159")
  end,
  [Faction.tbSectType.Outer] = function()
    return "strength_v", i18n.EasyGet("\229\138\155\233\135\143")
  end
}
local tbAttributeEnumToName = {
  [SecretBook.tbAttribute.StrengthOrDexterity] = function(emPlayerSectType)
    return SecretBook.tbFactionSectTypeToAttribute[emPlayerSectType]()
  end,
  [SecretBook.tbAttribute.Vitality] = function(emPlayerSectType)
    return "vitality_v", i18n.EasyGet("\228\189\147\232\180\168")
  end,
  [SecretBook.tbAttribute.Energy] = function(emPlayerSectType)
    return "energy_v", i18n.EasyGet("\232\186\171\230\179\149")
  end
}

function SecretBook:GetMagicAttributeName(pPlayer, nSecretAttribute)
  return self:GetMagicAttributeNameByFactionSect(nSecretAttribute, pPlayer.nFaction, pPlayer.nFactionSect)
end

function SecretBook:GetMagicAttributeNameByFactionSect(nSecretAttribute, nFaction, nFactionSect)
  local emSectType = Faction:GetSectType(nFaction, nFactionSect)
  return tbAttributeEnumToName[nSecretAttribute](emSectType)
end

function SecretBook:GetSkillId(nFaction, nSect, nSecretBookRank)
  return SecretBook.tbSkillData[nFaction][nSect][nSecretBookRank].nSkillId
end

function SecretBook:GetLevelUpCostXiuWei(nRank, nAfterUpgradeLevel)
  return SecretBook.tbCostXiuWei[nRank][nAfterUpgradeLevel].nCostXiuWei
end

function SecretBook:GetSkillLevel(pEquip)
  local nValue = pEquip.GetIntValue(self.tbItemSaveKey.SkillLevel)
  if 0 == nValue then
    nValue = 1
  end
  return nValue
end

function SecretBook:GetExtraSkillLevel(pPlayer, nRank)
  if pPlayer.tbSecretBookExtraSkillLevel and pPlayer.tbSecretBookExtraSkillLevel[nRank] then
    return pPlayer.tbSecretBookExtraSkillLevel[nRank]
  end
  return 0
end

function SecretBook:GetWearSecretBook(pPlayer, nRank)
  return Item.ExtEquip:GetEquipObj(pPlayer, self.tbRankToSlot[nRank])
end

function SecretBook:IsWearThisRankSecretBook(pPlayer, nSecretBookRank)
  return self:GetWearSecretBook(pPlayer, nSecretBookRank) ~= nil
end

function SecretBook:IsSecretBook(pEquip)
  return self:GetRank(pEquip) ~= nil
end

function SecretBook:GetRank(pEquip)
  local nSlot = Item:GetExtEquipSlot(pEquip)
  if nSlot > 0 then
    return self.tbSlotToRank[nSlot]
  else
    return nil
  end
end

function SecretBook:IsLegalRank(nRank)
  return self.tbEachRankCfg[nRank] ~= nil
end

function SecretBook:IsPlayerLevelEnoughWear(pPlayer, nRank)
  return pPlayer.nLevel >= SecretBook.tbEachRankCfg[nRank].nOpenLevel
end

function SecretBook:IsWearLevelEnough(nLevel, nRank)
  return nLevel >= self.tbEachRankCfg[nRank].nOpenLevel
end

function SecretBook:CanLevelUp(pPlayer, nRank)
  local pEquip = self:GetWearSecretBook(pPlayer, nRank)
  if nil ~= pEquip and self:IsPlayerLevelEnoughWear(pPlayer, nRank) then
    local nSecretBookLevel = self:GetLevel(pEquip)
    return nSecretBookLevel < SecretBook.Cfg.nMaxLevel and XiuWei:GetValue(pPlayer) >= self:GetLevelUpCostXiuWei(nRank, nSecretBookLevel + 1)
  else
    return false
  end
end

function SecretBook:CanUseLevelUpVolume(pPlayer, nRank)
  local pEquip = self:GetWearSecretBook(pPlayer, nRank)
  if nil == pEquip then
    return false, i18n.Get(self.tbText.szUseVolumeWhenNotEquipTips, i18n.Get(self.tbEachRankCfg[nRank].szSecretBookName))
  elseif self:GetLevel(pEquip) < self.Cfg.nMaxLevel then
    return false, i18n.Get(self.tbText.szUseVolumeWhenSecretBookNotFullLevelTips, self.Cfg.nMaxLevel)
  elseif self:GetSkillLevel(pEquip) < self.Cfg.nUseLevelUpVolumeMinSkillLevelRequire then
    return false, i18n.Get(self.tbText.szUseVolumeWhenSkillLevelToLowTips, self.Cfg.nUseLevelUpVolumeMinSkillLevelRequire, self.Cfg.nMaxSkillLevel - 1)
  elseif self:GetSkillLevel(pEquip) >= self.Cfg.nMaxSkillLevel then
    return false, i18n.Get(self.tbText.szUseVolumeWhenSkillLevelFullTips, self.Cfg.nMaxSkillLevel)
  else
    return true
  end
end

function SecretBook:OnUseOrUnuse(pPlayer, pEquip, bIsUseApply)
  self:OnUseOrUnuseHandleAttribute(pPlayer, pEquip, bIsUseApply)
  local nSkillLevel = self:GetSkillLevel(pEquip)
  local nRank = self:GetRank(pEquip)
  nSkillLevel = nSkillLevel + self:GetExtraSkillLevel(pPlayer, nRank)
  self:ApplyOrRemoveSkill(pPlayer, self:GetSkillId(pPlayer.nFaction, pPlayer.nFactionSect, nRank), nSkillLevel, bIsUseApply)
end

function SecretBook:ApplyAttrib(pPlayer, pEquip)
  self:OnUseOrUnuse(pPlayer, pEquip, true)
end

function SecretBook:RemoveAttrib(pPlayer, pEquip)
  self:OnUseOrUnuse(pPlayer, pEquip, false)
end

function SecretBook:OnUseOrUnuseHandleAttribute(pPlayer, pEquip, bIsUseApply)
  for nAttribute, nValue in ipairs(self:GetAllAttributeValue(pEquip)) do
    self:ApplyOrRemoveAnAttributeByEnum(pPlayer, nAttribute, nValue, bIsUseApply)
  end
end

function SecretBook:ApplyOrRemoveSkill(pPlayer, nSkillId, nSkillLevel, bIsApply)
  if bIsApply then
    pPlayer.GetNpc().AddSkill(nSkillId, nSkillLevel, 0, 1)
  else
    pPlayer.GetNpc().RemoveSkill(nSkillId)
  end
end

function SecretBook:ApplyOrRemoveAnAttributeByEnum(pPlayer, nAttribute, nValue, bIsApply)
  local szName = self:GetMagicAttributeName(pPlayer, nAttribute)
  self:ApplyOrRemoveAnAttributeByIdentify(pPlayer.GetNpc(), szName, nValue, bIsApply, pPlayer.dwID)
end

function SecretBook:ApplyOrRemoveAnAttributeByIdentify(pNpc, szIdentify, nValue, bIsApply, nPlayerId)
  if bIsApply then
    pNpc.ApplyMagicAttrib(szIdentify, nValue, 0, 0)
    if nPlayerId then
      AttributeHelp:LogAttrApply(nPlayerId, "SecretBook", szIdentify, nValue, 0, 0)
    end
  else
    pNpc.RemoveMagicAttrib(szIdentify, nValue, 0, 0)
    if nPlayerId then
      AttributeHelp:LogAttrRemove(nPlayerId, "SecretBook", szIdentify, nValue, 0, 0)
    end
  end
end

function SecretBook:CalcFightPower(pPlayer)
  local nSumPower = 0
  for nRank = 1, Lib:CountTB(self.tbRank) do
    local pEquip = self:GetWearSecretBook(pPlayer, nRank)
    if nil ~= pEquip then
      nSumPower = nSumPower + self:GetFightPower(pEquip)
    end
  end
  return nSumPower
end

function SecretBook:GetFightPower(pEquip)
  return self:GetFightPowerByLevelAndRank(self:GetLevel(pEquip), self:GetRank(pEquip))
end

function SecretBook:GetFightPowerByLevelAndRank(nLevel, nRank)
  return nLevel * self.tbEachRankCfg[nRank].nFightPowerALevel
end

function SecretBook:onApplyRobotNpcEquip(pPlayerAsync, pNpc, pEquip)
  local nFaction = pPlayerAsync.GetFaction()
  local nFactionSect = pPlayerAsync.GetFactionSect()
  local tbAttributeList = self:GetAllAttributeValue(pEquip)
  if tbAttributeList and next(tbAttributeList) then
    for nAttribute, nValue in ipairs(tbAttributeList) do
      local szName = self:GetMagicAttributeNameByFactionSect(nAttribute, nFaction, nFactionSect)
      self:ApplyOrRemoveAnAttributeByIdentify(pNpc, szName, nValue, true)
    end
  end
  local nSkillId = self:GetSkillId(nFaction, nFactionSect, self:GetRank(pEquip))
  local nSkillLevel = self:GetSkillLevel(pEquip)
  if pPlayerAsync.tbSecretBookExtraSkillLevel then
    local nRank = self:GetRank(pEquip)
    nSkillLevel = nSkillLevel + (pPlayerAsync.tbSecretBookExtraSkillLevel[nRank] or 0)
  end
  if nSkillId and nSkillLevel then
    pNpc.AddSkill(nSkillId, nSkillLevel, 0, 1)
  end
end

function SecretBook:onApplyEquip(pPlayer, pEquip)
  self:ApplyAttrib(pPlayer, pEquip)
end

function SecretBook:onRemoveEquip(pPlayer, pEquip)
  self:RemoveAttrib(pPlayer, pEquip)
end
