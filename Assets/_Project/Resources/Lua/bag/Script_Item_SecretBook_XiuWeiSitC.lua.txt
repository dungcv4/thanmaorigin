Require("CommonScript/Item/SecretBook/SecretBookDef.lua")

function XiuWei:OnS2CBeginSit()
  Ui.ProgressBar:OpenAutoProgressBar(i18n.Get(self.tbText.szXiuWeiSitInProgress), self.Cfg.nSitTime * 1000, self.EndProgress, {self})
end

function XiuWei:EndProgress()
  Ui.ProgressBar:Close()
end

function XiuWei:OnS2CSitAddXiuWei(nAddValue)
  local szMsg = i18n.Get(self.tbText.szSitGetXiuWeiTips, nAddValue)
  me.CenterMsg(szMsg)
  ChatMgr:OnChannelMessage(ChatMgr.ChannelType.System, 0, ChatMgr.SystemMsgType.System, "", 0, 0, 0, szMsg)
end
