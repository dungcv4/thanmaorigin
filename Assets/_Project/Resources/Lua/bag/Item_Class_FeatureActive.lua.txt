local tbFeatureActive = Item:GetClass("FeatureActive")
tbFeatureActive.EXT_PARAM_1 = 1

function tbFeatureActive:OnUse(pItem)
  if not pItem.dwTemplateId then
    return 0
  end
  if Horse:ActiveHorseFeature(me, pItem.dwTemplateId) then
    local szName = KItem.GetItemShowInfo(pItem.dwTemplateId, me.nFaction, me.nFactionSect, me.nSex)
    me.CenterMsg(i18n.EasyGet("\232\142\183\229\190\151{0}", i18n.Parse(szName)))
    return 1
  end
  return 0
end
