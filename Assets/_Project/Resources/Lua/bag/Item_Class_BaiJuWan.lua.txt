local tbItem = Item:GetClass("BaiJuWan")
tbItem.nAddTime = 28800
tbItem.nPrice = 48

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  local bOpen, szMsg = OfflineExp:CheckIsOpen(me)
  if not bOpen then
    me.CenterMsg(szMsg)
    return
  end
  local nRet = OfflineExp:OnUseBaiJuWan(me, self.nAddTime, OfflineExp.OfflineExpType.Pay)
  if not nRet then
    return 0
  end
  return 1
end
