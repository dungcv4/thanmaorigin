local tbItemClass = Item:GetClass("RedStone")
tbItemClass.PARAM_NPCSHOPID = 1

function tbItemClass:OnClientUse(it)
  local nNpcShopId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPCSHOPID)
  Ui:OpenWindow("UINpcShop", nNpcShopId, nil, nil, nil, nil, Currency.TYPE.RedStone, true)
  return 0
end

function tbItemClass:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "RedStone")
end
