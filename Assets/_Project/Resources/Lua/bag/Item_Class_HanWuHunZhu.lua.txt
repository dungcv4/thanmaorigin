local tbItem = Item:GetClass("HanWuHunZhu")

function tbItem:OnClientUse(it)
  local tbInfo = Item:GetClass("CountAddProp"):GetAddDegreeInfo("HanWuYiJi")
  if not tbInfo then
    me.CenterMsg(i18n.EasyGet("\231\188\186\229\176\145\229\175\146\230\173\166\233\173\130\231\143\160\233\133\141\231\189\174"))
    return 0
  end
  local nCurNum = DegreeCtrl:GetDegree(me, "HanWuYiJi")
  local nTotalNum = DegreeCtrl:GetMaxDegree("HanWuYiJi", me)
  if nCurNum >= nTotalNum then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\143\130\228\184\142\230\172\161\230\149\176\229\183\178\232\190\190\228\184\138\233\153\144"))
    return 0
  end
  if DegreeCtrl:GetDegree(me, "HanWuHunZhu") > 0 then
    return 1
  else
    me.CenterMsg(i18n.EasyGet("\230\130\168\228\187\138\230\151\165\229\183\178\231\187\143\228\189\191\231\148\168\232\191\135\229\175\146\230\173\166\233\173\130\231\143\160\228\186\134"))
    return 0
  end
end

function tbItem:OnUse(it)
  if DegreeCtrl:GetDegree(me, "HanWuHunZhu") <= 0 then
    return 0
  end
  if DegreeCtrl:ReduceDegree(me, "HanWuHunZhu", 1) then
    DegreeCtrl:AddDegree(me, "HanWuYiJi", 1)
    me.CallClientScript("HanWuYiJi:RefreshSignUpView")
    me.CenterMsg(i18n.EasyGet("\229\175\146\230\173\166\233\129\151\232\191\185\229\143\130\228\184\142\230\172\161\230\149\176\229\162\158\229\138\160\228\184\128\230\172\161"))
    return 1
  else
    return 0
  end
end
