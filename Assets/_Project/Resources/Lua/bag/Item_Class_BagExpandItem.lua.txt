Require("CommonScript/Item/ItemDefine.lua")
local tbBagExpandItem = Item:GetClass("BagExpandItem")
tbBagExpandItem.PARAM_EXPAND_BAG_COUNT = 1

function tbBagExpandItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbBagExpandItem:OnUseMany(it, nCount)
  if not me then
    return 0
  end
  local nOrgCount = me.GetMaxCountInBag()
  local nExpandCount = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_EXPAND_BAG_COUNT)
  local nExtMaxCnt = 0
  if Item.ZhenYuan:IsSystemOpen(me) then
    nExtMaxCnt = Item.ZhenYuan.nBagMaxCnt
  end
  local nRemainCount = Item.MAX_BAG_EXPAND + nExtMaxCnt - nOrgCount
  local nCurEx = me.GetBagExpandCount()
  local nCanUse = math.min(math.floor(nRemainCount / nExpandCount), nCount)
  if nCanUse > 0 then
    me.SetBagExpandCount(nCurEx + nExpandCount * nCount)
    return nCanUse
  else
    me.CallClientScript("Item:ExpandBagCountSucc", false, i18n.EasyGet("\232\131\140\229\140\133\229\174\185\233\135\143\229\183\178\232\190\190\233\129\147\229\133\183\228\189\191\231\148\168\228\184\138\233\153\144"))
    return 0
  end
end

function tbBagExpandItem:GetClientDefaultUseCount(nItemId)
  if not me then
    return 0
  end
  local pItem = me.GetItemInBag(nItemId)
  local nOrgCount = me.GetMaxCountInBag()
  local nExpandCount = KItem.GetItemExtParam(pItem.dwTemplateId, self.PARAM_EXPAND_BAG_COUNT)
  local nExtMaxCnt = 0
  if Item.ZhenYuan:IsSystemOpen(me) then
    nExtMaxCnt = Item.ZhenYuan.nBagMaxCnt
  end
  local nRemainCount = Item.MAX_BAG_EXPAND + nExtMaxCnt - nOrgCount
  local nCanUse = math.min(math.floor(nRemainCount / nExpandCount), pItem.nCount)
  if nCanUse <= 0 then
    me.CallClientScript("Item:ExpandBagCountSucc", false, i18n.EasyGet("\232\131\140\229\140\133\229\174\185\233\135\143\229\183\178\232\190\190\233\129\147\229\133\183\228\189\191\231\148\168\228\184\138\233\153\144"))
    return 0
  end
  return nCanUse
end
