local tbItem = Item:GetClass("FixedTimeFrameBox")
tbItem.PARAM_AWARD_ID = 1
tbItem.PARAM_AWARD_KEY = 2
tbItem.PARAM_DATETIME_ID = 3
tbItem.OFFSET = 14400

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  local nAwardId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARD_ID)
  local nAwardKey = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARD_KEY)
  local nParamId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_DATETIME_ID)
  local nTime = Item.FixedTimeFramBox:GetTime(nParamId)
  if not (nAwardId and nAwardKey) or not nTime then
    LogErr("[FixedTimeFramBox] OnUseMany config is wrong", it.dwTemplateId, nAwardId, nAwardKey, nParamId)
    return 0
  end
  local nCreateTime = GetServerCreateTime() - self.OFFSET
  local nSpecifiedTime = nTime - self.OFFSET
  local nSpecifiedDay = Lib:GetLocalDay(nSpecifiedTime)
  local nOffsetDay = nSpecifiedDay - Lib:GetLocalDay(nCreateTime) + 1
  if nOffsetDay <= 0 then
    nOffsetDay = 1
  end
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  local nNum, nRet = 0
  local szMsg = it.szName
  local nSuccessRet = 1
  local tbAwardListForTrack, tbLastAwards = {}
  for _ = 1, nCount do
    nRet = Award:CommonSendAward(me, nAwardId, nAwardKey, it.dwTemplateId, szMsg, {nAwardTime = nOffsetDay}, true)
    if nSuccessRet == nRet then
      nNum = nNum + 1
      tbLastAwards = Award:FetchLastAwards()
      Lib:MergeTable(tbAwardListForTrack, tbLastAwards)
    end
  end
  local nConsumeItemTemplateId = it.dwTemplateId
  local nExpendItemId = 0
  local nExpendItemNum = 0
  tbAwardListForTrack = Award:MergetSameTypeAward(tbAwardListForTrack)
  local tbMoneyForTrack, tbItemForTrack = Award:GetAwardListForMBILog(tbAwardListForTrack, true, {
    "Gold",
    "YuanBao",
    "Coin"
  })
  Lib:TrackEvent(me.dwID, Env.track_openbox_open, nConsumeItemTemplateId, nCount, nExpendItemId, nExpendItemNum * nCount, tbMoneyForTrack.Gold or 0, tbMoneyForTrack.YuanBao or 0, tbMoneyForTrack.Coin or 0, Lib:EncodeJson(tbItemForTrack) or "")
  return nNum
end

function tbItem:OnAutoUse(pPlayer, nItemTemplateId, nCount, szMsg, tbItemExtParams, nLogReason, nLogReason2)
  local nAwardId = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_AWARD_ID)
  local nAwardKey = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_AWARD_KEY)
  local nParamId = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_DATETIME_ID)
  local nTime = Item.FixedTimeFramBox:GetTime(nParamId)
  if not (nAwardId and nAwardKey) or not nTime then
    LogErr("[FixedTimeFramBox] OnAutoUse config is wrong", nItemTemplateId, nAwardId, nAwardKey, nParamId, nCount)
    return false
  end
  local nCreateTime = GetServerCreateTime() - self.OFFSET
  local nSpecifiedTime = nTime - self.OFFSET
  local nSpecifiedDay = Lib:GetLocalDay(nSpecifiedTime)
  local nOffsetDay = nSpecifiedDay - Lib:GetLocalDay(nCreateTime) + 1
  if nOffsetDay <= 0 then
    nOffsetDay = 1
  end
  local szItemName = Item:GetName(nItemTemplateId, pPlayer)
  local tbAwards = {}
  local tbTmpAward, bSuccess
  tbItemExtParams = tbItemExtParams or {}
  tbItemExtParams.nAwardTime = nOffsetDay
  local tbAwardListForTrack = {}
  for i = 1, nCount do
    bSuccess, tbTmpAward = Award:SendAutoUseAward(pPlayer, nAwardId, nAwardKey, nItemTemplateId, szItemName, true, false, true, tbItemExtParams, nLogReason, nLogReason2)
    if tbTmpAward then
      Lib:MergeTable(tbAwards, tbTmpAward)
      Lib:MergeTable(tbAwardListForTrack, tbTmpAward)
    end
  end
  local nExpendItemId = 0
  local nExpendItemNum = 0
  tbAwardListForTrack = Award:MergetSameTypeAward(tbAwardListForTrack)
  local tbMoneyForTrack, tbItemForTrack = Award:GetAwardListForMBILog(tbAwardListForTrack, true, {
    "Gold",
    "YuanBao",
    "Coin"
  })
  Lib:TrackEvent(pPlayer.dwID, Env.track_openbox_open, nItemTemplateId, nCount, nExpendItemId, nExpendItemNum * nCount, tbMoneyForTrack.Gold or 0, tbMoneyForTrack.YuanBao or 0, tbMoneyForTrack.Coin or 0, Lib:EncodeJson(tbItemForTrack) or "")
  return true, tbAwards
end
