local tbMercenary = Item:GetClass("GlobalDomainBattleMercenary")
tbMercenary.EXT_PARAM_NPC = 1

function tbMercenary:OnUse(pItem)
  local bRet, szErr = GlobalDomainBattle:CanHire(me, pItem)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  local szNpcName = KNpc.GetNameByTemplateId(KItem.GetItemExtParam(pItem.dwTemplateId, self.EXT_PARAM_NPC))
  me.MsgBox(i18n.EasyGet("\230\152\175\229\144\166\229\156\168\229\189\147\229\137\141\228\189\141\231\189\174\229\143\172\229\148\164{0}", szNpcName), {
    {
      i18n.szOk,
      self.SureUse,
      self,
      pItem.dwId
    },
    {
      i18n.szCancel
    }
  })
end

function tbMercenary:SureUse(nItemId)
  local pItem = me.GetItemInAllSpace(nItemId)
  if not pItem then
    me.CenterMsg(i18n.EasyGet("\231\137\169\229\147\129\229\183\178\229\164\177\230\149\136"))
    return
  end
  local bRet, szErr = GlobalDomainBattle:HireReq(me, pItem)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
end

function tbMercenary:OnClientUse()
  return 1
end

function tbMercenary:Give(nItemTemplateId, nItemId)
  Ui:CloseWindow("UIItemTips")
  Ui:OpenWindow("UIExperienceBook", nItemTemplateId, nItemId, "GlobalDomainbattleItem", false, false, "Legion")
end

function tbMercenary:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    Item:ClientUseItem(nItemId)
  end
  
  tbSetting.szSecondName = i18n.EasyGet("\232\181\160\233\128\129")
  
  function tbSetting.fnSecond()
    tbMercenary:Give(nTemplateId, nItemId)
  end
  
  return tbSetting
end
