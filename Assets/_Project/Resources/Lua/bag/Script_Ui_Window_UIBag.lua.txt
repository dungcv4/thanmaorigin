function tbWnd:OnCreate()
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self:SetJingHuoNum()
  self.tbwndJingZhuInfo = Ui:CreatePanel("UIwndJingZhuInfo", tbControls.wndJingZhuInfo, self)
  self.tbwndSeriesActive = Ui:CreatePanel("UISeriesActive", tbControls.wndSeriesActive, self)
  self.tbwndQiHunInfo = Ui:CreatePanel("UIwndQiHunInfo", tbControls.wndQiHunInfo, self)
  self.tbSubAttribute = Ui:CreatePanel("UISubAttribute", tbControls.wndSubAttribute, self)
  self.tbClothesPanelAttribute = Ui:CreatePanel("UIClothesPanelAttribute", tbControls.wndClosetPanelAttribute, self)
  self.tbClosetShowSetting = Ui:CreatePanel("UIClosetShowSetting", tbControls.wndClosetShowSetting, self)
  self:SetClickEmptyToClose(tbControls.wndSelectHorse)
  self:SetClickEmptyToClose(tbControls.wndClosetShowSetting)
  self:SetClickEmptyToClose(tbControls.wndClosetPanelAttribute)
end

function tbWnd:OnOpen(nSelectPanel, ...)
  if DingFengPK:CheckInDingFengPK() then
    nSelectPanel = 2
  end
  if not self.tbTabControl:OnOpen(nSelectPanel, ...) then
    return
  end
  self.pPanel:SetActive(tbControls.wndJingZhuInfo, false)
  self.tbMoneyPanel:SetMoneyValue()
  self.tbMoneyPanel:SetIconsShow()
  self:RegistRedPoint()
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  self:UnRegistRedPoint()
  if self.nShowModelTimer then
    Timer:Close(self.nShowModelTimer)
    self.nShowModelTimer = nil
  end
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnAttribute, {
    RedPointMgr.EVENT.PotentialPointLeft
  })
  local tbEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmEnhance, RedPointMgr.EVENT.AmuletEnhance do
    table.insert(tbEventList, nRedPoint)
  end
  table.insert(tbEventList, RedPointMgr.EVENT.ZhenFaLevelUp)
  for nRedPoint = RedPointMgr.EVENT.ZhenFaZJLevelUpBegin, RedPointMgr.EVENT.ZhenFaZJLevelUpEnd do
    table.insert(tbEventList, nRedPoint)
  end
  table.insert(tbEventList, RedPointMgr.EVENT.ShenBingBagEntrance)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnBag, tbEventList)
  local tbHorseEventList = {
    RedPointMgr.EVENT.HorseLevel
  }
  for i = 1, Horse.nHorseEquipCount do
    table.insert(tbHorseEventList, RedPointMgr.EVENT.HorseEquipGrade1 + i - 1)
  end
  for i = RedPointMgr.EVENT.HorseSkillLevelBegin, RedPointMgr.EVENT.HorseSkillLevelEnd do
    table.insert(tbHorseEventList, i)
  end
  for i = RedPointMgr.EVENT.HorseSkillStoneBegin, RedPointMgr.EVENT.HorseSkillStoneEnd do
    table.insert(tbHorseEventList, i)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnHorse, tbHorseEventList)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnCloset, {
    RedPointMgr.EVENT.FashionPrettyStageAward
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnAttribute)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnBag)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnHorse)
end

function tbWnd:ShowModel()

    Timer:Close(self.nShowModelTimer)
    self.nShowModelTimer = nil
  end
  self.nShowModelTimer = Timer:Register(1, self.DoShowModel, self)
end

  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_BAG, "ShowModel")
