Require("CommonScript/Item/ItemDefine.lua")
local tbQingChunQuan = Item:GetClass("QingChunQuan")

function tbQingChunQuan:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbQingChunQuan:OnTimeOut(pItem, pPlayer)
  local nItemTid = pItem.dwTemplateId
  local nCount = pItem.nCount
  local szName = pItem.szName
  local szCurrencyType, nNum = Item.tbItemSell:GetSellCurrencyValue(nItemTid)
  if not szCurrencyType then
    return
  end
  if nCount ~= pPlayer.ConsumeItem(pItem, nCount, Env.LogWay_TimeOut) then
    Log("qingchunquan:OnTimeOut", pPlayer.szName, "pid:", pPlayer.dwID, "item:", nItemTid, "count:", nCount, "errConsumeItem")
    return
  end
  local nTotalNum = nNum * nCount
  local tbAward = {
    {szCurrencyType, nTotalNum}
  }
  pPlayer.SendAward(tbAward, true, true, Env.LogWay_TimeOut, nil, nil)
  local szMsg = i18n.EasyGet("\230\130\168\230\156\137{0}\228\184\170{1}\229\183\178\232\191\135\230\156\159\239\188\140\232\191\148\232\191\152\230\130\168{2}{3}", nCount, szName, nTotalNum, Currency:GetName(szCurrencyType))
  pPlayer.Msg(szMsg)
  Log("qingchunquan:OnTimeOut", pPlayer.szName or "", "pid:" .. (pPlayer.dwID or 0), "item:" .. nItemTid, "count:" .. nCount)
end

function tbQingChunQuan:OnUseMany(it, nCount)
  if not it.dwTemplateId then
    return 0
  end
  local nQingChunQuanID = 28
  local pInstance = ActivityMgr:GetInstanceById(nQingChunQuanID)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\228\184\141\229\156\168\230\180\187\229\138\168\230\156\159"))
    return
  end
  local nOptType = KItem.GetItemExtParam(it.dwTemplateId, 1)
  if 1 == nOptType then
    pInstance:HandInQuan(0, nCount, me.dwID)
  elseif 2 == nOptType then
    pInstance:HandInQuan(nCount, 0, me.dwID)
  end
end

function tbQingChunQuan:GetClientDefaultUseCount(nItemId)
  local pItem = me.GetItemInBag(nItemId)
  local nResult = math.min(pItem.nCount, Item.UseManyLimit.QingChunQuan)
  return nResult
end
