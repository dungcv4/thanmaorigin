local tbItem = Item:GetClass("SnowFightSpecialTicket")

function tbItem:OnClientUse(it)
  local pSnowFightAct = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.SnowFightActivity)
  if not pSnowFightAct or not pSnowFightAct:IsOpen() then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
    return 0
  end
  local nHasUseSpecialTicketToday = me.GetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nUseSpecialTicketCountToday)
  local nTotalUseSpecialTicket = me.GetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nTotalUseSpecialTicketCount)
  if false then
    me.CenterMsg(i18n.EasyGet("{0}\230\175\143\229\164\169\229\143\170\232\131\189\228\189\191\231\148\168{1}\230\172\161\239\188\140\232\175\183\230\152\142\230\151\165\229\134\141\229\176\157\232\175\149\227\128\130", Item:GetName(pSnowFightAct.tbConfig.nSpecialTicketItemId, me), pSnowFightAct.tbConfig.nMaxUseSpecialTicketCountPerDay))
    return 0
  elseif nTotalUseSpecialTicket >= pSnowFightAct.tbConfig.nMaxUseSpecialTicketCount then
    me.CenterMsg(i18n.EasyGet("{0}\229\183\178\232\190\190\230\180\187\229\138\168\230\156\159\233\151\180\231\154\132\228\189\191\231\148\168\228\184\138\233\153\144", Item:GetName(pSnowFightAct.tbConfig.nSpecialTicketItemId, me)))
    return 0
  else
    return 1
  end
end

function tbItem:OnUse(it)
  local pSnowFightAct = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.SnowFightActivity)
  if not pSnowFightAct or not pSnowFightAct:IsOpen() then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
    return 0
  end
  local nHasUseSpecialTicketToday = me.GetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nUseSpecialTicketCountToday)
  local nTotalUseSpecialTicket = me.GetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nTotalUseSpecialTicketCount)
  if false then
    me.CenterMsg(i18n.EasyGet("{0}\230\175\143\229\164\169\229\143\170\232\131\189\228\189\191\231\148\168{1}\230\172\161\239\188\140\232\175\183\230\152\142\230\151\165\229\134\141\229\176\157\232\175\149\227\128\130", Item:GetName(pSnowFightAct.tbConfig.nSpecialTicketItemId, me), pSnowFightAct.tbConfig.nMaxUseSpecialTicketCountPerDay))
    return 0
  elseif nTotalUseSpecialTicket >= pSnowFightAct.tbConfig.nMaxUseSpecialTicketCount then
    me.CenterMsg(i18n.EasyGet("{0}\229\183\178\232\190\190\230\180\187\229\138\168\230\156\159\233\151\180\231\154\132\228\189\191\231\148\168\228\184\138\233\153\144", Item:GetName(pSnowFightAct.tbConfig.nSpecialTicketItemId, me)))
    return 0
  end
  me.SetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nUseSpecialTicketCountToday, nHasUseSpecialTicketToday + 1)
  me.SetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nTotalUseSpecialTicketCount, nTotalUseSpecialTicket + 1)
  local nRemainJoinCount = me.GetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nRemainJoinCount)
  me.SetUserValue(SnowFight.SaveKey, SnowFight.SaveID.nRemainJoinCount, nRemainJoinCount + 1)
  me.CenterMsg(i18n.EasyGet("{0}\228\189\191\231\148\168\230\136\144\229\138\159\239\188\140\230\130\168\229\162\158\229\138\160\228\186\1341\230\172\161\230\175\148\232\181\155\230\172\161\230\149\176", Item:GetName(it.dwTemplateId, me)))
  return 1
end
