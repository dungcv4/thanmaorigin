Require("CommonScript/Item/EquipSeriesActive/EquipSeriesActiveCfg.lua")
local EquipSeriesActive = Item.EquipSeriesActive
EquipSeriesActive.bAttrApplyLock = false
EquipSeriesActive.bAttrRemoveLock = false

function EquipSeriesActive:ApplyAttrib(pPlayer, pEquip)
  if pEquip.nEquipPos < Item.EQUIPPOS_HELM or pEquip.nEquipPos > Item.EQUIPPOS_AMULET then
    return
  end
  if self.bAttrApplyLock then
    return
  end
  self:UpdatePerEquipAttr(pPlayer, pEquip, true)
end

function EquipSeriesActive:RemoveAttrib(pPlayer, pEquip)
  if pEquip.nEquipPos < Item.EQUIPPOS_HELM or pEquip.nEquipPos > Item.EQUIPPOS_AMULET then
    return
  end
  if self.bAttrRemoveLock then
    return
  end
  self:UpdatePerEquipAttr(pPlayer, pEquip, false)
end

function EquipSeriesActive:StatSeries(pPlayer)
  local tbSeriesList = {}
  for nSeries = Item.emSERIES_TYPE.SERIES_NONE + 1, Item.emSERIES_TYPE.SERIES_NUM - 1 do
    tbSeriesList[nSeries] = 0
  end
  for nPos = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
    local pEquip = pPlayer.GetEquipByPos(nPos)
    if pEquip then
      local nSeries = pEquip.GetSeries()
      tbSeriesList[nSeries] = tbSeriesList[nSeries] + 1
    end
  end
  return tbSeriesList
end

function EquipSeriesActive:StatSeriesByAsyncData(pAsyncData)
  local tbSeriesList = {}
  for nSeries = Item.emSERIES_TYPE.SERIES_NONE + 1, Item.emSERIES_TYPE.SERIES_NUM - 1 do
    tbSeriesList[nSeries] = 0
  end
  local tbEquips = pAsyncData.GetEquips()
  for nPos = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
    local pEquip = KItem.GetItemObj(tbEquips[nPos])
    if pEquip then
      local nSeries = pEquip.GetSeries()
      tbSeriesList[nSeries] = tbSeriesList[nSeries] + 1
    end
  end
  return tbSeriesList
end

function EquipSeriesActive:UpdatePerEquipAttr(pPlayer, pEquip, bActive)
  local nEquipSeries = pEquip.GetSeries()
  local tbSeriesList = self:StatSeries(pPlayer)
  if 2 == tbSeriesList[nEquipSeries] then
    local tbAttribInfo = self.tbSeriesAttribList[nEquipSeries]
    if bActive then
      pPlayer.ApplyMagicAttrib(table.unpack(tbAttribInfo))
      AttributeHelp:LogAttrApply(pPlayer.dwID, "EquipSeriesActive", table.unpack(tbAttribInfo))
    else
      pPlayer.RemoveMagicAttrib(table.unpack(tbAttribInfo))
      AttributeHelp:LogAttrRemove(pPlayer.dwID, "EquipSeriesActive", table.unpack(tbAttribInfo))
    end
  end
end

function EquipSeriesActive:UpdateAllEquipAttr(pNpc, bActive, tbSeries)
  for i = 1, 5 do
    if tbSeries[i] >= 2 then
      local tbAttribInfo = self.tbSeriesAttribList[i]
      if bActive then
        pNpc.ApplyMagicAttrib(table.unpack(tbAttribInfo))
      else
        pNpc.RemoveMagicAttrib(table.unpack(tbAttribInfo))
      end
    end
  end
end

function EquipSeriesActive:ApplyAllItemAttrib(pPlayer, bStart)
  self.bAttrApplyLock = bStart
  if not bStart then
    local tbSeriesList = self:StatSeries(pPlayer)
    for nEquipSeries, nEquipNum in pairs(tbSeriesList) do
      if nEquipNum >= 2 then
        local tbAttribInfo = self.tbSeriesAttribList[nEquipSeries]
        pPlayer.ApplyMagicAttrib(table.unpack(tbAttribInfo))
        AttributeHelp:LogAttrApply(pPlayer.dwID, "EquipSeriesActive", table.unpack(tbAttribInfo))
      end
    end
  end
end

function EquipSeriesActive:RemoveAllItemAttrib(pPlayer, bStart)
  self.bAttrRemoveLock = bStart
  if not bStart then
    local tbSeriesList = self:StatSeries(pPlayer)
    for nEquipSeries, nEquipNum in pairs(tbSeriesList) do
      if nEquipNum >= 2 then
        local tbAttribInfo = self.tbSeriesAttribList[nEquipSeries]
        pPlayer.RemoveMagicAttrib(table.unpack(tbAttribInfo))
        AttributeHelp:LogAttrRemove(pPlayer.dwID, "EquipSeriesActive", table.unpack(tbAttribInfo))
      end
    end
  end
end
