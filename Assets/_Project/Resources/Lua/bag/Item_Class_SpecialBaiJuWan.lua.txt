local tbItem = Item:GetClass("SpecialBaiJuWan")
tbItem.nAddTime = 28800
tbItem.nPrice = 240

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  local bOpen, szMsg = OfflineExp:CheckIsOpen(me)
  if not bOpen then
    me.CenterMsg(szMsg)
    return
  end
  if not OfflineExp:CheckSpecialBaiJuWanIsOpen(me) then
    me.CenterMsg(i18n.EasyGet("\232\190\190\229\136\176\227\128\144{0}\227\128\145\230\137\141\229\143\175\228\189\191\231\148\168", Vip:GetVipName(Vip:GetVipCanUseSpecialBaiJuWWanLevel())))
    return
  end
  local bRet = OfflineExp:OnUseBaiJuWan(me, self.nAddTime, OfflineExp.OfflineExpType.SpecialPay)
  if not bRet then
    return 0
  end
  return 1
end
