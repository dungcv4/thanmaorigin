Require("CommonScript/Item/Class/Equip.lua")
Item.MAIN_EQUIP_CLASS_NAME = "MainEquip"
local tbMainEquip = Item:GetClass(Item.MAIN_EQUIP_CLASS_NAME, 1)
tbMainEquip = tbMainEquip or Item:NewClass(Item.MAIN_EQUIP_CLASS_NAME, "equip")
tbMainEquip.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_BEGIN = 1
tbMainEquip.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_END = 6
tbMainEquip.EQUIP_RANDOM_ATTRIB_VALUE_BEGIN = 7
tbMainEquip.EQUIP_RANDOM_ATTRIB_VALUE_END = 15

function tbMainEquip:GetSeriesActiveAttr(pEquip)
  local tbAttrList = {}
  if pEquip.szClass ~= Item.MAIN_EQUIP_CLASS_NAME then
    return tbAttrList
  end
  for nId = self.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_BEGIN, self.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_END do
    table.insert(tbAttrList, pEquip.GetIntValue(nId))
  end
  return tbAttrList
end

function tbMainEquip:SetSeriesActiveAttr(pEquip, tbAttrList)
  if pEquip.szClass ~= Item.MAIN_EQUIP_CLASS_NAME then
    return
  end
  for nId = self.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_BEGIN, self.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_END do
    local nAttr = tbAttrList[nId - self.EQUIP_SERIES_ACTIVE_ATTRIB_VALUE_BEGIN + 1]
    if nAttr then
      pEquip.SetIntValue(nId, nAttr)
    else
      pEquip.SetIntValue(nId, 0)
    end
  end
end

function tbMainEquip:GetRandomAttr(pEquip)
  local tbRandomAttr = Item:GetClass("EquipForgeSeed"):GetRandomAttrInfo(pEquip, self.EQUIP_RANDOM_ATTRIB_VALUE_BEGIN - 1)
  return tbRandomAttr
end

function tbMainEquip:SetRandomAttr(pEquip, tbRandomAttr)
  for nIndex, tbV in ipairs(tbRandomAttr) do
    Item:GetClass("EquipForgeSeed"):SetRandomAttrInfo(pEquip, nIndex, tbV[1], tbV[2], self.EQUIP_RANDOM_ATTRIB_VALUE_BEGIN - 1)
  end
end

function tbMainEquip:GetDetailLog(pEquip)
  if pEquip.szClass ~= Item.MAIN_EQUIP_CLASS_NAME then
    return
  end
  local tbLog = {}
  tbLog.SeriesActiveAttrs = table.concat(self:GetSeriesActiveAttr(pEquip), ",")
  return tbLog
end

function tbMainEquip:GetChatLinkInfo(pEquip)
  if pEquip.szClass ~= Item.MAIN_EQUIP_CLASS_NAME then
    return
  end
  local tbInfo = {}
  tbInfo.tbSeriesActiveAttrib = self:GetSeriesActiveAttr(pEquip)
  tbInfo.tbRandomAttr = self:GetRandomAttr(pEquip)
  return tbInfo
end
