Require("CommonScript/Item/ItemDefine.lua")
local tbPartnerSkillBook = Item:GetClass("PartnerSkillBook")

function tbPartnerSkillBook:OnUse(it)
end

function tbPartnerSkillBook:GetUseSetting(nTemplateId, nItemId)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local tbSetting = {}
  
  local function func()
    RemoteServer.CallPartnerFunc("DestroySkillBook", nTemplateId, 1)
  end
  
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    Partner:OpenPartnerMainUI(1)
  end
  
  tbSetting.szSecondName = i18n.EasyGet("\230\139\134\232\167\163")
  
  function tbSetting.fnSecond()
    if 1 == pItem.nCount then
      local nItemCount = Partner.tbSkillBookSetting.tbBookInfo[nTemplateId].nRemnantsReturnCount
      Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\229\176\134{0}\230\156\172\230\138\128\232\131\189\228\185\166\230\139\134\232\167\163\228\184\186{1}\228\184\170\230\138\128\232\131\189\230\174\139\231\175\135\239\188\159", 1, nItemCount), {
        {func},
        {}
      }, {
        i18n.szOk,
        i18n.szCancel
      })
    else
      Ui:OpenWindow("UIItemBatchPanel", nItemId, nTemplateId)
    end
  end
  
  return tbSetting
end
