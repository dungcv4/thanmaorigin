local tbItem = Item:GetClass("LimitTimeMonthCard")

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  local nAddTime = KItem.GetItemExtParam(it.dwTemplateId, 1)
  if not nAddTime or nAddTime <= 0 then
    return
  end
  if WeekMonthCard:CheckMonthCardState(me) then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\183\178\230\156\137\230\156\136\229\141\161\231\138\182\230\128\129\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168"))
    return
  end
  local nLastEndTime = WeekMonthCard:GetLimitTimeMonthCardEndTime(me)
  local nCurTime = GetTime()
  local nEndTime = 0
  if nLastEndTime < nCurTime then
    nEndTime = nCurTime + nAddTime
  else
    nEndTime = nLastEndTime + nAddTime
  end
  me.SetUserValue(WeekMonthCard.SAVE_GROUP, WeekMonthCard.LimitTimeMonthEndTime, nEndTime)
  me.RemoveSkillState(WeekMonthCard.nLimitTimeBuffId)
  me.AddSkillState(WeekMonthCard.nLimitTimeBuffId, 1, FightSkill.STATE_TIME_TYPE.state_time_truetime, nEndTime, 1)
  me.CallClientScript("WeekMonthCard:OnUseLimitTimeRsp", nAddTime)
  return 1
end

function tbItem:OnTimeOut(pItem, pPlayer)
  local nItemTid = pItem.dwTemplateId
  local nCount = pItem.nCount
  local szName = pItem.szName
  local szCurrencyType, nNum = Item.tbItemSell:GetSellCurrencyValue(nItemTid)
  if not szCurrencyType then
    return
  end
  if nCount ~= pPlayer.ConsumeItem(pItem, nCount, Env.LogWay_TimeOut) then
    Log("LimitTimeMonthCard:OnTimeOut", pPlayer.szName, "pid:" .. pPlayer.dwID, "item:" .. nItemTid, "count:" .. nCount, "errConsumeItem")
    return
  end
  local nTotalNum = nNum * nCount
  local tbMail = {
    To = pPlayer.dwID,
    Title = i18n.EasyGet("{0}\232\191\135\230\156\159", szName),
    Text = i18n.EasyGet("\230\130\168\230\156\137{0}\228\184\170{1}\229\183\178\232\191\135\230\156\159\239\188\140\232\191\148\232\191\152\230\130\168{2}{3}", nCount, szName, nTotalNum, Currency:GetName(szCurrencyType)),
    From = i18n.EasyGet("\231\179\187\231\187\159"),
    tbAttach = {
      {szCurrencyType, nTotalNum}
    },
    nLogReason = Env.LogWay_TimeOut,
    tbParams = {LogReason2 = nItemTid}
  }
  Mail:SendSystemMail(tbMail)
  Log("LimitTimeMonthCard:OnTimeOut", pPlayer.szName, "pid:" .. pPlayer.dwID, "item:" .. nItemTid, "count:" .. nCount)
end
