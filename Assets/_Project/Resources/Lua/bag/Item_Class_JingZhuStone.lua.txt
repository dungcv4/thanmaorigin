local tbItem = Item:GetClass("JingZhuStone")

function tbItem:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbRet = {}
  tbRet.szFirstName = i18n.EasyGet("\231\178\190\233\147\184")
  
  function tbRet.fnFirst()
    Ui:OpenWindow("UIEquipment", Ui.UIEquipment.tbSubPanelIndex.PANEL_TYPE_JINGZHU)
    Ui:CloseWindow("UIItemTips")
  end
  
  if Item.EquipJingZhu.tbSynthesisInfo[nTemplateId] then
    tbRet.szSecondName = i18n.EasyGet("\229\144\136\230\136\144")
    
    function tbRet.fnSecond()
      local tbSynThesisConfig = Item.EquipJingZhu.tbSynthesisInfo[nTemplateId]
      if not tbSynThesisConfig then
        return me.CenterMsg(i18n.EasyGet("\229\144\136\230\136\144\233\133\141\231\189\174\233\148\153\232\175\175"))
      end
      local nConsumeNum = tbSynThesisConfig[2]
      local nTargetId = tbSynThesisConfig[3]
      local nTargetNum = tbSynThesisConfig[4]
      local nNeedCoin = tbSynThesisConfig[5]
      local nCurCoin = me.GetMoney("Coin")
      if nNeedCoin > nCurCoin then
        return me.CenterMsg(i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179"))
      end
      local nCurItemNum = me.GetItemCountByTypeInBag(nTemplateId)
      local nItemCount = math.floor(nCurItemNum / nConsumeNum)
      if nItemCount < 1 then
        return me.CenterMsg(i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179"))
      end
      local nCoinCount = math.floor(nCurCoin / nNeedCoin)
      if 1 == nCoinCount or 1 == nItemCount then
        RemoteServer.RequestEquipJingZhu("JingZhuSynthesisReq", nTemplateId, 1)
      else
        local nMax = nItemCount < nCoinCount and nItemCount or nCoinCount
        Ui:OpenWindow("UIJingZhuCompose", nTemplateId, nMax, nTargetId, nTargetNum)
      end
      Ui:CloseWindow("UIItemTips")
    end
  end
  return tbRet
end
