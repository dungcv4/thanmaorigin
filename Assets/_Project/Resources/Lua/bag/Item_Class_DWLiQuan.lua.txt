local tbItemClass = Item:GetClass("DWLiQuan")

function tbItemClass:OnClientUse(pItem)
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.DragonBoatGift)
  if pInstance and pInstance:IsOpen() then
    for nIndex, nRedPointEventId in ipairs(pInstance:GetRedPointEventId()) do
      if RedPointMgr:CheckEvent(nRedPointEventId) then
        RedPointMgr:FadeRedPoint(nRedPointEventId)
      end
    end
    Ui:OpenWindow("UINpcShop", pInstance.tbConfig.nNpcShopId, nil, nil, nil, nil, Currency.TYPE.DWLiQuan, true)
  else
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
    Ui:CloseWindow("UIItemTips")
  end
  return 0
end
