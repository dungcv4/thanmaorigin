local tbWeddingInvitation = Item:GetClass("WeddingInvitation")

function tbWeddingInvitation:OnUse(pItem)
  local bInSafe = ActivityCheck:CheckIsInSafeArea(me)
  if not bInSafe then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186"))
    return 0
  end
  local dwPlayerId1 = pItem.GetIntValue(Marry.InvitationParam.Player1)
  local dwPlayerId2 = pItem.GetIntValue(Marry.InvitationParam.Player2)
  local nTime = pItem.GetIntValue(Marry.InvitationParam.Time)
  local szID = Marry:GenID(dwPlayerId1, dwPlayerId2)
  if nTime > GetTime() then
    me.CenterMsg(i18n.EasyGet("\229\169\154\231\164\188\229\176\154\230\156\170\229\188\128\229\167\139"))
    return
  elseif not Wedding:CheckWeddingOpen(szID) then
    me.CenterMsg(i18n.EasyGet("\229\169\154\231\164\188\229\183\178\231\187\147\230\157\159"))
    return
  end
  Wedding:TransferIn(me, szID)
  me.CallClientScript("Ui:CloseWindow", "UIBag")
  me.CallClientScript("Ui:CloseWindow", "UIItemTips")
  return 0
end

function tbWeddingInvitation:GetTip(Param)
  local pItem
  if type(Param) == "table" then
    pItem = KItem.GetItemObj(Param.tbItemExtParams.dwId)
  else
    pItem = Param
  end
  if not pItem then
    return
  end
  local szRet = ""
  local dwPlayerId1 = pItem.GetIntValue(Marry.InvitationParam.Player1)
  local dwPlayerId2 = pItem.GetIntValue(Marry.InvitationParam.Player2)
  local nTime = pItem.GetIntValue(Marry.InvitationParam.Time)
  local nLevel = pItem.GetIntValue(Marry.InvitationParam.Level)
  local szName = i18n.Get(Marry.Cfg.WeddingMapCfg[nLevel].szName)
  local bAllHit, nSession, tbRoleInfoList = Player.RoleInfoCache:GetCache({dwPlayerId1, dwPlayerId2})
  if bAllHit then
    szRet = i18n.Get(Marry.Cfg.Msg.INVITATION_TIPS, tbRoleInfoList[dwPlayerId1].szName, tbRoleInfoList[dwPlayerId2].szName, Lib:TimeDesc11Ex(nTime), szName)
  end
  return nil, szRet
end

function tbWeddingInvitation:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local dwPlayerId1 = pItem.GetIntValue(Marry.InvitationParam.Player1)
  local dwPlayerId2 = pItem.GetIntValue(Marry.InvitationParam.Player2)
  local nTime = pItem.GetIntValue(Marry.InvitationParam.Time)
  local tbRet
  if me.dwID == dwPlayerId1 or me.dwID == dwPlayerId2 then
    tbRet = {}
    tbRet.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
    
    function tbRet.fnFirst()
      Item:ClientUseItem(nItemId)
    end
    
    tbRet.szSecondName = i18n.EasyGet("\233\130\128\232\175\183")
    
    function tbRet.fnSecond()
      Marry:TryGetInviteInfo()
    end
    
    tbRet.szThirdName = i18n.EasyGet("\229\136\134\228\186\171")
    
    function tbRet.fnThird()
      Ui:SetItemToChat(nItemId)
      Ui:CloseWindow("UIBag")
      Ui:CloseWindow("UIItemTips")
    end
    
    tbRet.bHideShareButton = true
    tbRet.bHideHawkButton = true
    tbRet.bHideSellButton = true
    tbRet.btnMarketSellButton = true
  end
  return tbRet
end
