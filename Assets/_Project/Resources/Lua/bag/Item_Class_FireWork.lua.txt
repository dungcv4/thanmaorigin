local tbItem = Item:GetClass("FireWork")
tbItem.SaveKey = GetGroupKey("FireWorkItem")
tbItem.INTERVAL = 30
tbItem.LAST_USING_TIME_KEY = 1
tbItem.PARAM_SKILLID = 1
tbItem.PARAM_SKILLLEVEL = 2
tbItem.PARAM_AWARDID = 3
tbItem.PARAM_AWARDKEY = 4

function tbItem:OnUse(it)
  if it.nCount <= 0 then
    return 0
  end
  if not TongParty:IsPartyOpen(me.dwTongId) then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return 0
  end
  if me.nMapTemplateId ~= Tong.Def.nMainMapTemplateId then
    me.CenterMsg(i18n.EasyGet("\228\184\141\229\156\168\229\184\174\228\188\154\233\162\134\229\156\176\228\184\173"))
    return 0
  end
  local nNow = GetTime()
  local nLastUsingTime = me.GetUserValue(self.SaveKey, self.LAST_USING_TIME_KEY)
  if nNow - nLastUsingTime < self.INTERVAL then
    me.CenterMsg(i18n.EasyGet("\230\175\143{0}\231\167\146\228\187\133\229\143\175\228\187\165\231\130\185\231\135\1311\230\172\161\231\131\159\232\138\177", self.INTERVAL))
    return 0
  end
  local nSkillID = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_SKILLID)
  local nSkillLevel = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_SKILLLEVEL)
  local nAwardID = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARDID)
  local nAwardKey = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARDKEY)
  if not (nSkillID and not (nSkillID <= 0) and nSkillLevel and not (nSkillLevel < 0) and nAwardID and not (nAwardID < 0) and nAwardKey) or nAwardKey < 0 then
    me.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183{0}\231\188\186\229\176\145\230\137\169\229\177\149\229\143\130\230\149\176", it.dwTemplateId))
    return 0
  end
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168"))
    return 0
  end
  me.SetUserValue(self.SaveKey, self.LAST_USING_TIME_KEY, nNow)
  Award:GiveAward(me.dwID, nAwardID, nAwardKey, {
    nLogReason = Env.LogWay_FireWork
  })
  Event:OnEvent(Event.emEVENT_OBFIREWORK_USING, me.dwID)
  local pNpc = me.GetNpc()
  if pNpc then
    local _, nPosX, nPosY = me.GetWorldPos()
    pNpc.CastSkill(nSkillID, nSkillLevel, nPosX, nPosY)
  end
  return 1
end
