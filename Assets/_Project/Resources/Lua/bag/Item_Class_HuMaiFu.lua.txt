local tbHuMaiFu = Item:GetClass("HuMaiFu")
tbHuMaiFu.EXT_PARAM_VALUE = 1

function tbHuMaiFu:OnCreate()
end

function tbHuMaiFu:OnInit()
end

function tbHuMaiFu:OnUse(pItem)
  return 0
end

function tbHuMaiFu:OnUseMany(pItem, nCount)
  return 0
end

function tbHuMaiFu:GetAddAttrData(pItem)
  local nAddAttrCount = pItem.GetIntValue(1)
  local nAddAttrPercent1 = pItem.GetIntValue(2)
  local nAddAttrPercent2 = pItem.GetIntValue(3)
  local nAddAttrPercent3 = pItem.GetIntValue(4)
  return {
    nAddAttrCount,
    nAddAttrPercent1,
    nAddAttrPercent2,
    nAddAttrPercent3
  }
end

function tbHuMaiFu:SetAddAttrData(pItem, tbValue)
  local nAddAttrCount, nAddAttrPercent1, nAddAttrPercent2, nAddAttrPercent3 = table.unpack(tbValue)
  pItem.SetIntValue(1, nAddAttrCount)
  pItem.SetIntValue(2, nAddAttrPercent1)
  pItem.SetIntValue(3, nAddAttrPercent2)
  pItem.SetIntValue(4, nAddAttrPercent3)
end

function tbHuMaiFu:GetUseSetting(nTemplateId, nItemId, szOpt)
  local tbRet
  tbRet = {}
  if "ItemBox" == szOpt then
    tbRet.szFirstName = i18n.EasyGet("\232\163\133\229\164\135")
    
    function tbRet.fnFirst()
      JingMai:OpenJingMaiWindow()
    end
  end
  return tbRet
end

function tbHuMaiFu:GetChatLinkInfo(pItem)
  if pItem.szClass ~= "HuMaiFu" then
    return
  end
  local tbInfo = {}
  tbInfo.tbHuMaiFuData = self:GetAddAttrData(pItem)
  return tbInfo
end

function tbHuMaiFu:GetItemAttr(pItem)
  local tbHMFData = self:GetAddAttrData(pItem)
  local tbRet = self:GetItemAttrByTemplateId(pItem.dwTemplateId, tbHMFData)
  return tbRet
end

function tbHuMaiFu:GetItemAttrByTemplateId(dwTemplateId, tbHMFData)
  return JingMai:GetHuMaiFuAddAttrDesc(tbHMFData)
end
