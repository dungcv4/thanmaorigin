local tbItemClass = Item:GetClass("ActivityCurrency")
local tbExtParam = {BIG_TYPE = 1, SMALL_TYPE = 2}

function tbItemClass:OnClientUse(pItem)
  local nBigType = KItem.GetItemExtParam(pItem.dwTemplateId, tbExtParam.BIG_TYPE)
  local nSmallType = KItem.GetItemExtParam(pItem.dwTemplateId, tbExtParam.SMALL_TYPE)
  Ui:OpenWindowWithCallback("UIShop", function(caller, tbWnd)
    tbWnd:TrySelectNode(nBigType, nSmallType)
  end, self, Shop.ShopType2ViewIndex.LimitBuy)
  return 0
end
