local tbWnd = Ui:CreateWindow("UIItemSell")
tbWnd.tbControls = {
  btnClose = "imgBG/btnClose",
  UIItemGrid = "imgBG/panelItem/UIItemGrid",
  txtItemName = "imgBG/panelItem/txtItemName",
  UIAddAndSubGroup = "imgBG/panelSell/UIAddAndSubGroup",
  imgCostMoney = "imgBG/panelSell/Cost/imgMoney",
  txtCostCount = "imgBG/panelSell/Cost/txtCostCount",
  imgHaveMoney = "imgBG/panelSell/Have/imgMoney",
  txtHaveCount = "imgBG/panelSell/Have/txtHaveCount",
  btnSell = "imgBG/panelSell/btnSell"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnSell = function(self)
    local function fnConfirm()
      RemoteServer.OnItemSellRequest("SellPer", self.nItemId, self.nSellCount)
      
      self:Close()
    end
    
    local pItem = me.GetItemInAllSpace(self.nItemId)
    if not pItem then
      return
    end
    if 1 == pItem.IsEquip() and pItem.nItemType >= Item.EQUIP_HELM and pItem.nItemType <= Item.EQUIP_AMULET then
      local tbCurRandomAttr = Item:GetClass(Item.MAIN_EQUIP_CLASS_NAME):GetRandomAttr(pItem)
      if tbCurRandomAttr[1] then
        me.MsgBox(i18n.EasyGet("\232\163\133\229\164\135\230\156\137\233\153\132\229\138\160\229\177\158\230\128\167\239\188\140\230\152\175\229\144\166\229\155\158\230\148\182\239\188\159"), {
          {
            i18n.szOk,
            fnConfirm
          },
          {
            i18n.szCancel
          }
        })
      else
        fnConfirm()
      end
    else
      fnConfirm()
    end
  end
}

function tbWnd:OnCreate()
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, self.UIAddAndSubGroup)
  self.AddAndSubGroup:BindEvent(function(nValue)
    self.nSellCount = self:UpdateNumberInput(nValue)
    return self.nSellCount
  end)
  self.AddAndSubGroup:SetNumPadPos(-148, 12)
  self.AddAndSubGroup:SetEnable(true)
  self.ItemGrid = Ui:NewExistItemGrid(self, self.UIItemGrid)
end

function tbWnd:OnOpen(nItemId, nTemplateId)
  self.nItemId = nItemId
  self.nTemplateId = nTemplateId
  local pItem = KItem.GetItemObj(self.nItemId)
  local szName = KItem.GetItemShowInfo(nTemplateId, me.nFaction, me.nFactionSect, me.GetNpc().GetBodySex())
  self.pPanel:Label_SetText(self.txtItemName, i18n.Parse(szName))
  local szCurrencyType, nNum = Item.tbItemSell:GetSellCurrencyValue(nTemplateId, nItemId)
  local tbCustomData = Item.tbItemSell:GetCustomSellData(szCurrencyType)
  self.nPerPrice = nNum
  local szIcon = tbCustomData.funcGetIcon(szCurrencyType)
  self.pPanel:Sprite_SetSprite(self.imgCostMoney, szIcon)
  self.pPanel:Sprite_SetSprite(self.imgHaveMoney, szIcon)
  self.ItemGrid:SetItem(nItemId, nTemplateId)
  self.ItemGrid:SetClickEnable(false)
  self.ItemGrid:LabelSuffix_SetActive(false)
  local nHaveMoney = tbCustomData.funcGetValue(me, szCurrencyType)
  self.pPanel:Label_SetText(self.txtHaveCount, nHaveMoney)
  self.nSellCount = 1
  local nHaveNum = pItem.nCount
  self.AddAndSubGroup:SetNumRange(1, nHaveNum)
  self.AddAndSubGroup:SetInputValue(self.nSellCount or 1)
  self.pPanel:Label_SetText(self.txtCostCount, self.nPerPrice * self.nSellCount)
end

function tbWnd:UpdateNumberInput(nValue)
  self.pPanel:Label_SetText(self.txtCostCount, nValue * self.nPerPrice)
  return nValue
end
