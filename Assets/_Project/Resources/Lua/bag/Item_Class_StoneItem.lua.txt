local tbStone = Item:GetClass("Stone")

function tbStone:OnCreate()
end

function tbStone:GetUseSetting(nTemplateId, nItemId, szOpt)
  local tbRet = {}
  if szOpt and "ItemBox" == szOpt then
    tbRet.szFirstName = i18n.EasyGet("\233\149\182\229\181\140")
    
    function tbRet.fnFirst()
      if not Item.Stone:CheckIsOpen() then
        me.CenterMsg(i18n.EasyGet("\229\188\128\230\156\141\231\172\172{0}\229\164\169\229\188\128\229\144\175\229\174\157\231\159\179\233\149\182\229\181\140\229\138\159\232\131\189", Lib:CalcTimeFrameOpenDay(Item.Stone.szShowTimeFrame)))
        return
      end
      Require("Script/Ui/Window/UIEquipment.lua")
      Ui:OpenWindow("UIEquipment", Ui.UIEquipment.tbSubPanelIndex.PANEL_TYPE_STONE, nil, nTemplateId)
      Ui:CloseWindow("UIStoneTips")
    end
    
    if Item.Stone:CheckIsCanCompose(nTemplateId) then
      tbRet.szSecondName = i18n.EasyGet("\229\144\136\230\136\144")
      
      function tbRet.fnSecond()
        local function fnBid()
          RemoteServer.OnEquipStoneRequest("ComposeInBag", nTemplateId)
        end
        
        local ComposeId, nCostCoin = Item.Stone:GetComposeInfo(nTemplateId)
        local szName = KItem.GetItemShowInfo(nTemplateId, me.nFaction, me.nFactionSect, me.nSex)
        local szName2 = KItem.GetItemShowInfo(ComposeId, me.nFaction, me.nFactionSect, me.nSex)
        Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\230\182\136\232\128\151{0}\233\147\182\228\184\164\229\176\134{1}\228\184\170{2}\229\144\136\230\136\144\228\184\1861\228\184\170{3}?", nCostCoin, Item.Stone.nComposeNeed, i18n.Parse(szName), i18n.Parse(szName2)), {
          {fnBid},
          {}
        }, {
          i18n.szOk,
          i18n.szCancel
        })
      end
    end
    if Item.Stone:CheckIsCanSplit(nTemplateId) then
      tbRet.szThirdName = i18n.EasyGet("\230\139\134\232\167\163")
      
      function tbRet.fnThird()
        local nCnt = me.GetItemCountByTypeInBag(nTemplateId)
        local nMaxCanUse = math.min(nCnt, Item.Stone.MAX_SPLIT_NUM)
        Ui:OpenWindow("UIItemBatchPanel", nil, nTemplateId, "SplitStone", nMaxCanUse)
      end
    end
    if not Item.Stone:CheckIsCanSplit(nTemplateId) and Item.Stone:CheckIsCanExchange(nTemplateId) then
      tbRet.szThirdName = i18n.EasyGet("\229\133\145\230\141\162")
      
      function tbRet.fnThird()
        local nCnt = me.GetItemCountByTypeInBag(nTemplateId)
        local nMaxCanUse = math.min(nCnt, Item.Stone.MAX_EXCHANGE_NUM)
        Ui:OpenWindow("UIItemBatchPanel", nil, nTemplateId, "ExchangeStone", nMaxCanUse)
      end
    end
  elseif szOpt and "StoneInsert" == szOpt then
    if Ui:GetWindow("UIEquipment"):CheckCurHoleIsHadStone() then
      tbRet.szFirstName = i18n.EasyGet("\230\155\191\230\141\162")
    else
      tbRet.szFirstName = i18n.EasyGet("\233\149\182\229\181\140")
    end
    
    function tbRet.fnFirst()
      if Ui:WindowVisible("UIEquipment") then
        Ui:GetWindow("UIEquipment"):PutOnEquipStone(nItemId)
        Ui:CloseWindow("UIStoneTipsCompare")
      end
    end
    
    if Item.Stone:CheckIsCanCompose(nTemplateId) then
      tbRet.szSecondName = i18n.EasyGet("\229\144\136\230\136\144")
      
      function tbRet.fnSecond()
        local function fnBid()
          if RedPointMgr:CheckEvent(RedPointMgr.EVENT.StoneCompose) then
            RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.StoneCompose)
          end
          RemoteServer.OnEquipStoneRequest("ComposeInBag", nTemplateId)
        end
        
        local ComposeId, nCostCoin = Item.Stone:GetComposeInfo(nTemplateId)
        local szName = KItem.GetItemShowInfo(nTemplateId, me.nFaction, me.nFactionSect, me.nSex)
        local szName2 = KItem.GetItemShowInfo(ComposeId, me.nFaction, me.nFactionSect, me.nSex)
        Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\230\182\136\232\128\151{0}\233\147\182\228\184\164\229\176\134{1}\228\184\170{2}\229\144\136\230\136\144\228\184\1861\228\184\170{3}?", nCostCoin, Item.Stone.nComposeNeed, i18n.Parse(szName), i18n.Parse(szName2)), {
          {fnBid},
          {}
        }, {
          i18n.szOk,
          i18n.szCancel
        })
      end
    end
    if Item.Stone:CheckIsCanSplit(nTemplateId) then
      tbRet.szThirdName = i18n.EasyGet("\230\139\134\232\167\163")
      
      function tbRet.fnThird()
        local nCnt = me.GetItemCountByTypeInBag(nTemplateId)
        local nMaxCanUse = math.min(nCnt, Item.Stone.MAX_SPLIT_NUM)
        Ui:OpenWindow("UIItemBatchPanel", nil, nTemplateId, "SplitStone", nMaxCanUse)
      end
    end
    if not Item.Stone:CheckIsCanSplit(nTemplateId) and Item.Stone:CheckIsCanExchange(nTemplateId) then
      tbRet.szThirdName = i18n.EasyGet("\229\133\145\230\141\162")
      
      function tbRet.fnThird()
        local nCnt = me.GetItemCountByTypeInBag(nTemplateId)
        local nMaxCanUse = math.min(nCnt, Item.Stone.MAX_EXCHANGE_NUM)
        Ui:OpenWindow("UIItemBatchPanel", nil, nTemplateId, "ExchangeStone", nMaxCanUse)
      end
    end
    tbRet.nComposeRedPointPosition = 2
  elseif szOpt and "StoneForge" == szOpt then
    tbRet.szFirstName = i18n.EasyGet("\230\148\190\229\133\165")
    
    function tbRet.fnFirst()
      if Ui:WindowVisible("UIStoneCompose") then
        Ui:GetWindow("UIStoneCompose"):OnForgeStone(nTemplateId)
        Ui:CloseWindow("UIStoneTipsCompare")
      end
    end
    
    tbRet.bHideShareButton = true
  elseif szOpt and "Warehouse_ItemBag" == szOpt then
    tbRet.szFirstName = i18n.EasyGet("\229\173\152\229\133\165")
    
    function tbRet.fnFirst()
      if Ui:WindowVisible("UIWarehouse") then
        Ui:GetWindow("UIWarehouse"):MoveItemFromBagToBox(nItemId)
        Ui:CloseWindow("UIStoneTipsCompare")
      end
    end
    
    tbRet.bHideShareButton = true
    tbRet.bHideHawkButton = true
    tbRet.bHideSellButton = true
    tbRet.btnMarketSellButton = true
  elseif szOpt and "Warehouse_ItemBox" == szOpt then
    tbRet.szFirstName = i18n.EasyGet("\229\143\150\229\135\186")
    
    function tbRet.fnFirst()
      if Ui:WindowVisible("UIWarehouse") then
        Ui:GetWindow("UIWarehouse"):MoveItemFromBoxToBag(nItemId)
        Ui:CloseWindow("UIStoneTipsCompare")
      end
    end
    
    tbRet.bHideShareButton = true
    tbRet.bHideHawkButton = true
    tbRet.bHideSellButton = true
    tbRet.btnMarketSellButton = true
  end
  return tbRet
end
