local tbItemClass = Item:GetClass("LifeSkillMaterial")

function tbItemClass:OnClientUse(pItem)
  local nType = KItem.GetItemExtParam(pItem.dwTemplateId, 1)
  local nLevel = KItem.GetItemExtParam(pItem.dwTemplateId, 2)
  if not nType or not nLevel then
    return -1
  end
  
  local function func(tbWnd)
    if not tbWnd then
      return
    end
    tbWnd:OpenLifeSkillIndex(nType)
    tbWnd:OnSelectLifeSkillLevel(nLevel)
  end
  
  local szWndName = "UILifeSkill"
  if Ui:WindowVisible(szWndName) then
    return func(Ui:GetWindow(szWndName))
  end
  Ui:OpenWindowWithCallback(szWndName, function(_, tbWnd)
    func(tbWnd)
  end)
  return 0
end
