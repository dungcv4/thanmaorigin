Require("CommonScript/FieldPractice/FieldPractice.lua")
local tbItem = Item:GetClass("QiFuLing")
tbItem.nMaxUseCount = 6
tbItem.nGoldShellCount = 7

function tbItem:OnUse(it)
  if SERVER_RUNNING_MODE == SERVER_MODE.ZONE_SERVER and not GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
    return
  end
  if GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) and GlobalHangUp:IsForbiddenTime() then
    me.CenterMsg(GlobalHangUp:GetForbiddenUsingItemMsg())
    return
  end
  local nMaxUseCount = self:GetMaxUseCount(me)
  if nMaxUseCount <= FieldPractice:GetUseShellCount(me) then
    me.CenterMsg(i18n.EasyGet("\230\175\143\230\151\165\230\156\128\229\164\154\228\189\191\231\148\168{0}\228\184\170{1}", nMaxUseCount, it.szName))
    return
  end
  local tbSkillState = me.GetNpc().GetSkillState(FightSkill.LuckyAttrBuffId.QIFULING_BUFFID)
  local nCurFrame = GetFrame()
  local nAddFrame = 1800 * Env.GAME_FPS
  local nFinalFrame = nCurFrame + nAddFrame
  if tbSkillState and nCurFrame < tbSkillState.nEndFrame then
    nFinalFrame = tbSkillState.nEndFrame + nAddFrame
  else
    FieldPractice:SetGetShellCount(me, 0)
  end
  me.AddSkillState(FightSkill.LuckyAttrBuffId.QIFULING_BUFFID, 1, FightSkill.STATE_TIME_TYPE.state_time_leavemap, nFinalFrame - nCurFrame, 1)
  FieldPractice:OnUseQiFuLing(me, 1800)
  FieldPractice:AddGetShellCount(me, self.nGoldShellCount)
  FieldPractice:AddUseShellCount(me, 1)
  me.CenterMsg(i18n.EasyGet("\228\189\160\228\189\191\231\148\168\228\186\134\228\184\128\228\184\170{0}", it.szName))
  Lib:TrackEvent(me.dwID, Env.track_baiguolu_use, FieldPractice:GetUseShellCount(me))
  return 1
end

function tbItem:GetTip(it)
  if not it.dwId then
    return ""
  end
  local nCount = FieldPractice:GetUseShellCount(me)
  local szTip = i18n.EasyGet("\228\187\138\230\151\165\232\191\152\229\143\175\228\189\191\231\148\1680\228\184\170\239\188\136\230\172\161\230\151\1654:00\229\143\175\229\134\141\230\172\161\228\189\191\231\148\168\239\188\137")
  local nMaxUseCount = self:GetMaxUseCount(me)
  if nCount < nMaxUseCount then
    szTip = i18n.EasyGet("\228\187\138\230\151\165\232\191\152\229\143\175\228\189\191\231\148\168{0}\228\184\170", nMaxUseCount - nCount)
  end
  return szTip
end

function tbItem:GetMaxUseCount(pPlayer)
  return self.nMaxUseCount + Vip:GetVipAddBaiGuoLuNum(pPlayer.GetVipLevel())
end

function tbItem:GetUseTxt(pPlayer)
  local nCount = FieldPractice:GetUseShellCount(pPlayer)
  local nMaxUseCount = self:GetMaxUseCount(pPlayer)
  return string.format("%s/%s", nCount, nMaxUseCount)
end
