function GuanYin:GetSkillIdByTemplateId(nTemplateId)
  local tbBaseAttrib = KItem.GetEquipBaseProp(nTemplateId).tbBaseAttrib
  for _, tbAnAttrib in ipairs(tbBaseAttrib) do
    if tbAnAttrib.szName == "add_skill_level" then
      return tbAnAttrib.tbValue[1]
    end
  end
  return nil
end

function GuanYin:LoadGuanYinData()
  self.tbGuanYinData = Lib:LoadTabFileEx("Setting/Item/Equip/GuanYin.tab", "dsd", "TemplateId", {
    "TemplateId",
    "Name",
    "BigIcon"
  })
end

function tbGuanYinEquip:OpenQuickUseItem(nItemId)
  local pUsingEquip = me.GetEquipByPos(Item.EQUIPPOS_GUANYIN)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  if pUsingEquip and pItem.nValue <= pUsingEquip.nValue then
    return
  end
  Ui:OpenQuickUseItem(nItemId, i18n.EasyGet("\231\169\191\228\184\138"), nil, function(tbUIQuickUseItem)
    Item.Equip:UseEquip(false, nItemId)
    tbUIQuickUseItem:Close()
  end)
end
