local BOOK_CARD_ID_EXTRA_PARAM = 1
local ENTRANCE_EXTRA_PARAM = 2
local tbItem = Item:GetClass("SouvenirCardBook")

function tbItem:GetUseSetting(nTemplateId, nItemId)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local tbRet = {}
  local nBookCardId = KItem.GetItemExtParam(nTemplateId, BOOK_CARD_ID_EXTRA_PARAM)
  local nEntranceIndex = KItem.GetItemExtParam(nTemplateId, ENTRANCE_EXTRA_PARAM)
  tbRet.szFirstName = i18n.EasyGet("\230\159\165\233\152\133")
  
  function tbRet.fnFirst()
    Ui:OpenWindow("UIVietnamNationalGallery", nBookCardId, nEntranceIndex)
    Ui:CloseWindow("UIItemTips")
  end
  
  return tbRet
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "SouvenirCardBook")
end
