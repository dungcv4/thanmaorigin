Item.tbItemSell = Item.tbItemSell or {}
local tbItemSell = Item.tbItemSell
tbItemSell.bIsSelling = false

function tbItemSell:SellPerRsp()
end

function tbItemSell:SellManyRsp()
  self:SetBatchSellStatus(false)
  if Ui:WindowVisible("UIItemBatchSellPanel") then
    Ui:GetWindow("UIItemBatchSellPanel"):ResetSelectItem()
  end
  if Ui:WindowVisible("UIBag") then
    Ui:GetWindow("UIBag"):OnDelItem()
  end
end

function tbItemSell:IsDoingBatchSell()
  return self.bIsSelling
end

function tbItemSell:SetBatchSellStatus(bStatus)
  self.bIsSelling = bStatus
end
