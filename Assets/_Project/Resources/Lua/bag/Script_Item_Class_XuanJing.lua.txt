local tbItem = Item:GetClass("XuanJing")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbBaseProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbBaseProp then
    return
  end
  
  local function fnSplit()
    Ui:OpenWindow("UIXuanJingSplit", nItemId)
  end
  
  local function fnCompose()
    Ui:OpenWindow("UIXuanJingCompose")
  end
  
  if tbBaseProp.nLevel < Item.EquipEnhance.SPLIT_LV_REQ then
    return {
      szFirstName = i18n.EasyGet("\229\144\136\230\136\144"),
      fnFirst = fnCompose
    }
  else
    return {
      szFirstName = i18n.EasyGet("\230\139\134\232\167\163"),
      fnFirst = fnSplit,
      szSecondName = i18n.EasyGet("\229\144\136\230\136\144"),
      fnSecond = fnCompose
    }
  end
end
