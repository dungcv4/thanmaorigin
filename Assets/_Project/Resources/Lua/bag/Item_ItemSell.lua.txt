function tbItemSell:GetCustomSellData(szCurrencyType)
  local nItemId = tonumber(szCurrencyType)
  if nItemId and Item:IsExist(nItemId) then
    return self.tbCustomObtainType.Item
  end
  local tbCustomData = self.tbCustomObtainType[szCurrencyType] or self.tbCustomObtainType.Default
  return tbCustomData
end

function tbItemSell:LoadItemSell()
  self.tbSellItemCfg = {}
  local tbCfg = Lib:LoadTabFileEx("Setting/Item/ItemSell.tab", "dsdss", nil, {
    "nTemplateId",
    "szCurrencyType",
    "nPrice",
    "szTimeFrame",
    "szTimeEnd"
  })
  for _, v in pairs(tbCfg) do
    self.tbSellItemCfg[v.nTemplateId] = v
  end
  Validator:Register(self.CheckItemSell, self)
end

function tbItemSell:CheckItemSell()
  for nTemplateId, v in pairs(self.tbSellItemCfg) do
    if not Item:IsExist(nTemplateId) then
      LogErr(string.format("load ItemSell.tab failed, nTemplateId %d not exist", nTemplateId))
      return false
    end
    if Lib:IsEmptyStr(v.szCurrencyType) then
      LogErr(string.format("load ItemSell.tab failed, nTemplateId %d szCurrencyType id empty!", nTemplateId))
      return false
    end
    local szType = ""
    local nItemId = tonumber(v.szCurrencyType)
    if nItemId and Item:IsExist(nItemId) then
      szType = "Item"
    elseif self.tbCustomObtainType[v.szCurrencyType] then
      szType = v.szCurrencyType
    else
      szType = "Default"
    end
    if "Default" == szType then
      if Lib:IsEmptyStr(self.tbCustomObtainType[szType].funcGetObtainName(v.szCurrencyType)) then
        LogErr(string.format("load ItemSell.tab failed, nTemplateId %d szCurrencyType=%s is bad!", nTemplateId, v.szCurrencyType))
        return false
      end
    elseif "Item" == szType then
    elseif not self.tbCustomObtainType[v.szCurrencyType] then
      LogErr(string.format("load ItemSell.tab failed, nTemplateId %d szCurrencyType=%s is bad!", nTemplateId, v.szCurrencyType))
      return false
    end
  end
  return true
end

tbItemSell:LoadItemSell()

function tbItemSell:CheckCanSellEquip(pPlayer, pItem)
  local bRet, szTips = self:CheckCanSellItem(pItem.dwTemplateId)
  if not bRet then
    return false, szTips
  end
  local bIsEquip = pItem.IsEquip()
  if not bIsEquip then
    return true
  end
  local nItemId = pItem.nItemId
  local nEquipPos = pItem.nEquipPos
  local tbEquips = pPlayer.GetEquips(1)
  if tbEquips[nEquipPos] and tbEquips[nEquipPos] == nItemId then
    return false, i18n.EasyGet("\232\163\133\229\164\135\231\154\132\233\129\147\229\133\183\228\184\141\232\131\189\229\155\158\230\148\182")
  end
  return true
end

function tbItemSell:CheckCanSellItem(nTemplateId)
  if self.tbSellItemCfg[nTemplateId] then
    if self.tbSellItemCfg[nTemplateId].szTimeFrame ~= "" and 0 == GetTimeFrameState(self.tbSellItemCfg[nTemplateId].szTimeFrame) then
      return false, i18n.EasyGet("\232\175\165\233\129\147\229\133\183\228\184\141\232\131\189\229\155\158\230\148\182")
    end
    if "" ~= self.tbSellItemCfg[nTemplateId].szTimeEnd and Lib:GetConfigDate2Time(self.tbSellItemCfg[nTemplateId].szTimeEnd) > GetTime() then
      return false, i18n.EasyGet("\232\175\165\233\129\147\229\133\183\228\184\141\232\131\189\229\155\158\230\148\182")
    end
    return self.tbSellItemCfg[nTemplateId]
  end
  return false, i18n.EasyGet("\232\175\165\233\129\147\229\133\183\228\184\141\232\131\189\229\155\158\230\148\182")
end

function tbItemSell:GetSellCurrencyValue(nTemplateId, nItemId)
  local tbData = self:CheckCanSellItem(nTemplateId)
  if not tbData then
    return
  end
  local tbCustomData = self:GetCustomSellData(tbData.szCurrencyType)
  local nObtainNum = tbCustomData.funcGetObtainNum(nItemId, tbData)
  if not nObtainNum then
    return
  end
  return tbData.szCurrencyType, nObtainNum
end
