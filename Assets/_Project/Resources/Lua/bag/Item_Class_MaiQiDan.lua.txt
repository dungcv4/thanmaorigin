local tbMaiQiDan = Item:GetClass("MaiQiDan")
tbMaiQiDan.EXT_PARAM_VALUE = 1

function tbMaiQiDan:OnCreate()
end

function tbMaiQiDan:OnInit()
end

function tbMaiQiDan:GetAddMaiQiValue(pItem)
  return KItem.GetItemExtParam(pItem.dwTemplateId, tbMaiQiDan.EXT_PARAM_VALUE)
end

function tbMaiQiDan:OnUse(pItem)
  local nAddValue = KItem.GetItemExtParam(pItem.dwTemplateId, tbMaiQiDan.EXT_PARAM_VALUE)
  local nOldValue = JingMai:GetMaiQiValue(me)
  local tbCfg = JingMai:GetRestoreMaiQiCfg()
  if nOldValue + nAddValue > tbCfg.nMaiQiUpperLimit then
    me.CenterMsg(i18n.EasyGet("\232\132\137\230\176\148\232\182\133\229\135\186\228\184\138\233\153\144\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168"))
    return 0
  end
  JingMai:AddMaiQiValue(me, nAddValue)
  me.CallClientScript("JingMai:ShowMaiQiAward", nAddValue)
  return 1
end

function tbMaiQiDan:OnUseMany(pItem, nCount)
  if nCount > pItem.nCount then
    return 0
  end
  local nAddValue = KItem.GetItemExtParam(pItem.dwTemplateId, tbMaiQiDan.EXT_PARAM_VALUE) * nCount
  local nOldValue = JingMai:GetMaiQiValue(me)
  local tbCfg = JingMai:GetRestoreMaiQiCfg()
  if nOldValue + nAddValue > tbCfg.nMaiQiUpperLimit then
    me.CenterMsg(i18n.EasyGet("\232\132\137\230\176\148\232\182\133\229\135\186\228\184\138\233\153\144\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168"))
    return 0
  end
  JingMai:AddMaiQiValue(me, nAddValue)
  me.CallClientScript("JingMai:ShowMaiQiAward", nAddValue)
  return nCount
end
