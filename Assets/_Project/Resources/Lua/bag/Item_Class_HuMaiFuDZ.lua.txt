function HuMaiFuDZ:OnUse(pItem)
  local nHuMaiFuPos = KItem.GetItemExtParam(pItem.dwTemplateId, self.HMFPos)
  local nAttrQuality1 = KItem.GetItemExtParam(pItem.dwTemplateId, self.AttrQuality1)
  local nAttrQuality2 = KItem.GetItemExtParam(pItem.dwTemplateId, self.AttrQuality2)
  local nAttrQuality3 = KItem.GetItemExtParam(pItem.dwTemplateId, self.AttrQuality3)
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag < 1 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161\239\188\140\232\175\183\229\133\136\230\184\133\231\144\134\232\131\140\229\140\133"))
    return 0
  end
  if JingMai:DoIdentifyHuMaiFuDZ(me, nHuMaiFuPos, {
    nAttrQuality1,
    nAttrQuality2,
    nAttrQuality3
  }) then
    return 1
  end
  return 0
end
