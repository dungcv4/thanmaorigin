function tbItem:OnClientUse()
  return 1
end

function tbItem:OnUse(pItem)
  if not pItem then
    return 0
  end
  local nTemplateId = pItem.dwTemplateId
  local nItemId = pItem.dwId
  if not nTemplateId or not nItemId then
    return 0
  end
  local nActivityId = KItem.GetItemExtParam(nTemplateId, 1)
  local bRet, szErr = self:CheckUseWipeOutCard(pItem, nActivityId)
  if not bRet then
    if szErr then
      me.CenterMsg(szErr)
    end
    return 0
  end
  self:UseWipeOutCard(pItem, nActivityId)
  if not bRet then
    if szErr then
      me.CenterMsg(szErr)
    end
    return 0
  end
  return 1
end

function tbItem:CheckUseWipeOutCard(pItem, nActivityId)
  local tbFuncs = tbItem.tbFuncs[nActivityId]
  if not tbFuncs then
    return false, i18n.EasyGet("\233\129\147\229\133\183\231\177\187\229\158\139\228\184\141\229\173\152\229\156\168")
  end
  local fnCheck = tbFuncs.fnCheck
  if not fnCheck then
    return false, i18n.EasyGet("\233\129\147\229\133\183\231\177\187\229\158\139\228\184\141\229\173\152\229\156\168")
  end
  local bRet, szErr = fnCheck(pItem)
  return bRet, szErr
end

function tbItem:UseWipeOutCard(pItem, nActivityId)
  local tbFuncs = tbItem.tbFuncs[nActivityId]
  if not tbFuncs then
    return false, i18n.EasyGet("\233\129\147\229\133\183\231\177\187\229\158\139\228\184\141\229\173\152\229\156\168")
  end
  local fnUse = tbFuncs.fnUse
  if not fnUse then
    return false, i18n.EasyGet("\233\129\147\229\133\183\231\177\187\229\158\139\228\184\141\229\173\152\229\156\168")
  end
  local bRet, szErr = fnUse(pItem)
  return bRet, szErr
end

function tbItem:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    Item:ClientUseItem(nItemId)
    Ui:CloseWindow("UIItemTips")
  end
  
  return tbSetting
end
