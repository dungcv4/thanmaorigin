Require("CommonScript/Item/ItemDefine.lua")
local tbZongzi = Item:GetClass("Zongzi")

function tbZongzi:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.JingWangJiuXiActivity)
    if pInstance then
      pInstance:GotoHandInZongzi()
    else
      me.CenterMsg(i18n.EasyGet("\228\184\141\229\156\168\230\180\187\229\138\168\230\156\159"))
    end
  end
  
  return tbSetting
end
