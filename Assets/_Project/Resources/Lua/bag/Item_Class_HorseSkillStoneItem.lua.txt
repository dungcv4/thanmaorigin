local tbHorseSkillStoneItem = Item:GetClass("HorseSkillStoneItem")
tbHorseSkillStoneItem.EXT_PARAM_SKILL_INDEX = 1

function tbHorseSkillStoneItem:OnCreate(pSkillStone)
  local nSeries = pSkillStone.GetSeries()
  if nSeries <= Item.emSERIES_TYPE.SERIES_NONE or nSeries >= Item.emSERIES_TYPE.SERIES_NUM then
    local nSeries = MathRandom(Item.emSERIES_TYPE.SERIES_NONE + 1, Item.emSERIES_TYPE.SERIES_NUM - 1)
    pSkillStone.SetSeries(nSeries)
  end
end

function tbHorseSkillStoneItem:OnInit(pEquip)
  local nValue = pEquip.nOrgValue
  local nFightPower = pEquip.nBaseFightPower
  pEquip.nFightPower = nFightPower
  pEquip.SetSingleValue(nValue)
end

function tbHorseSkillStoneItem:GetTip(pEquip, pPlayer, bIsCompare, tbItemExtParams)
  local tbReturn = {}
  table.insert(tbReturn, {
    nil,
    nil,
    nil,
    nil,
    i18n.Get(HorseSkillMgr.HorseSkillCfg.szHorseSkillStoneDesc)
  })
  return tbReturn
end

function tbHorseSkillStoneItem:GetTipByTemplate(nTemplateId)
  local tbReturn = {}
  table.insert(tbReturn, {
    nil,
    nil,
    nil,
    nil,
    i18n.Get(HorseSkillMgr.HorseSkillCfg.szHorseSkillStoneDesc)
  })
  return tbReturn
end
