Require("CommonScript/Item/SecretBook/XiuWeiCfg.lua")

function XiuWei:LoadSitPosition()
  self.tbSitPosData = {}
  for nFaction, tbFactionData in pairs(self.tbEachFactionData) do
    if Faction:IsFactionOpen(nFaction) then
      local tbPos = Lib:LoadTabFile(tbFactionData.szSitPosFile, {PosX = 1, PosY = 1})
      if #tbPos <= 0 then
        LogErr("xiuwei sit pos file path", tbFactionData.szSitPosFile, "have no point")
      end
      self.tbSitPosData[nFaction] = tbPos
    end
  end
end

XiuWei:LoadSitPosition()
