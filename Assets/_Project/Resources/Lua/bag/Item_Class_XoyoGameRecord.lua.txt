local tbXoyoGameRecord = Item:GetClass("XoyoGameRecord")

function tbXoyoGameRecord:OnCreate()
end

function tbXoyoGameRecord:GetUseSetting(nTemplateId, nItemId)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local tbRet = {}
  tbRet.szFirstName = i18n.EasyGet("\230\159\165\233\152\133")
  
  function tbRet.fnFirst()
    Ui:OpenWindow("UIXiaoYaoLu", pItem)
    Ui:CloseWindow("UIItemTips")
  end
  
  return tbRet
end

function tbXoyoGameRecord:OnClientUse(pItem)
  Ui:OpenWindow("UIXiaoYaoLu", pItem)
  return 0
end

function tbXoyoGameRecord:OnUse(pItem)
  return 0
end

function tbXoyoGameRecord:GetCollectCountStr(nCount)
  local tbCardId = XoyoGame.XoyoGameRecordCfg.tbCardId
  local nMaxCount = #tbCardId
  local szColor500 = Ui.tbColor:GetColorHexBySettingId(500)
  local sMsg2 = i18n.EasyGet("<color=#{0}>\229\183\178\230\148\182\233\155\134\239\188\136{1}/{2}\239\188\137\229\188\160\229\141\161\231\137\135</color>", szColor500, nCount, nMaxCount)
  return sMsg2
end

function tbXoyoGameRecord:GetTip(pItem)
  if not pItem.dwId then
    return
  end
  local nCount = self:GetHadCardNum(pItem)
  return self:GetCollectCountStr(nCount)
end

function tbXoyoGameRecord:GetHadCardNum(pItem)
  if pItem.szClass ~= "XoyoGameRecord" then
    return 0
  end
  local nCount = 0
  local tbCardId = XoyoGame.XoyoGameRecordCfg.tbCardId
  for i = 1, #tbCardId do
    local nItemTemplateId = pItem.GetIntValue(i)
    if nItemTemplateId == tbCardId[i] then
      nCount = nCount + 1
    end
  end
  return nCount
end

function tbXoyoGameRecord:GetChatLinkInfo(pItem)
  if pItem.szClass ~= "XoyoGameRecord" then
    return
  end
  local tbInfo = {}
  tbInfo.XYGCardCount = self:GetHadCardNum(pItem)
  return tbInfo
end
