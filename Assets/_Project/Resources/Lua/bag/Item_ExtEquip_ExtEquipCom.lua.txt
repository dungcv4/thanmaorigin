Require("CommonScript/Item/ExtEquip/ExtEquipCfg.lua")
local ExtEquip = Item.ExtEquip
ExtEquip.EXT_EQUIPTYPE_NAME = {
  [ExtEquip.CFG.SLOT.SECRET_BOOK_LOW] = "SecretBook",
  [ExtEquip.CFG.SLOT.SECRET_BOOK_MIDDLE] = "SecretBook",
  [ExtEquip.CFG.SLOT.SECRET_BOOK_HIGH] = "SecretBook"
}

function Item:OnUseExtEquip(pPlayer, pEquip)
  if CLIENT then
    EventNotify.OnNotify(EventNotify.emNOTIFY_USE_UNUSE_EQUIP, true, Item.EQUIPPOS_EXTEND, pEquip.dwId)
  end
end

function Item:OnUnuseExtEquip(pPlayer, pEquip)
  if CLIENT then
    EventNotify.OnNotify(EventNotify.emNOTIFY_USE_UNUSE_EQUIP, false, Item.EQUIPPOS_EXTEND, pEquip.dwId)
  end
end

function Item:GetExtEquipMaxSlot()
  return Lib:CountTB(ExtEquip.CFG.SLOT)
end

function Item:GetExtEquipSlot(pEquip)
  return ExtEquip:GetSlot(pEquip)
end

function ExtEquip:GetEquipObj(pPlayer, nSlot)
  return pPlayer.GetEquipByPos(Item.EQUIPPOS_EXTEND, nSlot)
end

function ExtEquip:GetSlot(pEquip)
  if pEquip.nEquipPos ~= Item.EQUIPPOS_EXTEND then
    return 0
  end
  return pEquip.nDetailType
end

function ExtEquip:GetSlotByTemplateId(nItemTid)
  local nEquipPos = KItem.GetEquipPos(nItemTid)
  if not nEquipPos or nEquipPos ~= Item.EQUIPPOS_EXTEND then
    return 0
  end
  local nDetailType = KItem.GetDetailType(nItemTid)
  if not nDetailType then
    return 0
  end
  return nDetailType
end
