local tbItem = Item:GetClass("SpeechBalloon")
tbItem.PARAM_ID = 1
tbItem.PARAM_TIMEOUT = 2

function tbItem:OnUse(item)
  local nResult = self:UseSpeechBalloonItem(me, item.dwTemplateId, 1)
  return nResult
end

function tbItem:UseSpeechBalloonItem(pPlayer, nTemplateId, nCount)
  local nNum = pPlayer.GetItemCountByTypeInBag(nTemplateId)
  if nNum <= 0 then
    pPlayer.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183{0}\230\149\176\233\135\143\228\184\141\232\182\179", nTemplateId))
    return 0
  end
  local nSpeechBalloonId = KItem.GetItemExtParam(nTemplateId, self.PARAM_ID)
  if not SpeechBalloon:CheckId(nSpeechBalloonId) then
    pPlayer.CenterMsg(i18n.EasyGet("\232\175\165\232\129\138\229\164\169\230\176\148\230\179\161\228\184\141\229\173\152\229\156\168\239\188\129"))
    return 0
  end
  local nLimitTime = KItem.GetItemExtParam(nTemplateId, self.PARAM_TIMEOUT)
  return SpeechBalloon:AddSpeechBalloon(pPlayer, nSpeechBalloonId, nLimitTime)
end
