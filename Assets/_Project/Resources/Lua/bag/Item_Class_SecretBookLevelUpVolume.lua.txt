Require("CommonScript/Item/SecretBook/SecretBookCfg.lua")
local tbItem = Item:GetClass("SecretBookLevelUpVolume")
local SecretBook = Item.SecretBook
tbItem.nRankExtParam = 1

function tbItem:OnUse(pItem)
  local pPlayer = me
  local nRank = KItem.GetItemExtParam(pItem.dwTemplateId, self.nRankExtParam)
  local szErr = ""
  if SecretBook:IsLegalRank(nRank) then
    local bRet
    bRet, szErr = SecretBook:CanUseLevelUpVolume(pPlayer, nRank)
    if bRet then
      if MODULE_GAMESERVER then
        SecretBook:AddSkillLevel(pPlayer, SecretBook:GetWearSecretBook(pPlayer, nRank))
      end
      return 1
    end
  end
  if "" ~= szErr then
    pPlayer.CenterMsg(szErr)
  end
  return 0
end
