local tbItem = Item:GetClass("MingYangTianXiaBox")

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  local nLastAwardCount = self:GetLastAwardCount(me)
  local nSeniorAwardCount = math.min(nCount, nLastAwardCount)
  local nNormalAwardCount = nCount - nSeniorAwardCount
  self:AddUseCount(me, nCount)
  if nSeniorAwardCount > 0 then
    for i = 1, nSeniorAwardCount do
      Award:GiveAward(me.dwID, TongBattle.JiJieCfg.nSeniorAwardId, TongBattle.JiJieCfg.nSeniorAwardKey, {
        nLogReason = Env.LogWay_MingYangTianXia
      })
    end
  end
  if nNormalAwardCount > 0 then
    for i = 1, nNormalAwardCount do
      Award:GiveAward(me.dwID, TongBattle.JiJieCfg.nNormalAwardId, TongBattle.JiJieCfg.nNormalAwardKey, {
        nLogReason = Env.LogWay_MingYangTianXia
      })
    end
  end
  return nCount
end

function tbItem:GetLastAwardCount(pPlayer)
  local nCurDegree = pPlayer.GetUserValue(TongBattle.SAVE_GROUP, TongBattle.SAVE_BOX_COUNT)
  local nLastAwardCount = TongBattle.JiJieCfg.nMingYangAwardCount - nCurDegree
  if nLastAwardCount < 0 then
    nLastAwardCount = 0
  end
  return nLastAwardCount
end

function tbItem:GetUseCount(pPlayer)
  return pPlayer.GetUserValue(TongBattle.SAVE_GROUP, TongBattle.SAVE_BOX_COUNT)
end

function tbItem:AddUseCount(pPlayer, nAdd)
  local nCurCount = self:GetUseCount(pPlayer)
  pPlayer.SetUserValue(TongBattle.SAVE_GROUP, TongBattle.SAVE_BOX_COUNT, nCurCount + nAdd)
end

function tbItem:GetClientDefaultUseCount(nItemId)
  local pItem = me.GetItemInBag(nItemId)
  local nResult = math.min(pItem.nCount, Item.UseManyLimit.MingYangTianXiaBox)
  return nResult
end
