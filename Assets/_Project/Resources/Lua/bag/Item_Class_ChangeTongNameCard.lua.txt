local tbItemClass = Item:GetClass("ChangeTongNameCard")

function tbItemClass:OnUse(pItem)
  if not pItem.dwTemplateId then
    return
  end
  if not me.dwKinId or 0 == me.dwKinId then
    return
  end
  local tbTongData = Tong:GetTongById(me.dwTongId)
  local bRet = tbTongData:AddChangeNameRemainCount(1)
  if bRet then
    me.CenterMsg(i18n.EasyGet("\229\162\158\229\138\160\228\186\134\228\184\128\230\172\161\229\184\174\228\188\154\230\148\185\229\144\141\230\156\186\228\188\154"))
  end
  return bRet and 1 or 0
end
