Require("CommonScript/ShenBingPu/ShenBingPuCfg.lua")
local tbShenBingPu = Item:GetClass("ShenBingPuItem")
local ShenBingPu = Item.ShenBingPu

function tbShenBingPu:OnCreate(pItem)
  self:InitStarLevel(pItem)
  self:InitFixAttr(pItem)
end

function tbShenBingPu:InitStarLevel(pItem)
  if ShenBingPu.tbItemCfg[pItem.dwTemplateId] then
    ShenBingPu:SetStarLevel(pItem, ShenBingPu.tbItemCfg[pItem.dwTemplateId].nStarLevel)
  else
    Lib:LogErrFormat("ShenBingPuItem: InitStarLevel falied, unexist nTemplateId = %d", pItem.dwTemplateId)
  end
end

function tbShenBingPu:InitFixAttr(pItem)
  local tbItemCfg = ShenBingPu.tbItemCfg[pItem.dwTemplateId]
  if not tbItemCfg then
    return
  end
  for i = 1, Item.ShenBingPu.CFG.nMainAttrNum do
    local nFixedAttrId = tbItemCfg["nFixedAttr" .. i .. "Id"]
    if nFixedAttrId then
      local nValue1 = MathRandom(tbItemCfg["nFixedAttr" .. i .. "MinValue1"], tbItemCfg["nFixedAttr" .. i .. "MaxValue1"])
      local nValue2 = MathRandom(tbItemCfg["nFixedAttr" .. i .. "MinValue2"], tbItemCfg["nFixedAttr" .. i .. "MaxValue2"])
      local nValue3 = 0
      local szAttrType = ShenBingPu:GetAttrTypeStrByAttrId(nFixedAttrId)
      if "basic_damage_v" == szAttrType then
        nValue3 = nValue1
      else
        nValue3 = MathRandom(tbItemCfg["nFixedAttr" .. i .. "MinValue3"], tbItemCfg["nFixedAttr" .. i .. "MaxValue3"])
      end
      local tbNewFixAttr = {
        nAttrId = nFixedAttrId,
        nAttrValue1 = nValue1,
        nAttrValue2 = nValue2,
        nAttrValue3 = nValue3
      }
      ShenBingPu:SetFixedAttr(pItem, tbNewFixAttr)
    end
  end
end

function tbShenBingPu:GetRandomAttrib(pEquip)
  local tbTip = {}
  local tbAllRandAttrib
  if pEquip then
    local szQualityColor = Item:GetQualityColor(pEquip.nQuality)
    tbAllRandAttrib = ShenBingPu:GetAllRandAttr(me, pEquip)
    for nIndex, tbMA in ipairs(tbAllRandAttrib) do
      tbTip[nIndex] = {}
      local szMinValue, szMaxValue = ShenBingPu:GetRandAttrValRange(pEquip.dwTemplateId, tbMA.nAttrId, true)
      szMinValue = string.gsub(szMinValue, "+", "")
      szMaxValue = string.gsub(szMaxValue, "+", "")
      local szAttribDesc = FightSkill:GetMagicDesc(tbMA.szAttrType, {
        tbMA.nAttrValue1,
        tbMA.nAttrValue2,
        tbMA.nAttrValue3
      })
      if szAttribDesc and "" ~= szAttribDesc then
        table.insert(tbTip[nIndex], string.format("<color=#%s>%s</color> ", Ui.tbColor:GetColorHexByDefineName(szQualityColor), szAttribDesc))
        table.insert(tbTip[nIndex], string.format("<color=#%s>(%s - %s)</color>", Ui.tbColor:GetColorHexByDefineName("Gray"), szMinValue, szMaxValue))
      end
    end
    return tbTip
  else
    return nil
  end
end

function tbShenBingPu:GetChatLinkInfo(pEquip)
  if pEquip.szClass ~= "ShenBingPuItem" then
    return
  end
  local tbInfo = {}
  tbInfo.szClass = pEquip.szClass
  tbInfo.tbFixAttr = self:GetFixAttrib(pEquip)
  tbInfo.tbRandomAttr = self:GetRandomAttrib(pEquip)
  tbInfo.nFightPower = ShenBingPu:GetWeaponFightPower(me, pEquip)
  return tbInfo
end

function tbShenBingPu:GetTip(pEquip, pPlayer)
  return self:GetFixAttrib(pEquip), self:GetRandomAttrib(pEquip)
end

function tbShenBingPu:GetTipByTemplate(tbLinkInfo)
  return tbLinkInfo.tbFixAttr, tbLinkInfo.tbRandomAttr
end

function tbShenBingPu:GetFixAttrib(pEquip)
  local tbTip = {}
  local tbAllAttr, tbBaseAttr, tbAttrRange
  if pEquip then
    tbAllAttr = ShenBingPu:GetAllFixedAttr(me, pEquip) or {}
    tbBaseAttr = ShenBingPu:GetAllFixedAttr(me, pEquip, 1) or {}
    tbAttrRange = ShenBingPu:GetAllFixedAttrValRange(pEquip.dwTemplateId)
    local tbAttrDiff = ShenBingPu:GetWeaponAttrDiffByBatch(tbAllAttr, tbBaseAttr) or {}
    for i, tbDiff in ipairs(tbAttrDiff) do
      tbTip[i] = {}
      if tbDiff.szAttrDesc and tbDiff.szAttrDesc ~= "" then
        local szQualityColor = Item:GetQualityColor(pEquip.nQuality)
        table.insert(tbTip[i], string.format("<color=#%s>%s\239\188\154+%s</color> ", Ui.tbColor:GetColorHexByDefineName(szQualityColor), tbDiff.szAttrDesc, tbDiff.nAttr2Value))
        table.insert(tbTip[i], string.format("<color=#%s>(%s - %s)</color>", Ui.tbColor:GetColorHexByDefineName("Gray"), tbAttrRange[i].nMinValue, tbAttrRange[i].nMaxValue))
        if tbDiff.nAttr1Value ~= tbDiff.nAttr2Value then
          table.insert(tbTip[i], i18n.EasyGet("<color=#{0}>(\229\188\186\229\140\150 +{1})</color>", Ui.tbColor:GetColorHexByDefineName("Green"), tbDiff.nAttr1Value - tbDiff.nAttr2Value))
        end
      end
    end
  end
  return tbTip
end

function tbShenBingPu:GetRecyclePrice(nTemplate)
  local tbItemCfg = ShenBingPu.tbItemCfg[nTemplate]
  if tbItemCfg then
    return tbItemCfg.nRecycleMaterialId, tbItemCfg.nRecycleMaterialNum
  end
end
