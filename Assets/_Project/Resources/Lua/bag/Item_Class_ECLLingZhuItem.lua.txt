local ECLLingZhuItem = Item:GetClass("ECLLingZhuItem")
ECLLingZhuItem.ECL_RANDOM_ATTR_BEGIN = 1
ECLLingZhuItem.ECL_ATTR_OFFSET = 4

function ECLLingZhuItem:OnCreate()
end

function ECLLingZhuItem:OnClientUse(pItem)
end

function ECLLingZhuItem:OnUseMany(it, nCount)
  return 0
end

function ECLLingZhuItem:OnUse(pItem)
  return 0
end

function ECLLingZhuItem:GetUseSetting(nTemplateId, nItemId, szOpt)
  local tbRet = {}
  if "ItemBox" == szOpt then
    tbRet.szFirstName = i18n.EasyGet("\230\183\172\231\129\181")
    
    function tbRet.fnFirst()
      if not EquipCuiLing:IsSystemOpen() then
        return
      end
      Ui:OpenWindow("UIEquipment", Ui.UIEquipment.tbSubPanelIndex.PANEL_TYPE_CUILING)
      Ui:CloseWindow("UIItemTips")
    end
  end
  return tbRet
end

function ECLLingZhuItem:SetRandomAttrInfo(pItem, nIndex, nAttrId, tbV, nOffSet)
  nOffSet = nOffSet or 0
  pItem.SetIntValue((nIndex - 1) * self.ECL_ATTR_OFFSET + self.ECL_RANDOM_ATTR_BEGIN + nOffSet, nAttrId)
  for j = 1, 3 do
    pItem.SetIntValue((nIndex - 1) * self.ECL_ATTR_OFFSET + self.ECL_RANDOM_ATTR_BEGIN + j + nOffSet, tbV[j] or 0)
  end
end

function ECLLingZhuItem:SetRandomAttrInfo2(pItem, tbAttr, nOffSet)
  nOffSet = nOffSet or 0
  for i = 1, 3 do
    if tbAttr[i] then
      pItem.SetIntValue((i - 1) * self.ECL_ATTR_OFFSET + self.ECL_RANDOM_ATTR_BEGIN + nOffSet, tbAttr[i][1])
      for j = 1, 3 do
        pItem.SetIntValue((i - 1) * self.ECL_ATTR_OFFSET + self.ECL_RANDOM_ATTR_BEGIN + j + nOffSet, tbAttr[i][2][j])
      end
    end
  end
end

function ECLLingZhuItem:GetRandomAttrInfo(pItem, nOffSet)
  nOffSet = nOffSet or 0
  local tbRet = {}
  for i = 1, 3 do
    local nAttrId = pItem.GetIntValue((i - 1) * self.ECL_ATTR_OFFSET + self.ECL_RANDOM_ATTR_BEGIN + nOffSet)
    if nAttrId > 0 then
      local tbV = {}
      for j = 1, 3 do
        tbV[j] = pItem.GetIntValue((i - 1) * self.ECL_ATTR_OFFSET + self.ECL_RANDOM_ATTR_BEGIN + j + nOffSet)
      end
      tbRet[i] = {nAttrId, tbV}
    end
  end
  return tbRet
end

function ECLLingZhuItem:GetChatLinkInfo(pItem)
  if pItem.szClass ~= "ECLLingZhuItem" then
    return
  end
  local tbInfo = {}
  tbInfo.tbLingZhuData = self:GetRandomAttrInfo(pItem)
  return tbInfo
end

function ECLLingZhuItem:GetItemAttr(pItem)
  local tbLingZhuData = self:GetRandomAttrInfo(pItem)
  local tbRet = self:GetItemAttrByTemplateId(pItem.dwTemplateId, tbLingZhuData)
  return tbRet
end

function ECLLingZhuItem:GetItemAttrByTemplateId(dwTemplateId, tbLingZhuData)
  return EquipCuiLing:GetLingZhuAddAttrDesc(tbLingZhuData)
end
