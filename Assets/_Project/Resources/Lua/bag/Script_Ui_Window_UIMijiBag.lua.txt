Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
local tbWnd = Ui:CreateWindow("UIMijiBag")
local SecretBook = Item.SecretBook
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  ItemTemplate = "node/imgBG/Content/MijiList/maskView/Content/ItemTemplate",
  btnEquip = "node/imgBG/Content/btnAdd"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
    Guide:ClickWindowCloseCallSkipGuide(Guide.tbClickCloseButtonSkipGuide.UIMijiBag)
  end,
  btnEquip = function(self)
    if self.nNowSelect ~= nil then
      SecretBook:UICallUseSecretBook(self.tbItemIdList[self.nNowSelect])
    end
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.ItemTemplate)
end

tbWnd.szSelectEffectKey = "imgSelected"

function tbWnd:OnOpen(nRank, tbPItem)
  self.nRank = nRank
  self.nNowSelect = nil
  self.tbScrollView:Reset()
  self.tbItemIdList = {}
  for nIndex, pItem in ipairs(tbPItem) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      local tbItemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      tbElement.pPanel:SetActive(self.szSelectEffectKey, false)
      tbItemGrid:SetItem(pItem.dwId, pItem.dwTemplateId, nil, me.nFaction, me.nFactionSect, me.nSex)
      tbItemGrid:SetClickEnable(true, self.OnClickItem, self, nIndex, tbItemGrid)
      tbItemGrid.szItemOpt = "ItemBox"
    end)
    self.tbItemIdList[nIndex] = pItem.dwId
  end
  self:SetSelectIndex(nil, true)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self.OnSyncSecretBookCloseWindow, self)
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self)
end

function tbWnd:OnClickItem(nIndex, tbItemGrid)
  self:SetSelectIndex(nIndex, false)
  Item:ShowItemDetail(tbItemGrid)
end

function tbWnd:SetSelectIndex(nIndex, bInitCall)
  local function fnSetEffectActive(nInnerIndex, bActive)
    self.tbScrollView.tbElementList[nInnerIndex].pPanel:SetActive(self.szSelectEffectKey, bActive)
  end
  
  local bSelectChange = self.nNowSelect ~= nIndex
  if bSelectChange then
    if self.nNowSelect ~= nil then
      fnSetEffectActive(self.nNowSelect, false)
    end
    self.nNowSelect = nIndex
    if self.nNowSelect ~= nil then
      fnSetEffectActive(self.nNowSelect, true)
    end
  end
  if bSelectChange or bInitCall then
    self.pPanel:Button_SetEnable(tbControls.btnEquip, self.nNowSelect ~= nil)
  end
end

function tbWnd:OnSyncSecretBookCloseWindow(nItemId, bUpdateAll)
  local pEquip = KItem.GetItemObj(nItemId)
  if nil ~= pEquip then
    local nRank = SecretBook:GetRank(pEquip)
    if nil ~= nRank and nRank == self.nRank then
      Ui:CloseWindow(self.ClassName)
    end
  end
end
