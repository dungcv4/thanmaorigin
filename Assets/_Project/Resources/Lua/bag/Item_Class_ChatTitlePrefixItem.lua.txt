local tbChatTitlePrefixItem = Item:GetClass("ChatTitlePrefixItem")
tbChatTitlePrefixItem.PARAM_ID = 1
tbChatTitlePrefixItem.PARAM_TIMEOUT = 2

function tbChatTitlePrefixItem:OnUse(pItem)
  local nResult = self:UseChatTitlePrefixItem(me, pItem.dwTemplateId, 1)
  return nResult
end

function tbChatTitlePrefixItem:UseChatTitlePrefixItem(pPlayer, nTemplateId, nCount)
  local nNum = pPlayer.GetItemCountByTypeInBag(nTemplateId)
  if nNum <= 0 then
    pPlayer.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\230\149\176\233\135\143\228\184\141\232\182\179"))
    return 0
  end
  local nChatTitlePrefixTempId = KItem.GetItemExtParam(nTemplateId, self.PARAM_ID)
  local nLimitTime = KItem.GetItemExtParam(nTemplateId, self.PARAM_TIMEOUT)
  return ChatMgr:GainNewChatTitlePrefix(pPlayer, nChatTitlePrefixTempId, nLimitTime)
end
