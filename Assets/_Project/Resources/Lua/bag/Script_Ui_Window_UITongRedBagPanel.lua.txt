function tbWnd:OnCreate()
  self.selectMemberList = self.selectMemberList or {}
  self.REDBAGMIN = RedPacketMgr.Cfg.nMinAmount
  self.REDBAGMAX = RedPacketMgr.Cfg.nMaxAmount
  self.tbRedBagPanelList = Ui:NewButtonGroup(self, {
    [1] = tbControls.BtnJinbi,
    [2] = tbControls.BtnYuanbao
  }, true, function(self, nIndex)
    self:GeneratedRedBagPanel(nIndex)
  end)
  self.tbRecieveYuanbaoScrollView = Ui:NewLoopScrollView(self, tbControls.RecieveYuanbaoTemplate)
  self.tbRecieveJinbiScrollView = Ui:NewLoopScrollView(self, tbControls.RecieveJinBiTemplate)
  self.tbSendYuanbaoScrollView = Ui:NewLoopScrollView(self, tbControls.ItemCell, "Item")

end

function tbWnd:HideBtn()

  self.pPanel:SetActive(tbControls.YuanbaoNumUIAddAndSubGroup .. "/btnReduce", false)
end

function tbWnd:OnActivate()
  self.nRoleInfoSession = nil
  self.tbRoleInfoList = nil
  self.tbRedBagPanelList:Select(1)
  RedPacketMgr:TryFetchAll()
  self:Refresh()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.BtnJinbi, {
    RedPointMgr.EVENT.TongCostRedBag
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.BtnYuanbao, {
    RedPointMgr.EVENT.TongYuanBaoRedBagSend,
    RedPointMgr.EVENT.TongYuanBaoRedBagRecv

end

function tbWnd:OnDeactivate()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.BtnJinbi)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.BtnYuanbao)
end

function tbWnd:OnUpdateCostRedBagUI()
  local CostRedBagData = RedPacketMgr:GetTongCache("Gold")
  if not CostRedBagData then
    return
  end
  local nCount = #CostRedBagData
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local tbRedBag = CostRedBagData[nIndex]
    local tbElem = tbLoopScroll:GetElement(nIndex)
    tbElem.tbRedBag = tbRedBag
    local pPanel = tbElem.pPanel
    pPanel:Label_SetText("txtTitle", tbRedBag:GetName())
    pPanel:Label_SetText("txtDescription", tbRedBag:GetDesc())
    local btnTxt = tbRedBag:CanOpen() and i18n.EasyGet("\233\162\134\229\143\150") or i18n.EasyGet("\230\159\165\231\156\139")
    pPanel:Label_SetText("btnRecieve/Text", btnTxt)
    Ui:Button_BindEvent(tbElem.pPanel, "btnRecieve", function()
      RedPacketMgr:OpenRedPacket(tbRedBag.nId)
    end)
  end
  
  self.tbRecieveJinbiScrollView:SetElement(nCount, fnSetItem)
end

function tbWnd:GeneratedRedBagPanel(nIndex)
  if 1 == nIndex then
    self.pPanel:SetActive(tbControls.YuanbaoPanel, false)
    self.pPanel:SetActive(tbControls.JinbiPanel, true)
    self:OnUpdateCostRedBagUI()
    self:showRedBackMain()
  elseif 2 == nIndex then
    self.pPanel:SetActive(tbControls.YuanbaoPanel, true)
    self.pPanel:SetActive(tbControls.JinbiPanel, false)
    self:OnUpdateYuanbaoRedBagList()
  end
end

function tbWnd:SelectRedBag()

    self.pPanel:SetActive(tbControls.chooseTypePanel, false)
    self.pPanel:SetActive(tbControls.LuckyPanel, true)
  elseif true == self.pPanel:Toggle_GetChecked(tbControls.normalSelect) then
    self.pPanel:SetActive(tbControls.chooseTypePanel, false)
    self.pPanel:SetActive(tbControls.NormalPanel, true)
  elseif true == self.pPanel:Toggle_GetChecked(tbControls.specialSelect) then
    self.pPanel:SetActive(tbControls.chooseTypePanel, false)
    self.pPanel:SetActive(tbControls.SpecialPanel, true)
  end

function tbWnd:checkNormalMoneyInput(nNum)
  local nMaxNum = self.REDBAGMAX
  local nMinNum = self.REDBAGMIN
  if 0 == nMaxNum then
    nMaxNum = 1
  end
  if nNum < nMinNum then
    me.CenterMsg(i18n.Get(Shop.Terms.InputLimit))
    nNum = nMinNum
  end
  if nMaxNum < nNum then
    me.CenterMsg(i18n.Get(Shop.Terms.InputLimit))
    nNum = nMaxNum
  end
  return nNum
end

function tbWnd:SendLuckyRedBag()
  if self.nLuckyMoneyValue > me.GetMoney("Gold") then
    me.CenterMsg(i18n.EasyGet("\233\135\145\229\184\129\228\184\141\232\182\179"))
    return
  end
  local des = self.pPanel:Input_GetText(tbControls.LuckyInputField)
  
  local function fnBid()
    local nAmount = self.nLuckyMoneyValue
    local szText = self:GetFinalDes(des)
    local nMaxReceiver = self.nLuckyRedBagCount
    if nAmount < nMaxReceiver then
      me.CenterMsg(i18n.EasyGet("\231\186\162\229\140\133\230\149\176\233\135\143\228\184\141\229\144\136\230\179\149!"))
      return
    end
    local bSendSuccess = RedPacketMgr:Send(RedPacketMgr.Category.Tong, RedPacketMgr.TYPE.RANDOM_AMOUNT, nAmount, nMaxReceiver, nil, szText)
    if not bSendSuccess then
      return
    end
    self:showRedBackMain()
    self.pPanel:Input_SetText(tbControls.LuckyInputField, "")
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\232\138\177\232\180\185{0}\233\135\145\229\184\129\229\143\145\233\128\129\231\186\162\229\140\133?", self.nLuckyMoneyValue), {
    {fnBid},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

  end
  return finalDes

function tbWnd:sendSpecialRedBag()
  self.selectMemberList = self.selectMemberList or {}
  if #self.selectMemberList * self.nSpecialMoneyValue > me.GetMoney("Gold") then
    me.CenterMsg(i18n.EasyGet("\233\135\145\229\184\129\228\184\141\232\182\179"))
    return
  end
  if 0 == #self.selectMemberList then
    me.CenterMsg(i18n.EasyGet("\232\175\183\233\128\137\230\139\169\230\140\135\229\174\154\228\186\186"))
    return
  end
  local des = self.pPanel:Input_GetText(tbControls.SpecialInputField)
  
  local function fnBid()
    local nRedBagCount = #self.selectMemberList
    local nAmount = self.nSpecialMoneyValue * nRedBagCount
    local szText = self:GetFinalDes(des)
    local nMaxReceiver = nRedBagCount
    local tbSpecialReceiver = {}
    for k, v in pairs(self.selectMemberList) do
      tbSpecialReceiver[v.nMemberId] = true
    end
    local bSendSuccess = RedPacketMgr:Send(RedPacketMgr.Category.Tong, RedPacketMgr.TYPE.EXCLUSIVE, nAmount, nMaxReceiver, tbSpecialReceiver, szText)
    if not bSendSuccess then
      return
    end
    me.CenterMsg(i18n.EasyGet("\228\184\147\229\177\158\231\186\162\229\140\133\229\143\145\233\128\129\230\136\144\229\138\159"))
    self:showRedBackMain()
    self.pPanel:Input_SetText(tbControls.SpecialInputField, "")
  end
  
  local sum = self.nSpecialMoneyValue * #self.selectMemberList
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\232\138\177\232\180\185{0}\233\135\145\229\184\129\229\143\145\233\128\129\231\186\162\229\140\133?", sum), {
    {fnBid},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function tbWnd:OnUpdataMyYuanbaoRedBag()
  self.pPanel:SetActive(tbControls.SendPanel, true)
  self.pPanel:SetActive(tbControls.SendSetting, false)
  local tbRedBagList = RedPacketMgr:GetTongCache("YuanBao", RedPacketMgr.fnOnlySelf)
  if not tbRedBagList then
    return
  end
  self.nRow = 3
  local nCount = Lib:CountTB(tbRedBagList)
  self.pPanel:SetActive(tbControls.NoSendYuanbao, 0 == nCount)
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local nCol = nCount - self.nRow * (nIndex - 1)
    if nCol > self.nRow then
      nCol = self.nRow
    end
    for i = 1, self.nRow do
      if i <= nCol then
        local itemObj = tbLoopScroll:GetGridElement(nIndex, i)
        local curIndex = (nIndex - 1) * self.nRow + i
        itemObj.pPanel:SetActive("Main", true)
        local tbRedBag = tbRedBagList[curIndex]
        local pPanel = itemObj.pPanel
        pPanel:Label_SetText("txtDescription", tbRedBag:GetDesc())
        pPanel:Label_SetText("RedBagValue/txtCoinCost", tbRedBag.nAmount)
        local hasSend = tbRedBag.nSendTime > 0
        pPanel:SetActive("btnSend", false)
        pPanel:SetActive("btnCheck", false)
        pPanel:SetActive("btnRecieve", false)
        if hasSend then
          if tbRedBag:CanOpen() then
            pPanel:SetActive("btnRecieve", true)
          else
            pPanel:SetActive("btnCheck", true)
          end
        else
          pPanel:SetActive("btnSend", true)
        end
        Ui:Button_BindEvent(pPanel, "btnSend", function()
          self.pPanel:SetActive(tbControls.SendPanel, false)
          self:OnClickYuanbaoRedBagSendShow(tbRedBag)
        end)
        Ui:Button_BindEvent(pPanel, "btnCheck", function()
          RedPacketMgr:ShowRedPacket(tbRedBag.nId)
        end)
        Ui:Button_BindEvent(pPanel, "btnRecieve", function()
          RedPacketMgr:OpenRedPacket(tbRedBag.nId)
        end)
      else
        tbLoopScroll:GetGridElement(nIndex, i).pPanel:SetActive("Main", false)
      end
    end
  end
  
  self.tbSendYuanbaoScrollView:SetElement(math.ceil(nCount / self.nRow), fnSetItem)
end

function tbWnd:OnUpdateOtherYuanbaoRedBag()
  local tbRedBagList = RedPacketMgr:GetTongCache("YuanBao", RedPacketMgr.fnExcludeSelf)
  if not tbRedBagList then
    return
  end
  if not (not self.bForceUpdateRoleInfo and self.tbRoleInfoList) or not next(self.tbRoleInfoList) then
    self.bForceUpdateRoleInfo = false
    local tbPlayerIdList = {}
    for _, tbRedBag in ipairs(tbRedBagList) do
      table.insert(tbPlayerIdList, tbRedBag.nOwner)
    end
    local bAllHit, nSession, tbRoleInfoList = Player.RoleInfoCache:GetCache(tbPlayerIdList)
    if not bAllHit then
      self.nRoleInfoSession = nSession
      return
    end
    self.tbRoleInfoList = tbRoleInfoList
  end
  local tbRoleInfoList = self.tbRoleInfoList
  local nCount = #tbRedBagList
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local tbRedBag = tbRedBagList[nIndex]
    local tbElem = tbLoopScroll:GetElement(nIndex)
    tbElem.tbRedBag = tbRedBag
    local pPanel = tbElem.pPanel
    pPanel:Label_SetText("txtPlayerName", i18n.EasyGet("{0}\231\154\132\231\186\162\229\140\133", tbRoleInfoList[tbRedBag.nOwner].szName))
    pPanel:Label_SetText("txtDescription", tbRedBag:GetDesc())
    pPanel:Label_SetText("RedBagValue/txtCoinCost", tbRedBag.nAmount)
    local hasSend = tbRedBag.nSendTime > 0
    pPanel:SetActive("imgMark", not hasSend)
    local bCanOpen = tbRedBag:CanOpen()
    if hasSend then
      pPanel:SetActive("btnRecieve", bCanOpen)
      pPanel:SetActive("btncheck", not bCanOpen)
    else
      pPanel:SetActive("btnRecieve", false)
      pPanel:SetActive("btncheck", false)
    end
    
    local function fn()
      RedPacketMgr:OpenRedPacket(tbRedBag.nId)
    end
    
    Ui:Button_BindEvent(tbElem.pPanel, "btnRecieve", fn)
    Ui:Button_BindEvent(tbElem.pPanel, "btncheck", fn)
  end
  
  self.tbRecieveYuanbaoScrollView:SetElement(nCount, fnSetItem)
end

  self.YuanbaoRedBagNumAddAndSubGroup:SetEnable(false)

    bShowMulti = false
  end
  self.pPanel:SetActive(tbControls.BtnYuanbaoAdd, bShowMulti)
end
