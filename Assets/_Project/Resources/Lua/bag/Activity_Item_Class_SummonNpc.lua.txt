local tbItem = Item:GetClass("SummonNpc")
tbItem.PARAM_NPC_TEMPLATEID = 1

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  local nNpcTemplateId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPC_TEMPLATEID)
  if not nNpcTemplateId then
    LogErr("[SummonNpc] OnUseMany npc dwTemplateId is not exist", it.dwTemplateId, nNpcTemplateId)
    return 0
  end
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.SummonNpc)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\228\184\141\229\156\168\230\180\187\229\138\168\230\156\159\233\151\180"))
    return 0
  end
  return pInstance:OnUseMany(nCount, nNpcTemplateId)
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "SummonNpc")
end
