local tbWnd = Ui:CreateWindow("UIWeddingRedBagOpenGold")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ROLE_STAY_INFO,
      self.onSyncRoleInfo
    }
  }
end

local tbControls = {
  Open = "Open",
  OpentxtName = "Open/img_redbagopen02_Mask/Content/txtName1",
  OpenPortraitObj = "Open/img_redbagopen02_Mask/Content/PortraitObj",
  OpentxtContent = "Open/img_redbagopen02_Mask/Content/PanelContent1/txtContent",
  OpentxtGetMoney = "Open/img_redbagopen02_Mask/Content/PanelGetMoney/txtGetMoney",
  OpentxtReceiveDetail = "Open/img_redbagopen02_Mask/Content/PanelReceiveDetail/txtReceiveDetail",
  OpenScrollView = "Open/img_redbagopen02_Mask/Content/PanelReceiveDetail/ScrollView",
  OpentxtNoMoney = "Open/img_redbagopen02_Mask/Content/txtNoMoney",
  ReceiveScrollViewTemplate = "Open/img_redbagopen02_Mask/Content/PanelReceiveDetail/ScrollView/ViewPort/List/Template",
  Unopen = "Unopened",
  UnopentxtName = "Unopened/Content/txtName2",
  UnopenPortraitObj = "Unopened/Content/PortraitObj",
  UnopentxtContent = "Unopened/Content/PanelContent2/txtContent",
  UnopenBtnOpen = "Unopened/Content/BtnOpen",
  btnCloseAnimation = "btnCloseAnimation"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCloseAnimation = function(self)
    self.pPanel:SetActive(tbWnd.btnCloseAnimation, false)
    local Animator = self.pPanel:GetAnimator("")
    Animator:Update(Env.GAME_FPS * 1)
  end
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.ReceiveScrollView = Ui:NewScrollView(self, tbControls.ReceiveScrollViewTemplate)
end

function tbWnd:OnOpen(tbData)
  local nValue = 0
  local tbPlayerIdList = {}
  table.insert(tbPlayerIdList, tbData.nOwner)
  for _, tbRecord in pairs(tbData.tbRecord) do
    table.insert(tbPlayerIdList, tbRecord.nId)
    if me.dwID == tbRecord.nId then
      nValue = tbRecord.nAmount
    end
  end
  local bAllHit, nSession, tbRoleInfoList = Player.RoleInfoCache:GetCache(tbPlayerIdList)
  self.tbData = tbData
  self.nSession = nSession
  self.tbRoleInfoList = tbRoleInfoList
  self.pPanel:Label_SetText(tbControls.OpentxtContent, tbData.szDesc)
  if 0 == nValue then
    self.pPanel:Label_SetText(tbControls.OpentxtGetMoney, i18n.EasyGet("\230\156\170\230\138\162\229\136\176"))
  else
    self.pPanel:Label_SetText(tbControls.OpentxtGetMoney, nValue)
  end
  local szReceiveInfo = i18n.EasyGet("\229\183\178\233\162\134\229\143\150{0}/{1},\229\133\177{2}", #tbData.tbRecord, tbData.nCount, tbData.nValue)
  self.pPanel:Label_SetText(tbControls.OpentxtReceiveDetail, szReceiveInfo)
  self:ShowSender()
  self:ShowRecord()
end

function tbWnd:InitHead(pGrid, tbData)
  if tbData then
    local szFactionIcon = Faction:GetIcon(tbData.nFaction)
    local szHead, szAtlas = PlayerPortrait:GetIcon(tbData.nPortrait)
    self.pPanel:Label_SetText(pGrid .. "/Level/Text", tbData.nLevel)
    self.pPanel:Sprite_SetSprite(pGrid .. "/imgFactionIcon", szFactionIcon)
    self.pPanel:Sprite_SetSprite(pGrid .. "/Portrait/Image", szHead)
  end
end

function tbWnd:onSyncRoleInfo(nSession, tbRoleInfoList)
  if nSession == self.nSession then
    self.nSession = nil
    self.tbRoleInfoList = tbRoleInfoList
    self:ShowSender()
    self:ShowRecord()
  end
end

function tbWnd:ShowSender()
  local tbOwnerData = self.tbRoleInfoList[self.tbData.nOwner]
  if tbOwnerData then
    self:InitHead(tbControls.OpenPortraitObj, tbOwnerData)
    self.pPanel:Label_SetText(tbControls.OpentxtName, i18n.EasyGet("{0}\231\154\132\231\186\162\229\140\133", tbOwnerData.szName))
  else
    self.pPanel:Label_SetText(tbControls.OpentxtName, i18n.EasyGet("\229\138\160\232\189\189\228\184\173\231\154\132\231\186\162\229\140\133..."))
  end
end

function tbWnd:ShowRecord()
  self.ReceiveScrollView:Clear()
  for index, tbItem in pairs(self.tbData.tbRecord) do
    self.ReceiveScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:Label_SetText("txtMoney", tbItem.nAmount)
      local tbRoleInfo = self.tbRoleInfoList[tbItem.nId]
      if tbRoleInfo then
        tbElement.pPanel:Label_SetText("txtName", tbRoleInfo.szName)
        local szHeadPath = string.format("Open/img_redbagopen02_Mask/Content/PanelReceiveDetail/ScrollView/ViewPort/List/Element%d/PortraitObj", index)
        self:InitHead(szHeadPath, tbRoleInfo)
      else
        tbElement.pPanel:Label_SetText("txtName", i18n.EasyGet("\229\138\160\232\189\189\228\184\173..."))
      end
    end)
  end
end
