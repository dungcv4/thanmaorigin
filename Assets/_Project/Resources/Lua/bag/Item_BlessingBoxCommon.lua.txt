Item.BlessingBox = Item.BlessingBox or {}
local BlessingBox = Item.BlessingBox

function BlessingBox:InitConfig()
  self.tbBlessingBox = Lib:EasyLoadTabFile("Setting/Item/BlessingBox.tab", {"nParamId"})
  local tbAwardList = Lib:EasyLoadTabFile("Setting/Item/BlessingBoxAwardList.tab")
  for _, tbInfo in ipairs(tbAwardList) do
    assert(self.tbBlessingBox[tbInfo.nParamId], string.format("BlessingBox:InitConfig nParamId=%d not exist", tbInfo.nParamId))
    self.tbBlessingBox[tbInfo.nParamId].tbAwardList = self.tbBlessingBox[tbInfo.nParamId].tbAwardList or {}
    self.tbBlessingBox[tbInfo.nParamId].tbAwardList[tbInfo.nAwardKey] = tbInfo
    self.tbBlessingBox[tbInfo.nParamId].tbMaxStepList = self.tbBlessingBox[tbInfo.nParamId].tbMaxStepList or {}
    tbInfo.nMaxStep = tbInfo.nMaxStep or 0
    tbInfo.nMaxNum = tbInfo.nMaxNum or 0
    if tbInfo.nMaxStep > 0 then
      table.insert(self.tbBlessingBox[tbInfo.nParamId].tbMaxStepList, {
        tbInfo.nAwardKey,
        tbInfo.nMaxStep
      })
    end
  end
  for _, tbInfo in pairs(self.tbBlessingBox) do
    table.sort(tbInfo.tbMaxStepList, function(tbFirst, tbSecond)
      if tbFirst[2] ~= tbSecond[2] then
        return tbFirst[2] > tbSecond[2]
      end
      return tbFirst[1] > tbSecond[1]
    end)
  end
  self.tbBlessingBox = SetReadOnly(self.tbBlessingBox)
end

BlessingBox:InitConfig()

function BlessingBox:GetCfgById(nParamId)
  return self.tbBlessingBox[nParamId]
end

function BlessingBox:UseBlessingBox(dwId, nCount)
  RemoteServer.UseManyItem(dwId, nCount)
end
