local tbCountAddProp = Item:GetClass("CountAddProp")

function tbCountAddProp:LoadSetting()
  local tbField = {
    "TemplateId",
    "AddDegree",
    "AddCount",
    "CostDegree"
  }
  self.tbAllPropInfo = Lib:LoadTabFileEx("Setting/Item/Other/CountAddProp.tab", "dsds", "TemplateId", tbField)
  self.tbAllPropInfo = SetReadOnly(self.tbAllPropInfo)
  assert(self.tbAllPropInfo)
end

tbCountAddProp:LoadSetting()

function tbCountAddProp:GetAddDegreeInfo(szDegree)
  for _, tbInfo in pairs(self.tbAllPropInfo) do
    if tbInfo.AddDegree == szDegree then
      return tbInfo
    end
  end
end

function tbCountAddProp:GetAddDegreeInfoByTID(nItemTemplateId)
  for _, tbInfo in pairs(self.tbAllPropInfo) do
    if tbInfo.TemplateId == nItemTemplateId then
      return tbInfo
    end
  end
  return nil
end
