function EquipEnhance:SetNotShowValue(isOn)

function EquipEnhance:OnLoginC()

end

function EquipEnhance:GetGridEffectPath(nLevel)
  local szRet
  for i, szPath in pairs(self.tbLevel2GridEffect) do
    if i <= nLevel then
      szRet = szPath
    end
  end
  return szRet
end

function EquipEnhance:EnhanceReq(pPlayer, nPos, tbItemList)
  local bRet, szErr = self:CanEnhanceWithReminderWindow(pPlayer, nPos, tbItemList)
  if not bRet then
    pPlayer.CenterMsg(szErr)
    return
  end
  local nOrgLv = self:GetEquipEnhanceLevel(pPlayer, nPos)
  
  local function fnCallBack()
    RemoteServer.RequestEquipEnhance("EnhanceReq", nPos, tbItemList)
  end
  
  if self.tbLimitEnhancePercent[nOrgLv] then
    local pEquip = pPlayer.GetEquipByPos(nPos)
    local nUseVal, nReqVal, nNewLv, nPercent = self:CalcEnhanceRate(nPos, nOrgLv, tbItemList, pEquip, pPlayer)
    local nSetTime = me.GetUserValue(Item.EquipEnhance.SAVE_ENHANCE_GROUP, Item.EquipEnhance.NOT_SHOW_TIP) or 0
    if Lib:IsDiffDay(0, nSetTime, GetTime()) and nPercent < 100 then
      local szText = i18n.EasyGet("\230\173\164\230\147\141\228\189\156\229\173\152\229\156\168\229\164\177\232\180\165\233\163\142\233\153\169\239\188\140\229\188\186\229\140\150\229\164\177\232\180\165\229\176\134\230\141\159\229\164\177\229\188\186\229\140\150\230\157\144\230\150\153\239\188\140\230\152\175\229\144\166\231\161\174\232\174\164\229\188\186\229\140\150\239\188\159")
      Ui:OpenWindow("UIMessageBox", szText, {
        {fnCallBack},
        {}
      }, {
        i18n.szOk,
        i18n.szCancel
      }, "EquipEnhance", nil, i18n.EasyGet("\228\187\138\230\151\165\228\184\141\229\134\141\230\143\144\233\134\146"))
    else
      fnCallBack()

  else
    fnCallBack()

end

function EquipEnhance:EnhanceRsp(bSuccess, nPos, nOrgLv, nNewLv)
  if bSuccess then
    self:RefreshAttrib(me, nPos, nOrgLv, nNewLv)
  end
  if Ui:WindowVisible("UIEquipment") then
    Ui:GetWindow("UIEquipment"):NotifyEquipEnhance("Enhance", bSuccess, nPos)
  end
  self:OnCheckRedPoint()
end

function EquipEnhance:ComposeReq(pPlayer, tbItemList)
  local bRet, szErr = self:CanComposeWithReminderWindow(pPlayer, tbItemList)
  if not bRet then
    pPlayer.CenterMsg(szErr)
    return false
  end
  RemoteServer.RequestEquipEnhance("ComposeReq", tbItemList)
  return true
end

  if Ui:WindowVisible("UIEquipment") then
    Ui:GetWindow("UIEquipment"):NotifyXuanJingCompose()
  end
end

function EquipEnhance:SplitReq(pPlayer, nItemId, nNum)

function EquipEnhance:QuickCheckEnhance(pPlayer, pEquip)
  local nHasVal = 0
  for _, nItemTid in pairs(self.XUANJING_ITEM) do
    local nNum = pPlayer.GetItemCountInBags(nItemTid)
    if 0 ~= nNum then
      nHasVal = nHasVal + KItem.GetBaseValue(nItemTid) * nNum
    end
  end
  local nHasMoney = pPlayer.GetMoney(self.MONEY_TYPE)
  local nReqVal, nReqMoney = self:CalcEnhanceCost(pPlayer, pEquip, self:GetEquipEnhanceLevel(pPlayer, pEquip.nEquipPos))
  if nReqVal and nReqMoney and nHasVal >= nReqVal and nHasMoney >= nReqMoney then
    return true
  else
    return false
  end
end

function EquipEnhance:CanShowEnhanceButtonInUIEquipTips(nItemId)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return false
  end
  local nEquipPos = pItem.nEquipPos
  if nEquipPos < Item.EQUIPPOS_HELM or nEquipPos > Item.EQUIPPOS_AMULET then
    return false
  end
  return true
end

function EquipEnhance:OnCheckRedPoint()
  local nHasVal = 0
  for _, nItemTid in pairs(self.XUANJING_ITEM) do
    local nNum = me.GetItemCountInBags(nItemTid)
    if 0 ~= nNum then
      nHasVal = nHasVal + KItem.GetBaseValue(nItemTid) * nNum
    end
  end
  local nHasMoney = me.GetMoney(self.MONEY_TYPE)
  for nPos, nRedPoint in pairs(self.EQUIPPOS_2_REDPOINT) do
    local pEquip = me.GetEquipByPos(nPos)
    if pEquip then
      local nEnhanceLevel = self:GetEquipEnhanceLevel(me, pEquip.nEquipPos)
      if not Item.EquipWuLian:IsWuLianOpenC(nEnhanceLevel, pEquip.nLevel) then
        local nReqVal, nReqMoney = self:CalcEnhanceCost(me, pEquip, nEnhanceLevel)
        local nDegree = DegreeCtrl:GetDegree(me, self.POS_2_DEGREE[nPos])
        if nReqVal and nReqMoney and nHasVal >= nReqVal and nHasMoney >= nReqMoney and nDegree > 0 then
          RedPointMgr:LightRedPoint(nRedPoint)
        else
          RedPointMgr:FadeRedPoint(nRedPoint)
        end
      else
        local nReqVal, nReqMoney = Item.EquipWuLian:CalcWuLianCost(me, pEquip, Item.EquipWuLian:GetEquipWuLianLevel(me, pEquip.nEquipPos))
        local nDegree = DegreeCtrl:GetDegree(me, self.POS_2_DEGREE[nPos])
        if nReqVal and nReqMoney and nHasVal >= nReqVal and nHasMoney >= nReqMoney and nDegree > 0 then
          RedPointMgr:LightRedPoint(nRedPoint)
        else
          RedPointMgr:FadeRedPoint(nRedPoint)
        end
      end
    else
      RedPointMgr:FadeRedPoint(nRedPoint)
    end
  end
end

function EquipEnhance:OnSyncItemLimitTime()
  if not PlayerEvent.bLogin then
    return
  end
  self:OnCheckRedPoint()
end
