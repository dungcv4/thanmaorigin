local tbItem = Item:GetClass("TongFundItem")
tbItem.PARAM_TONGFUN = 1

function tbItem:OnUse(it)
  local nAddFund = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_TONGFUN)
  if nAddFund <= 0 then
    return
  end
  if me.dwTongId <= 0 then
    me.CenterMsg(i18n.EasyGet("\230\178\161\230\156\137\229\184\174\228\188\154\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168{0}", i18n.Parse(it.szName)))
    return
  end
  Tong:ChangePersonalStock(me.dwID, nAddFund, Env.LogWay_UseItem, it.dwTemplateId)
  me.CenterMsg(i18n.EasyGet("\228\189\160\228\189\191\231\148\168\228\186\134\228\184\128\228\184\170{0}\239\188\140\230\156\172\229\184\174\229\146\140\228\184\170\228\186\186\232\181\132\228\186\167\229\144\132\229\162\158\229\138\160{1}", i18n.Parse(it.szName), nAddFund))
  Tong:NotifyUpdatePlayerMemberListUI(me.dwTongId)
  return 1
end

function tbItem:OnUseMany(it, nCount)
  if nCount <= 0 then
    return 0
  end
  local nAddFund = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_TONGFUN)
  if nAddFund <= 0 then
    return
  end
  nAddFund = nAddFund * nCount
  if me.dwTongId <= 0 then
    me.CenterMsg(i18n.EasyGet("\230\178\161\230\156\137\229\184\174\228\188\154\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168{0}", i18n.Parse(it.szName)))
    return
  end
  Tong:ChangePersonalStock(me.dwID, nAddFund, Env.LogWay_UseItem, it.dwTemplateId)
  me.CenterMsg(i18n.EasyGet("\228\189\160\228\189\191\231\148\168\228\186\134{0}\228\184\170{1}\239\188\140\230\156\172\229\184\174\229\146\140\228\184\170\228\186\186\232\181\132\228\186\167\229\144\132\229\162\158\229\138\160{2}", nCount, i18n.Parse(it.szName), nAddFund))
  Tong:NotifyUpdatePlayerMemberListUI(me.dwTongId)
  return nCount
end
