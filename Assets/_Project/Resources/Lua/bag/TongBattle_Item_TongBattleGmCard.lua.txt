local tbItem = Item:GetClass("TongBattleGmCard")

function tbItem:OnClientUse(it)
  Ui:CloseWindow("UIBag")
  Ui:CloseWindow("UIItemTips")
  return 1
end

function tbItem:OnUse(it)
  local tbOptList = {}
  self:GetTongBattleWatch(tbOptList)
  self:GetFirstTongWatch(tbOptList)
  table.insert(tbOptList, {
    Text = i18n.EasyGet("\230\136\145\229\143\170\230\152\175\230\157\165\231\156\139\231\156\139")
  })
  Dialog:ShowNpcDialog({
    Text = i18n.EasyGet("\232\175\183\233\128\137\230\139\169\232\167\130\230\136\152\231\154\132\230\175\148\232\181\155"),
    OptList = tbOptList
  }, me, TongBattle.NPC_ID)
  return 0
end

function tbItem:GetTongBattleWatch(tbOptList)
  if Lib:IsEmptyTB(TongBattle.tbWatchData) then
    return
  end
  for nMapId, tbData in pairs(TongBattle.tbWatchData) do
    local szMatch = (tbData.tbTong[1] and tbData.tbTong[1].szTongName or "") .. " vs " .. (tbData.tbTong[2] and tbData.tbTong[2].szTongName or "")
    table.insert(tbOptList, {
      Text = szMatch,
      Callback = TongBattle.WatchMatch,
      Param = {
        TongBattle,
        me,
        nMapId
      }
    })
  end
end

function tbItem:GetFirstTongWatch(tbOptList)
  if not FirstTong.tbWatchData or Lib:IsEmptyTB(FirstTong.tbWatchData) then
    return
  end
  for nMapId, tbData in pairs(FirstTong.tbWatchData) do
    local szMatch = table.concat(tbData.tbTong, " vs ")
    table.insert(tbOptList, {
      Text = szMatch,
      Callback = FirstTong.WatchMatch,
      Param = {
        FirstTong,
        me,
        nMapId
      }
    })
  end
end
