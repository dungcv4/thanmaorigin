Require("CommonScript/FieldPractice/FieldPractice.lua")
local tbDef = FieldPractice.tbDef
local tbItem = Item:GetClass("XiuLianDan")
tbItem.nExtAddResiduTime = 1

function tbItem:CheckCanUse(pPlayer, nAddResiduTime)
  local bRet, szMsg = FieldPractice:CheckCanUseXiuLianDan(pPlayer, nAddResiduTime)
  if not bRet then
    return false, szMsg
  end
  local nCount = FieldPractice:GetXiuLianDanCounts(pPlayer)
  if nCount <= 0 then
    return false, i18n.EasyGet("\228\187\138\230\151\165\228\184\141\229\143\175\229\134\141\230\156\141\231\148\168\228\191\174\231\130\188\228\184\185\228\186\134\239\188\136\230\172\161\230\151\1654:00\229\143\175\231\148\168\239\188\137")
  end
  return true, ""
end

function tbItem:OnUse(it)
  if SERVER_RUNNING_MODE == SERVER_MODE.ZONE_SERVER and not GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
    return
  end
  local nAddResiduTime = KItem.GetItemExtParam(it.dwTemplateId, self.nExtAddResiduTime)
  if nAddResiduTime <= 0 then
    Log("Error XiuLianDan AddResiduTime", nAddResiduTime)
    return
  end
  local bRet, szMsg = self:CheckCanUse(me, nAddResiduTime)
  if not bRet then
    me.CenterMsg(szMsg)
    return
  end
  local nCount = FieldPractice:GetXiuLianDanCounts(me)
  FieldPractice:SetXiuLianDanCounts(me, nCount - 1)
  FieldPractice:AddXiuLianTime(me, nAddResiduTime)
  Event:OnEvent(Event.emEVENT_USE_XIU_LIAN_DAN, me.dwID)
  me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\230\136\144\229\138\159\239\188\140\229\162\158\229\138\160\228\186\134{0}\229\136\134\233\146\159\228\191\174\231\130\188\230\151\182\233\151\180", math.floor(nAddResiduTime / 60)))
  return 1
end

function tbItem:GetTip(it)
  if not it.dwId then
    return ""
  end
  local nCount = FieldPractice:GetXiuLianDanCounts(me)
  local szTip = i18n.EasyGet("\228\187\138\230\151\165\232\191\152\229\143\175\228\189\191\231\148\168{0}\228\184\170", nCount)
  if nCount <= 0 then
    szTip = string.format("%s%s", szTip, i18n.EasyGet("<color=#ffff0e>\239\188\136\230\172\161\230\151\1654:00\229\143\175\229\134\141\230\172\161\228\189\191\231\148\168\239\188\137</color>"))
  end
  return szTip
end
