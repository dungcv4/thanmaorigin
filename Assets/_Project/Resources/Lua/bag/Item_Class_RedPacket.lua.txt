local tbItem = Item:GetClass("RedPacket")
tbItem.PARAM_REDPACKET_EVENTID = 1

function tbItem:OnUse(item)
  return self:OnUseMany(item, 1)
end

function tbItem:OnUseMany(item, nCount)
  local pPlayer = me
  if nCount > item.nCount then
    pPlayer.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\230\149\176\233\135\143\228\184\141\232\182\179"))
    return 0
  end
  return self:SendRedPacket(pPlayer, item.dwTemplateId, nCount)
end

function tbItem:OnAutoUse(pPlayer, nTemplateId, nCount)
  self:SendRedPacket(pPlayer, nTemplateId, nCount)
  return true
end

function tbItem:SendRedPacket(pPlayer, nTemplateId, nCount)
  local nEventId = KItem.GetItemExtParam(nTemplateId, self.PARAM_REDPACKET_EVENTID)
  local bOk, szMsg
  local nCostNum = 0
  for i = 1, nCount do
    bOk, szMsg = RedPacketMgr:SendCommonRedPacket(pPlayer, nEventId)
    if not bOk then
      pPlayer.CenterMsg(szMsg)
      break
    end
    nCostNum = nCostNum + 1
  end
  if nCostNum ~= nCount then
    LogErr("[RedPacketItem] SendRedPacket fail, not all success", pPlayer.dwID, nTemplateId, nCostNum, nCostNum)
  end
  return nCostNum
end
