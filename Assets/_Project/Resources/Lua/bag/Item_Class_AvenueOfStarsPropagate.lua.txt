local tbItem = Item:GetClass("AvenueOfStarsPropagate")
if CLIENT then
  function tbItem:OnClientUse(pItem)
    AvenueOfStars:SendSlogan_C(pItem.dwTemplateId)
    
    return 0
  end
end
if MODULE_GAMESERVER then
  function tbItem:OnUse(pItem)
    AvenueOfStars:SendSlogan_S(me, pItem.dwTemplateId)
    
    return 0
  end
end
