local tbWuXingYin = Item:GetClass("WuXingYin")
local tbEquip = Item:GetClass("equip")
Item.EQUIP_WUXINGYIN_FIXATTR_LV_BEGIN = 1
Item.EQUIP_WUXINGYIN_FIXATTR_LV_END = 2
Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_BEGIN = 3
Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_END = 8
Item.WUXINGYIN_CLASS_NAME = "WuXingYin"
tbWuXingYin.WuXingYinOpenLevel = 35
tbWuXingYin.WuXingYinOpenTimeFrame = "OpenDay2B"

function tbWuXingYin:OnCreate(pEquip)
end

function tbWuXingYin:OnInit(pEquip)
  local nValue = pEquip.nOrgValue
  local nFightPower = pEquip.nBaseFightPower
  local nFixAttrFight = self:GetFixAttrFightPower(pEquip)
  local nRandAttrFight = self:GetRandAttrFightPower(pEquip)
  pEquip.nFightPower = nFightPower + nFixAttrFight + nRandAttrFight
  pEquip.nLevelFightPower = nFixAttrFight
  pEquip.nGradeFightPower = nFightPower
  pEquip.nAttrFightPower = nRandAttrFight
  pEquip.SetSingleValue(nValue)
end

function tbWuXingYin:UpdateFightPower(pEquip)
  local nFixAttrFight = self:GetFixAttrFightPower(pEquip)
  local nRandAttrFight = self:GetRandAttrFightPower(pEquip)
  pEquip.nLevelFightPower = nFixAttrFight
  pEquip.nAttrFightPower = nRandAttrFight
  pEquip.nFightPower = pEquip.nBaseFightPower + nFixAttrFight + nRandAttrFight
end

function tbWuXingYin:GetFixAttrFightPower(pEquip)
  local nFightPower = 0
  for nId = Item.EQUIP_WUXINGYIN_FIXATTR_LV_BEGIN, Item.EQUIP_WUXINGYIN_FIXATTR_LV_END do
    local nAttr = pEquip.GetIntValue(nId)
    local nLevel, _ = Item.WuXingYin:UnpackAttr(nAttr)
    local szDegree = Item.WuXingYin.FIXATTR_INDEX[nId].szDegree
    for i = 0, nLevel - 1 do
      local nMaxDegree = Item.WuXingYin.tbFixAttrUp[i][szDegree]
      nFightPower = nFightPower + math.floor(nMaxDegree * 0.5)
    end
  end
  return nFightPower
end

function tbWuXingYin:GetRandAttrFightPower(pEquip)
  local nFightPower = 0
  for nId = Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_BEGIN, Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_END do
    local nAttr = pEquip.GetIntValue(nId)
    if 0 ~= nAttr then
      local nGroup, nLevel = Item.WuXingYin:UnpackAttr(nAttr)
      local tbInfo = Item.WuXingYin:GetAttrFromPool(nGroup, nLevel)
      nFightPower = nFightPower + tbInfo.nFightPower
    end
  end
  return nFightPower
end

function tbWuXingYin:GetFightPowerByAttrList(tbAttrList)
  local nFightPower = 0
  for nId = Item.EQUIP_WUXINGYIN_FIXATTR_LV_BEGIN, Item.EQUIP_WUXINGYIN_FIXATTR_LV_END do
    local nAttr = tbAttrList[nId]
    local nLevel, _ = Item.WuXingYin:UnpackAttr(nAttr)
    local szDegree = Item.WuXingYin.FIXATTR_INDEX[nId].szDegree
    for i = 0, nLevel - 1 do
      local nMaxDegree = Item.WuXingYin.tbFixAttrUp[i][szDegree]
      nFightPower = nFightPower + math.floor(nMaxDegree * 0.5)
    end
  end
  for nId = Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_BEGIN, Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_END do
    local nAttr = tbAttrList[nId]
    if 0 ~= nAttr then
      local nGroup, nLevel = Item.WuXingYin:UnpackAttr(nAttr)
      local tbInfo = Item.WuXingYin:GetAttrFromPool(nGroup, nLevel)
      nFightPower = nFightPower + tbInfo.nFightPower
    end
  end
  return nFightPower
end

function tbWuXingYin:GetTip(pEquip, pPlayer, bIsCompare)
  local tbtips = self:FormatTips(pEquip.dwTemplateId, pEquip)
  return tbtips
end

function tbWuXingYin:GetTipByTemplate(nTemplateId, tbLinkInfo)
  local tbTips = {}
  local tbInfo = KItem.GetEquipBaseProp(nTemplateId)
  local tbAttrib = tbInfo.tbBaseAttrib
  local tbBaseProp = Item:GetItemBaseProp(nTemplateId)
  table.insert(tbTips, {
    i18n.Get(Item.Equip.tbText.szBaseAttribute)
  })
  for i, tbMA in ipairs(tbAttrib) do
    if tbLinkInfo and tbLinkInfo.tbWuXingYinAttr then
      local _, tbValue = Item.WuXingYin:GetFixAttrByValue(i, tbLinkInfo.tbWuXingYinAttr[i])
      for k, nV in ipairs(tbMA.tbValue) do
        tbMA.tbValue[k] = nV + tbValue[k]
      end
    end
    local szDesc = tbEquip:GetMagicAttribDesc(tbMA.szName, tbMA.tbValue, tbMA.nActiveReq)
    table.insert(tbTips, {
      {szDesc}
    })
  end
  if not tbLinkInfo or not tbLinkInfo.tbWuXingYinAttr then
    return tbTips
  end
  local tbRandAttr = {}
  for i = Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_BEGIN, Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_END do
    local szCurName, szCurValue, nQuality = Item.WuXingYin:GetRandAttrByValue(tbBaseProp.nLevel, tbLinkInfo.tbWuXingYinAttr[i])
    local szColor = Item.WuXingYin.ATTRCOLOR[nQuality]
    local szMagic = string.format("%s : %s", szCurName, szCurValue)
    if szCurName then
      table.insert(tbRandAttr, {szMagic, szColor})
    end
  end
  if next(tbRandAttr) then
    table.insert(tbTips, {
      i18n.EasyGet("\233\154\143\230\156\186\229\177\158\230\128\167")
    })
    for index, tbValue in ipairs(tbRandAttr) do
      table.insert(tbTips, {tbValue})
    end
  end
  return tbTips
end

function tbWuXingYin:FormatTips(nTemplateId, pEquip)
  local tbTips = {}
  local tbAttrib = KItem.GetEquipBaseProp(nTemplateId).tbBaseAttrib
  local tbBaseProp = Item:GetItemBaseProp(nTemplateId)
  table.insert(tbTips, {
    i18n.Get(Item.Equip.tbText.szBaseAttribute)
  })
  for i, tbMA in ipairs(tbAttrib) do
    if pEquip.nEquipPos == Item.EQUIPPOS_WUXINGYIN then
      local tbValue = Item.WuXingYin:GetFixAttrLvValue(pEquip.dwId, i)
      for k, nV in ipairs(tbMA.tbValue) do
        tbMA.tbValue[k] = nV + tbValue[k]
      end
    end
    local szDesc = tbEquip:GetMagicAttribDesc(tbMA.szName, tbMA.tbValue, tbMA.nActiveReq)
    table.insert(tbTips, {
      {szDesc}
    })
  end
  local tbRandAttr = Item.WuXingYin:GetRandAttr(pEquip, pEquip.nLevel)
  if next(tbRandAttr) then
    table.insert(tbTips, {
      i18n.EasyGet("\233\154\143\230\156\186\229\177\158\230\128\167")
    })
    for key, tbInfo in pairs(tbRandAttr) do
      table.insert(tbTips, {
        {
          tbInfo.szMagic,
          tbInfo.szColor
        }
      })
    end
  end
  return tbTips
end

function tbWuXingYin:GetWuXingYinAttr(pEquip)
  local tbAttrList = {}
  if pEquip.szClass ~= Item.WUXINGYIN_CLASS_NAME then
    return tbAttrList
  end
  for nId = Item.EQUIP_WUXINGYIN_FIXATTR_LV_BEGIN, Item.EQUIP_WUXINGYIN_RANDATTR_VALUE_END do
    table.insert(tbAttrList, pEquip.GetIntValue(nId))
  end
  return tbAttrList
end

function tbWuXingYin:GetDetailLog(pEquip)
  if pEquip.szClass ~= Item.WUXINGYIN_CLASS_NAME then
    return
  end
  local tbLog = {}
  tbLog.tbWuXingYinAttr = table.concat(self:GetWuXingYinAttr(pEquip), ",")
  return tbLog
end

function tbWuXingYin:GetChatLinkInfo(pEquip)
  if pEquip.szClass ~= Item.WUXINGYIN_CLASS_NAME then
    return
  end
  local tbInfo = {}
  tbInfo.tbWuXingYinAttr = self:GetWuXingYinAttr(pEquip)
  return tbInfo
end
