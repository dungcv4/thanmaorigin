local tbItemClass = Item:GetClass("ArenaAddTimes")
tbItemClass.szDegreeKey = "Arena"

function tbItemClass:OnUse(pItem)
  if not self:CheckCanUse(me) then
    me.CenterMsg(i18n.EasyGet("\232\191\152\230\156\137\229\137\169\228\189\153\230\172\161\230\149\176\239\188\140\230\151\160\233\156\128\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  local nAddTime = KItem.GetItemExtParam(pItem.dwTemplateId, 1)
  DegreeCtrl:AddDegree(me, self.szDegreeKey, nAddTime)
  Arena.Mgr:SyncPlayerInfo(me)
  me.CenterMsg(i18n.EasyGet("{0}\228\189\191\231\148\168\230\136\144\229\138\159!", i18n.Parse(pItem.szName)))
  return 1
end

function tbItemClass:CheckCanUse(pPlayer)
  if 0 == DegreeCtrl:GetDegree(pPlayer, self.szDegreeKey) then
    return true
  end
  return false
end

function tbItemClass:OnClientUse(it)
  if not self:CheckCanUse(me) then
    me.CenterMsg(i18n.EasyGet("\232\191\152\230\156\137\229\137\169\228\189\153\230\172\161\230\149\176\239\188\140\230\151\160\233\156\128\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  return 1
end
