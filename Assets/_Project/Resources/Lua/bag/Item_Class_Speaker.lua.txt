local tbSpeaker = Item:GetClass("Speaker")

function tbSpeaker:OnUse(it)
  return self:OnAutoUse(me, it.dwTemplateId, 1)
end

function tbSpeaker:OnUseMany(it, nCount)
  return self:OnAutoUse(me, it.dwTemplateId, nCount)
end

function tbSpeaker:OnAutoUse(pPlayer, nItemTemplateId, nCount)
  local nBuyChatCount = KItem.GetItemExtParam(nItemTemplateId, 1)
  local nType = KItem.GetItemExtParam(nItemTemplateId, 2)
  local nTotalBuyChatCount = nBuyChatCount * nCount
  if nType == ChatMgr.ChannelType.Public then
    ChatMgr:AddPublicChatCount(pPlayer, nTotalBuyChatCount)
    pPlayer.CenterMsg(i18n.EasyGet("\229\162\158\229\138\160\228\186\134{0}\230\172\161\228\184\150\231\149\140\233\162\145\233\129\147\229\143\145\232\168\128\230\156\186\228\188\154", nTotalBuyChatCount))
    return nCount
  elseif nType == ChatMgr.ChannelType.Color then
    ChatMgr:AddColorChatCount(pPlayer, nTotalBuyChatCount)
    pPlayer.CenterMsg(i18n.EasyGet("\229\162\158\229\138\160\228\186\134{0}\230\172\161\229\189\169\232\129\138\233\162\145\233\129\147\229\143\145\232\168\128\230\156\186\228\188\154", nTotalBuyChatCount))
    pPlayer.CallClientScript("ChatMgr:UpdateColorMsgCount")
    return nCount
  elseif nType == ChatMgr.ChannelType.Zone then
    ChatMgr:AddZoneChatCount(pPlayer, nTotalBuyChatCount)
    pPlayer.CenterMsg(i18n.EasyGet("\229\162\158\229\138\160\228\186\134{0}\230\172\161\232\183\168\230\156\141\233\162\145\233\129\147\229\143\145\232\168\128\230\156\186\228\188\154", nTotalBuyChatCount))
    return nCount
  else
    pPlayer.CenterMsg(i18n.EasyGet("\230\156\170\231\159\165\229\150\135\229\143\173\231\177\187\229\158\139"))
  end
end
