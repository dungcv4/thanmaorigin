local tbItem = Item:GetClass("DomainBattleTransform")

function tbItem:OnUse(pItem)
  local bRet, szErr = DomainBattle:CanTransform(me, pItem)
  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  DomainBattle:DoTransform(me, pItem)
end

function tbItem:OnClientUse()
  return 1
end

function tbItem:Give(nItemTemplateId, nItemId)
  Ui:CloseWindow("UIItemTips")
  Ui:OpenWindow("UIExperienceBook", nItemTemplateId, nItemId, "DomainbattleItem", false, false, "Tong")
end

function tbItem:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    Item:ClientUseItem(nItemId)
  end
  
  tbSetting.szSecondName = i18n.EasyGet("\232\181\160\233\128\129")
  
  function tbSetting.fnSecond()
    tbItem:Give(nTemplateId, nItemId)
  end
  
  return tbSetting
end
