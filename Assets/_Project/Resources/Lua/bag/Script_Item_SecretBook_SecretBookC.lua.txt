Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
local SecretBook = Item.SecretBook

function SecretBook:GetSecretBookInBag(nRank)
  local tbSecretInBag = me.GetItemListInBag(self.szClass)
  local tbRet = {}
  for _, pEquip in ipairs(tbSecretInBag) do
    if self:GetRank(pEquip) == nRank then
      table.insert(tbRet, pEquip)
    end
  end
  return tbRet
end

function SecretBook:GetCommonSkillSecretPanelUI()
  local tbWnd = Ui:GetWindow("UISkill")
  if nil ~= tbWnd and tbWnd.tbTabControl.nPanelType == tbWnd.PANEL_TYPE_COMMON then
    local tbWndCommonSkill = tbWnd.tbTabControl.tbPanelList[tbWnd.PANEL_TYPE_COMMON]
    if nil ~= tbWndCommonSkill and tbWndCommonSkill.tbTabControl.nPanelType == tbWndCommonSkill.PANEL_TYPE_SECRETBOOK then
      return tbWndCommonSkill.tbTabControl.tbPanelList[tbWndCommonSkill.PANEL_TYPE_SECRETBOOK]
    end
  end
end

function SecretBook:UICallUseSecretBook(dwEquipId)
  local pEquip = KItem.GetItemObj(dwEquipId)
  if nil ~= pEquip and self:IsSecretBook(pEquip) then
    Item.Equip:UseEquip(false, dwEquipId)
  end
end

function SecretBook:OnS2CSyncLevelUp(nRank)
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:OnSyncLevelUp()
  end
  self:RefreshARankCanLevelUpRedPoint(nRank, false)
end

function SecretBook:OnS2CLevelUpApplyAttribute(nAttribute, nAddValue)
  self:ApplyOrRemoveAnAttributeByEnum(me, nAttribute, nAddValue, true)
  local _, szAttributeName = self:GetMagicAttributeName(me, nAttribute)
  me.CenterMsg(string.format("%s+%d", szAttributeName, nAddValue))
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:PlayEffect_AddAttribute()
  end
end

function SecretBook:OnS2CSkillLevelUp(nRank)
  me.CenterMsg(i18n.Get(SecretBook.tbText.szSkillLevelIncrease, i18n.Get(SecretBook.tbEachRankCfg[nRank].szSecretBookName)))
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:PlayEffect_SkillLevelUp()
  end
end

function SecretBook:OnS2CExtraSkillLevel(nRank, nLevel)
  if not me.tbSecretBookExtraSkillLevel then
    me.tbSecretBookExtraSkillLevel = {}
  end
  me.tbSecretBookExtraSkillLevel[nRank] = nLevel
end

function SecretBook:OnS2CSellReturnXiuWei(nReturnValue)
  me.CenterMsg(i18n.Get(self.tbText.szReturnXiuWeiTips, nReturnValue))
end

function SecretBook:ClientApplyOrRemoveAnAttributeByIdentify(szIdentify, nValue, bIsApply)
  self:ApplyOrRemoveAnAttributeByIdentify(me, szIdentify, nValue, bIsApply)
end

function SecretBook:ClientApplyOrRemoveSkill(nSkillId, nSkillLevel, bIsApply)
  self:ApplyOrRemoveSkill(me, nSkillId, nSkillLevel, bIsApply)
end

SecretBook.tbLevelEnoughRedPointRank = SecretBook.tbLevelEnoughRedPointRank or {}
SecretBook.tbCanLevelUpRedPointRank = SecretBook.tbCanLevelUpRedPointRank or {}

function SecretBook:OnS2CRefreshCanLevelUpRedPoint()
  for nRank = 1, Lib:CountTB(self.tbRank) do
    self:RefreshARankCanLevelUpRedPoint(nRank, true)
  end
end

function SecretBook:OnUseOrUnuseEquip(bIsUse, nPos, dwEquipId)
  local pEquip = KItem.GetItemObj(dwEquipId)
  if nil ~= pEquip and self:IsSecretBook(pEquip) then
    local nRank = self:GetRank(pEquip)
    if bIsUse then
      self:FadeLevelEnoughRedPoint(nRank)
    end
    self:RefreshARankCanLevelUpRedPoint(nRank, false)
  end
end

function SecretBook:OnPlayerLevelUp()
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:OnPlayerLevelUp()
  end
  self:LevelUpCheckRedPoint()
end

function SecretBook:OnSyncItem(nItemId, bUpdateAll)
  local pEquip = KItem.GetItemObj(nItemId)
  if nil ~= pEquip and self:IsSecretBook(pEquip) then
    local tbPanel = self:GetCommonSkillSecretPanelUI()
    if nil ~= tbPanel then
      tbPanel:OnSyncSecretBook(pEquip)
    end
  end
end

function SecretBook:OnSyncXiuWeiValue(nOldValue, nNewValue)
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:OnSyncXiuWeiValue()
  end
  if nOldValue < nNewValue then
    local nLastLightTime = LocalData:GetRoleNumber("SecretBookLevelUpRedPoint", 0)
    if Lib:IsDiffDay(14400, nLastLightTime, GetTime()) then
      for nRank = 1, Lib:CountTB(self.tbRank) do
        self:RefreshARankCanLevelUpRedPoint(nRank, true)
      end
    end
  else
    for nRank = 1, Lib:CountTB(self.tbRank) do
      self:RefreshARankCanLevelUpRedPoint(nRank, false)
    end
  end
end

function SecretBook:LevelUpCheckRedPoint()
  if Ui.bInZone then
    return
  end
  local nNewLevel = me.nLevel
  for nRank = 1, Lib:CountTB(self.tbRank) do
    if nNewLevel == self.tbEachRankCfg[nRank].nUiShowOpenLevel then
      self:LightLevelEnoughRedPoint(nRank)
    end
  end
end

function SecretBook:LightLevelEnoughRedPoint(nRank)
  local bIsEmptyBefore = Lib:CountTB(self.tbLevelEnoughRedPointRank) <= 0
  self.tbLevelEnoughRedPointRank[nRank] = true
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:RegisterLevelEnoughRedPoint(nRank)
  end
  if bIsEmptyBefore then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.SecretBookLevelEnough)
  end
end

function SecretBook:FadeLevelEnoughRedPoint(nRank)
  self.tbLevelEnoughRedPointRank[nRank] = nil
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:UnRegisterLevelEnoughRedPoint(nRank)
  end
  if Lib:CountTB(self.tbLevelEnoughRedPointRank) <= 0 then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.SecretBookLevelEnough)
  end
end

function SecretBook:LightCanLevelUpRedPoint(nRank)
  local bIsEmptyBefore = Lib:CountTB(self.tbCanLevelUpRedPointRank) <= 0
  self.tbCanLevelUpRedPointRank[nRank] = true
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:RegisterLevelUpRedPoint(nRank)
  end
  if bIsEmptyBefore then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.SecretBookCanLevelUp)
    LocalData:SetRoleNumber("SecretBookLevelUpRedPoint", GetTime())
  end
end

function SecretBook:FadeCanLevelUpRedPoint(nRank)
  self.tbCanLevelUpRedPointRank[nRank] = nil
  local tbPanel = self:GetCommonSkillSecretPanelUI()
  if nil ~= tbPanel then
    tbPanel:UnRegisterLevelUpRedPoint(nRank)
  end
  if Lib:CountTB(self.tbCanLevelUpRedPointRank) <= 0 then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.SecretBookCanLevelUp)
  end
end

function SecretBook:RefreshARankCanLevelUpRedPoint(nRank, bCanLight)
  local bOldValue = self.tbCanLevelUpRedPointRank[nRank] ~= nil
  local bNewValue = self:CanLevelUp(me, nRank)
  if not bOldValue and bNewValue and bCanLight then
    self:LightCanLevelUpRedPoint(nRank)
  elseif bOldValue and not bNewValue then
    self:FadeCanLevelUpRedPoint(nRank)
  end
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_LEVEL, SecretBook.OnPlayerLevelUp, SecretBook)
EventNotify:RegistNotify(EventNotify.emNOTIFY_USE_UNUSE_EQUIP, SecretBook.OnUseOrUnuseEquip, SecretBook)
EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, SecretBook.OnSyncItem, SecretBook)
EventNotify:RegistNotify(EventNotify.emNOTIFY_XIU_WEI_CHANGE, SecretBook.OnSyncXiuWeiValue, SecretBook)
