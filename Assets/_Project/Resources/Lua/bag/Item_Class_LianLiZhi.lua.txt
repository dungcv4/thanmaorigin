local tbLianLiZhi = Item:GetClass("LianLiZhi")

function tbLianLiZhi:OnClientUse(it)
  if not Marry.tbMarryData.dwLoverID then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\230\178\161\230\156\137\228\190\160\228\190\163"))
    return 0
  end
  local tbInfo = FriendShip:GetFriendDataInfo(Marry.tbMarryData.dwLoverID)
  if not tbInfo or tbInfo.nState ~= Player.emPLAYER_STATE_NORMAL then
    me.CenterMsg(i18n.EasyGet("\228\190\160\228\190\163\229\189\147\229\137\141\228\184\141\229\156\168\231\186\191"))
    return 0
  end
  Ui:OpenWindow("UIGiftSystem", Marry.tbMarryData.dwLoverID, it.dwTemplateId)
  Ui:CloseWindow("UIItemTips")
  Ui:CloseWindow("UIBag")
  return 0
end
