local tbQiHun = Item:GetClass("QiHun")

function tbQiHun:OnCreate()
end

function tbQiHun:OnInit()
end

function tbQiHun:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function funcCloseWnd()
    Ui:CloseWindow("UIItemTips")
    Ui:CloseWindow("UIBag")
  end
  
  local function fnPutOn()
    Ui:OpenWindow("UIEquipment", Ui.UIEquipment.tbSubPanelIndex.PANEL_TYPE_QIHUN)
    funcCloseWnd()
  end
  
  local function fnCompose()
    Ui:OpenWindow("UIQiHun", 1, nItemTemplateId)
    funcCloseWnd()
  end
  
  local function fnReforge()
    Ui:OpenWindow("UIQiHun", 2, nItemTemplateId)
    funcCloseWnd()
  end
  
  local tbCfg = {
    [1] = {
      szName = "szFirstName",
      func = "fnFirst"
    },
    [2] = {
      szName = "szSecondName",
      func = "fnSecond"
    },
    [3] = {
      szName = "szThirdName",
      func = "fnThird"
    }
  }
  local tbSetting = {}
  local nIndex = 1
  local tbEquip = me.GetEquips()
  if Item.QiHun:IsSystemOpen(me) and Lib:CountTB(tbEquip) > 0 then
    tbSetting[tbCfg[nIndex].szName] = i18n.EasyGet("\233\149\182\229\181\140")
    tbSetting[tbCfg[nIndex].func] = fnPutOn
    nIndex = nIndex + 1
  end
  local nLevel = Item.QiHun:GetLevelByTemplateId(nItemTemplateId)
  if nLevel and Item.QiHun:CanCompose(me, nLevel) then
    tbSetting[tbCfg[nIndex].szName] = i18n.EasyGet("\229\144\136\230\136\144")
    tbSetting[tbCfg[nIndex].func] = fnCompose
    nIndex = nIndex + 1
  end
  if nLevel and Item.QiHun:CanReforge(me, nLevel) then
    tbSetting[tbCfg[nIndex].szName] = i18n.EasyGet("\233\135\141\233\147\184")
    tbSetting[tbCfg[nIndex].func] = fnReforge
  end
  return tbSetting
end
