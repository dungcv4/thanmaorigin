function QiHun:GetQiHunItemNum(nMinLevel, nMaxLevel)
  local tbRet = {}
  nMinLevel = nMinLevel or 1
  nMaxLevel = nMaxLevel or #self.tbLevelInfo
  for i = nMinLevel, nMaxLevel do
    local nTId = self.tbLevelInfo[i][1]
    local nCount = me.GetItemCountByTypeInBag(nTId)
    if nCount > 0 then
      table.insert(tbRet, {
        nTId,
        nCount,
        i
      })
    end
  end
  return tbRet
end

    RemoteServer.OnEquipQiHunRequest("ApplyTakeOff")
  end
  
  if not bRet then
    local bOk, nLeftTime = self:IsWaitTakeOffTime(me)
    if bOk then
      me.MsgBox(i18n.EasyGet("\233\156\128\232\166\129\231\173\137\229\190\133{0}\230\137\141\232\131\189\229\137\165\231\166\187", Lib:TimeDesc3(nLeftTime)))

      me.MsgBox(szErr, {
        {
          i18n.EasyGet("\231\148\179\232\175\183\229\137\165\231\166\187"),
          fnAgree
        },
        {
          i18n.szCancel
        }
      })

    return
  end
  RemoteServer.OnEquipQiHunRequest("TakeOff", nEquipPos)
end

function QiHun:ReforgeRsp(nintial_level, nOldLevel, nNewLevel)
  if self.nAutoCount and self.nAutoCount >= 0 then
    if self.nAutoDoneCount >= self.nAutoCount then
      me.CenterMsg(i18n.EasyGet("\229\153\168\233\173\130\230\181\139\232\175\149\229\174\140\230\136\144, \230\181\139\232\175\149\231\173\137\231\186\167{0}\239\188\140 \230\181\139\232\175\149\230\172\161\230\149\176{1}", self.nAutoLevel, self.nAutoDoneCount), true)
      self.nAutoCount = nil
      self.nAutoDoneCount = nil
      self.nAutoLevel = nil
      QiHun:ReforgeGetBackReq()
      return
    end
    if nNewLevel then
      if nNewLevel - nintial_level >= self.nGetBackLevel then
        self:ReforgeGetBackReq()
        self.nAutoDoneCount = self.nAutoDoneCount + 1
        self:ReforgeReq(self.nAutoLevel)
      else
        if self:GetCurReforgeTimes(me) >= self.nReforgeMaxCount then
          self:ReforgeGetBackReq()
        end
        self:ReforgeReq(nNewLevel)
      end
    else
      self.nAutoDoneCount = self.nAutoDoneCount + 1
      self:ReforgeReq(self.nAutoLevel)
    end
    return
  end
  self:OnCheckRedPoint()
  EventNotify.OnNotify(EventNotify.emNOTIFY_QIHUN_REFINE_RESULT, nintial_level, nOldLevel, nNewLevel)
end

function QiHun:CanShowQiHunInsertButtonInUIEquipTips(nItemId)
  if not self:IsSystemOpen() then
    return false
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return false
  end
  local nEquipPos = pItem.nEquipPos
  if nEquipPos < Item.EQUIPPOS_HELM or nEquipPos > Item.EQUIPPOS_AMULET then
    return false
  end
  if me.nLevel < self.nQiHunPutOnLevelLimit then
    return false
  end
  return true
end

function QiHun:OnCheckRedPoint()
  for nPos, nRedPoint in pairs(self.tbEquipPos2RedPoint) do
    if self:CheckRedPointByEquipPos(me, nPos) then
      RedPointMgr:LightRedPoint(nRedPoint)
    else
      RedPointMgr:FadeRedPoint(nRedPoint)
    end
  end
end

function QiHun:CheckRedPointByEquipPos(pPlayer, nPos)
  local pEquip = pPlayer.GetEquipByPos(nPos)
  if not pEquip then
    return false
  end
  if pPlayer.nLevel < self.nQiHunPutOnLevelLimit then
    return false
  end
  if not self:GetPunchState(pPlayer, nPos) then
    return false
  end
  local nPutOnLevel = self:GetPutOnLevel(pPlayer, nPos)
  local tbHadQiHun = self:GetQiHunItemNum()
  local nMaxPutOnLevel = QiHun:GetMaxPunOnLevel()
  for _, tbV in ipairs(tbHadQiHun) do
    if nMaxPutOnLevel >= tbV[3] and (not nPutOnLevel or nPutOnLevel < tbV[3]) then
      return true
    end
  end
  return false
end

function QiHun:GMStopAutoReforge()
  self.nAutoCount = self.nAutoDoneCount
end

function QiHun:GMAddQiHunRsp(bRet)
  if bRet then
    QiHun:ReforgeReq(self.nAutoLevel)
  else
    self.nGetBackLevel = nil
    self.nAutoCount = nil
    self.nAutoDoneCount = nil
    self.nAutoLevel = nil
  end
end

function QiHun:GMAutoReforge(nLevel, nCount, nGetBackLevel)
  if nCount <= 0 then
    return
  end
  self.nGetBackLevel = nGetBackLevel
  self.nAutoCount = nCount
  self.nAutoDoneCount = 1
  self.nAutoLevel = nLevel
  local nTemplateId = self.tbLevelInfo[nLevel][1]
  local nHadNum = me.GetItemCountByTypeInBag(nTemplateId)
  if nCount > nHadNum then
    RemoteServer.OnEquipQiHunRequest("GMAddQiHun", nLevel, nCount - nHadNum)
    return
  end
  QiHun:ReforgeReq(nLevel)
end
