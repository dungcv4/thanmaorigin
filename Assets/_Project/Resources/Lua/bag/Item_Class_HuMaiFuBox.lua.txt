local tbHuMaiFuBox = Item:GetClass("HuMaiFuBox")
tbHuMaiFuBox.IDENTIFY_TYPE = 1
tbHuMaiFuBox.BIND_TYPE = 2

function tbHuMaiFuBox:OnCreate()
end

function tbHuMaiFuBox:OnInit()
end

function tbHuMaiFuBox:OnUse(pItem)
  local nIdentifyType = KItem.GetItemExtParam(pItem.dwTemplateId, self.IDENTIFY_TYPE)
  local nBindType = KItem.GetItemExtParam(pItem.dwTemplateId, self.BIND_TYPE)
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag < 1 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161\239\188\140\232\175\183\229\133\136\230\184\133\231\144\134\232\131\140\229\140\133"))
    return 0
  end
  if JingMai:DoIdentifyHuMaiFu(me, nIdentifyType, nBindType, 3) then
    return 1
  end
  return 0
end

function tbHuMaiFuBox:OnUseMany(pItem, nCount)
  local nLeftBag = me.GetFreeBagCount()
  if nCount > nLeftBag then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161\239\188\140\232\175\183\229\133\136\230\184\133\231\144\134\232\131\140\229\140\133"))
    return 0
  end
  local nIdentifyType = KItem.GetItemExtParam(pItem.dwTemplateId, self.IDENTIFY_TYPE)
  local nBindType = KItem.GetItemExtParam(pItem.dwTemplateId, self.BIND_TYPE)
  local nUseNum = 0
  for i = 1, nCount do
    if JingMai:DoIdentifyHuMaiFu(me, nIdentifyType, nBindType, 3) then
      nUseNum = nUseNum + 1
    end
  end
  return nUseNum
end
