Require("CommonScript/Item/EquipForge/EquipForgeCfg.lua")
local EquipForge = Item.EquipForge
EquipForge.IS_SYS_OPEN = true
EquipForge.nAttrOtherNeedNum = 1
EquipForge.bAttrApplyLock = false
EquipForge.bAttrRemoveLock = false
EquipForge.EXT_PARAM_SEED_TO_EQUIP = 1
EquipForge.tbFiveElementsCoexist = {
  [Item.emSERIES_TYPE.SERIES_WOOD] = 1,
  [Item.emSERIES_TYPE.SERIES_FIRE] = 2,
  [Item.emSERIES_TYPE.SERIES_EARTH] = 3,
  [Item.emSERIES_TYPE.SERIES_METAL] = 4,
  [Item.emSERIES_TYPE.SERIES_WATER] = 5
}

function EquipForge:CanSell(pPlayer, nItemId)
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\231\179\187\231\187\159\230\156\170\229\188\128\230\148\190")
  end
  local pItem = pPlayer.GetItemInBag(nItemId)
  if not pItem then
    return false, i18n.EasyGet("\231\137\169\229\147\129\228\184\141\229\173\152\229\156\168")
  end
  local nEquipTemplateId = KItem.GetItemExtParam(pItem.dwTemplateId, self.EXT_PARAM_SEED_TO_EQUIP)
  local nPos = KItem.GetEquipPos(nEquipTemplateId)
  if not nPos then
    return false, i18n.EasyGet("\232\163\133\229\164\135\228\184\141\229\173\152\229\156\168")
  end
  local nQuality = self:GetQuality(pItem)
  local nPrice = self.SELL_PRICE[nPos] and self.SELL_PRICE[nPos][nQuality] and self.SELL_PRICE[nPos][nQuality][pItem.nLevel] or nil
  if not nPrice then
    return false, i18n.EasyGet("\229\135\186\229\148\174\228\187\183\230\160\188\233\148\153\232\175\175")
  end
  return true
end

function EquipForge:CanIdentify(pPlayer, nItemId, nSeries, nCount)
  nCount = nCount or 1
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\231\179\187\231\187\159\230\156\170\229\188\128\230\148\190")
  end
  if nSeries <= Item.emSERIES_TYPE.SERIES_NONE or nSeries >= Item.emSERIES_TYPE.SERIES_NUM then
    return false, i18n.EasyGet("\228\186\148\232\161\140\233\128\137\230\139\169\233\148\153\232\175\175")
  end
  local pItem = pPlayer.GetItemInBag(nItemId)
  if not pItem then
    return false, i18n.EasyGet("\231\137\169\229\147\129\228\184\141\229\173\152\229\156\168")
  end
  if nCount > pItem.nCount then
    return false, i18n.EasyGet("\231\137\169\229\147\129\230\149\176\233\135\143\228\184\141\232\182\179")
  end
  if pItem.nUseLevel > pPlayer.nTotalLevel then
    return false, i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179\239\188\140\228\184\141\232\131\189\233\137\180\229\174\154")
  end
  local nEquipTemplateId = KItem.GetItemExtParam(pItem.dwTemplateId, self.EXT_PARAM_SEED_TO_EQUIP)
  local nPos = KItem.GetEquipPos(nEquipTemplateId)
  if not nPos then
    return false, i18n.EasyGet("\232\163\133\229\164\135\228\184\141\229\173\152\229\156\168")
  end
  local nQuality = self:GetQuality(pItem)
  local nCost = self.IDENTIFY_COST[nPos] and self.IDENTIFY_COST[nPos][nQuality] and self.IDENTIFY_COST[nPos][nQuality][pItem.nLevel] or nil
  if not nCost then
    return false, i18n.EasyGet("\233\137\180\229\174\154\232\180\185\231\148\168\233\148\153\232\175\175")
  end
  if nCost > pPlayer.GetMoney("Coin") then
    return false, i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179")
  end
  if nCount > pPlayer.GetFreeBagCount() then
    return false, i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161")
  end
  return true
end

function EquipForge:CanTransfer(pPlayer, nSrcEquipId, nSrcAttrIdx, nDstEquipId, nDstAttrIdx)
  if not self.IS_SYS_OPEN then
    return false, i18n.EasyGet("\231\179\187\231\187\159\230\156\170\229\188\128\230\148\190")
  end
  if pPlayer.IsMaskPlayerAttr() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\228\184\141\229\143\175\230\180\151\231\187\131")
  end
  if nSrcEquipId == nDstEquipId then
    return false, i18n.EasyGet("\228\184\141\229\143\175\230\180\151\231\187\131")
  end
  local pSrcEquip = pPlayer.GetItemInAllSpace(nSrcEquipId)
  local pDstEquip = pPlayer.GetItemInAllSpace(nDstEquipId)
  if not pSrcEquip or not pDstEquip then
    return false, i18n.EasyGet("\232\163\133\229\164\135\228\184\141\229\173\152\229\156\168")
  end
  if pSrcEquip.nEquipPos ~= pDstEquip.nEquipPos then
    return false, i18n.EasyGet("\229\143\170\232\131\189\230\180\151\231\187\131\229\136\176\231\155\184\229\144\140\233\131\168\228\189\141\231\154\132\232\163\133\229\164\135\228\184\138")
  end
  if pSrcEquip.nEquipPos == pSrcEquip.nPos or pDstEquip.nEquipPos ~= pDstEquip.nPos then
    return false, i18n.EasyGet("\229\143\170\232\131\189\229\176\134\232\131\140\229\140\133\231\154\132\232\163\133\229\164\135\230\180\151\231\187\131\229\136\176\231\155\184\229\144\140\233\131\168\228\189\141\231\154\132\232\163\133\229\164\135\228\184\138")
  end
  if pDstEquip.nLevel < pSrcEquip.nLevel then
    return false, i18n.EasyGet("\229\189\147\229\137\141\230\137\128\233\128\137\232\163\133\229\164\135\231\173\137\233\152\182\230\155\180\233\171\152\239\188\140\230\151\160\230\179\149\230\180\151\231\187\131\239\188\140\229\187\186\232\174\174\231\169\191\230\136\180")
  end
  if pDstEquip.nLevel == pSrcEquip.nLevel and pDstEquip.nQuality < pSrcEquip.nQuality then
    return false, i18n.EasyGet("\232\175\165\232\163\133\229\164\135\229\177\158\230\128\167\230\155\180\233\171\152\239\188\140\230\151\160\230\179\149\230\180\151\231\187\131\239\188\140\229\187\186\232\174\174\231\169\191\230\136\180")
  end
  if pDstEquip.nLevel > pSrcEquip.nLevel and pDstEquip.nOrgValue < pSrcEquip.nOrgValue then
    return false, i18n.EasyGet("\232\175\165\232\163\133\229\164\135\229\177\158\230\128\167\230\155\180\233\171\152\239\188\140\230\151\160\230\179\149\230\180\151\231\187\131\239\188\140\229\187\186\232\174\174\231\169\191\230\136\180")
  end
  local tbSrcAttrList = self:GetAttr(pSrcEquip)
  local tbDstAttrList = self:GetAttr(pDstEquip)
  local tbSrcAttr = tbSrcAttrList[nSrcAttrIdx]
  local tbDstAttr = tbDstAttrList[nDstAttrIdx]
  if not tbSrcAttr or not tbDstAttr then
    return false, i18n.EasyGet("\229\177\158\230\128\167\230\160\143\228\189\141\233\148\153\232\175\175")
  end
  local nSrcGroupId, nSrcLevel = table.unpack(tbSrcAttr)
  local nDstGroupId, nDstLevel = table.unpack(tbDstAttr)
  if nSrcGroupId == nDstGroupId and nSrcLevel <= nDstLevel then
    return false, i18n.EasyGet("\229\177\158\230\128\167\231\173\137\231\186\167\233\153\144\229\136\182")
  end
  if nSrcGroupId <= 0 then
    return false
  end
  for nIdx, tbAttr in ipairs(tbDstAttrList) do
    if nIdx ~= nDstAttrIdx and tbAttr[1] == nSrcGroupId then
      return false, i18n.EasyGet("\229\177\158\230\128\167\228\184\141\229\143\175\233\135\141\229\164\141")
    end
  end
  local szColor = self:ColorAttr(pSrcEquip.nEquipPos, pSrcEquip.nLevel, nSrcLevel)
  local nCost = self.TRANSFER_COST[szColor] and self.TRANSFER_COST[szColor][pSrcEquip.nEquipPos] and self.TRANSFER_COST[szColor][pSrcEquip.nEquipPos][pSrcEquip.nLevel] or nil
  if not nCost then
    return false, i18n.EasyGet("\230\180\151\231\187\131\232\180\185\231\148\168\233\148\153\232\175\175")
  end
  if nCost > pPlayer.GetMoney("Coin") then
    return false, i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179")
  end
  return true
end

function EquipForge:ApplyAttrib(pPlayer, pEquip)
  if pEquip.nEquipPos < Item.EQUIPPOS_HELM or pEquip.nEquipPos > Item.EQUIPPOS_AMULET then
    return
  end
  if self.bAttrApplyLock then
    return
  end
  local tbSeriesList = self:StatSeries(pPlayer)
  local nSeries = pEquip.GetSeries()
  self:UpdatePerEquipAttr(pPlayer.GetNpc(), self:GetAttr(pEquip), tbSeriesList, nSeries, pPlayer.nFaction, true, pPlayer.dwID)
end

function EquipForge:RemoveAttrib(pPlayer, pEquip)
  if pEquip.nEquipPos < Item.EQUIPPOS_HELM or pEquip.nEquipPos > Item.EQUIPPOS_AMULET then
    return
  end
  if self.bAttrRemoveLock then
    return
  end
  local tbSeriesList = self:StatSeries(pPlayer)
  local nSeries = pEquip.GetSeries()
  self:UpdatePerEquipAttr(pPlayer.GetNpc(), self:GetAttr(pEquip), tbSeriesList, nSeries, pPlayer.nFaction, false, pPlayer.dwID)
end

function EquipForge:GetQuality(pItem)
  return self.EQUIP_TO_FORGE_QUALITY[pItem.nQuality]
end

function EquipForge:GetAttr(pEquip)
  local tbAttrList = Item:GetClass(Item.MAIN_EQUIP_CLASS_NAME):GetSeriesActiveAttr(pEquip)
  local tbUnpackAttrList = {}
  for nIndex, nAttr in pairs(tbAttrList) do
    if 0 ~= nAttr then
      tbUnpackAttrList[nIndex] = {
        self:UnpackAttr(nAttr)
      }
    else
      tbUnpackAttrList[nIndex] = {0, 0}
    end
  end
  return tbUnpackAttrList
end

function EquipForge:PackAttr(nGroupId, nLevel)
  return (nGroupId << 16) + nLevel
end

function EquipForge:UnpackAttr(nAttr)
  local nGroupId = nAttr >> 16
  local nLevel = nAttr & 255
  return nGroupId, nLevel
end

function EquipForge:ColorAttr(nEquipPos, nEquipLv, nAttrLv)
  local szColorName, nColorVal = table.unpack(self.IDENTIFY_ATTR_COLOR[1])
  for nIdx, tbInfo in ipairs(self.IDENTIFY_ATTR_COLOR) do
    local nColorLv = self.IDENTIFY_ATTR_LV[tbInfo[1]][nEquipPos][nEquipLv]
    if nAttrLv >= nColorLv then
      szColorName, nColorVal = table.unpack(tbInfo)
    else
      break
    end
  end
  return szColorName, nColorVal
end

function EquipForge:StatSeries(pPlayer)
  local tbSeriesList = {}
  for nSeries = Item.emSERIES_TYPE.SERIES_NONE + 1, Item.emSERIES_TYPE.SERIES_NUM - 1 do
    tbSeriesList[nSeries] = 0
  end
  for nPos = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
    local pEquip = pPlayer.GetEquipByPos(nPos)
    if pEquip then
      local nSeries = pEquip.GetSeries()
      tbSeriesList[nSeries] = tbSeriesList[nSeries] + 1
    end
  end
  return tbSeriesList
end

function EquipForge:ApplyAllItemAttrib(pPlayer, bStart)
  self.bAttrApplyLock = bStart
  if not bStart then
    local tbEquipList = {}
    for nPos = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
      local pEquip = pPlayer.GetEquipByPos(nPos)
      if pEquip then
        tbEquipList[nPos] = pEquip
      end
    end
    local tbSeriesList = self:StatSeries(pPlayer)
    self:UpdateAllEquipAttr(pPlayer.GetNpc(), tbSeriesList, tbEquipList, true, pPlayer.nFaction, pPlayer.dwID)
  end
end

function EquipForge:RemoveAllItemAttrib(pPlayer, bStart)
  self.bAttrRemoveLock = bStart
  if not bStart then
    local tbEquipList = {}
    for nPos = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
      local pEquip = pPlayer.GetEquipByPos(nPos)
      if pEquip then
        tbEquipList[nPos] = pEquip
      end
    end
    local tbSeriesList = self:StatSeries(pPlayer)
    self:UpdateAllEquipAttr(pPlayer.GetNpc(), tbSeriesList, tbEquipList, false, pPlayer.nFaction, pPlayer.dwID)
  end
end

function EquipForge:UpdateAllEquipAttr(pNpc, tbSeriesList, tbEquipList, bIsApply, nFaction, nPlayerId)
  for i = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
    local pEquip = tbEquipList[i]
    if pEquip then
      self:UpdatePerEquipAttr(pNpc, self:GetAttr(pEquip), tbSeriesList, pEquip.GetSeries(), nFaction, bIsApply, nPlayerId)
    end
  end
end

function EquipForge:UpdatePerEquipAttr(pNpc, tbAttrList, tbSeriesList, nEquipSeries, nFaction, bIsApply, nPlayerId)
  for k, v in ipairs(tbAttrList) do
    local tbSetting = FightSkill:GetExternAttrib(table.unpack(v))
    if tbSetting then
      for _, tbInfo in ipairs(tbSetting) do
        local szMagic = tbInfo.szAttribName
        szMagic = self.ATTR_CONVERT[szMagic] and self.ATTR_CONVERT[szMagic][nEquipSeries] or szMagic
        if bIsApply then
          pNpc.ApplyMagicAttrib(szMagic, table.unpack(tbInfo.tbValue))
          if nPlayerId then
            AttributeHelp:LogAttrApply(nPlayerId, "EquipForge", szMagic, table.unpack(tbInfo.tbValue))
          end
        else
          pNpc.RemoveMagicAttrib(szMagic, table.unpack(tbInfo.tbValue))
          if nPlayerId then
            AttributeHelp:LogAttrRemove(nPlayerId, "EquipForge", szMagic, table.unpack(tbInfo.tbValue))
          end
        end
      end
    end
  end
end
