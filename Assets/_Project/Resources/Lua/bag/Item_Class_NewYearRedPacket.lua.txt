local tbItem = Item:GetClass("NewYearRedPacket")

function tbItem:OnUse(pItem)
  return self:OnUseMany(pItem, 1)
end

function tbItem:OnUseMany(pItem, nCount)
  local pInst = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.PasswordRedPacket)
  if not pInst then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\167\139"))
    return
  end
  nCount = pInst:OnUseRedPacket(me, nCount)
  return nCount
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "NewYearRedPacket")
end
