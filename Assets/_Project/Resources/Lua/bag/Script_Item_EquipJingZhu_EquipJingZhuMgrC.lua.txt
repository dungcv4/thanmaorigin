Require("CommonScript/Item/EquipJingZhu/EquipJingZhuCfg.lua")
Require("Script/Ui/Common/RedPointMgr.lua")
local EquipJingZhu = Item.EquipJingZhu
EquipJingZhu.tbEquipPos2RedPoint = {
  [Item.EQUIPPOS_HELM] = RedPointMgr.EVENT.HelmJingZhu,
  [Item.EQUIPPOS_ARMOR] = RedPointMgr.EVENT.ArmorJingZhu,
  [Item.EQUIPPOS_BELT] = RedPointMgr.EVENT.BeltJingZhu,
  [Item.EQUIPPOS_CUFF] = RedPointMgr.EVENT.CuffJingZhu,
  [Item.EQUIPPOS_BOOTS] = RedPointMgr.EVENT.BootsJingZhu,
  [Item.EQUIPPOS_WEAPON] = RedPointMgr.EVENT.WeaponJingZhu,
  [Item.EQUIPPOS_NECKLACE] = RedPointMgr.EVENT.NecklaceJingZhu,
  [Item.EQUIPPOS_RING] = RedPointMgr.EVENT.RingJingZhu,
  [Item.EQUIPPOS_PENDANT] = RedPointMgr.EVENT.PendantJingZhu,
  [Item.EQUIPPOS_AMULET] = RedPointMgr.EVENT.AmuletJingZhu
}

function EquipJingZhu:JingZhuReq(nEquipPos)
  RemoteServer.RequestEquipJingZhu("JingZhuReq", nEquipPos)
end

function EquipJingZhu:JingZhuBreakReq(nEquipPos)
  RemoteServer.RequestEquipJingZhu("JingZhuBreakReq", nEquipPos)
end

function EquipJingZhu:OnS2CJingZhuResult(bSuccess, nPos, nRate, nSynthesisCostCoin)
  local szParam = ""
  local nNowLevel = self:GetJingZhuLevel(me, nPos)
  if bSuccess then
    self:RefreshAttrib(me, nPos, nNowLevel - 1, nNowLevel)
    szParam = "success"
  else
    szParam = "fail"
  end
  if Ui:WindowVisible("UIEquipment") then
    Ui:GetWindow("UIEquipment"):NotifyEquipJingZhu(szParam, nPos)
  end
  self:OnCheckRedPoint()
  local tbWnd = Ui:GetWindow("UIAutoJingzhu")
  if nil ~= tbWnd then
    tbWnd:OnSyncJingZhuResult(bSuccess, nPos, nRate, nSynthesisCostCoin, nNowLevel)
  end
end

function EquipJingZhu:OnS2CJingZhuReSet(nPos, nOldLevel, nNewLevel)
  self:RefreshAttrib(me, nPos, nOldLevel, nNewLevel)
end

function EquipJingZhu:OnSyncEquipJingZhuBreakLevel(nPos, nSynthesisCostCoin)
  Ui:OpenUpgradeEffect(i18n.EasyGet("\231\170\129\231\160\180\230\136\144\229\138\159"), {x = 100})
  if Ui:WindowVisible("UIEquipment") then
    Ui:GetWindow("UIEquipment"):NotifyEquipJingZhu(nil, nPos)
  end
  self:OnCheckRedPoint()
  local tbWnd = Ui:GetWindow("UIAutoJingzhu")
  if nil ~= tbWnd then
    local nNowLevel = EquipJingZhu:GetJingZhuLevel(me, nPos)
    tbWnd:OnSyncBreakResult(nPos, nSynthesisCostCoin, nNowLevel)
  end
end

function EquipJingZhu:OnSyncEquipJingZhuGradeScore(nOldGradeLevel, nJingZhuGradeScore)
  self:RefreshGradeScoreAttrib(me, nOldGradeLevel, nJingZhuGradeScore)
  local tbWnd = Ui:GetWindow("UIBag")
  if tbWnd then
    tbWnd:UpdateJZGradeScore(nJingZhuGradeScore)
  end
end

function EquipJingZhu:CheckRedPointByEquipPos(pPlayer, nPos)
  if self:CanJingZhu(pPlayer, nPos) then
    return true
  end
  if self:CanJZBreak(pPlayer, nPos) then
    return true
  end
  return false
end

function EquipJingZhu:OnCheckRedPoint()
  for nPos, nRedPoint in pairs(self.tbEquipPos2RedPoint) do
    if self:CheckRedPointByEquipPos(me, nPos) then
      RedPointMgr:LightRedPoint(nRedPoint)
    else
      RedPointMgr:FadeRedPoint(nRedPoint)
    end
  end
end

function EquipJingZhu:CanShowJingZhuButtonInUIEquipTips(nItemId)
  if not Item.EquipJingZhu:CheckIsOpen() then
    return false
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return false
  end
  local nEquipPos = pItem.nEquipPos
  if nEquipPos < Item.EQUIPPOS_HELM or nEquipPos > Item.EQUIPPOS_AMULET then
    return false
  end
  return true
end

function EquipJingZhu:OnSyncItemLimitTime()
  if not PlayerEvent.bLogin then
    return
  end
  self:OnCheckRedPoint()
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, EquipJingZhu)
EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, EquipJingZhu.OnCheckRedPoint, EquipJingZhu)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM_TIME_LIMIT, EquipJingZhu)
EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM_TIME_LIMIT, EquipJingZhu.OnSyncItemLimitTime, EquipJingZhu)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_MONEY, EquipJingZhu)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_MONEY, EquipJingZhu.OnCheckRedPoint, EquipJingZhu)
