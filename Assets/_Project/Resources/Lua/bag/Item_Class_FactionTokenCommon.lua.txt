Require("CommonScript/Item/SecretBook/SecretBookCfg.lua")
local tbItem = Item:GetClass("FactionToken")

function tbItem:OnUseMany(pItem, nCount)
  if MODULE_GAMESERVER then
    self:AddXiuWeiValue(nCount)
  end
  return nCount
end

function tbItem:OnAutoUse(pPlayer, nItemTemplateId, nCount)
  if MODULE_GAMESERVER then
    self:AddXiuWeiValue(nCount)
  end
  return nCount
end

function tbItem:AddXiuWeiValue(nUseCount)
  local nAddValue = nUseCount * XiuWei.Cfg.nFactionTokenAddValue
  XiuWei:AddValue(me, nAddValue)
  me.CenterMsg(i18n.Get(XiuWei.tbText.szUseTokenSuccessTips, nAddValue))
end
