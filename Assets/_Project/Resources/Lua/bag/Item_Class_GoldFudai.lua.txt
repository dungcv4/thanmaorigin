local tbItem = Item:GetClass("GoldFuDai")
tbItem.nMonthCardSeniorAwardCount = 50
tbItem.nNormalSeniorAwardCount = 30
tbItem.nSeniorAwardId = 1058
tbItem.nSeniorAwardKey = 1
tbItem.nSeniorRoomAwardId = 1060
tbItem.nSeniorRoomAwardKey = 1
tbItem.nNormalAwardId = 1059
tbItem.nNormalAwardKey = 1

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  local tbBaseProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbBaseProp then
    return
  end
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  else
    return nil
  end
end

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  local nLastAwardCount = self:GetLastAwardCount(me)
  local nSeniorAwardCount = math.min(nCount, nLastAwardCount)
  local nNormalAwardCount = nCount - nSeniorAwardCount
  if nSeniorAwardCount > 0 then
    FieldPractice:AddFudaiSeniorAwardCount(me, nSeniorAwardCount)
    for i = 1, nSeniorAwardCount do
      local bMonthCardState = WeekMonthCard:CheckMonthCardState(me)
      if not bMonthCardState then
        Award:GiveAward(me.dwID, self.nSeniorRoomAwardId, self.nSeniorRoomAwardKey, {
          nLogReason = Env.LogWay_GoldFudai
        })
      else
        Award:GiveAward(me.dwID, self.nSeniorAwardId, self.nSeniorAwardKey, {
          nLogReason = Env.LogWay_GoldFudai
        })
      end
    end
  end
  if nNormalAwardCount > 0 then
    for i = 1, nNormalAwardCount do
      Award:GiveAward(me.dwID, self.nNormalAwardId, self.nNormalAwardKey, {
        nLogReason = Env.LogWay_GoldFudai
      })
    end
  end
  return nCount
end

function tbItem:CheckMonthCard(pPlayer)
  return WeekMonthCard:CheckMonthCardState(pPlayer)
end

function tbItem:CheckWorkRoom(pPlayer)
  return Studio:CheckStudioSetting(pPlayer.dwID, Studio.ModuleCfg.Fudai)
end

function tbItem:GetLastAwardCount(pPlayer)
  local nCurDegree = FieldPractice:GetFudaiSeniorAwardCount(pPlayer)
  local bMonthCard = self:CheckMonthCard(pPlayer)
  local nMaxSeniorAwardCount = bMonthCard and self.nMonthCardSeniorAwardCount or self.nNormalSeniorAwardCount
  local nLastAwardCount = nMaxSeniorAwardCount - nCurDegree
  if nLastAwardCount < 0 then
    nLastAwardCount = 0
  end
  return nLastAwardCount, bMonthCard, self.nMonthCardSeniorAwardCount - self.nNormalSeniorAwardCount
end

function tbItem:GetClientDefaultUseCount(nItemId)
  local pItem = me.GetItemInBag(nItemId)
  local nLastAwardCount = self:GetLastAwardCount(me)
  local nResult = 0
  if nLastAwardCount > 0 then
    nResult = math.min(pItem.nCount, nLastAwardCount)
  else
    nResult = math.min(pItem.nCount, Item.UseManyLimit.GoldFuDai)
  end
  return nResult
end

  if it.nCount > 1 then

end

function tbItem:GetTip(pItem)
  local pPlayer = me
  local nUseCountToday = FieldPractice:GetFudaiSeniorAwardCount(pPlayer)
  return i18n.EasyGet("\228\187\138\230\151\165\229\183\178\228\189\191\231\148\168{0}\230\172\161", nUseCountToday), nil, false
end
