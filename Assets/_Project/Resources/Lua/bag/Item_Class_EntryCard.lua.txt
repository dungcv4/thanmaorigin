local tbItem = Item:GetClass("EntryCard")

function tbItem:OnClientUse(it)
  local szName = KItem.GetItemShowInfo(it.dwTemplateId, me.nFaction, me.nFactionSect, me.nSex)
  local tbInfo = Item:GetClass("CountAddProp"):GetAddDegreeInfoByTID(it.dwTemplateId)
  if not tbInfo then
    me.CenterMsg(i18n.EasyGet("\231\188\186\229\176\145{0}\233\133\141\231\189\174", szName))
    return 0
  end
  local nCurNum = DegreeCtrl:GetDegree(me, tbInfo.AddDegree)
  local nTotalNum = DegreeCtrl:GetMaxDegree(tbInfo.AddDegree, me)
  if nCurNum >= nTotalNum then
    me.CenterMsg(i18n.EasyGet("\229\183\178\232\190\190\229\143\130\232\181\155\230\172\161\230\149\176\228\184\138\233\153\144\239\188\140\232\175\183\229\133\136\230\182\136\232\128\151\229\137\169\228\189\153\230\172\161\230\149\176\229\144\142\229\134\141\228\189\191\231\148\168"))
    return 0
  end
  if DegreeCtrl:GetDegree(me, tbInfo.CostDegree) > 0 then
    return 1
  end
  me.CenterMsg(i18n.EasyGet("\230\156\172\230\151\165\228\189\191\231\148\168{0}\230\172\161\230\149\176\229\183\178\232\190\190\228\184\138\233\153\144\239\188\140\232\175\183\230\152\142\229\164\169\229\134\141\232\175\149", szName))
  return 0
end

function tbItem:OnUse(it)
  local szName = KItem.GetItemShowInfo(it.dwTemplateId, me.nFaction, me.nFactionSect, me.nSex)
  local tbInfo = Item:GetClass("CountAddProp"):GetAddDegreeInfoByTID(it.dwTemplateId)
  if not tbInfo then
    me.CenterMsg(i18n.EasyGet("\231\188\186\229\176\145{0}\233\133\141\231\189\174", szName))
    return 0
  end
  if DegreeCtrl:ReduceDegree(me, tbInfo.CostDegree, 1) then
    DegreeCtrl:AddDegree(me, tbInfo.AddDegree, 1)
    me.CallClientScript("me.BuyTimesSuccess", i18n.EasyGet("{0}\228\189\191\231\148\168\230\136\144\229\138\159\239\188\140\229\162\158\229\138\1601\230\172\161\229\143\130\228\184\142\230\172\161\230\149\176", szName))
    return 1
  else
    return 0
  end
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "EntryCard")
end
