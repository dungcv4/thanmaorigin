local tbSanShengShi = Item:GetClass("SanShengShi")

function tbSanShengShi:UseItem()
  local tbInfo = SanShengShi.NpcPos.SanShengShiNpc
  local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(tbInfo.nNpcID, tbInfo.nTID, true)
  
  local function fnCallback()
    local function __timer_Callback()
      local nNpcId = KNpc.GetNpcIdByTemplateId(tbInfo.nNpcID)
      
      if nNpcId then
        Operation:SimpleTap(nNpcId)
      end
    end
    
    Timer:Register(5, __timer_Callback)
  end
  
  Ui:CloseWindow("UIItemTips")
  Ui:CloseWindow("UIBag")
  AutoPath:GotoAndCallEx(tbInfo.nTID, nPosX, nPosY, fnCallback, nWalkNearLength, nil, tbInfo.nNpcID)
  return 0
end

function tbSanShengShi:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    tbSanShengShi:UseItem()
  end
  
  return tbSetting
end
