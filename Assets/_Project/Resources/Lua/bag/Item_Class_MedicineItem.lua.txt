function tbItem:GetMaxMedicineCount()
  return self.nMaxMedicineCount
end

function tbItem:GetMedicineCountAtMaxLevel(pPlayer)
  local nCount = 0
  local nMaxLevel = 0
  local tbList = pPlayer.GetItemListInMedicineBag()
  for _, pItem in pairs(tbList) do
    if nMaxLevel < pItem.nLevel then
      nMaxLevel = pItem.nLevel
      nCount = pItem.nCount
    elseif pItem.nLevel == nMaxLevel then
      nCount = nCount + pItem.nCount
    end
  end
  return nCount
end

function tbItem:GetMedicineItemIDAtMaxLevel(pPlayer)
  local nItemID
  local nMaxLevel = 0
  local tbList = pPlayer.GetItemListInMedicineBag()
  for _, pItem in pairs(tbList) do
    if nMaxLevel < pItem.nLevel then
      nMaxLevel = pItem.nLevel
      nItemID = pItem.dwId
    end
  end
  return nItemID
end

function tbItem:ClearUserValueOfZoneCostMoneyForMedicine(pPlayer)
  MedicineMgr:ClearZoneCostMoney(pPlayer.dwID)
  Log("---->", self.szKeyOfZoneCostMoneyForMedicine, MedicineMgr:GetZoneCostMoney(pPlayer.dwID))
end

function tbItem:AddUserValueOfZoneCostMoneyForMedicine(pPlayer, nAdd)
  if SERVER_RUNNING_MODE == SERVER_MODE.ZONE_SERVER then
    if Map:IsDisableSyncZoneMedicine(pPlayer.nMapTemplateId) then
      return
    end
    MedicineMgr:SetPlayerCostMoneyData(pPlayer, MedicineMgr:GetPlayerCostMoneyData(pPlayer) + nAdd)
    Log("---->", self.szKeyOfZoneCostMoneyForMedicine, MedicineMgr:GetPlayerCostMoneyData(pPlayer))
  end
end

function tbItem:GetUserValueOfZoneCostMoneyForMedicine(pPlayer)
  local nCostMoney = MedicineMgr:GetZoneCostMoney(pPlayer.dwID)
  if nCostMoney > 0 then
    return nCostMoney
  end
end

function tbItem:ClearUserValueOfZoneMedicineCount(pPlayer)
  MedicineMgr:ClearZoneMedicineCount(pPlayer.dwID)
  Log("---->", self.szKeyOfZoneMedicineCount, MedicineMgr:GetZoneMedicineCount(pPlayer.dwID))
end

function tbItem:UpdateUserValueOfZoneMedicineCount(pPlayer)
  if SERVER_RUNNING_MODE == SERVER_MODE.ZONE_SERVER then
    if Map:IsDisableSyncZoneMedicine(pPlayer.nMapTemplateId) then
      return
    end
    local nCount = self:GetMedicineCountAtMaxLevel(pPlayer)
    MedicineMgr:SetPlayerMedicineData(pPlayer, nCount + self.nZoneMedicineCountChangeFlag)
    Log("---->", self.szKeyOfZoneMedicineCount, MedicineMgr:GetPlayerMedicineData(pPlayer))
  end
end

function tbItem:GetUserValueOfZoneMedicineCount(pPlayer)
  local nCount = MedicineMgr:GetZoneMedicineCount(pPlayer.dwID)
  if nCount >= self.nZoneMedicineCountChangeFlag then
    return nCount - self.nZoneMedicineCountChangeFlag
  end
end

function tbItem:SyncZoneMedicineCount(pPlayer)
  local nZoneCount = self:GetUserValueOfZoneMedicineCount(pPlayer)
  if nZoneCount then
    nZoneCount = math.min(nZoneCount, self.nMaxMedicineCount)
    local nLocalCount = self:GetMedicineCountAtMaxLevel(pPlayer)
    local nDelta = nZoneCount - nLocalCount
    Log("----> SyncZoneMedicineCount Beg", nZoneCount, nLocalCount)
    if nDelta > 0 then
      local tbItemExtParams = {}
      local tbAward = {
        {
          "Item",
          self.nDefaultMedicineTId,
          nDelta,
          tbItemExtParams
        }
      }
      pPlayer.SendAward(tbAward, false, false, Env.LogWay_SyncZoneConsume, 0, "Medicine")
    elseif nDelta < 0 then
      local nItemID = self:GetMedicineItemIDAtMaxLevel(pPlayer)
      if nItemID then
        local pItem = pPlayer.GetItemInAllSpace(nItemID)
        if pItem then
          pPlayer.ConsumeItem(pItem, -nDelta, Env.LogWay_SyncZoneConsume)
        end
      end
    end
    Log("----> SyncZoneMedicineCount End", self:GetMedicineCountAtMaxLevel(pPlayer))
    self:ClearUserValueOfZoneMedicineCount(pPlayer)
  end
end

function tbItem:SyncZoneCostMoneyForMedicine(pPlayer)
  local nCostMoney = self:GetUserValueOfZoneCostMoneyForMedicine(pPlayer)
  if nCostMoney and nCostMoney > 0 then
    if nCostMoney > 0 then
      Log("----> SyncZoneCostMoneyForMedicine Beg", pPlayer.GetMoney(self.szMoneyType), nCostMoney)
      pPlayer.CostMoney(self.szMoneyType, nCostMoney, Env.LogWay_SyncZoneConsume, self.nDefaultMedicineTId)
      Log("----> SyncZoneCostMoneyForMedicine End", pPlayer.GetMoney(self.szMoneyType))
    end
    self:ClearUserValueOfZoneCostMoneyForMedicine(pPlayer)
  end
end

function tbItem:CheckUsable(it)
  local tbMagicAttrSetting = self.tbMagicAttrSetting[it.dwTemplateId]
  if not tbMagicAttrSetting then
    return
  end
  if MODULE_GAMESERVER then
    local nTotalCDTime = tbMagicAttrSetting.WaitTime or 0
    if nTotalCDTime > 0 then
      local pNpc = me.GetNpc()
      local nLastUseTime = pNpc.GetLastTimeUsedMedicine(it.dwTemplateId)
      local nTimeOfCDClose = nLastUseTime + nTotalCDTime
      if nTimeOfCDClose > GetTime() then
        return 0
      end
    end
  end
  return 1
end

function tbItem:CheckUseMedicine(dwTemplateId)
  local pMeNpc = me.GetNpc()
  if not pMeNpc then
    return false
  end
  if pMeNpc.HaveRefFlag(Npc.RefFlagType.npc_forbid_useitem) and me.GetDoing() ~= Npc.Doing.float then
    return false
  end
  if not GoodwillValue:CheckValueCanUseMedicine(me) then
    return false
  end
  self.nChangeValue = -1
  local tbChangeValue = tbItem.tbMedicineChangeValue[dwTemplateId]
  if tbChangeValue then
    local myLevel = me.nTotalLevel
    for i = 1, #tbChangeValue do
      if myLevel < tbChangeValue[i][1] then
        self.nChangeValue = tbChangeValue[i][2]
        break
      end
    end
  end
  return true
end

function tbItem:GetCurMedicineTId(pPlayer)
  assert(pPlayer)
  local tid = self.nDefaultMedicineTId
  local tbChangeValue = self.tbMedicineChangeValue[self.nDefaultMedicineTId]
  if tbChangeValue then
    local myLevel = pPlayer.nTotalLevel
    for i = 1, #tbChangeValue do
      if myLevel < tbChangeValue[i][1] then
        tid = tbChangeValue[i][3]
        break
      end
    end
  end
  return tid
end

function tbItem:OnUse(it)
  if not it.dwTemplateId or it.dwTemplateId <= 0 then
    return
  end
  return self:UseItembyTemplateId(it.dwTemplateId)
end

function tbItem:UseItembyTemplateId(dwTemplateId)
  if not dwTemplateId or dwTemplateId <= 0 then
    return
  end
  local tbMagicAttrSetting = self.tbMagicAttrSetting[dwTemplateId]
  if not tbMagicAttrSetting then
    return
  end
  local bCanUse = self:CheckUseMedicine(dwTemplateId)
  if not bCanUse then
    return
  end
  self.nChangeValue = self.nChangeValue or -1
  local bAddFlag = false
  local nMATime = 0
  for i = 1, 10 do
    local szAttrib = tbMagicAttrSetting[string.format("Attrib%dType", i)]
    local nMA1 = tbMagicAttrSetting[string.format("Attrib%dValue1", i)]
    if self.nChangeValue >= 0 then
      nMA1 = self.nChangeValue
    end
    local nMA2 = tbMagicAttrSetting[string.format("Attrib%dValue2", i)]
    local nMA3 = tbMagicAttrSetting[string.format("Attrib%dValue3", i)] or 0
    if not szAttrib or "" == szAttrib then
      break
    end
    me.ApplyMagicAttrib(szAttrib, nMA1, nMA2, nMA3)
    if "lifepotion_v" == szAttrib or "manapotion_v" == szAttrib then
      bAddFlag = true
      nMATime = nMA2
    end
  end
  if bAddFlag and MODULE_GAMESERVER then
    me.AddSkillState(tbItem.nLiftSkillState, 1, 0, nMATime)
  end
  local nSkillId = KItem.GetItemExtParam(dwTemplateId, self.ExtParam1) or 0
  if nSkillId > 0 then
    local nSkillLv = KItem.GetItemExtParam(dwTemplateId, self.ExtParam2) or 0
    if nSkillLv <= 0 then
      nSkillLv = 1
    end
    local nSkillType = KItem.GetItemExtParam(dwTemplateId, self.ExtParam3) or 0
    local nSkillTime = KItem.GetItemExtParam(dwTemplateId, self.ExtParam4) or 0
    local nSkillRemove = KItem.GetItemExtParam(dwTemplateId, self.ExtParam5) or 0
    Log("MedicineItem::OnUse add skill state 2", nSkillId, nSkillLv, nSkillType, nSkillTime, nSkillRemove)
    me.AddSkillState(nSkillId, nSkillLv, nSkillType, nSkillTime, 1)
  end
  local nBuffSkillId = KItem.GetItemExtParam(dwTemplateId, tbItem.ExtParam6) or 0
  local nBuffLevel = KItem.GetItemExtParam(dwTemplateId, tbItem.ExtParam7) or 0
  if nBuffSkillId < 0 then
    local pNpc = me.GetNpc()
    local _, nPosX, nPosY = pNpc.GetWorldPos()
    if nBuffLevel <= 0 then
      nBuffLevel = 1
    end
    pNpc.CastSkill(nBuffSkillId, nBuffLevel, nPosX, nPosY)
  end
  local nTotalCDTime = tbMagicAttrSetting.WaitTime or 0
  if nTotalCDTime > 0 then
    if MODULE_GAMESERVER then
      local pNpc = me.GetNpc()
      pNpc.SetLastTimeUsedMedicine(dwTemplateId)
      me.CallClientScript("MedicineMgr:NotifyMedicineCD", dwTemplateId, nTotalCDTime, nTotalCDTime)
    end
    if MODULE_GAMECLIENT then
      MedicineMgr:NotifyMedicineCD(dwTemplateId, nTotalCDTime, nTotalCDTime)
    end
  end
  return 1, true
end

function tbItem:NotifyMedicineCD(dwTemplateId, pPlayer, nTotalCDTime)
  if nil == pPlayer then
    pPlayer = me
  end
  local pNpc = pPlayer.GetNpc()
  local nLastUseTime = pNpc.GetLastTimeUsedMedicine(dwTemplateId)
  local nTimeOfCDClose = nLastUseTime + nTotalCDTime
  if nTimeOfCDClose > GetTime() then
    local nRemainTime = nTimeOfCDClose - GetTime()
    pPlayer.CallClientScript("MedicineMgr:NotifyMedicineCD", dwTemplateId, nRemainTime, nTotalCDTime)
    return true
  else
    return false
  end
end
