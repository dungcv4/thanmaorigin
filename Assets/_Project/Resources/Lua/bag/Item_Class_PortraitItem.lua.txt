local tbPortraitItem = Item:GetClass("PortraitItem")

function tbPortraitItem:OnUse(pItem)
  local tbExtParam = {
    nBoyPortrait = 1,
    nGirlPortrait = 2,
    nTime = 3
  }
  local nPortrait = KItem.GetItemExtParam(pItem.dwTemplateId, me.nSex + 1)
  local nTime = KItem.GetItemExtParam(pItem.dwTemplateId, tbExtParam.nTime)
  local bCanAdd, szErr = PlayerPortrait:CheckCanAdd(me, nPortrait)
  if not bCanAdd then
    me.CenterMsg(szErr)
    return
  end
  PlayerPortrait:AddPortrait(me, nPortrait, nTime)
  return 1
end
