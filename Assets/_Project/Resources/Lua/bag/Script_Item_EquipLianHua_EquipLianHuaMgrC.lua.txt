function EquipLianHua:OnLoginC()
  self:LoadFightPowerSetting()
  self.szLHGradeScoreTimeFrame = ""
  for _, tbInfo in ipairs(self.tbGradeScoreInfo) do
    if 1 == GetTimeFrameState(tbInfo.szTimeFrame) then
      self.szLHGradeScoreTimeFrame = tbInfo.szTimeFrame
    else
      break
    end
  end
end

function EquipLianHua:CheckIsShowWindow()
  if not self:CheckIsOpen() then
    return false
  end
  local tbEquip = me.GetEquips()
  for k, v in pairs(tbEquip) do
    if k <= Item.EQUIPPOS_AMULET then
      local pEquip = KItem.GetItemObj(v)
      if pEquip and pEquip.nLevel >= Item.EquipLianHua.tbMinLeveLimit[k] then
        return true
      end
    end
  end
  return false
end

function EquipLianHua:LianHuaReq(nEquipPos)
  RemoteServer.OnEquipLianHuaRequest("LianHuaReq", nEquipPos)
end

function EquipLianHua:LianHuaRsp(nEquipPos, nOldLevel)
  local pEquip = me.GetEquipByPos(nEquipPos)
  if pEquip and not me.IsMaskPlayerAttr() then
    self:ApplyAttrib(me, pEquip, self:GetLianHuaLevel(me, pEquip.nEquipPos))
  end
  if 0 == nOldLevel then
    self:OnLianHuaActive(me, nEquipPos)
  end
  me.CenterMsg(i18n.EasyGet("\231\130\188\229\140\150\230\136\144\229\138\159"))
  if Ui:WindowVisible("UIEquipment") then
    Ui:GetWindow("UIEquipment"):NotifyEquipLianHua("showeffect")
  end
  self:OnCheckRedPoint()
  Item.Stone:OnCheckRedPoint()
  EventNotify.OnNotify(EventNotify.emNOTIFY_EQUIP_LIANHUA_SUCCEED)
end

function EquipLianHua:FactionChange(nOldFaction, nFaction)
  local tbEquipList = {}
  local tbLianhuaLevel = {}
  for i = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
    local pEquip = me.GetEquipByPos(i)
    tbLianhuaLevel[i] = self:GetLianHuaLevel(me, i)
    if pEquip then
      tbEquipList[i] = pEquip
    end
  end
  Item.EquipLianHua:UpdateAllEquipAttr(me.GetNpc(), tbEquipList, false, tbLianhuaLevel)
  Item.EquipLianHua:UpdateAllEquipAttr(me.GetNpc(), tbEquipList, true, tbLianhuaLevel)
end

function EquipLianHua:OnCheckRedPoint()
  for nPos, nRedPoint in pairs(self.tbEquipPos2RedPoint) do
    if self:CanLianHua(me, nPos) then
      RedPointMgr:LightRedPoint(nRedPoint)
    else
      RedPointMgr:FadeRedPoint(nRedPoint)
    end
  end
end

function EquipLianHua:OnSyncItemLimitTime()
  if not PlayerEvent.bLogin then
    return
  end
  self:OnCheckRedPoint()
end
