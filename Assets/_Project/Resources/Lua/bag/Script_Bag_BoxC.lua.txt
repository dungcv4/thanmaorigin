local SORT_KEY = Ui:GetUi_SORT_KEY()
local tbToggle = Ui.tbToggle

local function BagItemSortFunc(bagItemDataA, bagItemDataB)
  if bagItemDataA.nSortKey ~= bagItemDataB.nSortKey then
    return bagItemDataA.nSortKey < bagItemDataB.nSortKey
  end
  if bagItemDataA.nDetailType ~= bagItemDataB.nDetailType then
    return bagItemDataA.nDetailType < bagItemDataB.nDetailType
  end
  if bagItemDataA.nLevel ~= bagItemDataB.nLevel then
    return bagItemDataA.nLevel > bagItemDataB.nLevel
  end
  if bagItemDataA.dwTemplateId ~= bagItemDataB.dwTemplateId then
    return bagItemDataA.dwTemplateId < bagItemDataB.dwTemplateId
  end
  return bagItemDataA.nItemId < bagItemDataB.nItemId
end

function Warehouse:GetItemBagList(nItemType)
  self.tbItemBagList = {}
  if self.tbItemBagList[nItemType] == nil then
    self.tbItemBagList[nItemType] = {}
    local tbBagItemData = self:GetItemBagByItemType(nItemType)
    if nil == tbBagItemData then
      return nil
    end
    local tbItemBagList = {}
    for nItemId, bagItemData in pairs(tbBagItemData) do
      table.insert(tbItemBagList, bagItemData)
    end
    table.sort(tbItemBagList, BagItemSortFunc)
    self.tbItemBagList[nItemType] = tbItemBagList
  end
  return self.tbItemBagList[nItemType]
end

function Warehouse:GetItemBagByItemType(nItemType)
  self:InitItemList()
  local tbBagItemData = self.tbItemBag[nItemType]
  return tbBagItemData
end

function Warehouse:InitItemList()
  self.tbItemBag = {}
  local tbItem = me.GetItemListInBox()
  for nIdx, pItem in ipairs(tbItem) do
    Warehouse:UpdateItemListSingle(pItem)
  end
end

function Warehouse:UpdateItemListSingle(pItem)
  local nItemId = pItem.dwId
  local bagItemData = {}
  bagItemData.nSortKey = SORT_KEY[pItem.nItemType] or 1000
  bagItemData.nDetailType = pItem.nDetailType
  bagItemData.nLevel = pItem.nLevel
  bagItemData.nItemId = nItemId
  bagItemData.dwTemplateId = pItem.dwTemplateId
  self.tbItemBag[tbToggle.TogAll] = self.tbItemBag[tbToggle.TogAll] or {}
  local tbBagItemData = self.tbItemBag[tbToggle.TogAll]
  tbBagItemData[nItemId] = bagItemData
  local tbClassifyInfo = Item:GetDetailTypeClassify(pItem.nItemType, pItem.nDetailType)
  if tbClassifyInfo then
    for key, nClassify in pairs(Ui.tbKey2Classify) do
      if 1 == tbClassifyInfo[key] then
        if "bInEquip" == key then
          if not Item.ZhenYuan:IsZhenYuanEquip(pItem) then
            self.tbItemBag[nClassify] = self.tbItemBag[nClassify] or {}
            local tbBagItemData = self.tbItemBag[nClassify]
            tbBagItemData[nItemId] = bagItemData
          end
        else
          self.tbItemBag[nClassify] = self.tbItemBag[nClassify] or {}
          local tbBagItemData = self.tbItemBag[nClassify]
          tbBagItemData[nItemId] = bagItemData
        end
      end
    end
  else
    LogErr(string.format("DetailTypeClassify.tab: DetailType:%d%s not Define Classify!", pItem.nDetailType, i18n.Parse(pItem.szName)))
  end
end
