local tbItemClass = Item:GetClass("LianHuaMaterial")

function tbItemClass:OnClientUse(pItem)
  if not Item.EquipLianHua:CheckIsOpen() then
    me.CenterMsg(i18n.EasyGet("\231\179\187\231\187\159\229\176\154\230\156\170\229\188\128\229\144\175"), false)
  end
  local nEquipPos = KItem.GetItemExtParam(pItem.dwTemplateId, 1)
  local nLevel = KItem.GetItemExtParam(pItem.dwTemplateId, 2)
  local nCurLevel = Item.EquipLianHua:GetLianHuaLevel(me, nEquipPos)
  if not nEquipPos then
    Ui:OpenWindow("UIEquipment")
    return 0
  end
  local pMyEquip = me.GetEquipByPos(nEquipPos)
  if not pMyEquip then
    me.CenterMsg(i18n.EasyGet("\232\175\165\233\131\168\228\189\141\230\156\170\231\169\191\230\136\180\232\163\133\229\164\135"), false)
    return -1
  end
  local nMaxLevel = pMyEquip.nLevel
  if nMaxLevel ~= nLevel then
    me.CenterMsg(i18n.EasyGet("\230\157\144\230\150\153\231\173\137\231\186\167\228\184\142\229\189\147\229\137\141\232\163\133\229\164\135\231\154\132\231\130\188\229\140\150\231\173\137\231\186\167\228\184\141\229\140\185\233\133\141", nLevel), false)
    return -1
  end
  if nCurLevel >= nMaxLevel then
    me.CenterMsg(i18n.EasyGet("\229\183\178\231\130\188\229\140\150\232\135\179\229\189\147\229\137\141\230\156\128\233\171\152\231\186\167\239\188\140\230\155\180\230\141\162\230\155\180\233\171\152\233\152\182\232\163\133\229\164\135\229\143\175\231\187\167\231\187\173\231\130\188\229\140\150"), false)
    return -1
  end
  Ui:OpenWindow("UIEquipment", Ui.UIEquipment.tbSubPanelIndex.PANEL_TYPE_LIANHUA, nEquipPos)
  return 0
end
