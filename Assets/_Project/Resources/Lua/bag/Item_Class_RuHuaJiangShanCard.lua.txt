local tbItem = Item:GetClass("RuHuaJiangShanCard")
tbItem.PARAM_CARDBOOK_ID = 1
tbItem.PARAM_AWARD_ID = 2
tbItem.PARAM_AWARD_KEY = 3

function tbItem:OnAutoUse(pPlayer, nItemTemplateId, nCount)
  local nCardBookId = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_CARDBOOK_ID)
  local nOldCollectKindNum = Item.SouvenirCard:GetCollectKindNum(pPlayer, nCardBookId)
  local nOldRareCardNum = Item.SouvenirCard:GetCardCollectNumByQuality(pPlayer, nCardBookId, 6)
  local nHasCount = Item.SouvenirCard:GetCollectNum(pPlayer, nCardBookId, nItemTemplateId)
  local nTotalNum = 0
  if nHasCount > 0 then
    local szName = KItem.GetItemShowInfo(nItemTemplateId, pPlayer.nFaction, pPlayer.nFactionSect, pPlayer.nSex)
    local szCurrencyType, nNum = Item.tbItemSell:GetSellCurrencyValue(nItemTemplateId)
    if not szCurrencyType or not nNum then
      return false
    end
    nTotalNum = nNum * nCount
    local tbAward = {
      {
        szCurrencyType,
        nTotalNum,
        bSysMsg = 1
      }
    }
    local tbAwardParams = {
      nLogReason = Env.LogWay_RuHuaJiangShanCard
    }
    tbAwardParams.szMsg = i18n.EasyGet("\229\155\189\229\186\134\229\141\161\230\138\152\232\191\148\233\147\182\228\184\164")
    Award:GenerateDefaultAwardParam(tbAwardParams)
    Award:SendAward(pPlayer.dwID, tbAward, tbAwardParams)
    local szMsg = i18n.EasyGet("{0}\229\141\161\233\135\141\229\164\141\230\191\128\230\180\187\239\188\140\229\183\178\229\155\158\230\148\182\228\184\186{1}{2}", szName, nTotalNum, Currency:GetName(szCurrencyType))
    pPlayer.CenterMsg(szMsg)
  end
  local nAwardId = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_AWARD_ID)
  local nAwardKey = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_AWARD_KEY)
  Award:GiveAward(pPlayer.dwID, nAwardId, nAwardKey, {
    nLogReason = Env.LogWay_RuHuaJiangShanCard
  })
  local nSaveKey = Item.SouvenirCard:GetSaveKey(nCardBookId)
  local nSaveId = Item.SouvenirCard:GetSaveId(nCardBookId, nItemTemplateId)
  pPlayer.SetUserValue(nSaveKey, nSaveId, nHasCount + nCount)
  local nNowCollectKindNum = Item.SouvenirCard:GetCollectKindNum(pPlayer, nCardBookId)
  local nNowRareCardNum = Item.SouvenirCard:GetCardCollectNumByQuality(pPlayer, nCardBookId, 6)
  if nNowCollectKindNum ~= nOldCollectKindNum or nNowRareCardNum ~= nOldRareCardNum then
    Event:OnEvent(Event.emEVENT_VNNATION_UPDATE_RANKVAL, pPlayer.dwID, Item.SouvenirCard:GetCollectKindNum(pPlayer, nCardBookId), Item.SouvenirCard:GetCardCollectNumByQuality(pPlayer, nCardBookId, 6))
  end
  Lib:TrackEvent(pPlayer.dwID, Env.track_guoqing_get, nItemTemplateId, nHasCount, nCount, nTotalNum)
  return true
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "RuHuaJiangShanCard")
end
