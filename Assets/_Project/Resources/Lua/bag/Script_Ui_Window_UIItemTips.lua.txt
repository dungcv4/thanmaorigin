function tbWnd:SetItem(nItemId, nTemplateId)
  if not nTemplateId and not nItemId then
    return
  end
  local pItem
  if nItemId and nItemId > 0 then
    pItem = KItem.GetItemObj(nItemId)
    if not pItem then
      return
    end
    nTemplateId = pItem.dwTemplateId
  end
  local tbInfo = Item:GetItemBaseProp(nTemplateId)
  if not tbInfo then
    return
  end
  local tbFuncSet = tbItemClassSetting[tbInfo.szClass]
  if tbFuncSet then
    tbFuncSet(self, nItemId, nTemplateId)
    return
  end
  self.nTemplateId = nTemplateId
  self.nItemId = nItemId
  self.itemGrid:SetItem(self.nItemId, self.nTemplateId, {bShowTip = false})
  local szName, _, _, nQuality = KItem.GetItemShowInfo(nTemplateId, self.nFaction or me.nFaction, self.nFactionSect or me.nFactionSect, self.nSex or me.nSex)
  local szNameColor = Item:GetQualityColor(nQuality) or "White"
  if szName then
    if EDITOR then
      szName = string.format("%s (ID:%d)", i18n.Parse(szName), nTemplateId)
    end
    self.pPanel:Label_SetText(self.txtName, i18n.Parse(szName))
  end
  local requiteLevel = i18n.EasyGet("\233\156\128\230\177\130\231\173\137\231\186\167\239\188\154\230\151\160\233\153\144\229\136\182")
  if tbInfo.nRequireLevel then
    if ZongShiSkill:IsZongShiByLevel(tbInfo.nRequireLevel) then
      requiteLevel = i18n.EasyGet("\233\156\128\230\177\130\231\173\137\231\186\167\239\188\154{0}", Lib:GetLevelText(tbInfo.nRequireLevel))
    else
      requiteLevel = i18n.EasyGet("\233\156\128\230\177\130\231\173\137\231\186\167\239\188\154{0}\231\186\167", Lib:GetLevelText(tbInfo.nRequireLevel))
    end
  end
  self.pPanel:Label_SetText(self.txtLevelRequire, requiteLevel)
  self.pPanel:SetActive(self.txtLevelRequire, true)
  local szClassName = tbInfo.szClass
  local tbClass = Item.tbClass[szClassName]
  self.szClassName = szClassName
  self:UpdateItemCount()
  self:SetItemAttr(pItem, tbClass)
  local szIntro1
  local sMsg2 = ""
  local bShowCenter = true
  if tbClass and tbClass.GetTip then
    szIntro1, sMsg2, bShowCenter = tbClass:GetTip(pItem or {
      dwTemplateId = self.nTemplateId,
      tbItemExtParams = self.tbItemExtParams
    })
    szIntro1 = i18n.Parse(szIntro1)
    sMsg2 = i18n.Parse(sMsg2)
  end
  if szIntro1 and "" ~= szIntro1 then
    self.pPanel:SetActive(self.txtLabel1, true)
    self.pPanel:Label_SetText(self.txtLabel1, szIntro1)
    self.pPanel:Label_SetColorByName(self.txtLabel1, Item.tbTipsColorDef.Lable1.ColorName or "Red")
    if self.nItemId and self.nItemId > 0 then
      self.pPanel:SetActive(self.btnCenter, true)
    end
  else
    self.pPanel:SetActive(self.txtLabel1, false)
    self.pPanel:SetActive(self.btnCenter, false)
  end
  if self.tbLinkInfo then
    if self.tbLinkInfo.RandomAttrInfo then
      self:SetItemAttrByLinkInfo(self.tbLinkInfo.RandomAttrInfo, tbClass)
    end
    if self.tbLinkInfo.tbHuMaiFuData then
      self:SetItemAttrByLinkInfo(self.tbLinkInfo.tbHuMaiFuData, tbClass)
    end
    if self.tbLinkInfo.tbLingZhuData then
      self:SetItemAttrByLinkInfo(self.tbLinkInfo.tbLingZhuData, tbClass)
    end
    if self.tbLinkInfo.tbWXHuShiData then
      self:SetItemAttrByLinkInfo(self.tbLinkInfo.tbWXHuShiData, tbClass)
    end
    if self.tbLinkInfo.XYGCardCount then
      sMsg2 = tbClass:GetCollectCountStr(self.tbLinkInfo.XYGCardCount)
    end
  end
  local szSource = i18n.Parse(tbInfo.szSource)
  if szSource and "" ~= szSource then
    self.pPanel:SetActive(self.txtSource, true)
    self.pPanel:Label_SetText(self.txtSource, i18n.EasyGet("\232\142\183\229\143\150\233\128\148\229\190\132\239\188\154{0}", szSource))
    self.pPanel:Label_SetColorByName(self.txtSource, Item.tbTipsColorDef.ItemSource.ColorName or "Red")
  end
  local bCanSell = self:CanSellItem(pItem)
  self.pPanel:SetActive(self.btnSell, bCanSell)
  if pItem and self.szOpt ~= "Market_ItemBox" and self.szOpt ~= "HanWu_Item" then
    self.pPanel:SetActive(self.btnShare, true)
  else
    self.pPanel:SetActive(self.btnShare, false)
  end
  local nCanMarketStall = false
  self.nMarketType = nil
  nCanMarketStall, self.nMarketType = MarketStallBase:CheckMarketSellItemTypeSettings(pItem)
  self.pPanel:SetActive(self.btnMarketSell, nCanMarketStall)
  local nCanMoveToBox = false
  local nVipState = false
  if pItem and self.szOpt == "ItemBox" then
    local tbBaseInfo = Item:GetItemBaseProp(pItem.dwTemplateId)
    if tbBaseInfo.bForbidToBox and 0 == tbBaseInfo.bForbidToBox then
      nCanMoveToBox = true
    end
    local weekCardLeft = WeekMonthCard:GetWeekCardLeftDay(me)
    local bMonthCardState = WeekMonthCard:CheckMonthCardState(me)
    if weekCardLeft > 0 or bMonthCardState then
      nVipState = true
    end
  end
  self.pPanel:SetActive(self.btnMoveToBox, nCanMoveToBox and nVipState)
  self.pPanel:Label_SetText(self.btnMoveToBox .. "/Text", i18n.EasyGet("\231\167\187\229\133\165\228\187\147\229\186\147"))
  if pItem then
    local bCanTrade, szTradeWay = self:GetItemTradeText(pItem)
    self.pPanel:SetActive(self.txtTradable, bCanTrade)
    self.pPanel:SetActive(self.txtNonTradable, not bCanTrade)
    if bCanTrade then
      self.pPanel:Label_SetText(self.txtTradable, szTradeWay)
      self.pPanel:Label_SetColorByName(self.txtTradable, Item.tbTipsColorDef.ItemTransaction.ColorName or "Red")
    else
      self.pPanel:Label_SetColorByName(self.txtNonTradable, Item.tbTipsColorDef.NoItemTransaction.ColorName or "Red")
    end
  else
    self.pPanel:SetActive(self.txtTradable, false)
    self.pPanel:SetActive(self.txtNonTradable, false)
  end
  local szIntro2 = i18n.Parse(tbInfo.szIntro)
  local szIntroBottom
  if tbClass then
    if tbClass.GetIntroBottom then
      szIntroBottom = tbClass:GetIntroBottom(nTemplateId)
    end
    if tbClass.GetIntrol then
      local szIntro = tbClass:GetIntrol(nTemplateId)
      if szIntro and "" ~= szIntro then
        szIntro2 = szIntro
      end
    end
    local tbUseSetting
    if self.szOpt and self.tbButtonOption[self.szOpt] then
      tbUseSetting = {}
      if self.tbButtonOption[self.szOpt].fnCondition and self.tbButtonOption[self.szOpt].fnCondition(self) then
        tbUseSetting = self.tbButtonOption[self.szOpt].params
      end
      if self.tbButtonOption[self.szOpt].bHideShareButton then
        self.pPanel:SetActive(self.btnShare, false)
      end
      local bShowSell = not self.tbButtonOption[self.szOpt].bHideSellButton
      bCanSell = bCanSell and bShowSell
      self.pPanel:SetActive(self.btnSell, bCanSell)
      local bShowMarket = not self.tbButtonOption[self.szOpt].bHideMarketButton
      bShowMarket = nCanMarketStall and bShowMarket
      self.pPanel:SetActive(self.btnMarketSell, bShowMarket)
    end
    if "" == szClassName then
      if tbUseSetting then
        self:SetButtonFunc(tbUseSetting.szFirstName, tbUseSetting.fnFirst, tbUseSetting.szSecondName, tbUseSetting.fnSecond, tbUseSetting.szThirdName, tbUseSetting.fnThird)
      end
    else
      if not tbUseSetting and tbClass.GetUseSetting then
        tbUseSetting = tbClass:GetUseSetting(nTemplateId, self.nItemId, self.szOpt)
      end
      if tbUseSetting then
        if tbUseSetting.bHideShareButton then
          self.pPanel:SetActive(self.btnShare, false)
        end
        self:SetButtonFunc(tbUseSetting.szFirstName, tbUseSetting.fnFirst, tbUseSetting.szSecondName, tbUseSetting.fnSecond, tbUseSetting.szThirdName, tbUseSetting.fnThird)
      elseif self.nItemId and self.nItemId > 0 then
        if pItem and pItem.nCount > 1 and Item:GetItemUseManyLimit(szClassName) and tbClass.OnUseMany and Item:CheckIsOpenManyUseUi(szClassName) then
          self:SetButtonFunc(i18n.EasyGet("\228\189\191\231\148\168"), self.UseMany)
        elseif tbClass.OnUseAll and tbClass.OnUse then
          self:SetButtonFunc(i18n.EasyGet("\229\133\168\233\131\168\228\189\191\231\148\168"), self.UseAll, i18n.EasyGet("\228\189\191\231\148\168"), self.UseItem)
        elseif tbClass.OnUse or tbClass.OnClientUse then
          self:SetButtonFunc(i18n.EasyGet("\228\189\191\231\148\168"), self.UseItem)
        end
      end
    end
    if tbClass.GetTipMsg and pItem then
      local nRet, nOwnCount, nResDailyCount = tbClass:GetTipMsg(pItem)
      if nOwnCount > 0 then
        local szMsg = i18n.EasyGet("\229\137\169\228\189\153\228\189\191\231\148\168\230\172\161\230\149\176\239\188\154{0}", nOwnCount)
        self.pPanel:Label_SetText(self.txtLeftTime, szMsg)
        self.pPanel:SetActive(self.txtLeftTime, true)
      end
      if 1 ~= nRet then
        self.pPanel:SetActive(self.txtWarning, true)
      end
      self.pPanel:SetActive(self.btnCenter, 1 == nRet)
    end
    if tbClass.GetAttrInfo then
      local szAttrDes = tbClass:GetAttrInfo(nTemplateId)
      if szAttrDes and "" ~= szAttrDes then
        szIntro2 = string.format([[
%s
%s]], szIntro2, szAttrDes)
      end
    end
    if tbClass.GetUseTimes then
      local nUseTime, nLeaveTime = tbClass:GetUseTimes(nTemplateId)
      local szMsg = i18n.EasyGet("\228\189\191\231\148\168\230\149\176\233\135\143\239\188\154{0}/{1}", nUseTime, nLeaveTime)
      self.pPanel:Label_SetText(self.txtLeftTime, szMsg)
      self.pPanel:SetActive(self.txtLeftTime, true)
    end
    if tbClass.GetUseTimesAndTimeLimit then
      local nUseTimes, nTimeLimit = tbClass:GetUseTimesAndTimeLimit(nTemplateId)
      if nUseTimes then
        local szMsg = i18n.EasyGet("\229\137\169\228\189\153\228\189\191\231\148\168\230\172\161\230\149\176\239\188\154<color=#28ff28>{0}</color>\230\172\161", nUseTimes)
        if nTimeLimit then
          szMsg = szMsg .. i18n.EasyGet("\239\188\136<color=#b60000>{0}</color>\229\164\169\229\144\142\229\136\183\230\150\176\239\188\137", nTimeLimit)
        end
        self.pPanel:Label_SetText(self.txtLeftTime, szMsg)
        self.pPanel:SetActive(self.txtLeftTime, true)
      end
    end
    if tbClass.GetWeekLeftTimes then
      local nLeftTimes, nMaxUseTimes = tbClass:GetWeekLeftTimes(me, nTemplateId)
      if nLeftTimes then
        local szMsg = i18n.EasyGet("\229\137\169\228\189\153\228\189\191\231\148\168\230\172\161\230\149\176\239\188\154<color=#28ff28>{0}/{1}</color>\230\172\161\239\188\136\230\175\143\229\145\168\228\184\128\229\136\183\230\150\176\239\188\137", nLeftTimes, nMaxUseTimes)
        self.pPanel:Label_SetText(self.txtLeftTime, szMsg)
        self.pPanel:SetActive(self.txtLeftTime, true)
        self.pPanel:SetActive(self.btnCenter, nLeftTimes > 0)
      end
    end
    if tbClass.GetPriceTips then
      local nAppraise = tbClass:GetPriceTips(me, nTemplateId)
      if nAppraise then
        self.pPanel:Label_SetText(self.txtAppraiseValue, nAppraise)
        self.pPanel:SetActive(self.txtAppraise, true)
      else
        self.pPanel:SetActive(self.txtAppraise, false)
      end
    else
      self.pPanel:SetActive(self.txtAppraise, false)
    end
    if bCanSell or "HuMaiFu" == szClassName then
      local szCurrency, nPrice = Item.tbItemSell:GetSellCurrencyValue(self.nTemplateId, self.nItemId)
      if szCurrency then
        self.pPanel:Label_SetText(self.txtPriceValue, nPrice)
        self.pPanel:SetActive(self.txtPrice, true)
        local tbCustomData = Item.tbItemSell:GetCustomSellData(szCurrency)
        local szIcon = tbCustomData.funcGetIcon(szCurrency)
        self.pPanel:Sprite_SetSprite(self.imgSellMoney, szIcon)
      else
        self.pPanel:SetActive(self.txtPrice, false)
      end
    else
      self.pPanel:SetActive(self.txtPrice, false)
    end
  end
  if pItem and pItem.nUseDelayMins > 0 then
    local nCurTime = GetTime()
    local nGenItemTime = GetItemGenTime(pItem.dwId)
    local nDelatTime = nGenItemTime + pItem.nUseDelayMins * 60 - nCurTime
    if nDelatTime > 0 then

function tbWnd:OnFixedUpdate()
  local pItem = KItem.GetItemObj(self.nItemId or 0)
  if pItem then
    local bHas, nGenTime, nTimeOut = Item:GetTimeOut(pItem)
    if bHas then
      local bIsTimeOut = Item:CheckIsTimeOut(pItem)
      if bIsTimeOut then
        self.pPanel:Label_SetText(self.txtItemTime, i18n.EasyGet("\229\183\178\229\164\177\230\149\136"))
      else
        local nLeaveSec = nTimeOut - GetTime()
        local nLeaveTime = Lib:TimeDesc2(nLeaveSec)
        if nLeaveSec < 600 then
          nLeaveTime = string.format("<color=#b60000>%s</color>", nLeaveTime)
        else
          nLeaveTime = string.format("<color=#ffff0e>%s</color>", nLeaveTime)
        end
        self.pPanel:Label_SetText(self.txtItemTime, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154{0}", nLeaveTime))
      end
      self.pPanel:SetActive(self.txtItemTime, true)
    end
  elseif self.nTimeOutValue then
    if "string" == type(self.nTimeOutValue) and not Lib:IsEmptyStr(self.nTimeOutValue) then
      self.nTimeOutValue = Lib:ParseConfigDateTime(self.nTimeOutValue) or 0
    end
    if type(self.nTimeOutValue) == "number" and self.nTimeOutValue > 0 then
      local nLeaveSec = self.nTimeOutValue - GetTime()
      if nLeaveSec <= 0 then
        self.pPanel:Label_SetText(self.txtItemTime, i18n.EasyGet("\229\183\178\229\164\177\230\149\136"))
      elseif nLeaveSec < 600 then
        local szLeaveSec = string.format("<color=#b60000>%s</color>", Lib:TimeDesc2(nLeaveSec))
        self.pPanel:Label_SetText(self.txtItemTime, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154{0}", szLeaveSec))
      else
        local szLeaveSec = string.format("<color=#ffff0e>%s</color>", Lib:TimeDesc2(nLeaveSec))
        self.pPanel:Label_SetText(self.txtItemTime, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154{0}", szLeaveSec))
      end

    end
  end
end

      pPanel:Label_SetColorByName("Attribute1/txtAttr", nColor or "White")
    end
  end
end

function tbWnd:OnSyncRoleInfo(tbExistList)
  if not self.szClassName then
    return
  end
  local pItem
  if self.nItemId then
    pItem = KItem.GetItemObj(self.nItemId)
  end
  local sMsg2 = ""
  local tbClass = Item.tbClass[self.szClassName]
  if tbClass and tbClass.GetTip then
    _, sMsg2 = tbClass:GetTip(pItem or {
      dwTemplateId = self.nTemplateId,
      tbItemExtParams = self.tbItemExtParams
    })
    sMsg2 = i18n.Parse(sMsg2)
  end
  if sMsg2 and "" ~= sMsg2 then
    self:SetMainIntro(sMsg2)
  end
end
