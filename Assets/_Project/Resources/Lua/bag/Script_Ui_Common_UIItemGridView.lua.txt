local tbWnd = {}
Ui:SetClass("UIItemGridView", tbWnd)

function Ui:NewItemGridView(tbOwner, ...)
  local tbClass = Ui:NewClass(Ui:GetClass("UIItemGridView"))
  tbClass.tbOwner = tbOwner
  tbClass:OnCreate(...)
  return tbClass
end

function tbWnd:OnCreate(szItemTemplateKey)
  self.szItemTemplateKey = szItemTemplateKey
  self.ItemTemplateObj = self.tbOwner.pPanel:GetObject(self.szItemTemplateKey)
  self.tbOwner.pPanel:SetActive(self.szItemTemplateKey, false)
  self.tbItemList = {}
  self.tbItemKey = {}
end

function tbWnd:SetItem(nIndex, tbItem, bEnableClick, funcClickFunc, ...)
  if nIndex > #self.tbItemList then
    self.tbItemList[nIndex] = self:CreateGrid(nIndex)
  end
  local tbGrid = self.tbItemList[nIndex]
  tbGrid:SetActive(true)
  if "item" == tbItem[1] or "Item" == tbItem[1] then
    tbGrid:SetItemByTemplate(tbItem[2], tbItem[3] or 1, me.nFaction, me.nFactionSect, nil, nil, nil, tbItem[4])
  else
    tbGrid:SetGenericItem(tbItem)
  end
  tbGrid:SetClickEnable(bEnableClick, funcClickFunc, ...)
  return tbGrid
end

function tbWnd:SetEmpty(nIndex, bShowBg)
  if nIndex > #self.tbItemList then
    self.tbItemList[nIndex] = self:CreateGrid(nIndex)
  end
  local tbGrid = self.tbItemList[nIndex]
  tbGrid:SetActive(true)
  tbGrid:SetEmpty(bShowBg)
  return tbGrid
end

function tbWnd:CreateGrid(nIndex)
  local szItemKey = self.tbOwner.pPanel:CloneObject(self.szItemTemplateKey, "Element" .. nIndex)
  self.tbItemKey[nIndex] = szItemKey
  local tbGrid = Ui:NewExistItemGrid(self.tbOwner, szItemKey .. "/UIItemGrid")
  tbGrid:SetEmpty()
  return tbGrid
end

function tbWnd:Reset()
  for i = 1, #self.tbItemList do
    self.tbItemList[i]:SetActive(false)
  end
end

function tbWnd:ResetFrom(nStartIndex)
  for i = nStartIndex, #self.tbItemList do
    self.tbItemList[i]:SetActive(false)
  end
end

function tbWnd:Clear()
  for i = 1, #self.tbItemList do
    self.tbItemList[i]:UnInitPanel()
    self.tbOwner.pPanel:DeleteObject(self.tbItemKey[i])
  end
  self.tbItemList = {}
  self.tbItemKey = {}
end

function tbWnd:GetItemKey(nIndex)
  if nIndex > #self.tbItemKey then
    return
  end
  return self.tbItemKey[nIndex]
end

function tbWnd:GetUIItemGridByIndex(nIndex)
  return self.tbItemList[nIndex]
end

function tbWnd:HideRange(nStartIndex, nEndIndex)
  nStartIndex = math.max(nStartIndex, 1)
  if not nEndIndex then
    nEndIndex = #self.tbItemList
  else
    nEndIndex = math.min(#self.tbItemList, nEndIndex)
  end
  for i = nStartIndex, nEndIndex do
    self.tbOwner.pPanel:SetActive(self.tbItemKey[i], false)
  end
end

function tbWnd:ShowRange(nStartIndex, nEndIndex)
  nStartIndex = math.max(nStartIndex, 1)
  if not nEndIndex then
    nEndIndex = #self.tbItemList
  else
    nEndIndex = math.min(#self.tbItemList, nEndIndex)
  end
  for i = nStartIndex, nEndIndex do
    self.tbOwner.pPanel:SetActive(self.tbItemKey[i], true)
  end
end
