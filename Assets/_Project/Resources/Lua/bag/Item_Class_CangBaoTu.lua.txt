function tbItem:GetTip(it)
  if not it.dwId then
    return ""
  end
  local nMapTemplateId = it.GetIntValue(self.PARAM_MAPID)
  local nPosX = it.GetIntValue(self.PARAM_POSX)
  local nPosY = it.GetIntValue(self.PARAM_POSY)
  if not nMapTemplateId then
    Log("CangBaoTu GetTip nMapTemplateId is null!!", nMapTemplateId)
    return ""
  end
  local tbMapSetting = Map:GetMapSetting(nMapTemplateId)
  if not tbMapSetting then
    Log("CangBaoTu GetTip tbMapSetting is null!!", nMapTemplateId)
    return ""
  end
  local nItemType = KItem.GetItemExtParam(it.dwTemplateId, tbItem.EXT_PARAM_1)
  local szTip = i18n.EasyGet("\232\151\143\229\174\157\231\130\185\239\188\154{0}({1}, {2})", i18n.Parse(tbMapSetting.MapName), Lib:MathRound(nPosX * Map.nShowPosScale), Lib:MathRound(nPosY * Map.nShowPosScale))
  return szTip
end

end

function tbItem:RandomPos(pItem, pPlayer)
  local nItemType = KItem.GetItemExtParam(pItem.dwTemplateId, tbItem.EXT_PARAM_1)
  local nLevel = pPlayer and pPlayer.nLevel or self.tbLevel[nItemType].nMinLevel
  local tbCurInfo = {
    tbUsedMap = {}
  }
  if pPlayer and pPlayer.tbCangBaoTuInfo and pPlayer.tbCangBaoTuInfo[nItemType] then
    tbCurInfo = pPlayer.tbCangBaoTuInfo[nItemType]
  end
  local tbPosInfo = {}
  local nTotalCount = 0
  if tbCurInfo.nMapTemplateId then
    local nLastPosIndex = pItem.GetIntValue(self.PARAM_LASTPOSINDEX)
    local tbTemp = Lib:CopyTB(self.tbAllPos[nItemType][tbCurInfo.nMapTemplateId])
    if tbTemp[nLastPosIndex] then
      table.remove(tbTemp, nLastPosIndex)
    end
    tbPosInfo = {tbTemp}
    nTotalCount = #tbTemp
  else
    if self.tbAllPos[nItemType] == nil then
      LogErr("CangBaoTu RandomPos error", nItemType)
      if pItem then
        LogErr("CangBaoTu RandomPos log ", pItem.dwTemplateId)
      end
    end
    for nMapTemplateId, tbInfo in pairs(self.tbAllPos[nItemType] or {}) do
      if not tbCurInfo.tbUsedMap[nMapTemplateId] and nLevel >= tbInfo.nLevel then
        table.insert(tbPosInfo, tbInfo)
        nTotalCount = nTotalCount + #tbInfo
      end
    end
  end
  if nTotalCount <= 0 then
    LogErr(string.format("CangBaoTu RandomPos nTotalCount = %s, nItemType = %s, nLevel = %s", nTotalCount, nItemType, nLevel))
  end
  if Lib:CountTB(tbPosInfo) <= 0 then
    LogErr(string.format("CangBaoTu RandomPos tbPosInfo empty, nItemType = %s, nLevel = %s", nItemType, nLevel))
  end
  local nRandom = MathRandom(nTotalCount)
  local tbPos
  for _, tbInfo in pairs(tbPosInfo) do
    tbPos = tbInfo[nRandom]
    if tbPos then
      break
    end
    nRandom = nRandom - #tbInfo
  end
  local nMapId = tbPos[1]
  local nX = tbPos[2]
  local nY = tbPos[3]
  local nRandIndex = 0
  if self.tbAllPos and self.tbAllPos[nItemType] and self.tbAllPos[nItemType][nMapId] then
    for i, tbInfo in pairs(self.tbAllPos[nItemType][nMapId]) do
      if tbInfo[2] == nX and tbInfo[3] == nY then
        nRandIndex = i
        break
      end
    end
  end
  pItem.SetIntValue(self.PARAM_MAPID, nMapId)
  pItem.SetIntValue(self.PARAM_POSX, nX)
  pItem.SetIntValue(self.PARAM_POSY, nY)
  pItem.SetIntValue(self.PARAM_LASTPOSINDEX, nRandIndex)
end

function tbItem:GetCangBaoTuPos(pItem)
  local nMapTemplateId = pItem.GetIntValue(self.PARAM_MAPID)
  local nPosX = pItem.GetIntValue(self.PARAM_POSX)
  local nPosY = pItem.GetIntValue(self.PARAM_POSY)
  return nMapTemplateId, nPosX, nPosY
end

function tbItem:CheckCanWaBao(pPlayer, nItemId)
  if not pPlayer then
    return false, i18n.EasyGet("\231\142\169\229\174\182\228\184\141\229\173\152\229\156\168")
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem or pItem.szClass ~= "CangBaoTu" then
    return false, i18n.EasyGet("\229\146\166\239\188\140\232\151\143\229\174\157\229\155\190\229\145\162\239\188\129")
  end
  local nMapTemplateId, nPosX, nPosY = self:GetCangBaoTuPos(pItem)
  local _, nX, nY = pPlayer.GetWorldPos()
  if pPlayer.nMapTemplateId ~= nMapTemplateId or math.abs(nPosX - nX) > Npc.DIALOG_DISTANCE or math.abs(nPosY - nY) > Npc.DIALOG_DISTANCE then
    return false, i18n.EasyGet("\230\173\164\229\164\132\230\178\161\230\156\137\229\174\157\232\151\143\239\188\140\230\141\162\228\184\170\229\156\176\230\150\185\232\175\149\232\175\149\229\144\167\239\188\129")
  end
  return true, szMsg, pItem
end

function tbItem:RecordPlayerInfo(pPlayer, nItemType)
  if not self.tbAllPos[nItemType][pPlayer.nMapTemplateId] then
    return
  end
  pPlayer.tbCangBaoTuInfo = pPlayer.tbCangBaoTuInfo or {}
  pPlayer.tbCangBaoTuInfo[nItemType] = pPlayer.tbCangBaoTuInfo[nItemType] or {
    tbUsedMap = {}
  }
  if not pPlayer.tbCangBaoTuInfo[nItemType].nMapTemplateId or pPlayer.tbCangBaoTuInfo[nItemType].nMapTemplateId ~= pPlayer.nMapTemplateId then
    pPlayer.tbCangBaoTuInfo[nItemType].nMapTemplateId = pPlayer.nMapTemplateId
    pPlayer.tbCangBaoTuInfo[nItemType].nCurCount = 0
  end
  pPlayer.tbCangBaoTuInfo[nItemType].nCurCount = pPlayer.tbCangBaoTuInfo[nItemType].nCurCount + 1
  if pPlayer.tbCangBaoTuInfo[nItemType].nCurCount < self.tbSetting[nItemType].MIN_POS_COUNT_ONE_MAP then
    return
  end
  if pPlayer.tbCangBaoTuInfo[nItemType].nCurCount < self.tbSetting[nItemType].MAX_POS_COUNT_ONE_MAP and MathRandom(1000) <= self.tbSetting[nItemType].CHANGE_MAP_RATE then
    return
  end
  pPlayer.tbCangBaoTuInfo[nItemType].nMapTemplateId = nil
  pPlayer.tbCangBaoTuInfo[nItemType].nCurCount = nil
  pPlayer.tbCangBaoTuInfo[nItemType].tbUsedMap[pPlayer.nMapTemplateId] = 1
  local nCanUseMapCount = 0
  for _, tbInfo in pairs(self.tbAllPos[nItemType]) do
    if pPlayer.nLevel >= tbInfo.nLevel then
      nCanUseMapCount = nCanUseMapCount + 1
    end
  end
  if nCanUseMapCount <= Lib:CountTB(pPlayer.tbCangBaoTuInfo[nItemType].tbUsedMap) then
    pPlayer.tbCangBaoTuInfo[nItemType].tbUsedMap = {}
  end
end

function tbItem:CheckDungeon(pPlayer)
  tbItem.tbDungeonData[me.dwID] = tbItem.tbDungeonData[me.dwID] or {nCurTimes = 0}
  local tbDData = tbItem.tbDungeonData[me.dwID]
  if tbDData.nCurTimes >= tbItem.nMinDungeonTimes then
    tbDData.nCurTimes = 0
    tbDData.bHasDungeon = false
  end
  tbDData.nCurTimes = tbDData.nCurTimes + 1
  if tbDData.bHasDungeon then
    return
  end
  local nRate = 0
  for _, tbInfo in pairs(self.tbDungeonRate) do
    nRate = tbInfo[2]
    if pPlayer.nLevel <= tbInfo[1] then
      break
    end
  end
  if nRate < MathRandom(100000) then
    return
  end
  tbDData.bHasDungeon = true
  
  local function fnCallBack(dwRoleId)
    local pPlayer = KPlayer.GetPlayerObjById(dwRoleId)
    if not pPlayer then
      return
    end
    Fuben.DungeonFubenMgr:CreateFuben(pPlayer, true)
  end
  
  Timer:Register(Env.GAME_FPS * self.nDungeonWaiteTime + 1, fnCallBack, pPlayer.dwID)
  return true
end

function tbItem:CheckDaoBaoZei(pPlayer)
  local nRate = 0
  for _, tbInfo in ipairs(self.tbDBZRate) do
    nRate = tbInfo[2]
    if pPlayer.nLevel <= tbInfo[1] then
      break
    end
  end
  local nHit = MathRandom(self.MAX_DBZ_RATE)
  if nRate < nHit then
    return
  end
  self:RandomDBZ(pPlayer)
  return true
end

function tbItem:RandomDBZ(pPlayer)
  local nCount = 1
  local nMapID, nPox, nPoy = pPlayer.GetWorldPos()
  for i = 1, nCount do
    local tbOffsetPos = self.tbDBZOffsetPos[i] or {0, 0}
    local pNpc = KNpc.Add(self.nDBZId, pPlayer.nLevel, -1, nMapID, nPox + tbOffsetPos[1], nPoy + tbOffsetPos[2], 0, 0)
    if pNpc then
      CangBaoTu:AddTimerObj(CangBaoTu.ObjType_Npc, pNpc.nId)
      pPlayer.CallClientScript("CangBaoTu:SetAutoFight", true)
    end
  end
end

function tbItem:OnEndProgress(nItemId)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  if not self:CheckNeedHuanHua(me, pItem.dwTemplateId) then
    return
  end
  local bRet, szMsg, pItem = self:CheckCanWaBao(me, nItemId)
  if not bRet then
    me.CenterMsg(szMsg)
    return
  end
  bRet = false
  local szName = pItem.szName
  local nCount = pItem.nCount
  local nItemDwId = pItem.dwId
  local nRandomByLevelKindId = pItem.nRandomByLevelKindId
  local ndwTemplateId = pItem.dwTemplateId
  local nItemType = KItem.GetItemExtParam(ndwTemplateId, tbItem.EXT_PARAM_1)
  if nItemType == self.TYPE_SENIOR then
    local bAllReady, szTips = self:CheckTeamMember(pItem)
    if not bAllReady then
      me.CenterMsg(szTips)
      return
    end
  end
  local nWaBaoLogReason = Env.LogWay_LongMaiTuWaBao
  local nConsumeCount = me.ConsumeItem(pItem, 1, nWaBaoLogReason)
  if 1 ~= nConsumeCount then
    return
  end
  local nEvent = self:RandEvent(nItemType, me)
  if nEvent == self.EventType_Award then
    self:SendCangBaoTuAward(ndwTemplateId)
  elseif nEvent == self.EventType_Box then
    self:CreateBox(me)
  elseif nEvent == self.EventType_DBZ then
    self:RandomDBZ(me)
  end
  if self.TYPE_NORMAL == nItemType then
  elseif self.TYPE_SENIOR == nItemType then
    self:AddImitity()
  end
  CangBaoTu:AddDailyTimes(me)
  ActivePoint:FinishTarget(me, "DigTreasure")
  Event:OnEvent(Event.emEVENT_CANGBAOTU, me)
  self:RecordPlayerInfo(me, nItemType)
  if nEvent ~= self.EventType_Award and self.TYPE_NORMAL == nItemType or nCount <= 1 or self.TYPE_SENIOR == nItemType then
    self:RandomPosById(nItemDwId, me)
    me.CallClientScript("CangBaoTu:OnEndProgress", nItemDwId, false)
  else
    self:RandomPosById(nItemDwId, me)
    me.CallClientScript("CangBaoTu:OnEndProgress", nItemDwId, true)
    me.CallClientScript("CangBaoTu:AutoUse", nItemDwId)
  end
  if nCount <= 1 then
    me.CallClientScript("Ui:CloseWindow", "UIQuickUseItem")
  end
  local nUseTemplateId = ndwTemplateId
  local nLogResult = nEvent
  local tbTeamList = {}
  if me.dwTeamID > 0 then
    local tbMember = TeamMgr:GetMembers(me.dwTeamID)
    for _, nPlayerId in pairs(tbMember) do
      local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
      if pPlayer then
        table.insert(tbTeamList, nPlayerId)
      end
    end
    Lib:MBIPLog(me.dwID, Env.cangbaotu_usetu, nUseTemplateId, nLogResult, table.unpack(tbTeamList))
  else
    table.insert(tbTeamList, me.dwID)
    Lib:MBIPLog(me.dwID, Env.cangbaotu_usetu, nUseTemplateId, nLogResult, table.unpack(tbTeamList))
  end
  local tbLastAwards = Award:FetchLastAwards()
  local tbMoneyForTrack, tbItemForTrack = Award:GetAwardListForMBILog(tbLastAwards, false)
  Lib:CallBack({
    Event.OnEvent,
    Event,
    Event.emEVENT_XUI_XIAN_TASK,
    me,
    "CangBao",
    1
  })
  Lib:TrackEvent(me.dwID, Env.track_wabao_join, me.szDeviceId or "", me.szIP or "", me.szMAC or "", tbMoneyForTrack[1] or "", tbMoneyForTrack[2] or 0, tbItemForTrack[1] or 0, tbItemForTrack[2] or 0)
end

function tbItem:SendCangBaoTuAward(ndwTemplateId)
  local nItemType = KItem.GetItemExtParam(ndwTemplateId, tbItem.EXT_PARAM_1)
  local nAwardId = KItem.GetItemExtParam(ndwTemplateId, tbItem.EXT_PARAM_2)
  local nAwardKey = KItem.GetItemExtParam(ndwTemplateId, tbItem.EXT_PARAM_3)
  if nAwardKey <= 0 then
    nAwardKey = Award:GetLevelAwardKey(me.nLevel)
  end
  local nExtAwardId = KItem.GetItemExtParam(ndwTemplateId, tbItem.EXT_PARAM_4)
  local nExtAwardKey = KItem.GetItemExtParam(ndwTemplateId, tbItem.EXT_PARAM_5)
  local tbBaseInfo = Item:GetItemBaseProp(ndwTemplateId)
  local szItemName = tbBaseInfo.szName
  local bResult = self:SendCaptainAward(nAwardId, nAwardKey, me.dwID, szItemName, ndwTemplateId)
  if not bResult then
    me.CenterMsg(i18n.EasyGet("\229\146\166\239\188\140\230\140\150\229\174\157\229\164\177\232\180\165\228\186\134\239\188\129"))
    return
  end
  if self.TYPE_SENIOR == nItemType then
    self:SendTeamAward(nExtAwardId, nExtAwardKey)
  end
end

function tbItem:SendCaptainAward(nAwardId, nAwardKey, nPlayerID, szItemName, ndwTemplateId)
  if not (nAwardId and nAwardKey) or nAwardId <= 0 or nAwardKey <= 0 then
    Lib:LogFormat("ItemTemplateId:%s AwardId or AwardKey not configured", szItemName)
    return
  end
  local nWaBaoLogReason = Env.LogWay_LongMaiTuWaBao
  local nRet = Award:GiveAward(nPlayerID, nAwardId, nAwardKey, {nLogReason = nWaBaoLogReason})
  return nRet
end

function tbItem:SendTeamAward(nAwardId, nAwardKey)
  if not (nAwardId and nAwardKey) or nAwardId <= 0 or nAwardKey <= 0 then
    Lib:LogFormat("ItemTemplateId:%d AwardId or AwardKey not configured", ndwTemplateId)
    return
  end
  local nTeamID = me.dwTeamID
  if not nTeamID or 0 == nTeamID then
    return
  end
  local tbTeamMember = TeamMgr:GetMembers(nTeamID) or {}
  for _, dwID in pairs(tbTeamMember) do
    if dwID ~= me.dwID then
      local pPlayer = KPlayer.GetPlayerObjById(dwID)
      if pPlayer then
        local nDegreee = DegreeCtrl:GetDegree(pPlayer, "CangBaoTuAssist")
        if not nDegreee or nDegreee <= 0 then
          pPlayer.CenterMsg(i18n.EasyGet("\228\187\138\230\151\165\229\141\143\229\138\169\230\140\150\229\174\157\230\172\161\230\149\176\229\183\178\231\187\143\231\148\168\229\174\140\239\188\129"))
        else
          DegreeCtrl:ReduceDegree(pPlayer, "CangBaoTuAssist", 1)
          Award:GiveAward(dwID, nAwardId, nAwardKey, {
            nLogReason = Env.LogWay_CangBaoTuWaBao
          })
        end
      end
    end
  end
end

function tbItem:AddImitity()
  if not me.dwTeamID or 0 == me.dwTeamID then
    return
  end
  local tbMember = TeamMgr:GetMembers(me.dwTeamID)
  if not next(tbMember) then
    return
  end
  for _, dwID in pairs(tbMember) do
    if dwID ~= me.dwID then
      local bIsFriend = FriendShip:IsFriend(me.dwID, dwID)
      if bIsFriend then
        FriendShip:AddImitity(me.dwID, dwID, self.SeniorCanBaoTuImitity, Env.LogWay_CangBaoTuWaBao)
      end
    end
  end
end

function tbItem:CheckBoss()
  local nHit = MathRandom(self.nBossTotalRate)
  if nHit <= self.nBossRate then
    local nMapTemplateId, nPosX, nPosY = me.GetWorldPos()
    local tbOffsetPos = self.BossOffsetPos[1] or {0, 0}
    local bRet = RandomBoss:RandomOneBoss(nMapTemplateId, nPosX + tbOffsetPos[1], nPosY + tbOffsetPos[2], self.szBossType, me.dwID)
    if not bRet then
      Log("player random cangbaotuboss fail!!", me.dwID, nMapTemplateId, nPosX, nPosY)
      return
    end
    Dialog:SendBlackBoardMsg(me, self.szBossBlackMsg)
    return true
  end
end

function tbItem:RandomWeaponByType(nWeaponType)
  local tbAwardInfo = self.tbWeaponAward[nWeaponType]
  if not tbAwardInfo then
    return
  end
  local nTotalRate = 0
  for nLevel, tbInfo in pairs(self.tbWeaponRate) do
    if nLevel <= me.nLevel then
      local nRate = tbInfo[nWeaponType] or 0
      nTotalRate = nTotalRate + nRate
    end
  end
  if 0 == nTotalRate then
    Log("[cangbaotu RandomWeaponByType nTotalRate is zero !]", nWeaponType)
    return
  end
  local nHitRate = MathRandom(nTotalRate)
  local tbRowInfo = {}
  local tbAward = {}
  for _, tbRow in ipairs(tbAwardInfo) do
    if tbRow.nAwardLevel <= me.nLevel then
      nHitRate = nHitRate - tbRow.nRate
      if nHitRate <= 0 then
        tbAward = tbRow.tbAward
        tbRowInfo = tbRow
        break
      end
    end
  end
  if next(tbAward) then
    if 1 == tbRowInfo.nAdd2Auction then
    else
      me.SendAward(tbAward, false, true, Env.LogWay_CangBaoTuWaBao)
    end
    local nItemId = tbAward[1][2]
    local szItemName = KItem.GetItemShowInfo(nItemId) or ""
    local szMsg = i18n.EasyGet("\227\128\140{0}\227\128\141\232\183\159\233\154\143\231\157\128\233\171\152\231\186\167\232\151\143\229\174\157\229\155\190\231\154\132\230\140\135\229\188\149\239\188\140\231\171\159\231\132\182\229\143\145\231\142\176\228\186\134{1}\239\188\129", me.szName, szItemName)
    if tbRowInfo.bKinNotice and me.dwKinId and me.dwKinId > 0 then
      ChatMgr:SendPlayerMsg(ChatMgr.ChannelType.Kin, me.dwID, me.szName, me.nFaction, me.nPortrait, me.nLevel, szMsg)
    end
    if tbRowInfo.bSystemNotice then
      KPlayer.SendWorldNotify(1, 1000, szMsg, ChatMgr.SystemMsgType.System, 1000)
    end
  end
end

function tbItem:IsIllegal(nItemType, nMapTemplateId, nPosX, nPosY, pPlayer)
  if not self.tbAllPos[nItemType][nMapTemplateId] then
    if pPlayer.tbCangBaoTuInfo and pPlayer.tbCangBaoTuInfo[nItemType] then
      pPlayer.tbCangBaoTuInfo[nItemType] = nil
    end
    return false
  end
  for _, tbPos in ipairs(self.tbAllPos[nItemType][nMapTemplateId]) do
    if tbPos[2] and tbPos[3] and tbPos[4] and tbPos[2] == nPosX and tbPos[3] == nPosY then
      return true
    end
  end
  return false
end

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  if not self:CheckNeedHuanHua(me, it.dwTemplateId) then
    return
  end
  if 1 ~= GetTimeFrameState("OpenLevel39") then
    return
  end
  local nItemType = KItem.GetItemExtParam(it.dwTemplateId, tbItem.EXT_PARAM_1)
  local tbBaseInfo = Item:GetItemBaseProp(it.dwTemplateId)
  local nRequireLevel = tbBaseInfo.nRequireLevel
  if nRequireLevel > me.nLevel then
    me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\233\152\133\229\142\134\229\176\154\230\181\133\239\188\140\233\156\128\232\166\129\231\173\137\231\186\167\232\190\190\229\136\176{0}\231\186\167", nRequireLevel))
    return
  end
  if nItemType == self.TYPE_SENIOR then
    local bAllReady, szTips = self:CheckTeamMember(it, true)
    if not bAllReady then
      me.CenterMsg(szTips)
      return
    end
  end
  if me.CheckNeedArrangeBag() then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\233\129\147\229\133\183\230\149\176\233\135\143\232\191\135\229\164\154\239\188\140\232\175\183\230\149\180\231\144\134\228\184\128\228\184\139\239\188\129"))
    return
  end
  local nMapTemplateId, nPosX, nPosY = self:GetCangBaoTuPos(it)
  if not self:IsIllegal(nItemType, nMapTemplateId, nPosX, nPosY, me) then
    self:RandomPos(it, me)
    nMapTemplateId, nPosX, nPosY = self:GetCangBaoTuPos(it)
  end
  local bRet = self:CheckCanWaBao(me, it.dwId)
  if not bRet then
    it.SetIntValue(self.PARAM_MAPID, nMapTemplateId)
    it.SetIntValue(self.PARAM_POSX, nPosX)
    it.SetIntValue(self.PARAM_POSY, nPosY)
    me.CallClientScript("CangBaoTu:OnUseItem", it.dwId, nMapTemplateId, nPosX, nPosY)
    return
  end
  local nItemtype = KItem.GetItemExtParam(it.dwTemplateId, tbItem.EXT_PARAM_1)
  if nItemtype == self.TYPE_SENIOR then
    local bAllReady, szTips = self:CheckTeamMember(it)
    if not bAllReady then
      me.CenterMsg(szTips)
      me.CallClientScript("Ui:OpenQuickUseItem", it.dwId, i18n.EasyGet("\228\189\191\231\148\168"))
      return
    end
  end
  me.CallClientScript("CangBaoTu:OnStartProgress")
  Progress:BeginProgress(me, i18n.EasyGet("\230\140\150\229\174\157\228\184\173"), self.tbSetting[nItemtype].nWaBaoTime, self.OnEndProgress, {
    self,
    it.dwId
  })
  return 0
end

function tbItem:CheckTeamMember(it, bCheckCommon)
  if not me.dwTeamID or 0 == me.dwTeamID then
    return false, i18n.EasyGet("\233\156\128{0}\228\186\186\228\187\165\228\184\138\231\187\132\233\152\159\229\137\141\229\190\128\230\140\150\229\174\157", self.nNeedTeamate)
  end
  local tbMember = TeamMgr:GetMembers(me.dwTeamID)
  if not next(tbMember) then
    return false, i18n.EasyGet("\233\156\128\232\166\129\233\152\159\229\143\139\229\141\143\229\138\169\230\137\141\232\131\189\230\140\150\229\174\157")
  end
  if Lib:CountTB(tbMember) < self.nNeedTeamate then
    return false, i18n.EasyGet("\233\156\128{0}\228\186\186\228\187\165\228\184\138\231\187\132\233\152\159\229\137\141\229\190\128\230\140\150\229\174\157", self.nNeedTeamate)
  end
  if bCheckCommon then
    return true
  end
  local nMapTemplateId = self:GetCangBaoTuPos(it)
  local nMapMemberCount = 0
  for _, dwID in pairs(tbMember) do
    local pPlayer = KPlayer.GetPlayerObjById(dwID)
    if pPlayer and pPlayer.nMapId == nMapTemplateId then
      nMapMemberCount = nMapMemberCount + 1
    end
  end
  if nMapMemberCount < Lib:CountTB(tbMember) then
    return false, i18n.EasyGet("\230\137\128\230\156\137\233\152\159\229\145\152\229\136\176\233\189\144\229\144\142\230\137\141\232\131\189\230\140\150\229\174\157")
  end
  return true
end

function tbItem:RandEvent(nItemType, pPlayer)
  local nEventType = self.EventType_Award
  local tbEventList = self.tbTrapEvents[nItemType]
  local nRand = MathRandom(10000)
  for _, tbInfo in ipairs(tbEventList) do
    nRand = nRand - tbInfo.nRate
    if nRand <= 0 then
      nEventType = tbInfo.nEventType
      break
    end
  end
  if nEventType == self.EventType_Box then
    local nDegree = DegreeCtrl:GetDegree(pPlayer, "CangBaoTuBoxRefresh")
    if nDegree <= 0 then
      nEventType = self.EventType_Award
    else
      DegreeCtrl:ReduceDegree(pPlayer, "CangBaoTuBoxRefresh", 1)
    end
  end
  if nEventType == self.EventType_DBZ then
    local nDegree = DegreeCtrl:GetDegree(pPlayer, "CangBaoTuDBZRefresh")
    if nDegree <= 0 then
      nEventType = self.EventType_Award
    else
      DegreeCtrl:ReduceDegree(pPlayer, "CangBaoTuDBZRefresh", 1)
    end
  end
  return nEventType
end
