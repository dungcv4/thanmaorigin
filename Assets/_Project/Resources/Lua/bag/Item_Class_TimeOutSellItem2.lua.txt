local tbItem = Item:GetClass("TimeOutSellItem2")
tbItem.PARAM_AWARD_ID = 1
tbItem.PARAM_AWARD_KEY = 2

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  local nAwardId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARD_ID)
  local nAwardKey = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_AWARD_KEY)
  local tbAwardStudio = StudioRandomItem:GetStudioAward(me, it.dwTemplateId)
  if tbAwardStudio then
    nAwardId = tbAwardStudio.nAwardId
    nAwardKey = tbAwardStudio.nAwardKey
  end
  local bTimeOut, nGenTime, nDynamicTimeOut = Item:GetTimeOut(it)
  if bTimeOut and nDynamicTimeOut < GetTime() then
    me.CenterMsg(i18n.EasyGet("\229\183\178\229\164\177\230\149\136"))
    return 0
  end
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  local nNum, nRet = 0
  local szMsg = it.szName
  local nSuccessRet = 1
  local tbAwardListForTrack, tbLastAwards = {}
  for _ = 1, nCount do
    nRet = Award:CommonSendAward(me, nAwardId, nAwardKey, it.dwTemplateId, szMsg, {}, true)
    if nSuccessRet == nRet then
      nNum = nNum + 1
      tbLastAwards = Award:FetchLastAwards()
      Lib:MergeTable(tbAwardListForTrack, tbLastAwards)
    end
  end
  local nConsumeItemTemplateId = it.dwTemplateId
  local nExpendItemId = 0
  local nExpendItemNum = 0
  tbAwardListForTrack = Award:MergetSameTypeAward(tbAwardListForTrack)
  local tbMoneyForTrack, tbItemForTrack = Award:GetAwardListForMBILog(tbAwardListForTrack, true, {
    "Gold",
    "YuanBao",
    "Coin"
  })
  Lib:TrackEvent(me.dwID, Env.track_openbox_open, nConsumeItemTemplateId, nCount, nExpendItemId, nExpendItemNum * nCount, tbMoneyForTrack.Gold or 0, tbMoneyForTrack.YuanBao or 0, tbMoneyForTrack.Coin or 0, Lib:EncodeJson(tbItemForTrack) or "")
  return nNum
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "TimeOutSellItem2")
end
