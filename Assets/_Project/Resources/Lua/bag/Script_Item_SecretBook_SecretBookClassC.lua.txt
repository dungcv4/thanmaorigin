Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
local SecretBook = Item.SecretBook
local tbSecretBook = Item:GetClass(SecretBook.szClass)

function tbSecretBook:InnerGetTips(nRank, nSecretBookLevel, tbAttributeValue, nSkillLevel, nFaction, nFactionSect, nExtraLevel)
  nExtraLevel = nExtraLevel or 0
  local nAllSkillLevel = nSkillLevel + nExtraLevel
  local szSkillLevel = tostring(nSkillLevel)
  if nExtraLevel > 0 then
    szSkillLevel = string.format("%d\239\188\136%d+%d\239\188\137", nAllSkillLevel, nSkillLevel, nExtraLevel)
  end
  local tbRet = {}
  table.insert(tbRet, {
    i18n.Get(SecretBook.tbText.szSecretBookLevelShow, nSecretBookLevel, SecretBook.Cfg.nMaxLevel)
  })
  table.insert(tbRet, {
    i18n.Get(SecretBook.tbText.szSecretBookAttributeText)
  })
  for nAttributeIndex, nAttributeValue in ipairs(tbAttributeValue) do
    local szMagicAttributeIdentifier = SecretBook:GetMagicAttributeNameByFactionSect(nAttributeIndex, nFaction, nFactionSect)
    local szAttributeText = FightSkill:GetMagicDesc(szMagicAttributeIdentifier, {nAttributeValue})
    table.insert(tbRet, {
      {szAttributeText}
    })
  end
  table.insert(tbRet, {
    i18n.Get(SecretBook.tbText.szSecretBookSkillText)
  })
  local nSkillId = SecretBook:GetSkillId(nFaction, nFactionSect, nRank)
  local tbSkillInfo = FightSkill.tbAllSkillSetting[nSkillId]
  local szSkillName = i18n.Parse(tbSkillInfo.szSkillName)
  local szSkillDesc = i18n.Parse(tbSkillInfo.szDesc)
  table.insert(tbRet, {
    nil,
    nil,
    nil,
    nil,
    i18n.Get(SecretBook.tbText.szSecretBookSkillDesc, szSkillName, szSkillLevel, SecretBook.Cfg.nMaxSkillLevel, szSkillDesc)
  })
  local szNowLevelDesc = FightSkill:GetSkillMagicDesc(nSkillId, nAllSkillLevel)
  local szMaxLevelDesc = FightSkill:GetSkillMagicDesc(nSkillId, SecretBook.Cfg.nMaxSkillLevel)
  local szLevelCompareText = i18n.Get(SecretBook.tbText.szSecretBookLevelCompareDesc, szSkillLevel, szNowLevelDesc, SecretBook.Cfg.nMaxSkillLevel, szMaxLevelDesc)
  table.insert(tbRet, {
    nil,
    nil,
    nil,
    nil,
    szLevelCompareText
  })
  return tbRet
end

function tbSecretBook:GetTip(pEquip, pPlayer, bIsCompare, tbItemExtParams)
  if SecretBook:IsSecretBook(pEquip) then
    local nRank = SecretBook:GetRank(pEquip)
    local nSecretBookLevel = SecretBook:GetLevel(pEquip)
    local tbAttributeValue = SecretBook:GetAllAttributeValue(pEquip)
    local nSkillLevel = SecretBook:GetSkillLevel(pEquip)
    local nExtraLevel = SecretBook:GetExtraSkillLevel(pPlayer, nRank)
    local nFaction = not bIsCompare and tbItemExtParams and tbItemExtParams.nFaction or pPlayer.nFaction
    local nFactionSect = not bIsCompare and tbItemExtParams and tbItemExtParams.nFactionSect or pPlayer.nFactionSect
    return self:InnerGetTips(nRank, nSecretBookLevel, tbAttributeValue, nSkillLevel, nFaction, nFactionSect, nExtraLevel)
  else
    LogErr("tbSecretBook:GetTip, arg pEquip is not SecretBook")
    return nil
  end
end

function tbSecretBook:GetTipByTemplate(nTemplateId, tbLinkInfo)
  if nil ~= tbLinkInfo then
    return self:InnerGetTips(tbLinkInfo.nRank, tbLinkInfo.nSecretBookLevel, tbLinkInfo.tbAttributeValue, tbLinkInfo.nSkillLevel, tbLinkInfo.nFaction, tbLinkInfo.nFactionSect)
  else
    return {}
  end
end

function tbSecretBook:GetChatLinkInfo(pEquip)
  if SecretBook:IsSecretBook(pEquip) then
    local tbLinkInfo = {}
    tbLinkInfo.nRank = SecretBook:GetRank(pEquip)
    tbLinkInfo.nSecretBookLevel = SecretBook:GetLevel(pEquip)
    tbLinkInfo.tbAttributeValue = SecretBook:GetAllAttributeValue(pEquip)
    tbLinkInfo.nSkillLevel = SecretBook:GetSkillLevel(pEquip)
    return tbLinkInfo
  else
    return nil
  end
end
