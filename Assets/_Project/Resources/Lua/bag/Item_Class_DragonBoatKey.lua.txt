Require("CommonScript/Item/ItemDefine.lua")
local tbDragonBoatKey = Item:GetClass("DragonBoatKey")

function tbDragonBoatKey:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "DragonBoatKey")
end

function tbDragonBoatKey:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    if Ui:WindowVisible("UIItemTips") then
      Ui:CloseWindow("UIItemTips")
    end
    Ui:OpenWindow("UIDuanWuGift")
  end
  
  return tbSetting
end
