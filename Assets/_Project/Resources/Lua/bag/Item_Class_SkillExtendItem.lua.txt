local tbSkillExtendItem = Item:GetClass("SkillExtendItem")
local tbTemplatedIdLinkLevelUpGroup = {
  [2060] = {6, 7},
  [2061] = {8, 9},
  [2062] = {10, 11},
  [2063] = {12, 1},
  [2064] = {2, 3},
  [2065] = {4, 5}
}

function tbSkillExtendItem:UseItem()
  Ui:OpenWindow("UISkill")
  Ui:CloseWindow("UIItemTips")
  Ui:CloseWindow("UIBag")
  return 0
end

function tbSkillExtendItem:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    tbSkillExtendItem:UseItem()
  end
  
  return tbSetting
end

function tbSkillExtendItem:GetTip(pItem)
  if not tbTemplatedIdLinkLevelUpGroup[pItem.dwTemplateId] then
    return ""
  end
  local nSkillId1 = FightSkill:GetSkillIdByLevelUpGroup(me, tbTemplatedIdLinkLevelUpGroup[pItem.dwTemplateId][1])
  local nSkillId2 = FightSkill:GetSkillIdByLevelUpGroup(me, tbTemplatedIdLinkLevelUpGroup[pItem.dwTemplateId][2])
  local szTip1 = self:GetTipsBySkillId(nSkillId1)
  local szTip2 = self:GetTipsBySkillId(nSkillId2)
  return szTip1 .. "\n" .. szTip2
end

function tbSkillExtendItem:GetTipsBySkillId(nSkillId)
  local nGroupId = FightSkill:GetSkillLevelUpGroupId(nSkillId)
  local nMaxLevel = FightSkill:GetSkillLevelExtendLimit(nSkillId)
  local nMyBreakLevel = FightSkill:GetSkillMaxLevel(me, nSkillId)
  local nDefaultMaxLevel = FightSkill:GetDefaultSkillMaxLevel(nGroupId)
  local nGainLevel = FightSkill:GetSkillGainLevel(nSkillId)
  nGainLevel = nGainLevel <= 0 and 1 or nGainLevel
  local szDesc = i18n.EasyGet("{0}\231\186\167\230\138\128\232\131\189\229\183\178\228\189\191\231\148\168\230\172\161\230\149\176\239\188\154{1}/{2}", nGainLevel, nMyBreakLevel - nDefaultMaxLevel, nMaxLevel - nDefaultMaxLevel)
  return szDesc
end
