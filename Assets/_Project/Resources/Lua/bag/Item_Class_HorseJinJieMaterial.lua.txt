local tbItemClass = Item:GetClass("HorseJinJieMaterial")

function tbItemClass:OnClientUse(pItem)
  local bRet, szMsg = Horse:IsHorseOpen(me)
  if not bRet then
    me.CenterMsg(szMsg, false)
    return -1
  end
  local nCurGrade = Horse:GetHorseGrade(me)
  local nMaterialGrade = KItem.GetItemExtParam(pItem.dwTemplateId, 1)
  if nCurGrade >= nMaterialGrade then
    me.CenterMsg(i18n.EasyGet("\233\152\182\230\149\176\229\183\178\232\182\133\232\191\135"), false)
    return -1
  end
  Ui:OpenWindow("UIHorseJinjie", nMaterialGrade)
  return 0
end

function tbItemClass:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function fnClientUse()
    Item:ClientUseItem(nItemId)
    Ui:CloseWindow("UIItemTips")
  end
  
  local function fnExchange()
    Horse:ExchangeJinJieMaterialReq(nItemId)
    Ui:CloseWindow("UIItemTips")
  end
  
  local tbSetting = {}
  if nItemId then
    tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
    tbSetting.fnFirst = fnClientUse
    if Horse:CanJinJieMaterialExchange(me, nItemId) then
      tbSetting.szSecondName = i18n.EasyGet("\229\133\145\230\141\162")
      tbSetting.fnSecond = fnExchange
    end
  end
  return tbSetting
end
