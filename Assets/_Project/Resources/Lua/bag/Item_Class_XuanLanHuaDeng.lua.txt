tbItem.tbMapClassList = {"city", "tong"}

function tbItem:OnUse(it)
  if 0 == me.dwKinId and 0 == me.dwTongId then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\138\160\229\133\165\229\174\182\230\151\143\230\136\150\229\184\174\228\188\154\229\144\142\228\189\191\231\148\168"))
    return 0
  end
  local bIsCanUse = false
  local szMeMapClass = Map:GetMapSetting(me.nMapTemplateId).Class
  for _, szMapClass in ipairs(self.tbMapClassList) do
    if szMeMapClass == szMapClass then
      bIsCanUse = true
      break
    end
  end
  if false == bIsCanUse then
    me.CenterMsg(i18n.EasyGet("\230\151\160\230\179\149\229\156\168\230\173\164\229\156\176\229\155\190\228\189\191\231\148\168\239\188\140\232\175\183\229\137\141\229\190\128\228\184\187\229\159\142\230\136\150\229\184\174\228\188\154\233\162\134\229\156\176"))
    return 0
  end
  local nRandomMax = #self.tbTextList
  local nRandomIndex = math.random(1, nRandomMax)
  if 0 ~= me.dwTongId then
    ChatMgr:SendSystemMsg(ChatMgr.SystemMsgType.Tong, self.tbTextList[nRandomIndex](me.szName), me.dwTongId)
  elseif 0 ~= me.dwKinId then
    ChatMgr:SendSystemMsg(ChatMgr.SystemMsgType.Kin, self.tbTextList[nRandomIndex](me.szName), me.dwKinId)
  end
  local _, nPosX, nPosY = me.GetNpc().GetWorldPos()
  me.GetNpc().CastSkill(10368, 1, nPosX, nPosY)
  me.CallClientScript("Ui:CloseWindow", "UIItemTips")
  me.CallClientScript("Ui:CloseWindow", "UIBag")
  return 1
end
