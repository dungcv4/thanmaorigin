local tbItem = Item:GetClass("PartnerExpItem")

function tbItem:OnUse(pItem)
end

function tbItem:GetExpInfo(nItemId, nTemplateId)
  if 0 ~= nItemId then
    local pItem = KItem.GetItemObj(nItemId)
    if not pItem or pItem.szClass ~= "PartnerExpItem" then
      return 0
    end
    nTemplateId = pItem.dwTemplateId
  end
  local tbBaseInfo = Item:GetItemBaseProp(nTemplateId)
  if not tbBaseInfo or tbBaseInfo.szClass ~= "PartnerExpItem" then
    return 0
  end
  local nBaseExp = tbBaseInfo.nValue * Partner.nExpItemValue2PartnerExp
  return nBaseExp
end

function tbItem:OnClientUse(pItem)
  Partner:OpenPartnerMainUI(1)
  return 0
end
