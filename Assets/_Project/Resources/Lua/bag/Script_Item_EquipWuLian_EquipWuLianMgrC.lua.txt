local EquipWuLian = Item.EquipWuLian
EquipWuLian.tbWuLianLevelData = EquipWuLian.tbWuLianLevelData or {}

function EquipWuLian:OnLoginC()
  self:LoadFightPowerSetting()
end

function EquipWuLian:WuLianReqC(pPlayer, nPos, tbItemList)
  local bRet, szErr = self:CanWuLianWithReminderWindow(pPlayer, nPos, tbItemList)
  if not bRet then
    pPlayer.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestEquipWuLian("WuLianReqS", nPos, tbItemList)
end

function EquipWuLian:WuLianRsp(bSuccess, nPos, nOrgLv, nNewLv)
  if bSuccess then
    self:RefreshAttrib(me, nPos, nOrgLv, nNewLv)
  end
  if Ui:WindowVisible("UIEquipment") then
    Ui:GetWindow("UIEquipment"):NotifyEquipEnhance("WuLian", bSuccess, nPos)
  end
  Item.EquipEnhance:OnCheckRedPoint()
end

function EquipWuLian:QuickCheckWuLian(pPlayer, pEquip)
  local nHasVal = 0
  for _, nItemTid in pairs(Item.EquipEnhance.XUANJING_ITEM) do
    local nNum = pPlayer.GetItemCountInBags(nItemTid)
    if 0 ~= nNum then
      nHasVal = nHasVal + KItem.GetBaseValue(nItemTid) * nNum
    end
  end
  local nHasMoney = pPlayer.GetMoney(Item.EquipEnhance.MONEY_TYPE)
  local nReqVal, nReqMoney = self:CalcWuLianCost(pPlayer, pEquip, self:GetEquipWuLianLevel(pPlayer, pEquip.nEquipPos))
  if nReqVal and nReqMoney and nHasVal >= nReqVal and nHasMoney >= nReqMoney then
    return true
  else
    return false
  end
end

function EquipWuLian:CanShowWuLianButtonInUIEquipTips(nItemId)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return false
  end
  local nEquipPos = pItem.nEquipPos
  if nEquipPos < Item.EQUIPPOS_HELM or nEquipPos > Item.EQUIPPOS_AMULET then
    return false
  end
  return true
end

function EquipWuLian:IsWuLianOpenC(nEnhanceLv, nEquipLevel)
  if nEnhanceLv < Item.EquipEnhance.nMaxLevel then
    return false
  end
  if nEquipLevel < self.WULIAN_LV[1].LV_REQ then
    return false
  end
  if 1 ~= GetTimeFrameState(self.WULIAN_LV[1].TIMEFRAME_REQ) then
    return false
  end
  return true
end
