local tbPotentialPointItem = Item:GetClass("PotentialPointItem")

function tbPotentialPointItem:OnCreate()
end

function tbPotentialPointItem:OnClientUse(pItem)
  local nTemplateId = pItem.dwTemplateId
  local nMaxUseCount = KItem.GetItemExtParam(nTemplateId, 1)
  local szGroupKey = KItem.GetItemExtParam(nTemplateId, 3)
  local nUseCount = me.GetUserValue(szGroupKey, 1)
  Ui:CloseWindow("UIItemTips")
  if nMaxUseCount <= nUseCount then
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\230\172\161\230\149\176\229\183\178\232\190\190\228\184\138\233\153\144"))
    return 0
  else
    return 1
  end
end

function tbPotentialPointItem:OnUse(pItem)
  local a = Item:GetIntro(pItem.dwTemplateId)
  local nTemplateId = pItem.dwTemplateId
  local nMaxUseCount = KItem.GetItemExtParam(nTemplateId, 1)
  local szGroupKey = KItem.GetItemExtParam(nTemplateId, 3)
  local nUseCount = me.GetUserValue(szGroupKey, 1)
  if nMaxUseCount <= nUseCount then
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\230\172\161\230\149\176\229\183\178\232\190\190\228\184\138\233\153\144"))
    return 0
  else
    local nAddNum = KItem.GetItemExtParam(nTemplateId, 2)
    me.CenterMsg(i18n.EasyGet("\230\176\184\228\185\133\229\162\158\229\138\160{0}\231\130\185\230\189\156\232\131\189\231\130\185", nAddNum))
    me.AddPotentialPoint(nAddNum)
    me.SetUserValue(szGroupKey, 1, nUseCount + 1)
    return 1
  end
end

function tbPotentialPointItem:GetTip(pItem)
  local nTemplateId = pItem.dwTemplateId
  local nMaxUseCount = KItem.GetItemExtParam(nTemplateId, 1)
  local szGroupKey = KItem.GetItemExtParam(nTemplateId, 3)
  local nUseCount = me.GetUserValue(szGroupKey, 1)
  local szUseCount = i18n.EasyGet("\229\183\178\228\189\191\231\148\168\230\172\161\230\149\176\239\188\154{0}/{1}", nUseCount, nMaxUseCount)
  return szUseCount
end
