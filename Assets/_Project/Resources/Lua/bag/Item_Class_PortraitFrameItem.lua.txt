local tbPortraitFrameItem = Item:GetClass("PortraitFrame")
tbPortraitFrameItem.PARAM_FRAMEID = 1
tbPortraitFrameItem.PARAM_FRAMETIMEOUT = 2

function tbPortraitFrameItem:OnUse(item)
  local nResult = self:UsePortraitFrameItem(me, item.dwTemplateId, 1)
  return nResult
end

function tbPortraitFrameItem:UsePortraitFrameItem(pPlayer, nTemplateId, nCount)
  local nNum = pPlayer.GetItemCountByTypeInBag(nTemplateId)
  if nNum <= 0 then
    pPlayer.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183{0}\230\149\176\233\135\143\228\184\141\232\182\179", nTemplateId))
    return 0
  end
  local nPortraitFrameID = KItem.GetItemExtParam(nTemplateId, self.PARAM_FRAMEID)
  if not PlayerPortraitFrame:CheckPFIdExist(nPortraitFrameID) then
    pPlayer.CenterMsg(i18n.EasyGet("\232\175\165\229\164\180\229\131\143\230\161\134\228\184\141\229\173\152\229\156\168\239\188\129"))
    return 0
  end
  local nLimitTime = KItem.GetItemExtParam(nTemplateId, self.PARAM_FRAMETIMEOUT)
  return PlayerPortraitFrame:AddPortraitFrame(pPlayer, nPortraitFrameID, nLimitTime)
end
