local tbTitleItem = Item:GetClass("TitleItem")
tbTitleItem.PARAM_TITLEID = 1
tbTitleItem.PARAM_TITLETTIMEOUT = 2

function tbTitleItem:OnUse(item)
  local nResult = self:UseTitleItem(me, item.dwTemplateId, 1, false, item.dwId)
  return nResult
end

function tbTitleItem:OnAutoUse(pPlayer, nTemplateId, nCount)
  local nResult = self:UseTitleItem(pPlayer, nTemplateId, nCount, true)
  return nResult
end

function tbTitleItem:UseTitleItem(pPlayer, nTemplateId, nCount, bAutoUse, nItemId)
  local nNum = pPlayer.GetItemCountByTypeInBag(nTemplateId)
  if not bAutoUse and nNum <= 0 then
    pPlayer.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183{0}\230\149\176\233\135\143\228\184\141\232\182\179", nTemplateId))
    return 0
  end
  local nTitleID = KItem.GetItemExtParam(nTemplateId, self.PARAM_TITLEID)
  local tbTitleInfo = PlayerTitle:GetTitleTemplate(nTitleID)
  if not tbTitleInfo then
    pPlayer.CenterMsg(i18n.EasyGet("\231\167\176\229\143\183{0}\228\184\141\229\173\152\229\156\168", nTitleID))
    return 0
  end
  if not bAutoUse and PlayerTitle:HasTitleActivated(pPlayer, nTitleID) then
    pPlayer.MsgBox(i18n.EasyGet("\230\130\168\229\183\178\230\191\128\230\180\187\231\167\176\229\143\183\227\128\144{0}\227\128\145\239\188\140\229\134\141\230\172\161\230\191\128\230\180\187\229\144\142\231\167\176\229\143\183\230\151\182\230\149\136\230\151\160\230\179\149\229\143\160\229\138\160\239\188\140\230\152\175\229\144\166\231\161\174\232\174\164\228\189\191\231\148\168\239\188\159", tbTitleInfo.Name), {
      {
        i18n.szOk,
        self.OnActivateTittle,
        self,
        pPlayer.dwID,
        nTemplateId,
        nItemId
      },
      {
        i18n.szCancel
      }
    })
    return 0
  end
  self:OnActivateTittle(pPlayer.dwID, nTemplateId, nItemId)
  return 0
end

function tbTitleItem:OnActivateTittle(dwPlayerId, nTemplateId, nItemId)
  local pPlayer = KPlayer.GetPlayerObjById(dwPlayerId)
  if not pPlayer then
    return
  end
  local nTitleID = KItem.GetItemExtParam(nTemplateId, self.PARAM_TITLEID)
  local nTitleTimeout = KItem.GetItemExtParam(nTemplateId, self.PARAM_TITLETTIMEOUT)
  local tbTitleInfo = PlayerTitle:GetTitleTemplate(nTitleID)
  if nTitleTimeout > 0 then
    nTitleTimeout = GetTime() + nTitleTimeout
  end
  if nItemId then
    local pItem = KItem.GetItemObj(nItemId)
    assert(pItem)
    assert(1 == pPlayer.ConsumeItem(pItem, 1, Env.LogWay_UseItem, false))
  end
  local bRet = PlayerTitle:AddTitle(pPlayer, nTitleID, (nTitleTimeout > 0 and {nTitleTimeout} or {nil})[1])
  if not bRet then
    pPlayer.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\230\191\128\230\180\187\231\167\176\229\143\183{0}", tbTitleInfo.Name))
    return 0
  end
  pPlayer.CallClientScript("PlayerTitle:OpenTitleWithTitleID", nTitleID)
end
