Require("CommonScript/Item/ItemDefine.lua")
local tbZongziMaterial = Item:GetClass("ZongziMaterial")

function tbZongziMaterial:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  local nActivityID = 13
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    Ui:OpenWindow("UIBaoZongZi", nActivityID)
  end
  
  return tbSetting
end
