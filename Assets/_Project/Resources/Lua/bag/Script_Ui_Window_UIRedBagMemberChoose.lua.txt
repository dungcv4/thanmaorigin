local tbWnd = Ui:CreateWindow("UIRedBagMemberChoose")

function tbWnd:NotifyEvents()
  return {}
end

tbWnd.tbControls = {
  memberItem = "imgBG/imgList/List/Template",
  imgList = "imgBG/imgList",
  NoMember = "imgBG/NoMember",
  btnConfirm = "imgBG/btnConfirm",
  ToggleSelectAll = "imgBG/ToggleSelectAll",
  btnClose = "imgBG/btnClose"
}
tbWnd.tbBuilding2Txt = {
  [Kin.Def.Building_Main] = "node/imgBG/KinInfoPanel/Building/Zhudian/LevelNum"
}
tbWnd.tbMemberSortFunc = {}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnConfirm = function(self)
    self:ConfirmClick()
  end
}

function tbWnd:OnCreate()
  self.tbMemberSelectScrollView = Ui:NewScrollView(self, self.memberItem)
end

function tbWnd:OnOpen(WindowType)
  if not WindowType then
    return
  end
  self.WindowType = WindowType
  self:SetJobTable()
  self:UpdateShowList()
  self:refresh()
  Ui:Toggle_BindEvent(self.pPanel, self.ToggleSelectAll, function(self, isOn)
    self:SetSelectMemberAll(isOn)
  end, self)
  self.pPanel:Toggle_SetChecked(self.ToggleSelectAll, false)
end

function tbWnd:refresh()
  self:ShowMemberInfo()
end

function tbWnd:SetJobTable()
  self.tbJobName = {}
  if self.WindowType == "Kin" then
    self.tbJobName = Kin.Def.Career_Name
  else
    self.tbJobName = Tong.Def.CareerName
  end
end

function tbWnd:SetSelectMemberAll(showAll)
  self.tbMemberList = self.tbMemberList or {}
  for i = 1, #self.tbMemberList do
    self.pPanel:Toggle_SetChecked(string.format("%s/List/Element%d/togglePush", self.imgList, i), showAll)
  end
end

function tbWnd:ConfirmClick()
  self.tbMemberList = self.tbMemberList or {}
  self.selectMemberList = {}
  for i = 1, #self.tbMemberList do
    if self.pPanel:Toggle_GetChecked(string.format("%s/List/Element%d/togglePush", self.imgList, i)) == true then
      table.insert(self.selectMemberList, self.tbMemberList[i])
    end
  end
  if self.WindowType == "Kin" then
    local UIKin = Ui:GetClass("UIKin")
    if UIKin then
      UIKin:UpdateSelectInfo(self.selectMemberList)
    end
  else
    local UITong = Ui:GetClass("UITong")
    if UITong then
      UITong:UpdateSelectInfo(self.selectMemberList)
    end
  end
  self:Close()
end

function tbWnd:ShowMemberInfo()
  self.tbMemberList = self:GetShowList() or {}
  if not self.tbMemberList then
    print("tbMemberList is nil")
  else
    print("tbMemberSortFunc not nil ")
  end
  self.pPanel:SetActive(self.NoMember, 0 == #self.tbMemberList)
  self.tbMemberSelectScrollView:Clear()
  for i = 1, #self.tbMemberList do
    self.tbMemberSelectScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtName", self.tbMemberList[i].szName)
      Ui:SetVipLevelEffect(pPanel, self.tbMemberList[i].nVipLevel, "txtName")
      pPanel:Label_SetText("txtJob", i18n.Get(self.tbJobName[self.tbMemberList[i].nCareer]) or "")
      pPanel:Label_SetText("txtHistory", self.tbMemberList[i].nTotalContribution or 0)
      if 1 == self.tbMemberList[i].nIsOnLine then
        pPanel:Label_SetText("txtOffline", i18n.EasyGet("\229\156\168\231\186\191"))
      else
        pPanel:Label_SetText("txtOffline", "<color=#352620>" .. i18n.EasyGet("\231\166\187\231\186\191") .. "</color>")
      end
      if self.SelectMemberMap[self.tbMemberList[i].nMemberId] then
        print("togglePush true")
        pPanel:Toggle_SetChecked("togglePush", true)
      else
        pPanel:Toggle_SetChecked("togglePush", false)
      end
    end)
  end
end

function tbWnd:GetShowList()
  if self.WindowType == "Kin" then
    local MemberList = Kin:GetMemberList() or {}
    return self:ExceptMeShowList(MemberList)
  else
    local MemberList = Tong:GetMemberList() or {}
    return self:ExceptMeShowList(MemberList)
  end
end

function tbWnd:ExceptMeShowList(MemberList)
  local data = {}
  for k, v in pairs(MemberList) do
    if v.nMemberId ~= me.dwID then
      table.insert(data, v)
    end
  end
  return data
end

function tbWnd:UpdateShowList()
  if self.WindowType == "Kin" then
    Kin:UpdateMemberList()
  else
    Tong:UpdateMemberList()
  end
  self:GetSelectMemberMap()
end

function tbWnd:GetSelectMemberMap()
  self.SelectMemberMap = {}
  if self.WindowType == "Kin" then
    local UIKin = Ui:GetClass("UIKin")
    if UIKin then
      self.selectMemberList = UIKin:GetSelectInfo()
    end
  else
    local UITong = Ui:GetClass("UITong")
    if UITong then
      self.selectMemberList = UITong:GetSelectInfo()
    end
  end
  self.selectMemberList = self.selectMemberList or {}
  for k, v in pairs(self.selectMemberList) do
    print("v.nMemberId", v.nMemberId)
    self.SelectMemberMap[v.nMemberId] = true
  end
end
