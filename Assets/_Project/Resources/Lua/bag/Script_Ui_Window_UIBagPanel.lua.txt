    self:UpdateItemList()

function tbWnd:UpdateBagCount(nCurItemKind)
  local nMaxCount = me.GetMaxCountInBag()
  self.pPanel:Label_SetText(tbControls.txtItemCount, string.format("%d/%d", self.nAllItemCount or 0, nMaxCount or 0))
  if not Item.ZhenYuan:IsSystemOpen(me) and nMaxCount >= Item.MAX_BAG_EXPAND then
    self.pPanel:SetActive(self.btnAdd, false)
  elseif Item.ZhenYuan:IsSystemOpen(me) and nMaxCount >= Item.MAX_BAG_EXPAND + Item.ZhenYuan.nBagMaxCnt then
    self.pPanel:SetActive(self.btnAdd, false)
  else
    self.pPanel:SetActive(self.btnAdd, true)
  end
  
  local function fnCallBack()
    if 1 == nCurItemKind and self.nAllItemCount >= me.GetMaxCountInBag() then
      me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\231\154\132\232\131\140\229\140\133\230\187\161\228\186\134\239\188\140\230\149\180\231\144\134\228\184\128\228\184\139\229\144\167"))
      Timer:Close(self.nMsgTimer)
      self.nMsgTimer = nil
    end
  end
  
  if not self.nMsgTimer then
    self.nMsgTimer = Timer:Register(Env.GAME_FPS, fnCallBack, self)
  end
end

function tbWnd:UpdateItemList()
  local tbAllItemBagList = Bag:GetItemBagList(tbToggle.TogAll)
  if nil ~= tbAllItemBagList then
    self.nAllItemCount = Lib:CountTB(tbAllItemBagList)
  else
    self.nAllItemCount = 0
  end
  local nCurItemKind = self.tbItemKindGroup:GetKey()
  self.nCurItemKind = nCurItemKind
  self.tbItemList = Bag:GetItemBagList(nCurItemKind)
  self:UpdateItemSV()
  self:UpdateBagCount(nCurItemKind)
end

function tbWnd:UpdateItemSV(tbSearchResult)
  local nCount = 0
  if nil == tbSearchResult and nil ~= self.tbItemList then
    tbSearchResult = self.tbItemList
  end
  tbSearchResult = tbSearchResult or {}
  nCount = Lib:CountTB(tbSearchResult)
  local nMaxCount = me.GetMaxCountInBag()
  
  local function fnSetItem(tbLoopScroll, nIndex)
    local nCol = nMaxCount - self.nRow * (nIndex - 1)
    if nCol > self.nRow then
      nCol = self.nRow
    end
    for i = 1, self.nRow do
      if i <= nCol then
        local itemObj = tbLoopScroll:GetGridElement(nIndex, i)
        local curIndex = (nIndex - 1) * self.nRow + i
        itemObj.pPanel:SetActive("Main", true)
        local itemGrid = Ui:NewExistItemGrid(itemObj, "UIItemGrid")
        self:SaveItemGrid(nIndex, i, itemGrid)
        if curIndex <= nCount then
          itemGrid:SetActive(true)
          local tbItem = tbSearchResult[curIndex]
          local nItemId = tbItem and tbItem.nItemId
          local nShowCount = tbItem and tbItem.nShowCount
          itemGrid.szItemOpt = "ItemBox"
          itemGrid:SetItem(nItemId, nil, {
            bShowTip = true,
            bShowSeries = self.bShowSeries
          }, nil, nil, nil, nil, nShowCount)
          itemGrid:SetClickEnable(true)
        else
          itemGrid:SetEmpty()
          itemGrid:SetActive(false)
        end
      else
        tbLoopScroll:GetGridElement(nIndex, i).pPanel:SetActive("Main", false)
      end
    end
  end
  
  self.tbBagScrollView:SetElement(math.ceil(nMaxCount / self.nRow), fnSetItem)
end

function tbWnd:OnSyncItem(nItemId, bUpdateAll)
  if self.bIsWndOpen then
    self:UpdateEquip()
    if 1 == bUpdateAll then
      Bag:ClearDirtyBagItemData()
      self:UpdateAllItem()
    else
      local pItem = me.GetItemInAllSpace(nItemId)
      if Item.TB_MEDICINE_LIST[pItem.dwTemplateId] == tostring(pItem.dwTemplateId) then
        return
      end
      if self.nCurItemKind == tbToggle.TogNormal and 0 == tbRefreshCondition[self.nCurItemKind][pItem.szClass] then
        return
      end
      if self.nCurItemKind == tbToggle.TogNormal or self.nCurItemKind == tbToggle.TogAll or self.nCurItemKind == tbToggle.TogStone or 1 == tbRefreshCondition[self.nCurItemKind][pItem.szClass] or self.nCurItemKind == tbToggle.TogPartner then
        Bag:ClearDirtyBagItemData()
        self:UpdateAllItem()
      end
    end
  end
end

function tbWnd:SaveItemGrid(nRow, nCol, itemGrid)
  self.tbItemGridSave = self.tbItemGridSave or {}
  self.tbItemGridSave[nRow] = self.tbItemGridSave[nRow] or {}
  self.tbItemGridSave[nRow][nCol] = itemGrid
end

function tbWnd:GetItemGrid(nRow, nCol)
  if self.tbItemGridSave ~= nil and self.tbItemGridSave[nRow] ~= nil then
    return self.tbItemGridSave[nRow][nCol]
  else
    return nil
  end
end

function tbWnd:UpdateOneItem(nItemId)
  local nCount = Lib:CountTB(self.tbItemList)
  local nMaxCount = me.GetMaxCountInBag()
  local nNum = math.ceil(nMaxCount / self.nRow)
  for nIndex = 1, nNum do
    for i = 1, self.nRow do
      local curIndex = (nIndex - 1) * self.nRow + i
      if nCount < curIndex then
        return
      end
      if self.tbItemList[curIndex].nItemId == nItemId then
        local itemObj = self.tbBagScrollView:GetGridElement(nIndex, i)
        if itemObj then
          local itemGrid = Ui:NewExistItemGrid(itemObj, "UIItemGrid")
          itemGrid:SetActive(true)
          local tbItem = self.tbItemList[curIndex]
          local nItemId = tbItem and tbItem.nItemId
          local nShowCount = tbItem and tbItem.nShowCount
          itemGrid.szItemOpt = "ItemBox"
          itemGrid:SetItem(nItemId, nil, {
            bShowTip = true,
            bShowSeries = self.bShowSeries
          }, nil, nil, nil, nil, nShowCount)
          itemGrid:SetClickEnable(true)
        end
        return
      end
    end
  end
end

function tbWnd:SwipeModel(nOffset)
  self.nSwipeOffset = self.nSwipeOffset or 0
  self.nSwipeOffset = self.nSwipeOffset + nOffset * 0.1
  local nDir = (self.nNpcModelDir - self.nSwipeOffset) % 8
  if nDir == self.nNpcModelDir then
    return
  end
  self.nNpcModelDir = nDir
  self.nSwipeOffset = 0
  if self.NpcModel then
    self.NpcModel:SetDir(math.floor(self.nNpcModelDir))
  end
end

function tbWnd:FindBagItemGridByTemplateID(nTemplateId)
  local nTargetIndex
  for nIndex, value in ipairs(self.tbItemList) do
    if value.dwTemplateId == nTemplateId then
      nTargetIndex = nIndex
      break
    end
  end
  if nil ~= nTargetIndex then
    local nRow = math.floor((nTargetIndex - 1) / self.nRow) + 1
    local nCol = (nTargetIndex - 1) % self.nRow + 1
    return self:GetItemGrid(nRow, nCol)
  else
    return nil
  end
end

function tbWnd:OnLateUpdate()
  if self.bNeedUpdateEquip then
    self:DoUpdateEquip()
    self.bNeedUpdateEquip = false
  end
  if self.bNeedOnSyncItem then
    self:OnSyncItem(self.nTempItemId, self.bTempUpdateAll)
    self.bNeedOnSyncItem = false
  end
end

function tbWnd:DoUpdateEquip()
  self.tbEquipPanel:Reset()
  local tbEquip = me.GetEquips(1)
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    if 0 ~= tbEquip[i - 1] then
      self.tbEquipPanel:SetEquip(i, tbEquip[i - 1])
    end
  end
  if 0 ~= tbEquip[Item.EQUIPPOS_GUANYIN] then
    self.tbEquipPanel:SetEquip(Item.EQUIPPOS_GUANYIN + 1, tbEquip[Item.EQUIPPOS_GUANYIN])
  end
  if 0 ~= tbEquip[Item.EQUIPPOS_MANTLE] then
    self.tbEquipPanel:SetEquip(Item.EQUIPPOS_MANTLE + 1, tbEquip[Item.EQUIPPOS_MANTLE])
  end
  if 0 ~= tbEquip[Item.EQUIPPOS_WUXINGYIN] then
    self.tbEquipPanel:SetEquip(Item.EQUIPPOS_WUXINGYIN + 1, tbEquip[Item.EQUIPPOS_WUXINGYIN])
  end
  self.tbEquipPanel:ResetFrom(Item.EQUIPPOS_MAIN_NUM + 1)
end

function tbWnd:OnSyncItemEvent(nItemId, bUpdateAll)
  if 1 == bUpdateAll then
    Bag:ClearDirtyBagItemData()
    self.nTempItemId = nItemId
    self.bTempUpdateAll = bUpdateAll
    self.bNeedOnSyncItem = true
  else
    self:OnSyncItem(nItemId, bUpdateAll)
  end
end

function tbWnd:SearchItem(nItemType, szSearchText)
  if not self.tbItemList then
    return {}
  end
  local tbItemInfo = {}
  for nIndex, tbData in pairs(self.tbItemList) do
    tbItemInfo[nIndex] = {}
    local tbItemCfg = Item:GetItemBaseProp(tbData.dwTemplateId)
    tbItemInfo[nIndex].szName = i18n.Parse(tbItemCfg.szName)
    if nItemType == tbToggle.TogStone then
      local tbStoneAttrCfg = Item.Stone:GetStoneAttri(tbData.dwTemplateId)
      if tbStoneAttrCfg[1] then
        tbItemInfo[nIndex].szAttrDesc1 = FightSkill:GetMagicDescSplit(tbStoneAttrCfg[1][1], {
          tbStoneAttrCfg[1][2],
          tbStoneAttrCfg[1][3],
          tbStoneAttrCfg[1][4]
        })
      end
      if tbStoneAttrCfg[2] then
        tbItemInfo[nIndex].szAttrDesc2 = FightSkill:GetMagicDescSplit(tbStoneAttrCfg[2][1], {
          tbStoneAttrCfg[2][2],
          tbStoneAttrCfg[2][3],
          tbStoneAttrCfg[2][4]
        })
      end
      tbItemInfo[nIndex].tbStonePos = {}
      for i = Item.EQUIP_HELM, Item.EQUIP_AMULET do
        local nEquipPos = Item.EQUIPTYPE_POS[i]
        if Item.Stone.tbEquipPos2Stone[nEquipPos][tbData.dwTemplateId] then
          table.insert(tbItemInfo[nIndex].tbStonePos, i18n.Get(Item:GetNotExtEquipName(i)))
        end
      end
    end
  end
  local tbSearchResult = {}
  for nIndex, tbData in pairs(tbItemInfo) do
    if nItemType == tbToggle.TogStone then
      if string.find(tbData.szName, szSearchText) ~= nil then
        table.insert(tbSearchResult, self.tbItemList[nIndex])
      elseif tbData.szAttrDesc1 and string.find(tbData.szAttrDesc1, szSearchText) ~= nil then
        table.insert(tbSearchResult, self.tbItemList[nIndex])
      elseif tbData.szAttrDesc2 and string.find(tbData.szAttrDesc2, szSearchText) ~= nil then
        table.insert(tbSearchResult, self.tbItemList[nIndex])
      elseif tbData.tbStonePos then
        for _, szEquipPos in ipairs(tbData.tbStonePos) do
          if string.find(szEquipPos, szSearchText) ~= nil then
            table.insert(tbSearchResult, self.tbItemList[nIndex])
            break
          end
        end
      end
    elseif string.find(tbData.szName, szSearchText) ~= nil then
      table.insert(tbSearchResult, self.tbItemList[nIndex])
    end
  end
  return tbSearchResult
end

function tbWnd:UpdateShenBing()
  if 1 ~= GetTimeFrameState(Item.ShenBingPu.CFG.szOpenLevel) or me.nLevel < Item.ShenBingPu.CFG.nLevelLimit then
    self.pPanel:SetActive(tbControls.btnShenBing, false)
  else
    local nCurHonorLevel = Item.ShenBingPu:GetHornorLevel(me)
    local szEffectName = Item.ShenBingPu.CFG.HONOR[nCurHonorLevel] and Item.ShenBingPu.CFG.HONOR[nCurHonorLevel][3] or ""
    self.pPanel:SetActive(tbControls.btnShenBing .. "/ShenBing01", nCurHonorLevel <= 0)
    local nIndex = 2
    local szKey = string.format("ShenBing%02d", nIndex)
    while self.pPanel:IsObjectExist(tbControls.btnShenBing .. "/" .. szKey) do
      self.pPanel:SetActive(tbControls.btnShenBing .. "/" .. szKey, szKey == szEffectName)
      nIndex = nIndex + 1
      szKey = string.format("ShenBing%02d", nIndex)
    end
    self.pPanel:SetActive(tbControls.btnShenBing, true)
  end
end
