local tbItem = Item:GetClass("ExpBookItem")
tbItem.PARAM_LEVEL = 1
tbItem.PARAM_MAX_EXP = 2
tbItem.PARAM_EXP = 3

function tbItem:GetTip(it)
  if not it.dwId then
    return ""
  end
  local param_level = it.GetIntValue(self.PARAM_LEVEL)
  local param_max_exp = it.GetIntValue(self.PARAM_MAX_EXP)
  local exp = it.GetIntValue(self.PARAM_EXP)
  local szLevelTip = i18n.EasyGet("\231\187\143\233\170\140\228\185\166\231\173\137\231\186\167:{0}", Lib:GetLevelText(param_level))
  local szMaxExpTip = i18n.EasyGet("\230\156\128\229\164\167\229\143\175\229\173\152\229\130\168\231\187\143\233\170\140:{0}", param_max_exp)
  local szCurExpTip = i18n.EasyGet("\229\189\147\229\137\141\229\183\178\231\187\143\229\173\152\229\130\168\231\187\143\233\170\140:{0}", exp)
  local szTip = i18n.EasyGet([[
{0}
{1}
{2} ]], szLevelTip, szMaxExpTip, szCurExpTip)
  return szTip
end

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  return {}
end
