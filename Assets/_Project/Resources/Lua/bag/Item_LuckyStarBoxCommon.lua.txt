Item.LuckyStarBox = Item.LuckyStarBox or {}
local LuckyStarBox = Item.LuckyStarBox

function LuckyStarBox:InitConfig()
  local tbField = {
    "nParamId",
    "szSaveKey",
    "nAwardId",
    "szKeyList",
    "nMaxTime",
    "nItemTemplateId",
    "nVersion"
  }
  self.tbLuckyStarBox = Lib:LoadTabFileEx("Setting/Item/LuckyStarBox.tab", "dsdsddd", "nParamId", tbField)
  for _, tbInfo in pairs(self.tbLuckyStarBox or {}) do
    local tbKeyList = Lib:GetTableFromString(tbInfo.szKeyList)
    tbInfo.tbKeyList = {}
    for k, v in pairs(tbKeyList or {}) do
      table.insert(tbInfo.tbKeyList, {k, v})
    end
    table.sort(tbInfo.tbKeyList, function(tbFirst, tbSecond)
      return tbFirst[1] < tbSecond[1]
    end)
  end
end

LuckyStarBox:InitConfig()

function LuckyStarBox:GetCfgById(nParamId)
  return self.tbLuckyStarBox[nParamId]
end
