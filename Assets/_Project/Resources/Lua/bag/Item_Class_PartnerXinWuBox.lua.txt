local tbPartnerXinWuBox = Item:GetClass("PartnerXinWuBox", 1)
tbPartnerXinWuBox = tbPartnerXinWuBox or Item:NewClass("PartnerXinWuBox", "RandItem")
local tbParam = {
  "Key",
  "AwardSubType"
}
local szType = "dd"
local tbXinWuInfo = Lib:LoadTabFileEx("Setting/Award/Awards/Items/BoxItem.tab", szType, "Key", tbParam)

function tbPartnerXinWuBox:GetUseSetting(nItemTemplateId, nItemId)
  local tbBaseProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbBaseProp then
    return
  end
  
  local function fnPreview()
    local nAwardKey = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_AWARDKEY)
    Ui:OpenWindow("UIPartnerGalleryDetail", Partner.tbAllPartnerXinWuInfo[tbXinWuInfo[nAwardKey].AwardSubType].nId)
  end
  
  local function fnUse()
    Item:ClientUseItem(nItemId)
    if me.GetItemCountInBags(nTemplateId) > 1 then
      return
    end
    Ui:CloseWindow("UIItemTips")
  end
  
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return {
      szFirstName = i18n.EasyGet("\233\162\132\232\167\136"),
      fnFirst = fnPreview
    }
  else
    return {
      szFirstName = i18n.EasyGet("\233\162\132\232\167\136"),
      fnFirst = fnPreview,
      szSecondName = i18n.EasyGet("\228\189\191\231\148\168"),
      fnSecond = fnUse
    }
  end
end
