Item.Equip = Item.Equip or {}
local Equip = Item.Equip
Equip.EQUIPED_TYPE = {
  NOT = 0,
  ME = 1,
  ASYNCDATA = 2
}

function Equip:OpenWearEquipUI(nItemId, nTemplateId)
  Ui:OpenWindow("UIQuickUseItem", nItemId, i18n.EasyGet("\231\169\191\228\184\138"), function(tbItemGrid)
    tbItemGrid:SetItem(nItemId, nTemplateId)
    tbItemGrid:SetClickEnable(true)
  end, function(tbUIQuickUseItem)
    self:UseEquip(false, nItemId)
    tbUIQuickUseItem:Close()
    Guide:TryShowQueueHeadEquipWearUI()
  end)
end

function Equip:UseEquip(bIgnoreWeaponLimit, nItemId)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local nEquipPos = KItem.GetEquipPos(pItem.dwTemplateId)
  if nEquipPos == Item.EQUIPPOS_HORSE and me.GetDoing() == Npc.Doing.sit then
    me.CenterMsg(i18n.EasyGet("\230\137\147\229\157\144\230\151\182\228\184\141\232\131\189\232\163\133\229\164\135\233\169\172\229\140\185"))
    return
  end
  if not bIgnoreWeaponLimit and nEquipPos == Item.EQUIPPOS_WEAPON and not KFightSkill.IsWeaponFitFactionSect(me.nFaction, me.nFactionSect, pItem.nWeaponType) then
    me.MsgBox(i18n.EasyGet("\229\189\147\229\137\141\231\169\191\230\136\180\231\154\132\232\163\133\229\164\135\228\184\141\231\172\166\229\144\136\229\189\147\229\137\141\233\151\168\230\180\190\230\180\190\231\179\187\239\188\140\229\176\134\230\151\160\230\179\149\228\189\191\231\148\168\230\138\128\232\131\189\239\188\140\230\152\175\229\144\166\231\161\174\232\174\164\231\187\167\231\187\173\231\169\191\230\136\180"), {
      {
        i18n.szOk,
        self.UseEquip,
        self,
        true,
        nItemId
      },
      {
        i18n.szCancel
      }
    })
    return
  end
  local bRet = Player:UseEquip(nItemId)
  if not bRet then
    return
  end
end
