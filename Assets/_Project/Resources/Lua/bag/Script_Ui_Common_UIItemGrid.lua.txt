function tbWnd:DealTip_SetActive(bActive, dwTemplateId)
  if bActive and dwTemplateId then
    local nTradeLimitWay = Trade:GetTradeLimitWay(dwTemplateId)
    if Lib:GetBit(nTradeLimitWay, Item.eTRADELIMITWAY.P2P) then
      self.pPanel:SetActive("TagCanDeal", true)
      return
    end
    if Lib:GetBit(nTradeLimitWay, Item.eTRADELIMITWAY.MARKETSTALL) then
      self.pPanel:SetActive("TagMarketStall", true)
      return
    end
  else
    self.pPanel:SetActive("TagCanDeal", false)
    self.pPanel:SetActive("TagMarketStall", false)
  end
end

function tbWnd:SetDigitalItem(szType, nValue, tbControlFlag)
  self:ResetValue()
  nValue = nValue or 0
  tbControlFlag = tbControlFlag or {bShowCount = true}
  self.szDigitalType = szType
  self.nCount = nValue
  local szIcon
  if Currency:Valid(szType) then
    szIcon = Currency:GetIcon(szType)
  elseif self.tbOtherConfig[szType] then
    if "BasicExp" == szType then
      local nBaseExp = me.GetBaseAwardExp()
      nValue = nValue * nBaseExp
    end
    szIcon = self.tbOtherConfig[szType].szIcon
  else
    Log("[UIItemGrid] SetDigitalItem Type Error", szType)
    return
  end
  local nQuality = Item:GetDigitalItemQuality(szType, nValue)
  self:BaseColor_SetSpriteByQuality(nQuality)
  self:BaseColor_SetActive(true)
  self:ItemIcon_SetSprite(szIcon)
  self:ItemIcon_SetActive(true)
  self:EquipComposeIcon_SetActive(false)
  if nValue > 0 then
    local OHT = 100000
    local szValue = nValue >= OHT and math.floor(nValue / 10000) .. i18n.EasyGet("\228\184\135") or tostring(nValue)
    self:LabelSuffix_SetText(szValue)
  end
  self:TagNew_SetActive(false)
  self:CDLabyer_SetActive(tbControlFlag.bShowCDLayer)
  self:TagTip_SetActive(false)
  self:LabelSuffix_SetActive(tbControlFlag.bShowCount and nValue > 1)
  self:SetCenterText("")
  self:SetLTCornerText("")
  self:SetLTCornerImg("")
  self:StoneTagTip_SetSprite()
  self:ZFZhenJueTypeTagTip_SetSprite()
  self:ImgRightTop_SetSprite()
  self:UdateRightTopLabel()
  self:ZYImgRankBgUpdate()
  self:DivineImgLvUpdate()
end

function tbWnd:SetGenericItem(tbData)
  if not tbData or not next(tbData) then
    Log("ItemGrid:SetGenericItem Error, param is error")
    return
  end
  local szType = tbData[1]
  local fnSetGrid = self.tbSetGridFunc[Player.AwardType[szType]]
  if fnSetGrid then
    fnSetGrid(self, table.unpack(tbData, 2))
  else
    self:SetDigitalItem(table.unpack(tbData))
  end
end

function tbWnd:Update()
  local szName, nIcon, nView, szCount
  local szFrameColor = "itemframebg"
  local szColor, pItem, nQuality
  local itemType = ""
  local szCenterText = ""
  local szLTCornerText = ""
  local szLTCornerImg = ""
  if self.nItemId and KItem.GetItemObj(self.nItemId) then
    pItem = KItem.GetItemObj(self.nItemId)
    nQuality = pItem.nQuality
    if pItem then
      self.nTemplate = pItem.dwTemplateId
      if 1 == pItem.IsEquip() and (pItem.nPos == pItem.nEquipPos or self.pAsyncData) then
        local nLianHuaLevel = -1
        local nEquipPos = pItem.nEquipPos
        if nEquipPos >= Item.EQUIPPOS_HELM and nEquipPos <= Item.EQUIPPOS_AMULET then
          if self.tbItemExtParams.nLianHuaLevel then
            nLianHuaLevel = self.tbItemExtParams.nLianHuaLevel
          elseif self.pAsyncData then
            nLianHuaLevel = self.pAsyncData.GetLianHua(nEquipPos + 1)
          else
            nLianHuaLevel = Item.EquipLianHua:GetLianHuaLevel(me, nEquipPos)
          end
        end
        if nLianHuaLevel == pItem.nLevel then
          nQuality = Item.EquipLianHua.nLianHuaQuality
        end
      end
      szName, nIcon, nView = pItem.GetItemShowInfo(self.nFaction, self.nFactionSect, self.nBodySex)
      if pItem.nMaxCount > 1 and pItem.nCount > 1 or type(self.nCount) == "string" then
        szCount = tostring(pItem.nCount)
      end
      itemType = pItem.szClass
    end
  elseif self.nTemplate then
    szName, nIcon, nView, nQuality = KItem.GetItemShowInfo(self.nTemplate, self.nFaction, self.nFactionSect, self.nBodySex)
    if szName and (type(self.nCount) == "number" and self.nCount > 1 or type(self.nCount) == "string") then
      szCount = tostring(self.nCount)
    end
    local tbBaseCfg = Item:GetItemBaseProp(self.nTemplate)
    if tbBaseCfg then
      itemType = tbBaseCfg.szClass
    end
  end
  self:SetCenterText(szCenterText)
  self:SetLTCornerText(szLTCornerText)
  self:SetLTCornerImg(szLTCornerImg)
  self:ZYImgRankBgUpdate()
  self:DivineImgLvUpdate()
  szCount = self.nShowCount and tostring(self.nShowCount) or szCount
  if nIcon then
    self:ItemIcon_SetSpriteByID(self.tbControlFlag.bBigIcon and nView or nIcon)
    self:ItemIcon_SetActive(true)
    self:BaseColor_SetSpriteByQuality(nQuality)
    self:BaseColor_SetActive(true)
  else
    self:ItemIcon_SetActive(false)
    self:BaseColor_SetActive(false)
    self:ItemIcon_SetSprite("")
  end
  if pItem and 1 == pItem.IsEquip() then
    self:LabelSuffix_SetActive(false)
  else
    self:LabelSuffix_SetActive(true)
  end
  self:LabelSuffix_SetText(szCount)
  self:SetTagTip()
  local bUseable = true
  if pItem and not self.pAsyncData then
    bUseable = 1 == Item:CheckUsable(pItem, pItem.szClass)
  end
  if not bUseable then
    self:CDLabyer_SetActive(true)
  else
    self:CDLabyer_SetActive(self.tbControlFlag.bShowCDLayer)
  end
  self:StoneTagTip_SetSprite()
  self:ZFZhenJueTypeTagTip_SetSprite()
  self:ImgRightTop_SetSprite()
  self:UdateRightTopLabel()
  local isEquipCompose = "EquipCompose" == itemType
  self:EquipComposeIcon_SetActive(isEquipCompose)
end

function tbWnd:SetTagTip()
  self:DealTip_SetActive(false)
  self:SeriesTip_SetActive(false)
  if not self.tbControlFlag.bShowTip then
    self:TagTip_SetActive(false)
    return
  end
  local szImage
  local showHigher = false
  local bShowArrow = false
  local szRBCornerText = ""
  if self.nItemId then
    local pItem = KItem.GetItemObj(self.nItemId)
    if pItem and 1 == pItem.IsEquip() then
      local pWearThisPosEquip
      if pItem.nEquipPos ~= Item.EQUIPPOS_EXTEND then
        pWearThisPosEquip = me.GetEquipByPos(pItem.nEquipPos)
      else
        pWearThisPosEquip = Item.ExtEquip:GetEquipObj(me, Item.ExtEquip:GetSlot(pItem))
      end
      local nEquipingId
      if nil ~= pWearThisPosEquip then
        nEquipingId = pWearThisPosEquip.dwId
      end
      if not nEquipingId then
        if pItem.nUseLevel <= me.nTotalLevel then
          szImage = i18n.EasyGet("\229\143\175\232\163\133\229\164\135")
          showHigher = true
        else
          szImage = string.format("<color=red>%s</color>", i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179"))
        end
      elseif nEquipingId ~= self.nItemId then
        local pCurEquip = KItem.GetItemObj(nEquipingId)
        local bEquipCompare = Item:CanEquip(pItem, pCurEquip)
        if bEquipCompare then
          if pItem.nUseLevel <= me.nTotalLevel then
            if pItem.nLevel > pCurEquip.nLevel or pItem.nLevel == pCurEquip.nLevel and pItem.nQuality > pCurEquip.nQuality or pItem.nLevel == pCurEquip.nLevel and pItem.nQuality == pCurEquip.nQuality and self:CheckRandomAttr(pCurEquip, pItem) or pItem.nLevel < pCurEquip.nLevel and pItem.nOrgValue > pCurEquip.nOrgValue then
              szImage = i18n.EasyGet("\229\143\175\232\163\133\229\164\135")
              showHigher = true
            end
          else
            szImage = string.format("<color=red>%s</color>", i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179"))
          end
        end
        if not showHigher and Item.EquipForge:IsHadTransferAttr(pItem, pCurEquip) then
          szImage = i18n.EasyGet("\229\143\175\230\180\151\231\187\131")
        end
      end
      self:SetSeriesTip(pItem.nItemType, pItem.GetSeries())
    elseif pItem and pItem.szClass == "EquipForgeSeed" then
      szImage = string.format("<color=red>%s</color>", i18n.EasyGet("\230\156\170\233\137\180\229\174\154"))
    elseif pItem and pItem.szClass == "WXHuShiItemSeed" then
      szImage = string.format("<color=red>%s</color>", i18n.EasyGet("\230\156\170\233\137\180\229\174\154"))
    elseif pItem and pItem.szClass == "ComposeMeterial" then
      if Compose.EntityCompose:CheckIsCanCompose(me, pItem.dwTemplateId) then
        szImage = i18n.EasyGet("\229\143\175\229\144\136\230\136\144")
      end
    elseif pItem and pItem.szClass == "ShenBingPuSeed" then
      szImage = string.format("<color=red>%s</color>", i18n.EasyGet("\230\156\170\233\137\180\229\174\154"))
    end
    if pItem then
      local _, nPos = me.GetItemInAllSpace(self.nItemId)
      if Item:IsInBag(nPos) or Item:IsInWarehouse(nPos) then
        local bRet = Item:GetItemTradeLimitDescBySettings(pItem.dwTemplateId, pItem.dwTradeLimitType)
        if bRet and (pItem.dwTradeLimitType == Item.eTRADELIMITTYPE.TRADE_TIMES and pItem.dwTradeLimitValue > 0 or pItem.dwTradeLimitType == Item.eTRADELIMITTYPE.NORMAL) then
          self:DealTip_SetActive(true, pItem.dwTemplateId)
        end
      end
    end
  end
  if szImage then
    self:TagTip_SetActive(true)
    self:TagTip_SetText(szImage)
  else
    self:TagTip_SetActive(false)
  end
  self:Higher_SetActive(showHigher)
  self:RedPoint_SetActive(bShowArrow)
  self:SetRBCornerText(szRBCornerText)
end

function tbWnd:CheckRandomAttr(pCurEquip, pItem)
  if pCurEquip.nEquipPos < Item.EQUIPPOS_HELM or pCurEquip.nEquipPos > Item.EQUIPPOS_AMULET then
    return false
  end
  local tbCurRandomAttr = Item:GetClass(Item.MAIN_EQUIP_CLASS_NAME):GetRandomAttr(pCurEquip)
  local tbItemRandomAttr = Item:GetClass(Item.MAIN_EQUIP_CLASS_NAME):GetRandomAttr(pItem)
  for i = 1, 2 do
    if tbItemRandomAttr[i] and not tbCurRandomAttr[i] then
      return true
    end
  end
  return false
end

function tbWnd:ImgRightTop_SetSprite()
  if self.nTemplate then
    local tbBaseCfg = Item:GetItemBaseProp(self.nTemplate)
    if tbBaseCfg then
      if tbBaseCfg.szClass == "HorseSkillStoneSeed" then
        self:ImgRightTop_SetActive(true)
        self:ImgRightTop_SetText(tbBaseCfg.nLevel)
      elseif tbBaseCfg.szClass == "HorseSkillStoneItem" then
        local nSeries
        if self.nItemId then
          local pItem = KItem.GetItemObj(self.nItemId)
          if pItem then
            nSeries = pItem.GetSeries()
          end
        end
        self:ImgRightTop_SetText(tbBaseCfg.nLevel)
        self:ImgRightTop_SetActive(true, nSeries)
      elseif tbBaseCfg.szClass == "ZhenFaSuiPian" and ZhenFa.tbItemId2GridLevel[self.nTemplate] then
        self:ImgRightTop_SetText(ZhenFa.tbItemId2GridLevel[self.nTemplate])
        self:ImgRightTop_SetActive(true)
      else
        self:ImgRightTop_SetActive(false)
      end
    else
      self:ImgRightTop_SetActive(false)
    end
  else
    self:ImgRightTop_SetActive(false)
  end
end
