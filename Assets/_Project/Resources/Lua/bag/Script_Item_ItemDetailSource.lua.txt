Item.tbDetailSource = Item.tbDetailSource or {}
Item.tbSourcePool = Item.tbDetailSource or {}

function Item:LoadDetailSource()
  local szPath = "Setting/Item/SourcePool.tab"
  Item.tbSourcePool = Lib:EasyLoadTabFile(szPath, {"nSourceId"})
  szPath = "Setting/Item/ItemDetailSource.tab"
  local tbDetailSource = Lib:EasyLoadTabFile(szPath, {
    "nItemTemplatedId",
    "nWayId"
  })
  for nItemTemplatedId, tbWay in pairs(tbDetailSource) do
    Item.tbDetailSource[nItemTemplatedId] = {}
    Item.tbDetailSource[nItemTemplatedId].tbWay = {}
    for nWayId, tbInfo in ipairs(tbWay) do
      Item.tbDetailSource[nItemTemplatedId].szDesc = Item.tbDetailSource[nItemTemplatedId].szDesc or tbInfo.szDesc
      Item.tbDetailSource[nItemTemplatedId].tbWay[nWayId] = {
        szWay = tbInfo.szWay,
        szTextSource = tbInfo.szTextSource,
        szGridSource = tbInfo.szGridSource,
        szJumpSource = tbInfo.szJumpSource
      }
    end
  end
  szPath = "Setting/Item/JumpSourceCfg.tab"
  local tbJumpSource = Lib:EasyLoadTabFile(szPath, {
    "nItemTemplatedId",
    "nWayId",
    "nSourceId"
  })
  for nItemTemplatedId, tbWay in pairs(tbJumpSource) do
    for nWayId, tbSource in pairs(tbWay) do
      assert(Item.tbDetailSource[nItemTemplatedId] and Item.tbDetailSource[nItemTemplatedId].tbWay[nWayId])
      local tbTemp = Item.tbDetailSource[nItemTemplatedId].tbWay[nWayId]
      tbTemp.tbJumpSource = tbTemp.tbJumpSource or {}
      for nSourceId, tbInfo in pairs(tbSource) do
        table.insert(tbTemp.tbJumpSource, {
          nSourceId = nSourceId,
          nNums = tbInfo.nNums,
          szRedPoint = tbInfo.szRedPoint,
          szTimeFrameStart = tbInfo.szTimeFrameStart,
          szTimeFrameEnd = tbInfo.szTimeFrameEnd,
          szOpenTimeStr = tbInfo.szOpenTimeStr,
          szEndTimeStr = tbInfo.szEndTimeStr,
          nShow = tbInfo.nShow,
          szRemindMsg = tbInfo.szRemindMsg
        })
      end
    end
  end
end

function Item:GetSourceDesc(nItemTemplatedId)
  if not Item.tbDetailSource[nItemTemplatedId] then
    return ""
  end
  return Item.tbDetailSource[nItemTemplatedId].szDesc or ""
end

function Item:GetProduceWay(nItemTemplatedId)
  if not Item.tbDetailSource[nItemTemplatedId] then
    return
  end
  return Item.tbDetailSource[nItemTemplatedId].tbWay
end

function Item:GetSourceFromPool(nSourceId)
  if not Item.tbSourcePool[nSourceId] then
    return
  end
  return Item.tbSourcePool[nSourceId]
end

Item:LoadDetailSource()
