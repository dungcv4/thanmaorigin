Require("CommonScript/Validator.lua")
Item.SelectAward = Item.SelectAward or {}
local SelectAward = Item.SelectAward

function SelectAward:InitConfig()
  local szFilepath = "Setting/Item/SelectAward.tab"
  local tbSelectAward = Lib:EasyLoadTabFile(szFilepath)
  self.tbSelectAward = {}
  for _, tbInfo in pairs(tbSelectAward or {}) do
    local tbSelectAward = {}
    local nItemBindType = tbInfo.nBindType and tbInfo.nBindType or 0
    tbSelectAward.tbAward = {
      [1] = tbInfo.szType,
      [2] = tbInfo.nItemId,
      [3] = tbInfo.nItemNum,
      [4] = {nBindType = nItemBindType}
    }
    tbSelectAward.TimeFrameStart = tbInfo.szTimeFrameStart
    tbSelectAward.TimeFrameEnd = tbInfo.szTimeFrameEnd
    if not self.tbSelectAward[tbInfo.nParamId] then
      self.tbSelectAward[tbInfo.nParamId] = {}
    end
    table.insert(self.tbSelectAward[tbInfo.nParamId], tbSelectAward)
  end
  Validator:Register(function()
    local tbField = {
      "TemplateId",
      "ClassName",
      "ExtParam1"
    }
    local tbItem = Lib:LoadTabFileEx("Setting/Item/Item.tab", "dsd", nil, tbField)
    for _, tbInfo in ipairs(tbItem) do
      if tbInfo.ClassName == "SelectAward" and not SelectAward.tbSelectAward[tbInfo.ExtParam1] then
        LogErr(string.format("SelectAward config not exists, itemid:%d, nParamId:%d", tbInfo.TemplateId, tbInfo.ExtParam1))
        return false
      end
    end
    return true
  end)
end

SelectAward:InitConfig()

function SelectAward:GetCfgById(nParamId)
  if not self.tbSelectAward[nParamId] then
    return
  end
  local tbSelectAward = {}
  for _, tbInfo in ipairs(self.tbSelectAward[nParamId]) do
    if tbInfo.TimeFrameStart and tbInfo.TimeFrameStart ~= "" and tbInfo.TimeFrameEnd and "" ~= tbInfo.TimeFrameEnd then
      if 1 == GetTimeFrameState(tbInfo.TimeFrameStart) and 1 ~= GetTimeFrameState(tbInfo.TimeFrameEnd) then
        table.insert(tbSelectAward, tbInfo.tbAward)
      end
    elseif tbInfo.TimeFrameStart and tbInfo.TimeFrameStart ~= "" and (not tbInfo.TimeFrameEnd or "" == tbInfo.TimeFrameEnd) then
      if 1 == GetTimeFrameState(tbInfo.TimeFrameStart) then
        table.insert(tbSelectAward, tbInfo.tbAward)
      end
    elseif tbInfo.TimeFrameEnd and "" ~= tbInfo.TimeFrameEnd and (not tbInfo.TimeFrameStart or tbInfo.TimeFrameStart == "") then
      if 1 ~= GetTimeFrameState(tbInfo.TimeFrameEnd) then
        table.insert(tbSelectAward, tbInfo.tbAward)
      end
    else
      table.insert(tbSelectAward, tbInfo.tbAward)
    end
  end
  if #tbSelectAward <= 0 then
    return
  end
  return tbSelectAward
end
