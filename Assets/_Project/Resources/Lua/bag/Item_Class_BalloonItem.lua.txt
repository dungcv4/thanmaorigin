Require("Setting/Npc/Ai/ChildrensBalloon/BalloonCfg.lua")
local tbItem = Item:GetClass("BalloonItem")
local tbBalloonCfg = Npc.Balloon.Cfg

function tbItem:OnUse(it)
  local nActivityId = KItem.GetItemExtParam(it.dwTemplateId, 1)
  local tbActivity = ActivityMgr:GetInstanceById(nActivityId)
  if nil == tbActivity then
    return 0
  end
  local nFreeNum = tbActivity:GetEveryDayFreeNum(me)
  if nFreeNum >= tbActivity.tbConfig.nFreeNum then
    me.CenterMsg(i18n.EasyGet("\228\187\138\230\151\165\233\135\138\230\148\190\230\172\161\230\149\176\229\183\178\231\148\168\229\174\140"))
    return 0
  end
  if me.nLevel < tbActivity.tbConfig.nLevel then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179"))
    return 0
  end
  local pNpc = KNpc.GetNearestNpcByTemplateId(me.GetNpc(), tbBalloonCfg.nBalloonNpcTemplateId, tbBalloonCfg.nPlantRadius)
  if pNpc then
    me.CenterMsg(i18n.EasyGet("\233\153\132\232\191\145\230\156\137\229\133\182\228\187\150\230\176\148\231\144\131\239\188\140\232\175\183\231\167\187\230\173\165\229\136\171\229\164\132\233\135\138\230\148\190\229\144\167"))
    return 0
  end
  if not self:CheckMap(me) then
    me.CenterMsg(i18n.Get(tbBalloonCfg.szNotProperMap))
    return 0
  end
  local nMapId, nLocX, nLocY = me.GetWorldPos()
  local pNewNpc = KNpc.Add(tbBalloonCfg.nBalloonNpcTemplateId, 1, -1, nMapId, nLocX, nLocY, 0, 0, 0)
  if pNewNpc then
    pNewNpc.nFatherPlayerId = me.dwID
    pNewNpc.nActivityId = nActivityId
    pNewNpc.nBlowNum = 0
    pNewNpc.nPrickNum = 0
    pNewNpc.SetName(i18n.Get(tbBalloonCfg.FeatureName[tbBalloonCfg.InteractType[1]], me.szName))
    pNewNpc.SetDir(tbBalloonCfg.nToward)
    tbActivity:SetEveryDayFreeNum(me, nFreeNum + 1)
    tbActivity:AddNpc(pNewNpc.nId)
    tbActivity:BroadcastNpc(me)
    Award:GiveAward(me.dwID, tbActivity.tbConfig.nAwardId, tbActivity.tbConfig.nAwardKey, {
      nLogReason = tbActivity.tbConfig.szLogWay
    })
    me.CenterMsg(i18n.Get(tbBalloonCfg.GrowText.balloonUnLive))
    me.CallClientScript("Ui:CloseWindow", "UIBag")
    me.CallClientScript("Ui:CloseWindow", "UIItemTips")
    Lib:TrackEvent(me.dwID, Env.track_childrenday_chuiqiqiu, me.dwID, me.nMapTemplateId, 0)
    return 1
  else
    me.CenterMsg(i18n.EasyGet("\230\176\148\231\144\131\233\135\138\230\148\190\229\164\177\232\180\165\239\188\140\232\175\183\231\168\141\229\144\142\229\134\141\232\175\149"))
    return 0
  end
end

function tbItem:OnClientUse(it)
  if not me then
    return 0
  end
  if not self:CheckMap(me) then
    me.CenterMsg(i18n.Get(tbBalloonCfg.szNotProperMap))
    return 0
  end
  return 1
end

function tbItem:CheckMap(pPlayer)
  for _, nTemplateId in ipairs(tbBalloonCfg.tbUseMap) do
    if pPlayer.nMapTemplateId == nTemplateId then
      return true
    end
  end
  return false
end
