Require("CommonScript/ChuanGong/ChuanGongDef.lua")
local tbItem = Item:GetClass("ChuanGongAdd")

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  local recieveCount = me.GetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_RECIEVE_COUNT)
  local sendCount = me.GetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_SEND_COUNT)
  if DegreeCtrl:GetDegree(me, "ChuanGongAdd") < ChuanGong.UseItem then
    return
  end
  local szMsg = i18n.Get(ChuanGong.tbText.szShowBothTwoTimesAdd, ChuanGong.ADD_SEND_TIMES, ChuanGong.ADD_ACCEPT_TIMES)
  if not DegreeCtrl:ReduceDegree(me, "ChuanGongAdd", 1) then
    me.CenterMsg(i18n.Get(ChuanGong.tbText.szReduceTimesFail))
    LogErr("ChuanGongAdd:OnUse reduce degree: ChuanGongAdd fail")
    return
  end
  if recieveCount < ChuanGong.ADD_ACCEPT_TIMES and sendCount < ChuanGong.ADD_SEND_TIMES then
    me.SetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_RECIEVE_COUNT, ChuanGong.ADD_ACCEPT_TIMES)
    me.SetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_SEND_COUNT, ChuanGong.ADD_SEND_TIMES)
  elseif recieveCount < ChuanGong.ADD_ACCEPT_TIMES then
    me.SetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_RECIEVE_COUNT, ChuanGong.ADD_ACCEPT_TIMES)
    szMsg = i18n.Get(ChuanGong.tbText.szRecieveTimesAdd, ChuanGong.ADD_ACCEPT_TIMES)
  elseif sendCount < ChuanGong.ADD_SEND_TIMES then
    me.SetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_SEND_COUNT, ChuanGong.ADD_SEND_TIMES)
    szMsg = i18n.Get(ChuanGong.tbText.szSendTimesAdd, ChuanGong.ADD_SEND_TIMES)
  end
  Kin:UpdateKinMemberInfo(me.dwKinId, me.dwID, "tbChuanGong", ChuanGong:GetInfo(me.dwID))
  if me.dwTongId > 0 then
    Tong:UpdateTongMemberInfo(me.dwTongId, me.dwID, "tbChuanGong", ChuanGong:GetInfo(me.dwID))
  end
  me.CenterMsg(szMsg)
  return 1
end

function tbItem:OnClientUse(it)
  if not it.dwTemplateId then
    return 0
  end
  local recieveCount = me.GetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_RECIEVE_COUNT)
  local sendCount = me.GetUserValue(ChuanGong.SAVE_GROUP, ChuanGong.tbSaveKeyEnum.KEY_ADD_SEND_COUNT)
  if recieveCount >= ChuanGong.ADD_ACCEPT_TIMES and sendCount >= ChuanGong.ADD_SEND_TIMES then
    me.CenterMsg(i18n.Get(ChuanGong.tbText.szStillHaveExtraEffectUnuse))
    return 0
  end
  if DegreeCtrl:GetDegree(me, "ChuanGongAdd") < ChuanGong.UseItem then
    me.CenterMsg(i18n.Get(ChuanGong.tbText.szTodayTimesReachUpperLimit))
    return 0
  end
  local szBoxMsg = i18n.Get(ChuanGong.tbText.szAskHaveExtraRecieveTimes)
  if recieveCount < ChuanGong.ADD_ACCEPT_TIMES then
    szBoxMsg = i18n.Get(ChuanGong.tbText.szAskHaveExtraSendTimes)
  end
  
  local function fnConfirm()
    RemoteServer.UseItem(it.dwId)
  end
  
  if recieveCount < ChuanGong.ADD_ACCEPT_TIMES and sendCount < ChuanGong.ADD_SEND_TIMES then
    return 1
  else
    me.MsgBox(szBoxMsg, {
      {
        i18n.szOk,
        fnConfirm
      },
      {
        i18n.szCancel
      }
    })
    return 0
  end
end
