local tbItem = Item:GetClass("ZongShiXinDe")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function fnUse()
    local nUseCount = ZongShiSkill:GetXinDeUseCount(me)
    local nServerMaxLevel = ZongShiSkill:GetServerMaxZongShiLevel() or 0
    if nServerMaxLevel <= 0 then
      me.CenterMsg(i18n.EasyGet("\229\174\151\229\184\136\231\173\137\231\186\167\230\156\170\229\188\128\229\144\175"))
      return
    end
    local nMaxUseCount = (math.floor(nServerMaxLevel / 10) + 1) * 5
    if nUseCount >= nMaxUseCount then
      me.CenterMsg(i18n.EasyGet("\229\183\178\232\190\190\229\136\176\228\189\191\231\148\168\228\184\138\233\153\144"))
      return
    end
    RemoteServer.RequestZongShiSkill("RequestUseXinDe")
  end
  
  return {
    szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnFirst = fnUse
  }
end

function tbItem:GetTip(pItem)
  local nServerMaxLevel = ZongShiSkill:GetServerMaxZongShiLevel() or 0
  local nMaxUseCount = (math.floor(nServerMaxLevel / 10) + 1) * 5
  local nUseCount = ZongShiSkill:GetXinDeUseCount(me)
  local szUseCount = i18n.EasyGet("\229\183\178\228\189\191\231\148\168\230\172\161\230\149\176\239\188\154{0}/{1}", nUseCount, nMaxUseCount)
  return szUseCount
end
