local tbExchangeItem = Item:GetClass("ExchangeItem")
local nExchangeCountKey = 2

function tbExchangeItem:OnCreate()
end

function tbExchangeItem:IsConsumeCountValid(dwTemplateId, nCount, nExchangeItemTempateId)
  local nItemTId
  for i = 1, 10, 2 do
    nItemTId = KItem.GetItemExtParam(dwTemplateId, i)
    if nItemTId == nExchangeItemTempateId then
      local tbData = Item.StrengthItem:GetCurExchangeData(me, nExchangeItemTempateId)
      if tbData then
        if tbData.nConsumeNum ~= nCount then
          return false, 0
        end
        return true, tbData.nTempateId
      end
      if KItem.GetItemExtParam(dwTemplateId, i + 1) == nCount then
        return true, nExchangeItemTempateId
      end
      return false, 0
    end
  end
  return false, 0
end

function tbExchangeItem:OnUseManyByTemplated(dwTemplateId, nCount, nExchangeItemTempateId)
  local bSuccess
  bSuccess, nExchangeItemTempateId = self:IsConsumeCountValid(dwTemplateId, nCount, nExchangeItemTempateId)
  if not bSuccess then
    return 0
  end
  if 0 == nExchangeItemTempateId then
    return 0
  end
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  local bCanUse = Item.StrengthItem:IsCurUnlock(me, nExchangeItemTempateId)
  if not bCanUse then
    me.CenterMsg(i18n.EasyGet("\232\175\165\233\129\147\229\133\183\230\151\160\230\179\149\229\133\145\230\141\162"))
    return 0
  end
  local nMaxExchangeCount = Item.StrengthItem:GetMaxExchangeCount(nExchangeItemTempateId)
  local nExchangeCount = Item.StrengthItem:GetExchangeCount(me, nExchangeItemTempateId)
  if nMaxExchangeCount <= nExchangeCount then
    me.CenterMsg(i18n.EasyGet("\232\175\165\231\137\169\229\147\129\229\133\145\230\141\162\230\172\161\230\149\176\229\183\178\229\136\176\232\190\190\228\184\138\233\153\144\227\128\130"))
    return 0
  end
  if nCount <= me.GetItemCountByTypeInBag(dwTemplateId) then
    local tbAwardList = {}
    local tbStoneAward = {
      "Item",
      nExchangeItemTempateId,
      1,
      {}
    }
    table.insert(tbAwardList, tbStoneAward)
    for _, tbInfo in pairs(tbAwardList) do
      tbInfo.bCenterMsg = 1
      tbInfo.bAwardMsgView = 1
    end
    me.SendAward(tbAwardList, true, false, Env.LogWay_ExchangeItem)
    me.CallClientScript("Item:OnUseCallBack", "ExchangeItem")
    return nCount
  else
    return 0
  end
end

function tbExchangeItem:OnClientUse(pItem)
  local tbExchangeList = {}
  for i = 1, 10, 2 do
    local nExchangeItemId = KItem.GetItemExtParam(pItem.dwTemplateId, i)
    local nExchangeItemConsumeNum = KItem.GetItemExtParam(pItem.dwTemplateId, i + 1)
    if 0 ~= nExchangeItemId and 0 ~= nExchangeItemConsumeNum then
      local itemExchangeData = {}
      itemExchangeData.nTempateId = nExchangeItemId
      itemExchangeData.nConsumeNum = nExchangeItemConsumeNum
      table.insert(tbExchangeList, itemExchangeData)
    else
      break
    end
  end
  Ui:OpenWindow("UIItemExchange", pItem.dwTemplateId, tbExchangeList)
  Ui:CloseWindow("UIItemTips")
  return 0
end

function tbExchangeItem:OnUseCallBack()
  self:RefreshViewClient()
end

function tbExchangeItem:RefreshViewClient()
  if Ui:WindowVisible("UIItemExchange") == true then
    Ui:GetWindow("UIItemExchange"):RefreshView()
  end
  if true == Ui:WindowVisible("UIItemExchangeJiNengTuPo") then
    Ui:GetWindow("UIItemExchangeJiNengTuPo"):RefreshView()
  end
end
