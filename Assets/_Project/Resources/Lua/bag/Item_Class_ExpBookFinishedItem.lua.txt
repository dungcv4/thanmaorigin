local tbItem = Item:GetClass("ExpBookFinishedItem")

function tbItem:OnCreate(it)
end

function tbItem:GetTip(it)
  if not it.dwId then
    return ""
  end
  local nLevel = self:GetBookLevelById(it.dwTemplateId)
  if not nLevel then
    return ""
  end
  local nGetExp = ExpBook.tbExpBookLevelInfo[nLevel].nMaxLevelExp
  local szLevelTip = i18n.EasyGet("\231\187\143\233\170\140\228\185\166\231\173\137\231\186\167:{0}", Lib:GetLevelText(nLevel))
  local szUseTimesTip = i18n.EasyGet("\228\187\138\230\151\165\232\191\152\229\143\175\228\187\165\228\189\191\231\148\168:{0}\230\172\161", DegreeCtrl:GetDegree(me, "ExpBookItemUse"))
  local szExpTip = i18n.EasyGet("\228\189\191\231\148\168\232\142\183\229\190\151\231\187\143\233\170\140:{0}", nGetExp)
  local szConditionTip = i18n.EasyGet("\227\128\144\228\189\191\231\148\168\230\157\161\228\187\182\227\128\145\n  1.30\231\186\167<=\231\142\169\229\174\182\231\173\137\231\186\167<=\231\187\143\233\170\140\228\185\166\231\173\137\231\186\167\n  2.\230\187\161\231\187\143\233\170\140\229\144\142\228\184\141\229\143\175\228\187\165\228\189\191\231\148\168")
  local szTip = string.format([[
%s
%s
%s

%s  ]], szLevelTip, szUseTimesTip, szExpTip, szConditionTip)
  return szTip
end

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  local pItem
  if nItemId then
    pItem = KItem.GetItemObj(nItemId)
    if not pItem then
      return
    end
  end
  if not pItem then
    return
  end
  
  local function fnGive()
    Ui:CloseWindow("UIItemTips")
    me.CallClientScript("Ui:OpenWindow", "UIExperienceBook", nItemTemplateId, nItemId, "ExperienceBook", true, true)
  end
  
  return {
    szSecondName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnSecond = "UseItem",
    szFirstName = i18n.EasyGet("\232\181\160\233\128\129"),
    fnFirst = fnGive
  }
end

function tbItem:GetBookLevelById(nItemTemplateId)
  for nLevel, tbV in pairs(ExpBook.tbExpBookLevelInfo) do
    if nItemTemplateId == tbV.nTemplateId then
      return nLevel
    end
  end
end

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  if me.nLevel < ExpBook.Cfg.MIN_USE_LEVEL then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}\231\186\167", ExpBook.Cfg.MIN_USE_LEVEL))
    return
  end
  local nLevel = self:GetBookLevelById(it.dwTemplateId)
  if not nLevel then
    return
  end
  if nLevel < me.nTotalLevel then
    me.CenterMsg(i18n.EasyGet("\230\151\160\230\179\149\228\189\191\231\148\168\228\189\142\228\186\142\232\135\170\232\186\171\231\173\137\231\186\167\231\154\132\231\187\143\233\170\140\228\185\166"))
    return
  end
  local nExpPercent = me.GetExp() / me.GetNextLevelExp()
  if nExpPercent >= 1 then
    me.CenterMsg(i18n.EasyGet("\228\189\160\229\183\178\231\187\143\230\152\175\230\187\161\231\187\143\233\170\140\231\138\182\230\128\129\230\151\160\230\179\149\228\189\191\231\148\168"))
    return
  end
  local nZongShiExp = ZongShiSkill:GetPlayerZongShiExp(me)
  local nZongShiExpPercent = nZongShiExp / me.GetNextLevelExp()
  if nZongShiExpPercent >= 1 then
    me.CenterMsg(i18n.EasyGet("\228\189\160\229\183\178\231\187\143\230\152\175\230\187\161\231\187\143\233\170\140\231\138\182\230\128\129\230\151\160\230\179\149\228\189\191\231\148\168"))
    return
  end
  if 0 == DegreeCtrl:GetDegree(me, "ExpBookItemUse") then
    me.CenterMsg(i18n.EasyGet("\232\182\133\232\191\135\230\175\143\230\151\165\230\156\128\229\164\167\228\189\191\231\148\168\230\172\161\230\149\176"))
    return
  end
  local nGetExp = ExpBook.tbExpBookLevelInfo[nLevel].nMaxLevelExp
  me.AddExperience(nGetExp, Env.LogWay_ExpBook)
  local szMsg = i18n.EasyGet("\230\136\144\229\138\159\228\189\191\231\148\168\228\184\128\230\156\172\231\187\143\233\170\140\228\185\166\239\188\140\232\142\183\229\190\151\228\186\134 {0} \231\187\143\233\170\140", nGetExp)
  me.CenterMsg(szMsg)
  me.Msg(szMsg)
  DegreeCtrl:ReduceDegree(me, "ExpBookItemUse", 1)
  return 1
end
