local tbItem = Item:GetClass("ShenBingPuMaterial")
local PANEL_ENHANCE = 1

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return {}
  end
  if not Item.ShenBingPu.CFG.ENHANCE_MATERIAL_VAL[nItemTemplateId] then
    return {}
  end
  
  local function fnEnhance()
    if me.nLevel < Item.ShenBingPu.CFG.nLevelLimit then
      me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179\230\151\160\230\179\149\228\189\191\231\148\168"))
      return
    end
    Ui:OpenWindow("UIDivineWeapon", PANEL_ENHANCE)
  end
  
  local function fnCompose()
    local bRet, szMsg = self:CheckShowComposeBtn(nItemTemplateId)
    if not bRet then
      if not Lib:IsEmptyStr(szMsg) then
        me.CenterMsg(szMsg)
      end
      return
    end
    local tbMaterialComposeCfg = Item.ShenBingPu.CFG.MATERIAL_COMPOSE[nItemTemplateId]
    local nConsumeNum = tbMaterialComposeCfg[2]
    local nTargetId = tbMaterialComposeCfg[3]
    local nTargetNum = tbMaterialComposeCfg[4]
    local nNeedCoin = tbMaterialComposeCfg[5]
    local nCurCoin = me.GetMoney("Coin")
    if nNeedCoin > nCurCoin then
      return me.CenterMsg(i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179"))
    end
    local nCurItemNum = me.GetItemCountByTypeInBag(nItemTemplateId)
    local nItemCount = math.floor(nCurItemNum / nConsumeNum)
    if nItemCount < 1 then
      return me.CenterMsg(i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179"))
    end
    local nCoinCount = math.floor(nCurCoin / nNeedCoin)
    if 1 == nCoinCount or 1 == nItemCount then
      self:DoComposeReq(nItemTemplateId, 1)
    else
      local nMax = nItemCount < nCoinCount and nItemCount or nCoinCount
      Ui:OpenWindow("UIJingZhuCompose_Common", nItemTemplateId, nMax, nTargetId, nTargetNum, i18n.EasyGet("\230\157\144\230\150\153\229\144\136\230\136\144"), {
        self.DoComposeReq,
        self
      })
    end
    Ui:CloseWindow("UIItemTips")
  end
  
  if self:CheckShowComposeBtn(nItemTemplateId) then
    return {
      szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
      fnFirst = fnEnhance,
      szSecondName = i18n.EasyGet("\229\144\136\230\136\144"),
      fnSecond = fnCompose
    }
  else
    return {
      szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
      fnFirst = fnEnhance
    }
  end
end

function tbItem:DoComposeReq(nTemplateId, nTargetNum)
  local ShenBingPuC = Item.ShenBingPu
  ShenBingPuC:CallServerScript(ShenBingPuC.Protocol.MaterialCompose, nTemplateId, nTargetNum)
end

function tbItem:CheckShowComposeBtn(nTemplateId)
  local tbMaterialComposeCfg = Item.ShenBingPu.CFG.MATERIAL_COMPOSE[nTemplateId]
  if not tbMaterialComposeCfg then
    return false, i18n.EasyGet("\232\175\165\230\157\144\230\150\153\229\176\154\230\156\170\229\188\128\229\144\175\229\144\136\230\136\144")
  end
  local szOpenTimeFrame = Item.ShenBingPu.CFG.OpenTimeFrame[tbMaterialComposeCfg[6]]
  if 0 == GetTimeFrameState(szOpenTimeFrame) then
    local nCurMaxLevel = Calendar:GetWorldLevelByTimeframe(szOpenTimeFrame)
    return false, i18n.EasyGet("\229\188\128\230\148\190{0}\231\173\137\231\186\167\228\184\138\233\153\144\229\144\142\229\188\128\229\144\175\229\144\136\230\136\144\229\138\159\232\131\189", nCurMaxLevel)
  end
  return true, ""
end
