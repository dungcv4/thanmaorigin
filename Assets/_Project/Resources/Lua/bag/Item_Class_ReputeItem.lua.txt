local tbItem = Item:GetClass("ReputeItem")

function tbItem:OnClientUse(pItem)
  return 1
end

function tbItem:OnUseMany(it, nCount)
  if not it.dwTemplateId or nCount <= 0 then
    return 0
  end
  local nReputeType = KItem.GetItemExtParam(it.dwTemplateId, 1)
  local szCurrencyType = Repute.tbReputeType2CurrencyType[nReputeType]
  local nPerValue = KItem.GetItemExtParam(it.dwTemplateId, 2)
  local nCurValue = me.GetMoney(szCurrencyType)
  local nMaxValue = Currency:GetCfgMaxValue(szCurrencyType)
  local nCanAddCount = math.floor((nMaxValue - nCurValue) / nPerValue)
  if nCanAddCount < 1 then
    me.CallClientScript("Repute:OnOpenMessageBox", nReputeType)
    return 0
  end
  local nAddCount = math.min(nCount, nCanAddCount)
  local nAddValue = nPerValue * nAddCount
  me.AddMoney(szCurrencyType, nAddValue, Env.LogWay_UseItem, 0, it.dwTemplateId, nAddCount)
  local szName = i18n.Get(Repute.tbCfgReputeInfo[nReputeType].szName)
  me.CenterMsg(i18n.EasyGet("\228\189\160\232\142\183\229\190\151\228\186\134{0}\231\130\185{1}", nAddValue, szName))
  return nAddCount
end

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  
  local function funcCloseUI()
    me.CallClientScript("Ui:CloseWindow", "UIBag")
    me.CallClientScript("Ui:CloseWindow", "UIItemTips")
  end
  
  local nReputeType = KItem.GetItemExtParam(it.dwTemplateId, 1)
  local szCurrencyType = Repute.tbReputeType2CurrencyType[nReputeType]
  local nValue = KItem.GetItemExtParam(it.dwTemplateId, 2)
  local nCurValue = me.GetMoney(szCurrencyType)
  local nMaxValue = Currency:GetCfgMaxValue(szCurrencyType)
  if nMaxValue < nCurValue + nValue then
    me.CallClientScript("Repute:OnOpenMessageBox", nReputeType)
    funcCloseUI()
    return 0
  end
  me.AddMoney(szCurrencyType, nValue, Env.LogWay_UseItem, 0, it.dwTemplateId, 1)
  local szName = i18n.Get(Repute.tbCfgReputeInfo[nReputeType].szName)
  me.CenterMsg(i18n.EasyGet("\228\189\160\232\142\183\229\190\151\228\186\134{0}\231\130\185{1}", nValue, szName))
  return 1
end
