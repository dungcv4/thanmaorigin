local tbZFZhenJue = Item:GetClass("ZFZhenJue")

function tbZFZhenJue:GetTip(pItem)
  local tbAttrCur = ZhenFa:GetZhenJueAttrByItemId(me, pItem.tbItemExtParams.nPos, pItem.tbItemExtParams.nHole)
  if not tbAttrCur then
    return
  end
  local szAttr = FightSkill:GetMagicDesc(tbAttrCur[1], {
    table.unpack(tbAttrCur, 2)
  })
  return szAttr
end

function tbZFZhenJue:GetUseSetting(pnTemplateId, nItemId, szOptItem)
  return {}
end
