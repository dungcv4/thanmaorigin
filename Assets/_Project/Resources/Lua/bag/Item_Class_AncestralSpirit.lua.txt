local tbItem = Item:GetClass("AncestralSpirit")

function tbItem:OnClientUse(it)
  local tbInfo = Item:GetClass("CountAddProp"):GetAddDegreeInfo("TowerDefense")
  if not tbInfo then
    me.CenterMsg(i18n.EasyGet("\231\188\186\229\176\145\229\133\136\231\165\150\228\185\139\233\173\130\233\133\141\231\189\174"))
    return 0
  end
  local nCurNum = DegreeCtrl:GetDegree(me, "TowerDefense")
  local nTotalNum = DegreeCtrl:GetMaxDegree("TowerDefense", me)
  if nCurNum >= nTotalNum then
    me.CenterMsg(i18n.EasyGet("\229\183\178\232\190\190\229\143\130\232\181\155\230\172\161\230\149\176\228\184\138\233\153\144\239\188\140\232\175\183\229\133\136\230\182\136\232\128\151\229\137\169\228\189\153\230\172\161\230\149\176\229\144\142\229\134\141\228\189\191\231\148\168"))
    return 0
  end
  if DegreeCtrl:GetDegree(me, "AncestralSpirit") > 0 then
    return 1
  else
    me.CenterMsg(i18n.EasyGet("\230\156\172\230\151\165\228\189\191\231\148\168\229\133\136\231\165\150\229\186\135\230\138\164\230\172\161\230\149\176\229\183\178\232\190\190\228\184\138\233\153\144\239\188\140\232\175\183\230\152\142\229\164\169\229\134\141\232\175\149"))
    return 0
  end
end

function tbItem:OnUse(it)
  if DegreeCtrl:ReduceDegree(me, "AncestralSpirit", 1) then
    DegreeCtrl:AddDegree(me, "TowerDefense", 1)
    me.CallClientScript("TowerDefense:RefreshSignUpView")
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\230\136\144\229\138\159"))
    return 1
  else
    return 0
  end
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "AncestralSpirit")
end
