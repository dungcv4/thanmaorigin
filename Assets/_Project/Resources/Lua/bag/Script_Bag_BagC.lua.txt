local SORT_KEY = Ui:GetUi_SORT_KEY()
local tbToggle = Ui.tbToggle
Bag.tbEquipBagSortKey = {MainEquip = 1, EquipForgeSeed = 2}
Bag.nDefaultSortKey2 = 1000
Bag.tbEquipSeedDetailType2ItemType = {
  [1] = Item.EQUIP_WEAPON,
  [2] = Item.EQUIP_HELM,
  [3] = Item.EQUIP_NECKLACE,
  [4] = Item.EQUIP_CUFF,
  [5] = Item.EQUIP_RING,
  [6] = Item.EQUIP_ARMOR,
  [7] = Item.EQUIP_PENDANT,
  [8] = Item.EQUIP_BELT,
  [9] = Item.EQUIP_AMULET,
  [10] = Item.EQUIP_BOOTS
}

function Bag.BagItemSortFunc(bagItemDataA, bagItemDataB)
  if bagItemDataA.nSortKey2 ~= Bag.nDefaultSortKey2 and bagItemDataB.nSortKey2 ~= Bag.nDefaultSortKey2 then
    return Bag.EquipItemSortFunc(bagItemDataA, bagItemDataB)
  end
  if bagItemDataA.nSortKey ~= bagItemDataB.nSortKey then
    return bagItemDataA.nSortKey < bagItemDataB.nSortKey
  end
  if bagItemDataA.nDetailType ~= bagItemDataB.nDetailType then
    return bagItemDataA.nDetailType < bagItemDataB.nDetailType
  end
  if bagItemDataA.nLevel ~= bagItemDataB.nLevel then
    return bagItemDataA.nLevel > bagItemDataB.nLevel
  end
  local nQualityA = Item:GetQuality(bagItemDataA.dwTemplateId)
  local nQualityB = Item:GetQuality(bagItemDataB.dwTemplateId)
  if nQualityA ~= nQualityB then
    return nQualityA > nQualityB
  end
  if bagItemDataA.dwTemplateId ~= bagItemDataB.dwTemplateId then
    return bagItemDataA.dwTemplateId < bagItemDataB.dwTemplateId
  end
  return bagItemDataA.nItemId < bagItemDataB.nItemId
end

function Bag.EquipItemSortFunc(bagItemDataA, bagItemDataB)
  if bagItemDataA.nUseLevel ~= bagItemDataB.nUseLevel and bagItemDataA.nSortKey2 ~= Bag.nDefaultSortKey2 and bagItemDataB.nSortKey2 ~= Bag.nDefaultSortKey2 and (bagItemDataA.nUseLevel <= me.nTotalLevel and bagItemDataB.nUseLevel > me.nTotalLevel or bagItemDataA.nUseLevel > me.nTotalLevel and bagItemDataB.nUseLevel <= me.nTotalLevel) then
    return bagItemDataA.nUseLevel < bagItemDataB.nUseLevel
  end
  if bagItemDataA.nSortKey ~= bagItemDataB.nSortKey then
    return bagItemDataA.nSortKey < bagItemDataB.nSortKey
  end
  if bagItemDataA.nSortKey2 ~= bagItemDataB.nSortKey2 then
    return bagItemDataA.nSortKey2 < bagItemDataB.nSortKey2
  end
  local nQualityA = Item:GetQuality(bagItemDataA.dwTemplateId)
  local nQualityB = Item:GetQuality(bagItemDataB.dwTemplateId)
  if nQualityA ~= nQualityB then
    return nQualityA > nQualityB
  end
  if bagItemDataA.nLevel ~= bagItemDataB.nLevel then
    return bagItemDataA.nLevel > bagItemDataB.nLevel
  end
  if bagItemDataA.nDetailType ~= bagItemDataB.nDetailType then
    return bagItemDataA.nDetailType < bagItemDataB.nDetailType
  end
  if bagItemDataA.dwTemplateId ~= bagItemDataB.dwTemplateId then
    return bagItemDataA.dwTemplateId < bagItemDataB.dwTemplateId
  end
  return bagItemDataA.nItemId < bagItemDataB.nItemId
end

function Bag.StoneItemSortFunc(bagItemDataA, bagItemDataB)
  if Item.Stone:IsStone(bagItemDataA.dwTemplateId) and Item.Stone:IsStone(bagItemDataB.dwTemplateId) then
    local tbStoneAttrCfgA = Item.Stone:GetStoneAttri(bagItemDataA.dwTemplateId)
    local tbStoneAttrCfgB = Item.Stone:GetStoneAttri(bagItemDataB.dwTemplateId)
    if #tbStoneAttrCfgA ~= #tbStoneAttrCfgB then
      return #tbStoneAttrCfgA > #tbStoneAttrCfgB
    end
    local nGradeA = Item.Stone:GetStoneGrade(bagItemDataA.dwTemplateId)
    local nGradeB = Item.Stone:GetStoneGrade(bagItemDataB.dwTemplateId)
    if nGradeA ~= nGradeB then
      return nGradeA > nGradeB
    end
    if bagItemDataA.nLevel ~= bagItemDataB.nLevel then
      return bagItemDataA.nLevel > bagItemDataB.nLevel
    end
    if bagItemDataA.nDetailType ~= bagItemDataB.nDetailType then
      return bagItemDataA.nDetailType < bagItemDataB.nDetailType
    end
    local nQualityA = Item:GetQuality(bagItemDataA.dwTemplateId)
    local nQualityB = Item:GetQuality(bagItemDataB.dwTemplateId)
    if nQualityA ~= nQualityB then
      return nQualityA > nQualityB
    end
    if bagItemDataA.dwTemplateId ~= bagItemDataB.dwTemplateId then
      return bagItemDataA.dwTemplateId < bagItemDataB.dwTemplateId
    end
    return bagItemDataA.nItemId < bagItemDataB.nItemId
  else
    return Bag.BagItemSortFunc(bagItemDataA, bagItemDataB)
  end
end

function Bag:GetAllItemBag()
  if self.tbItemBag == nil then
    Bag:InitItemList()
  end
  return self.tbItemBag
end

function Bag:GetItemBagByItemType(nItemType)
  if self.tbItemBag == nil then
    Bag:InitItemList()
  end
  local tbBagItemData = self.tbItemBag[nItemType]
  return tbBagItemData
end

function Bag:GetItemBagList(nItemType)
  if self.tbItemBagList == nil then
    self.tbItemBagList = {}
  end
  if self.tbItemBagList[nItemType] == nil then
    self.tbItemBagList[nItemType] = {}
    local tbBagItemData = self:GetItemBagByItemType(nItemType)
    if nil == tbBagItemData then
      return nil
    end
    local tbItemBagList = {}
    for nItemId, bagItemData in pairs(tbBagItemData) do
      table.insert(tbItemBagList, bagItemData)
    end
    if nItemType == Ui.tbToggle.TogStone then
      table.sort(tbItemBagList, Bag.StoneItemSortFunc)
    elseif nItemType == Ui.tbToggle.TogEquip then
      table.sort(tbItemBagList, Bag.EquipItemSortFunc)
    else
      table.sort(tbItemBagList, Bag.BagItemSortFunc)
    end
    self.tbItemBagList[nItemType] = tbItemBagList
  end
  return self.tbItemBagList[nItemType]
end

function Bag:UpdateItemListSingle(pItem)
  if self.tbItemBag == nil then
    self.tbItemBag = {}
  end
  local nItemId = pItem.dwId
  local bagItemData = {}
  bagItemData.nSortKey = SORT_KEY[pItem.nItemType] or 1000
  if pItem.szClass == "EquipForgeSeed" then
    bagItemData.nSortKey = SORT_KEY[Bag.tbEquipSeedDetailType2ItemType[pItem.nDetailType]] or 1000
  end
  bagItemData.nSortKey2 = Bag.tbEquipBagSortKey[pItem.szClass] or Bag.nDefaultSortKey2
  bagItemData.nDetailType = pItem.nDetailType
  bagItemData.nLevel = pItem.nLevel
  bagItemData.nItemId = nItemId
  bagItemData.dwTemplateId = pItem.dwTemplateId
  bagItemData.nUseLevel = pItem.nUseLevel
  bagItemData.bIsEquip = 1 == pItem.IsEquip()
  self.tbItemBag[tbToggle.TogAll] = self.tbItemBag[tbToggle.TogAll] or {}
  local tbBagItemData = self.tbItemBag[tbToggle.TogAll]
  tbBagItemData[nItemId] = bagItemData
  local tbClassifyInfo = Item:GetDetailTypeClassify(pItem.nItemType, pItem.nDetailType)
  if tbClassifyInfo then
    for key, nClassify in pairs(Ui.tbKey2Classify) do
      if 1 == tbClassifyInfo[key] then
        if "bInZhenYuanEquip" == key then
          if Item.ZhenYuan:IsZhenYuanEquip(pItem) then
            self.tbItemBag[nClassify] = self.tbItemBag[nClassify] or {}
            local tbBagItemData = self.tbItemBag[nClassify]
            tbBagItemData[nItemId] = bagItemData
          end
        elseif "bInEquip" == key then
          if not Item.ZhenYuan:IsZhenYuanEquip(pItem) then
            self.tbItemBag[nClassify] = self.tbItemBag[nClassify] or {}
            local tbBagItemData = self.tbItemBag[nClassify]
            tbBagItemData[nItemId] = bagItemData
          end
        else
          self.tbItemBag[nClassify] = self.tbItemBag[nClassify] or {}
          local tbBagItemData = self.tbItemBag[nClassify]
          tbBagItemData[nItemId] = bagItemData
        end
      end
    end
  else
    LogErr(string.format("DetailTypeClassify.tab: DetailType:%d%s not Define Classify!", pItem.nDetailType, i18n.Parse(pItem.szName)))
  end
end

function Bag:InitItemList()
  if self.tbItemBag == nil then
    self.tbItemBag = {}
  end
  local tbItem = me.GetItemListInBag()
  for nIdx, pItem in ipairs(tbItem) do
    Bag:UpdateItemListSingle(pItem)
  end
end

function Bag:OnLogin(bIsReconnect)
  self.tbItemBag = nil
  self.tbItemBagList = nil
  Bag:InitItemList()
end

function Bag:ClearDirtyBagItemData()
  self.tbItemBag = nil
  self.tbItemBagList = nil
end

function Bag:OnItemUpdate(nItemId, bUpdateAll, nAddCount)
  local bIsNeedClear = Ui:WindowVisible("UIBag") and not Ui:GetWindow("UIBag"):IsOpenBagPanel() or true
  if bIsNeedClear then
    self:ClearDirtyBagItemData()
  end
end

function Bag:OnDelItem(dwItemId, dwTemplateId)
  local bIsNeedClear = Ui:WindowVisible("UIBag") and not Ui:GetWindow("UIBag"):IsOpenBagPanel() or true
  if bIsNeedClear then
    self:ClearDirtyBagItemData()
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, Bag)
EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, Bag.OnItemUpdate, Bag)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_DEL_ITEM, Bag)
EventNotify:RegistNotify(EventNotify.emNOTIFY_DEL_ITEM, Bag.OnDelItem, Bag)
