local tbItem = Item:GetClass("TreasureRabbitBag")
tbItem.PARAM_AWARDID = 1
tbItem.PARAM_AWARDKEY = 2

function tbItem:OnUse(item)
  return self:OnUseMany(item, 1)
end

function tbItem:GenAward(nCount, nAwardId, nAwardKey)
  local tbAward = {}
  for i = 1, nCount do
    Lib:MergeTable(tbAward, Award:GetAward(nAwardId, nAwardKey))
  end
  tbAward = Award:MergetSameTypeAward(tbAward)
  return tbAward
end

function tbItem:OnUseMany(item, nCount)
  local pPlayer = me
  local nLeftBag = pPlayer.GetFreeBagCount()
  if nLeftBag <= 0 then
    pPlayer.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\239\188\129"))
    return 0
  end
  if nCount > item.nCount then
    pPlayer.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\230\149\176\233\135\143\228\184\141\232\182\179"))
    return 0
  end
  local nId = ActivityMgr:GetOBTreasureRabbitId()
  if not nId then
    local nAwardId = KItem.GetItemExtParam(item.dwTemplateId, self.PARAM_AWARDID)
    local nAwardKey = KItem.GetItemExtParam(item.dwTemplateId, self.PARAM_AWARDKEY)
    if not (nAwardId and not (nAwardId <= 0) and nAwardKey) or nAwardKey <= 0 then
      pPlayer.CenterMsg(i18n.EasyGet("{0}\232\191\135\230\156\159\229\165\150\229\138\177\228\184\141\229\173\152\229\156\168", item.dwTemplateId))
      return 0
    end
    local tbAward = self:GenAward(nCount, nAwardId, nAwardKey)
    pPlayer.SendAward(tbAward, true, true, Env.LogWay_TreasureRabbitBag)
    return nCount
  end
  local pInstance = ActivityMgr:GetInstanceById(nId)
  local nLastAwardCount = pInstance:GetLastNumOfUsingBag(pPlayer)
  local tbConfig = pInstance:GetConfig()
  local nAdvancedCount = math.min(nCount, nLastAwardCount)
  local nBaseCount = nCount - nAdvancedCount
  local tbAward = {}
  if nAdvancedCount > 0 then
    pInstance:AddBagUsedNum(pPlayer, nAdvancedCount)
    Lib:MergeTable(tbAward, self:GenAward(nAdvancedCount, tbConfig.nBagAwardId, tbConfig.nBagAwardKey))
  end
  if nBaseCount > 0 then
    Lib:MergeTable(tbAward, self:GenAward(nBaseCount, tbConfig.nBagBaseAwardId, tbConfig.nBagBaseAwardKey))
  end
  tbAward = Award:MergetSameTypeAward(tbAward)
  pPlayer.SendAward(tbAward, true, true, Env.LogWay_TreasureRabbitBag)
  return nCount
end

function tbItem:OnClientUse(item)
  return 1
end
