Require("CommonScript/ZhenYuan/ZhenYuanOtherCfg.lua")
local tbItem = Item:GetClass("ZhenYuan")
local tbEquip = Item:GetClass("equip")

function tbItem:GetTip(pEquip, pPlayer, nEnhanLevel)
  return self:FormatTips(pEquip.dwTemplateId, pEquip, pPlayer, nEnhanLevel)
end

function tbItem:FormatTips(nTemplateId, pEquip, pPlayer, nEnhanLevel)
  local tbBaseAttrTips = {}
  local tbExeAttrTips = {}
  local tbSpecialTips = {}
  local tbLianHuaTips = {}
  local tbAttrib = KItem.GetEquipBaseProp(pEquip.dwTemplateId).tbBaseAttrib
  local nQuality = Item:GetQuality(nTemplateId)
  local szColor = Item:GetQualityColor(nQuality)
  local tbLianHuaAttr = {}
  if nEnhanLevel and nEnhanLevel > 0 then
    local nSlot = Item:GetExtEquipSlot(pEquip)
    local tbEnhanceCfg = Item.ZhenYuan:GetHoleEnhanceCfg(nSlot, nEnhanLevel)
    if tbEnhanceCfg then
      for i = 1, tbEnhanceCfg.nAttrNum do
        local szAttrName = tbEnhanceCfg["szAttribType" .. i]
        local tbValue = tbEnhanceCfg["tbnAttribValue" .. i]
        tbLianHuaAttr[szAttrName] = tbValue
      end
    end
  end
  for i, tbMA in ipairs(tbAttrib) do
    local szDesc = tbEquip:GetMagicAttribDesc(tbMA.szName, tbMA.tbValue, tbMA.nActiveReq)
    if nEnhanLevel and nEnhanLevel > 0 and tbLianHuaAttr[tbMA.szName] then
      local _, szAttrValue = FightSkill:GetMagicDescSplit(tbMA.szName, tbLianHuaAttr[tbMA.szName])
      local szDesc2 = i18n.EasyGet("<color=#{0}>(\231\130\188\229\140\150{1})</color>", Ui.tbColor:GetColorHexByDefineName("NewGreen"), szAttrValue)
      table.insert(tbLianHuaTips, szDesc2)
    end
    szDesc = string.format("<color=#%s>%s</color>", Ui.tbColor:GetColorHexByDefineName(szColor), szDesc)
    table.insert(tbBaseAttrTips, szDesc)
  end
  local tbRandAttr = Item.ZhenYuan:GetEquipRandomAttr(pEquip)
  local tbMinAndMaxAttr = Item.ZhenYuan:GetMaxAndMinAttr(pEquip)
  if next(tbRandAttr) then
    for i = 1, #tbRandAttr do
      local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRandAttr[i].szAttrType, tbRandAttr[i].tbAttrValue)
      local szMagic = ""
      if tbMinAndMaxAttr[szAttrType] then
        local _, szMinAttrValue = FightSkill:GetMagicDescSplit(tbRandAttr[i].szAttrType, tbMinAndMaxAttr[szAttrType][1])
        local _, szMaxAttrValue = FightSkill:GetMagicDescSplit(tbRandAttr[i].szAttrType, tbMinAndMaxAttr[szAttrType][2])
        local szMinStr = string.gsub(szMinAttrValue, "+", "")
        local szMaxStr = string.gsub(szMaxAttrValue, "+", "")
        szMagic = string.format("<color=#%s>%s : %s</color><color=#%s>  (%s - %s)</color>", Ui.tbColor:GetColorHexByDefineName(tbRandAttr[i].szColor), szAttrType, szAttrValue, Ui.tbColor:GetColorHexByDefineName("Gray"), szMinStr, szMaxStr)
        table.insert(tbExeAttrTips, szMagic)
      end
    end
  end
  local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[pEquip.dwTemplateId]
  if tbItemCfg and tbItemCfg.nSkillId and tbItemCfg.nSkillId > 0 then
    local szNowLevelDesc = FightSkill:GetSkillMagicDesc(tbItemCfg.nSkillId, tbItemCfg.nSkillLv)
    table.insert(tbSpecialTips, i18n.Parse(szNowLevelDesc))
  end
  return tbBaseAttrTips, tbExeAttrTips, tbSpecialTips, tbLianHuaTips
end

function tbItem:GetChatLinkInfo(pEquip)
  if pEquip.szClass ~= "ZhenYuan" then
    return
  end
  local tbInfo = {}
  tbInfo.tbRandAttr = Item.ZhenYuan:GetEquipRandomAttr(pEquip)
  tbInfo.tbMinAndMaxAttr = Item.ZhenYuan:GetMaxAndMinAttr(pEquip)
  tbInfo.nPower = Item.ZhenYuan:GetEquipPower(pEquip)
  return tbInfo
end

function tbItem:GetTipByTemplate(nTemplateId, tbLinkInfo)
  local tbBaseAttrTips = {}
  local tbExeAttrTips = {}
  local tbSpecialTips = {}
  local tbInfo = KItem.GetEquipBaseProp(nTemplateId)
  local tbAttrib = tbInfo.tbBaseAttrib
  local nQuality = Item:GetQuality(nTemplateId)
  local szColor = Item:GetQualityColor(nQuality)
  for i, tbMA in ipairs(tbAttrib) do
    local szDesc = tbEquip:GetMagicAttribDesc(tbMA.szName, tbMA.tbValue, tbMA.nActiveReq)
    szDesc = string.format("<color=#%s>%s</color>", Ui.tbColor:GetColorHexByDefineName(szColor), szDesc)
    table.insert(tbBaseAttrTips, szDesc)
  end
  if not tbLinkInfo or not tbLinkInfo.tbRandAttr then
    local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[nTemplateId]
    if tbItemCfg then
      table.insert(tbExeAttrTips, i18n.EasyGet("<color=#{0}>\233\154\143\230\156\186\232\142\183\229\190\151{1}\230\157\161\229\177\158\230\128\167,\229\177\158\230\128\167\229\128\188\228\184\141\229\155\186\229\174\154</color>", Ui.tbColor:GetColorHexByDefineName("NewGreen"), tbItemCfg.nExtAttrNum))
    end
    if tbItemCfg and tbItemCfg.nSkillId and tbItemCfg.nSkillId > 0 then
      local szNowLevelDesc = FightSkill:GetSkillMagicDesc(tbItemCfg.nSkillId, tbItemCfg.nSkillLv)
      table.insert(tbSpecialTips, i18n.Parse(szNowLevelDesc))
    end
    return tbBaseAttrTips, tbExeAttrTips, tbSpecialTips, {}
  end
  local tbRandAttr = tbLinkInfo.tbRandAttr
  local tbMinAndMaxAttr = tbLinkInfo.tbMinAndMaxAttr
  if next(tbRandAttr) then
    for i = 1, #tbRandAttr do
      local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRandAttr[i].szAttrType, tbRandAttr[i].tbAttrValue)
      local szMagic = ""
      if tbMinAndMaxAttr[szAttrType] then
        local _, szMinAttrValue = FightSkill:GetMagicDescSplit(tbRandAttr[i].szAttrType, tbMinAndMaxAttr[szAttrType][1])
        local _, szMaxAttrValue = FightSkill:GetMagicDescSplit(tbRandAttr[i].szAttrType, tbMinAndMaxAttr[szAttrType][2])
        local szMinStr = string.gsub(szMinAttrValue, "+", "")
        local szMaxStr = string.gsub(szMaxAttrValue, "+", "")
        szMagic = string.format("<color=#%s>%s : %s</color><color=#%s>  (%s - %s)</color>", Ui.tbColor:GetColorHexByDefineName(tbRandAttr[i].szColor), szAttrType, szAttrValue, Ui.tbColor:GetColorHexByDefineName("Gray"), szMinStr, szMaxStr)
        table.insert(tbExeAttrTips, szMagic)
      end
    end
  end
  local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[nTemplateId]
  if tbItemCfg and tbItemCfg.nSkillId and tbItemCfg.nSkillId > 0 then
    local szNowLevelDesc = FightSkill:GetSkillMagicDesc(tbItemCfg.nSkillId, tbItemCfg.nSkillLv)
    table.insert(tbSpecialTips, i18n.Parse(szNowLevelDesc))
  end
  return tbBaseAttrTips, tbExeAttrTips, tbSpecialTips, {}
end

function tbItem:CheckUsable(pItem)
  return 1
end
