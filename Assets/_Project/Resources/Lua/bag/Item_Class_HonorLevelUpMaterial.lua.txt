local tbItemClass = Item:GetClass("HonorLevelUpMaterial")

function tbItemClass:OnClientUse(pItem)
  local tbSetting = HonorLevel.LEVEL_CFG[HonorLevel:GetStarLevel(me) + 1]
  if not tbSetting then
    me.CenterMsg(i18n.EasyGet("\229\183\178\230\153\139\229\141\135\232\135\179\230\156\128\233\171\152\231\173\137\231\186\167"), false)
    return -1
  elseif tbSetting.ITEM_ID == nil or tbSetting.ITEM_ID ~= pItem.dwTemplateId then
    me.CenterMsg(i18n.EasyGet("\230\157\144\230\150\153\228\184\141\229\140\185\233\133\141\229\189\147\229\137\141\230\153\139\229\141\135\231\173\137\231\186\167"), false)
    return -1
  end
  Ui:OpenWindow("UIHonorLevel")
  return 0
end

function tbItemClass:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  
  local function fnClientUse()
    Item:ClientUseItem(nItemId)
    Ui:CloseWindow("UIItemTips")
  end
  
  local function fnExchange()
    Ui:CloseWindow("UIItemTips")
    if 1 == pItem.nCount then
      HonorLevel:ExchangeReq(nItemId, 1)
    else
      Ui:OpenWindow("UIItemBatchPanel", nItemId, nItemTemplateId)
    end
  end
  
  local tbSetting = {}
  if nItemId then
    tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
    tbSetting.fnFirst = fnClientUse
    if HonorLevel:CanExchange(me, nItemId, 1) then
      tbSetting.szSecondName = i18n.EasyGet("\229\133\145\230\141\162")
      tbSetting.fnSecond = fnExchange
    end
  end
  return tbSetting
end
