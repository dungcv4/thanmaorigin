Item.ExchangeItem = Item.ExchangeItem or {}
local ExchangeItem = Item.ExchangeItem
ExchangeItem.MAX_EXCHANGE_ELEMENT_NUM = 2

function ExchangeItem:InitConfig()
  local tbField = {
    "nParamId",
    "nId",
    "szExchangeList"
  }
  local tbExchangeList = Lib:LoadTabFileEx("Setting/Item/ExchangeItem.tab", "dds", nil, tbField)
  self.tbExchangeList = {}
  local nParamId
  for _, tbInfo in ipairs(tbExchangeList or {}) do
    nParamId = tbInfo.nParamId
    self.tbExchangeList[nParamId] = self.tbExchangeList[nParamId] or {}
    tbInfo.tbExchangeList = Lib:GetAwardFromString(tbInfo.szExchangeList)
    if #tbInfo.tbExchangeList == self.MAX_EXCHANGE_ELEMENT_NUM then
      table.insert(self.tbExchangeList[nParamId], tbInfo)
    end
  end
end

ExchangeItem:InitConfig()

function ExchangeItem:GetCfgById(nParamId)
  return self.tbExchangeList[nParamId]
end

function ExchangeItem:GetExchangeItemList(nParamId, nItemTemplateId)
  if not self.tbExchangeList[nParamId] then
    return {}
  end
  local tbExchangeList = {}
  local tbTmpList
  for _, tbInfo in ipairs(self.tbExchangeList[nParamId]) do
    tbTmpList = {}
    for _, tbDetailInfo in ipairs(tbInfo.tbExchangeList) do
      if tbDetailInfo[2] == nItemTemplateId then
        table.insert(tbTmpList, 1, tbDetailInfo)
      else
        table.insert(tbTmpList, tbDetailInfo)
      end
    end
    if #tbTmpList == self.MAX_EXCHANGE_ELEMENT_NUM then
      table.insert(tbExchangeList, {
        tbTmpList,
        tbInfo.nId
      })
    end
  end
  return tbExchangeList
end
