function Mantle:Init()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self.OnPlayerFirstLoginToday, self)
end

function Mantle:OnPlayerFirstLoginToday()
  local tbEquip = me.GetEquips(1)
  if tbEquip[Item.EQUIPPOS_MANTLE] and 0 ~= tbEquip[Item.EQUIPPOS_MANTLE] then
    return
  end
  local nHonorLevel = HonorLevel:GetStarLevel(me)
  if nHonorLevel < 1 then
    return
  end
  if not RedPointMgr:CheckEvent(RedPointMgr.EVENT.Mantle) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.Mantle)
  end
end

Mantle:Init()
