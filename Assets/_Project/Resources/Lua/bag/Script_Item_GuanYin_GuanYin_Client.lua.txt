Item.GuanYin = Item.GuanYin or {}
local GuanYin = Item.GuanYin

function GuanYin:CheckRedPoint()
  Log("GuanYin:CheckRedPoint")
  local pEquip = me.GetEquipByPos(Item.EQUIPPOS_GUANYIN)
  if pEquip then
    return
  end
  if not RedPointMgr:CheckEvent(RedPointMgr.EVENT.GuanYin) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.GuanYin)
  end
end

function GuanYin:ClickGuanYinBtn()
  local _, nVersion = Tong:GetMemberList()
  RemoteServer.OnTongRequest("SyncTongMemberInfo", nVersion)
  RemoteServer.RequestClickGuanYinBtn()
end
