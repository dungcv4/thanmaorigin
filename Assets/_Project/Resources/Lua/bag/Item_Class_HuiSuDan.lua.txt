local tbHuiSuDan = Item:GetClass("HuiSuDan")

function tbHuiSuDan:OnUse(pItem)
  local tbData = Marry:GetMarryData(me)
  if tbData.nWeddingState ~= Marry.WEDDING_STATE.CANREBOOOK or GetTime() < tbData.nWeddingTime + Wedding.STATE_INFO[0].nTime then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\228\184\141\229\143\175\232\161\165\229\138\158\229\169\154\231\164\188"))
    return 0
  end
  local pLover = KPlayer.GetPlayerObjById(tbData.dwLoverID)
  if not pLover then
    me.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\228\190\160\228\190\163\228\184\141\229\156\168\231\186\191"))
    return 0
  end
  if 0 == me.dwTeamID then
    me.CenterMsg(i18n.EasyGet("\232\175\183\228\184\142\230\130\168\231\154\132\228\190\160\228\190\163\231\187\132\233\152\159\228\189\191\231\148\168"))
    return 0
  end
  local tbMembers = TeamMgr:GetMembers(me.dwTeamID)
  if 2 ~= #tbMembers then
    me.CenterMsg(i18n.Get(Marry.Cfg.Msg.SEX_LIMIT))
    return 0
  end
  local nLoverIndex = 0
  for nIndex, dwPlayerId in ipairs(tbMembers) do
    if dwPlayerId == tbData.dwLoverID then
      nLoverIndex = nIndex
    end
  end
  if 0 == nLoverIndex then
    me.CenterMsg(i18n.EasyGet("\232\175\183\228\184\142\230\130\168\231\154\132\228\190\160\228\190\163\231\187\132\233\152\159\228\189\191\231\148\168"))
    return 0
  end
  Marry:SetReBookData(me)
  Marry:SetReBookData(pLover)
  me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\230\136\144\229\138\159!"))
  Log("HuiSuDan:OnUse", me.dwID)
  return 1
end
