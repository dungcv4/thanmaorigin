local tbItem = Item:GetClass("ActivePointItem")
tbItem.PARAM_ADDPOINT = 1

function tbItem:GetClientDefaultUseCount(nItemId)
  local pItem = me.GetItemInBag(nItemId)
  if not pItem or pItem.nCount <= 0 then
    me.CenterMsg(i18n.EasyGet("\229\175\185\228\184\141\232\181\183\239\188\140\230\130\168\230\178\161\230\156\137{0}", pItem.szName))
    return 0
  end
  local nCurActivePoint = ActivePoint:GetCount(me)
  local nMaxCanUseCount = self:GetMaxCanUseCount(pItem.dwTemplateId)
  if nCurActivePoint >= ActivePoint.Def.nMaxCount then
    me.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\230\180\187\232\183\131\229\186\166\229\183\178\230\187\161"))
    return 0
  else
    return nMaxCanUseCount
  end
end

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  local nAddPointPerUse = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_ADDPOINT)
  local nCurActivePoint = ActivePoint:GetCount(me)
  local nMaxCanUseCount = self:GetMaxCanUseCount(it.dwTemplateId)
  if nCurActivePoint >= ActivePoint.Def.nMaxCount then
    me.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\230\180\187\232\183\131\229\186\166\229\183\178\230\187\161"))
    return 0
  elseif nMaxCanUseCount <= 0 then
    local szName = KItem.GetItemShowInfo(nItemId)
    me.CenterMsg(i18n.EasyGet("\229\175\185\228\184\141\232\181\183\239\188\140\230\130\168\230\178\161\230\156\137{0}", szName))
    return 0
  else
    local nMaxUse = math.min(nMaxCanUseCount, nCount)
    ActivePoint:AddCount(me, nMaxUse * nAddPointPerUse)
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\230\136\144\229\138\159\239\188\140\230\130\168\229\162\158\229\138\160\228\186\134{0}\231\130\185\230\180\187\232\183\131\229\186\166", ActivePoint:GetCount(me) - nCurActivePoint))
    return nMaxUse
  end
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "ActivePointItem")
end

function tbItem:GetMaxCanUseCount(nItemTemplateId)
  local nAddPointPerUse = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_ADDPOINT)
  local nCurActivePoint = ActivePoint:GetCount(me)
  local nHasCount = me.GetItemCountInBags(nItemTemplateId)
  local nCanUseCount = math.ceil((ActivePoint.Def.nMaxCount - nCurActivePoint) / nAddPointPerUse)
  return math.min(nHasCount, nCanUseCount)
end

function tbItem:IsOverFlow(it, nCount)
  local nAddPointPerUse = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_ADDPOINT)
  local nCurActivePoint = ActivePoint:GetCount(me)
  return nAddPointPerUse * nCount + nCurActivePoint > ActivePoint.Def.nMaxCount
end
