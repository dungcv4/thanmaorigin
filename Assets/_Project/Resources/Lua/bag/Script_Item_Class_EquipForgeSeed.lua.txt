local tbItem = Item:GetClass("EquipForgeSeed")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function fnIdentify()
    Ui:OpenWindow("UISelectSeries", nItemId, function(nSeries, nCount)
      Item.EquipForge:IdentifyReq(me, nItemId, nSeries, nCount)
    end)
  end
  
  local tbSetting = {}
  tbSetting.szFirstName = i18n.EasyGet("\233\137\180\229\174\154")
  tbSetting.fnFirst = fnIdentify
  return tbSetting
end

function tbItem:GetPriceTips(pPlayer, nItemTemplateId)
  local EquipForge = Item.EquipForge
  local tbTips = {}
  local nEquipTemplateId = KItem.GetItemExtParam(nItemTemplateId, EquipForge.EXT_PARAM_SEED_TO_EQUIP)
  local tbEquipProp = Item:GetItemBaseProp(nEquipTemplateId)
  if not tbEquipProp then
    return
  end
  local nPos = KItem.GetEquipPos(nEquipTemplateId)
  if not nPos then
    return
  end
  local tbItemProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbItemProp then
    return
  end
  local nQuality = EquipForge:GetQuality(tbItemProp)
  local nCost = EquipForge.IDENTIFY_COST[nPos] and EquipForge.IDENTIFY_COST[nPos][nQuality] and EquipForge.IDENTIFY_COST[nPos][nQuality][tbEquipProp.nLevel] or nil
  return nCost
end
