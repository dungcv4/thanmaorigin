Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
local SecretBook = Item.SecretBook
local XiuWei = XiuWei or {}

function XiuWei:GetValue(pPlayer)
  local nOverValue = pPlayer.GetUserValue(SecretBook.GROUP_KEY, SecretBook.tbSaveKey.XiuWeiValueIfOverLimit)
  local nBeginTime = pPlayer.GetUserValue(SecretBook.GROUP_KEY, SecretBook.tbSaveKey.RestoreBeginTime)
  return self:CalcXiuWei(nBeginTime, nOverValue, self:GetRestoreOneCostTime())
end

function XiuWei:CalcXiuWei(nBeginTime, nOverValue, nOnePointCostTime)
  if nOverValue >= XiuWei.Cfg.nXiuWeiUpperLimit then
    return nOverValue
  else
    local nPassedTime = GetTime() - nBeginTime
    local nAlreadyRestorePoint = math.floor(nPassedTime / nOnePointCostTime)
    return math.min(nAlreadyRestorePoint, self.Cfg.nXiuWeiUpperLimit)
  end
end

function XiuWei:CalcXiuWeiOld(nEndTime, nOverValue)
  if nOverValue >= SecretBook.tbDataReconstruct.nLimitXiuWei then
    return nOverValue
  else
    local nFullTime = SecretBook.tbDataReconstruct.nLimitXiuWei * SecretBook.tbDataReconstruct.nRecordIncreaseOneCostTime
    local nHaveRestoreTimeFromZero = nFullTime - (nEndTime - GetTime())
    local nAlreadyRestorePoint = math.floor(nHaveRestoreTimeFromZero / SecretBook.tbDataReconstruct.nRecordIncreaseOneCostTime)
    return math.min(nAlreadyRestorePoint, SecretBook.tbDataReconstruct.nLimitXiuWei)
  end
end

function XiuWei:CheckCanXiuWeiSit(pPlayer)
  if DegreeCtrl:GetDegree(pPlayer, "XiuWeiSit") <= 0 then
    return false, i18n.Get(self.tbText.szLackSitTimes)
  end
  local tbCfg = self:GetRandomTable()
  if Lib:CountTB(tbCfg) <= 0 then
    return false, i18n.EasyGet("\228\184\141\230\187\161\232\182\179\230\157\161\228\187\182")
  end
  return true
end

function XiuWei:GetRestoreOneCostTime()
  local nCostTime = self.tbRestoreOneXiuWeiTime[1].nCostTime
  for _, tbEntry in ipairs(self.tbRestoreOneXiuWeiTime) do
    if 1 ~= GetTimeFrameState(tbEntry.szOpenTimeFrame) then
      break
    end
    nCostTime = tbEntry.nCostTime
  end
  return nCostTime
end

function XiuWei:GetRandomTable()
  local tbTmp = {}
  for _, tbRandom in ipairs(self.tbSitAddXiuWeiProbability) do
    if 1 ~= GetTimeFrameState(tbRandom.szOpenTimeFrame) then
      break
    end
    tbTmp = tbRandom
  end
  local tbRet = {}
  for _, tbKV in pairs(tbTmp.tbRandom or {}) do
    tbRet[tbKV[1]] = tbKV[2]
  end
  return tbRet
end
