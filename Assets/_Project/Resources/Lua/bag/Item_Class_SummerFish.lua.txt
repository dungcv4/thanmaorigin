Require("CommonScript/Activity/SummerActivity/SummerFishing.lua")
local tbItem = Item:GetClass("SummerFish")

function tbItem:OnCreate(pItem)
end

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  local tbBaseProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbBaseProp then
    return
  end
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  else
    return nil
  end
end

function tbItem:OnAutoUse(pPlayer, nItemTemplateId, nCount)
  pPlayer.ConsumeItemInBag(nItemTemplateId, nCount, Env.LogWay_UseItem)
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.SummerFishing)
  if pInstance then
    local nFishWeight = pPlayer.GetUserValue(pInstance.nSaveId, pInstance.SaveID.FishWeightHistory)
    pPlayer.CallClientScript("Ui:AddShowAwards", {
      [1] = "Item",
      [2] = nItemTemplateId,
      [3] = nCount,
      [4] = {nWeight = nFishWeight}
    })
  end
  return true
end

function tbItem:OnUse(it)
  return self:OnUseMany(it, 1)
end

function tbItem:OnUseMany(it, nCount)
  return nCount
end

function tbItem:GetTip(tbInfo)
  if tbInfo and tbInfo.tbItemExtParams and tbInfo.tbItemExtParams.nWeight then
    return i18n.EasyGet("\232\191\153\230\157\161\233\177\188\230\156\137{0}\230\150\164\233\135\141", tbInfo.tbItemExtParams.nWeight), ""
  end
end
