local tbWnd = Ui:CreateWindow("UIItemBuyAndUse")
tbWnd.tbControls = {
  txtTitle = "imgBG/txtTitle",
  txtDetail = "imgBG/txtDetail",
  txtHaveValue = "imgBG/txtHaveValue",
  Btn1 = "imgBG/btn1",
  Btn2 = "imgBG/btn2",
  Btn3 = "imgBG/btn3",
  btnClose = "imgBG/btnClose"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnItemChange,
      self
    },
    {
      EventNotify.emNOTIFY_DEL_ITEM,
      self.OnItemChange,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.Item = Ui:NewExistItemGrid(self, "imgBG/UIItemGrid")
end

function tbWnd:OnOpen(nTemplateId, szTitle, tbBtns)
  self.nTemplateId = nTemplateId
  self.szTitle = szTitle
  self.tbBtns = tbBtns
  if not nTemplateId or not tbBtns then
    Lib:LogFormat("Open ItemBuyAndUsePanel error, param error: nTemplateId, tbBtns = %s, %s", tostring(nTemplateId), tostring(tbBtns))
    return
  end
  for i = 1, 3 do
    local szBtn = string.format("Btn%s", i)
    self.pPanel:SetActive(self[szBtn], false)
  end
  self:Show()
end

function tbWnd:Show()
  self:ShowTitle()
  self:ShowItemInfo()
  self:ShowButtons()
end

function tbWnd:ShowTitle()
  if self.szTitle then
    self.pPanel:Label_SetText(self.txtDetail, self.szTitle)
  end
  local szName = KItem.GetItemShowInfo(self.nTemplateId, me.nFaction, me.nFactionSect, me.GetNpc().GetBodySex())
  self.pPanel:Label_SetText(self.txtTitle, szName or "")
end

function tbWnd:ShowItemInfo()
  self.Item:SetItemByTemplate(self.nTemplateId)
  self.Item:SetClickEnable(true)
  self:UpdateItemCount()
end

function tbWnd:UpdateItemCount()
  local nCount = me.GetItemCountInBags(self.nTemplateId)
  self.pPanel:Label_SetText(self.txtHaveValue, nCount)
end

function tbWnd:ShowButtons()
  for i, v in pairs(self.tbBtns) do
    local szBtn = string.format("Btn%s", i)
    local szDesc = v.szDesc
    local fnOnClick = v.fnOnClick
    self.pPanel:Label_SetText(self[szBtn] .. "/Text", szDesc)
    Ui:Button_BindEvent(self.pPanel, self[szBtn], function()
      fnOnClick(self.nTemplateId)
    end)
    self.pPanel:SetActive(self[szBtn], true)
  end
end

function tbWnd:OnItemChange(dwItemId, dwTemplateId)
  if dwTemplateId == self.nTemplateId then
    self:UpdateItemCount()
  else
    local pItem = me.GetItemInAllSpace(dwItemId)
    if pItem and pItem.dwTemplateId == self.nTemplateId then
      self:UpdateItemCount()
    end
  end
end

function tbWnd:OnItemDel(dwItemId, dwTemplateId)
  if dwTemplateId == self.nTemplateId then
    self:UpdateItemCount()
  end
end
