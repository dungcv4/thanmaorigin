Item.Obj = Item.Obj or {}
local Obj = Item.Obj
Obj.tbObj2NpcList = Obj.tbObj2NpcList or {}

function Item:OnObjAdd(...)
  Obj:OnAdd(...)
end

function Item:OnObjDel(...)
  Obj:OnDel(...)
end

function Obj:OnAdd(nObjId, nBuffId, nMapId, nPosX, nPosY)
  self:OnDel(nObjId)
  local nNpcTid = self.tbBuffList[nBuffId] and self.tbBuffList[nBuffId].nObjId or nil
  if not nNpcTid then
    return
  end
  local pNpc = KNpc.Add(nNpcTid, 1, Npc.emSERIES_TYPE.SERIES_NONE, nMapId, nPosX, nPosY, 0)
  if not pNpc then
    return
  end
  local nShowBuffId = self.tbBuffList[nBuffId].nShowBuffId
  if nShowBuffId and nShowBuffId > 0 then
    pNpc.AddSkillState(nShowBuffId, 1, 0, Env.GAME_FPS * self.tbBuffList[nBuffId].nDeathTime)
  end
  self.tbObj2NpcList[nObjId] = pNpc.nId
end

function Obj:OnDel(nObjId)
  local nNpcId = self.tbObj2NpcList[nObjId]
  if not nNpcId then
    return
  end
  self.tbObj2NpcList[nObjId] = nil
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc then
    return
  end
  pNpc.Delete()
end
