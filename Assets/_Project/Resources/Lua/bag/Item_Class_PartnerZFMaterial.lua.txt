local tbItem = Item:GetClass("PartnerZFMaterial")
tbItem.PARAM_TYPE = 1

function tbItem:OnClientUse(pItem)
  if not PartnerZhenFa:CanShowUIEntrance() then
    me.CenterMsg(i18n.Get(PartnerZhenFa.tbText.SysNotOpen))
    return -1
  end
  local nPanelType = KItem.GetItemExtParam(pItem.dwTemplateId, self.PARAM_TYPE)
  Ui:OpenWindow("UIPartnerNew", 5, 0, nPanelType)
  return 0
end
