Item.FixedTimeFramBox = Item.FixedTimeFramBox or {}
local FixedTimeFramBox = Item.FixedTimeFramBox

function FixedTimeFramBox:InitConfig()
  local tbField = {"nParamId", "szDateTime"}
  local tbDateTime = Lib:LoadTabFileEx("Setting/Item/FixedTimeFrameBox.tab", "ds", "nParamId", tbField)
  self.tbDateTime = {}
  for nParamId, tbInfo in pairs(tbDateTime or {}) do
    self.tbDateTime[nParamId] = Lib:ParseDateTime(tbInfo.szDateTime)
  end
end

FixedTimeFramBox:InitConfig()

function FixedTimeFramBox:GetTime(nParamId)
  return self.tbDateTime[nParamId]
end
