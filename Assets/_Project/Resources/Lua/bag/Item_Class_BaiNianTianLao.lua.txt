BaiNianTianLao.tbYeWaiMapTemplateId = {
  14,
  31,
  89,
  90
}

function BaiNianTianLao:CheckMapTemplateId()
  for _, value in pairs(BaiNianTianLao.tbYeWaiMapTemplateId) do
    if value == me.nMapTemplateId then
      return true
    end
  end
  return false
end

local tbItem = Item:GetClass("BaiNianTianLao")
tbItem.tbSetting = {nOpenTime = 3}

function tbItem:OnClientUse(pItem)
  if not BaiNianTianLao:CheckMapTemplateId() then
    local Count = #BaiNianTianLao.tbYeWaiMapTemplateId
    local nIndex = MathRandom(1, Count)
    Map:SwitchMap(BaiNianTianLao.tbYeWaiMapTemplateId[nIndex])
    Ui:CloseWindow("UIBag")
    Ui:CloseWindow("UIItemTips")
    Ui:OpenQuickUseItem(pItem.dwId, i18n.EasyGet("\228\189\191\231\148\168"))
    return 0
  end
  return 1
end

function tbItem:OnUse(it)
  if not it.dwTemplateId then
    return
  end
  local tbBaseInfo = Item:GetItemBaseProp(it.dwTemplateId)
  local nRequireLevel = tbBaseInfo.nRequireLevel
  if nRequireLevel > me.nLevel or BaiNianTianLao.nLevelLimit > me.nLevel then
    me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\233\152\133\229\142\134\229\176\154\230\181\133\239\188\140\233\156\128\232\166\129\231\173\137\231\186\167\232\190\190\229\136\176{0}\231\186\167", nRequireLevel))
    return 0
  end
  local bRet, szMsg = ActivityCheck:CheckCanEnterActivityEx(me)
  if not bRet then
    me.CenterMsg(szMsg)
    return 0
  end
  
  local function funcCloseUI()
    me.CallClientScript("Ui:CloseWindow", "UIBag")
    me.CallClientScript("Ui:CloseWindow", "UIItemTips")
  end
  
  if me.nMapTemplateId == BaiNianTianLao.nMainMapTemplateId then
    funcCloseUI()
    me.CenterMsg(i18n.EasyGet("\229\175\185\228\184\141\232\181\183\239\188\140\232\151\143\229\174\157\230\180\158\229\143\170\232\131\189\229\156\168\230\140\135\229\174\154\229\156\176\229\155\190\229\188\128\229\144\175\227\128\130"))
    return 0
  end
  if Map:IsFuben(me.nMapTemplateId) then
    me.CenterMsg(i18n.EasyGet("\232\175\165\229\156\186\230\153\175\230\151\160\230\179\149\228\189\191\231\148\168\239\188\140\232\175\183\229\156\168\229\159\142\229\184\130\230\136\150\232\128\133\233\135\142\229\164\150\229\156\176\229\155\190\228\189\191\231\148\168\227\128\130"))
    return 0
  end
  if not BaiNianTianLao:CheckMapTemplateId() then
    funcCloseUI()
    me.CallClientScript("Ui:OpenQuickUseItem", it.dwId, i18n.EasyGet("\228\189\191\231\148\168"))
    return 0
  end
  if me.nFightMode == Npc.FIGHT_MODE.emFightMode_None then
    funcCloseUI()
    me.CenterMsg(i18n.EasyGet("\232\175\183\231\167\187\230\173\165\229\136\176\230\156\172\229\156\176\229\155\190\230\136\152\230\150\151\229\140\186\229\159\159\229\188\128\229\144\175\232\151\143\229\174\157\230\180\158\239\188\129"))
    me.CallClientScript("Ui:OpenQuickUseItem", it.dwId, i18n.EasyGet("\228\189\191\231\148\168"))
    return
  end
  local nMapId, nX, nY = me.GetWorldPos()
  local bRet, szTip = BaiNianTianLao:CheckMapCanOpenCangBaoDong(nMapId, me.dwID)
  if not bRet then
    me.CallClientScript("Ui:CloseWindow", "UIItemTips")
    me.CenterMsg(szTip)
    return 0
  end
  funcCloseUI()
  Progress:BeginProgress(me, i18n.EasyGet("\232\151\143\229\174\157\230\180\158\229\188\128\229\144\175\228\184\173"), self.tbSetting.nOpenTime, self.OnEndProgress, {
    self,
    me.dwID,
    it.dwId
  })
  return 0
end

function tbItem:OnEndProgress(dwID, nItemId)
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local pPlayer = KPlayer.GetPlayerObjById(dwID)
  if not pPlayer then
    return
  end
  local nConsumeCount = pPlayer.ConsumeItem(pItem, 1, Env.LogWay_BaiNianTianLao)
  if 1 ~= nConsumeCount then
    return
  end
  pPlayer.CallClientScript("Ui:CloseWindow", "UIQuickUseItem")
  BaiNianTianLao:CreateCangBaoDongNpc()
end

function tbItem:GetTip(pItem)
  local nMaxUseCount = DegreeCtrl:GetMaxDegree("BaiNianTiaoLao", me)
  local nHaveCount = DegreeCtrl:GetDegree(me, "BaiNianTiaoLao")
  local szUseCount = i18n.EasyGet("\229\183\178\229\141\143\229\138\169\230\172\161\230\149\176\239\188\154{0}/{1}", nMaxUseCount - nHaveCount, nMaxUseCount)
  return szUseCount
end
