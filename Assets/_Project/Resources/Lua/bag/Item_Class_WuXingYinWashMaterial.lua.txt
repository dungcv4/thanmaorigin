local tbItemClass = Item:GetClass("WuXingYinWashMaterial")

function tbItemClass:OnClientUse(pItem)
  local pEquip = me.GetEquipByPos(Item.EQUIPPOS_WUXINGYIN)
  if not pEquip then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\232\163\133\229\164\135\228\186\148\232\161\140\229\141\176"), false)
    return -1
  end
  local nPanelType = KItem.GetItemExtParam(pItem.dwTemplateId, 1)
  if not Item.WuXingYin:IsOpen(me, nPanelType) then
    me.CenterMsg(i18n.EasyGet("\231\179\187\231\187\159\229\176\154\230\156\170\229\188\128\229\144\175"), false)
    return -1
  end
  Ui:OpenWindow("UIWuxingyin", nPanelType, pEquip.dwId)
  return 0
end
