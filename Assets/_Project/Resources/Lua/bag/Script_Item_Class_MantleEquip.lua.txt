Require("CommonScript/Item/Class/Equip.lua")
local tbMantleEquip = Item:GetClass("MantleEquip", 1)
tbMantleEquip = tbMantleEquip or Item:NewClass("MantleEquip", "equip")

function tbMantleEquip:OpenQuickUseItem(nItemId)
  local tbUsingEquips = me.GetEquips(1)
  local nMantleId = tbUsingEquips[Item.EQUIPPOS_MANTLE]
  local pUsingMantle
  if nMantleId and nMantleId > 0 then
    pUsingMantle = KItem.GetItemObj(nMantleId)
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  if pUsingMantle and pItem.nLevel <= pUsingMantle.nLevel then
    return
  end
  Ui:OpenQuickUseItem(nItemId, i18n.EasyGet("\231\169\191\228\184\138"), nil, function(tbUIQuickUseItem)
    Item.Equip:UseEquip(false, nItemId)
    tbUIQuickUseItem:Close()
  end)
end
