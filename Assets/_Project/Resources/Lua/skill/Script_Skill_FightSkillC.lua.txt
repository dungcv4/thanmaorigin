function FightSkill:SplitMagicDesc()
  local tbSkillMagicList = {}
  local tbBuffMagicList = {}
  for SkillId, tbInfo in pairs(FightSkill.tbSkillDesc) do
    local tbSkillMagicName, tbBuffMagicName
    if tbInfo.SkillMagicDesc then
      tbSkillMagicName = Lib:SplitStr(i18n.Parse(tbInfo.SkillMagicDesc), "|")
      tbSkillMagicList[SkillId] = tbSkillMagicName
    end
    if tbInfo.BuffMagicDesc ~= "" then
      tbBuffMagicName = Lib:SplitStr(i18n.Parse(tbInfo.BuffMagicDesc), "|")
      tbBuffMagicList[SkillId] = tbBuffMagicName
    elseif tbInfo.SkillMagicDesc then
      tbBuffMagicList[SkillId] = tbSkillMagicName
    end
  end
  for SkillId, tbInfo in pairs(FightSkill.tbSkillDesc) do
    tbInfo.SkillMagicDesc = tbSkillMagicList[SkillId]
    tbInfo.BuffMagicDesc = tbBuffMagicList[SkillId]
  end
end

function FightSkill:GetSkillMagicDesc(nSkillId, nSkillLevel)

  if not tbSetting then
    return ""
  end
  local szMagicDesc = FightSkill:GetSkillMagicDescEx(nSkillId, nSkillLevel)
  return szMagicDesc, tbSetting.ClassName
end

function FightSkill:GetSkillMagicDescEx(nSkillId, nSkillLevel, tbSkillMagic, nLastSkillId)
  FightSkill:TrySplitMagicDesc()
  local tbMagicSeq = {}
  local tbMagicDesc = {}
  local szDesc
  local bIgnore = false
  if not tbSkillMagic then
    tbSkillMagic = KFightSkill.GetSkillAllMagic(nSkillId, nSkillLevel)
    for _, tbMA in pairs(tbSkillMagic) do
      for i = 1, 3 do
        tbMA.tbValue[i] = tbMA.tbValue[i] / 100
      end
    end
    tbMagicSeq = FightSkill.tbSkillDesc[nSkillId].SkillMagicDesc
  else
    tbMagicSeq = FightSkill.tbSkillDesc[nSkillId].BuffMagicDesc
    bIgnore = true
  end
  for key, szAttribName in ipairs(tbMagicSeq) do
    local _, _, szText = string.find(szAttribName, "<(.+)>")
    local _, _, szFn = string.find(szAttribName, "{(.+)}")
    if szText then
      szDesc = szText .. "\n"
    elseif szFn then
      local tbParam = {}
      local szToolFunc
      _, _, szToolFunc, tbParam[1], tbParam[2] = string.find(szFn, "(.+):(.+);(.+)")
      if not szToolFunc then
        _, _, szToolFunc, tbParam[1] = string.find(szFn, "(.+):(.+)")
      end
      if FightSkill.tbMagic2Fn[szToolFunc] then
        szDesc = FightSkill.tbMagic2Fn[szToolFunc](nSkillId, nSkillLevel, nLastSkillId, table.unpack(tbParam))
      end
    elseif FightSkill.tbMagic2Fn[szAttribName] then
      szDesc = FightSkill.tbMagic2Fn[szAttribName](nSkillId, nSkillLevel, nLastSkillId)
    else
      szDesc = FightSkill.tbMagic2Fn.Default(nSkillId, nSkillLevel, szAttribName, tbSkillMagic, bIgnore)
    end
    table.insert(tbMagicDesc, szDesc)
  end
  local result = table.concat(tbMagicDesc)
  if not nLastSkillId then
    result = "\n" .. result
  end
  return result
end

function FightSkill:GetMagicShieldDesc(nSkillId, nSkillLevel)
  local tbStateEffect = FightSkill:GetStateEffectBySkill(nSkillId, nSkillLevel)
  if not tbStateEffect then
    return ""
  end
  local szMagicDesc = FightSkill:GetSkillMagicDescEx(nSkillId, nSkillLevel)
  local tbSkillMagic = KFightSkill.GetSkillAllMagic(nSkillId, nSkillLevel)
  local tbMagic = {
    0,
    0,
    0
  }
  for _, tbInfo in pairs(tbSkillMagic) do
    for i = 1, 3 do
      tbInfo.tbValue[i] = tbInfo.tbValue[i] / 100
    end
    if tbInfo.szName == "magicshield" then
      tbMagic = tbInfo.tbValue
    end
  end
  return string.format(szMagicDesc, math.floor(tbMagic[1] * me.nDexterity / 100))
end

function FightSkill:GetSkillStateMagicDesc(nSkillId, nSkillLevel, tbSkillMagic)
  local tbStateEffect = FightSkill:GetStateEffectBySkill(nSkillId, nSkillLevel)
  if not tbStateEffect then
    return ""
  end
  local funDescFun = self.tbStateFunDesc[nSkillId]

    return funDescFun(self, nSkillId, nSkillLevel)
  end
  local szMagicDesc = FightSkill:GetSkillMagicDescEx(nSkillId, nSkillLevel, tbSkillMagic)
  return szMagicDesc
end

function FightSkill:GetSkillShowTipInfo(nSkillId, nSkillLevel, nSkillMaxLevel)

  end

  local bMax = nSkillLevel >= nMaxLevel
  local szCurMagicDesc = FightSkill:GetSkillMagicDesc(nSkillId, nSkillLevel) or ""
  local szNextMagicDesc = nMaxLevel >= nSkillLevel + 1 and FightSkill:GetSkillMagicDesc(nSkillId, nSkillLevel + 1) or ""

    nId = nSkillId,
    nLevel = nSkillLevel,
    nMaxLevel = nMaxLevel,
    bMax = bMax,
    szIcon = tbSkillInfo.Icon or "",
    szIconAltlas = tbSkillInfo.IconAtlas or "",
    szName = tbSkillInfo.SkillName or "",
    szDesc = tbSkillInfo.Desc or "",
    szProperty = tbSkillInfo.Property or "",
    nCD = tbSkillInfo.TimePerCast or 0,
    bPassive = tbSkillInfo.SkillType == FightSkill.SkillTypeDef.skill_type_passivity,
    nRadius = tbSkillInfo.AttackRadius or 0,
    szCurMagicDesc = szCurMagicDesc or "",
    szNextMagicDesc = szNextMagicDesc or ""

function FightSkill:IsShowSkillStateIcon(nSkillId, nSkillLevel)
  local tbStateEffect = FightSkill:GetStateEffectBySkill(nSkillId, nSkillLevel)
  if not tbStateEffect then
    return false
  end
  if Lib:IsEmptyStr(tbStateEffect.Icon) then
    return false
  end
  return true
end

function FightSkill:IsShowSkillState(nSkillId, nSkillLevel)
  local tbStateEffect = FightSkill:GetStateEffectBySkill(nSkillId, nSkillLevel)
  if not tbStateEffect then
    return false
  end
  if Lib:IsEmptyStr(tbStateEffect.MagicDesc) or Lib:IsEmptyStr(tbStateEffect.Icon) then
    return false
  end
  return true
end

function FightSkill:FormatMagicDesc(nSkillId, nSkillLevel)
  local szCurMagicDesc, szNextMagicDesc, szClassName
  if nSkillLevel <= 0 then
    szCurMagicDesc = ""
    szNextMagicDesc, szClassName = self:GetSkillMagicDesc(nSkillId, 1)
  else
    szCurMagicDesc, szClassName = self:GetSkillMagicDesc(nSkillId, nSkillLevel)
    szNextMagicDesc = self:GetSkillMagicDesc(nSkillId, nSkillLevel + 1)
  end
  return szCurMagicDesc, szNextMagicDesc, szClassName
end

function FightSkill:GetPlayerOneSkillInfo(tbFactionSkill, nSortInUI, nFactionSect)
  local tbInfo = tbFactionSkill[nSortInUI]
  local nSkillId = tbInfo.SkillId
  local nSkillLevelTotal, nSkillLevel = 1, 1
  if me.nFactionSect == nFactionSect then
    nSkillLevelTotal, nSkillLevel = me.GetSkillLevel(nSkillId)
  end
  local nSkillInfoLevel = nSkillLevel
  if not nSkillInfoLevel or nSkillInfoLevel <= 0 then
    nSkillInfoLevel = 1
  end
  local tbSkillInfo = self:GetSkillSetting(nSkillId, nSkillInfoLevel)
  local nReqLevel, nCoin, nExp, nPoint = self:GetSkillLevelUpNeed(nSkillId, nSkillLevel)
  local bMax = nSkillLevel >= self:GetSkillMaxLevel(me, nSkillId)
  local szCurMagicDesc, szNextMagicDesc, szClassName = self:FormatMagicDesc(nSkillId, nSkillLevelTotal)
  local tbSkillListInfo = {
    nId = nSkillId,
    nLevel = nSkillLevelTotal,
    nReqLevel = nReqLevel,
    nCoin = nCoin,
    nExp = nExp,
    nPoint = nPoint,
    bMax = bMax,
    nSkillGroup = tbInfo.SkillGroup,
    szName = tbSkillInfo.SkillName or "",
    szDesc = tbSkillInfo.Desc or "",
    szProperty = tbSkillInfo.Property or "",
    nCD = tbSkillInfo.TimePerCast or 0,
    bPassive = tbSkillInfo.SkillType == self.SkillTypeDef.skill_type_passivity or tbSkillInfo.SkillType == self.SkillTypeDef.skill_type_inst_missile,
    nRadius = tbSkillInfo.AttackRadius or 0,
    nGainLevel = tbInfo.GainLevel,
    nSort = tbInfo.SortInUI,
    szBtnName = tbInfo.BtnName,
    szBtnIcon = tbInfo.BtnIcon,
    szIconAltlas = tbInfo.IconAltlas,
    szCurMagicDesc = szCurMagicDesc or "",
    szNextMagicDesc = szNextMagicDesc or "",
    szClassName = szClassName or ""
  }
  tbSkillListInfo.bCanLevelUp = nReqLevel <= me.nLevel and tbSkillListInfo.nLevel >= 1 and not tbSkillListInfo.bMax
  if tbSkillInfo.IsAura then
    tbSkillListInfo.bPassive = true
  end
  return tbSkillListInfo
end

function FightSkill:TrySplitMagicDesc()
  if not FightSkill.bSplit then
    FightSkill.bSplit = true
    FightSkill:SplitMagicDesc()
    SetReadOnly(FightSkill.tbSkillDesc, true)
  end
end

function FightSkill:GetSkillMagicDescAndValue(nSkillID, nSkillLevel)
  FightSkill:TrySplitMagicDesc()
  local tbResult = {}
  local tbState = KFightSkill.GetSkillAllMagic(nSkillID, nSkillLevel)
  local tbMagicSeq = FightSkill.tbSkillDesc[Marry.Cfg.nBuffSkillID].SkillMagicDesc
  for i, szAttribName in ipairs(tbMagicSeq) do
    for _, tbAttr in ipairs(tbState) do
      if tbAttr.szName == szAttribName then
        for k = 1, 3 do
          tbAttr.tbValue[k] = tbAttr.tbValue[k] / 100
        end
        local szFinalName, szFinalValue = FightSkill:GetMagicDescSplit(tbAttr.szName, tbAttr.tbValue)
        tbResult[i] = {szFinalName = szFinalName, szFinalValue = szFinalValue}
      end
    end
  end
  return tbResult
end

function FightSkill:CheckBaseSkillLeveUp(pPlayer, nSkillId, nBaseLevel)
  if pPlayer.IsMaskPlayerFactSkill() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\230\151\160\230\179\149\229\138\160\231\130\185")
  end
  if nBaseLevel < 0 then
    return false, i18n.EasyGet("\230\178\161\230\156\137\232\142\183\229\190\151\232\175\165\230\138\128\232\131\189")
  end
  local nFactionLimit = KFightSkill.GetFactionLimit(nSkillId)
  if pPlayer.nFaction ~= nFactionLimit and nFactionLimit > 0 then
    return false, i18n.EasyGet("\233\151\168\230\180\190\228\184\141\231\172\166\229\144\136")
  end
  local tbSkillUpInfo = FightSkill:GetSkillUpInfo(nSkillId, nBaseLevel + 1)
  if not tbSkillUpInfo then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167!")
  end
  local nReqLevel, _, _, nNeedPoint = FightSkill:GetSkillLevelUpNeed(nSkillId, nBaseLevel)
  local nSkillPoint = pPlayer.GetSkillPoint()
  if nNeedPoint > nSkillPoint then
    return false, i18n.EasyGet("\230\138\128\232\131\189\231\130\185\228\184\141\232\182\179")
  end
  if nReqLevel > pPlayer.nLevel then
    return false, i18n.EasyGet("\228\184\187\232\167\146\232\190\190\229\136\176{0}\231\186\167\239\188\140\230\137\141\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\230\173\164\230\138\128\232\131\189", nReqLevel)
  end
  if nBaseLevel >= FightSkill:GetSkillMaxLevel(pPlayer, nSkillId) then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167")
  end
  return true, "", nBaseLevel, nNeedPoint
end
