local tbWnd = Ui:CreateClass("UICommonSkill_SecretBook", "UIPanel")
Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
local SecretBook = Item.SecretBook
local tbControls = {
  MijiGrid1 = "MijiCell/MijiGrid1",
  MijiGrid2 = "MijiCell/MijiGrid2",
  MijiGrid3 = "MijiCell/MijiGrid3",
  MijiGrid4 = "MijiCell/MijiGrid4",
  MijiGrid5 = "MijiCell/MijiGrid5",
  SkillIcon1 = "SkillIcon1",
  SkillIcon2 = "SkillIcon2",
  SkillIcon3 = "SkillIcon3",
  SkillIcon4 = "SkillIcon4",
  SkillIcon5 = "SkillIcon5",
  MijiLevelPanel = "MijiLevelUp",
  NoWearPanel = "Empty",
  txtName = "MijiLevelUp/txtName",
  RightUIItemGrid = "MijiLevelUp/UIItemGrid",
  txtLevelValue = "MijiLevelUp/Level/txtLevelValue",
  txtSkillLevelValue = "MijiLevelUp/SkillLevel/txtSkillLevelValue",
  txtXiuWeiOrFullLevelTip = "MijiLevelUp/Exp/txtExpTip",
  txtXiuWeiValue = "MijiLevelUp/Exp/txtExpValue",
  btnLevelUp = "MijiLevelUp/btnLevelUp",
  btnChange = "MijiLevelUp/btnChange",
  txtTip1 = "MijiLevelUp/txtTip1",
  txtTip2 = "MijiLevelUp/txtTip2",
  txtTip3 = "MijiLevelUp/txtTip3",
  EffectLevelUP_1 = "MijiLevelUp/Effect/LevelUP_1",
  EffectLevelUP_5 = "MijiLevelUp/Effect/LevelUP_5"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnChange = function(self)
    if self.nNowSelectRank ~= nil then
      self:OpenSecretBookBag(self.nNowSelectRank)
    end
  end,
  btnLevelUp = function(self)
    self:OnClickLevelUp()
  end
}
tbWnd.tbSecretBookRankCfg = {
  [SecretBook.tbRank.Low] = {
    szGridPath = tbControls.MijiGrid1,
    szSkillPath = tbControls.SkillIcon1,
    bShowSlot = 1 == SecretBook.tbEachRankCfg[SecretBook.tbRank.Low].bEnableShowSlot
  },
  [SecretBook.tbRank.Middle] = {
    szGridPath = tbControls.MijiGrid2,
    szSkillPath = tbControls.SkillIcon2,
    bShowSlot = 1 == SecretBook.tbEachRankCfg[SecretBook.tbRank.Middle].bEnableShowSlot
  },
  [SecretBook.tbRank.High] = {
    szGridPath = tbControls.MijiGrid3,
    szSkillPath = tbControls.SkillIcon3,
    bShowSlot = 1 == SecretBook.tbEachRankCfg[SecretBook.tbRank.High].bEnableShowSlot
  },
  [SecretBook.tbRank.JiaLanJing] = {
    szGridPath = tbControls.MijiGrid4,
    szSkillPath = tbControls.SkillIcon4,
    bShowSlot = 1 == SecretBook.tbEachRankCfg[SecretBook.tbRank.JiaLanJing].bEnableShowSlot
  },
  [SecretBook.tbRank.JiuYinZhenJing] = {
    szGridPath = tbControls.MijiGrid5,
    szSkillPath = tbControls.SkillIcon5,
    bShowSlot = 1 == SecretBook.tbEachRankCfg[SecretBook.tbRank.JiuYinZhenJing].bEnableShowSlot
  }
}

function tbWnd:OnCreate()
  self.tbLeftItemGridList = {}
  local tbCfg = {}
  for nRank = 1, Lib:CountTB(SecretBook.tbRank) do
    tbCfg = self.tbSecretBookRankCfg[nRank]
    if tbCfg.bShowSlot then
      local szGridKey = tbCfg.szGridPath
      self.tbLeftItemGridList[nRank] = Ui:NewExistItemGrid(self, szGridKey .. "/UIItemGrid")
      Ui:Button_BindEvent(self.pPanel, szGridKey .. "/Empty", self.OnClickAddIcon, self, nRank)
      local szSkillIconKey = tbCfg.szSkillPath
      Ui:Button_BindEvent(self.pPanel, szSkillIconKey .. "/btnSkill", self.OnClickSkillIcon, self, nRank)
    end
    self.pPanel:SetActive(tbCfg.szGridPath, tbCfg.bShowSlot)
    self.pPanel:SetActive(tbCfg.szSkillPath, tbCfg.bShowSlot)
  end
  self:InitLeftPanel()
  self.tbRightItemGrid = Ui:NewExistItemGrid(self, tbControls.RightUIItemGrid)
  self.pPanel:SetActive(tbControls.RightUIItemGrid, true)
end

function tbWnd:OnOpen()
  self:UpdateLeftPanel()
  self:AutoSetSelectRankByOrder()
  self:HideEffect()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_FACTION_SECT, self.OnChangeFactionOrSect, self)
  for nRank, _ in pairs(SecretBook.tbLevelEnoughRedPointRank) do
    self:RegisterLevelEnoughRedPoint(nRank)
  end
  for nRank, _ in pairs(SecretBook.tbCanLevelUpRedPointRank) do
    self:RegisterLevelUpRedPoint(nRank)
  end
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_FACTION_SECT, self)
  for nRank = 1, Lib:CountTB(SecretBook.tbRank) do
    self:UnRegisterLevelEnoughRedPoint(nRank)
  end
  for nRank, _ in pairs(SecretBook.tbCanLevelUpRedPointRank) do
    self:UnRegisterLevelUpRedPoint(nRank)
  end
end

function tbWnd:InitLeftPanel()
  for nRank = 1, Lib:CountTB(SecretBook.tbRank) do
    if self.tbSecretBookRankCfg[nRank].bShowSlot then
      local tbItemGrid = self.tbLeftItemGridList[nRank]
      if tbItemGrid then
        tbItemGrid:SetClickEnable(true, self.OnClickLeftSecretBookGrid, self, nRank, tbItemGrid)
        tbItemGrid.szItemOpt = "PlayerEquip"
      end
    end
  end
  self:UpdateSkillIcon()
end

function tbWnd:UpdateLeftPanel()
  local tbCfg = {}
  for nRank = 1, Lib:CountTB(SecretBook.tbRank) do
    tbCfg = self.tbSecretBookRankCfg[nRank]
    if tbCfg.bShowSlot then
      local szGridKey = tbCfg.szGridPath
      local bTimeFrameIsOpenThisRank = 0 ~= GetTimeFrameState(SecretBook.tbEachRankCfg[nRank].szOpenTimeFrame)
      self.pPanel:SetActive(szGridKey, bTimeFrameIsOpenThisRank)
      local bSkillIconActive = false
      if bTimeFrameIsOpenThisRank then
        local pEquip = SecretBook:GetWearSecretBook(me, nRank)
        local bLevelEnough = me.nLevel >= SecretBook.tbEachRankCfg[nRank].nUiShowOpenLevel
        local bWear = nil ~= pEquip
        self.pPanel:SetActive(szGridKey .. "/UIItemGrid", bWear)
        self.pPanel:SetActive(szGridKey .. "/Empty", not bWear and bLevelEnough)
        self.pPanel:SetActive(szGridKey .. "/Lock", not bLevelEnough and not bWear)
        self.pPanel:Label_SetText(szGridKey .. "/Lock/txtLock", i18n.EasyGet("{0}\231\186\167\229\188\128\230\148\190", SecretBook.tbEachRankCfg[nRank].nUiShowOpenLevel))
        local tbItemGrid = self.tbLeftItemGridList[nRank]
        if bWear then
          tbItemGrid:SetItem(pEquip.dwId, pEquip.dwTemplateId, nil, me.nFaction, me.nFactionSect, me.nSex)
        end
        bSkillIconActive = bWear
      end
      local szSkillKey = tbCfg.szSkillPath
      self.pPanel:SetActive(szSkillKey, bSkillIconActive)
    end
  end
  self:UpdateSkillIcon()
end

function tbWnd:UpdateSkillIcon()
  local tbCfg = {}
  for nRank = 1, Lib:CountTB(SecretBook.tbRank) do
    tbCfg = self.tbSecretBookRankCfg[nRank]
    if tbCfg.bShowSlot then
      local szSkillKey = tbCfg.szSkillPath
      local nSkillId = SecretBook:GetSkillId(me.nFaction, me.nFactionSect, nRank)
      local szButtonKey = szSkillKey .. "/btnSkill"
      self.pPanel:Sprite_SetSprite(szButtonKey, FightSkill:GetSkillIconAtlas(nSkillId))
    end
  end
end

function tbWnd:OnClickAddIcon(nRank)
  if GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\230\151\160\230\179\149\232\191\155\232\161\140\230\173\164\230\147\141\228\189\156"))
    return
  end
  local nHaveSecretBookCount = #SecretBook:GetSecretBookInBag(nRank)
  if nHaveSecretBookCount <= 0 and not Task:IsTaskFinished(me, SecretBook.Cfg.nSecretBookSubTaskId) then
    Ui:OpenWindowWithCallback("UITask", function(_, tbWnd)
      tbWnd:SelectSubTaskInWindow(SecretBook.Cfg.nSecretBookSubTaskId)
    end)
    Ui:CloseWindow("UISkill")
  else
    self:OpenSecretBookBag(nRank)
  end
  SecretBook:FadeLevelEnoughRedPoint(nRank)
end

function tbWnd:OpenSecretBookBag(nRank)
  if GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\230\151\160\230\179\149\232\191\155\232\161\140\230\173\164\230\147\141\228\189\156"))
    return
  end
  local tbPItem = SecretBook:GetSecretBookInBag(nRank)
  if #tbPItem > 0 then
    Ui:OpenWindow("UIMijiBag", nRank, tbPItem)
  else
    local szSecretBookName = i18n.Parse(SecretBook.tbEquipCfgData[nRank].Name)
    
    local function fnBuy()
      Ui:OpenWindow("UIStore", Shop.ShopTypeStr.HunShi, Ui.UIStoreGood.tbHunShiSubPanelIndex.SecretBook)
    end
    
    Ui:OpenWindow("UIMessageBox", i18n.Get(SecretBook.tbText.szNoSecretBookTips, szSecretBookName), {
      {fnBuy},
      {}
    }, {
      i18n.Get(SecretBook.tbText.szBuy),
      i18n.szCancel
    })
  end
end

function tbWnd:OnClickLeftSecretBookGrid(nRank, tbItemGrid)
  local nOldSelectRank = self.nNowSelectRank
  self:UpdateSelectRank(nRank, true)
  if nRank == nOldSelectRank then
    tbItemGrid:DefaultClick()
    SecretBook:FadeCanLevelUpRedPoint(nRank)
  end
end

function tbWnd:OnClickSkillIcon(nRank)
  local nSkillId = SecretBook:GetSkillId(me.nFaction, me.nFactionSect, nRank)
  local pEquip = SecretBook:GetWearSecretBook(me, nRank)
  local nSkillLevel = SecretBook:GetSkillLevel(pEquip)
  local nExtraLevel = SecretBook:GetExtraSkillLevel(me, nRank)
  local tbSkillInfo = FightSkill.tbAllSkillSetting[nSkillId]
  local tbIncomeArgSkillInfo = {
    szIconAltlas = FightSkill:GetSkillIconAtlas(nSkillId),
    bMax = nSkillLevel < SecretBook.Cfg.nMaxSkillLevel,
    nLevel = nSkillLevel,
    nExtraLevel = nExtraLevel,
    szName = tbSkillInfo.szSkillName,
    szClassName = "",
    szNextMagicDesc = FightSkill:GetSkillMagicDesc(nSkillId, nSkillLevel + nExtraLevel + 1),
    szCurMagicDesc = FightSkill:GetSkillMagicDesc(nSkillId, nSkillLevel + nExtraLevel),
    nId = nSkillId,
    nCD = 0,
    bPassive = true,
    nRadius = 0,
    szDesc = tbSkillInfo.szDesc,
    nMaxLevel = SecretBook.Cfg.nMaxSkillLevel
  }
  Ui:OpenWindow("UISkillShow", tbIncomeArgSkillInfo, true, false)
end

function tbWnd:AutoSetSelectRankByOrder()
  local nStartSelectIndex
  for nRank = 1, Lib:CountTB(SecretBook.tbRank) do
    if self.tbSecretBookRankCfg[nRank].bShowSlot then
      local pEquip = SecretBook:GetWearSecretBook(me, nRank)
      if pEquip then
        nStartSelectIndex = nStartSelectIndex or nRank
        if SecretBook:GetLevel(pEquip) < SecretBook.Cfg.nMaxLevel then
          nStartSelectIndex = nRank
          break
        end
      end
    end
  end
  self:UpdateSelectRank(nStartSelectIndex, false)
end

function tbWnd:SetNowSelectRank(nValue)
  local function fnSetEffectActive(bActive)
    self.pPanel:SetActive(self.tbSecretBookRankCfg[self.nNowSelectRank].szGridPath .. "/imgSelected", bActive)
  end
  
  local bSelectChange = self.nNowSelectRank ~= nValue
  if bSelectChange then
    if self.nNowSelectRank ~= nil then
      fnSetEffectActive(false)
    end
    local nOldValue = self.nNowSelectRank
    self.nNowSelectRank = nValue
    if self.nNowSelectRank ~= nil then
      fnSetEffectActive(true)
    end
    local bOldHaveLevelUpRedPoint = nil ~= nOldValue and nil ~= SecretBook.tbCanLevelUpRedPointRank[nOldValue]
    local bNewHaveLevelUpRedPoint = nil ~= nValue and nil ~= SecretBook.tbCanLevelUpRedPointRank[nValue]
    if not bOldHaveLevelUpRedPoint and bNewHaveLevelUpRedPoint then
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUp, {
        RedPointMgr.EVENT.SecretBookCanLevelUp
      })
    elseif bOldHaveLevelUpRedPoint and not bNewHaveLevelUpRedPoint then
      RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUp)
    end
  end
end

function tbWnd:UpdateSelectRank(nRank, bIsClickCall)
  local nOldSelectRank = self.nNowSelectRank
  self:SetNowSelectRank(nRank)
  if nOldSelectRank ~= nRank or not bIsClickCall then
    self:UpdateRightPanel()
  end
end

function tbWnd:UpdateRightPanel()
  local pEquip
  local bWear = self.nNowSelectRank ~= nil
  if bWear then
    pEquip = SecretBook:GetWearSecretBook(me, self.nNowSelectRank)
  end
  bWear = bWear and nil ~= pEquip
  self.pPanel:SetActive(tbControls.MijiLevelPanel, bWear)
  self.pPanel:SetActive(tbControls.NoWearPanel, not bWear)
  if bWear then
    self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(pEquip.szName))
    local nLevel = SecretBook:GetLevel(pEquip)
    self.pPanel:Label_SetText(tbControls.txtLevelValue, string.format("%d/%d", nLevel, SecretBook.Cfg.nMaxLevel))
    local nSkillLevel = SecretBook:GetSkillLevel(pEquip)
    local nRank = SecretBook:GetRank(pEquip)
    local nExtraLevel = SecretBook:GetExtraSkillLevel(me, nRank)
    if nExtraLevel > 0 then
      self.pPanel:Label_SetText(tbControls.txtSkillLevelValue, i18n.EasyGet("{0}\239\188\136{1}+{2}\239\188\137/{3}", nSkillLevel + nExtraLevel, nSkillLevel, nExtraLevel, SecretBook.Cfg.nMaxSkillLevel))
    else
      self.pPanel:Label_SetText(tbControls.txtSkillLevelValue, i18n.EasyGet("{0}/{1}", nSkillLevel, SecretBook.Cfg.nMaxSkillLevel))
    end
    local bIsNotFullLevel = nLevel < SecretBook.Cfg.nMaxLevel
    self.pPanel:SetActive(tbControls.txtXiuWeiValue, bIsNotFullLevel)
    local bCanLevelUp
    if bIsNotFullLevel then
      local nLevelUpCostXiuWei = SecretBook:GetLevelUpCostXiuWei(self.nNowSelectRank, nLevel + 1)
      bCanLevelUp = nLevelUpCostXiuWei <= XiuWei:GetValue(me)
      self.pPanel:Label_SetText(tbControls.txtXiuWeiValue, string.format("%d/%d", XiuWei:GetValue(me), nLevelUpCostXiuWei))
      self.pPanel:Label_SetText(tbControls.txtXiuWeiOrFullLevelTip, i18n.Get(SecretBook.tbText.szNoFullLevelText))
      self.pPanel:Label_SetText(tbControls.txtTip1, i18n.Get(SecretBook.tbText.szLevelUp1Tips))
      self.pPanel:Label_SetText(tbControls.txtTip2, i18n.Get(SecretBook.tbText.szLevelUp5Tips))
    else
      bCanLevelUp = false
      self.pPanel:Label_SetText(tbControls.txtXiuWeiOrFullLevelTip, i18n.Get(SecretBook.tbText.szFullLevelText))
      self.pPanel:Label_SetText(tbControls.txtTip1, i18n.Get(SecretBook.tbText.szXiuWeiValueTips, XiuWei:GetValue(me)))
      self.pPanel:Label_SetText(tbControls.txtTip2, "")
    end
    self.pPanel:Label_SetText(tbControls.txtTip3, i18n.EasyGet("\230\175\1435\229\136\134\233\146\159\232\135\170\229\138\168\229\155\158\229\164\141{0}\231\130\185\228\191\174\228\184\186", self:GetXiuWeiRestoreTips()))
    self.pPanel:Button_SetEnable(tbControls.btnLevelUp, bCanLevelUp)
    self.tbRightItemGrid:SetItem(pEquip.dwId, pEquip.dwTemplateId, nil, me.nFaction, me.nFactionSect, me.nSex)
    self.tbRightItemGrid:SetClickEnable(true)
  end
end

function tbWnd:GetXiuWeiRestoreTips()
  local nShowPoint = XiuWei.tbRestoreOneXiuWeiTime[1].nShowPoint
  for _, tbEntry in ipairs(XiuWei.tbRestoreOneXiuWeiTime) do
    if 1 ~= GetTimeFrameState(tbEntry.szOpenTimeFrame) then
      break
    end
    nShowPoint = tbEntry.nShowPoint
  end
  return nShowPoint
end

function tbWnd:HideEffect()
  self.pPanel:SetActive(tbControls.EffectLevelUP_1, false)
  self.pPanel:SetActive(tbControls.EffectLevelUP_5, false)
end

function tbWnd:PlayEffect_AddAttribute()
  self.pPanel:SetActive(tbControls.EffectLevelUP_1, false)
  self.pPanel:SetActive(tbControls.EffectLevelUP_1, true)
end

function tbWnd:PlayEffect_SkillLevelUp()
  self.pPanel:SetActive(tbControls.EffectLevelUP_5, false)
  self.pPanel:SetActive(tbControls.EffectLevelUP_5, true)
end

function tbWnd:OnSyncSecretBook(pEquip)
  local nRank = SecretBook:GetRank(pEquip)
  self:UpdateLeftPanel()
  if SecretBook:IsWearThisRankSecretBook(me, nRank) then
    self:UpdateSelectRank(nRank, false)
  else
    self:AutoSetSelectRankByOrder()
  end
  self:HideEffect()
  self:UpdateRightPanel()
end

function tbWnd:OnSyncLevelUp()
  self:UpdateRightPanel()
end

function tbWnd:OnSyncXiuWeiValue(nOldValue, nNewValue)
  self:UpdateRightPanel()
end

function tbWnd:OnChangeFactionOrSect(nFaction, nFactionSect)
  self:UpdateSkillIcon()
end

function tbWnd:OnPlayerLevelUp()
  self:UpdateLeftPanel()
end

function tbWnd:RegisterLevelEnoughRedPoint(nRank)
  if self.tbSecretBookRankCfg[nRank].bShowSlot then
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.tbSecretBookRankCfg[nRank].szGridPath .. "/Empty", {
      RedPointMgr.EVENT.SecretBookLevelEnough
    })
  end
end

function tbWnd:UnRegisterLevelEnoughRedPoint(nRank)
  if self.tbSecretBookRankCfg[nRank].bShowSlot then
    RedPointMgr:DynamicUnRegist(self.pPanel, self.tbSecretBookRankCfg[nRank].szGridPath .. "/Empty")
  end
end

function tbWnd:RegisterLevelUpRedPoint(nRank)
  if self.tbSecretBookRankCfg[nRank].bShowSlot then
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.tbSecretBookRankCfg[nRank].szGridPath .. "/UIItemGrid", {
      RedPointMgr.EVENT.SecretBookCanLevelUp
    })
    if self.nNowSelectRank == nRank then
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUp, {
        RedPointMgr.EVENT.SecretBookCanLevelUp
      })
    end
  end
end

function tbWnd:UnRegisterLevelUpRedPoint(nRank)
  if self.tbSecretBookRankCfg[nRank].bShowSlot then
    RedPointMgr:DynamicUnRegist(self.pPanel, self.tbSecretBookRankCfg[nRank].szGridPath .. "/UIItemGrid")
    if self.nNowSelectRank == nRank then
      RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUp)
    end
  end
end

function tbWnd:OnClickLevelUp()
  if GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
    me.CenterMsg(i18n.EasyGet("\230\138\128\232\131\189\229\141\135\231\186\167\229\164\177\232\180\165\239\188\140\232\175\183\232\191\148\229\155\158\229\142\159\230\156\141\230\147\141\228\189\156"))
    return
  end
  if self.nNowSelectRank ~= nil then
    RemoteServer.RequestSecretBook(SecretBook.tbC2SProtocol.LevelUp, self.nNowSelectRank)
    SecretBook:FadeCanLevelUpRedPoint(self.nNowSelectRank)
  end
end
