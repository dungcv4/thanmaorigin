Require("CommonScript/Item/SecretBook/SecretBookDef.lua")
local tbWnd = Ui:CreateClass("UICommonSkill", "UIPanel")
local tbControls = {
  btnMiji = "TapList/btnMiji",
  btnGuanxian = "TapList/btnGuanxian",
  panelMiji = "Miji",
  panelGuanxian = "Guanxian"
}
tbWnd.tbControls = tbControls
tbWnd.PANEL_TYPE_SECRETBOOK = 1
tbWnd.PANEL_TYPE_GUANXIAN = 2
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_TYPE_SECRETBOOK] = {
    CLASS = "UICommonSkill_SecretBook",
    OBJ = tbControls.panelMiji,
    BTN = tbControls.btnMiji,
    OpenFunc = function()
      return 0 ~= GetTimeFrameState(Item.SecretBook.Cfg.szUIOpenTimeFrame)
    end
  },
  [tbWnd.PANEL_TYPE_GUANXIAN] = {
    CLASS = "UICommonSkill_GuanXian",
    OBJ = tbControls.panelGuanxian,
    BTN = tbControls.btnGuanxian,
    OpenFunc = function()
      return true
    end
  }
}
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
end

function tbWnd:OnOpen(nSelectPanel, ...)
  if nil == nSelectPanel then
    for nIndex, tbPanelSetting in ipairs(self.PANEL_SETTING) do
      if tbPanelSetting.OpenFunc() then
        nSelectPanel = nIndex
        break
      end
    end
  end
  if nil ~= nSelectPanel then
    self.tbTabControl:OnOpen(nSelectPanel, ...)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnMiji, {
    RedPointMgr.EVENT.SecretBookLevelEnough,
    RedPointMgr.EVENT.SecretBookCanLevelUp
  })
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnMiji)
end
