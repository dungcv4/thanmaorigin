local tbWnd = Ui:CreateWindow("UILifeSkill")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  panelLiftSkill = "node/imgBG/Content/LifeSkillView",
  ToggleRareEquipment = "node/imgBG/Content/ToggleRareEquipment"
}
tbWnd.tbControls = tbControls
tbWnd.PANEL_TYPE_LIFESKILL = 1
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_TYPE_LIFESKILL] = {
    CLASS = "UILifeSkillView",
    OBJ = tbControls.panelLiftSkill,
    BTN = nil,
    OpenFunc = function()
      return true
    end
  }
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("LifeSkillHelp")
  end
}

function tbWnd:OnCreate()
  self.tbClass2Panel = {}
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL] = Ui:CreatePanel(tbWnd.PANEL_SETTING[tbWnd.PANEL_TYPE_LIFESKILL].CLASS, tbWnd.PANEL_SETTING[tbWnd.PANEL_TYPE_LIFESKILL].OBJ, self)
  local bIsDoubleRate = LifeSkill:GetDoubleRate()
  Ui:Toggle_UnBindEvent(self.pPanel, tbControls.ToggleRareEquipment)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleRareEquipment, bIsDoubleRate)
  Ui:Toggle_BindEvent(self.pPanel, tbControls.ToggleRareEquipment, function(self, isOn)
    self.pPanel:Toggle_SetChecked(tbControls.ToggleRareEquipment, isOn)
    LifeSkill:SetDoubleRate(isOn)
    self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:OnDoubleRateChange()
  end, self)
end

function tbWnd:OnOpen()
  self.pPanel:Button_SetEnable(tbControls.btnHelp, true)
  self.pPanel:SetActive(tbControls.ToggleRareEquipment, LifeSkill:IsDoubleRateOpen())
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:OnOpen()
end

function tbWnd:OpenLifeSkillIndex(nIndex)
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:SetLiftSkillSelect(nIndex)
end

function tbWnd:UpdateView()
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:OnModifySpecialty()
end

function tbWnd:OnClose()
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:OnClose()
end

function tbWnd:OnCraftResult(nProductTemplateId)
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:OnCraftResult(nProductTemplateId)
end

function tbWnd:OnSelectLifeSkillLevel(nLevel)
  self.tbClass2Panel[tbWnd.PANEL_TYPE_LIFESKILL]:SetLifeSkillLevel(nLevel)
end

function tbWnd:OnSetToggleRareEquipmentShow(bActive)
  self.pPanel:SetActive(tbControls.ToggleRareEquipment, bActive)
end
