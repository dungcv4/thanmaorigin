local tbWnd = Ui:CreateClass("UIOtherZongShiSkill", "UIPanel")
local tbControls = {
  txtTitle = "Master/txt_Tittle",
  txtTitleInfo = "Master/txt_Info",
  btnReset = "Master/btnReset",
  btnCenter = "Master/Scroll View/Viewport/Content/Template/Cont_Center/btnSkill%d",
  btn01 = "Master/Scroll View/Viewport/Content/Template/Cont_01/btnSkill%d",
  btn02 = "Master/Scroll View/Viewport/Content/Template/Cont_02/btnSkill%d",
  btn03 = "Master/Scroll View/Viewport/Content/Template/Cont_03/btnSkill%d",
  btn04 = "Master/Scroll View/Viewport/Content/Template/Cont_04/btnSkill%d",
  SkillName = "/txtSkillName",
  SkillNum = "/txtSkillNum",
  SkillImage = "/skillImage",
  SkillUnopen = "/skillUnopen",
  SkillUnopen_Effect = "/skillUnopen_Effect",
  SkillActiveStateText = "/txtState",
  ImgLine = "/ImgLine0%d",
  ImgLineOpen = "/ImgLineOpen0%d",
  ImgLineOpen_Effect = "/ImgLineOpen0%d_Effect"
}
tbWnd.tbOnClick = {}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self.pAsyncRole = self.tbOwner.pAsyncRole
  self.nFactionSect = self.pAsyncRole.GetFactionSect()
  self.nFaction = self.pAsyncRole.GetFaction()
  self.tbZongShiSkillSetting = ZongShiSkill.tbZongShiSkillSetting or {}
  self.tbZongShiSkillData = {}
  self.tbContList = {}
  self.tbContBtnName = {
    [0] = tbControls.btnCenter,
    [1] = tbControls.btn01,
    [2] = tbControls.btn02,
    [3] = tbControls.btn03,
    [4] = tbControls.btn04
  }
  self:OnSyncPlayerData()
end

function tbWnd:OnClose()
end

function tbWnd:OnSyncPlayerData()
  self.pPanel:Label_SetText(tbControls.txtTitleInfo, "")
  self.pPanel:SetActive(tbControls.btnReset, false)
  self:UpdateContData()
end

function tbWnd:onClickViewSkill(nSkillId)
  Ui:OpenWindow("UISkillShow_MasterSkill", nSkillId, true)
end

function tbWnd:UpdateContData()
  for nSkillId, tbSkillData in pairs(self.tbZongShiSkillSetting) do
    local nSkillLevel = self.pAsyncRole.GetZongShiSkill(ZongShiSkill.tbZongShiSkill2SaveId[nSkillId]) or 0
    local _, nMaxLevel = ZongShiSkill:GetSkillLevelInfoById(me, nSkillId)
    self.tbZongShiSkillData[nSkillId] = {nSkillLevel, nMaxLevel}
    self.tbContList[tbSkillData.nSect] = self.tbContList[tbSkillData.nSect] or {}
    self.tbContList[tbSkillData.nSect][nSkillId] = tbSkillData
  end
  for i = 0, 4 do
    for nSkillId, tbSkillData in pairs(self.tbContList[i]) do
      local nPosId = tbSkillData.nUIShowPos
      local szBtnName = string.format(self.tbContBtnName[i], nPosId)
      self.pPanel:Label_SetText(szBtnName .. tbControls.SkillName, i18n.Parse(tbSkillData.szName))
      if self.tbZongShiSkillData[nSkillId][1] ~= self.tbZongShiSkillData[nSkillId][2] then
        self.pPanel:Label_SetText(szBtnName .. tbControls.SkillNum, string.format("%d/%d", self.tbZongShiSkillData[nSkillId][1], self.tbZongShiSkillData[nSkillId][2]))
      else
        self.pPanel:Label_SetText(szBtnName .. tbControls.SkillNum, string.format("<color=#FFEF00>%d/%d</color>", self.tbZongShiSkillData[nSkillId][1], self.tbZongShiSkillData[nSkillId][2]))
      end
      local szIconAltlas = FightSkill:GetSkillIconAtlas(nSkillId)
      self.pPanel:Sprite_SetSprite(szBtnName .. tbControls.SkillImage, szIconAltlas)
      self.pPanel:SetActive(szBtnName .. tbControls.SkillUnopen, self.tbZongShiSkillData[nSkillId][1] <= 0)
      self.pPanel:SetActive(szBtnName .. tbControls.SkillUnopen_Effect, self.tbZongShiSkillData[nSkillId][1] == self.tbZongShiSkillData[nSkillId][2])
      self.pPanel:SetActive(szBtnName .. tbControls.SkillActiveStateText, self.tbZongShiSkillData[nSkillId][1] <= 0)
      self.pPanel:Label_SetText(szBtnName .. tbControls.SkillActiveStateText, "")
      Ui:Button_BindEvent(self.pPanel, szBtnName, self.onClickViewSkill, self, nSkillId)
      if 8 == nPosId then
        local szEffectName = ""
        if 1 == i then
          szEffectName = "/QingLong"
        elseif 2 == i then
          szEffectName = "/ZhuQue"
        elseif 3 == i then
          szEffectName = "/BaiHu"
        elseif 4 == i then
          szEffectName = "/XuanWu"
        end
        self.pPanel:SetActive(szBtnName .. szEffectName, self.tbZongShiSkillData[nSkillId][1] > 0)
      end
      local tbSkillLineData = ZongShiSkill.tbUIZongShiSkillLineSetting[i][nPosId]
      if tbSkillLineData then
        for j = 1, tbSkillLineData.nLineCount do
          local nLineTarget = tbSkillLineData["nLine" .. j .. "Target"]
          local nTargetSect = math.floor(nLineTarget / 10)
          local nTargetPos = math.floor(nLineTarget % 10)
          local tbTargetSkill = ZongShiSkill.tbZongShiSkillUIShowPos[nTargetSect][nTargetPos]
          local nCurSkillLevel = self.pAsyncRole.GetZongShiSkill(ZongShiSkill.tbZongShiSkill2SaveId[nSkillId]) or 0
          local nTargetSkillLevel = self.pAsyncRole.GetZongShiSkill(ZongShiSkill.tbZongShiSkill2SaveId[tbTargetSkill[2]]) or 0
          local szLineName = string.format(tbControls.ImgLine, j)
          local szLineOpenName = string.format(tbControls.ImgLineOpen, j)
          local szLineOpenEffect = string.format(tbControls.ImgLineOpen_Effect, j)
          self.pPanel:SetActive(szBtnName .. szLineName, true)
          self.pPanel:SetActive(szBtnName .. szLineOpenName, nCurSkillLevel > 0 and nTargetSkillLevel > 0)
          self.pPanel:SetActive(szBtnName .. szLineOpenEffect, nCurSkillLevel > 0 and nTargetSkillLevel > 0)
        end
      end
    end
  end
end
