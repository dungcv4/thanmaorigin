local ErrorMsg = {
  [LifeSkill.ERROR.NOT_LEARN_RECIPE] = LifeSkill.MSG.NOT_LEARN_RECIPE,
  [LifeSkill.ERROR.NOT_ENOUGH_MATERIAL] = LifeSkill.MSG.NOT_ENOUGH_MATERIAL,
  [LifeSkill.ERROR.NOT_ENOUGH_MONEY] = LifeSkill.MSG.NOT_ENOUGH_MONEY,
  [LifeSkill.ERROR.NOT_ENOUGH_BAG] = LifeSkill.MSG.NOT_ENOUGH_BAG,
  [LifeSkill.ERROR.NOT_LEARN_SKILL] = LifeSkill.MSG.NOT_LEARN_SKILL,
  [LifeSkill.ERROR.MAX_SKILL_LEVEL] = LifeSkill.MSG.MAX_SKILL_LEVEL,
  [LifeSkill.ERROR.NOT_SPECIAL_SKILL] = LifeSkill.MSG.NOT_SPECIAL_SKILL,
  [LifeSkill.ERROR.LEVELUP_SUCCESS] = LifeSkill.MSG.LEVELUP_SUCCESS,
  [LifeSkill.ERROR.NEXT_LEVEL_REQUIRE_TIPS] = LifeSkill.MSG.NEXT_LEVEL_REQUIRE_TIPS
}
LifeSkill.bUseNewPlan = true

function LifeSkill:GetErrorMsg(nErrorCode, ...)
  local szMsg = ErrorMsg[nErrorCode]
  if szMsg then
    return i18n.Get(szMsg, ...)
  else
    assert(szMsg, nErrorCode)
  end
end

function LifeSkill:OnServerResult(nErrorCode, tbParam)
  local szMsg = self:GetErrorMsg(nErrorCode, table.unpack(tbParam or {}))
  if szMsg then
    me.CenterMsg(szMsg, true)
  end
end

function LifeSkill:C2SLevelUp(nSkillId)
  local tbData = self:GetClientData()
  local tbSkill = self.tbSkills[nSkillId]
  if not tbSkill then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NOT_LEARN_SKILL))
    return
  end
  local nCurLevel = tbData.tbSkills[nSkillId]
  if not nCurLevel then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NOT_LEARN_SKILL))
    return
  end
  local tbLevel = self.tbLevel[nSkillId][nCurLevel + 1]
  if not tbLevel then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.MAX_SKILL_LEVEL))
    return
  end
  if me.nTotalLevel < tbLevel.nPlayerLevel then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NEXT_LEVEL_REQUIRE_TIPS, Lib:GetLevelText(tbLevel.nPlayerLevel)))
    return
  end
  if me.GetMoney(LifeSkill.LevelUpCurrencyType) < tbLevel.nCost then
    GainCurrency:ReadyForJump(LifeSkill.LevelUpCurrencyType)
    return
  end
  RemoteServer.OnLifeSkillRequest(LifeSkill.Protocol.C2SLevelUp, nSkillId)
end

function LifeSkill:OnLevelUp(nSkillId, nLevel)
  if not self.tbClientData then
    return
  end
  self.tbClientData.tbSkills[nSkillId] = nLevel
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_LEVELUP, nSkillId, nLevel)
  self:HandleLifeSkillRedPoint()
end

function LifeSkill:C2SCraft(nRecipeId)
  if not nRecipeId then
    return
  end
  local tbRecipeInfo = self.tbRecipes[nRecipeId]
  if not tbRecipeInfo then
    return
  end
  local tbData = self:GetClientData()
  if tbData.tbSkills[tbRecipeInfo.nSkill] < tbRecipeInfo.nLevel then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NOT_LEARN_RECIPE))
    return
  end
  if not self:CheckMaterial(tbRecipeInfo) then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NOT_ENOUGH_MATERIAL))
    return
  end
  local nCost = tbRecipeInfo.nCost
  if tbData.nSpecialty ~= tbRecipeInfo.nSkill then
    nCost = math.ceil(tbRecipeInfo.nCost * LifeSkill:GetCraftCostCoefficient())
  end
  if nCost > me.GetMoney(tbRecipeInfo.szCurrencyType) then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NOT_ENOUGH_MONEY, nCost, Currency:GetName(tbRecipeInfo.szCurrencyType)))
    return
  end
  if me.GetFreeBagCount() < tbRecipeInfo.nNeedBagCount then
    me.CenterMsg(i18n.Get(LifeSkill.MSG.NOT_ENOUGH_BAG))
    return
  end
  RemoteServer.OnLifeSkillRequest(LifeSkill.Protocol.C2SCraft, nRecipeId, self.bDoubleRate)
end

function LifeSkill:OnModifySpecialty(nSkillId)
  if not self.tbClientData then
    return
  end
  if nSkillId then
    self.tbClientData.nSpecialty = nSkillId
  else
    self.tbClientData.nSpecialty = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_LIFESKILL_MODIFY_SPECIALTY)
  self:HandleLifeSkillRedPoint()
end

function LifeSkill:RequestData()
  RemoteServer.OnLifeSkillRequest(LifeSkill.Protocol.C2SRequestData)
end

function LifeSkill:OnSyncData(tbPlayerData)
  self.tbClientData = tbPlayerData
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_DATA)
  self:HandleLifeSkillRedPoint()
end

function LifeSkill:OnLogin(bIsReconnect)
  self.tbClientData = nil
  self.bDoubleRate = false
  self:RequestData()
end

function LifeSkill:GetDoubleRate()
  return self.bDoubleRate
end

function LifeSkill:SetDoubleRate(bDoubleRate)
  self.bDoubleRate = bDoubleRate
end

function LifeSkill:GetClientData()
  return self.tbClientData
end

function LifeSkill:CanShow()
  return self.tbClientData ~= nil and nil ~= next(self.tbClientData.tbSkills)
end

function LifeSkill:HandleLifeSkillRedPoint()
  local tbPlayerData = LifeSkill:GetClientData()
  if not tbPlayerData then
    return
  end
  local nSpecialtySkill = tbPlayerData.nSpecialty
  if not nSpecialtySkill or 0 == nSpecialtySkill then
    return
  end
  local nLevel = tbPlayerData.tbSkills[nSpecialtySkill]
  if not nLevel then
    return
  end
  local isHaveRedPoint = LifeSkill:CheckIsCanLifeSkillLvUp(nSpecialtySkill, nLevel)
  if true == isHaveRedPoint then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.LifeSkill)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.LifeSkill)
  end
end

function LifeSkill:OnPlayerLvChange()
  self:HandleLifeSkillRedPoint()
end

function LifeSkill:OnMoneyChange()
  self:HandleLifeSkillRedPoint()
end

function LifeSkill:OnS2CSetSpecialtyComfirm(nSkillId)
  local function fnComfirm()
    RemoteServer.OnLifeSkillRequest(LifeSkill.Protocol.C2SComfirmSetSpecialty, nSkillId)
  end
  
  local tbInfo = self:GetLifeSkillInfo(nSkillId)
  Ui:OpenWindow("UIMessageBox", i18n.Get(self.MSG.SET_SPECIALTY_COMFIRM, i18n.Parse(tbInfo.szName)), {
    {fnComfirm},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function LifeSkill:SetNotShowValue(bIsNotShowToday)
  RemoteServer.OnLifeSkillRequest(self.Protocol.C2SSetTodaySecondConfirm, not bIsNotShowToday)
end

function LifeSkill:OnSyncFlag(bUseNewPlan)

end
