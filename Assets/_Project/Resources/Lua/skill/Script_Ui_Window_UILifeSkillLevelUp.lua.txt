local tbWnd = Ui:CreateWindow("UILifeSkillLevelUp")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_LIFESKILL_LEVELUP,
      self.Close
    },
    {
      EventNotify.emNOTIFY_SYNC_LIFESKILL_DATA,
      self.Close
    }
  }
end

tbWnd.tbControls = {
  btnClose = "imgBG/btnClose",
  txtSkill = "imgBG/SkillLevel/txtSkill",
  txtLevel = "imgBG/SkillLevel/imgLevel/txtLevel",
  imgSkill = "imgBG/SkillLevel/imgSkill",
  txtNextSkill = "imgBG/SkillNextLevel/txtSkill",
  txtNextLevel = "imgBG/SkillNextLevel/imgLevel/txtLevel",
  imgNextSkill = "imgBG/SkillNextLevel/imgSkill",
  imgMoney = "imgBG/imgMoney",
  txtCost = "imgBG/txtCost",
  btnLevelUp = "imgBG/btnLevelUp",
  txtDes = "imgBG/txtDes"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnLevelUp = function(self)
    LifeSkill:C2SLevelUp(self.nSkill)
    self:Close()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nSkill)
  self.nSkill = nSkill
  local tbData = LifeSkill:GetClientData()
  local nLevel = tbData.tbSkills[nSkill]
  local tbSkillInfo = LifeSkill:GetLifeSkillInfo(nSkill)
  local szName = i18n.Parse(tbSkillInfo.szName)
  self.pPanel:Label_SetText(self.txtSkill, szName)
  self.pPanel:Sprite_SetSprite(self.imgSkill, tbSkillInfo.szNormalIcon)
  self.pPanel:Label_SetText(self.txtLevel, nLevel)
  self.pPanel:Label_SetText(self.txtNextSkill, szName)
  self.pPanel:Sprite_SetSprite(self.imgNextSkill, tbSkillInfo.szNormalIcon)
  self.pPanel:Label_SetText(self.txtNextLevel, nLevel + 1)
  local tbLevelInfo = LifeSkill.tbLevel[nSkill][nLevel + 1]
  self.pPanel:Label_SetText(self.txtCost, tbLevelInfo.nCost)
  self.pPanel:Sprite_SetSprite(self.imgMoney, Currency:GetIcon(LifeSkill.LevelUpCurrencyType))
  self.pPanel:Label_SetText(self.txtDes, i18n.Get(LifeSkill.MSG.NEXT_LEVEL_REQUIRE_TIPS, Lib:GetLevelText(tbLevelInfo.nPlayerLevel)))
  return false
end
