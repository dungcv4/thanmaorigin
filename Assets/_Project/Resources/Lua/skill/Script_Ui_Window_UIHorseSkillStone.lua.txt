function tbWnd:OnCreate()
  self:SetClickEmptyToClose(tbControls.wndJumpTo)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  self:SetClickEmptyToClose(tbControls.wndReplace)
  self.pPanel:SetActive(tbControls.wndReplace, false)
  self.tbStoneGridList = {}
  for nIndex, tbV in ipairs(tbControls.tbStoneInfoKey) do
    self.tbStoneGridList[nIndex] = Ui:NewExistItemGrid(self, tbV.UIItemGrid)
  end
  self.tbObtainScrollView = Ui:NewScrollView(self, tbControls.ObtainTemplate)
  self.tbStoneScrollView = Ui:NewSuperListView(self, tbControls.StoneTemplate, self.SetSkillStoneElement)
  for nIndex = 1, HorseSkillMgr.nMaxSkillStoneNum do
    Ui:Button_BindEvent(self.pPanel, tbControls.tbStoneInfoKey[nIndex].btnEquip, self.UpdateStoneSV, self, nIndex)
    Ui:Button_BindEvent(self.pPanel, tbControls.tbStoneInfoKey[nIndex].btnChange, self.UpdateStoneSV, self, nIndex)
  end
end

function tbWnd:OnDestroy()
  for nIndex = 1, HorseSkillMgr.nMaxSkillStoneNum do
    Ui:Button_UnBindEvent(self.pPanel, tbControls.tbStoneInfoKey[nIndex].btnEquip)
    Ui:Button_UnBindEvent(self.pPanel, tbControls.tbStoneInfoKey[nIndex].btnChange)
  end
end

function tbWnd:OnOpen()
  self:UpdateView()
  for nIndex = 1, HorseSkillMgr.nMaxSkillStoneNum do
    local nRedPoint = RedPointMgr.EVENT.HorseSkillStoneBegin + nIndex - 1
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.tbStoneInfoKey[nIndex].btnEquip, {nRedPoint})
  end
end

function tbWnd:OnClose()
  for nIndex = 1, HorseSkillMgr.nMaxSkillStoneNum do
    RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.tbStoneInfoKey[nIndex].btnEquip)
  end
end

function tbWnd:UpdateView()
  for nSlot = Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1, Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE5 do
    local pSkillStone = me.GetEquipByPos(Item.EQUIPPOS_EXTEND, nSlot)
    local nIndex = nSlot - Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1 + 1
    if pSkillStone then
      self.pPanel:SetActive(tbControls.tbStoneInfoKey[nIndex].Empty, false)
      self.pPanel:SetActive(tbControls.tbStoneInfoKey[nIndex].info, true)
      local szStoneName = Item:GetName(pSkillStone.dwTemplateId, me)
      self.pPanel:Label_SetText(tbControls.tbStoneInfoKey[nIndex].txtStoneName, szStoneName)
      local nSeries = pSkillStone:GetSeries()
      for nType, szKey in pairs(tbControls.tbStoneInfoKey[nIndex].Series) do
        self.pPanel:SetActive(szKey, nSeries == nType)
        self.pPanel:SetActive(self.tbSeriesIconKey[nIndex][nType], nSeries == nType)
      end
      for i = 1, #tbControls.tbStoneInfoKey[nIndex].tbStar do
        self.pPanel:SetActive(tbControls.tbStoneInfoKey[nIndex].tbStar[i], i <= pSkillStone.nQuality - 1)
      end
      self.tbStoneGridList[nIndex]:SetItem(pSkillStone.dwId, pSkillStone.dwTemplateId)
      self.tbStoneGridList[nIndex]:SetClickEnable(true, function()
        local tbInfo = {
          szItemOpt = "HorseSkillStone",
          nItemId = pSkillStone.dwId,
          nTemplate = pSkillStone.dwTemplateId
        }
        Item:ShowItemDetail(tbInfo)
      end)
    else
      self.pPanel:SetActive(tbControls.tbStoneInfoKey[nIndex].Empty, true)
      self.pPanel:SetActive(tbControls.tbStoneInfoKey[nIndex].info, false)
      self.pPanel:Label_SetText(tbControls.tbStoneInfoKey[nIndex].txtStone, i18n.Get(HorseSkillMgr.tbHorseSkillStoneName[nIndex]))
    end
  end
  local nSkillId, tbCfg, nShowLevel = HorseSkillMgr:GetSkillStoneIdAndLevel(me)
  if nSkillId and nSkillId > 0 then
    self.pPanel:SetActive(tbControls.NotActive, false)
    self.pPanel:SetActive(tbControls.Activated, true)
    local szSkillName = FightSkill:GetSkillName(nSkillId)
    local nSkillGrade = HorseSkillMgr:GetStoneSkillGrade(me)
    self.pPanel:Label_SetText(tbControls.txtSkillName, i18n.EasyGet("{0}\233\152\182{1}", nSkillGrade, i18n.Parse(szSkillName)))
    local szSkillIcon = FightSkill:GetSkillIconAtlas(nSkillId)
    self.pPanel:Sprite_SetSprite(tbControls.Img_Skill, szSkillIcon)
    self.pPanel:Label_SetText(tbControls.txtLv, i18n.EasyGet("\230\138\128\232\131\189\231\173\137\231\186\167\239\188\154{0}", nShowLevel))
    local nRealSkillLevel = tbCfg[2]
    local szDesc = FightSkill:GetSkillMagicDesc(nSkillId, nRealSkillLevel)
    self.pPanel:Label_SetText(tbControls.txtAttrib, i18n.Parse(szDesc))
    local nGrade = HorseSkillMgr:GetStoneSkillGrade(me)
    local nMaxLevel = #HorseSkillMgr.tbPassiveSkillLevelCfg[nGrade]
    if nShowLevel >= nMaxLevel then
      self.pPanel:Label_SetText(tbControls.txtNub, i18n.EasyGet("\229\189\147\229\137\141\229\183\178\229\136\176\230\156\128\233\171\152\231\186\167"))
    else
      local tbConfig = HorseSkillMgr.tbPassiveSkillLevelCfg[nGrade][nShowLevel + 1]
      local nAllStar = HorseSkillMgr:GetStoneSkillStar(me)
      self.pPanel:Label_SetText(tbControls.txtNub, i18n.EasyGet("{0}/{1}", nAllStar, tbConfig[1]))
    end
  else
    self.pPanel:SetActive(tbControls.NotActive, true)
    self.pPanel:SetActive(tbControls.Activated, false)
  end
end

function tbWnd:UpdateObtainSV(tbData)
  self.pPanel:SetActive(tbControls.wndJumpTo, true)
  
  local function fnSetItem(tbValue, tbElement)
    local pPanel = tbElement.pPanel
    pPanel:Label_SetText("txtName", tbValue.szTitle())
    pPanel:Label_SetText("txtDesc", tbValue.szDesc())
    if tbValue.funcObtain then
      pPanel:SetActive("imgArrow", true)
      Ui:Button_BindEvent(pPanel, "Main", function()
        self.pPanel:SetActive(tbControls.wndJumpTo, false)
        tbValue.funcObtain()
      end)
    else
      pPanel:SetActive("imgArrow", false)
    end
  end
  
  self.tbObtainScrollView:Reset()
  for i, tbValue in ipairs(tbData) do
    self.tbObtainScrollView:SetElement(i, function(tbElement)
      fnSetItem(tbValue, tbElement)
    end)
  end
end

function tbWnd:UpdateStoneSV(nStoneIndex)
  local tbItemList = HorseSkillMgr:GetSkillStoneList(nStoneIndex)
  self.pPanel:SetActive(tbControls.wndReplace, true)
  self.tbStoneScrollView:SetAllElement(tbItemList)
  self.tbStoneScrollView:MovePanelToFirstItem()
end

function tbWnd:SetSkillStoneElement(tbElement, pItem, nIndex)
  local pPanel = tbElement.pPanel
  local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  itemGrid:SetItem(pItem.dwId, pItem.dwTemplateId)
  itemGrid:SetClickEnable(true)
  pPanel:Label_SetText("txtStoneName", Item:GetName(pItem.dwTemplateId, me))
  local nSeries = pItem.GetSeries()
  for i, szStr in ipairs(tbControls.tbSeriesMap) do
    pPanel:SetActive("txtStoneName/Series/" .. szStr, nSeries == i)
  end
  for i = 1, HorseSkillMgr.nMaxSkillStoneStar do
    pPanel:SetActive("starsCell/imgStarsIcon0" .. i, i <= pItem.nQuality - 1)
  end
  Ui:Button_BindEvent(pPanel, "btnAdd", function()
    RemoteServer.UseEquip(pItem.dwId)
    self.pPanel:SetActive(tbControls.wndReplace, false)
  end)
end
