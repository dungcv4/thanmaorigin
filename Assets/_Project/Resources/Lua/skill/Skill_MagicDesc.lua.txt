function FightSkill:FromatMagicDesc(szFormat, tbMagic)
  local bHide
  
  local function DisposeMagic(varText)
    local nId = tonumber(varText)
    if nId then
      return tbMagic[nId]
    else
      local szFormatDesc, bNeedHide = self:Parse(varText, tbMagic)
      if bNeedHide then
        bHide = true
      end
      return szFormatDesc
    end
  end
  
  local szDesc = i18n.Parse(szFormat)
  szDesc = string.gsub(szDesc, "{(.-)}", DisposeMagic)
  if "" == szDesc then
    szDesc = nil
  end
  if bHide then
    szDesc = nil
  end
  return szDesc
end

function FightSkill:Parse(varText, tbMagic)
  local tbId = {}
  local szToolFunc, _
  _, _, szToolFunc, tbId[1], tbId[2], tbId[3] = string.find(varText, "(.+):(.+);(.+);(.+)")
  if not szToolFunc then
    _, _, szToolFunc, tbId[1], tbId[2] = string.find(varText, "(.+):(.+);(.+)")
  end
  if not szToolFunc then
    _, _, szToolFunc, tbId[1] = string.find(varText, "(.+):(.+)")
  end
  if not szToolFunc then
    error("skill desc param error")
    return
  end
  local tbValue = {}
  for i = 1, 3 do
    if not tbId[i] then
      break
    end
    local nId = tonumber(tbId[i])
    if nId then
      tbValue[i] = tbMagic[nId]
    else
      tbValue[i] = tbId[i]
    end
  end
  return FightSkill.TOOL_FUNC[szToolFunc](FightSkill.TOOL_FUNC, table.unpack(tbValue))
end

function FightSkill:GetMagicDesc(szMagicName, tbMagic)
  local szFinalName, szFinalValue = self:GetMagicDescSplit(szMagicName, tbMagic)
  if nil == szFinalName then
    return nil
  end
  if nil == szFinalValue then
    return szFinalName
  end
  return string.format("%s : %s", szFinalName, szFinalValue)
end

function FightSkill:GetMagicDescSplit(szMagicName, tbMagic)
  if not self.tbAllMagicDesc then
    self:LoadMagicDesc()
  end
  local tbDesc = self.tbAllMagicDesc[szMagicName]
  local szFinalName, szFinalValue
  if not tbDesc then
    return nil, nil
  else
    local fnFunc = DESC_FUNC[tbDesc.szMagicDescFunc]
    if fnFunc then
      szFinalName, szFinalValue = fnFunc(DESC_FUNC, tbDesc, tbMagic)
    else
      if tbDesc.szMagicName then
        szFinalName = self:FromatMagicDesc(tbDesc.szMagicName, tbMagic)
      end
      if tbDesc.szMagicValue and tbMagic and 0 ~= Lib:CountTB(tbMagic) then
        szFinalValue = self:FromatMagicDesc(tbDesc.szMagicValue, tbMagic)
      end
    end
  end
  return szFinalName, szFinalValue
end

function FightSkill:LoadMagicDesc()
  local tbData = Lib:LoadTabFileEx("Setting/Skill/MagicDesc.tab", "ssss", nil, {
    "szMagicAttrib",
    "szMagicName",
    "szMagicValue",
    "szMagicDescFunc"
  }, nil, nil, false)
  local tbAllMagicDesc = {}
  for nRow, tbInfo in ipairs(tbData) do
    tbInfo.nRow = nRow
    tbAllMagicDesc[tbInfo.szMagicAttrib] = tbInfo
  end
  self.tbAllMagicDesc = SetReadOnly(tbAllMagicDesc)
end

function FightSkill:StringValueAddResult(szValue1, szValue2)
  local nValue1, bPer1 = string.gsub(szValue1, "[%%]", "")
  local nValue2, bPer2 = string.gsub(szValue2, "[%%]", "")
  local nResult = nValue2 + nValue1
  local szFront = nResult > 0 and "+" or ""
  local szBack = (bPer1 > 0 or bPer2 > 0) and "%" or ""
  return szFront .. tostring(nResult) .. szBack
end

function FightSkill:StringValueSubResult(szValue1, szValue2)
  local nValue1, bPer1 = string.gsub(szValue1, "[%%]", "")
  local nValue2, bPer2 = string.gsub(szValue2, "[%%]", "")
  local nResult = nValue2 - nValue1
  if 0 == nResult then
    return ""
  end
  local szFront = nResult > 0 and "+" or ""
  local szBack = (bPer1 > 0 or bPer2 > 0) and "%" or ""
  return szFront .. tostring(nResult) .. szBack
end

function FightSkill:StringValueSubNumberResult(szValue1, szValue2)
  local nValue1, _ = string.gsub(szValue1, "[%%]", "")
  local nValue2, _ = string.gsub(szValue2, "[%%]", "")
  local nResult = nValue1 - nValue2
  return nResult
end

function FightSkill:GetAttribTypeName(attribType)
  if not self.tbAllMagicDesc then
    self:LoadMagicDesc()
  end
  return i18n.Parse(self.tbAllMagicDesc[attribType].szMagicName)
end

function FightSkill:GetTableValueMainIndex(szMagicName)
  if not self.tbAllMagicDesc then
    self:LoadMagicDesc()
  end
  local tbDesc = self.tbAllMagicDesc[szMagicName]
  if not tbDesc or not tbDesc.szMagicValue then
    return
  end
  local nRet
  local varText = tbDesc.szMagicValue
  
  local function DisposeMagic(varText)
    local nId = tonumber(varText)
    if nId then
      nRet = nId
    else
      local tbId = {}
      local szToolFunc, _
      _, _, szToolFunc, tbId[1], tbId[2], tbId[3] = string.find(varText, "(.+):(.+);(.+);(.+)")
      if not szToolFunc then
        _, _, szToolFunc, tbId[1], tbId[2] = string.find(varText, "(.+):(.+);(.+)")
      end
      if not szToolFunc then
        _, _, szToolFunc, tbId[1] = string.find(varText, "(.+):(.+)")
      end
      if not szToolFunc then
        print("skill desc param error")
        return varText
      end
      local tbRet = {}
      for i = 1, 3 do
        if not tbId[i] then
          break
        end
        tbRet[i] = tonumber(tbId[i])
      end
      nRet = tbRet[1]
      return varText
    end
  end
  
  varText = i18n.Parse(varText)
  string.gsub(varText, "{(.-)}", DisposeMagic)
  return nRet
end
