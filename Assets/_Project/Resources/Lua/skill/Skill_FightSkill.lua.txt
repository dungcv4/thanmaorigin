function FightSkill:ResetSkillPointFightPower()
  local nTotalFightPower = 0
  local nTotalAttributePower = 0
  local nSkillPoint = 0
  while true do
    nSkillPoint = nSkillPoint + 1
    local tbInfo = self.tbSkillPointFightPower[nSkillPoint]
    if not tbInfo then
      return
    end
    self.nMaxSkillPoint = nSkillPoint
    nTotalFightPower = nTotalFightPower + tbInfo.FightPower
    nTotalAttributePower = nTotalAttributePower + tbInfo.AttributePower
    tbInfo.nTotalFightPower = nTotalFightPower
    tbInfo.nTotalAttributePower = nTotalAttributePower
  end
end

function FightSkill:LoadSetting()
  local tbSkillIniSet = Lib:LoadIniFile("Setting/Skill/SkillSetting.ini")

  self.tbSkillIniSet.nFullAnger = tonumber(tbSkillIniSet.Mix.FullAnger)

  self.tbSkillIniSet.nStealManaMax = math.floor(tonumber(tbSkillIniSet.Mix.StealManaMax) / 100)

    "GroupId",
    "SkillLevel",
    "ReqLevel",

    "FightPower",
    "SkillPoint",
    "KinContrib",

    "szExtendTimeFrame",
    "nExtendItemId",
    "nExtendCost"
  })

  self.tbSkillDefaultMaxLevelList = {}
  self.tbSkillLevelUpGroup = {}
  self.tbSkillExtendSetting = {}

    self.tbSkillLevelUpGroup[v.GroupId] = self.tbSkillLevelUpGroup[v.GroupId] or {}

      v.ReqLevel,

      v.Exp,
      v.FightPower,

    self.tbSkillDefaultMaxLevelList[v.GroupId] = self.tbSkillDefaultMaxLevelList[v.GroupId] or 0
    self.tbSkillExtendSetting[v.GroupId] = self.tbSkillExtendSetting[v.GroupId] or {}

      bNeedExtend = v.bNeedExtend,
      szExtendFrame = v.szExtendTimeFrame,
      nExtendItemId = v.nExtendItemId,
      nExtendCost = v.nExtendCost
    }

      self.tbSkillMaxLevelList[v.GroupId] = v.SkillLevel
    end

      self.tbSkillDefaultMaxLevelList[v.GroupId] = v.SkillLevel

    "SkillName",
    "IconAtlas",
    "LockRange",
    "AttackRadius",

    "HitSkillID",
    "CollisionSkillId",
    "FlySkillId",
    "VanishedSkillId",

    "CostValue",

    "IsPeaceCanUse"
  })

  self.tbAllSkillSetting = {}
  for _, tbSkillInfo in ipairs(tbAllSkillTB) do

      nSkillId = tbSkillInfo.SkillId,
      szSkillName = tbSkillInfo.SkillName,
      szIconAtlas = tbSkillInfo.IconAtlas,
      nLockRange = Lib:Map2Logic(tbSkillInfo.LockRange) or self.nDefLockRange,
      nAttackRadius = tbSkillInfo.AttackRadius or 0,
      nStartSkillID = tbSkillInfo.StartSkillID,
      nHitSkillID = tbSkillInfo.HitSkillID,
      nCollisionSkillId = tbSkillInfo.CollisionSkillId,
      nFlySkillId = tbSkillInfo.FlySkillId,
      nVanishedSkillId = tbSkillInfo.VanishedSkillId,
      nCostType = tbSkillInfo.CostType,
      nCostValue = tbSkillInfo.CostValue,
      szDesc = tbSkillInfo.Desc
    }

  local tbFactionSkill = Lib:LoadTabFile("Setting/Skill/FactionSkill.tab", {
    Faction = 1,
    FactionSect = 1,
    SkillId = 1,
    SortInUI = 1,
    IsAnger = 1,
    LevelUpGroup = 1,
    GainLevel = 1,
    SkillGroup = 1,
    CastType = 0
  }, nil, false)
  assert(tbFactionSkill)
  self.tbFactionSkillSetting = {}
  self.tbLevelUpGroup2Skill = {}

    local nFactionId = tbSkillInfo.Faction

    self.tbLevelUpGroup2Skill[nLevelUpGroup][nFactionId] = self.tbLevelUpGroup2Skill[nLevelUpGroup][nFactionId] or {}
    self.tbLevelUpGroup2Skill[nLevelUpGroup][nFactionId][nFactionSect] = tbSkillInfo.SkillId
  end
  local tbSkillPublic = Lib:LoadTabFile("Setting/Skill/SkillPublic.tab", {SkillId = 1, PublicCd = 1})

  for _, tbSkillInfo in ipairs(tbSkillPublic) do
    self.tbFactionSkillPublicCd[tbSkillInfo.SkillId] = tbSkillInfo.PublicCd

    "Faction",

      self.tbSkillLevelUp[v.SkillId] = self.tbSkillLevelUpGroup[v.LevelUpGroup]
    end

  for nSkillId, tbSkillData in pairs(self.tbSkillLevelUp) do

    for k, v in pairs(tbSkillData) do

function FightSkill:GetSkillLockRange(nSkillId)
  if not nSkillId or nSkillId <= 0 then
    return self.nDefLockRange
  end
  local tbSkillSetting = self.tbAllSkillSetting[nSkillId]
  if not tbSkillSetting then
    return self.nDefLockRange
  end
  return tbSkillSetting.nLockRange
end

  if self.tbFactionSkillPublicCd and self.tbFactionSkillPublicCd[nSkillId] then
    return self.tbFactionSkillPublicCd[nSkillId]
  end
  return 0
end

function FightSkill:IsSelfSkill(nSkillId)
  local tbSkillInfo = self:GetSkillSetting(nSkillId)
  return tbSkillInfo.TargetSelf
end

function FightSkill:GetSkillPointFightPower(nSkillPoint, bIsAttribute)
  if nSkillPoint <= 0 then
    return 0
  end
  local tbInfo = self:GetSkillPointPowerInfo(nSkillPoint)
  if bIsAttribute then
    return tbInfo.nTotalAttributePower or 0
  end
  return tbInfo.nTotalFightPower or 0
end

function FightSkill:GetSkillPointPowerInfo(nSkillPoint)
  local tbInfo = self.tbSkillPointFightPower[nSkillPoint]
  if tbInfo then
    return tbInfo
  end
  local tbCurInfo = self.tbSkillPointFightPower[self.nMaxSkillPoint]
  local nTotalFightPower = tbCurInfo.nTotalFightPower
  local nTotalAttributePower = tbCurInfo.nTotalAttributePower
  for nI = self.nMaxSkillPoint + 1, nSkillPoint do
    local tbFightPowerInfo = {}
    tbFightPowerInfo.SkillPoint = nI
    tbFightPowerInfo.FightPower = tbCurInfo.FightPower
    tbFightPowerInfo.AttributePower = tbCurInfo.AttributePower
    nTotalFightPower = nTotalFightPower + tbFightPowerInfo.FightPower
    nTotalAttributePower = nTotalAttributePower + tbFightPowerInfo.AttributePower
    tbFightPowerInfo.nTotalFightPower = nTotalFightPower
    tbFightPowerInfo.nTotalAttributePower = nTotalAttributePower
    self.tbSkillPointFightPower[nI] = tbFightPowerInfo
  end
  self.nMaxSkillPoint = nSkillPoint
  return self.tbSkillPointFightPower[nSkillPoint]
end

function FightSkill:GetAttackRadius(nSkillId)
  local tbSkillInfo = self:GetSkillSetting(nSkillId)
  return tbSkillInfo.AttackRadius
end

function FightSkill:GetSkillLevelUpNeed(nSkillId, nLevel)
  nLevel = nLevel + 1
  if self.tbSkillLevelUp[nSkillId] and self.tbSkillLevelUp[nSkillId][nLevel] then
    local tbSkill = self.tbSkillLevelUp[nSkillId][nLevel]
    return tbSkill[1], tbSkill[2], tbSkill[3], tbSkill[5], tbSkill[6]
  else
    return 0, 0, 0, 0, 0
  end
end

function FightSkill:GetSkillUpInfo(nSkillId, nLevel)
  local tbSkillInfo = self.tbSkillLevelUp[nSkillId]
  if not tbSkillInfo then
    return
  end
  return tbSkillInfo[nLevel]
end

function FightSkill:GetSkillFightPower(nSkillId, nLevel)
  if self.tbSkillLevelUp[nSkillId] and self.tbSkillLevelUp[nSkillId][nLevel] then
    local tbSkill = self.tbSkillLevelUp[nSkillId][nLevel]
    return tbSkill[4]
  else
    return 0
  end
end

function FightSkill:GetSkillMaxLevel(pPlayer, nSkillId)
  if self.tbAllFactionSkill[nSkillId] then
    local nGroupId = self:GetSkillLevelUpGroupId(nSkillId)
    return self:GetDefaultSkillMaxLevel(nGroupId) + self:GetExtendSkillMaxLevel(pPlayer, nGroupId)
  end
  return self.tbSkillMaxLevel[nSkillId] or 15
end

function FightSkill:GetAsyncSkillMaxLevel(pAsyncRole, nSkillId)
  local nGroupId = self:GetSkillLevelUpGroupId(nSkillId)
  local nExtendLevel = 0
  if pAsyncRole then
    nExtendLevel = pAsyncRole.GetPlayerSkillExtendLevel(nGroupId) or 0
  end
  return self:GetDefaultSkillMaxLevel(nGroupId) + nExtendLevel
end

function FightSkill:GetDefaultSkillMaxLevel(nGroupId)
  return self.tbSkillDefaultMaxLevelList[nGroupId] or 0
end

function FightSkill:GetExtendSkillMaxLevel(pPlayer, nGroupId)
  local nExtendSKillLevel = pPlayer.GetUserValue(self.SKILL_EXTEND_GROUP, nGroupId)
  return nExtendSKillLevel or 0
end

function FightSkill:GetTrueSkillMaxLevel(nUpGroupID)
  return self.tbSkillMaxLevelList[nUpGroupID] or 15
end

function FightSkill:GetSkillLevel(pPlayer, nSkillID)
  return pPlayer.GetSkillLevel(nSkillID)
end

function FightSkill:CheckSkillLeveUp(pPlayer, nSkillId)
  if pPlayer.IsMaskPlayerFactSkill() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\230\151\160\230\179\149\229\138\160\231\130\185")
  end
  local _, nBaseLevel = pPlayer.GetSkillLevel(nSkillId)
  if nBaseLevel < 0 then
    return false, i18n.EasyGet("\230\178\161\230\156\137\232\142\183\229\190\151\232\175\165\230\138\128\232\131\189")
  end
  local nFactionLimit = KFightSkill.GetFactionLimit(nSkillId)
  if pPlayer.nFaction ~= nFactionLimit and nFactionLimit > 0 then
    return false, i18n.EasyGet("\233\151\168\230\180\190\228\184\141\231\172\166\229\144\136")
  end
  local tbSkillUpInfo = FightSkill:GetSkillUpInfo(nSkillId, nBaseLevel + 1)
  if not tbSkillUpInfo then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167!")
  end
  local nReqLevel, _, _, nNeedPoint = FightSkill:GetSkillLevelUpNeed(nSkillId, nBaseLevel)
  local nSkillPoint = pPlayer.GetSkillPoint()
  if nNeedPoint > nSkillPoint then
    return false, i18n.EasyGet("\230\138\128\232\131\189\231\130\185\228\184\141\232\182\179")
  end
  if nReqLevel > pPlayer.nLevel then
    return false, i18n.EasyGet("\228\184\187\232\167\146\232\190\190\229\136\176{0}\231\186\167\239\188\140\230\137\141\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\230\173\164\230\138\128\232\131\189", nReqLevel)
  end
  if nBaseLevel >= FightSkill:GetSkillMaxLevel(pPlayer, nSkillId) then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167")
  end
  return true, "", nBaseLevel, nNeedPoint
end

function FightSkill:CheckSkillBatchLevelUp(pPlayer, nSkillId, nAddLevel)
  if pPlayer.IsMaskPlayerFactSkill() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\230\151\160\230\179\149\229\138\160\231\130\185")
  end
  local _, nBaseLevel = pPlayer.GetSkillLevel(nSkillId)
  if nBaseLevel < 0 then
    return false, i18n.EasyGet("\230\178\161\230\156\137\232\142\183\229\190\151\232\175\165\230\138\128\232\131\189")
  end
  local nFactionLimit = KFightSkill.GetFactionLimit(nSkillId)
  if pPlayer.nFaction ~= nFactionLimit and nFactionLimit > 0 then
    return false, i18n.EasyGet("\233\151\168\230\180\190\228\184\141\231\172\166\229\144\136")
  end
  if nBaseLevel >= FightSkill:GetSkillMaxLevel(pPlayer, nSkillId) then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167")
  end
  local nFinalLevel = nBaseLevel + nAddLevel
  if nFinalLevel > FightSkill:GetSkillMaxLevel(pPlayer, nSkillId) then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167")
  end
  local tbSkillUpInfo = FightSkill:GetSkillUpInfo(nSkillId, nFinalLevel)
  if not tbSkillUpInfo then
    return false, i18n.EasyGet("\229\183\178\232\190\190\230\156\128\229\164\167\231\173\137\231\186\167!")
  end
  local nMaxReqLevel = 0
  local nTotalNeedPoint = 0
  for nAdd = 1, nAddLevel do
    local nReqLevel, _, _, nNeedPoint = FightSkill:GetSkillLevelUpNeed(nSkillId, nBaseLevel + nAdd - 1)
    nTotalNeedPoint = nTotalNeedPoint + nNeedPoint
    nMaxReqLevel = math.max(nMaxReqLevel, nReqLevel)
  end
  local nSkillPoint = pPlayer.GetSkillPoint()
  if nTotalNeedPoint > nSkillPoint then
    return false, i18n.EasyGet("\230\138\128\232\131\189\231\130\185\228\184\141\232\182\179")
  end
  if nMaxReqLevel > pPlayer.nLevel then
    return false, i18n.EasyGet("\228\184\187\232\167\146\232\190\190\229\136\176{0}\231\186\167\239\188\140\230\137\141\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\230\173\164\230\138\128\232\131\189", nMaxReqLevel)
  end
  return true, "", nBaseLevel, nTotalNeedPoint
end

function FightSkill:GetSkillShowInfo(nSkillId)
  local tbSkillInfo = self:GetSkillSetting(nSkillId)
  if not tbSkillInfo then
    return
  end
  local szAtlas = tbSkillInfo.IconAtlas
  if Lib:IsEmptyStr(szAtlas) then
    szAtlas = "UI/Atlas/SkillIcon/SkillIcon.prefab"
  end
  local tbIcon = {
    szIconAtlas = szAtlas,
    szIconSprite = tbSkillInfo.Icon or "Skill3"
  }
  return tbIcon, tbSkillInfo.SkillName
end

function FightSkill:GetFactionSkill(nFaction, nFactionSect)
  if self.tbFactionSkillSetting and self.tbFactionSkillSetting[nFaction] then
    return self.tbFactionSkillSetting[nFaction][nFactionSect]
  end
end

function FightSkill:GetFactionSkillInfo(nFaction, nFactionSect, nSkillId)
  local tbFactionSkill = self:GetFactionSkill(nFaction, nFactionSect)
  if not tbFactionSkill then
    return
  end
  for k, tbSkillInfo in pairs(tbFactionSkill) do
    if tbSkillInfo.SkillId == nSkillId then
      return tbSkillInfo
    end
  end
end

function FightSkill:IsFactionSkill(nSkillId, nFactionSect, nFaction)
  local tbFactionSkill = FightSkill:GetFactionSkill(nFaction, nFactionSect)
  if tbFactionSkill then
    for k, tbSkillInfo in pairs(tbFactionSkill) do
      if tbSkillInfo.SkillId == nSkillId then
        return true
      end
    end
  end
  return false
end

function FightSkill:GetSortInUI(nFaction, nFactionSect, nSkillId)
  local tbFactionSkill = self:GetFactionSkill(nFaction, nFactionSect)
  if not tbFactionSkill then
    return -1
  end
  for k, tbSkillInfo in pairs(tbFactionSkill) do
    if tbSkillInfo.SkillId == nSkillId then
      return tbSkillInfo.SortInUI
    end
  end
  return -1
end

function FightSkill:GetFactionSkillNotCommonLimitLevel(nFaction, nFactionSect, nLevel)
  self.tbFactionSkillNotCommonLimitLevel = self.tbFactionSkillNotCommonLimitLevel or {}
  self.tbFactionSkillNotCommonLimitLevel[nFaction] = self.tbFactionSkillNotCommonLimitLevel[nFaction] or {}
  self.tbFactionSkillNotCommonLimitLevel[nFaction][nFactionSect] = self.tbFactionSkillNotCommonLimitLevel[nFaction][nFactionSect] or {}
  if not self.tbFactionSkillNotCommonLimitLevel[nFaction][nFactionSect][nLevel] then
    self.tbFactionSkillNotCommonLimitLevel[nFaction][nFactionSect][nLevel] = {}
    local tbFactionSkill = FightSkill:GetFactionSkill(nFaction, nFactionSect)
    for _, tbSkillInfo in pairs(tbFactionSkill) do
      if nLevel >= tbSkillInfo.GainLevel then
        table.insert(self.tbFactionSkillNotCommonLimitLevel[nFaction][nFactionSect][nLevel], tbSkillInfo.SkillId)
      end
    end
  end
  return self.tbFactionSkillNotCommonLimitLevel[nFaction][nFactionSect][nLevel]
end

function FightSkill:GetFactionSkillNotCommon(nFaction, nFactionSect)
  self.tbFactionSkillNotCommon = self.tbFactionSkillNotCommon or {}
  self.tbFactionSkillNotCommon[nFaction] = self.tbFactionSkillNotCommon[nFaction] or {}
  if not self.tbFactionSkillNotCommon[nFaction][nFactionSect] then
    self.tbFactionSkillNotCommon[nFaction][nFactionSect] = {}
    local tbFactionSkill = FightSkill:GetFactionSkill(nFaction, nFactionSect)
    for k, tbSkillInfo in pairs(tbFactionSkill) do
      table.insert(self.tbFactionSkillNotCommon[nFaction][nFactionSect], tbSkillInfo.SkillId)
    end
  end

function FightSkill:GetLevelData(szSkillName, nLevel)
  assert(szSkillName and nLevel)
  if not self.tbClass[szSkillName] then
    print(string.format("[ERROR][skill error]\226\128\152%s\226\128\153 not found, className not found \239\188\129\239\188\129\239\188\129\239\188\129\239\188\129\239\188\129", szSkillName))
    return {}
  end
  local tbSkill = assert(self.tbClass[szSkillName], "[ERROR]Skill{" .. tostring(szSkillName) .. "} not found!")
  local tbRet = {}
  if 0 ~= #tbSkill.tbMagics then
    for _, tbMagic in ipairs(tbSkill.tbMagics) do
      if type(tbMagic) ~= "table" then
        Log("[ERROR][skill error]", szSkillName, nLevel)
      end
      local tbData = {
        szName = tbMagic.szMagicName
      }
      local tbProp = {}
      for nI = 1, self.MAGIC_VALUE_NUM do
        tbData["nV" .. nI] = 0
      end
      if type(tbMagic.tbValue) == "function" then
        tbProp = tbMagic.tbValue(nLevel)
      else
        for i = 1, self.MAGIC_VALUE_NUM do
          local nTemp = Lib.Calc:Link(nLevel, tbMagic.tbValue[i], true)
          tbProp[i] = math.floor(nTemp * 100)
        end
      end
      for nI, nValue in pairs(tbProp) do
        tbData["nV" .. nI] = nValue
      end
      tbRet[#tbRet + 1] = tbData
    end
  else
    for szMagicName, tbMagicProp in pairs(tbSkill.tbMagics) do
      if "skill_desc" ~= szMagicName then
        local tbData = {szName = szMagicName}
        local tbProp = {}
        for nI = 1, self.MAGIC_VALUE_NUM do
          tbData["nV" .. nI] = 0
        end
        if type(tbMagicProp) == "function" then
          tbProp = tbMagicProp(nLevel)
        else
          for i = 1, self.MAGIC_VALUE_NUM do
            local nTemp = Lib.Calc:Link(nLevel, tbMagicProp[i], true)
            tbProp[i] = math.floor(nTemp * 100)
          end
        end
        for nI, nValue in pairs(tbProp) do
          tbData["nV" .. nI] = nValue
        end
        tbRet[#tbRet + 1] = tbData
      end
    end
  end
  return tbRet
end

function FightSkill:GetFormatValue(szFormat, nLevel)
  local fnData = load("return " .. szFormat)
  if not fnData then
    error(string.format("load failed! FMT = %s, Level = %d", szFormat, nLevel))
  end
  local tbPoint = {
    fnData()
  }
  if #tbPoint <= 1 then
    local varValue = tbPoint[1]
    if "string" == type(varValue) then
      return self:GetFormatValue(varValue, nLevel)
    elseif type(varValue) == "number" then
      return varValue
    else
      error("format error!")
    end
  end
  return Lib.Calc:Link(nLevel, tbPoint)
end

end

function FightSkill:GetAllSkillIdByLuaName(szLuaName)
  if not self.tbSkillIdClass then
    local tbSkill = Lib:LoadTabFile("Setting/Skill/Skill.tab", {SkillId = 1})
    self.tbSkillIdClass = {}
    for k, v in pairs(tbSkill) do
      self.tbSkillIdClass[v.ClassName] = self.tbSkillIdClass[v.ClassName] or {}
      table.insert(self.tbSkillIdClass[v.ClassName], v.SkillId)
    end
  end
  local tbSkillId = {}
  if self.tbLuaNameClass then
    local tbSingleLuaNameClass = self.tbLuaNameClass[szLuaName]
    if tbSingleLuaNameClass then
      for k, skillName in pairs(tbSingleLuaNameClass) do
        local tbSkillNameClas = self.tbSkillIdClass[skillName]
        if tbSkillNameClas then
          tbSkillId = Lib:MergeTable(tbSkillId, tbSkillNameClas)
        end
      end
    elseif me then
      me.CenterMsg("not found luaName:" .. szLuaName)
    end
  end
  self.tbSkillIdClass = SetReadOnly(self.tbSkillIdClass)
  return tbSkillId
end

function FightSkill:CanSkillLevelExtend(pPlayer, nSkillId)

  if not bRet then
    return bRet, szRet
  end
  local nItemId, nCostNum = self:GetSkillExtendCost(pPlayer, nSkillId)

function FightSkill:CheckSkillLevelExtendOpen(pPlayer, nSkillId)
  if not pPlayer then
    return false, i18n.EasyGet("\231\142\169\229\174\182\228\184\141\229\173\152\229\156\168")
  end
  if not nSkillId then
    return false, i18n.EasyGet("\230\138\128\232\131\189\228\184\141\229\173\152\229\156\168")
  end
  local _, nBaseLevel = pPlayer.GetSkillLevel(nSkillId)
  if nBaseLevel < 0 then
    return false, i18n.EasyGet("\230\178\161\230\156\137\232\142\183\229\190\151\232\175\165\230\138\128\232\131\189")
  end
  local bSameFactionSect = self:IsFactionSkill(nSkillId, pPlayer.nFactionSect, pPlayer.nFaction)
  if not bSameFactionSect then
    return false, i18n.EasyGet("\233\151\168\230\180\190\232\183\175\231\186\191\228\184\141\231\172\166\229\144\136")
  end
  local nFactionLimit = KFightSkill.GetFactionLimit(nSkillId)
  if pPlayer.nFaction ~= nFactionLimit and nFactionLimit > 0 then
    return false, i18n.EasyGet("\233\151\168\230\180\190\228\184\141\231\172\166\229\144\136")
  end
  local nSkillLevelExtendLimit = self:GetSkillLevelExtendLimit(nSkillId)
  local nMaxSkillLevel = self:GetSkillMaxLevel(pPlayer, nSkillId)
  if nSkillLevelExtendLimit <= nMaxSkillLevel then
    return false, i18n.EasyGet("\229\189\147\229\137\141\230\138\128\232\131\189\229\183\178\231\187\143\230\137\169\229\177\149\229\136\176\228\184\138\233\153\144")
  end
  if not self:CheckSkillExtendTimeFrame(pPlayer, nSkillId) then
    return false, i18n.EasyGet("\229\189\147\229\137\141\230\151\182\233\151\180\232\189\180\230\156\170\229\188\128\230\148\190")
  end
  return true, ""
end

function FightSkill:GetSkillGainLevel(nSkillId)
  return self.tbAllFactionSkill[nSkillId].GainLevel
end

function FightSkill:GetSkillIdByLevelUpGroup(pPlayer, nGroupId)

end
