ZongShiSkill = ZongShiSkill or {}

function ZongShiSkill:OnPlayerLvChange()
  self:UpdateUISkillView(false)
end

function ZongShiSkill:OnPlayerLogin()
  self:UpdateUISkillView(false)
end

function ZongShiSkill:UpdateUISkillView(bUpdateSkill)
  local nSkillPointTotalNum, nSkillPointUsedNum = ZongShiSkill:GetZongShiPoint(me)
  if nSkillPointUsedNum < nSkillPointTotalNum then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ZongShiSkill)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZongShiSkill)
  end
  if Ui:WindowVisible("UISkill") then
    Ui:GetWindow("UISkill"):UpdateZongShiView(bUpdateSkill)
  end
  if Ui:WindowVisible("UISkillShow_MasterSkill") then
    Ui:GetWindow("UISkillShow_MasterSkill"):UpdateUIShow()
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_LEVEL, ZongShiSkill)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_LEVEL, ZongShiSkill.OnPlayerLvChange, ZongShiSkill)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, ZongShiSkill)
EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, ZongShiSkill.OnPlayerLogin, ZongShiSkill)
