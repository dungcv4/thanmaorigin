  btnClose = function(self)
    self:Close()
  end,
  btnBuyItem = function(self)
  end,
  btnStarUp = function(self)
    WuXue:RequestLearnWuXue(self.nCurSelectWuXueId)
  end,
  btnEffectSkillIcon1 = function(self)

  btnEffectSkillIcon4 = function(self)

  end,
  btnEffectSkillIcon5 = function(self)
    self:UpdateWuXueDetail(self.nCurSelectWuXueId, 5)

  btnEffectSkillIcon6 = function(self)
    self:UpdateWuXueDetail(self.nCurSelectWuXueId, 6)
  end

function tbWnd:OnCreate()
  self.tbWuXueListScrollView = Ui:NewScrollView(self, tbControls.goWuXueListTemplate, function(tbOwner, nIndex, element)
    self.nCurSelectWuXueIndex = nIndex
    self.nCurSelectWuXueId = self.tbIndex2WuXueId[nIndex]
    self:UpdateWuXueInfo(self.nCurSelectWuXueId)
  end, false, false)
  self.tbSkillDetail1ScrollView = Ui:NewScrollView(self, tbControls.goSkillDetail1)
  self.goStarUpGrid = Ui:NewExistItemGrid(self, tbControls.goItemGrid)
  self.goStarUpGrid:SetClickEnable(true)
end

function tbWnd:UpdateUI()
  self.tbWuXueListScrollView:Clear()
  self.tbIndex2WuXueId = {}
  local tbWuXueLearned = WuXue:GetWuXueLearnedCount(me)
  for _, tbWuXueInfo in pairs(tbWuXueLearned) do
    table.insert(self.tbIndex2WuXueId, tbWuXueInfo.nWuXueId)
  end
  for nIndex, nWuXueId in ipairs(self.tbIndex2WuXueId) do
    self.tbWuXueListScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local nWuXueLearnState = WuXue:GetWuXueSkillLearnState(me, nWuXueId)
      for i = 1, nWuXueLearnState - 1 do
        pPanel:SetActive(string.format("StarGrid/Start0%d/imgLight", i), true)
      end
      pPanel:Label_SetText("txtSkillName", i18n.Parse(WuXue.tbWuXueSkillInfo[nWuXueId].szWuXueName))
      local nActiveSkillId, nPassiveSkillId = WuXue:GetSkillIdByWuXueId(nWuXueId)
      local szSkillIcon = FightSkill:GetSkillIconAtlas(nActiveSkillId)
      pPanel:Sprite_SetSprite("SkillIcon01/btnSkill", szSkillIcon)
      local bInstalled, nType, nIndex = WuXue:CheckWuXueInstalled(me, nWuXueId)
      if not bInstalled then
        pPanel:SetActive("imgMark", false)
      else
        pPanel:SetActive("imgMark", true)
        if 1 == nType then
          pPanel:Label_SetText("imgMark/Text", i18n.EasyGet("\228\184\187\229\138\168"))
        elseif 2 == nType then
          pPanel:Label_SetText("imgMark/Text", i18n.EasyGet("\232\162\171\229\138\168"))
        end
      end
    end)
  end
  for nIndex, nWuXueId in ipairs(self.tbIndex2WuXueId) do
    if nWuXueId == self.nCurSelectWuXueId then
      self.nCurSelectWuXueIndex = nIndex
      break
    end
  end
  self.tbWuXueListScrollView:SelectElement(self.nCurSelectWuXueIndex)
end

end

function tbWnd:UpdateWuXueInfo(nWuXueId)
  local nActiveSkillId, nPassiveSkillId = WuXue:GetSkillIdByWuXueId(nWuXueId)
  local szSkillIcon = FightSkill:GetSkillIconAtlas(nActiveSkillId)
  local nCurLeranState = WuXue:GetWuXueSkillLearnState(me, nWuXueId)
  for i = 1, 5 do
    self.pPanel:SetActive(string.format(tbControls.goEffectSkillIconLine, i), false)
    if i <= nCurLeranState - 2 then
      self.pPanel:SetActive(string.format(tbControls.goEffectSkillIconLine, i), true)
    end
  end
  for i = 1, 6 do
    self.pPanel:Sprite_SetSprite(string.format(tbControls.imgWuXueEffectSkillIcon, i), szSkillIcon)
    for j = 1, i do
      self.pPanel:SetActive(string.format(tbControls.goEffectStarLight, i, j), true)
    end
  end
  self:UpdateWuXueDetail(nWuXueId, 1)
end

function tbWnd:UpdateWuXueDetail(nWuXueId, nStar)
  local nActiveSkillId, nPassiveSkillId = WuXue:GetSkillIdByWuXueId(nWuXueId)
  local szSkillIcon = FightSkill:GetSkillIconAtlas(nActiveSkillId)
  self.pPanel:Sprite_SetSprite(tbControls.imgSkillIcon, szSkillIcon)
  self.pPanel:Label_SetText(tbControls.txtSkillName, i18n.Parse(WuXue.tbWuXueSkillInfo[nWuXueId].szWuXueName))
  for i = nStar + 1, 6 do
    self.pPanel:SetActive(string.format(tbControls.goStarLight, i), false)
  end
  for i = 1, nStar do
    self.pPanel:SetActive(string.format(tbControls.goStarLight, i), true)
  end
  local nItemTemplateId = WuXue.tbWuXueSkillInfo[nWuXueId].nItemTemplateId
  self.goStarUpGrid:SetItemByTemplate(nItemTemplateId, 0)
  local nItemCount = me.GetItemCountInBags(nItemTemplateId)
  self.pPanel:Label_SetText(tbControls.txtCost, string.format("%d/%d", nItemCount, 1))
  local tbSkillDetail1 = WuXue:GenerateSkillDescDiffText(nActiveSkillId, nStar)
  self.tbSkillDetail1ScrollView:Clear()
  for nIndex, tbDescList in pairs(tbSkillDetail1) do
    self.tbSkillDetail1ScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtCont", tbDescList[1] .. "\239\188\154")
      pPanel:Label_SetText("txtCont/txtNum", tbDescList[2])
    end)
  end
  local szDesc = WuXue.tbWuXueSkillLevelDescList[nWuXueId][nStar + 1]
  if szDesc then
    self.pPanel:Label_SetText(tbControls.txtSkillDetail2, szDesc)
  else
    self.pPanel:Label_SetText(tbControls.txtSkillDetail2, "")
  end
end
