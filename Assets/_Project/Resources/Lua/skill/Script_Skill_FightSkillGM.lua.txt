FightSkill.GM = FightSkill.GM or {}
local GM = FightSkill.GM

function GM:AddTestSkill(nSkillId, nSkillLv)
  if not WINDOWS or 1 ~= ALLOW_GM then
    return
  end
  if 0 ~= nSkillLv then
    me.nTestSkillCache = nSkillId
    me.CenterMsg(string.format("\229\183\178\230\183\187\229\138\160\230\138\128\232\131\189%d", nSkillId))
  else
    me.nTestSkillCache = nil
    me.CenterMsg(string.format("\229\183\178\229\136\160\233\153\164\230\138\128\232\131\189%d", nSkillId))
  end
end

function GM:UseTestSkill()
  if not WINDOWS or 1 ~= ALLOW_GM then
    return
  end
  local nSkillId = me.nTestSkillCache
  if not nSkillId then
    return
  end
  if me.nFightMode == Npc.FIGHT_MODE.emFightMode_None then
    me.CenterMsg("\229\174\137\229\133\168\229\140\186\229\134\133\228\184\141\232\131\189\228\189\191\231\148\168\232\175\165\230\138\128\232\131\189")
    return
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_USE_SKILL)
  Operation:Attack(nSkillId, false)
end

function GM:ShowTestSkill()
  if not WINDOWS or 1 ~= ALLOW_GM then
    return
  end
  local nSkillId = me.nTestSkillCache
  if not nSkillId then
    return
  end
  if Ui:WindowVisible("UISkillShow") then
    Ui:CloseWindow("UISkillShow")
    return
  end
  local nSkillLv = me.GetSkillLevel(nSkillId)
  local tbFactionSkill = FightSkill:GetFactionSkill(me.nFaction, me.nFactionSect)[1]
  local tbDummySkill = Lib:CopyTB(tbFactionSkill)
  tbDummySkill.SkillId = nSkillId
  local tbShowInfo = FightSkill:GetPlayerOneSkillInfo({tbDummySkill}, 1, me.nFactionSect)
  Ui:OpenWindow("UISkillShow", tbShowInfo, true, false)
end

function GM:ExtendSkill(nGroupId)
  local tbSkillList = FightSkill:GetPlayerAllSkillInfo(me.nFactionSect)
  FightSkill:SkillLevelExtend(tbSkillList[nGroupId].nId)
end
