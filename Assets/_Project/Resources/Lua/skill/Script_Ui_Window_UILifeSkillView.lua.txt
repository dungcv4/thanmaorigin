tbWnd.tbOnClick = {
  btnLifeSkillUp = function(self)
    local tbSkillData = self.tbCurSelectSkill
    if not tbSkillData then
      return
    end
    Ui:OpenWindow("UILifeSkillLevelUp", tbSkillData.nSkill)
  end,
  btnCraft = function(self)
    if self.nCurSelectRecipeId then
      local nRequipLevel = LifeSkill:GetProductRequieLvByRecipeId(self.nCurSelectRecipeId)
      if not nRequipLevel then
        return
      end
      local bLowLevel = me.nTotalLevel < nRequipLevel - LifeSkill.nRequireLevelOffset
      if bLowLevel and LifeSkill.bUseNewPlan then
        me.CenterMsg(i18n.EasyGet("\230\137\147\233\128\160\233\156\128\232\166\129\232\167\146\232\137\178\231\173\137\231\186\167{0}\231\186\167", nRequipLevel - LifeSkill.nRequireLevelOffset))
        return
      end
      
      local function fnAgree()
        LifeSkill:C2SCraft(self.nCurSelectRecipeId)
      end
      
      local tbRecipeInfo = LifeSkill.tbRecipes[self.nCurSelectRecipeId]
      local tbData = LifeSkill:GetClientData()
      local bSpecialty = tbData.nSpecialty == tbRecipeInfo.nSkill
      local bLevelLow = self.nCurSelectLifeSkillLv < LifeSkill:GetCanCraftMaxLevel(LifeSkill.tbRecipes[self.nCurSelectRecipeId].nSkill, me.nTotalLevel)
      local bHaveDegree = DegreeCtrl:GetDegree(me, "LifeSkillTodaySecondConfirm") > 0
      local szMsg
      if bLevelLow and not bSpecialty then
        szMsg = i18n.Get(LifeSkill.MSG.NOTSPECIALTY_LEVELLOW_COMFIRM)
      elseif bLevelLow then
        szMsg = i18n.Get(LifeSkill.MSG.LEVEL_LOW_COMFIRM)
      elseif not bSpecialty then
        szMsg = i18n.Get(LifeSkill.MSG.NOT_SPECIALTY_COMFIRM)
      end
      if szMsg and bHaveDegree then
        Ui:OpenWindow("UIMessageBox", szMsg, {
          {fnAgree},
          {}
        }, {
          i18n.Get(LifeSkill.MSG.LEVEL_LOW_COMFIRM_AGREE),
          i18n.szCancel
        }, "LifeSkillSecondConfirm", nil, i18n.Get(LifeSkill.MSG.TOGGLE_TEXT))
      else
        fnAgree()
      end
    end
  end,
  btnAdd = function(self)
    local tbSkillData = self.tbCurSelectSkill
    if not tbSkillData then
      return
    end
    local nNpcId = LifeSkill:GetLifeSkillInfo(tbSkillData.nSkill).nTeacherId
    if not self.nCurSelectRecipeId then
      return
    end
    local tbRecipeInfo = LifeSkill.tbRecipes[self.nCurSelectRecipeId]
    local nItemTemplateId = tbRecipeInfo.nMaterialId
    local tbWare = Shop:FindNpcShopWare(nNpcId, nItemTemplateId)
    if not tbWare then
      return
    end
    local nRemainCount = Shop:GetShopRemainCount(tbWare, me)
    local tbItem = {
      nWareId = tbWare.nId,
      nItemId = nItemTemplateId,
      szCurrencyType = tbWare.szCurrencyType,
      nPrice = tbWare.nPrice,
      nCount = 1,
      nRemainCount = nRemainCount
    }
    Ui:OpenWindow("UIShopBuySmall", tbItem, true)
  end,
  btnLZAdd = function(self)
    self:ShowWayButtons()
  end,
  btnJingHuoAdd = function(self)
    self:ShowWayButtons()
  end,
  btnSpecialSet = function(self)
    local tbPlayerData = LifeSkill:GetClientData()
    if tbPlayerData.nSpecialty then
      return
    end
    local tbElement = self.pLifeSkillView:GetElement(self.pLifeSkillView:GetIndex())
    LifeSkill:OnS2CSetSpecialtyComfirm(tbElement.tbData.nSkill)
  end,
  btnSpecialForget = function(self)
    local tbPlayerData = LifeSkill:GetClientData()
    if not tbPlayerData.nSpecialty then
      return
    end
    LifeSkill:ForgetSpecialty(tbPlayerData.nSpecialty)
  end,
  JumpToEmpty4Raycast = function(self)
    self.pPanel:SetActive(tbControls.wndJumpTo, false)
    self.tbWayScrollView:Clear()
  end,
  btnLZCreate = function(self)
    EquipCuiLing:LZSuiPianForgeC()
  end
}

function tbWnd:OnCreate()
  self.pLifeSkillView = Ui:NewScrollView(self, tbControls.LifeSkillTemplate, function(_, nIndex, tbElement)
    return self:OnSelectLifeSkill(nIndex, tbElement)
  end, false, false)
  self.tbProductView = {}
  for i, szControl in ipairs(self.tbProduct) do
    self.tbProductView[i] = Ui:NewExistItemGrid(self, szControl .. "/UIItemGrid")
  end
  self.tbLZSuiPianProductView = {}
  for i, szControl in ipairs(tbControls.tbProductLZSuiPian) do
    self.tbLZSuiPianProductView[i] = Ui:NewExistItemGrid(self, szControl .. "/UIItemGrid")
  end
  self.pRecipeTemplate = Ui:NewScrollView(self, tbControls.RecipeTemplate, function(_, nIndex, tbElement)
    return self:OnSelectRecipe(nIndex, tbElement)
  end, false, false)
  self.pPanel:Dropdown_BindEvent(tbControls.DropdownQuality, function(value)
    self.nCurSelectLifeSkillLv = value
    self:OnSelectLifeSkillLevel(value)
  end)
  self.pMaterial = Ui:NewExistItemGrid(self, tbControls.Material)
  self.pOtherProduct = Ui:NewExistItemGrid(self, tbControls.OtherProduct)
  self.pProductResult = Ui:NewExistItemGrid(self, tbControls.ProductResult)
  self.pLZProductResult = Ui:NewExistItemGrid(self, tbControls.LZProductResult)
  self.effectGrid = Ui:NewExistItemGrid(self, tbControls.effectUIItemGrid)
  self.MyEquipGrid = Ui:NewExistItemGrid(self, tbControls.MyEquipGrid)
  self.pPanel:SetActive(tbControls.MakeEffect, false)
  self.tbWayScrollView = Ui:NewScrollView(self, tbControls.WayTemplate)
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(tbControls.LZSuiPianMonthCard, false)
  self.pPanel:SetActive(tbControls.LZCostItem, false)
  self.pPanel:SetActive(tbControls.LZCostMoney, false)
  self.pLZProductResult:SetActive(false)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_DATA, self.OnSyncPlayerData, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_LIFESKILL_MODIFY_SPECIALTY, self.OnModifySpecialty, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_LEVELUP, self.OnLevelUp, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_CRAFT, self.OnUpdateCraftResult, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_MONEY, self.OnMoneyChange, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self.OnSyncItemEvent, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_DEL_ITEM, self.OnDelItem, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_SUIPIAN_FORGE_RESULT, self.OnLZSuiPianForgeRsp, self)
  self:OnSyncPlayerData()
  self:HideCraftResult()
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  EventNotify.OnNotify(EventNotify.emNOTIFY_LIFESKILL_WINDOW_OPEN)
end

  self.tbCurSelectSkill = nil
  self.nCurSelectRecipeId = nil
  self.nCurSelectLifeSkillLv = nil
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_DATA, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_LIFESKILL_MODIFY_SPECIALTY, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_LEVELUP, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_LIFESKILL_CRAFT, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_MONEY, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_DEL_ITEM, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_SUIPIAN_FORGE_RESULT, self)
  self:CloseTimerAndHideResult()

  self.MyEquipGrid:SetEmpty()
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  self.tbWayScrollView:Clear()
end

function tbWnd:UpdateCurrency(tbRecipeInfo)
  self.pPanel:Label_SetText(tbControls.txtToplimitEnergyValue, Currency:GetUserMaxValue(me, tbRecipeInfo.szCurrencyType))
  self.pPanel:Label_SetText(tbControls.txtLZToplimitEnergyValue, Currency:GetUserMaxValue(me, EquipCuiLing.szSuiPianForgeCurrency))
  self.pPanel:Label_SetText(tbControls.txtTotalEnergyValue, me.GetMoney(tbRecipeInfo.szCurrencyType))
  self.pPanel:Label_SetText(tbControls.txtLZTotalEnergyValue, me.GetMoney(EquipCuiLing.szSuiPianForgeCurrency))
  if WeekMonthCard:CheckMonthCardState(me) then
    self.pPanel:Label_SetColorByName(tbControls.txtState, "Green")
    self.pPanel:Label_SetText(tbControls.txtState, i18n.EasyGet("\239\188\136\229\183\178\230\191\128\230\180\187\239\188\137"))
  else
    self.pPanel:Label_SetColorByName(tbControls.txtState, "Red")
    self.pPanel:Label_SetText(tbControls.txtState, i18n.EasyGet("\239\188\136\230\156\170\230\191\128\230\180\187\239\188\137"))
  end
  local nDoubleRate = 1
  if LifeSkill:GetDoubleRate() then
    nDoubleRate = LifeSkill.DoubleRate
  end
  local nCost = math.ceil(tbRecipeInfo.nCost * nDoubleRate)
  local nCost2 = tbRecipeInfo.nCost2
  local tbData = LifeSkill:GetClientData()
  self.pPanel:Label_SetColorByName(tbControls.txtCostEnergyValue, "Green")
  self.pPanel:Label_SetColorByName(tbControls.txtLZCostEnergyValue, "Green")
  if tbData.nSpecialty ~= tbRecipeInfo.nSkill then
    nCost = math.ceil(tbRecipeInfo.nCost * LifeSkill:GetCraftCostCoefficient() * nDoubleRate)
    nCost2 = math.ceil(tbRecipeInfo.nCost2 * LifeSkill:GetCraftCostCoefficient())
    self.pPanel:Label_SetColorByID(tbControls.txtCostEnergyValue, 602)
  end
  self.pPanel:Label_SetText(tbControls.txtCostEnergyValue, nCost)
  self.pPanel:Label_SetText(tbControls.txtLZCostEnergyValue, EquipCuiLing.Cfg.nForgeSuiPianCostJingHuo)
  self.pPanel:Label_SetText(tbControls.txtCostMoney, nCost2)
end

function tbWnd:UpdateMaterial(tbRecipeInfo)
  self.pMaterial:SetItemByTemplate(tbRecipeInfo.nMaterialId)
  self.pMaterial:SetClickEnable(true)
  local nCount = me.GetItemCountInBags(tbRecipeInfo.nMaterialId)
  self.pPanel:Label_SetText(tbControls.MaterialCount, string.format("%d/%d", nCount, tbRecipeInfo.nMaterialCount))
  if nCount < tbRecipeInfo.nMaterialCount then
    self.pPanel:Label_SetColorByName(tbControls.MaterialCount, "NotEnough")
    self.pPanel:Button_SetEnable(tbControls.btnCraft, false)
  else
    self.pPanel:Label_SetColorByName(tbControls.MaterialCount, "Enough")
    self.pPanel:Button_SetEnable(tbControls.btnCraft, true)
  end
end

function tbWnd:OnSelectLifeSkill(nIndex, tbElement)
  local tbPlayerData = LifeSkill:GetClientData()
  if tbPlayerData and nIndex == Lib:CountTB(tbPlayerData.tbSkills) + 1 then
    Ui:OpenWindow("UIStoneCompose")
    Ui:CloseWindow("UILifeSkill")
    return
  end
  if tbPlayerData and nIndex == Lib:CountTB(tbPlayerData.tbSkills) + 2 then
    self.pPanel:SetActive(tbControls.panelCuiling, true)
    self.pPanel:SetActive(tbControls.ContLiftSkill, false)
    self.pPanel:SetActive(tbControls.panelCreate, false)
    self.tbOwner:OnSetToggleRareEquipmentShow(false)
    self:UpdateLingZhuView()
  else
    self.pPanel:SetActive(tbControls.panelCuiling, false)
    self.pPanel:SetActive(tbControls.ContLiftSkill, true)
    self.pPanel:SetActive(tbControls.panelCreate, true)
    self.tbOwner:OnSetToggleRareEquipmentShow(true)
    self:UpdateLifeSkillView(tbElement)
  end
end

function tbWnd:UpdateLifeSkillView(tbElement)
  local tbSkillData = tbElement.tbData
  if self.tbCurSelectSkill == nil then
    self.nCurSelectLifeSkillLv = nil
  elseif self.tbCurSelectSkill.nSkill ~= tbSkillData.nSkill then
    self.nCurSelectLifeSkillLv = nil
  end
  self.tbCurSelectSkill = tbSkillData
  local nSkillId = self.tbCurSelectSkill.nSkill
  local nNextLevelInfo = LifeSkill.tbLevel[nSkillId][tbSkillData.nLevel + 1]
  if not nNextLevelInfo then
    self.pPanel:SetActive(tbControls.btnLifeSkillUp, false)
  elseif me.nTotalLevel >= nNextLevelInfo.nPlayerLevel then
    self.pPanel:SetActive(tbControls.btnLifeSkillUp, true)
  else
    self.pPanel:SetActive(tbControls.btnLifeSkillUp, false)
  end
  self.pPanel:Label_SetText(tbControls.txtLifeSkillLevel, i18n.EasyGet("{0}\231\186\167", tbSkillData.nLevel))
  self.pPanel:Label_SetText(tbControls.txtLifeSkillName, tbSkillData.szName)
  local tbPlayerData = LifeSkill:GetClientData()
  local tbOptions = {}
  table.insert(tbOptions, i18n.Get(LifeSkill.MSG.DropdownAllOption))
  for i = 1, tbSkillData.nLevel do
    table.insert(tbOptions, i18n.Get(LifeSkill.MSG.DropdownOption, i))
  end
  local nCurSelectLifeSkillLv = nil ~= self.nCurSelectLifeSkillLv and self.nCurSelectLifeSkillLv or tbSkillData.nLevel
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownQuality, tbOptions)
  self.pPanel:Dropdown_SetValue(tbControls.DropdownQuality, nCurSelectLifeSkillLv)
  local nPlayerSpecialtyId = tbPlayerData.nSpecialty
  local bSpecialty = nPlayerSpecialtyId == self.tbCurSelectSkill.nSkill
  if bSpecialty and LifeSkill:CheckIsCanLifeSkillLvUp(nSkillId, self.tbCurSelectSkill.nLevel) then
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLifeSkillUp, {
      RedPointMgr.EVENT.LifeSkill
    })
  else
    RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLifeSkillUp)
  end
  self.pPanel:SetActive(tbControls.btnSpecialForget, bSpecialty and nPlayerSpecialtyId)
  self.pPanel:SetActive(tbControls.btnSpecialSet, not nPlayerSpecialtyId)
end

function tbWnd:OnSelectLifeSkillLevel(nValue)
  if not self.tbCurSelectSkill then
    return
  end
  local tbSkillData = self.tbCurSelectSkill
  local tbRecipes
  if 0 == nValue then
    tbRecipes = {}
    for i = 1, tbSkillData.nLevel do
      Lib:MergeTable(tbRecipes, LifeSkill.tbSkillRecipes[tbSkillData.nSkill][i])
    end
  else
    tbRecipes = LifeSkill.tbSkillRecipes[tbSkillData.nSkill][nValue]
  end
  local nIndex = 1
  self.pRecipeTemplate:Clear()
  for nElementIndex = 1, #tbRecipes do
    local tbRecipe = tbRecipes[nElementIndex]
    self.pRecipeTemplate:SetElement(nElementIndex, function(tbElement)
      tbElement.nRecipeId = tbRecipe.nRecipeId
      tbElement.pPanel:Label_SetText("txtEquipName", i18n.Parse(tbRecipe.szName))
      tbElement.pPanel:Label_SetText("txtDesc", i18n.Get(LifeSkill.MSG.PRODUCT_DESC_TIPS, Lib:GetLevelText(tonumber(tbRecipe.szDescParam))))
      tbElement.pPanel:Sprite_SetSprite("imgIcon", tbRecipe.szIcon)
      local bLowLevel = me.nTotalLevel + LifeSkill.nRequireLevelOffset < (tonumber(tbRecipe.szDescParam) or 0)
      tbElement.pPanel:SetActive("imgItemMask", bLowLevel and LifeSkill.bUseNewPlan)
      if bLowLevel and LifeSkill.bUseNewPlan then
        tbElement.pPanel:Label_SetColorByName("txtDesc", "Red")
      end
    end)
    if self.nCurSelectRecipeId == tbRecipe.nRecipeId then
      nIndex = nElementIndex
    end
  end
  self.pRecipeTemplate:SelectElement(nIndex)
end

function tbWnd:UpdatePossibilityAndGuaranteedText(tbRecipeInfo)
  for i = 1, #tbRecipeInfo.tbProducts do
    local szPossibilityText = math.floor(tbRecipeInfo.tbProductRate[i] / (LifeSkill.MaxRandValue / 100)) .. "%"
    self.pPanel:Label_SetText(self.tbProduct[i] .. "/txtPossibility", szPossibilityText)
  end
  self.pPanel:Label_SetText(tbControls.txtGuaranteedTips, "")
end

function tbWnd:OnSelectRecipe(nIndex, tbElement)
  local nRecipeId = tbElement.nRecipeId
  local bRecipeIdChange = self.nCurSelectRecipeId ~= nRecipeId
  self.nCurSelectRecipeId = nRecipeId
  local tbRecipeInfo = LifeSkill.tbRecipes[nRecipeId]
  local tbProduct = tbRecipeInfo.tbProducts
  for i = 1, #tbProduct do
    self.tbProductView[i]:SetItemByTemplate(tbProduct[i])
    self.tbProductView[i]:SetClickEnable(true)
  end
  self:UpdatePossibilityAndGuaranteedText(tbRecipeInfo)
  if bRecipeIdChange then
    self:CloseTimerAndHideResult()
  end
  self.pOtherProduct:SetItemByTemplate(tbRecipeInfo.nOtherId)
  self.pOtherProduct:SetClickEnable(true)
  local nByProduct = tbRecipeInfo.nOtherNums
  self.pOtherProduct:SetLabelSuffixText(nByProduct)
  local nRatePercent = LifeSkill:GetOtherRateByRecipeId(nRecipeId)
  self.pPanel:Label_SetText(tbControls.txtByProductRate, nRatePercent .. "%")
  self:UpdateMaterial(tbRecipeInfo)
  self:UpdateCurrency(tbRecipeInfo)
  local nEquipPos = LifeSkill:GetEquipPosByRecipeId(self.nCurSelectRecipeId)
  local pItem = me.GetEquipByPos(nEquipPos)
  if pItem then
    self.MyEquipGrid:SetItem(pItem.dwId, pItem.dwTemplateId)
    self.MyEquipGrid:SetClickEnable(true)
  else
    self.MyEquipGrid:SetEmpty()
  end
end

function tbWnd:SelectCurLifeSkill(tbSkillList, nTargetLifeSkillId)
  for i = 1, #tbSkillList do
    local tbLifeSkillData = tbSkillList[i]
    if tbLifeSkillData.nSkill == nTargetLifeSkillId then
      Timer:Register(3, function()
        self.pLifeSkillView:SelectElement(i)
      end)
      return
    end
  end
  Timer:Register(3, function()
    self.pLifeSkillView:SelectElement(1)
  end)
end

function tbWnd:SetLiftSkillSelect(nIndex)
  if self.pLifeSkillView then
    self.pLifeSkillView:SelectElement(nIndex)
  end
end

function tbWnd:OnModifySpecialty()
  local tbPlayerData = LifeSkill:GetClientData()
  if not tbPlayerData then
    return
  end
  local tbSkillList = {}
  for nId, nLevel in pairs(tbPlayerData.tbSkills) do
    local tbInfo = LifeSkill:GetLifeSkillInfo(nId)
    table.insert(tbSkillList, {
      szName = i18n.Parse(tbInfo.szBtnName),
      nSkill = nId,
      nLevel = nLevel,
      szNormalIcon = tbInfo.szNormalIcon,
      szSelectedIcon = tbInfo.szSelectedIcon
    })
  end
  for i = 1, self.pLifeSkillView:GetCount() do
    local tbElement = self.pLifeSkillView:GetElement(i)
    local bSpecialty = tbElement.tbData.nSkill == tbPlayerData.nSpecialty
    if tbElement.bSpecialty then
      tbElement.pPanel:SetActive("imgSpecial", false)
      RedPointMgr:DynamicUnRegist(self.pPanel, tbElement.szKey)
    end
    if bSpecialty then
      tbElement.pPanel:SetActive("imgSpecial", true)
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbElement.szKey, {
        RedPointMgr.EVENT.LifeSkill
      })
    end
  end
  self.pLifeSkillView:OnClickElement(self.pLifeSkillView:GetIndex())
end

function tbWnd:OnSyncPlayerData()
  for i = 1, self.pLifeSkillView:GetCount() do
    local tbElement = self.pLifeSkillView:GetElement(i)
    if tbElement.bSpecialty then
      RedPointMgr:DynamicUnRegist(self.pPanel, tbElement.szKey)
    end
  end
  self.pLifeSkillView:Clear()
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownQuality, {})
  local tbPlayerData = LifeSkill:GetClientData()
  if not tbPlayerData then
    return
  end
  local tbSkillList = {}
  for nId, nLevel in pairs(tbPlayerData.tbSkills) do
    local tbInfo = LifeSkill:GetLifeSkillInfo(nId)
    table.insert(tbSkillList, {
      szName = i18n.Parse(tbInfo.szBtnName),
      nSkill = nId,
      nLevel = nLevel,
      szNormalIcon = tbInfo.szNormalIcon,
      szSelectedIcon = tbInfo.szSelectedIcon
    })
  end
  for i = 1, #tbSkillList do
    local tbSkillData = tbSkillList[i]
    local bSpecialty = tbPlayerData.nSpecialty == tbSkillData.nSkill
    self.pLifeSkillView:SetElement(i, function(tbElement)
      if bSpecialty then
        RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbElement.szKey, {
          RedPointMgr.EVENT.LifeSkill
        })
      end
      tbElement.tbData = tbSkillData
      tbElement.bSpecialty = bSpecialty
      tbElement.pPanel:SetActive("imgSpecial", bSpecialty)
      self.pPanel:ButtonGroup_SetSprite(tbElement.szKey, 1, tbSkillData.szSelectedIcon, tbSkillData.szNormalIcon)
      tbElement.pPanel:Label_SetText("Text", tbSkillData.szName)
      tbElement.pPanel:Label_SetText("imgLevel/txtLevel", tbSkillData.nLevel)
    end)
  end
  if 1 == GetTimeFrameState(Item.Stone.tbLifeSkillUI.szOpenTimeFrame) then
    self.pLifeSkillView:SetElement(#tbSkillList + 1, function(tbElement)
      tbElement.tbData = {}
      tbElement.bSpecialty = false
      tbElement.pPanel:SetActive("imgSpecial", false)
      tbElement.pPanel:SetActive("imgLevel", false)
      tbElement.pPanel:SetActive("imgLevel/txtLevel", false)
      self.pPanel:ButtonGroup_SetSprite(tbElement.szKey, 1, "UI/atlas/skill/btn_Gem_normal.png", "UI/atlas/skill/btn_Gem_normal.png")
      tbElement.pPanel:Label_SetText("Text", i18n.EasyGet("\229\174\157\231\159\179"))
    end)
  end
  if EquipCuiLing:IsSystemOpen() then
    self.pLifeSkillView:SetElement(#tbSkillList + 2, function(tbElement)
      tbElement.tbData = {}
      tbElement.bSpecialty = false
      tbElement.pPanel:SetActive("imgSpecial", false)
      tbElement.pPanel:SetActive("imgLevel", false)
      tbElement.pPanel:SetActive("imgLevel/txtLevel", false)
      self.pPanel:ButtonGroup_SetSprite(tbElement.szKey, 1, "UI/atlas/skill/btn_Gem_normal.png", "UI/atlas/skill/btn_Gem_normal.png")
      tbElement.pPanel:Label_SetText("Text", i18n.EasyGet("\231\129\181\231\143\160\231\162\142\231\137\135"))
    end)
  end
  local nTargetLifeSkillId = self.tbCurSelectSkill ~= nil and self.tbCurSelectSkill.nSkill or tbPlayerData.nSpecialty
  self:SelectCurLifeSkill(tbSkillList, nTargetLifeSkillId)
end

function tbWnd:SelectSkill(nSkill)
  for i = 1, self.pLifeSkillView:GetCount() do
    local tbElement = self.pLifeSkillView:GetElement(i)
    if tbElement.tbData.nSkill == nSkill then
      self.pLifeSkillView:SelectElement(i)
      break
    end
  end
end

function tbWnd:OnLevelUp(nSkill, nLevel)
  local tbSkillData = self.tbCurSelectSkill
  if not tbSkillData then
    return
  end
  if tbSkillData.nSkill ~= nSkill then
    return
  end
  tbSkillData.nLevel = nLevel
  self.pPanel:Label_SetText(tbControls.txtLifeSkillLevel, i18n.EasyGet("{0}\231\186\167", nLevel))
  local tbOptions = {}
  table.insert(tbOptions, i18n.Get(LifeSkill.MSG.DropdownAllOption))
  for i = 1, tbSkillData.nLevel do
    table.insert(tbOptions, i18n.Get(LifeSkill.MSG.DropdownOption, i))
  end
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownQuality, tbOptions)
  self.pPanel:Dropdown_SetValue(tbControls.DropdownQuality, tbSkillData.nLevel)
  local nNextLevelInfo = LifeSkill.tbLevel[nSkill][nLevel + 1]
  if not nNextLevelInfo then
    self.pPanel:SetActive(tbControls.btnLifeSkillUp, false)
  elseif me.nTotalLevel >= nNextLevelInfo.nPlayerLevel then
    self.pPanel:SetActive(tbControls.btnLifeSkillUp, true)
  else
    self.pPanel:SetActive(tbControls.btnLifeSkillUp, false)
  end
  for i = 1, self.pLifeSkillView:GetCount() do
    local tbElement = self.pLifeSkillView:GetElement(i)
    if tbElement.tbData.nSkill == nSkill then
      tbElement.tbData.nLevel = nLevel
      tbElement.pPanel:Label_SetText("imgLevel/txtLevel", nLevel)
    end
  end
end

function tbWnd:OnSyncItemEvent(nItemId)
  if self.nCurSelectRecipeId then
    local pItem = me.GetItemInAllSpace(nItemId)
    local tbRecipeInfo = LifeSkill.tbRecipes[self.nCurSelectRecipeId]
    if pItem and tbRecipeInfo.nMaterialId == pItem.dwTemplateId then
      self:UpdateMaterial(tbRecipeInfo)
    end
  end
end

function tbWnd:OnDelItem(_, dwTemplateId)
  if self.nCurSelectRecipeId then
    local tbRecipeInfo = LifeSkill.tbRecipes[self.nCurSelectRecipeId]
    if tbRecipeInfo.nMaterialId == dwTemplateId then
      self:UpdateMaterial(tbRecipeInfo)
    end
  end
end

function tbWnd:ShowWayButtons()
  self.tbWayScrollView:Clear()
  for i = 1, #LifeSkill.CFG_WAY do
    local tbWayInfo = LifeSkill.CFG_WAY[i]
    local nLevelLimit = tonumber(tbWayInfo.Param3) or 0
    if nLevelLimit <= 0 or nLevelLimit > 0 and nLevelLimit <= me.nTotalLevel then
      self.tbWayScrollView:SetElement(i, function(tbElement)
        tbElement.pPanel:Label_SetText("txtName", i18n.Parse(tbWayInfo.szWayName))
        tbElement.pPanel:Label_SetText("txtDesc", i18n.Parse(tbWayInfo.szWayDesc))
        if tbWayInfo.fnOnTrack then
          Ui:Button_BindEvent(tbElement.pPanel, "", function()
            tbWayInfo.fnOnTrack()
            self.pPanel:SetActive(tbControls.wndJumpTo, false)
            self.tbWayScrollView:Clear()
          end)
        else
          tbElement.pPanel:SetActive("imgArrow", false)
        end
      end)
    end
  end
  self.pPanel:SetActive(tbControls.wndJumpTo, true)
end
