local tbExternAttribSetting = Lib:LoadTabFile("Setting/Item/ExternAttrib.tab", {
  GroupId = true,
  Level = true,
  FightPower = true,
  Attrib1Type = false,
  Attrib1Value1 = true,
  Attrib1Value2 = true,
  Attrib1Value3 = true,
  Attrib2Type = false,
  Attrib2Value1 = true,
  Attrib2Value2 = true,
  Attrib2Value3 = true,
  Attrib3Type = false,
  Attrib3Value1 = true,
  Attrib3Value2 = true,
  Attrib3Value3 = true,
  Attrib4Type = false,
  Attrib4Value1 = true,
  Attrib4Value2 = true,
  Attrib4Value3 = true,
  Attrib5Type = false,
  Attrib5Value1 = true,
  Attrib5Value2 = true,
  Attrib5Value3 = true
})

function FightSkill:InitExternAttribInfo()
  self.tbExternAttrib = {}
  for _, tbSetting in pairs(tbExternAttribSetting) do
    local nGroupId = tbSetting.GroupId
    if not self.tbExternAttrib[nGroupId] then
      self.tbExternAttrib[nGroupId] = {}
    end
    local tbGroupAttrib = self.tbExternAttrib[nGroupId]
    local nLevel = tbSetting.Level
    if not tbGroupAttrib[nLevel] then
      tbGroupAttrib[nLevel] = {}
    end
    local tbGroupLevelAttrib = tbGroupAttrib[nLevel]
    tbGroupLevelAttrib.nFightPower = tbSetting.FightPower
    for i = 1, 5 do
      local szAttribName = tbSetting["Attrib" .. i .. "Type"]
      if "" ~= szAttribName then
        local tbValue = {
          tbSetting["Attrib" .. i .. "Value1"],
          tbSetting["Attrib" .. i .. "Value2"],
          tbSetting["Attrib" .. i .. "Value3"]
        }
        tbGroupLevelAttrib[i] = {szAttribName = szAttribName, tbValue = tbValue}
      end
    end
  end
  self.tbExternAttrib = SetReadOnly(self.tbExternAttrib)
end

FightSkill:InitExternAttribInfo()

function FightSkill:GetExternAttrib(nGroupId, nLevel)
  if not tonumber(nGroupId) or not tonumber(nLevel) then
    return
  end
  if not self.tbExternAttrib[nGroupId] then
    return
  end
  return self.tbExternAttrib[nGroupId][nLevel]
end
