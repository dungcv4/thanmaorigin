local tbWnd = Ui:CreateClass("UICommonSkill_GuanXian", "UIPanel")
local tbControls = {
  NoGuanYinEmpty = "Empty",
  IconGuanxian = "IconGuanxian",
  txtName = "txtName",
  SkillIcon = "SkillIcon",
  btnSkill = "SkillIcon/btnSkill",
  txtSkillName = "txtSkillName",
  txtSkillDesc = "txtSkillDesc"
}
tbWnd.tbControls = tbControls

function tbWnd:OnOpen()
  self:UpdateInfo()
end

function tbWnd:UpdateInfo()
  local pEquip = me.GetEquipByPos(Item.EQUIPPOS_GUANYIN)
  local nTemplateId
  if nil ~= pEquip then
    nTemplateId = pEquip.dwTemplateId
  end
  local bHaveGuanYin = nil ~= nTemplateId
  self.pPanel:SetActive(tbControls.NoGuanYinEmpty, not bHaveGuanYin)
  self.pPanel:SetActive(tbControls.IconGuanxian, bHaveGuanYin)
  self.pPanel:SetActive(tbControls.txtName, bHaveGuanYin)
  local nSkillId
  if bHaveGuanYin then
    nSkillId = Item.GuanYin:GetSkillIdByTemplateId(nTemplateId)
  end
  local bSkillExsit = nil ~= nSkillId
  self.pPanel:SetActive(tbControls.SkillIcon, bSkillExsit)
  self.pPanel:SetActive(tbControls.txtSkillName, bSkillExsit)
  self.pPanel:SetActive(tbControls.txtSkillDesc, bSkillExsit)
  if bHaveGuanYin then
    local tbGuanYinInfo = Item.GuanYin.tbGuanYinData[nTemplateId]
    local _, szIconPath = Item:GetIcon(tbGuanYinInfo.BigIcon)
    self.pPanel:Sprite_SetSprite(tbControls.IconGuanxian, szIconPath)
    self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(tbGuanYinInfo.Name))
    if bSkillExsit then
      self.pPanel:Sprite_SetSprite(tbControls.btnSkill, FightSkill:GetSkillIconAtlas(nSkillId))
      local tbSkillInfo = FightSkill.tbAllSkillSetting[nSkillId]
      self.pPanel:Label_SetText(tbControls.txtSkillName, i18n.Parse(tbSkillInfo.szSkillName))
      self.pPanel:Label_SetText(tbControls.txtSkillDesc, i18n.Parse(tbSkillInfo.szDesc))
    end
  end
end
