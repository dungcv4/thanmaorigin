local tbWnd = Ui:CreateWindow("UIHorseSkillGallery")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  btnClose = "node/imgBG/btnClose",
  lvInfo = "node/imgBG/Content/lvInfo",
  txtNub = "node/imgBG/Content/starsInfo/txtNub",
  Template = "node/imgBG/Content/list/Scroll View/Viewport/Content/Template",
  DropdownQuality = "node/imgBG/Content/DropdownQualityLV",
  DropdownLevel = "node/imgBG/Content/DropdownQualityRank"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("ZuoQiBeiDongJiNeng")
  end
}

function tbWnd:OnCreate()
  local tbLevelOptions = {}
  local nOpenGrade = HorseSkillMgr:GetHSOpenGrade()
  for i = 1, nOpenGrade do
    table.insert(tbLevelOptions, i18n.EasyGet("{0}\233\152\182", i))
  end
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownLevel, tbLevelOptions)
  self.pPanel:Dropdown_BindEvent(tbControls.DropdownLevel, function(index)
    self.nLevelIndex = index + 1
    self:UpdateView()
  end)
  local tbStarOptions = {}
  for i = 1, #HorseSkillMgr.tbPassiveSkillLevelCfg[1] do
    table.insert(tbStarOptions, i18n.EasyGet("{0}\231\186\167", i))
  end
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownQuality, tbStarOptions)
  self.pPanel:Dropdown_BindEvent(tbControls.DropdownQuality, function(index)
    self.nStarIndex = index + 1
    self:UpdateView()
  end)
  self.tbSkillScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnDestroy()
end

function tbWnd:OnOpen()
  self.nLevelIndex = HorseSkillMgr:GetHSOpenGrade()
  self.nStarIndex = 1
  self.pPanel:Dropdown_SetValue(tbControls.DropdownLevel, self.nLevelIndex - 1)
  self.pPanel:Dropdown_SetValue(tbControls.DropdownQuality, self.nStarIndex - 1)
  self:UpdateView()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateView()
  self.pPanel:Label_SetText(tbControls.lvInfo, i18n.EasyGet("\230\138\128\232\131\189\231\173\137\231\186\167\239\188\154{0}", self.nStarIndex))
  self.pPanel:Label_SetText(tbControls.txtNub, HorseSkillMgr.tbPassiveSkillLevelCfg[self.nLevelIndex][self.nStarIndex][1])
  self:UpdateSkillSV()
end

function tbWnd:OnSkillClick()
end

function tbWnd:UpdateSkillSV()
  local nRealSkillLevel = HorseSkillMgr.tbPassiveSkillLevelCfg[self.nLevelIndex][self.nStarIndex][2]
  local tbSeriesMap = {
    [1] = "TxtMetal",
    [2] = "TxtWood",
    [3] = "TxtWater",
    [4] = "TxtFire",
    [5] = "TxtEarth"
  }
  self.tbSkillScrollView:Reset()
  for i, tbV in pairs(HorseSkillMgr.tbPassiveSkillId) do
    self.tbSkillScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local nSkillId = tbV.nSkillId
      local tbSeries = tbV.tbSeries
      for nIndex, nSeries in ipairs(tbSeries) do
        for j = 1, #tbSeriesMap do
          pPanel:SetActive(string.format("elementsCell/Series0%s/%s", nIndex, tbSeriesMap[j]), j == nSeries)
        end
      end
      local szSkillIcon = FightSkill:GetSkillIconAtlas(nSkillId)
      pPanel:Sprite_SetSprite("Img_Skill", szSkillIcon)
      local szSkillName = FightSkill:GetSkillName(nSkillId)
      pPanel:Label_SetText("txtSkillName", i18n.Parse(szSkillName))
      local szDesc = FightSkill:GetSkillMagicDesc(nSkillId, nRealSkillLevel)
      szDesc = i18n.Parse(szDesc)
      szDesc = string.gsub(szDesc, "\n", " ")
      pPanel:Label_SetText("txtSkillInfo", szDesc)
    end)
  end
end
