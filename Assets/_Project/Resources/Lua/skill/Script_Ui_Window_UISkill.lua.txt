local tbWnd = Ui:CreateWindow("UISkill")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/Content/FactionSkillView/btnHelp",
  btnZongShiHelp = "node/imgBG/Content/Master/btnHelp",
  btnFactionSkill = "node/imgBG/btnList/btnFactionSkill",
  btnLifeSkill = "node/imgBG/btnList/btnLifeSkill",
  btnKinSkill = "node/imgBG/btnList/btnKinSkill",
  btnCommonSkill = "node/imgBG/btnList/btnCommonSkill",
  btnZongShiSkill = "node/imgBG/btnList/btnMaster",
  btnMartialArts = "node/imgBG/btnList/btnMartialArts",
  panelFactionSkill = "node/imgBG/Content/FactionSkillView",
  panelLiftSkill = "node/imgBG/Content/LifeSkillView",
  panelListSkill = "node/imgBG/Content/KinSkillView",
  panelCommonSkill = "node/imgBG/Content/CommonSkillView",
  panelZongShiSkill = "node/imgBG/Content/Master"
}
tbWnd.tbControls = tbControls
tbWnd.PANEL_TYPE_FACTION = 1
tbWnd.PANEL_TYPE_KIN = 2
tbWnd.PANEL_TYPE_COMMON = 3
tbWnd.PANEL_TYPE_ZONGSHI = 4
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_TYPE_FACTION] = {
    CLASS = "UIFactionSkill",
    OBJ = tbControls.panelFactionSkill,
    BTN = tbControls.btnFactionSkill,
    OpenFunc = function()
      return true
    end
  },
  [tbWnd.PANEL_TYPE_KIN] = {
    CLASS = "UIKinSkill",
    OBJ = tbControls.panelListSkill,
    BTN = tbControls.btnKinSkill,
    OpenFunc = function()
      return Kin:IsKinSKillUnlock()
    end
  },
  [tbWnd.PANEL_TYPE_COMMON] = {
    CLASS = "UICommonSkill",
    OBJ = tbControls.panelCommonSkill,
    BTN = tbControls.btnCommonSkill,
    OpenFunc = function()
      return true
    end
  },
  [tbWnd.PANEL_TYPE_ZONGSHI] = {
    CLASS = "UIZongShiSkill",
    OBJ = tbControls.panelZongShiSkill,
    BTN = tbControls.btnZongShiSkill,
    OpenFunc = function()
      return ZongShiSkill:IsZongShiUnlock(me)
    end
  }
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
    Guide:ClickWindowCloseCallSkipGuide(Guide.tbClickCloseButtonSkipGuide.UISkill)
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("BattleSkillHelp")
  end,
  btnZongShiHelp = function(self)
    Ui:OnHelpClicked("ZongShiSkillHelp")
  end,
  btnMartialArts = function(self)
    Ui:OpenWindow("UIMartialArtsMain")
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
end

function tbWnd:OnOpen(nSelectPanel, bEnteringWithKin, bEnteringWithHanWu)
  if not self.tbTabControl:OnOpen(nSelectPanel, bEnteringWithKin, bEnteringWithHanWu) then
    return
  end
  self.pPanel:SetActive(self.btnCommonSkill, true)
  self.pPanel:SetActive(self.btnKinSkill, true)
  if true == bEnteringWithHanWu then
    self.pPanel:SetActive(self.btnCommonSkill, false)
    self.pPanel:SetActive(self.btnKinSkill, false)
    self.pPanel:SetActive(self.btnZongShiSkill, false)
    return false
  end
  if DingFengPK:CheckInDingFengPK() then
    self.pPanel:SetActive(self.btnKinSkill, false)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnFactionSkill, {
    RedPointMgr.EVENT.FactionSkill
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnCommonSkill, {
    RedPointMgr.EVENT.SecretBookLevelEnough,
    RedPointMgr.EVENT.SecretBookCanLevelUp
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnKinSkill, {
    RedPointMgr.EVENT.KinSkill
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnZongShiSkill, {
    RedPointMgr.EVENT.ZongShiSkill
  })
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnFactionSkill)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnCommonSkill)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnKinSkill)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnZongShiSkill)
end

function tbWnd:UpdateFactView()
  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_FACTION, "UpdateView")
end

function tbWnd:UpdateZongShiView(bUpdateSkill)
  self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_ZONGSHI, "OnSyncPlayerData", bUpdateSkill)
end

function tbWnd:OnPanelChange(nPanelType)
end
