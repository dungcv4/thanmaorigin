function tbWnd:OnOpen(tbInfo)
  self.nItemId = tbInfo.nItemId
  self.nTemplateId = tbInfo.nTemplateId
  self.nFaction = tbInfo.nFaction or me.nFaction
  self.nFactionSect = tbInfo.nFactionSect or me.nFactionSect
  self.nSex = tbInfo.nSex or me.GetNpc().GetBodySex()
  self.szOpt = tbInfo.szOpt
  self.nSeries = tbInfo.nSeries
  self.tbTradeInfo = nil
  if tbInfo.tbItemExtParams and tbInfo.tbItemExtParams.tbTradeInfo then
    self.tbTradeInfo = tbInfo.tbItemExtParams.tbTradeInfo
  end
  self.pPanel:Object_SetLocalPosition(self.panelRoot, tbInfo.pos)
  if 0 == self.nItemId then
    self.nItemId = nil
  end
  if self.nItemId then
    local pItem = KItem.GetItemObj(self.nItemId)
    self.pItem = pItem
    self.nTemplateId = pItem.dwTemplateId
  else
    self.pItem = nil
  end
  self.tbFuncOnButtonClick = {}
  local tbDetailInfo = {
    szOpt = self.szOpt,
    pItem = self.pItem,
    nItemId = self.nItemId,
    nTemplateId = self.nTemplateId,
    nFaction = self.nFaction,
    nFactionSect = self.nFactionSect,
    nSex = self.nSex,
    pAsyncData = tbInfo.pAsyncData,
    bIsCompare = self.ClassName == "UICompareTipsPartnerEquip",
    tbItemExtParams = tbInfo.tbItemExtParams,
    tbLinkInfo = tbInfo.tbLinkInfo,
    nSeries = self.nSeries
  }
  self.tbEquipDetail:Update(tbDetailInfo)
  self:UpdateButton()
  self:UpdateTradePanel()
end

function tbWnd:UpdateButton()
  local pItem = self.pItem
  local szOpt = self.szOpt
  if pItem and szOpt and self.tbButtonOption[szOpt] and not self.pAsyncData then
    self.tbButtonGroup:SetActiveAll(true)
    self.tbDynamicButtonsTypeToIndex = {}
    local nIndex = 1
    for _, tbButton in ipairs(self.tbButtonOption[szOpt]) do
      local fnVisible, szBtnName, fnCallback, fnCheckRedPoint, nButtonType = table.unpack(tbButton)
      if fnVisible and fnVisible(self) then
        local bLightRedPoint = fnCheckRedPoint and fnCheckRedPoint(self, pItem.dwId) or false
        self.tbButtonGroup:SetActive(nIndex, true)
        if type(szBtnName) == "string" then
          self.tbButtonGroup:SetButtonText(nIndex, szBtnName)
        elseif type(szBtnName) == "function" then
          self.tbButtonGroup:SetButtonText(nIndex, szBtnName(tbWnd))
        end
        self.tbFuncOnButtonClick[nIndex] = fnCallback
        if bLightRedPoint then
          self.tbButtonGroup:LightRedPoint(nIndex)
        else
          self.tbButtonGroup:FadeRedPoint(nIndex)
        end
        self.tbDynamicButtonsTypeToIndex[nButtonType] = nIndex
        nIndex = nIndex + 1
      end
    end
    for i = nIndex, #self.tbButtonGroupKey do
      self.tbButtonGroup:SetActive(i, false)
      self.tbButtonGroup:FadeRedPoint(i)
      self.tbFuncOnButtonClick[i] = nil
    end
    self.pPanel:SetActive(tbControls.btnShare, not self.tbButtonOption[szOpt].bHideShareButton)
    self.pPanel:SetActive(tbControls.btnHawk, false)
    local bShowSell = not self.tbButtonOption[szOpt].bHideSellButton
    local bCanSell, _ = Item.tbItemSell:CheckCanSellItem(pItem.dwTemplateId)
    self.pPanel:SetActive(tbControls.btnSell, bCanSell and bShowSell)
    local nCanMoveToBox = false
    local nVipState = false
    if pItem and self.szOpt == "ItemBox" then
      local tbBaseInfo = Item:GetItemBaseProp(pItem.dwTemplateId)
      if tbBaseInfo.bForbidToBox and 0 == tbBaseInfo.bForbidToBox then
        nCanMoveToBox = true
      end
      local weekCardLeft = WeekMonthCard:GetWeekCardLeftDay(me)
      local bMonthCardState = WeekMonthCard:CheckMonthCardState(me)
      if weekCardLeft > 0 or bMonthCardState then
        nVipState = true
      end
    end
    self.pPanel:SetActive(self.btnMoveToBox, nCanMoveToBox and nVipState)
    self.pPanel:Label_SetText(self.btnMoveToBox .. "/Text", i18n.EasyGet("\231\167\187\229\133\165\228\187\147\229\186\147"))
    return
  end
  self.tbButtonGroup:SetActiveAll(false)
  self.pPanel:SetActive(tbControls.btnShare, false)
  self.pPanel:SetActive(tbControls.btnHawk, false)
  self.pPanel:SetActive(tbControls.btnSell, false)
  self.pPanel:SetActive(tbControls.btnMoveToBox, false)
end

function tbWnd:UpdateTradePanel()
  if not self.tbTradeInfo then
    for _, szWndType in ipairs(self.tbTradePanel) do
      self.pPanel:SetActive(szWndType, false)
    end
    return
  end
  local bNotInTBPanel = true
  local szWndType = self.tbTradeInfo.szWndType
  for k, v in ipairs(self.tbTradePanel) do
    if szWndType == v then
      bNotInTBPanel = false
      break
    end
  end
  if bNotInTBPanel then
    local szWnd = Ui:CreatePanel(szWndType, szWndType, self)
    table.insert(self.tbTradePanel, szWndType)
    self.tbTradePanelObj = self.tbTradePanelObj or {}
    self.tbTradePanelObj[szWndType] = szWnd
  end
  if "UIMarketStallBuyPanel" == szWndType then
    self.tbTradePanelObj[szWndType]:OnOpen(self.tbTradeInfo.nStallType, self.tbTradeInfo.nSeTypeId, self.tbTradeInfo.nStallItemTempId, self.tbTradeInfo.tbItem)
  elseif "UIMarketStallSellPanel" == szWndType then
    self.tbTradePanelObj[szWndType]:OnOpen(self.tbTradeInfo.nStallType, self.tbTradeInfo.nShowType, self.tbTradeInfo.index, self.tbTradeInfo.bBag, self.tbTradeInfo.tbItem and self.tbTradeInfo.tbItem.nPrice, self.tbTradeInfo.nTimeout)
  elseif "UITradeItemPanel" == szWndType then
    self.tbTradePanelObj[szWndType]:OnOpen(self.tbTradeInfo.nItemId, self.tbTradeInfo.nOnTradeCount, self.tbTradeInfo.nMaxCount, self.tbTradeInfo.bOnTrade)
  end
  if tbWnd.szlastPanel then
    self.pPanel:SetActive(tbWnd.szlastPanel, false)
  end
  tbWnd.szlastPanel = szWndType
  self.pPanel:SetActive(szWndType, true)
end

function tbWnd:DecomposeEquip()
  local pItem = KItem.GetItemObj(self.nItemId)
  if not pItem then
    return
  end
  if not PartnerEquipC:CheckIsPartnerEquip(pItem) then
    return
  end
  
  local function fnOK(nItemId)
    PartnerEquipC:ReqPartnerEquipDecompose(nItemId)
    self:Close()
  end
  
  local nRewardNum = PartnerEquipC:GetNeedRewardMaterial(pItem)
  if not nRewardNum then
    return
  end
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\231\161\174\232\174\164\229\136\134\232\167\163\229\185\182\232\142\183\229\190\151{0}\231\142\132\231\159\179\239\188\159", nRewardNum), {
    {
      fnOK,
      self.nItemId
    },
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end
