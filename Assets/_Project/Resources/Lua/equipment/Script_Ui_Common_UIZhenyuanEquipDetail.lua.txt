function tbWnd:OnCreate()
  self.itemGrid = Ui:NewExistItemGrid(self, tbControls.ItemGridTemplate)
  self.tbScrollView = Ui:NewScrollView(self, tbControls.DescTemplate, nil, false, true)
  self.tbTradePanel = {}
  self.tbExtraAttrScrollView = Ui:NewScrollView(self, tbControls.Attribute1)
end

function tbWnd:Update(tbInfo)
  self.szOpt = tbInfo.szOpt
  self.pItem = tbInfo.pItem
  self.nItemId = tbInfo.nItemId
  self.nTemplateId = tbInfo.nTemplateId
  self.nFaction = tbInfo.nFaction
  self.nFactionSect = tbInfo.nFactionSect
  self.nSex = tbInfo.nSex
  self.pAsyncData = tbInfo.pAsyncData
  self.bIsCompare = tbInfo.bIsCompare
  self.nSeries = tbInfo.nSeries
  local tbItemInfo = Item:GetItemBaseProp(self.nTemplateId)
  tbInfo.tbItemExtParams = tbInfo.tbItemExtParams or {}
  self.nTimeOutValue = tbInfo.tbItemExtParams.nItemTimeOut or 0
  self.nTradeLimitType = tbInfo.tbItemExtParams.nTradeLimitType or tbItemInfo.nTradeLimitType
  self.nTradeLimitValue = tbInfo.tbItemExtParams.nTradeLimitValue or 0
  self.tbTradeInfo = tbInfo.tbItemExtParams.tbTradeInfo
  self.tbLinkInfo = tbInfo.tbLinkInfo
  self.tbItemExtParams = tbInfo.tbItemExtParams
  self.pPanel:SetActive(tbControls.Stars, false)
  self:UpdateSaleMoney()
  self:UpdateIcon()
  self:UpdateLevelAndDesc()
  self:UpdateBindStateDesc()
  self:UpdateEquiped()
  self:UpdateAttrib()
end

function tbWnd:GetItemGrid()
  return self.itemGrid
end

function tbWnd:UpdateSaleMoney()
  local szCurrency, nPrice = Item.tbItemSell:GetSellCurrencyValue(self.nTemplateId)
  if szCurrency then
    self.pPanel:SetActive(tbControls.SalePrice, true)
    self.pPanel:Label_SetText(tbControls.SalePrice .. "/txtSalePrice", nPrice)
    local szIcon = Currency:GetIcon(szCurrency)
    self.pPanel:Sprite_SetSprite(tbControls.SalePrice .. "/imgMoney", szIcon)
  else
    self.pPanel:SetActive(tbControls.SalePrice, false)
  end
end

function tbWnd:UpdateIcon()
  local szName = ""
  for nSeries, szPath in pairs(tbSeriesControls) do
    self.pPanel:SetActive(szPath, false)
  end
  if self.pItem then
    szName = KItem.GetItemShowInfo(self.pItem.dwTemplateId, self.nFaction or me.nFaction, self.nFactionSect or me.nFactionSect, self.nSex or me.nSex)
    local nSeries = self.pItem.GetSeries()
    local szSeriesPath = tbSeriesControls[nSeries]
    if 0 ~= nSeries then
      self.pPanel:SetActive(tbControls.txtSeries, true)
      self.pPanel:SetActive(szSeriesPath, true)
      self.pPanel:Label_SetText(szSeriesPath, Item.TB_SERIES_TEXT[nSeries])
    else
      self.pPanel:SetActive(tbControls.txtSeries, false)
    end
  else
    szName = KItem.GetItemShowInfo(self.nTemplateId, self.nFaction or me.nFaction, self.nFactionSect or me.nFactionSect, self.nSex or me.nSex)
    if self.nSeries and 0 ~= self.nSeries then
      local szSeriesPath = tbSeriesControls[self.nSeries]
      self.pPanel:SetActive(tbControls.txtSeries, true)
      self.pPanel:SetActive(szSeriesPath, true)
      self.pPanel:Label_SetText(szSeriesPath, Item.TB_SERIES_TEXT[self.nSeries])
    else
      self.pPanel:SetActive(tbControls.txtSeries, false)
    end
  end
  self:UpdateStar()
  local bShowTip = true
  if self.pAsyncData then
    bShowTip = false
  end
  local nLianHuaLevel = -1
  szName = i18n.Parse(szName)
  if EDITOR then
    szName = string.format("%s (ID:%d)", szName, self.nTemplateId)
  end
  self.pPanel:SetActive(tbControls.txtZhenyuanName, false)
  self.pPanel:Label_SetText(tbControls.txtTitle, szName)
  self.itemGrid:SetItem(self.nItemId, self.nTemplateId, {bShowTip = bShowTip}, self.nFaction, self.nFactionSect, self.nSex, self.pAsyncData, nil, {nLianHuaLevel = nLianHuaLevel})
end

function tbWnd:UpdateStar()
  local nStarNum = Item.ZhenYuan:GetMaxStar()
  for i = 1, #tbStarbgGroup do
    if i <= nStarNum then
      self.pPanel:SetActive(tbStarbgGroup[i], true)
    else
      self.pPanel:SetActive(tbStarbgGroup[i], false)
    end
  end
  local nEquipedType = self:GetHasEquippedType()
  if nEquipedType == Item.Equip.EQUIPED_TYPE.ME then
    local nSlot = Item.ExtEquip:GetSlot(self.pItem)
    local nLevel = Item.ZhenYuan:GetHoleEnhanLevel(me, nSlot)
    local nStar = 0
    if nLevel > 0 then
      nStar = math.floor(nLevel / 10)
    end
    local nStar2 = 0
    if nStar > 9 then
      nStar2 = nStar - 9
    end
    self.pPanel:SetActive(tbControls.Stars, true)
    for i = 1, #tbStarbgGroup do
      if nStar >= i then
        self.pPanel:SetActive(tbStarsGroup[i], true)
        self.pPanel:Sprite_SetSprite(tbStarsGroup[i], Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar1)
      else
        self.pPanel:SetActive(tbStarsGroup[i], false)
      end
      if nStar2 >= i then
        self.pPanel:Sprite_SetSprite(tbStarsGroup[i], Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar2)
      end
    end
  elseif nEquipedType == Item.Equip.EQUIPED_TYPE.ASYNCDATA then
    local nSlot = Item.ExtEquip:GetSlot(self.pItem)
    local nLevel = self.pAsyncData.GetZhenYuanHoleLevel(nSlot - Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL + 1)
    local nStar = 0
    if nLevel > 0 then
      nStar = math.floor(nLevel / 10)
    end
    local nStar2 = 0
    if nStar > 9 then
      nStar2 = nStar - 9
    end
    self.pPanel:SetActive(tbControls.Stars, true)
    for i = 1, #tbStarbgGroup do
      if nStar >= i then
        self.pPanel:SetActive(tbStarsGroup[i], true)
        self.pPanel:Sprite_SetSprite(tbStarsGroup[i], Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar1)
      else
        self.pPanel:SetActive(tbStarsGroup[i], false)
      end
      if nStar2 >= i then
        self.pPanel:Sprite_SetSprite(tbStarsGroup[i], Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar2)
      end
    end
  end
end

function tbWnd:UpdateLevelAndDesc()
  local tbInfo = Item:GetItemBaseProp(self.nTemplateId)
  local szColor = me.nTotalLevel >= tbInfo.nRequireLevel and "#52D455" or "#b60000"
  if ZongShiSkill:IsZongShiByLevel(tbInfo.nRequireLevel) then
    self.pPanel:Label_SetText(tbControls.txtLevelLimit, i18n.EasyGet("\233\156\128\230\177\130\231\173\137\231\186\167\239\188\154<color={0}>{1}</color>", szColor, Lib:GetLevelText(tbInfo.nRequireLevel)))
  else
    self.pPanel:Label_SetText(tbControls.txtLevelLimit, i18n.EasyGet("\233\156\128\230\177\130\231\173\137\231\186\167\239\188\154<color={0}>{1}</color>\231\186\167", szColor, Lib:GetLevelText(tbInfo.nRequireLevel)))
  end
  self.pPanel:Label_SetColorByName(tbControls.txtLevelLimit, Item.tbTipsColorDef.EquipLevel.ColorName or "Red")
  self.pPanel:Label_SetText(tbControls.txtRank, i18n.EasyGet("{0}\233\152\182", tbInfo.nLevel))
  self.pPanel:Label_SetColorByName(tbControls.txtRank, Item.tbTipsColorDef.EquipRank.ColorName or "Red")
  self.pPanel:SetActive(tbControls.txtLevelLimit, true)
  self.pPanel:SetActive(tbControls.txtRank, true)
end

function tbWnd:UpdateBindStateDesc()
  if self.pItem and 1 == self.pItem.dwTradeLimitType then
    self.pPanel:SetActive(tbControls.txtTradeLimitState, false)
    self.pPanel:SetActive(tbControls.txtTradable, true)
    self.pPanel:SetActive(tbControls.txtNonTradable, false)
  elseif not self.pItem and self.tbItemExtParams and self.tbItemExtParams.nBindType and self.tbItemExtParams.nBindType == Item.eBINDTYPE.UNBIND then
    self.pPanel:SetActive(tbControls.txtTradeLimitState, false)
    self.pPanel:SetActive(tbControls.txtTradable, true)
    self.pPanel:SetActive(tbControls.txtNonTradable, false)
  elseif not self.pItem and self.tbItemExtParams and self.tbItemExtParams.nTradeLimitType and self.tbItemExtParams.nTradeLimitType == Item.eBINDTYPE.UNBIND then
    self.pPanel:SetActive(tbControls.txtTradeLimitState, false)
    self.pPanel:SetActive(tbControls.txtTradable, true)
    self.pPanel:SetActive(tbControls.txtNonTradable, false)
  else
    self.pPanel:SetActive(tbControls.txtTradeLimitState, false)
    self.pPanel:SetActive(tbControls.txtTradable, false)
    self.pPanel:SetActive(tbControls.txtNonTradable, true)
    self.pPanel:Label_SetColorByName(tbControls.txtNonTradable, "NewOrange")
  end
end

function tbWnd:UpdateEquiped()
  local nEquipedType = self:GetHasEquippedType()
  self.pPanel:SetActive(tbControls.HasEquipped, nEquipedType == Item.Equip.EQUIPED_TYPE.ME)
end

function tbWnd:GetHasEquippedType()
  local nEquipedType = Item.Equip.EQUIPED_TYPE.NOT
  if self.pItem then
    if not self.pAsyncData then
      if self.pItem.nEquipPos == Item.EQUIPPOS_EXTEND then
        local nSlot = Item.ExtEquip:GetSlot(self.pItem)
        local pThisSlotWearEquip = Item.ExtEquip:GetEquipObj(me, nSlot)
        if nil ~= pThisSlotWearEquip and pThisSlotWearEquip.dwId == self.nItemId then
          nEquipedType = Item.Equip.EQUIPED_TYPE.ME
        end
      end
    else
      local nSlot = Item.ExtEquip:GetSlot(self.pItem)
      if self.tbItemExtParams[nSlot] and self.tbItemExtParams[nSlot] == self.nItemId then
        nEquipedType = Item.Equip.EQUIPED_TYPE.ASYNCDATA
      end
    end
  end
  return nEquipedType
end

function tbWnd:UpdateAttrib()
  local tbInfo = Item:GetItemBaseProp(self.nTemplateId)
  local tbClass = Item.tbClass.ZhenYuan
  local tbBaseAttrib, tbExtAttrib, tbSpecial, tbLianHuaTips
  local nFightPower = 0
  if self.pItem then
    local pPlayer = self.pAsyncData or me
    local nEquipedType = self:GetHasEquippedType()
    local nEnhanLevel = 0
    if nEquipedType == Item.Equip.EQUIPED_TYPE.ME then
      local nSlot = Item.ExtEquip:GetSlot(self.pItem)
      nFightPower = Item.ZhenYuan:GetEquipPower(self.pItem, pPlayer, nSlot)
      nEnhanLevel = Item.ZhenYuan:GetHoleEnhanLevel(me, nSlot)
    elseif nEquipedType == Item.Equip.EQUIPED_TYPE.ASYNCDATA then
      local nSlot = Item.ExtEquip:GetSlot(self.pItem)
      nEnhanLevel = self.pAsyncData.GetZhenYuanHoleLevel(nSlot - Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL + 1)
      nFightPower = Item.ZhenYuan:GetEquipPowerByEnhanLevel(self.pItem, nEnhanLevel, nSlot)
    else
      nFightPower = Item.ZhenYuan:GetEquipPower(self.pItem)
    end
    tbBaseAttrib, tbExtAttrib, tbSpecial, tbLianHuaTips = tbClass:GetTip(self.pItem, pPlayer, nEnhanLevel)
  else
    tbBaseAttrib, tbExtAttrib, tbSpecial, tbLianHuaTips = tbClass:GetTipByTemplate(self.nTemplateId, self.tbLinkInfo)
    local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[self.nTemplateId]
    if not tbItemCfg then
      nFightPower = 0
    end
    nFightPower = tbItemCfg.nFightPower
  end
  self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\230\136\152\229\138\155\229\128\188\239\188\154") .. tostring(nFightPower))
  self.pPanel:Label_SetColorByName(tbControls.txtFightPower, Item.tbTipsColorDef.EquipFightPower.ColorName or "Red")
  self:SetTips(tbBaseAttrib, tbExtAttrib, tbSpecial, tbLianHuaTips)
end

function tbWnd:SetTips(tbBaseAttrib, tbExtAttrib, tbSpecial, tbLianHuaTips)
  self.pPanel:SetActive(tbControls.DescTemplate, true)
  for i = 1, 2 do
    self.pPanel:Label_SetText(tbControls.Regular .. "/Attribute" .. i .. "/txtAttr", tbBaseAttrib[i] or "")
    if tbLianHuaTips[i] then
      self.pPanel:SetActive(tbControls.Regular .. "/Attribute" .. i .. "/Extra", true)
      self.pPanel:SetActive(tbControls.Regular .. "/Attribute" .. i .. "/Extra/txtExtra1", true)
      self.pPanel:Label_SetText(tbControls.Regular .. "/Attribute" .. i .. "/Extra/txtExtra1", tbLianHuaTips[i] or "")
    else
      self.pPanel:SetActive(tbControls.Regular .. "/Attribute" .. i .. "/Extra/txtExtra1", false)
    end
  end
  self.tbExtraAttrScrollView:Reset()
  for nIndex, szValue in ipairs(tbExtAttrib) do
    self.tbExtraAttrScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtAttr", i18n.EasyGet("{0}", szValue))
      pPanel:SetActive("Extra", false)
    end)
  end
  if #tbSpecial > 0 then
    self.pPanel:SetActive(tbControls.Special, true)
    self.pPanel:Label_SetText(tbControls.Special .. "/txtLine", tbSpecial[1])
  else
    self.pPanel:SetActive(tbControls.Special, false)
  end
end
