function tbWnd:OnCreate()
  self.tbLeftItemList = {}
  self.tbLeftItemGridView = Ui:NewItemGridView(self, tbControls.leftItemTemplate)
end

function tbWnd:OnOpen()
  self:UpdateLeftItem()
  self:UpdateRightPanel(Item.PartnerEquip.WEAPON)
end

function tbWnd:OnClose()
end

function tbWnd:UpdateLeftItem()
  self.tbLeftItemList = {}
  for i = Item.PartnerEquip.WEAPON, Item.PartnerEquip.BOOTS do
    if 0 ~= GetTimeFrameState(Item.PartnerEquip.CFG.OpenTimeFrame[i]) then
      local pItem = Item.ExtEquip:GetEquipObj(me, i)
      table.insert(self.tbLeftItemList, {
        nEquipType = i,
        nItemId = pItem and pItem.dwId or nil,
        nEquipLevel = pItem and PartnerEquipC:GetEquipLevel(pItem) or nil,
        nItemTemplateId = pItem and pItem.dwTemplateId or nil
      })
    end
  end
  for nIndex, tb in ipairs(self.tbLeftItemList) do
    self.tbLeftItemGridView:SetEmpty(nIndex)
    local szItemKey = self.tbLeftItemGridView:GetItemKey(nIndex)
    if tb.nItemTemplateId then
      self.tbLeftItemGridView:SetItem(nIndex, {
        "Item",
        tb.nItemTemplateId
      }, false)
      self.pPanel:SetActive(szItemKey .. "/Empty", false)
      if tb.nEquipLevel and tb.nEquipLevel > 0 then
        self.pPanel:SetActive(szItemKey .. "/txtLv", true)
        self.pPanel:Label_SetText(szItemKey .. "/txtLv", "+" .. tb.nEquipLevel)
      else
        self.pPanel:SetActive(szItemKey .. "/txtLv", false)
      end
      if PartnerEquipC:GetEnhanceRedPointData(tb.nEquipType) then
        RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szItemKey, {
          RedPointMgr.EVENT.PartnerEquipCanEnhance
        })
      end
      if PartnerEquipC:GetReforgeRedPointData(tb.nEquipType) then
        RedPointMgr:DynamicRegisteRedPoint(self.pPanel, szItemKey, {
          RedPointMgr.EVENT.PartnerEquipCanReforge
        })
      end
      if not PartnerEquipC:GetEnhanceRedPointData(tb.nEquipType) and not PartnerEquipC:GetReforgeRedPointData(tb.nEquipType) then
        RedPointMgr:DynamicUnRegist(self.pPanel, szItemKey)
      end
    else
      self.pPanel:SetActive(szItemKey .. "/txtLv", false)
      self.pPanel:SetActive(szItemKey .. "/Empty", true)
      RedPointMgr:DynamicUnRegist(self.pPanel, szItemKey)
    end
    self.pPanel:Label_SetText(szItemKey .. "/txtClothes", i18n.Get(Item.PartnerEquip.CFG.EquipPosDesc[tb.nEquipType]))
    self.pPanel:Button_SetEnable(szItemKey .. "/Button", true)
    self.pPanel:SetActive(szItemKey .. "/imgSelected", false)
    Ui:Button_BindEvent(self.pPanel, szItemKey .. "/Button", self.SelectLeftItem, self, tb.nEquipType)
  end
end

function tbWnd:OnEventNotify(...)
  self:UpdateLeftItem()
  self:UpdateRightPanel(...)
end

function tbWnd:UpdateRightPanel(nEquipType)
  if Lib:IsInteger(nEquipType) and nEquipType >= Item.PartnerEquip.WEAPON and nEquipType <= Item.PartnerEquip.BOOTS then
    self.nEquipType = nEquipType
  else
    self.nEquipType = self.nEquipType or Item.PartnerEquip.WEAPON
  end
  local nSuitAttrLevel = PartnerEquipC:GetPlayerSuitAttrLevel(me)
  if nSuitAttrLevel and nSuitAttrLevel > 0 then
    self.pPanel:Button_SetText(tbControls.btnSuitLook, i18n.EasyGet("\229\165\151\232\163\133\229\177\158\230\128\167{0}\231\186\167", nSuitAttrLevel), "txt")
    self.pPanel:SetActive(tbControls.btnSuitLook, true)
  else
    self.pPanel:Button_SetText(tbControls.btnSuitLook, i18n.EasyGet("\229\165\151\232\163\133\229\177\158\230\128\167\230\156\170\232\167\163\233\148\129"), "txt")
  end
  local pItem = Item.ExtEquip:GetEquipObj(me, self.nEquipType)
  if pItem then
    self:__UpdateNormal(pItem)
  else
    self:__UpdateEmpty()
  end
  self.pPanel:SetActive(tbControls.normal, nil ~= pItem)
  self.pPanel:SetActive(tbControls.empty, nil == pItem)
  local nTotalFightPower = PartnerEquipC:CalcFightPower(me)
  self.pPanel:Label_SetText(tbControls.txtTotalFightPowerValue, nTotalFightPower)
end

function tbWnd:__UpdateNormal(pItem)
  if not self.tbEquipGrid then
    self.tbEquipGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  else
    self.tbEquipGrid:Clear()
  end
  self.tbEquipGrid:SetItemByTemplate(pItem.dwTemplateId, 0)
  self.pPanel:Label_SetText(tbControls.txtEquipName, i18n.Parse(pItem.szName))
  local nEquipLevelNow = PartnerEquipC:GetEquipLevel(pItem)
  local bFullLevel = PartnerEquipC:IsEquipFullLevel(pItem)
  self.pPanel:Label_SetText(tbControls.txtLevelNow, i18n.EasyGet("{0}\231\186\167", nEquipLevelNow))
  if bFullLevel then
    self.pPanel:SetActive(tbControls.txtFullLevelTips, true)
    self.pPanel:SetActive(tbControls.txtLevelNext, false)
    self.pPanel:SetActive(tbControls.imgArrow, false)
  else
    self.pPanel:Label_SetText(tbControls.txtLevelNext, i18n.EasyGet("{0}\231\186\167", nEquipLevelNow + 1))
    self.pPanel:SetActive(tbControls.txtFullLevelTips, false)
    self.pPanel:SetActive(tbControls.imgArrow, true)
    self.pPanel:SetActive(tbControls.txtLevelNext, true)
  end
  self:__UpdateAttrBasicInfo(pItem)
  self:__UpdateAttrHideInfo(pItem)
  self:__UpdateAttrRandomInfo(pItem)
  self:__UpdateInteractButton(pItem)
end

function tbWnd:__UpdateEmpty()
  local pParentPanel = Ui:GetPanelByName("UIPartnerNew_Equip")
  if pParentPanel then
    Ui:Button_BindEvent(self.pPanel, tbControls.itemEmpty, function()
      pParentPanel:MainPanelButtonSelect(pParentPanel.tbPanelStateIndex.PANEL_PUTON, self.nSelectLeftEquipType)
    end)
    Ui:Button_BindEvent(self.pPanel, tbControls.btnEquip, function()
      pParentPanel:MainPanelButtonSelect(pParentPanel.tbPanelStateIndex.PANEL_PUTON, self.nSelectLeftEquipType)
    end)
  end
end

function tbWnd:__UpdateAttrBasicInfo(pItem)
  local tbBasicAttr = PartnerEquipC:GetAllBaseAttr(pItem) or {}
  local bFullLevel = PartnerEquipC:IsEquipFullLevel(pItem)
  for nIndex, tbAttr in ipairs(tbBasicAttr) do
    local szRoot = string.format(tbAttrBase.attrBase, nIndex)
    if bFullLevel then
      local szAttrDesc, nAttrValue, _ = PartnerEquipC:GetEquipAttrDiff(tbAttr, nil)
      self.pPanel:Label_SetText(szRoot .. tbAttrBase.txtAttrTip, szAttrDesc)
      self.pPanel:Label_SetText(szRoot .. tbAttrBase.txtAttrValueNow, "+" .. tostring(nAttrValue))
    else
      local tbNextAtrr = PartnerEquipC:GetEquipAttrNextStatusInfoByHoleIndex(pItem, nIndex + 1 - PartnerEquipC.BASE_ATTR_MIN)
      local szAttrDesc, nAttr1Value, nAttr2Value = PartnerEquipC:GetEquipAttrDiff(tbAttr, tbNextAtrr)
      self.pPanel:Label_SetText(szRoot .. tbAttrBase.txtAttrTip, szAttrDesc)
      self.pPanel:Label_SetText(szRoot .. tbAttrBase.txtAttrValueNow, "+" .. tostring(nAttr1Value))
      self.pPanel:Label_SetText(szRoot .. tbAttrBase.txtAttrValueNext, "+" .. tostring(nAttr2Value))
      self.pPanel:Label_SetText(szRoot .. tbAttrBase.txtAttrValueUp, "+" .. tostring(nAttr2Value - nAttr1Value))
    end
    self.pPanel:SetActive(szRoot .. tbAttrBase.txtAttrValueNext, not bFullLevel)
    self.pPanel:SetActive(szRoot .. tbAttrBase.txtAttrValueUp, not bFullLevel)
    self.pPanel:SetActive(szRoot .. tbAttrBase.imgArrow1, not bFullLevel)
  end
end

function tbWnd:__UpdateAttrRandomInfo(pItem)
  if not self.tbRandomAttrView then
    self.tbRandomAttrView = Ui:NewScrollView(self, tbAttrRandom.attrRandomTemplate)
  end
  self.tbRandomAttrView:Clear()
  local tbAllRandomAttr = PartnerEquipC:GetAllRandomAttr(pItem) or {}
  for i = 1, PartnerEquipC.MAX_RANDOMATTR_NUM do
    self.tbRandomAttrView:SetElement(i, function(tbElement, nIndex)
      local tbPrevAttrInfo
      if PartnerEquipC:CheckCanReforgeEquip(pItem) then
        local tbAttrCache = PartnerEquipC:GetEquipAttrInfoCache(pItem.dwId)
        if tbAttrCache and tbAttrCache.tbRandomAttr and tbAttrCache.tbRandomAttr[nIndex] then
          tbPrevAttrInfo = tbAttrCache.tbRandomAttr[nIndex]
        else
          tbPrevAttrInfo = tbAllRandomAttr[nIndex]
        end
      else
        tbPrevAttrInfo = tbAllRandomAttr[nIndex]
      end
      self:__SetRandomAttrElement(tbElement, tbAllRandomAttr[nIndex], tbPrevAttrInfo, nIndex, pItem)
    end)
  end
end

function tbWnd:__SetRandomAttrElement(tbElement, tbNowRandomAttr, tbPrevAttrInfo, nIndex, pItem)
  if not tbNowRandomAttr then
    tbElement.pPanel:SetActive(tbAttrRandom.attrNormal, false)
    tbElement.pPanel:SetActive(tbAttrRandom.attrLock, true)
    tbElement.pPanel:Label_SetText(tbAttrRandom.txtLock, i18n.EasyGet("\229\141\135\232\135\179{0}\231\186\167\229\143\175\232\167\163\233\148\129", nIndex * Item.PartnerEquip.CFG.nRandomAttrStageLevel))
  else
    tbElement.pPanel:SetActive(tbAttrRandom.attrNormal, true)
    tbElement.pPanel:SetActive(tbAttrRandom.attrLock, false)
    local szAttrDesc, nAttr1Val, nAttr2Val = PartnerEquipC:GetEquipAttrDiff(tbPrevAttrInfo, tbNowRandomAttr)
    local _, nMinVal, nMaxVal = PartnerEquipC:GetAttrRange(tbPrevAttrInfo.nAttrId)
    tbElement.pPanel:Label_SetText(tbAttrRandom.txtAttrTip, szAttrDesc)
    tbElement.pPanel:Label_SetText(tbAttrRandom.txtAttrValueNow, string.format("+%s", nAttr1Val))
    if nAttr1Val < nAttr2Val then
      tbElement.pPanel:Label_SetText(tbAttrRandom.txtAttrValueNext, "+" .. tostring(nAttr2Val))
      tbElement.pPanel:Label_SetText(tbAttrRandom.txtAttrValueUp, "+" .. tostring(nAttr2Val - nAttr1Val))
      tbElement.pPanel:SetActive(tbAttrRandom.imgArrow1, true)
      tbElement.pPanel:SetActive(tbAttrRandom.txtAttrValueNext, true)
      tbElement.pPanel:SetActive(tbAttrRandom.txtAttrValueUp, true)
    else
      tbElement.pPanel:SetActive(tbAttrRandom.imgArrow1, false)
      tbElement.pPanel:SetActive(tbAttrRandom.txtAttrValueNext, false)
      tbElement.pPanel:SetActive(tbAttrRandom.txtAttrValueUp, false)
    end
    if PartnerEquipC:IsEquipFullLevel(pItem) and pItem.nQuality == PartnerEquipC.eEquipQuality.Rare then
      local tbMaxRandomAttrInfo = PartnerEquipC:GetMaxRandomAttrInfo(tbNowRandomAttr.nAttrId)
      local _, nAttrVal, _ = PartnerEquipC:GetEquipAttrDiff(tbMaxRandomAttrInfo, nil)
      if nAttr2Val >= nAttrVal then
        tbElement.pPanel:Label_SetText(tbAttrRandom.txtLimit, i18n.EasyGet("\229\183\178\232\190\190\228\184\138\233\153\144"))
      else
        tbElement.pPanel:Label_SetText(tbAttrRandom.txtLimit, i18n.EasyGet("\239\188\136\228\184\138\233\153\144\239\188\154{0}\239\188\137", nAttrVal))
      end
    else
      tbElement.pPanel:Label_SetText(tbAttrRandom.txtLimit, string.format("\239\188\136%s - %s\239\188\137", nMinVal, nMaxVal))
    end
  end
end

function tbWnd:__UpdateAttrHideInfo(pItem)
  local bShowHideAttr = PartnerEquipC:IsEquipFullLevel(pItem) and pItem.nQuality == PartnerEquipC.eEquipQuality.Rare
  if bShowHideAttr then
    self.pPanel:SetActive(tbAttrHide.attrHide, true)
  else
    self.pPanel:SetActive(tbAttrHide.attrHide, true)
    self.pPanel:SetActive(tbAttrHide.attrLock, true)
    self.pPanel:SetActive(tbAttrHide.attrHideContent, false)
    self.pPanel:Label_SetText(tbAttrHide.attrLock .. "/txtLock", i18n.EasyGet("\231\168\128\230\156\137\232\163\133\229\164\135\229\141\135\232\135\179{0}\231\186\167\229\144\142\233\135\141\233\147\184\232\135\179\228\184\138\233\153\144\232\167\163\233\148\129", Item.PartnerEquip.CFG.nMaxEquipLevel))
    return
  end
  local nHideSkill = PartnerEquipC:GetHideSkillId(pItem)
  if not nHideSkill or nHideSkill <= 0 and not Item.PartnerEquip.SpecialSkill[nHideSkill] then
    self.pPanel:SetActive(tbAttrHide.attrLock, true)
    self.pPanel:SetActive(tbAttrHide.attrHideContent, false)
    self.pPanel:Label_SetText(tbAttrHide.attrLock .. "/txtLock", i18n.EasyGet("\231\168\128\230\156\137\232\163\133\229\164\135\229\141\135\232\135\179{0}\231\186\167\229\144\142\233\135\141\233\147\184\232\135\179\228\184\138\233\153\144\232\167\163\233\148\129", Item.PartnerEquip.CFG.nMaxEquipLevel))
  else
    self.pPanel:SetActive(tbAttrHide.attrLock, false)
    self.pPanel:SetActive(tbAttrHide.attrHideContent, true)
    local szSkillName, szSkillDesc = PartnerEquipC:GetHideSkillInfo(pItem)
    self.pPanel:Label_SetText(tbAttrHide.attrHideContent, string.format("[%s] %s", i18n.Parse(szSkillName), i18n.Parse(szSkillDesc)))
  end
end

function tbWnd:__OnBatchToggleChange(bIsOn)
  local nHasXuanShiNum = me.GetItemCountInBags(Item.PartnerEquip.CFG.nXuanShiTemplateId)
  local nHasCoinNum = me.GetMoney(Currency.TYPE.Coin)
  if self.pPanel:Toggle_GetChecked(tbUpgrade.Tog10) then
    local tbNeedCostItemBatch
    if self.nInteractItemLevel <= Item.PartnerEquip.CFG.nMaxEquipLevel - 10 then
      self.pPanel:Label_SetText(tbUpgrade.Tog10Text, i18n.EasyGet("\229\141\135\231\186\167{0}\230\172\161", 10), "Text")
      tbNeedCostItemBatch = PartnerEquipC:GetEnhanceCostItem(self.nInteractItemLevel, self.nInteractItemLevel + 10)
    else
      self.pPanel:Label_SetText(tbUpgrade.Tog10Text, i18n.EasyGet("\229\141\135\232\135\179\230\187\161\231\186\167"), "Text")
      tbNeedCostItemBatch = PartnerEquipC:GetEnhanceCostItem(self.nInteractItemLevel, Item.PartnerEquip.CFG.nMaxEquipLevel)
    end
    self.pPanel:Label_SetText(tbUpgrade.txtCost1, i18n.EasyGet("{0}/{1}", nHasXuanShiNum, tbNeedCostItemBatch.XuanShi))
    self.pPanel:Label_SetText(tbUpgrade.txtCost2, i18n.EasyGet("{0}/{1}", nHasCoinNum, tbNeedCostItemBatch.Coin))
    Ui:Button_BindEvent(self.pPanel, tbUpgrade.btnLevelUp, PartnerEquipC.ReqPartnerEquipEnhance, PartnerEquipC, self.nInteractItemId, PartnerEquipC.ProtocolSymbol.EnhanceBatch)
    self:SetToggleCheckWithOutTrigger(tbUpgrade.Tog1, false)
    self.szCheckTog = tbUpgrade.Tog10
    if nHasXuanShiNum < tbNeedCostItemBatch.XuanShi or nHasCoinNum < tbNeedCostItemBatch.Coin then
      RedPointMgr:DynamicUnRegist(self.pPanel, tbUpgrade.btnLevelUp)
      self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, false)
    else
      self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, true)
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbUpgrade.btnLevelUp, {
        RedPointMgr.EVENT.PartnerEquipCanEnhance
      })
    end
  elseif not self.pPanel:Toggle_GetChecked(tbUpgrade.Tog1) then
    RedPointMgr:DynamicUnRegist(self.pPanel, tbUpgrade.btnLevelUp)
    self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, false)
  end
end

  local nHasXuanShiNum = me.GetItemCountInBags(Item.PartnerEquip.CFG.nXuanShiTemplateId)
  local nHasCoinNum = me.GetMoney(Currency.TYPE.Coin)
  if self.pPanel:Toggle_GetChecked(tbUpgrade.Tog1) then
    local tbNeedCostItemOnce = PartnerEquipC:GetEnhanceCostItem(self.nInteractItemLevel, self.nInteractItemLevel + 1)
    self.pPanel:Label_SetText(tbUpgrade.txtCost1, i18n.EasyGet("{0}/{1}", nHasXuanShiNum, tbNeedCostItemOnce.XuanShi))
    self.pPanel:Label_SetText(tbUpgrade.txtCost2, i18n.EasyGet("{0}/{1}", nHasCoinNum, tbNeedCostItemOnce.Coin))
    self.pPanel:Label_SetText(tbUpgrade.Tog1Text, i18n.EasyGet("\229\141\135\231\186\167{0}\230\172\161", 1), "Text")
    self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, true)
    Ui:Button_BindEvent(self.pPanel, tbUpgrade.btnLevelUp, PartnerEquipC.ReqPartnerEquipEnhance, PartnerEquipC, self.nInteractItemId, PartnerEquipC.ProtocolSymbol.EnhanceOnce)
    self:SetToggleCheckWithOutTrigger(tbUpgrade.Tog10, false)
    self.szCheckTog = tbUpgrade.Tog1
    if nHasXuanShiNum < tbNeedCostItemOnce.XuanShi or nHasCoinNum < tbNeedCostItemOnce.Coin then
      RedPointMgr:DynamicUnRegist(self.pPanel, tbUpgrade.btnLevelUp)
      self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, false)
    else
      self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, true)
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbUpgrade.btnLevelUp, {
        RedPointMgr.EVENT.PartnerEquipCanEnhance
      })
    end
  elseif not self.pPanel:Toggle_GetChecked(tbUpgrade.Tog10) then
    RedPointMgr:DynamicUnRegist(self.pPanel, tbUpgrade.btnLevelUp)
    self.pPanel:Button_SetEnable(tbUpgrade.btnLevelUp, false)
  end
end

function tbWnd:SetToggleCheckWithOutTrigger(szTogglePath, bCheck)
  Ui:Toggle_UnBindEvent(self.pPanel, szTogglePath)
  self.pPanel:Toggle_SetChecked(szTogglePath, bCheck)
  if szTogglePath == tbUpgrade.Tog1 then
    Ui:Toggle_BindEvent(self.pPanel, szTogglePath, self.__OnOnceToggleChange, self)
  end
  if szTogglePath == tbUpgrade.Tog10 then
    Ui:Toggle_BindEvent(self.pPanel, szTogglePath, self.__OnBatchToggleChange, self)
  end
end

function tbWnd:__UpdateInteractButton(pItem)
  self.nInteractItemId = pItem.dwId
  self.nInteractItemLevel = PartnerEquipC:GetEquipLevel(pItem)
  if not PartnerEquipC:IsEquipFullLevel(pItem) then
    self.pPanel:SetActive(tbUpgrade.upgradeBtnCell, true)
    self.pPanel:SetActive(tbRebuild.rebuildBtnCell, false)
    if not self.tbUpgradeGrid1 then
      self.tbUpgradeGrid1 = Ui:NewExistItemGrid(self, tbUpgrade.UIItemGrid1)
      self.tbUpgradeGrid1:SetItemByTemplate(Item.PartnerEquip.CFG.nXuanShiTemplateId, nil)
    end
    if not self.tbUpgradeGrid2 then
      self.tbUpgradeGrid2 = Ui:NewExistItemGrid(self, tbUpgrade.UIItemGrid2)
      self.tbUpgradeGrid2:SetDigitalItem(Currency.TYPE.Coin)
    end
    Ui:Toggle_BindEvent(self.pPanel, tbUpgrade.Tog1, self.__OnOnceToggleChange, self)
    Ui:Toggle_BindEvent(self.pPanel, tbUpgrade.Tog10, self.__OnBatchToggleChange, self)
    Ui:Button_BindEvent(self.pPanel, tbUpgrade.btnAdd1, self.__JumpToXuanShiShop, self)
    Ui:Button_BindEvent(self.pPanel, tbUpgrade.btnAdd2, self.__JumpToCoinShop, self)
    self.pPanel:Toggle_SetChecked(self.szCheckTog or tbUpgrade.Tog1, false)
    self.pPanel:Toggle_SetChecked(self.szCheckTog or tbUpgrade.Tog10, false)
    self.pPanel:Toggle_SetChecked(self.szCheckTog or tbUpgrade.Tog1, true)
  elseif PartnerEquipC:CheckCanReforgeEquip(pItem) then
    local nHasXuanShiNum = me.GetItemCountInBags(Item.PartnerEquip.CFG.nXuanShiTemplateId)
    local nHasCoinNum = me.GetMoney(Currency.TYPE.Coin)
    self.pPanel:SetActive(tbUpgrade.upgradeBtnCell, false)
    self.pPanel:SetActive(tbRebuild.rebuildBtnCell, true)
    if not self.tbRebuildGrid1 then
      self.tbRebuildGrid1 = Ui:NewExistItemGrid(self, tbRebuild.UIItemGrid1)
      self.tbRebuildGrid1:SetItemByTemplate(Item.PartnerEquip.CFG.nXuanShiTemplateId, nil)
    end
    if not self.tbRebuildGrid2 then
      self.tbRebuildGrid2 = Ui:NewExistItemGrid(self, tbRebuild.UIItemGrid2)
      self.tbRebuildGrid2:SetDigitalItem(Currency.TYPE.Coin)
    end
    local tbNeedCostItemNum = PartnerEquipC:GetReforgeCostItem()
    self.pPanel:Label_SetText(tbRebuild.txtCost1, i18n.EasyGet("{0}/{1}", nHasXuanShiNum, tbNeedCostItemNum.XuanShi))
    self.pPanel:Label_SetText(tbRebuild.txtCost2, i18n.EasyGet("{0}/{1}", nHasCoinNum, tbNeedCostItemNum.Coin))
    self.pPanel:Button_SetText(tbRebuild.btnRebuild, i18n.EasyGet("\233\135\141\233\147\1841\230\172\161"), "Text")
    Ui:Button_BindEvent(self.pPanel, tbRebuild.btnRebuild, PartnerEquipC.ReqPartnerEquipReforge, PartnerEquipC, pItem.dwId)
    Ui:Button_BindEvent(self.pPanel, tbRebuild.btnAdd1, self.__JumpToXuanShiShop, self)
    Ui:Button_BindEvent(self.pPanel, tbRebuild.btnAdd2, self.__JumpToCoinShop, self)
  else
    self.pPanel:SetActive(tbUpgrade.upgradeBtnCell, false)
    self.pPanel:SetActive(tbRebuild.rebuildBtnCell, false)
  end
  if PartnerEquipC:GetEnhanceRedPointData(self.nEquipType) == self.nInteractItemId then
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbUpgrade.btnLevelUp, {
      RedPointMgr.EVENT.PartnerEquipCanEnhance
    })
  else
    RedPointMgr:DynamicUnRegist(self.pPanel, tbUpgrade.btnLevelUp)
  end
  if PartnerEquipC:GetReforgeRedPointData(self.nEquipType) == self.nInteractItemId then
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbRebuild.btnRebuild, {
      RedPointMgr.EVENT.PartnerEquipCanReforge
    })
  else
    RedPointMgr:DynamicUnRegist(self.pPanel, tbRebuild.btnRebuild)
  end
end

function tbWnd:SelectLeftItem(nEquipType)
  for nIndex, tb in ipairs(self.tbLeftItemList) do
    if tb.nEquipType == nEquipType then
      local szItemKey = self.tbLeftItemGridView:GetItemKey(nIndex)
      if self.nSelectLeftIndex then
        local szOldItemKey = self.tbLeftItemGridView:GetItemKey(self.nSelectLeftIndex)
        self.pPanel:SetActive(szOldItemKey .. "/imgSelected", false)
      end
      self.pPanel:SetActive(szItemKey .. "/imgSelected", true)
      self.nSelectLeftIndex = nIndex
      self.nSelectLeftEquipType = nEquipType
      self:__OnClickLeftItem(tb)
      break
    end
  end
end

function tbWnd:__OnClickLeftItem(tbLeftItemInfo)
  if not tbLeftItemInfo or not tbLeftItemInfo.nEquipType then
    return
  end
  if not tbLeftItemInfo.nItemId then
    local pParentPanel = Ui:GetPanelByName("UIPartnerNew_Equip")
    if pParentPanel then
      pParentPanel:MainPanelButtonSelect(pParentPanel.tbPanelStateIndex.PANEL_PUTON, self.nSelectLeftEquipType)
    end
  else
    self:UpdateRightPanel(tbLeftItemInfo.nEquipType)
  end
end

function tbWnd:__JumpToXuanShiShop()
  local tbWare = Shop:GetAWare2(Shop.ShopTypeStr.YuanBao, Item.PartnerEquip.CFG.nXuanShiTemplateId)
  if tbWare then
    Ui:OpenWindow("UIShopBuy", tbWare, true)
  end
end

function tbWnd:__JumpToCoinShop()
  Ui:OpenWindow("UIWelfare", "PANEL_TYPE_MONEY_TREE")
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end
