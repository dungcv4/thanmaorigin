function tbWnd:OnCreate()
  self:SetClickEmptyToClose(tbControls.wndCuiling)
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbwndLianHuaInfo = Ui:CreatePanel("UIWndLianHuaInfo", tbControls.wndLianHuaInfo, self)
  self.tbWndJumpTo = Ui:CreatePanel("UIEquipmentJumpTo", tbControls.wndJumpTo, self)
  self.tbLingZhuScrollView = Ui:NewSuperListView(self, tbControls.ZTemplate, self.SetLingZhuElement)
end

function tbWnd:IsPutOnEquip(bNeedTip)
  local tbEquip = me.GetEquips()
  for k, v in pairs(tbEquip) do
    if k <= Item.EQUIPPOS_AMULET then
      return true
    end
  end
  if bNeedTip then
    me.CenterMsg(i18n.EasyGet("\229\176\154\230\156\170\231\169\191\230\136\180\232\163\133\229\164\135\239\188\129"))
  end
  return false
end

function tbWnd:OnOpen(nSelectPanel, nEquipPos, nParam1)
  if not self.tbTabControl:OnOpen(nSelectPanel, nEquipPos, nParam1) then
    return
  end
  if not self:IsPutOnEquip(true) then
    self:Close()
    return
  end
  self.pPanel:SetActive(tbControls.wndLianHuaInfo, false)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  self.pPanel:SetActive(tbControls.wndCuiling, false)
  self:RegistRedPoint()
  if Item.EquipWuLian:CheckIsOpen() then
    self.pPanel:Label_SetText(tbControls.txtBtnEnhance, i18n.EasyGet("\229\188\186\229\140\150/\230\173\166\231\130\188"))
  else
    self.pPanel:Label_SetText(tbControls.txtBtnEnhance, i18n.EasyGet("\229\188\186\229\140\150"))
  end
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  self:UnRegistRedPoint()
end

function tbWnd:RegistRedPoint()
  local tbEnhanceEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmEnhance, RedPointMgr.EVENT.AmuletEnhance do
    table.insert(tbEnhanceEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnEnhance, tbEnhanceEventList)
  local tbInsetEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmStoneInsert, RedPointMgr.EVENT.AmuletStoneInsert do
    table.insert(tbInsetEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnInset, tbInsetEventList)
  local tbJingZhuEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmJingZhu, RedPointMgr.EVENT.AmuletJingZhu do
    table.insert(tbJingZhuEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnJingZhu, tbJingZhuEventList)
  local tbLianHuaEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmLianHua, RedPointMgr.EVENT.AmuletLianHua do
    table.insert(tbLianHuaEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLianHua, tbLianHuaEventList)
  local tbQiHunEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmQiHun, RedPointMgr.EVENT.AmuletQiHun do
    table.insert(tbQiHunEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnQiHun, tbQiHunEventList)
  local tbLingZhuEventList = {}
  for nRedPoint = RedPointMgr.EVENT.HelmCuiLing, RedPointMgr.EVENT.AmuletCuiLing do
    table.insert(tbLingZhuEventList, nRedPoint)
  end
  for nRedPoint = RedPointMgr.EVENT.HelmELZWenYang, RedPointMgr.EVENT.AmuletELZWenYang do
    table.insert(tbLingZhuEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnCuiLing, tbLingZhuEventList)
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnEnhance)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnJingZhu)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLianHua)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnQiHun)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnCuiLing)
end

function tbWnd:OnSyncItemEvent(nItemId, bUpdateAll, nAddCount)
  if not self:Visible() then
    return
  end
  self:NotifyEquipEnhance("Item", nItemId, bUpdateAll, nAddCount)
  local pEquip = KItem.GetItemObj(nItemId)
  if nil ~= pEquip and Item:IsCommonTenPos(pEquip.nEquipPos) then
    self:NotifyEquipJingZhu(nil, pEquip.nEquipPos)
  end
  self:NotifyEquipLianHua()
  self:NotifyEquipStone()
  self:QiHunUpdateView()
end

function tbWnd:OnDeleteItemEvent()
  self:NotifyEquipStone()
end

function tbWnd:ShowCuiLingInfo(nEquipPos, bActive)
  if bActive then
    self.pPanel:SetActive(tbControls.wndCuiling, bActive)
    local tbItem = EquipCuiLing:GetLingZhuList()
    table.sort(tbItem, function(a, b)
      return a.nQuality > b.nQuality
    end)
    self.tbLingZhuScrollView:SetAllElement(tbItem)
    self.tbLingZhuScrollView:MovePanelToFirstItem()
    self.nCuiLingEquipPos = nEquipPos
  else
    self.pPanel:SetActive(tbControls.wndCuiling, bActive)
  end
end

function tbWnd:SetLingZhuElement(tbElement, pItem, nIndex)
  local pPanel = tbElement.pPanel
  local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  itemGrid:SetItem(pItem.dwId, pItem.dwTemplateId)
  itemGrid:SetClickEnable(true, function()
  end)
  Ui:Button_BindEvent(pPanel, "btnAdd", function()
    EquipCuiLing:ECLPutOnC(self.nCuiLingEquipPos, pItem.dwId)
  end)
  local tbECLLingZhuItem = Item:GetClass("ECLLingZhuItem")
  local tbAttr = tbECLLingZhuItem:GetRandomAttrInfo(pItem)
  for i = 1, 3 do
    if tbAttr[i] then
      local tbAttrCfg = EquipCuiLing.tbLZAttrId2AttrType[tbAttr[i][1]]
      local szAttrType = tbAttrCfg.szAttrType
      local szDesc, szValueCur = FightSkill:GetMagicDescSplit(szAttrType, tbAttr[i][2])
      pPanel:Label_SetText(string.format("AttrWid/txtInfo%d", i), szDesc .. szValueCur)
      pPanel:SetActive(string.format("AttrWid/txtInfo%d", i), true)
    else
      pPanel:Label_SetText(string.format("AttrWid/txtInfo%d", i), "")
      pPanel:SetActive(string.format("AttrWid/txtInfo%d", i), false)
    end
  end
end
