FirstRecharge = FirstRecharge or {}
FirstRecharge.bHadOpenUIFirstRecharge = false

function FirstRecharge:ResetFirstRechargeHadOpenState()
  FirstRecharge.bHadOpenUIFirstRecharge = false
end

function FirstRecharge:OnBattleAloneEndUp()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_MAP_UI_LOADED, self.OnMapUILoaded, self)
end

function FirstRecharge:OnMapUILoaded()
  local nHasGet = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_FIRST_RECHARGE)
  local nCanGetAward = self:CheckHasCanGetAward()
  if nCanGetAward or 0 == nHasGet then
    Ui:OpenWindow("UIFirstRecharge")
  end
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_MAP_UI_LOADED, self)
end

function FirstRecharge:CheckCanGetByDay(nDay)
  if nDay > 3 then
    return false
  end
  local nGetFirst = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_FIRST_RECHARGE)
  if 1 ~= nGetFirst then
    return false
  end
  if self:CheckGetAlreadyByDay(nDay) then
    return false
  end
  local nTime = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_FIRST_RECHARGE_TIME)
  local nDayOffset = Lib:CalcDayOffset(14400, nTime)
  if nDay > nDayOffset + 1 then
    return false
  end
  return true
end

function FirstRecharge:GetAllAwards()
  for nDay = 1, 3 do
    if self:CheckCanGetByDay(nDay) then
      RemoteServer.RequestFirstChargeAward(nDay)
    end
  end
end

function FirstRecharge:CheckHasCanGetAward()
  local bHasAward = false
  for nDay = 1, 3 do
    if self:CheckCanGetByDay(nDay) then
      bHasAward = true
      break
    end
  end
  return bHasAward
end

function FirstRecharge:CheckGetAlreadyByDay(nDay)
  local nAward = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_AWARD)
  if not nAward then
    return false
  end
  local bHasAward = 0 ~= Recharge.tbFirstRechargeDayBit[nDay] & nAward
  if not bHasAward then
    return false
  end
  return true
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, FirstRecharge.ResetFirstRechargeHadOpenState, FirstRecharge)
