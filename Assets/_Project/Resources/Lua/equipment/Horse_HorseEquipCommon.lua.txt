function HorseEquipMgr:CalcFightPower(pPlayer, szType)
  local nRet = 0
  
  for i = Item.ExtEquip.CFG.SLOT.HORSEREIN, Item.ExtEquip.CFG.SLOT.HORSESHOE do
    local pHorseEquip = Item.ExtEquip:GetEquipObj(pPlayer, i)
    if pHorseEquip then
      nRet = nRet + pHorseEquip.nFightPower
    end
  end
  return nRet
end

FightPower:RegisterFPSystemModule({"HorseEquip"}, HorseEquipMgr)
