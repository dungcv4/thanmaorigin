local tbSeriesNumberToTextName = {
  [1] = "txtMetal",
  [2] = "txtWood",
  [3] = "txtWater",
  [4] = "txtFire",
  [5] = "txtEarth"

tbWnd.tbOnClick = {
  btnJingZhuInfo = function(self)
    if self.pAsyncData then
      if Ui:WindowVisible("UIViewRole") then
        Ui:GetWindow("UIViewRole"):SetGradeScoreStatus("JingZhu", self.pAsyncData.GetJingZhuGradeScore(), self.pAsyncData)
      end
    elseif Ui:WindowVisible("UIBag") then
      Ui:GetWindow("UIBag"):SetGradeScoreStatus("JingZhu", Item.EquipJingZhu:GetEquipJingZhuGradeScore(me))
    end
  end,
  btnInsetInfo = function(self)
    if self.pAsyncData then
      if Ui:WindowVisible("UIViewRole") then
        Ui:GetWindow("UIViewRole"):SetGradeScoreStatus("Stone", self.pAsyncData.GetStoneGradeScore(), self.pAsyncData)
      end
    elseif Ui:WindowVisible("UIBag") then
      Ui:GetWindow("UIBag"):SetGradeScoreStatus("Stone", Item.Stone:GetStoneGradeScoreLevel(me))
    end
  end,
  btnQiHunInfo = function(self)
    if self.pAsyncData then
      local nSumLevel = 0
      for i = Item.EQUIPPOS_HELM, Item.EQUIPPOS_AMULET do
        local nLevel = self.pAsyncData.GetQiHun(i + 1)
        if nLevel then
          nSumLevel = nSumLevel + nLevel
        end
      end
      local gradeRank = 0
      for k, v in pairs(Item.QiHun.tbGradeScoreInfo) do
        local levelBound = v[0][1]
        if nSumLevel < levelBound then
          break
        end
        gradeRank = gradeRank + 1
      end
      if Ui:WindowVisible("UIViewRole") then
        Ui:GetWindow("UIViewRole"):OpenQiHunInfo(gradeRank)
      end
    else
      local nGradeScoreLevel = Item.QiHun:GetQiHunGradeScoreLevel(me)
      if Ui:WindowVisible("UIBag") then
        Ui:GetWindow("UIBag"):OpenQiHunInfo(nGradeScoreLevel)
      end
    end
  end,
  btnSeriesText = function(self)
    if self.pAsyncData then
      if Ui:WindowVisible("UIViewRole") then
        Ui:GetWindow("UIViewRole"):OpenSeriesActive(self.pAsyncData)
      end
    elseif Ui:WindowVisible("UIBag") then
      Ui:GetWindow("UIBag"):OpenSeriesActive()
    end
  end
}

function tbWnd:OnCreate(nSex, szItemOpt)
  self.tbEnhanceEffect = {}
  self.tbGridList = {}
  self.tbGridTextList = {}
  self.tbEnhanceLvTextList = {}
  for i = 1, Item.EQUIPPOS_WIND + 1 do
    if "" ~= tbEquipPosKey[i] then
      local tbGrid = Ui:NewExistItemGrid(self, tbEquipPosKey[i] .. "/UIItemGrid")
      tbGrid.szItemOpt = szItemOpt
      self.tbGridList[i] = tbGrid
      self.tbGridTextList[i] = tbEquipPosKey[i] .. "/Text"
      if i < Item.EQUIPPOS_MAIN_NUM + 1 then
        self.tbEnhanceLvTextList[i] = tbEquipPosKey[i] .. "/txtJingZhuLv"
      end
      tbGrid:SetClickEnable(true)
      tbGrid:SetActive(false)
      self.pPanel:SetActive(self.tbGridTextList[i], true)
    end
  end
  self.UIBagPanel = Ui:GetWindow("UIBag")
  self:RegistRedPoint()
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(self.btnJingZhuInfo, Item.EquipJingZhu:CheckIsOpen())
  self:UpdateJZGradeScore(Item.EquipJingZhu:GetEquipJingZhuGradeScore(me))
  self:UpdateStoneGradeScore(Item.Stone:GetStoneGradeScoreLevel(me))
  self:SetFightPower(Stronger.nFightPower or 0)
  local szPlayerSeries = KPlayer.GetPlayerInitInfo(me.nFaction).nSeries
  for index, sztxt in ipairs(tbSeriesNumberToTextName) do
    self.pPanel:SetActive(self.btnSeriesText .. "/" .. sztxt, false)
  end
  self.pPanel:SetActive(self.btnSeriesText .. "/" .. tbSeriesNumberToTextName[szPlayerSeries], true)
  local nQiHunTotalLevel = Item.QiHun:GetQiHunCountLevel(me)
  self.pPanel:Label_SetText(self.goQiHunInfo, nQiHunTotalLevel)
end

function tbWnd:OnClose()
end

function tbWnd:UpdateJZGradeScore(nLevel)
  self.pPanel:Label_SetText(self.goJZGradeScoreLevel, nLevel)
end

function tbWnd:UpdateStoneGradeScore(nLevel)
  self.pPanel:Label_SetText(self.goStoneGradeScoreLevel, nLevel)
end

function tbWnd:RegistRedPoint()
  for nKey, tbEventList in pairs(tbRedPointEvent) do
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbEquipPosKey[nKey], tbEventList)
  end
end

function tbWnd:UnRegistRedPoint()
  for nKey, tbEventList in pairs(tbRedPointEvent) do
    RedPointMgr:DynamicUnRegist(self.pPanel, tbEquipPosKey[nKey])
  end
end

function tbWnd:Reset()
  for i = 1, Item.EQUIPPOS_MAIN_NUM do
    self.tbGridList[i]:SetActive(false)
    self.pPanel:SetActive(self.tbGridTextList[i], true)
    if i < Item.EQUIPPOS_MAIN_NUM + 1 then
      self.pPanel:SetActive(self.tbEnhanceLvTextList[i], false)
    end
  end
end

function tbWnd:ResetFrom(nStartIndex)
  for i = nStartIndex, Item.EQUIPPOS_MAIN_NUM do
    self.tbGridList[i]:SetActive(false)
    if i < Item.EQUIPPOS_MAIN_NUM + 1 then
      self.pPanel:SetActive(self.tbEnhanceLvTextList[i], false)
    end
    self.pPanel:SetActive(self.tbGridTextList[i], true)
  end
end

function tbWnd:OpenBagReset()
  self.pPanel:SetActive("EquipPanel/PanelEquip", true)
end

function tbWnd:SetEquip(nPos, nItemId, pAsyncData)
  self.pAsyncData = pAsyncData
  local tbGrid = self.tbGridList[nPos]
  if not tbGrid then
    return
  end
  if self.tbEnhanceEffect[nPos] then
    self.pPanel:SetActive(self.tbEnhanceEffect[nPos].szKey, false)
  end
  if nItemId and nItemId > 0 then
    local pItem = KItem.GetItemObj(nItemId)
    local nFaction, nFactionSect, nSex
    local bShowTip = true
    if pAsyncData then
      nFaction = pAsyncData.GetFaction()
      nFactionSect = pAsyncData.GetFactionSect()
      nSex = pAsyncData.GetSex()
      bShowTip = false
    end
    tbGrid:SetItem(nItemId, pItem.dwTemplateId, {bShowTip = bShowTip, bBigIcon = true}, nFaction, nFactionSect, nSex, pAsyncData)
    if nPos < Item.EQUIPPOS_MAIN_NUM + 1 then
      local nEnhanceLv
      if pAsyncData then
        nEnhanceLv = pAsyncData.GetEnhance(nPos)
      else
        nEnhanceLv = Item.EquipEnhance:GetEquipEnhanceLevel(me, pItem.nEquipPos)
      end
      self.pPanel:SetActive(self.tbEnhanceLvTextList[nPos], true)
      if nEnhanceLv > 0 then
        self.pPanel:Label_SetText(self.tbEnhanceLvTextList[nPos], "+" .. tostring(nEnhanceLv))
      else
        self.pPanel:Label_SetText(self.tbEnhanceLvTextList[nPos], "")
      end
      local szPath = Item.EquipEnhance:GetGridEffectPath(nEnhanceLv)
      if szPath then
        if not self.tbEnhanceEffect[nPos] then
          self.tbEnhanceEffect[nPos] = {}
          self.tbEnhanceEffect[nPos].szPath = szPath
          self.tbEnhanceEffect[nPos].szKey = self.pPanel:DynamicLoadGameObject(tbEquipPosKey[nPos] .. "/UIItemGrid/Enhance", szPath, "effect")
          self.pPanel:SetSortingOrder(self.tbEnhanceEffect[nPos].szKey, 2)
        elseif szPath ~= self.tbEnhanceEffect[nPos].szPath then
          self.pPanel:DeleteObject(self.tbEnhanceEffect[nPos].szKey)
          self.tbEnhanceEffect[nPos].szKey = self.pPanel:DynamicLoadGameObject(tbEquipPosKey[nPos] .. "/UIItemGrid/Enhance", szPath, "effect")
          self.pPanel:SetSortingOrder(self.tbEnhanceEffect[nPos].szKey, 2)
        end
        self.pPanel:SetActive(self.tbEnhanceEffect[nPos].szKey, true)
      end
    end
    tbGrid:SetActive(true)
    self.pPanel:SetActive(self.tbGridTextList[nPos], false)
  else
    tbGrid:SetActive(false)
    if nPos < Item.EQUIPPOS_MAIN_NUM + 1 then
      self.pPanel:SetActive(self.tbEnhanceLvTextList[nPos], false)
    end
    self.pPanel:SetActive(self.tbGridTextList[nPos], true)
  end
end

function tbWnd:SetFightPower(nValue)
  self.pPanel:Label_SetText(self.txtFightPower, nValue)
end

function tbWnd:OnLevelChange()
  self.pPanel:SetActive(self.btnJingZhuInfo, Item.EquipJingZhu:CheckIsOpen())
end
