function EquipCuiLing:ECLLingZhuComposeRsp(tbItemId)

    local pItem = KItem.GetItemObj(dwItemId)
    if pItem then
      local tbAward = {
        Player.AwardType2Name[Player.award_type_item],
        pItem.dwTemplateId,

    end

  EventNotify.OnNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_COMPOSE_SUCCESS)

function EquipCuiLing:ECLPutOnRsp(nEquipPos, bActiveSuit)
  local tbLingZhuData = self:GetPutOnLingZhuData(me, nEquipPos)
  self:ApplyLingZhuAttrib(me.GetNpc(), tbLingZhuData)
  EventNotify.OnNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_PUTON_SUCCESS)
  if bActiveSuit then
    Ui:OpenUpgradeEffect(i18n.EasyGet("\229\183\178\230\191\128\230\180\187\233\149\135\229\178\179\229\165\151\232\163\133"), {x = 100})
    self:ApplySuitActiveAttr(me.GetNpc())
    local nAllWYLevel = self:GetWenYangAllLevel(me)
    self:ApplyWenYangAttr(me.GetNpc(), nAllWYLevel, me.dwID)
  end
  if self:CheckCuiLingRedPointByEquipPos(me, nEquipPos) then
    RedPointMgr:LightRedPoint(self.tbEquipPos2RedPoint[nEquipPos])
  else
    RedPointMgr:FadeRedPoint(self.tbEquipPos2RedPoint[nEquipPos])
  end
end

end

function EquipCuiLing:GetLingZhuAddAttrDesc(tbData)
  local tbRet = {}
  table.insert(tbRet, {
    i18n.EasyGet("\229\159\186\231\161\128\229\177\158\230\128\167")
  })
  for i = 1, #tbData do
    local nAttrId = tbData[i][1]
    local tbMagic = tbData[i][2]
    local tbAttrCfg = self.tbLZAttrId2AttrType[nAttrId]
    local szMagic = tbAttrCfg.szAttrType
    local szDesc = FightSkill:GetMagicDesc(szMagic, tbMagic)
    table.insert(tbRet, {szDesc, "NewGreen"})
  end
  return tbRet
end

function EquipCuiLing:ResetAttr()
  self:OnUnEquip(me)
end

function EquipCuiLing:CheckCuiLingRedPointByEquipPos(pPlayer, nEquipPos)
  if self:CanCuiLing(pPlayer, nEquipPos) then
    return true
  end
  local tbItemList = pPlayer.GetItemListInBag("ECLLingZhuItem")
  if not tbItemList then
    return false
  end
  local tbLingZhuData = self:GetPutOnLingZhuData(pPlayer, nEquipPos)
  local nQuality = 0
  if tbLingZhuData.nTemplateId > 0 then
    nQuality = Item:GetQuality(tbLingZhuData.nTemplateId)
  end
  for _, pItem in ipairs(tbItemList) do
    if self:CanPutOnLingZhu(pPlayer, nEquipPos, pItem.dwId) and nQuality < pItem.nQuality then
      return true
    end
  end
  return false
end

function EquipCuiLing:OnCheckWenYangRedPoint(pPlayer, nEquipPos)
  if self:CanLZWenYang(pPlayer, nEquipPos) then
    return true
  end
  return false
end

function EquipCuiLing:OnCheckComposeRedPoint()
  if self:CanLingZhuCompose(me, 1) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.EquipLingZhuCompose)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.EquipLingZhuCompose)
  end
end

function EquipCuiLing:OnCheckRedPoint()
  for nEquipPos, nRedPoint in pairs(self.tbEquipPos2RedPoint) do
    if self:CheckCuiLingRedPointByEquipPos(me, nEquipPos) then
      RedPointMgr:LightRedPoint(nRedPoint)
    else
      RedPointMgr:FadeRedPoint(nRedPoint)
    end
  end
  for nEquipPos, nRedPoint in pairs(self.tbEquipPos2WYRedPoint) do
    if self:OnCheckWenYangRedPoint(me, nEquipPos) then
      RedPointMgr:LightRedPoint(nRedPoint)
    else
      RedPointMgr:FadeRedPoint(nRedPoint)
    end
  end
  self:OnCheckComposeRedPoint()
end

function EquipCuiLing:OnSyncItemLimitTime()
  if not PlayerEvent.bLogin then
    return
  end
  self:OnCheckRedPoint()
end
