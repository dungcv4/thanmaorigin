local tbWnd = Ui:CreateClass("UIPartnerNew_Equip", "UIPanel")
local tbControls = {
  infoPanel = "infoPanel",
  equipPanel = "equipPanel",
  dissolvePanel = "dissolvePanel",
  ToggleInfo = "equipToggleList/Togglnfo",
  TogglePutOn = "equipToggleList/ToggleGallery",
  ToggleDissolve = "equipToggleList/ToggleDecompose"
}
tbWnd.tbControls = tbControls
tbWnd.tbPanelStateIndex = {
  PANEL_INFO = 1,
  PANEL_PUTON = 2,
  PANEL_DISSOLVE = 3
}
tbWnd.tbTogglePath = {
  [tbWnd.tbPanelStateIndex.PANEL_INFO] = "equipToggleList/Togglnfo",
  [tbWnd.tbPanelStateIndex.PANEL_PUTON] = "equipToggleList/ToggleGallery",
  [tbWnd.tbPanelStateIndex.PANEL_DISSOLVE] = "equipToggleList/ToggleDecompose"
}
tbWnd.tbToggleCallBack = {
  [tbWnd.tbPanelStateIndex.PANEL_INFO] = function(tbSelf, isOn)
    if isOn then
      tbSelf:MainPanelButtonSelect(tbWnd.tbPanelStateIndex.PANEL_INFO)
    end
  end,
  [tbWnd.tbPanelStateIndex.PANEL_PUTON] = function(tbSelf, isOn)
    if isOn then
      tbSelf:MainPanelButtonSelect(tbWnd.tbPanelStateIndex.PANEL_PUTON)
    end
  end,
  [tbWnd.tbPanelStateIndex.PANEL_DISSOLVE] = function(tbSelf, isOn)
    if isOn then
      tbSelf:MainPanelButtonSelect(tbWnd.tbPanelStateIndex.PANEL_DISSOLVE)
    end
  end
}
tbWnd.tbPanelListRoot = {
  {
    name = tbControls.infoPanel,
    panel = "UIPartnerNew_EquipInfo"
  },
  {
    name = tbControls.equipPanel,
    panel = "UIPartnerNew_EquipPutOn"
  },
  {
    name = tbControls.dissolvePanel,
    panel = "UIPartnerNew_EquipDissolve"
  }
}

function tbWnd:OnCreate()
  self.tbPanelList = {}
  for i = 1, #self.tbPanelListRoot do
    self.tbPanelList[i] = Ui:CreatePanel(self.tbPanelListRoot[i].panel, self.tbPanelListRoot[i].name, self)
  end
end

function tbWnd:OnOpen()
  for nIndex, _ in ipairs(self.tbTogglePath) do
    Ui:Toggle_BindEvent(self.pPanel, self.tbTogglePath[nIndex], self.tbToggleCallBack[nIndex], self)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.tbTogglePath[self.tbPanelStateIndex.PANEL_PUTON], {
    RedPointMgr.EVENT.PartnerEquipBetter
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.tbTogglePath[self.tbPanelStateIndex.PANEL_INFO], {
    RedPointMgr.EVENT.PartnerEquipCanEnhance,
    RedPointMgr.EVENT.PartnerEquipCanReforge
  })
  self:MainPanelButtonSelect(self.tbPanelStateIndex.PANEL_INFO)
end

function tbWnd:MainPanelButtonSelect(nPanelType, ...)
  for nIndex, szPath in ipairs(self.tbTogglePath) do
    self.tbPanelList[nIndex]:SetActive(nPanelType == nIndex, ...)
    self.pPanel:SetActive(szPath .. "/Selected", nPanelType == nIndex)
    self.pPanel:Toggle_SetEnable(szPath, true)
    self.pPanel:ToggleGroup_SetSelect(szPath, nPanelType == nIndex)
  end
end

function tbWnd:OnEventNotify(...)
  for i = 1, #self.tbPanelList do
    if self.pPanel:IsActive(self.tbPanelListRoot[i].name) then
      self.tbPanelList[i]:OnEventNotify(...)
    end
  end
end

function tbWnd:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, self.tbTogglePath[self.tbPanelStateIndex.PANEL_PUTON])
  RedPointMgr:DynamicUnRegist(self.pPanel, self.tbTogglePath[self.tbPanelStateIndex.PANEL_INFO])
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end
