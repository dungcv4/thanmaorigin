function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ZHENYUAN_ITEM_CACHE,
      self.UpdateBagInfo
    },
    {
      EventNotify.emNOTIFY_USE_UNUSE_EQUIP,
      self.UpdateEquipInfo
    }
  }
end

function tbWnd:OnCreate()
  self.nSelectSeries = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL
  self.nSelectIndex = 1
  self.tbItemGrid = {}
  self.tbItemRed = {}
  self:RegistRedPoint()
  self.tbLeftItemList = Ui:NewScrollView(self, tbControls.LeftItemTemplate, function(self, nIndex, tbElement)
    self:OnUpdateSelectAttr(nIndex)
    if self.tbItemRed[self.nSelectSeries] and self.tbItemRed[self.nSelectSeries] == nIndex then
      self.tbItemRed[self.nSelectSeries] = nil
      self.nRedPanel = nil
      RedPointMgr:FadeRedPoint(Item.ZhenYuan.tbEquipPosRedPoint[self.nSelectSeries].nEquipRed)
    end
  end)
  self.tbSelectExtScrollView = Ui:NewScrollView(self, tbControls.SelectExtAttrTemplate, nil)
  self.tbSelectBaseScrollView = Ui:NewScrollView(self, tbControls.SelectBaseTemplate, nil)
  self.tbSelectSkillScrollView = Ui:NewScrollView(self, tbControls.SelectSkilltxtSpecial, nil)
  self.tbCurExtScrollView = Ui:NewScrollView(self, tbControls.CurExtAttrTemplate, nil)
  self.tbCurBaseScrollView = Ui:NewScrollView(self, tbControls.CurBaseAttrTemplate, nil)
  self.tbCurSkillScrollView = Ui:NewScrollView(self, tbControls.CurSkillTemplate, nil)
end

function tbWnd:OnOpen(nItemTemplateId)
  nItemTemplateId = nItemTemplateId or Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL
  self:UpHoleInfo()
  self:UpdateSubPanel(nItemTemplateId)
end

function tbWnd:OnClose()
  self.nRedPanel = nil
end

function tbWnd:OnDestroy()
  self.nSelectSeries = nil
  self.tbLeftItemList = nil
  self.nSelectIndex = nil
  self.tbItemGrid = nil
end

function tbWnd:RegistRedPoint()
  for nPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbZhenYuanRedPosKey[nPos], {
      Item.ZhenYuan.tbEquipPosRedPoint[nPos].nEquipRed
    })
  end
end

function tbWnd:GetPlayerCacheData()
  return Item.ZhenYuan:GetPlayerZhenYuanItemBySeries(self.nSelectSeries)
end

function tbWnd:GetItemListSkew(nPos)
  local pEquip = Item.ExtEquip:GetEquipObj(me, nPos)
  if pEquip then
    return 1
  end
  return 0
end

function tbWnd:IsSelctCurEquip()
  local pEquip = Item.ExtEquip:GetEquipObj(me, self.nSelectSeries)
  if pEquip and 1 == self.nSelectIndex then
    return true
  end
  return false
end

function tbWnd:UseEquip()
  if self:IsSelctCurEquip() then
    return
  end
  local nSkew = self:GetItemListSkew(self.nSelectSeries)
  local tbSeriesCache = self:GetPlayerCacheData()
  local tbItem = tbSeriesCache[self.nSelectIndex - nSkew]
  local pItem = me.GetItemInBag(tbItem.dwId)
  if pItem and pItem.nUseLevel > me.nTotalLevel then
    me.CenterMsg(i18n.EasyGet("\232\167\146\232\137\178\231\173\137\231\186\167\228\184\141\232\182\179"))
    return
  end
  RemoteServer.UseEquip(tbItem.dwId, -1)
end

function tbWnd:UnUseEquip()
  if not self:IsSelctCurEquip() then
    return
  end
  local pEquip = Item.ExtEquip:GetEquipObj(me, self.nSelectSeries)
  if pEquip then
    RemoteServer.UnuseEquipById(pEquip.dwId)
  end
end

function tbWnd:UpdateEquipInfo(bUse, nPos, dwId)
  if nPos ~= Item.EQUIPPOS_EXTEND then
    return
  end
  self:UpHoleInfo()
  self:UpdateSubPanel(self.nSelectSeries)
end

function tbWnd:UpdateBagInfo()
  self:UpdateSubPanel(self.nSelectSeries)
end

function tbWnd:UpdateSelectSeriesBtn(nItemTemplateId)
  local tbOldSeries = tbZhenYuanPosKey[self.nSelectSeries]
  local tbNewSeries = tbZhenYuanPosKey[nItemTemplateId]
  self.pPanel:SetActive(tbOldSeries.selectbtn, false)
  self.pPanel:SetActive(tbNewSeries.selectbtn, true)
end

function tbWnd:UpdateSelectItemBtn(nIndex)
  self.pPanel:SetActive(tbControls.Content .. "/Element" .. self.nSelectIndex .. "/imgSelected", false)
  self.pPanel:SetActive(tbControls.Content .. "/Element" .. nIndex .. "/imgSelected", true)
  self.nSelectIndex = nIndex
end

function tbWnd:UpdateSubPanel(nItemTemplateId)
  self:UpdateSelectSeriesBtn(nItemTemplateId)
  self.tbLeftItemList:Reset()
  self.nSelectSeries = nItemTemplateId
  local tbSeriesCache = self:GetPlayerCacheData()
  self:ReSetAttrList()
  local pEquip = Item.ExtEquip:GetEquipObj(me, nItemTemplateId)
  if #tbSeriesCache > 0 then
    self:UpdateLeftPanel()
    self:OnUpdateSelectAttr(self.nSelectIndex)
  elseif pEquip then
    self:UpdateLeftPanel()
    if self:IsSelctCurEquip() then
      self:SetSelectAttrPlan(false, true)
    end
  else
    self.pPanel:SetActive(tbControls.ZhenYuanInfo, false)
    self.pPanel:SetActive(tbControls.AllEmpty, true)
  end
end

function tbWnd:UpdateLeftPanel()
  self.pPanel:SetActive(tbControls.AllEmpty, false)
  self.pPanel:SetActive(tbControls.ZhenYuanInfo, true)
  self:UpdateItemList()
  self:UpdateSelectItemBtn(1)
  self:UpdateCurAttr()
end

function tbWnd:UpdateItemList()
  local tbSeriesCache = self:GetPlayerCacheData()
  local pEquip = Item.ExtEquip:GetEquipObj(me, self.nSelectSeries)
  local nWearPower = 0
  local nQuality = 0
  if pEquip then
    self.tbLeftItemList:SetElement(1, function(tbElement)
      local pPanel = tbElement.pPanel
      if not tbElement.pGrid then
        tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      end
      tbElement.pGrid:SetItemByTemplate(pEquip.dwTemplateId, 1, me.nFaction, me.nFactionSect, {bDisableQualityEffect = true})
      tbElement.pGrid:SetClickEnable(true, function()
      end)
      nQuality = Item:GetQuality(pEquip.dwTemplateId)
      local nFightPower = Item.ZhenYuan:GetEquipPower(pEquip, me, self.nSelectSeries)
      nWearPower = Item.ZhenYuan:GetEquipPower(pEquip)
      pPanel:Label_SetText("FightPower/txtFightPowerValue", i18n.Parse(tostring(nFightPower)))
      local szItemName = Item:GetName(pEquip.dwTemplateId, me)
      pPanel:Label_SetText("txtZhenYuanName", i18n.EasyGet("{0}", szItemName))
      pPanel:SetActive("UIItemGrid", true)
      pPanel:SetActive("StarList/Star", false)
      pPanel:SetActive("imgEquip", true)
    end)
  end
  if self.nRedPanel then
    local tbLastElement = self.tbLeftItemList:GetElement(self.nRedPanel)
    if tbLastElement then
      RedPointMgr:DynamicUnRegist(tbLastElement.pPanel, "UIItemGrid")
      self.nRedPanel = nil
    end
  end
  local bIsRed = false
  local nSkew = self:GetItemListSkew(self.nSelectSeries)
  for nIndex, tbValue in ipairs(tbSeriesCache) do
    self.tbLeftItemList:SetElement(nIndex + nSkew, function(tbElement)
      local pPanel = tbElement.pPanel
      if not tbElement.pGrid then
        tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      end
      tbElement.pGrid:SetItemByTemplate(tbValue.nTemplateId, 1, me.nFaction, me.nFactionSect, {bDisableQualityEffect = true})
      tbElement.pGrid:SetClickEnable(true, function()
        if self.tbItemRed[self.nSelectSeries] and self.tbItemRed[self.nSelectSeries] == nIndex + nSkew then
          self.tbItemRed[self.nSelectSeries] = nil
          self.nRedPanel = nil
          RedPointMgr:FadeRedPoint(Item.ZhenYuan.tbEquipPosRedPoint[self.nSelectSeries].nEquipRed)
        end
      end)
      local nFightPower = tbValue.nFightPower
      local pItem = me.GetItemInBag(tbValue.dwId)
      if pItem then
        tbElement.pGrid:CDLabyer_SetActive(pItem.nUseLevel > me.nTotalLevel)
        if tbValue.nQuality >= nQuality and pItem.nUseLevel <= me.nTotalLevel and not bIsRed then
          if tbValue.nQuality > nQuality then
            bIsRed = true
          end
          if tbValue.nFightPower > nWearPower then
            bIsRed = true
          end
          if bIsRed then
            self.tbItemRed[self.nSelectSeries] = nIndex + nSkew
            self.nRedPanel = nIndex + nSkew
            RedPointMgr:DynamicRegisteRedPoint(pPanel, "UIItemGrid", {
              Item.ZhenYuan.tbEquipPosRedPoint[self.nSelectSeries].nEquipRed
            })
          end
        end
      end
      pPanel:Label_SetText("FightPower/txtFightPowerValue", i18n.Parse(tostring(nFightPower)))
      local szItemName = Item:GetName(tbValue.nTemplateId, me)
      pPanel:Label_SetText("txtZhenYuanName", i18n.EasyGet("{0}", szItemName))
      pPanel:SetActive("UIItemGrid", true)
      pPanel:SetActive("StarList/Star", false)
      pPanel:SetActive("imgEquip", false)
    end)
  end
end

function tbWnd:UpdateCurAttr()
  self.tbCurSkillScrollView:Reset()
  self.tbCurBaseScrollView:Reset()
  self.tbCurExtScrollView:Reset()
  local pEquip = Item.ExtEquip:GetEquipObj(me, self.nSelectSeries)
  if pEquip and pEquip.dwId > 0 then
    self.tbCurSkillScrollView:SetElement(1, function(tbElement)
      local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[pEquip.dwTemplateId]
      if tbItemCfg and tbItemCfg.nSkillId and tbItemCfg.nSkillId > 0 then
        local szNowLevelDesc = FightSkill:GetSkillMagicDesc(tbItemCfg.nSkillId, tbItemCfg.nSkillLv)
        tbElement.pPanel:Label_SetText("txtSpecial", i18n.Parse(szNowLevelDesc))
      else
        tbElement.pPanel:Label_SetText("txtSpecial", i18n.EasyGet("\230\151\160"))
      end
      self.pPanel:ContentSizeFitter_Refresh(tbControls.CurContent)
    end)
    local tbRet1 = Item.ZhenYuan:GetEquipBaseAttr(pEquip)
    local tbRet2 = Item.ZhenYuan:GetEquipRandomAttr(pEquip)
    self.tbCurBaseScrollView:Reset()
    self.tbCurExtScrollView:Reset()
    for i = 1, #tbRet1 do
      self.tbCurBaseScrollView:SetElement(i, function(tbElement)
        local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRet1[i].szAttrType, tbRet1[i].tbAttrValue)
        tbElement.pPanel:Label_SetText("txtValueName", i18n.Parse(szAttrType))
        tbElement.pPanel:Label_SetColorByName("txtValueName", "NewGreen")
        tbElement.pPanel:Label_SetText("txtValue", i18n.Parse(szAttrValue))
        tbElement.pPanel:Label_SetColorByName("txtValue", "NewGreen")
        self.pPanel:ContentSizeFitter_Refresh(tbControls.CurContent)
      end)
    end
    for i = 1, #tbRet2 do
      self.tbCurExtScrollView:SetElement(i, function(tbElement)
        local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRet2[i].szAttrType, tbRet2[i].tbAttrValue)
        tbElement.pPanel:Label_SetText("txtValueName", i18n.Parse(szAttrType))
        tbElement.pPanel:Label_SetColorByName("txtValueName", tbRet2[i].szColor)
        tbElement.pPanel:Label_SetText("txtValue", i18n.Parse(szAttrValue))
        tbElement.pPanel:Label_SetColorByName("txtValue", tbRet2[i].szColor)
        self.pPanel:ContentSizeFitter_Refresh(tbControls.CurContent)
      end)
    end
    self.pPanel:SetActive(tbControls.zhenYuanNow, true)
    self.pPanel:SetActive(tbControls.EmptyNow, false)
  else
    self.pPanel:SetActive(tbControls.zhenYuanNow, false)
    self.pPanel:SetActive(tbControls.EmptyNow, true)
  end
end

function tbWnd:OnUpdateSelectAttr(nIndex)
  self:UpdateSelectItemBtn(nIndex)
  if self:IsSelctCurEquip() then
    self:SetSelectAttrPlan(false, true)
    return
  end
  local tbSeriesCache = self:GetPlayerCacheData()
  local nSkew = self:GetItemListSkew(self.nSelectSeries)
  local tbItem = tbSeriesCache[nIndex - nSkew]
  if not tbItem then
    return
  end
  self:SetSelectAttrPlan(true, false)
  local pItem = me.GetItemInBag(tbItem.dwId)
  self.tbSelectSkillScrollView:Reset()
  self.tbSelectBaseScrollView:Reset()
  self.tbSelectExtScrollView:Reset()
  if pItem then
    self.tbSelectSkillScrollView:SetElement(1, function(tbElement)
      local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[pItem.dwTemplateId]
      if tbItemCfg and tbItemCfg.nSkillId and tbItemCfg.nSkillId > 0 then
        local szNowLevelDesc = FightSkill:GetSkillMagicDesc(tbItemCfg.nSkillId, tbItemCfg.nSkillLv)
        tbElement.pPanel:Label_SetText("txtSpecial", i18n.Parse(szNowLevelDesc))
      else
        tbElement.pPanel:Label_SetText("txtSpecial", i18n.EasyGet("\230\151\160"))
      end
      self.pPanel:ContentSizeFitter_Refresh(tbControls.SelectContent)
    end)
    local tbRet1 = Item.ZhenYuan:GetEquipBaseAttr(pItem)
    local tbRet2 = Item.ZhenYuan:GetEquipRandomAttr(pItem)
    for i = 1, #tbRet1 do
      self.tbSelectBaseScrollView:SetElement(i, function(tbElement)
        local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRet1[i].szAttrType, tbRet1[i].tbAttrValue)
        tbElement.pPanel:Label_SetText("txtValueName", i18n.Parse(szAttrType))
        tbElement.pPanel:Label_SetColorByName("txtValueName", "NewGreen")
        tbElement.pPanel:Label_SetText("txtValue", i18n.Parse(szAttrValue))
        tbElement.pPanel:Label_SetColorByName("txtValue", "NewGreen")
        self.pPanel:ContentSizeFitter_Refresh(tbControls.SelectContent)
      end)
    end
    for i = 1, #tbRet2 do
      self.tbSelectExtScrollView:SetElement(i, function(tbElement)
        local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRet2[i].szAttrType, tbRet2[i].tbAttrValue)
        tbElement.pPanel:Label_SetText("txtValueName", i18n.Parse(szAttrType))
        tbElement.pPanel:Label_SetColorByName("txtValueName", tbRet2[i].szColor)
        tbElement.pPanel:Label_SetText("txtValue", i18n.Parse(szAttrValue))
        tbElement.pPanel:Label_SetColorByName("txtValue", tbRet2[i].szColor)
        self.pPanel:ContentSizeFitter_Refresh(tbControls.SelectContent)
      end)
    end
  end
end

function tbWnd:SetSelectAttrPlan(bZhenYuan, bEmpty)

  self.pPanel:SetActive(tbControls.btnEquip, bZhenYuan)
  self.pPanel:SetActive(tbControls.imgArrow, bZhenYuan)
  self.pPanel:SetActive(tbControls.EmptySelect, bEmpty)
  self.pPanel:SetActive(tbControls.btnUnEquip, bEmpty)
  self.pPanel:SetActive(tbControls.btnRecover, bZhenYuan)
end

function tbWnd:ReSetAttrList()
  self.tbSelectBaseScrollView:Reset()
  self.tbSelectExtScrollView:Reset()
  self.tbCurExtScrollView:Reset()
  self.tbCurBaseScrollView:Reset()
  self.tbSelectSkillScrollView:Reset()
end

function tbWnd:UpHoleInfo()
  for nPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local pEquip = Item.ExtEquip:GetEquipObj(me, nPos)
    if pEquip then
      self.pPanel:SetActive(tbZhenYuanPosKey[nPos].holeaddr, true)
      local itemGrid = self.tbItemGrid[nPos]
      if not itemGrid then
        itemGrid = Ui:NewExistItemGrid(self, tbZhenYuanPosKey[nPos].holeaddr .. "/UIItemGrid")
        self.tbItemGrid[nPos] = itemGrid
      end
      itemGrid:SetItemByTemplate(pEquip.dwTemplateId, 1, me.nFaction, me.nFactionSect, {bDisableQualityEffect = true})
      itemGrid:SetClickEnable(true, function()
        self:UpdateSubPanel(nPos)
      end)
    else
      self.pPanel:SetActive(tbZhenYuanPosKey[nPos].holeaddr, false)
    end
  end
  return true
end
