local tbWnd = Ui:CreateClass("UIEquipSuitInfo", "UIPanel")
local tbControls = {
  attrContent = "imgBg/attrContent",
  equipLvCell = "imgBg/equipLvCell",
  SuitClickEmptyToClose = "ClickEmptyToClose",
  txtEquipSuitTitle = "imgBg/txtTitle",
  attrSuitTemplate = "imgBg/attrContent/attrSuitTemplate",
  txtEquipSuitDesc = "imgBg/txtDesc",
  txtEquipSuitLevelNow = "imgBg/equipLvCell/txtLevelNow",
  txtEquipSuitLevelNext = "imgBg/equipLvCell/nextInfo/txtLevelNext",
  txtEquipSuitUpgrade = "imgBg/txtUpgrade",
  txtEquipSuitUnlocked = "imgBg/txtUnlocked",
  txtEquipSuitLock = "imgBg/txtLock",
  imgEquipSuitArrow = "imgBg/equipLvCell/nextInfo/imgArrow1 "
}
tbWnd.tbControls = tbControls

function tbWnd:UpdateView()
  local PartnerEquipC = Item.PartnerEquip
  self.nSuitLevel = PartnerEquipC:GetPlayerSuitAttrLevel(me)
  self.nHasReforgeFullNum = me.GetUserValue(PartnerEquipC.SAVE_GROUP, PartnerEquipC.SAVE_ID.REFORGE_FULL_NUM)
  self:UpdateEquipSuit()
  self.pPanel:SetActive("Main", true)
end

function tbWnd:UpdateEquipSuit()
  self.pPanel:SetActive(tbControls.txtEquipSuitLock, false)
  if not self.tbEquipSuitAttrView then
    self.tbEquipSuitAttrView = Ui:NewScrollView(self, tbControls.attrSuitTemplate)
  end
  local PartnerEquipC = Item.PartnerEquip
  local nMaxSuitLevel = #Item.PartnerEquipEnhance.SUIT_ATTR_VAL
  if self.nSuitLevel > 0 then
    local bNotMaxLevel = nMaxSuitLevel > self.nSuitLevel
    self.pPanel:SetActive(tbControls.txtEquipSuitLock, false)
    self.pPanel:SetActive(tbControls.txtEquipSuitDesc, bNotMaxLevel)
    self.pPanel:SetActive(tbControls.imgEquipSuitArrow, bNotMaxLevel)
    self.pPanel:SetActive(tbControls.txtEquipSuitLevelNext, bNotMaxLevel)
    self.pPanel:SetActive(tbControls.txtEquipSuitUpgrade, bNotMaxLevel)
    self.pPanel:SetActive(tbControls.txtEquipSuitUnlocked, bNotMaxLevel)
    self.pPanel:Label_SetText(tbControls.txtEquipSuitLevelNow, i18n.EasyGet("{0}\231\186\167", self.nSuitLevel))
    self.pPanel:Label_SetText(tbControls.txtEquipSuitTitle, i18n.EasyGet("\229\165\151\232\163\133\229\177\158\230\128\167{0}\231\186\167", self.nSuitLevel))
    if bNotMaxLevel then
      local nReq = Item.PartnerEquipEnhance.SUIT_ATTR_VAL[self.nSuitLevel + 1][5]
      self.pPanel:Label_SetText(tbControls.txtEquipSuitLevelNext, i18n.EasyGet("{0}\231\186\167", self.nSuitLevel + 1))
      self.pPanel:Label_SetText(tbControls.txtEquipSuitUpgrade, i18n.EasyGet("\229\141\135\231\186\167\230\157\161\228\187\182\239\188\154\232\167\163\233\148\129{0}\228\187\182\229\144\140\228\188\180\232\163\133\229\164\135\231\154\132\233\154\144\232\151\143\229\177\158\230\128\167", nReq))
      self.pPanel:Label_SetText(tbControls.txtEquipSuitUnlocked, i18n.EasyGet("\229\183\178\232\167\163\233\148\129\239\188\154{0}/{1}", self.nHasReforgeFullNum, nReq))
    end
    local tbSuitAttrNowList = PartnerEquipC:GetSuitAttrInfoByLevel(self.nSuitLevel)
    local tbSuitAttrNextList = PartnerEquipC:GetSuitAttrInfoByLevel(self.nSuitLevel + 1)
    self.tbSuitAttrDiffList = PartnerEquipC:GetEquipAttrDiffByBatch(tbSuitAttrNowList, tbSuitAttrNextList) or {}
    for nIndex, _ in ipairs(self.tbSuitAttrDiffList) do
      self.tbEquipSuitAttrView:SetElement(nIndex, function(tbElement, nOrder)
        self:_SetSuitAttrElementUnLock(tbElement, nOrder)
      end)
    end
  else
    self.pPanel:Label_SetText(tbControls.txtEquipSuitLevelNow, i18n.EasyGet("\230\156\170\230\191\128\230\180\187"))
    self.pPanel:Label_SetText(tbControls.txtEquipSuitLevelNext, i18n.EasyGet("{0}\231\186\167", self.nSuitLevel + 1))
    self.pPanel:Label_SetText(tbControls.txtEquipSuitTitle, i18n.EasyGet("\229\165\151\232\163\133\229\177\158\230\128\167"))
    local nReq = Item.PartnerEquipEnhance.SUIT_ATTR_VAL[self.nSuitLevel + 1][5]
    self.pPanel:Label_SetText(tbControls.txtEquipSuitUpgrade, i18n.EasyGet("\229\141\135\231\186\167\230\157\161\228\187\182\239\188\154\232\167\163\233\148\129{0}\228\187\182\229\144\140\228\188\180\232\163\133\229\164\135\231\154\132\233\154\144\232\151\143\229\177\158\230\128\167", nReq))
    self.pPanel:Label_SetText(tbControls.txtEquipSuitUnlocked, i18n.EasyGet("\229\183\178\232\167\163\233\148\129\239\188\154{0}/{1}", self.nHasReforgeFullNum, nReq))
    local tbSuitAttrNextList = PartnerEquipC:GetSuitAttrInfoByLevel(self.nSuitLevel + 1)
    self.tbSuitAttrDiffList = PartnerEquipC:GetEquipAttrDiffByBatch(tbSuitAttrNextList, tbSuitAttrNextList)
    for nIndex, _ in ipairs(self.tbSuitAttrDiffList) do
      self.tbEquipSuitAttrView:SetElement(nIndex, function(tbElement, nOrder)
        self:_SetSuitAttrElementLock(tbElement, nOrder)
      end)
    end
  end
end

function tbWnd:_SetSuitAttrElementLock(tbElement, nIndex)
  tbElement.pPanel:Label_SetText("newinfo/txtAttrTip", self.tbSuitAttrDiffList[nIndex].szAttrDesc)
  tbElement.pPanel:Label_SetText("newinfo/txtAttrValueNow", "0")
  tbElement.pPanel:Label_SetText("nextInfo/txtAttrValueNext", self.tbSuitAttrDiffList[nIndex].nAttr2Value)
  tbElement.pPanel:Label_SetText("nextInfo/txtAttrValueUp", "+" .. self.tbSuitAttrDiffList[nIndex].nAttr1Value)
  tbElement.pPanel:SetActive("nextInfo/imgArrow1", true)
  tbElement.pPanel:SetActive("nextInfo/txtAttrValueNext", true)
  tbElement.pPanel:SetActive("nextInfo/txtAttrValueUp", true)
end

function tbWnd:_SetSuitAttrElementUnLock(tbElement, nIndex)
  tbElement.pPanel:Label_SetText("newinfo/txtAttrTip", self.tbSuitAttrDiffList[nIndex].szAttrDesc)
  tbElement.pPanel:Label_SetText("newinfo/txtAttrValueNow", self.tbSuitAttrDiffList[nIndex].nAttr1Value)
  if self.tbSuitAttrDiffList[nIndex].nAttr2Value then
    tbElement.pPanel:Label_SetText("nextInfo/txtAttrValueNext", self.tbSuitAttrDiffList[nIndex].nAttr2Value)
    tbElement.pPanel:Label_SetText("nextInfo/txtAttrValueUp", "+" .. tostring(self.tbSuitAttrDiffList[nIndex].nAttr2Value - self.tbSuitAttrDiffList[nIndex].nAttr1Value))
  end
  tbElement.pPanel:SetActive("nextInfo/imgArrow1", self.tbSuitAttrDiffList[nIndex].nAttr2Value ~= nil)
  tbElement.pPanel:SetActive("nextInfo/txtAttrValueNext", self.tbSuitAttrDiffList[nIndex].nAttr2Value ~= nil)
  tbElement.pPanel:SetActive("nextInfo/txtAttrValueUp", self.tbSuitAttrDiffList[nIndex].nAttr2Value ~= nil)
end
