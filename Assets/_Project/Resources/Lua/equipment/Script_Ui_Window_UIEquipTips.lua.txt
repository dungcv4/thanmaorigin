function tbWnd:OnOpen(tbInfo)
  self.nItemId = tbInfo.nItemId
  self.nTemplateId = tbInfo.nTemplateId
  self.nFaction = tbInfo.nFaction or me.nFaction
  self.nFactionSect = tbInfo.nFactionSect or me.nFactionSect
  self.nSex = tbInfo.nSex or me.GetNpc().GetBodySex()
  self.szOpt = tbInfo.szOpt
  self.nSeries = tbInfo.nSeries
  self.pPanel:Object_SetLocalPosition(self.panelRoot, tbInfo.pos)
  if 0 == self.nItemId then
    self.nItemId = nil
  end
  if self.nItemId then
    local pItem = KItem.GetItemObj(self.nItemId)
    self.pItem = pItem
    self.nTemplateId = pItem.dwTemplateId
  else
    self.pItem = nil
  end
  self.tbFuncOnButtonClick = {}
  local tbDetailInfo = {
    szOpt = self.szOpt,
    pItem = self.pItem,
    nItemId = self.nItemId,
    nTemplateId = self.nTemplateId,
    nFaction = self.nFaction,
    nFactionSect = self.nFactionSect,
    nSex = self.nSex,
    pAsyncData = tbInfo.pAsyncData,
    bIsCompare = self.ClassName == "UICompareTips",
    tbItemExtParams = tbInfo.tbItemExtParams,
    tbLinkInfo = tbInfo.tbLinkInfo,
    nSeries = self.nSeries
  }
  self.tbEquipDetail:Update(tbDetailInfo)
  self:UpdateButton()
end

function tbWnd:CheckRedPoint_Enhance(nEquipId)
  local pEquip = KItem.GetItemObj(nEquipId)
  if not pEquip then
    return false
  end
  local nEnhanceLevel = Item.EquipEnhance:GetEquipEnhanceLevel(me, self.pItem.nEquipPos)
  if not Item.EquipWuLian:IsWuLianOpenC(nEnhanceLevel, pEquip.nLevel) then
    return Item.EquipEnhance:QuickCheckEnhance(me, pEquip)
  else
    return Item.EquipWuLian:QuickCheckWuLian(me, pEquip)
  end
end

function tbWnd:UpdateButton()
  local pItem = self.pItem
  local szOpt = self.szOpt
  if pItem and szOpt and self.tbButtonOption[szOpt] and not self.pAsyncData then
    self.tbButtonGroup:SetActiveAll(true)
    self.tbDynamicButtonsTypeToIndex = {}
    local nIndex = 1
    for _, tbButton in ipairs(self.tbButtonOption[szOpt]) do
      local fnVisible, szBtnName, fnCallback, fnCheckRedPoint, nButtonType = table.unpack(tbButton)
      if fnVisible and fnVisible(self) then
        local bLightRedPoint = fnCheckRedPoint and fnCheckRedPoint(self, pItem.dwId) or false
        self.tbButtonGroup:SetActive(nIndex, true)
        if type(szBtnName) == "string" then
          self.tbButtonGroup:SetButtonText(nIndex, szBtnName)
        elseif type(szBtnName) == "function" then
          self.tbButtonGroup:SetButtonText(nIndex, szBtnName(tbWnd))
        end
        self.tbFuncOnButtonClick[nIndex] = fnCallback
        if bLightRedPoint then
          self.tbButtonGroup:LightRedPoint(nIndex)
        else
          self.tbButtonGroup:FadeRedPoint(nIndex)
        end
        self.tbDynamicButtonsTypeToIndex[nButtonType] = nIndex
        nIndex = nIndex + 1
      end
    end
    for i = nIndex, #self.tbButtonGroupKey do
      self.tbButtonGroup:SetActive(i, false)
      self.tbButtonGroup:FadeRedPoint(i)
      self.tbFuncOnButtonClick[i] = nil
    end
    self.pPanel:SetActive(self.btnShare, not self.tbButtonOption[szOpt].bHideShareButton)
    local nCanMarketStall
    self.nMarketType = nil
    nCanMarketStall, self.nMarketType = MarketStallBase:CheckMarketSellItemTypeSettings(pItem)
    self.pPanel:SetActive(self.btnHawk, nCanMarketStall and not self.tbButtonOption[szOpt].bHideHawkButton)
    self.pPanel:Button_SetText(self.btnHawk, i18n.EasyGet("\230\145\134\230\145\138"), "Text")
    local bShowSell = not self.tbButtonOption[szOpt].bHideSellButton
    local bCanSell, _ = Item.tbItemSell:CheckCanSellItem(pItem.dwTemplateId)
    bCanSell = bCanSell or HonorLevel:CanSell(me, pItem.dwId)
    self.pPanel:SetActive(self.btnSell, bCanSell and bShowSell)
    local nCanMoveToBox = false
    local nVipState = false
    if pItem and self.szOpt == "ItemBox" then
      local tbBaseInfo = Item:GetItemBaseProp(pItem.dwTemplateId)
      if tbBaseInfo.bForbidToBox and 0 == tbBaseInfo.bForbidToBox then
        nCanMoveToBox = true
      end
      local weekCardLeft = WeekMonthCard:GetWeekCardLeftDay(me)
      local bMonthCardState = WeekMonthCard:CheckMonthCardState(me)
      if weekCardLeft > 0 or bMonthCardState then
        nVipState = true
      end
    end
    self.pPanel:SetActive(self.btnMoveToBox, nCanMoveToBox and nVipState)
    self.pPanel:Label_SetText(self.btnMoveToBox .. "/Text", i18n.EasyGet("\231\167\187\229\133\165\228\187\147\229\186\147"))
    return
  end
  self.tbButtonGroup:SetActiveAll(false)
  self.pPanel:SetActive(self.btnShare, false)
  self.pPanel:SetActive(self.btnHawk, false)
  self.pPanel:SetActive(self.btnSell, false)
  self.pPanel:SetActive(self.btnMoveToBox, false)
end

function tbWnd:CanForgeTransfer()
  if not self.nItemId then
    return false
  end
  local pSrcItem = KItem.GetItemObj(self.nItemId)
  if not pSrcItem then
    return false
  end
  if pSrcItem.nEquipPos < Item.EQUIPPOS_HELM or pSrcItem.nEquipPos > Item.EQUIPPOS_AMULET then
    return false
  end
  local pDstItem = me.GetEquipByPos(pSrcItem.nEquipPos)
  if not pDstItem then
    return false
  end
  return true
end

function tbWnd:ForgeTransfer()
  if not self.nItemId then
    return
  end
  local pSrcItem = KItem.GetItemObj(self.nItemId)
  if not pSrcItem then
    return
  end
  local pDstItem = me.GetEquipByPos(pSrcItem.nEquipPos)
  if pDstItem then
    if pSrcItem.nLevel > pDstItem.nLevel then
      me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\230\137\128\233\128\137\232\163\133\229\164\135\231\173\137\233\152\182\230\155\180\233\171\152\239\188\140\230\151\160\230\179\149\230\180\151\231\187\131\239\188\140\229\187\186\232\174\174\231\169\191\230\136\180"))
      return
    end
    if pSrcItem.nLevel == pDstItem.nLevel and pSrcItem.nQuality > pDstItem.nQuality then
      me.CenterMsg(i18n.EasyGet("\232\175\165\232\163\133\229\164\135\229\177\158\230\128\167\230\155\180\233\171\152\239\188\140\230\151\160\230\179\149\230\180\151\231\187\131\239\188\140\229\187\186\232\174\174\231\169\191\230\136\180"))
      return
    end
    if pSrcItem.nLevel < pDstItem.nLevel and pSrcItem.nOrgValue > pDstItem.nOrgValue then
      me.CenterMsg(i18n.EasyGet("\232\175\165\232\163\133\229\164\135\229\177\158\230\128\167\230\155\180\233\171\152\239\188\140\230\151\160\230\179\149\230\180\151\231\187\131\239\188\140\229\187\186\232\174\174\231\169\191\230\136\180"))
      return
    end
  end
  Ui:OpenWindow("UIReplaceAttrib", pSrcItem.dwId, pDstItem and pDstItem.dwId or 0)
  self:OnClickButtonClose()
end

function tbWnd:CanSell()
  if not self.nItemId then
    return false
  end
  local pItem = KItem.GetItemObj(self.nItemId)
  if not pItem then
    return false
  end
  if not Shop:CanItemSell(me, pItem.dwTemplateId) then
    return false
  end
  local nPrice, szMoneyType = Shop:GetSellSumPrice(me, self.nTemplateId, pItem and pItem.nCount or 1)
  self.nPrice = nPrice
  self.szMoneyType = szMoneyType
  if not self.szMoneyType then
    return false
  end
  if not self.nPrice then
    return false
  end
  return true
end

function tbWnd:SellItem()
  if not self.nItemId then
    return
  end
  if self:IsSecretBook(self.nItemId) then
    self:ShowSellSecretBook(self.nTemplateId, self.nItemId, 1)
    return
  end
  if Item.tbItemSell:CheckCanSellItem(self.nTemplateId) then
    Ui:OpenWindow("UIItemSell", self.nItemId, self.nTemplateId)
    self:OnClickButtonClose()
    return
  end
  if HonorLevel:CanSell(me, self.nItemId) then
    HonorLevel:SellReq(self.nItemId)
    self:OnClickButtonClose()
    return
  end
end

  local pItem = KItem.GetItemObj(self.nItemId)
  if not pItem then
    return false
  end
  local nEquipPos = KItem.GetEquipPos(pItem.dwTemplateId)
  if nEquipPos < Item.EQUIPPOS_HELM or nEquipPos > Item.EQUIPPOS_PENDANT then
    return false
  end
  if pItem.nHoleCount and pItem.nHoleCount <= 0 then
    return false
  end
  return true
end

function tbWnd:MoveItemFromBagToBox()
  local pItem = KItem.GetItemObj(self.nItemId)
  if not pItem then
    return
  end
  local tbBaseInfo = Item:GetItemBaseProp(pItem.dwTemplateId)
  if tbBaseInfo.bForbidToBox and 1 == tbBaseInfo.bForbidToBox then
    me.CenterMsg(i18n.EasyGet("\232\175\165\231\137\169\229\147\129\228\184\141\229\143\175\229\173\152\229\133\165\228\187\147\229\186\147"))
    return

  if me.GetFreeBoxCount() <= 0 then
    me.CenterMsg(i18n.EasyGet("\229\176\145\228\190\160\228\187\147\229\186\147\229\183\178\230\187\161\239\188\140\230\151\160\230\179\149\229\173\152\229\133\165\231\137\169\229\147\129"))
    return
  end
  if pItem.nPos ~= Item.EITEMPOS_BAG then
    me.CenterMsg(i18n.EasyGet("\232\175\165\231\137\169\229\147\129\228\184\141\229\156\168\232\131\140\229\140\133\229\134\133"))

    Ui:CloseWindow("UICompareTips")
  end
  self:Close()
end
