local tbControls = {
  panelEquipList = "panelEquipList",
  equipTemplate = "panelEquipList/imgList/List/Template",
  UIItemGridBefore = "panelCenter/EquipBefore/UIItemGrid",
  UIItemGridAfter = "panelCenter/EquipAfter/UIItemGrid",
  btnDelete = "panelCenter/EquipAfter/btnDelete",
  imgLock = "panelCenter/EquipAfter/imgLock",
  btnLock = "panelCenter/EquipAfter/imgLock/Empty4Raycast",
  txtEquitpName = "panelCenter/txtEquitpName",
  panelFull = "panelFull",
  CenterFull = "panelFull/Center",
  AttributeFullCuiLing = "panelFull/Center/Attribute/txtAttribute",
  btnChange = "panelFull/Center/btnChange",
  btnCompose = "panelFull/Center/btnCompose",
  txtFullSuitLevel = "panelFull/Right/SuitAttr/txtLevel",
  txtSuitFullTitle = "panelFull/Right/SuitAttr/txtSuit",
  txtWenyangLevel = "panelFull/Right/SuitAttr/txtValue",
  AttributeFullSuit = "panelFull/Right/SuitAttr/Attribute",
  txtTitleLianHua = "panelFull/Right/txtTitleLianHua",
  widConsume = "panelFull/Right/Template1",
  btnAdd = "panelFull/Right/Template1/btnAdd",
  UIItemGridConsume = "panelFull/Right/Template1/UIItemGrid",
  btnWenYnag = "panelFull/Right/btnChange",
  txtSuccessRate = "panelFull/Right/txtSuccessRate",
  txtTips = "panelFull/Right/txtValue",
  txtTips1 = "panelFull/Right/txtLV",
  txtCost = "panelFull/Right/Template1/txtCost",
  BGLingZhu = "panelCenter/EquipAfter/BGAfter",
  txtCoinCost = "panelFull/Right/txtCost",
  panelEmpty = "panelEmpty",
  txtEmptySuitLevel = "panelEmpty/SuitAttr/txtLevel",
  AttributeEmpthSuit = "panelEmpty/SuitAttr/Attribute",
  btnCuiling = "panelEmpty/btnCuiling"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCuiling = function(self)
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    EquipCuiLing:CuiLingReqC(nEquipPos)
  end,
  BGLingZhu = function(self)
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    self.tbOwner:ShowCuiLingInfo(nEquipPos, true)
  end,
  btnLock = function(self)
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    EquipCuiLing:CuiLingReqC(nEquipPos)
  end,
  btnDelete = function(self)
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    EquipCuiLing:ECLTakeOffC(nEquipPos)
  end,
  btnChange = function(self)
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    self.tbOwner:ShowCuiLingInfo(nEquipPos, true)
  end,
  btnCompose = function(self)
    Ui:OpenWindow("UICuilingCompose")
  end,
  btnWenYnag = function(self)
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    EquipCuiLing:ECLWenYangC(nEquipPos)
  end,
  btnAdd = function(self)
    self.tbOwner:ShowJumpToInfo(4)
  end
}

function tbWnd:OnCreate()
  self.tbEquipScrollView = Ui:NewScrollView(self, tbControls.equipTemplate, self.OnEquipClick, false, true)
  self.tbUIItemGridEquip = Ui:NewExistItemGrid(self, tbControls.UIItemGridBefore)
  self.tbUIItemGridEquip:SetClickEnable(false)
  self.tbUIItemGridLingZhu = Ui:NewExistItemGrid(self, tbControls.UIItemGridAfter)
  self.tbUIItemGridLingZhu:SetClickEnable(false)
  self.tbUIItemGridConsume = Ui:NewExistItemGrid(self, tbControls.UIItemGridConsume)
  self.tbUIItemGridConsume:SetClickEnable(false)
end

function tbWnd:OnOpen(nEquipPos)
  self.tbSVData = {}
  local tbEquip = me.GetEquips()
  for k, v in pairs(tbEquip) do
    if k <= Item.EQUIPPOS_AMULET then
      local pEquip = KItem.GetItemObj(v)
      table.insert(self.tbSVData, {
        k,
        v,
        Item.tbEquipSort[pEquip.nEquipPos]
      })
    end
  end
  if 0 == #self.tbSVData then
    me.CenterMsg(i18n.EasyGet("\229\176\154\230\156\170\231\169\191\230\136\180\232\163\133\229\164\135"))
    return 0
  end
  
  local function fnCmp(a, b)
    return a[3] < b[3]
  end
  
  table.sort(self.tbSVData, fnCmp)
  self.nEquipIndex = 1
  if nEquipPos then
    for k, v in ipairs(self.tbSVData) do
      if v[1] == nEquipPos then
        self.nEquipIndex = k
        break
      end
    end
  end
  self.tbEquipScrollView:ScrollToIndex(self.nEquipIndex, true)
  self:UpdateEquipSV()
  self:UpdateEquipDetailInfo()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_EQUIP_CUI_LING_SUCCESS, self.OnCuiLingRsp, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_PUTON_SUCCESS, self.OnPutOnRsp, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_TAKEOFF_SUCCESS, self.OnTakeOffRsp, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_WENYANG_RESULT, self.OnWenYangRsp, self)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnCompose, {
    RedPointMgr.EVENT.EquipLingZhuCompose
  })
end

function tbWnd:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, self.btnCompose)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_EQUIP_CUI_LING_SUCCESS, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_PUTON_SUCCESS, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_TAKEOFF_SUCCESS, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_EQUIP_LINGZHU_WENYANG_RESULT, self)
end

function tbWnd:UpdateView()
end

function tbWnd:EquipfnSetItem(i, tbElement)
  local nEquipPos, nEquipId = self.tbSVData[i][1], self.tbSVData[i][2]
  local pEquip = KItem.GetItemObj(nEquipId)
  local pPanel = tbElement.pPanel
  local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  itemGrid:SetItem(nEquipId, pEquip.dwTemplateId)
  itemGrid:SetClickEnable(true, function()
    self:OnEquipClick(i)
  end)
  tbElement.pPanel:SetActive("imgSelected", i == self.nEquipIndex)
  pPanel:Label_SetText("txtEquipName", Item:GetName(pEquip.dwTemplateId, me))
  local nWenYangLevel = EquipCuiLing:GetWenYangLevel(me, nEquipPos)
  pPanel:Label_SetText("txtWenyangLevel/txtLevelValue", nWenYangLevel)
  local tbLingZhuData = EquipCuiLing:GetPutOnLingZhuData(me, nEquipPos)
  if tbLingZhuData.nTemplateId > 0 then
    pPanel:SetActive("ImgSign", true)
  else
    pPanel:SetActive("ImgSign", false)
  end
  RedPointMgr:DynamicRegisteRedPoint(pPanel, "Main", {
    EquipCuiLing.tbEquipPos2RedPoint[nEquipPos],
    EquipCuiLing.tbEquipPos2WYRedPoint[nEquipPos]
  })
end

function tbWnd:UpdateEquipSV()
  self.tbEquipScrollView:Reset()
  for i, v in pairs(self.tbSVData) do
    self.tbEquipScrollView:SetElement(i, function(tbElement)
      self:EquipfnSetItem(i, tbElement)
    end)
  end
end

function tbWnd:OnEquipClick(nIndex)
  if self.nEquipIndex == nIndex then
    return
  end
  local tbOldElement = self.tbEquipScrollView:GetElement(self.nEquipIndex)
  if tbOldElement then
    tbOldElement.pPanel:SetActive("imgSelected", false)
  end
  self.nEquipIndex = nIndex
  local nEquipPos = self.tbSVData[self.nEquipIndex][1]
  local tbCurElement = self.tbEquipScrollView:GetElement(self.nEquipIndex)
  tbCurElement.pPanel:SetActive("imgSelected", true)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnChange, {
    EquipCuiLing.tbEquipPos2RedPoint[nEquipPos]
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnCuiling, {
    EquipCuiLing.tbEquipPos2RedPoint[nEquipPos]
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnWenYnag, {
    EquipCuiLing.tbEquipPos2WYRedPoint[nEquipPos]
  })
  self:UpdateEquipDetailInfo()
end

function tbWnd:UpdateEquipDetailInfo()
  if not self.tbSVData[self.nEquipIndex] then
    return
  end
  local nEquipPos = self.tbSVData[self.nEquipIndex][1]
  local pEquip = me.GetEquipByPos(nEquipPos)
  if not pEquip then
    return
  end
  self.pPanel:Label_SetText(self.txtEquitpName, Item:GetName(pEquip.dwTemplateId, me))
  self.tbUIItemGridEquip:SetItem(pEquip.dwId, pEquip.dwTemplateId)
  self.tbUIItemGridEquip:SetClickEnable(true)
  local nPutOnLingZhuCount = EquipCuiLing:GetPutOnLingZhuCount(me)
  if EquipCuiLing:GetCuiLingState(me, nEquipPos) then
    self.pPanel:SetActive(tbControls.panelFull, true)
    self.pPanel:SetActive(tbControls.panelEmpty, false)
    self.pPanel:SetActive(tbControls.imgLock, false)
    self.pPanel:SetActive(tbControls.txtTips, false)
    local tbAttrValue = {}
    local tbAttrNextValue = {}
    for i = 1, 3 do
      tbAttrValue[i] = {
        table.unpack(EquipCuiLing.tbCLSuitAttr[1][i], 2)
      }
      tbAttrNextValue[i] = {
        table.unpack(EquipCuiLing.tbCLSuitAttr[1][i], 2)
      }
    end
    local tbLingZhuData = EquipCuiLing:GetPutOnLingZhuData(me, nEquipPos)
    if nPutOnLingZhuCount >= Item.EQUIPPOS_MAIN_NUM then
      self.pPanel:SetActive(tbControls.widConsume, true)
      self.pPanel:SetActive(tbControls.btnWenYnag, true)
    else
      self.pPanel:SetActive(tbControls.widConsume, false)
      self.pPanel:SetActive(tbControls.btnWenYnag, false)
    end
    local nAllWYLevel = EquipCuiLing:GetWenYangAllLevel(me)
    local nNextWYLevel = nAllWYLevel + 1
    local nMaxLevel = EquipCuiLing:GetCurTimeFrameMaxWenYangLevel()
    nNextWYLevel = math.min(nNextWYLevel, nMaxLevel * Item.EQUIPPOS_MAIN_NUM)
    if tbLingZhuData.nTemplateId > 0 then
      self.pPanel:SetActive(tbControls.CenterFull, true)
      self.pPanel:SetActive(tbControls.btnDelete, true)
      self.pPanel:SetActive(tbControls.txtSuitFullTitle, true)
      self.pPanel:SetActive(tbControls.txtTips1, true)
      self.tbUIItemGridLingZhu:SetItemByTemplate(tbLingZhuData.nTemplateId)
      self.tbUIItemGridLingZhu:Plus_SetActive(false)
      self.tbUIItemGridLingZhu:SetClickEnable(true)
      self.pPanel:Label_SetText(tbControls.txtWenyangLevel, i18n.EasyGet("{0}->{1}", nAllWYLevel, nNextWYLevel))
      local tbCfg = EquipCuiLing:GetWenYangCfgByLevel(nAllWYLevel + 1)
      local nCurrencyTypeCost = tbCfg[2]
      self.pPanel:Label_SetText(tbControls.txtCoinCost, i18n.EasyGet("\232\138\177\232\180\185\233\147\182\228\184\164{0}", nCurrencyTypeCost))
      local nItemTemplateId = tbCfg[3]
      self.tbUIItemGridConsume:SetItemByTemplate(nItemTemplateId)
      self.tbUIItemGridConsume:SetClickEnable(true)
      local nCurNum = me.GetItemCountByTypeInBag(nItemTemplateId)
      local nNeedNum = tbCfg[4]
      if nCurNum >= nNeedNum then
        self.pPanel:Label_SetColorByName(tbControls.txtCost, "Enough")
      else
        self.pPanel:Label_SetColorByName(tbControls.txtCost, "NotEnough")
      end
      self.pPanel:Label_SetText(tbControls.txtCost, i18n.EasyGet("{0}/{1}", nCurNum, nNeedNum))
      local tbAttr = EquipCuiLing:GetPutOnLingZhuData(me, nEquipPos)
      for i = 1, 3 do
        if tbAttr[i] then
          local tbAttrCfg = EquipCuiLing.tbLZAttrId2AttrType[tbAttr[i][1]]
          local szAttrType = tbAttrCfg.szAttrType
          local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(szAttrType, tbAttr[i][2])
          self.pPanel:Label_SetText(string.format("%s%d", tbControls.AttributeFullCuiLing, i), szAttribMsg)
          self.pPanel:Label_SetText(string.format("%s%d/txtNum", tbControls.AttributeFullCuiLing, i), szValue)
        else
          self.pPanel:Label_SetText(string.format("%s%d", tbControls.AttributeFullCuiLing, i), "")
          self.pPanel:Label_SetText(string.format("%s%d/txtNum", tbControls.AttributeFullCuiLing, i), "")
        end
      end
      if nPutOnLingZhuCount >= Item.EQUIPPOS_MAIN_NUM then
        self.pPanel:Label_SetText(tbControls.txtSuccessRate, i18n.EasyGet("\230\136\144\229\138\159\231\142\135\239\188\154{0}%", tbCfg[5] / 10000))
        self.pPanel:Label_SetText(tbControls.txtTips1, i18n.EasyGet("\230\155\180\230\141\162\232\163\133\229\164\135\228\188\154\228\191\157\231\149\153\230\184\169\229\133\187\231\173\137\231\186\167"))
      else
        self.pPanel:Label_SetText(tbControls.txtTips1, i18n.EasyGet("\230\191\128\230\180\187\233\149\135\229\178\179\229\165\151\232\163\133\229\144\142\230\150\185\229\143\175\230\184\169\229\133\187\231\187\167\231\187\173\229\188\186\229\140\150"))
        self.pPanel:Label_SetText(tbControls.txtSuccessRate, "")
        self.pPanel:Label_SetText(tbControls.txtCoinCost, "")
      end
    else
      self.pPanel:SetActive(tbControls.btnDelete, false)
      self.pPanel:SetActive(tbControls.CenterFull, false)
      self.pPanel:SetActive(tbControls.txtSuitFullTitle, false)
      self.pPanel:SetActive(tbControls.txtTips1, false)
      self.pPanel:SetActive(tbControls.btnWenYnag, false)
      self.pPanel:SetActive(tbControls.widConsume, false)
      self.tbUIItemGridLingZhu:SetEmpty(true)
      self.tbUIItemGridLingZhu:Plus_SetActive(true)
      self.pPanel:Label_SetText(tbControls.txtCoinCost, "")
      self.pPanel:Label_SetText(tbControls.txtWenyangLevel, "")
      self.pPanel:Label_SetText(tbControls.txtSuccessRate, "")
    end
    self.pPanel:Label_SetText(tbControls.txtFullSuitLevel, i18n.EasyGet("\233\149\135\229\178\179\229\165\151\232\163\133\239\188\136{0}/{1}\239\188\137", nPutOnLingZhuCount, Item.EQUIPPOS_MAIN_NUM))
    if nAllWYLevel > 0 then
      for i = 1, 3 do
        for j = 1, 3 do
          tbAttrValue[i][j] = tbAttrValue[i][j] + EquipCuiLing.tbCLSuitAttr[2][nAllWYLevel][i][j]
        end
      end
    end
    if nNextWYLevel > 0 then
      for i = 1, 3 do
        for j = 1, 3 do
          tbAttrNextValue[i][j] = tbAttrNextValue[i][j] + EquipCuiLing.tbCLSuitAttr[2][nNextWYLevel][i][j]
        end
      end
    end
    for i = 1, 3 do
      local szAttribMsg, szValueCur = FightSkill:GetMagicDescSplit(EquipCuiLing.tbCLSuitAttr[1][i][1], tbAttrValue[i])
      local _, szValueNext = FightSkill:GetMagicDescSplit(EquipCuiLing.tbCLSuitAttr[1][i][1], tbAttrNextValue[i])
      self.pPanel:Label_SetText(string.format("%s%d/txtAttribute", tbControls.AttributeFullSuit, i), szAttribMsg)
      local szSubValue = FightSkill:StringValueSubResult(szValueCur, szValueNext)
      if tbLingZhuData.nTemplateId > 0 and nPutOnLingZhuCount >= Item.EQUIPPOS_MAIN_NUM then
        self.pPanel:Label_SetColorByName(string.format("%s%d/txtValue", tbControls.AttributeFullSuit, i), "NewGreen")
        self.pPanel:Label_SetText(string.format("%s%d/txtValue", tbControls.AttributeFullSuit, i), szValueCur .. szSubValue)
      else
        self.pPanel:Label_SetColorByName(string.format("%s%d/txtValue", tbControls.AttributeFullSuit, i), "JingMaiGray")
        self.pPanel:Label_SetText(string.format("%s%d/txtValue", tbControls.AttributeFullSuit, i), szValueCur)
      end
    end
  else
    self.pPanel:SetActive(tbControls.panelFull, false)
    self.pPanel:SetActive(tbControls.btnDelete, false)
    self.pPanel:SetActive(tbControls.panelEmpty, true)
    self.pPanel:SetActive(tbControls.imgLock, true)
    self.tbUIItemGridLingZhu:SetEmpty(true)
    self.pPanel:Label_SetText(tbControls.txtEmptySuitLevel, i18n.EasyGet("\233\149\135\229\178\179\229\165\151\232\163\133\239\188\136{0}/{1}\239\188\137", nPutOnLingZhuCount, Item.EQUIPPOS_MAIN_NUM))
    for i = 1, 3 do
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(EquipCuiLing.tbCLSuitAttr[1][i][1], {
        table.unpack(EquipCuiLing.tbCLSuitAttr[1][i], 2)
      })
      self.pPanel:Label_SetText(string.format("%s%d/txtAttribute", tbControls.AttributeEmpthSuit, i), szAttribMsg)
      self.pPanel:Label_SetText(string.format("%s%d/txtValue", tbControls.AttributeEmpthSuit, i), szValue)
    end
  end
end

function tbWnd:OnCuiLingRsp()
  self:UpdateEquipDetailInfo()
end

function tbWnd:OnPutOnRsp()
  local tbElement = self.tbEquipScrollView:GetElement(self.nEquipIndex)
  if tbElement then
    tbElement.pPanel:SetActive("ImgSign", true)
  end
  self.tbOwner:ShowCuiLingInfo(nil, false)
  self:UpdateEquipDetailInfo()
end

function tbWnd:OnTakeOffRsp()
  local tbElement = self.tbEquipScrollView:GetElement(self.nEquipIndex)
  if tbElement then
    tbElement.pPanel:SetActive("ImgSign", false)
  end
  self:UpdateEquipDetailInfo()
end

function tbWnd:OnWenYangRsp()
  local tbElement = self.tbEquipScrollView:GetElement(self.nEquipIndex)
  if tbElement then
    local nEquipPos = self.tbSVData[self.nEquipIndex][1]
    local nWenYangLevel = EquipCuiLing:GetWenYangLevel(me, nEquipPos)
    tbElement.pPanel:Label_SetText("txtWenyangLevel/txtLevelValue", nWenYangLevel)
  end
  self:UpdateEquipDetailInfo()
end
