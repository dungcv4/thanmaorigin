local tbWnd = Ui:CreateClass("UIEquipSuitInfoSmall", "UIPanel")
local tbControls = {
  txtLevel = "imgBg/txtLevel",
  info = "imgBg/attrContent/info",
  txtTitle = "imgBg/txtTitle",
  attrContent = "imgBg/attrContent"
}
tbWnd.tbControls = tbControls

function tbWnd:UpdateView(pAsyncData)
  local PartnerEquipC = Item.PartnerEquip
  local nSuitAttrLevel = pAsyncData.GetPartnerEquipSuitLv()
  if nSuitAttrLevel <= 0 then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\229\165\151\232\163\133\229\177\158\230\128\167"))
    self.pPanel:Label_SetText(tbControls.txtLevel, i18n.EasyGet("\230\156\170\230\191\128\230\180\187"))
    self.pPanel:SetActive(tbControls.attrContent, false)
    return
  else
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\229\165\151\232\163\133\229\177\158\230\128\167{0}\231\186\167", nSuitAttrLevel))
    self.pPanel:Label_SetText(tbControls.txtLevel, i18n.EasyGet("{0}\231\186\167", nSuitAttrLevel))
    self.pPanel:SetActive(tbControls.attrContent, true)
  end
  local tbAllSuitAttr = PartnerEquipC:GetSuitAttrInfoByLevel(nSuitAttrLevel)
  if not self.tbAttrView then
    self.tbAttrView = Ui:NewScrollView(self, tbControls.info)
  end
  for nIndex, tbAttr in ipairs(tbAllSuitAttr or {}) do
    self.tbAttrView:SetElement(nIndex, function(tbElement)
      self:SetAttrElement(tbElement, tbAttr)
    end)
  end
end

function tbWnd:SetAttrElement(tbElement, tbAttr)
  local szAttrDesc, szAttrValue, _ = Item.PartnerEquip:GetEquipAttrDiff(tbAttr, nil)
  tbElement.pPanel:Label_SetText("txtAttrTip", szAttrDesc)
  tbElement.pPanel:Label_SetText("txtAttrValue", "+" .. szAttrValue)
end
