Require("Script/Fuben/FubenUIController.lua")
QianQiongGongMgr.nMainMapTemplateId = 352

function QianQiongGongMgr:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    self.nMainMapTemplateId
  }, self)
end

function QianQiongGongMgr:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenFuBenUI()
end

function QianQiongGongMgr:OnLeave()
  FubenUIController:CloseFubenUI()
end

function QianQiongGongMgr:IsInFubenMap()
  return true
end

function QianQiongGongMgr:UpdateShowInfo(szProgress, tbLastTime)
  FubenUIController:ShowInfo(szProgress, nil, tbLastTime)
end

function QianQiongGongMgr:OnReadyGo(nRestTime)
  if Lib:IsInteger(nRestTime) then
    Ui:OpenWindow("UIReadyGo", nRestTime)
  end
end

function QianQiongGongMgr:CreateTeam()
  if not TeamMgr:CanTeam(me.nMapTemplateId) then
    me.CenterMsg(i18n.Get(TeamMgr.Terms.MapLimitWhenJoin))
    return
  end
  
  local function fnOK()
    TeamMgr:CreateOnePersonTeam()
    TeamMgr:SetShowTargetPanel()
    if (Map:IsFieldFightMap(me.nMapTemplateId) or Map:IsCityMap(me.nMapTemplateId)) and 0 ~= me.nFightMode then
      return
    end
    local panel = Ui:GetPanelByName("UIHudLeftPanelTeam")
    panel:OnActivate()
  end
  
  me.MsgBox(i18n.Get(TeamMgr.Terms.CreateConfirm), {
    {
      i18n.szOk,
      fnOK
    },
    {
      i18n.szCancel
    }
  })
end

function QianQiongGongMgr:OnSyncFubenType(nFubenType)
  if 0 == nFubenType then
    self.nFubenType = nil
    return
  end
  self.nFubenType = nFubenType
end

function QianQiongGongMgr:UseEntranceItem()
  local nItemTid = self.Cfg.nItemId
  if not nItemTid then
    LogErr("[QianQiongGongMgr:UseEntranceItem] bad fuben type!")
    return
  end
  local pItem = me.GetOneItemByTypeInBag(nItemTid)
  if not pItem then
    me.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\228\184\141\232\182\179"))
    return
  end
  RemoteServer.UseItem(pItem.dwId)
end

function QianQiongGongMgr:GetMapName()
  if me.nMapTemplateId ~= self.nMainMapTemplateId then
    return
  end
  return i18n.EasyGet("\229\141\131\231\144\188\229\174\171")
end

QianQiongGongMgr:RegisterFubenUIChecker()
