Require("Script/EventSystem/EventNotify.lua")
FubenUIController.tbChecker = FubenUIController.tbChecker or {}

function FubenUIController:IsShowTipWhenQuit()
  local tbChecker = self.tbChecker[me.nMapTemplateId]
  if tbChecker then
    if tbChecker.IsShowTipWhenQuit then
      return tbChecker:IsShowTipWhenQuit()
    end
    return true
  end
  return true
end

function FubenUIController:GetShowTipMsg()
  local tbChecker = self.tbChecker[me.nMapTemplateId]
  if tbChecker and tbChecker.UICheckerGetShowTipMsg then
    return tbChecker:UICheckerGetShowTipMsg()
  end
  return i18n.EasyGet("\230\130\168\231\161\174\229\174\154\233\128\128\229\135\186\229\189\147\229\137\141\229\156\176\229\155\190\229\144\151\239\188\159")
end

function FubenUIController:IsShowBtnScore()
  local tbChecker = self.tbChecker[me.nMapTemplateId]
  if tbChecker then
    if tbChecker.IsShowBtnScore then
      return tbChecker:IsShowBtnScore(me.nMapTemplateId)

    return false

end

function FubenUIController:OnLeave(nTemplateID, nMapID)
  local tbChecker = self.tbChecker[nTemplateID]
  if tbChecker and tbChecker.OnLeave then
    tbChecker:OnLeave()
  end
end

function FubenUIController:IsInFubenMap(pPlayer)
  local tbChecker = self.tbChecker[pPlayer.nMapTemplateId]
  if tbChecker then
    if tbChecker.IsInFubenMap then
      return tbChecker:IsInFubenMap(pPlayer)
    else
      return true
    end
  end
  local szMapType = Map:GetClassDesc(pPlayer.nMapTemplateId)
  if "battle" == szMapType or "fuben" == szMapType or "boss" == szMapType then
    return true
  end
  return false
end

function FubenUIController:OnFubenMapLoaded(pPlayer)
  local tbChecker = self.tbChecker[pPlayer.nMapTemplateId]
  if tbChecker and tbChecker.OnFubenMapLoaded then
    tbChecker:OnFubenMapLoaded(pPlayer)
  end
end

function FubenUIController:RegisterChecker(tbMapIds, tbChecker)
  for _, nMapId in pairs(tbMapIds) do
    self.tbChecker[nMapId] = tbChecker
  end
end

function FubenUIController:UnRegisterChecker(tbMapIds, tbChecker)
  for _, nMapId in pairs(tbMapIds) do
    self.tbChecker[nMapId] = nil
  end
end

function FubenUIController:OpenFuBenUI()
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if tbWnd then
    tbWnd:SetPanelState(tbWnd.panelTaskTeamFuBen, tbWnd.TP_FUBEN)
  end
  Ui:OpenHudFubenCustomWithTeam("UIHudLeftPanelDungeon")
end

function FubenUIController:OpenBattleUI()
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if not tbWnd then
    return
  end
  tbWnd:SetPanelState(tbWnd.panelTaskTeamFuBen, tbWnd.TP_FUBEN)
  tbWnd:OpenFubenWithTeam("UIHudLeftPanelBattle")
end

function FubenUIController:CloseFubenUI()
  self.szCondition = nil
  self.szProgress = nil
  self.tbLastTime = nil
  self.szIntro = nil
  self.nReceiveTime = nil
  Ui:FinishHudFuben()
end

function FubenUIController:ShowInfo(szCondition, szProgress, tbLastTime, szIntro)
  self.szCondition = szCondition
  self.szProgress = szProgress
  self.tbLastTime = tbLastTime
  self.szIntro = szIntro
  self.nReceiveTime = GetTime()
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if tbWnd then
    tbWnd:UpdateCustomFubenData("UIHudLeftPanelDungeon")
  end
end

function FubenUIController:RequireQuit()
  if self:IsShowTipWhenQuit() then
    local szTip = self:GetShowTipMsg()
    me.MsgBox(szTip, {
      {
        i18n.szOk,
        function()
          Dungeon:ExitDungeon()
        end
      },
      {
        i18n.szCancel
      }
    })
  else
    Dungeon:ExitDungeon()
  end
end

function FubenUIController:ShowHudCountdown(nEndTime, szText)
  local tbWnd = Ui:GetWindow("UIHud")
  if tbWnd then
    tbWnd:SetCountdownNotice(nEndTime, szText)
  end
end

function FubenUIController:OpenDefaultFuBenUI(szFubenType, tbTimerParam, tbDefaultContent)
  if tbTimerParam then
    self.tbTimerParam = tbTimerParam
    for i, tbParam in ipairs(self.tbTimerParam) do
      if nil ~= tbParam[1] then
        tbParam[1] = tbParam[1] + GetTime()
      end
    end
  end
  if tbDefaultContent then
    self.tbDefaultContent = tbDefaultContent
  end
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if not tbWnd then
    return
  end
  if tbWnd then
    tbWnd:SetPanelState(tbWnd.panelTaskTeamFuBen, tbWnd.TP_FUBEN)
  end
  tbWnd:OpenFubenWithTeam("UIDefaultFubenInfo", szFubenType)
end

function FubenUIController:CloseDefaultFubenUI()
  self.tbTimerParam = nil
  self.tbDefaultContent = nil
  Ui:FinishHudFuben()
end

function FubenUIController:ShowDefaultInfo(tbTimerParam, tbDefaultContent)
  self.tbTimerParam = tbTimerParam
  if self.tbTimerParam ~= nil then
    for i, tbParam in ipairs(self.tbTimerParam) do
      if nil ~= tbParam[1] then
        tbParam[1] = tbParam[1] + GetTime()
      end
    end
  end
  self.tbDefaultContent = tbDefaultContent
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if tbWnd then
    tbWnd:UpdateCustomFubenData("UIDefaultFubenInfo")
  end
end

function FubenUIController:ShowDefaultContentIndexInfo(nIndex, szContent)
  if self.tbDefaultContent and self.tbDefaultContent[nIndex] then
    self.tbDefaultContent[nIndex] = szContent
  end
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if tbWnd then
    tbWnd:UpdateContentData("UIDefaultFubenInfo")
  end
end

function FubenUIController:RegisterDefaultFubenPanelClickEvent(fnEvent, ...)
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if nil ~= tbWnd then
    local fubenSubPanel = tbWnd:GetSubPanel("UIHudLeftPanelFuben")
    if nil ~= fubenSubPanel then
      local defaultFubenSubPanel = fubenSubPanel:GetSubPanel("UIDefaultFubenInfo")
      if defaultFubenSubPanel and defaultFubenSubPanel.RegisterPanelClickEvent then
        defaultFubenSubPanel:RegisterPanelClickEvent(fnEvent, ...)
      end
    end
  end
end

function FubenUIController:UnRegisterDefaultFubenPanelClickEvent()
  local tbWnd = Ui:GetWindow("UIHudLeftFrameExt")
  if nil ~= tbWnd then
    local fubenSubPanel = tbWnd:GetSubPanel("UIHudLeftPanelFuben")
    if nil ~= fubenSubPanel then
      local defaultFubenSubPanel = fubenSubPanel:GetSubPanel("UIDefaultFubenInfo")
      if defaultFubenSubPanel and defaultFubenSubPanel.RegisterPanelClickEvent then
        defaultFubenSubPanel:UnRegisterPanelClickEvent()
      end
    end
  end
end

function FubenUIController:ShowRightCenterTimer(tbParam)
  self.tbRightCenterParam = nil
  if tbParam then
    self.tbRightCenterParam = {}
    self.tbRightCenterParam.szTitle = tbParam[1]
    self.tbRightCenterParam.nTime = tbParam[2] + GetTime()
    self:SetPlayerGuideCountDownShow(true)
  else
    self:SetPlayerGuideCountDownShow(false)
  end
end

function FubenUIController:SetPlayerGuideCountDownShow(bShow)
  self.bPlayerGuideCountDownShow = bShow
end

function FubenUIController:OnPlayerLogin(nReconnect)
  if 1 == nReconnect then
    self:OnFubenMapLoaded(me)
  end
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_MAP_LEAVE, FubenUIController.OnLeave, FubenUIController)
EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, FubenUIController.OnPlayerLogin, FubenUIController)
EventNotify:RegistNotify(EventNotify.emNOTIFY_ONSHOW_ACTIVITY_REMAINTIME, FubenUIController.SetPlayerGuideCountDownShow, FubenUIController)
