Dungeon.szFilePath = "Setting/Dungeon/DungeonResult/ScoreLevel.tab"
Dungeon.tbGradeLevel = {}
Dungeon.tbAddScore = {
  nKinScore = 2,
  tbFriendScore = {
    {10, 4},
    {1, 2}
  }
}
Dungeon.tbGradeLevel = Dungeon.tbGradeLevel or {}

function Dungeon:GetGradeInfo(nScore)
  if not nScore or nScore < 0 then
    return
  end
  local nGradeIndex = 0
  for i = 1, #self.tbGradeLevel do
    if nScore >= self.tbGradeLevel[i].nScore then
      nGradeIndex = i
      break
    end
  end
  if nGradeIndex > 1 then
    return self.tbGradeLevel[nGradeIndex], self.tbGradeLevel[nGradeIndex - 1]
  else
    return self.tbGradeLevel[nGradeIndex]
  end
end

function Dungeon:GetFriendScore(nImity)
  if nImity < 1 then
    return 0
  end
  for _, tbScore in pairs(self.tbAddScore.tbFriendScore) do
    if nImity >= tbScore[1] then
      return tbScore[2]
    end
  end
end

function Dungeon:LoadConfig()
  local tbTmpConfig = Lib:EasyLoadTabFile(self.szFilePath, {"nLevel"})
  for nLevel, tbField in pairs(tbTmpConfig) do
    if not self.tbGradeLevel[nLevel] then
      self.tbGradeLevel[nLevel] = {
        nLevel = nLevel,
        nScore = tbField.nScore,
        szGrade = tbField.szGrade
      }
    end
  end
end

function Dungeon:GetLevelScore(nLevel)
  local tbSetting = self.tbGradeLevel[nLevel]
  if not tbSetting then
    return 0
  end
  return tbSetting.nScore
end

function Dungeon:GetLevelDesc(nLevel)
  local tbSetting = self.tbGradeLevel[nLevel]
  if not tbSetting then
    return ""
  end
  return tbSetting.szGrade
end

Dungeon:LoadConfig()
