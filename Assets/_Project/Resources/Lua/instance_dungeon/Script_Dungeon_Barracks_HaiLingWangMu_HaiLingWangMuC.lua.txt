Require("CommonScript/BeachFight/BeachFightCfg.lua")
HaiLingWangMu.bIsCanExit = true

function HaiLingWangMu:OnSetTargetPosResp(nPosX, nPosY)

end

end

function HaiLingWangMu:OnChangeNewTargetResp(szTargetType)
  if HaiLingWangMu.Cfg.TargetCfg[szTargetType] then
    local szMsg = HaiLingWangMu.Cfg.TargetCfg[szTargetType].szBeforeMsg
    if not Lib:IsEmptyStr(szMsg) then
      me.CenterMsg(i18n.Get(szMsg))
    end
  end
end

function HaiLingWangMu:OnFinishTargetResp(szTargetType)
  if HaiLingWangMu.Cfg.TargetCfg[szTargetType] then
    local szMsg = HaiLingWangMu.Cfg.TargetCfg[szTargetType].szFinishMsg
    if not Lib:IsEmptyStr(szMsg) then
      me.CenterMsg(i18n.Get(szMsg))
    end
  end
end

    i18n.EasyGet("\231\173\137\229\190\133\229\137\175\230\156\172\229\188\128\229\167\139")
  })
end

  end

end

function HaiLingWangMu:IsNeedExitFubenButton()
  return self.bIsCanExit
end

function HaiLingWangMu:SetCanExit(bCanExit)
  self.IsCanExit = bCanExit

end

function HaiLingWangMu:OnJoin()
  local function fnOpenTeamWindowShowBarrack()
    Ui:OpenWindow("UITeam")
    
    local tbWnd = Ui:GetWindow("UITeam")
    if not tbWnd:SelectBigAndSmallButton("Barracks", "HaiLingWangMu") then
      LogErr("Barracks big button or HaiLingWangMu small button is not exsit in UITeam left")
    end
  end
  
  if not TeamMgr:HasTeam() then
    Ui:CloseWindow("UIActivity")
    fnOpenTeamWindowShowBarrack()
  elseif not TeamMgr:IsCaptain() then
    me.CenterMsg(TeamMgr.OnlyCaptainCanJoinHintMsg())
  elseif not TeamMgr:TeamFull() then
    Ui:CloseWindow("UIActivity")
    Ui:OpenWindow("UITeam")
    TeamMgr:SetTeamActivity(TeamMgr:GetActivityId("Barracks", "HaiLingWangMu"))
    local tbWnd = Ui:GetWindow("UITeam")
    tbWnd:SelectBigButton("myTeam")
  else
    Ui:CloseWindow("UIActivity")
    local nActivityId = TeamMgr:GetActivityId("Barracks", "HaiLingWangMu")
    if nil ~= nActivityId then
      TeamMgr:AutoSetAndEnterActivity(nActivityId)
    end
  end
end

function HaiLingWangMu:GoToNpc()
  if not self.nTargetNpcId then
    return
  end
  if TeamMgr:IsFollowTeammate() then
    return
  end
  if AutoPath:IsPathing() then
    return
  end
  if Ui:WindowVisible("UIProgressBar") then
    return
  end
  if Ui:WindowVisible("UINpcDialog") then
    return
  end
  if not self.bCheckFilg then
    return
  end
  self.bCheckFilg = false
  local pNpc = KNpc.GetNearestNpcByTemplateId(me.GetNpc(), self.nTargetNpcId, AFK:GetSetting():GetFightSetting().nSearchRadius)
  if pNpc then
    Operation:SimpleTap(pNpc.nId)
  end
end

function HaiLingWangMu:OnStartAutoTalkNpc(nTempNpcId)
  if not nTempNpcId then
    return false
  end
  self.nTargetNpcId = nTempNpcId
  self.nAutoTalkNpcTimer = Timer:Register(Env.GAME_FPS * self.nAutoTalkNpcTime, function()
    self.bCheckFilg = true
    self:GoToNpc()
    return true
  end)
end

function HaiLingWangMu:OnCloseAutoTalkNpcTimer()
  if self.nAutoTalkNpcTimer then
    Timer:Close(self.nAutoTalkNpcTimer)
    self.nAutoTalkNpcTimer = nil
  end
end

function HaiLingWangMu:OnAFKStateChange(bAFKState)
  if bAFKState == AFK.LeadFightStateTB.emStop then
    self:OnCloseAutoTalkNpcTimer()
  elseif bAFKState == AFK.LeadFightStateTB.emAutoFight then
    self:OnStartAutoTalkNpc(self.nTargetNpcId)
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_AFK_STATE, HaiLingWangMu)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_AFK_STATE, HaiLingWangMu.OnAFKStateChange, HaiLingWangMu)
