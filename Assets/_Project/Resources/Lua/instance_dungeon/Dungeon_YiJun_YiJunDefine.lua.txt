YiJun = YiJun or {}
YiJun.nLevelLimit = 5
YiJun.tbMainMapPos = {17445, 23590}
YiJun.fileNpcSetting = "Setting/Map/taozhugongyizhong/info/batch_npc_setting.txt"
YiJun.fileAutoSetting = "Setting/Map/taozhugongyizhong/info/batch_auto_setting.txt"
YiJun.tbNpcSetting = nil
YiJun.nTotalTime = 600
YiJun.tbGouhuoNpcPos = {45411, 59736}
YiJun.tbGouhuoNpcParam = {
  30,
  667,
  5
}
YiJun.nAwardId = 801
YiJun.nPassAwardKey = 1
YiJun.tbTimeScore = {
  {300, 100},
  {360, 90},
  {420, 80},
  {480, 70},
  {600, 60},
  {10000, 50}
}

function YiJun:CheckCanShow()
  if me.nLevel < YiJun.nLevelLimit then
    return false
  end
  return true
end

if not YiJun.tbNpcSetting then
  YiJun.tbNpcSetting = {}
  local tbNpcSetting = {}
  tbNpcSetting = Lib:LoadTabFileEx(YiJun.fileNpcSetting, "dddddd", nil, {
    "Batch",
    "NpcType",
    "NpcTemplateId",
    "PosX",
    "PosY",
    "Sequence"
  }, nil, nil, false)
  if not tbNpcSetting then
    error("cannot read" .. YiJun.fileNpcSetting)
    return
  end
  for _, tbNpcInfo in ipairs(tbNpcSetting) do
    if not YiJun.tbNpcSetting[tbNpcInfo.Batch] then
      YiJun.tbNpcSetting[tbNpcInfo.Batch] = {}
    end
    table.insert(YiJun.tbNpcSetting[tbNpcInfo.Batch], {
      tbNpcInfo.NpcType,
      tbNpcInfo.NpcTemplateId,
      tbNpcInfo.PosX,
      tbNpcInfo.PosY,
      tbNpcInfo.Sequence
    })
  end
end

function YiJun:InitAutoCfg()
  if not self.tbAutoSetting then
    self.tbAutoSetting = {}
    local tbAutoSetting = {}
    tbAutoSetting = Lib:LoadTabFileEx(YiJun.fileAutoSetting, "ddd", nil, {
      "Batch",
      "PosX",
      "PosY"
    }, nil, nil, false)
    if not tbAutoSetting then
      error("cannot read" .. YiJun.fileAutoSetting)
      return
    end
    for _, tbAutoInfo in ipairs(tbAutoSetting) do
      if not self.tbAutoSetting[tbAutoInfo.Batch] then
        self.tbAutoSetting[tbAutoInfo.Batch] = {}
      end
      local tbAutoData = {}
      tbAutoData.nPosX = tbAutoInfo.PosX
      tbAutoData.nPosY = tbAutoInfo.PosY
      self.tbAutoSetting[tbAutoInfo.Batch] = tbAutoData
    end
  end
end

YiJun:InitAutoCfg()
