Require("Script/Fuben/FubenUIController.lua")

function KinLevel:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    KinLevel.nMainMapTemplateId
  }, self)
end

function KinLevel:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenFuBenUI()
end

function KinLevel:OnLeave()
  FubenUIController:CloseFubenUI()
end

function KinLevel:IsInFubenMap()
  return true
end

function KinLevel:IsNeedExitFubenButton()
  return self.IsCanExit or false
end

function KinLevel:SetCanExit(bCanExit)
  self.IsCanExit = bCanExit
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function KinLevel:EnterKinLevelByCaptainReq(nTargetMapId, nTargetMapTemplateId)
  RemoteServer.OnKinLevelRequest(KinLevel.Protocol.C2SJoinFubenReq, nTargetMapId)
end

function KinLevel:EnterKinLevelByCaptainRes(bIsCanEnter)
  if false == bIsCanEnter and TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
end

function KinLevel:Init()
  if not self.bInit then
    self:RegisterFubenUIChecker()
    self.bInit = true
    self.IsCanExit = false
    local tbKinLevelMapClass = Map:GetClass(KinLevel.nMainMapTemplateId)
    
    function tbKinLevelMapClass.TransmitFrom(tbKinLevelMapClass, nFromMapId, nFromTemplateId, nTargetMapId, nTargetMapTemplateId)
      return KinLevel:EnterKinLevelByCaptainReq(nTargetMapId, nTargetMapTemplateId)
    end
  end
end

function KinLevel:RequestSignKinLevel()
  local function func()
    RemoteServer.OnKinLevelRequest(KinLevel.Protocol.C2SSignUp)
  end
  
  local bRet, szMsg = ActivityCheck:CheckCanEnterActivityEx(me, nil, func, true, false, true, nil, {}, false)
  if not bRet then
    me.CenterMsg(szMsg)
  end
end

KinLevel:Init()
