Require("CommonScript/Dungeon/LouLan/LoulanCfg.lua")
Loulan.nMainMapTemplateId = 4
Loulan.nBossMapTemplateId = 5

function Loulan:CheckCanEnter(pPlayer)
  if not TeamMgr:HasTeam(pPlayer) then
    local szMsg = i18n.Get(Loulan.Msg.ERR_NOTEAM)
    return false, szMsg
  end
  if not TeamMgr:IsCaptain(pPlayer) then
    return false, TeamMgr.OnlyCaptainCanJoinHintMsg()
  end
  local nTeamId = pPlayer.dwTeamID
  local tbMembers = TeamMgr:GetMembers(nTeamId)
  local nLimitMember = Loulan.Def.nMinMember - 1
  if MODULE_GAMESERVER then
    nLimitMember = Loulan.Def.nMinMember
  else
  end
  if nLimitMember > #tbMembers then
    local szMsg = i18n.Get(Loulan.Msg.ERR_LESSMEMBER, Loulan.Def.nMinMember)
    return false, szMsg
  end
  local bMemberCanEnter, szMsg = self:CheckMemberEnter(tbMembers)
  return bMemberCanEnter, szMsg
end

function Loulan:CheckMemberEnter(tbMembers)
  if not tbMembers then
    return false
  end
  for index, value in ipairs(tbMembers) do
    local dwPlayerId = value
    if not MODULE_GAMESERVER then
      dwPlayerId = value.nPlayerID
    end
    local pMember = KPlayer.GetPlayerObjById(dwPlayerId)
    if pMember then
      if pMember.nLevel < Loulan.Def.nOpenLevel then
        local szMsg = i18n.Get(Loulan.Msg.ERR_LEVELLIMIT_MEMBER, pMember.szName, Loulan.Def.nOpenLevel)
        return false, szMsg
      end
      if DegreeCtrl:GetDegree(pMember, "Loulan") < 1 then
        local szMsg = i18n.Get(Loulan.Msg.ERR_NODEGREE_MEMBER, pMember.szName)
        return false, szMsg
      end
      local bRet, szErr = ActivityCheck:CheckCanEnterActivityEx(pMember, nil, nil, false, false, true, nil, nil, false)
      if not bRet then
        return false, i18n.EasyGet("\233\152\159\229\145\152{0}", pMember.szName) .. szErr
      end
    end
  end
  return true
end

function Loulan:GetCurCfgLevel()
  local nLevel = 0
  local nServerMaxLevel = GetMaxTotalLevel()
  for nMaxLevel, tbInfo in pairs(Loulan.tbNpcLvSetting) do
    if nMaxLevel <= nServerMaxLevel and nMaxLevel > nLevel then
      nLevel = nMaxLevel
    end
  end
  return nLevel
end

function Loulan:CheckCanShow()
  return me.nLevel >= Loulan.Def.nOpenLevel
end
