Require("Script/Fuben/FubenUIController.lua")
FuNiuMountain = FuNiuMountain or {}
FuNiuMountain.bIsCanExit = false

function FuNiuMountain:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    self.nMainMapTemplateId
  }, self)
end

function FuNiuMountain:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenDefaultFuBenUI()
  self.bIsCanExit = false
  RemoteServer.FuNiuMountainSendProcessInfo()
end

function FuNiuMountain:OnLeave()
  FubenUIController:CloseFubenUI()
  self.bIsCanExit = false
end

function FuNiuMountain:IsNeedExitFubenButton()
  return self.bIsCanExit
end

FuNiuMountain:RegisterFubenUIChecker()

function FuNiuMountain:ReceiveServer(nProtocol, ...)
  if nProtocol == FuNiuMountain.S2CProtocol.S2CUpdateProcessBatch then
    self:OnProcessBatch(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2COnBatchStart then
    self:OnBatchStart(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2COnBatchEnd then
    self:OnBatchEnd(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2COnUpdateSchedule then
    self:OnUpdateSchedule(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2COnFinishDungeon then
    self:OnFinishDungeon(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2COnTrapPoint then
    self:OnTrapPoint(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2COnStart then
    self:OnStart(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2SetTargetPos then
    Dungeon:SetTargetPos(...)
  elseif nProtocol == FuNiuMountain.S2CProtocol.S2ClearTargetPos then
    Dungeon:ClearTargetPos()
  end
end

function FuNiuMountain:OnProcessBatch(nStepEnum)
  if nStepEnum == FuNiuMountain.tbEachStepEnum.GatherMaple then
    me.CenterMsg(i18n.Get(self.tbText.CenterMsgGetAMaple))
  elseif nStepEnum == FuNiuMountain.tbEachStepEnum.OpenGate then
    me.CenterMsg(i18n.Get(self.tbText.CenterMsgInOpenGate))
  elseif nStepEnum == FuNiuMountain.tbEachStepEnum.GetKnives then
    me.CenterMsg(i18n.Get(self.tbText.CenterMsgInGetKnives))
  end
end

function FuNiuMountain:OnBatchStart(nStepEnum)
  local szText
  if nStepEnum == FuNiuMountain.tbEachStepEnum.ThroughMine then
    szText = i18n.Get(self.tbText.CenterMsgStartThroughMine)
  elseif nStepEnum == FuNiuMountain.tbEachStepEnum.TakeBoat then
    szText = i18n.Get(self.tbText.CenterMsgStartTakeBoat)
  elseif nStepEnum == FuNiuMountain.tbEachStepEnum.DefeatGrandArtisan then
    szText = i18n.Get(self.tbText.CenterMsgStartGrandArtisan)
  end
  if nil ~= szText then
    me.CenterMsg(szText)
  end
end

function FuNiuMountain:OnBatchEnd(nStepEnum)
  local szText
  if nStepEnum == FuNiuMountain.tbEachStepEnum.GatherMaple then
    szText = i18n.Get(self.tbText.CenterMsgEndGatherMaple)
  end
  if nil ~= szText then
    me.CenterMsg(szText)
  end
end

function FuNiuMountain:OnUpdateSchedule(nStepEnum, tbEachBatchFinishSchedule, nLeftTime)
  self:UpdateLeftPanelUI(nStepEnum, tbEachBatchFinishSchedule, nLeftTime)
end

function FuNiuMountain:OnFinishDungeon()
  self:UpdateUIFinishState(FuNiuMountain.nFromEndToQuitTime)
end

function FuNiuMountain:OnTrapPoint(nTrapPointIndex, bTouchTeleport)
  local szText
  if bTouchTeleport then
    if nTrapPointIndex == FuNiuMountain.tbTrapPoint.Gate1 or nTrapPointIndex == FuNiuMountain.tbTrapPoint.Gate2 then
      szText = i18n.Get(self.tbText.CenterMsgNotOpenGate)
    elseif nTrapPointIndex == FuNiuMountain.tbTrapPoint.LeaveMapleForest then
      szText = i18n.Get(self.tbText.CenterMsgNotGatherEnough, self.tbEachBatchCounter[self.tbEachStepEnum.GatherMaple])
    end
  elseif nTrapPointIndex == FuNiuMountain.tbTrapPoint.MineEntrance then
    szText = i18n.Get(self.tbText.CenterMsgNotKillMinion)
  end
  if nil ~= szText then
    me.CenterMsg(szText)
  end
end

function FuNiuMountain:OnStart()
  me.CenterMsg(i18n.Get(self.tbText.CenterMsgAndTeamEnterDungeon))
end

function FuNiuMountain:UpdateUIFinishState(nRestTime)
  local tbTimeInfo
  if nil ~= nRestTime then
    tbTimeInfo = {
      nRestTime,
      i18n.Get(self.tbText.LeftUIEndStageRestTime)
    }
  else
    tbTimeInfo = {}
  end
  FubenUIController:ShowDefaultInfo({tbTimeInfo}, {
    i18n.Get(self.tbText.LeftUIEnd)
  })
  self.bIsCanExit = true
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function FuNiuMountain:UpdateUIBeforeStartState()
  self.bIsCanExit = false
  FubenUIController:ShowDefaultInfo(nil, {
    i18n.Get(self.tbText.LeftUIStart)
  })
end

function FuNiuMountain:UpdateLeftPanelUI(nStepEnum, tbEachBatchFinishSchedule, nLeftTime)
  if nStepEnum == FuNiuMountain.tbEachStepEnum.Start then
    self:UpdateUIBeforeStartState()
  elseif nStepEnum == FuNiuMountain.tbEachStepEnum.End then
    self:UpdateUIFinishState(nLeftTime)
  else
    local tbTexts
    if nStepEnum == self.tbEachStepEnum.GatherMaple then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIGatherMaple, tbEachBatchFinishSchedule[nStepEnum], self.tbEachBatchCounter[nStepEnum])
      }
    elseif nStepEnum == self.tbEachStepEnum.DefeatMinions then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIDefeatMinions)
      }
    elseif nStepEnum == self.tbEachStepEnum.ThroughMine then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIThroughMine)
      }
    elseif nStepEnum == self.tbEachStepEnum.OpenGate then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIOpenGate, tbEachBatchFinishSchedule[nStepEnum], self.tbEachBatchCounter[nStepEnum])
      }
    elseif nStepEnum == self.tbEachStepEnum.KillOverwatchChief or nStepEnum == self.tbEachStepEnum.KillOverwatch then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIKillOverwatch, tbEachBatchFinishSchedule[self.tbEachStepEnum.KillOverwatch], self.tbEachBatchCounter[self.tbEachStepEnum.KillOverwatch]),
        i18n.Get(self.tbText.LeftUIKillOverwatchChief, tbEachBatchFinishSchedule[self.tbEachStepEnum.KillOverwatchChief], self.tbEachBatchCounter[self.tbEachStepEnum.KillOverwatchChief])
      }
    elseif nStepEnum == self.tbEachStepEnum.TakeBoat then
      tbTexts = {
        i18n.Get(self.tbText.LeftUITakeBoat)
      }
    elseif nStepEnum == self.tbEachStepEnum.GetKnives then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIGetKnives, tbEachBatchFinishSchedule[nStepEnum], self.tbEachBatchCounter[nStepEnum])
      }
    elseif nStepEnum == self.tbEachStepEnum.TurnOnStoneCutter then
      tbTexts = {
        i18n.Get(self.tbText.LeftUITurnOnStoneCutter)
      }
    elseif nStepEnum == self.tbEachStepEnum.DefeatGrandArtisan then
      tbTexts = {
        i18n.Get(self.tbText.LeftUIDefeatGrandArtisan)
      }
    end
    if nil == tbTexts then
      tbTexts = {}
    end
    if #tbTexts >= 1 then
      tbTexts[1] = i18n.Get(self.tbText.LeftUICurrentGoal) .. tbTexts[1]
    end
    local tbTimerParam
    if nil ~= nLeftTime then
      tbTimerParam = {
        {
          nLeftTime,
          i18n.Get(self.tbText.LeftUIFormalStageRestTime)
        }
      }
    else
      tbTimerParam = {}
    end
    FubenUIController:ShowDefaultInfo(tbTimerParam, tbTexts)
  end
end

function FuNiuMountain:OnJoin()
  local function fnOpenTeamWindowShowBarrack()
    Ui:OpenWindowWithCallback("UITeam", function(_, tbWnd)
      if not tbWnd:SelectBigAndSmallButton("Barracks", "FuNiuMountain") then
        LogErr("Barracks big button or Funiumountain small button is not exsit in UITeam left")
      end
    end)
  end
  
  if not TeamMgr:HasTeam() then
    Ui:CloseWindow("UIActivity")
    fnOpenTeamWindowShowBarrack()
  elseif not TeamMgr:IsCaptain() then
    me.CenterMsg(TeamMgr.OnlyCaptainCanJoinHintMsg())
  elseif not TeamMgr:TeamFull() then
    Ui:CloseWindow("UIActivity")
    Ui:OpenWindowWithCallback("UITeam", function(_, tbWnd)
      TeamMgr:SetTeamActivity(TeamMgr:GetActivityId("Barracks", "FuNiuMountain"))
      tbWnd:SelectBigButton("myTeam")
    end)
  else
    Ui:CloseWindow("UIActivity")
    local nActivityId = TeamMgr:GetActivityId("Barracks", "FuNiuMountain")
    if nil ~= nActivityId then
      TeamMgr:AutoSetAndEnterActivity(nActivityId)
    end
  end
end
