function pFubenMgrC:IsNeedStateCancelBtn()
  return self.bHenshin, {
    function()
      self:RequestHelper("CancelHenshinReq")
    end
  }, self.nHenshinBuffLeftTime
end

function pFubenMgrC:IsInGuardXiangYang()
  return me.nMapTemplateId == self.Cfg.nMapTid
end

function pFubenMgrC:_UpdateOwnInfo(nSocre, nRank)
  self.tbOwnScoreInfo = {nRank = nRank, nSocre = nSocre}
end

function pFubenMgrC:DoEnterFuben()
  local bRet, szMsg = ActivityCheck:CheckCanEnterActivityEx(me, nil, nil, true, false, true, nil, {}, false)
  if not bRet then
    me.CenterMsg(szMsg)
    return
  end
  if not Tong:HasTong() then
    me.CenterMsg(i18n.EasyGet("\228\189\160\230\178\161\230\156\137\229\138\160\229\133\165\229\184\174\228\188\154"))
    return
  end
  
  local function fnOk()
    TeamMgr:Quite()
    self:RequestHelper("EnterMapReq")
  end
  
  if TeamMgr:HasTeam() and not TeamMgr:IsCaptain(me.dwID) then
    me.MsgBox(i18n.EasyGet("\230\152\175\229\144\166\231\166\187\229\188\128\233\152\159\228\188\141\239\188\140\229\143\130\229\138\160\229\184\174\230\180\190\229\137\175\230\156\172\230\180\187\229\138\168\239\188\159"), {
      {
        i18n.szOk,
        function()
          fnOk()
        end
      },
      {
        i18n.szCancel
      }
    })
    return
  end
  self:RequestHelper("EnterMapReq")
end

function pFubenMgrC:RequestHelper(szProtocol)
  self.tbReqTimeMap = self.tbReqTimeMap or {}
  local nLastReqTsp = self.tbReqTimeMap[szProtocol]
  if not nLastReqTsp or nLastReqTsp and GetTime() >= nLastReqTsp + 1 then
    self.tbReqTimeMap[szProtocol] = GetTime()
    RemoteServer.RequestGuardXiangYang(szProtocol)
    return
  end
  me.CenterMsg(i18n.EasyGet("\230\147\141\228\189\156\229\164\170\233\162\145\231\185\129\228\186\134\239\188\140\232\175\183\231\168\141\229\144\142\229\134\141\232\175\149"))
end

function pFubenMgrC:IsNeedExitFubenButton()

end

function pFubenMgrC:IsShowBtnScore()
  if me.nMapTemplateId == self.Cfg.nMapTid then
    return true, function()
      RemoteServer.RequestGuardXiangYang("GetRankInfoReq")
    end
  end
  return false
end

function pFubenMgrC:ReqRankInfo()
  RemoteServer.RequestGuardXiangYang("GetRankInfoReq")
end

function pFubenMgrC:OnSyncRankInfo(tbRankData)
  tbRankData.tbOwnRank = {
    szName = me.szName,
    nScore = self.tbOwnScoreInfo.nSocre or 0,
    nRank = self.tbOwnScoreInfo.nRank or 0
  }
  if Ui:WindowVisible("UIHudOutputInfo") then
    Ui:GetWindow("UIHudOutputInfo"):UpdateRank(2, tbRankData)
  end
end
