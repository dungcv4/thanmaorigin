Require("Script/Fuben/FubenUIController.lua")

function YiJun:OnJoin()
  if not TeamMgr:HasTeam() then
    Ui:OpenWindow("UITeam", TeamMgr.ACTIVITY_ID.YiJun)
    return
  elseif not TeamMgr:IsCaptain() then
    me.CenterMsg(TeamMgr.OnlyCaptainCanJoinHintMsg())
    return
  end
  local nActivityId = TeamMgr:GetActivityId("YiJun", "YiJun")
  if TeamMgr:HasTeam() and TeamMgr:IsCaptain() and TeamMgr:TeamFull() and nil ~= nActivityId then
    TeamMgr:AutoSetAndEnterActivity(nActivityId)
    return
  end
  Ui:OpenWindowWithCallback("UITeam", function(_, tbWnd)
    TeamMgr:SetTeamActivity(nActivityId)
    tbWnd:SelectBigButton("myTeam")
  end)
end

function YiJun:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    self.nMainMapTemplateId
  }, self)
end

function YiJun:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenFuBenUI()
end

function YiJun:OnLeave()
  FubenUIController:CloseFubenUI()
  self.IsCanExit = nil
end

function YiJun:SetCanExit(bCanExit)
  self.IsCanExit = bCanExit
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function YiJun:IsNeedExitFubenButton()
  return self.IsCanExit or false
end

YiJun:RegisterFubenUIChecker()
