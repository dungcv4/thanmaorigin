Require("Script/Fuben/FubenUIController.lua")

function Loulan:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    self.nMainMapTemplateId,
    self.nBossMapTemplateId
  }, self)
end

function Loulan:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenFuBenUI()
end

function Loulan:OnLeave()
  FubenUIController:CloseFubenUI()
  self.IsCanExit = nil
end

function Loulan:IsNeedExitFubenButton()
  return self.IsCanExit or false
end

function Loulan:RequieExit()
  FubenUIController:RequireQuit()
end

function Loulan:SetCanExit(bCanExit)
  self.IsCanExit = bCanExit
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

Loulan:RegisterFubenUIChecker()

function Loulan:StartFollow()
  local tbCaptain = TeamMgr:GetCaptainData()
  if tbCaptain then
    TeamMgr:StartFollowMember(tbCaptain.nPlayerID, tbCaptain.nNpcID)
  end
end

function Loulan:ShowTeamUI()
  if not TeamMgr:HasTeam() then
    Ui:OpenWindow("UITeam", TeamMgr.ACTIVITY_ID.LouLan)
    return
  elseif not TeamMgr:IsCaptain(me.dwID) then
    me.CenterMsg(TeamMgr.OnlyCaptainCanJoinHintMsg())
    return
  end
  local tbMembers = TeamMgr:GetMembers()
  local nActivityId = TeamMgr:GetActivityId("LouLan", "LouLan")
  if TeamMgr:HasTeam() then
    TeamMgr:SetTeamActivity(nActivityId)
    Ui:OpenWindow("UITeam")
  else
    if not TeamMgr:CanTeam(me.nMapTemplateId) then
      me.CenterMsg(i18n.Get(TeamMgr.Terms.MapLimitWhenJoin))
      return
    end
    TeamMgr:CreateOnePersonTeam(nActivityId)
    Ui:OpenWindow("UITeam")
  end
end
