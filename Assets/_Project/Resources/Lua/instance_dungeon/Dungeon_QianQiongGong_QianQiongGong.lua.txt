local QianQiongGongComm = QianQiongGongMgr or {}

function QianQiongGongComm:LoadConfig()
  self.tbNpcSetting = self.tbNpcSetting or {}
  self.tbGuanQiaNpcSetting = self.tbGuanQiaNpcSetting or {}
  local szObstaclePos = Map:GetMapPosFile(self.Cfg.nFubenTemplateId, "obstacle_pos.txt")
  local szNpcSettingFile = Map:GetMapPosFile(self.Cfg.nFubenTemplateId, "phasenpc_info.txt")
  local szNpcGuanQiaSettingFile = Map:GetMapPosFile(self.Cfg.nFubenTemplateId, "barriernpc_pos.txt")
  local szHusongNpcPathFile = Map:GetMapPosFile(self.Cfg.nFubenTemplateId, "husong_npc_path.txt")
  local tbSetting = Lib:LoadTabFileEx(szNpcSettingFile, "dddd", nil, {
    "NpcTemplateId",
    "PosX",
    "PosY",
    "nPhase"
  }, nil, nil, false)
  if not tbSetting then
    LogErr("[QianQiongGongMgr::LoadConfig] load failed!", szNpcSettingFile)
    return
  end
  local tbGuanQiaSetting = Lib:LoadTabFileEx(szNpcGuanQiaSettingFile, "dddddd", nil, {
    "NpcTemplateId",
    "PosX",
    "PosY",
    "ObstId",
    "nPhase",
    "Direction"
  }, nil, nil, false)
  if not tbGuanQiaSetting then
    LogErr("[QianQiongGongMgr::LoadConfig] load failed!", szNpcGuanQiaSettingFile)
    return
  end
  local tbHusongNpcSetting = Lib:LoadTabFileEx(szHusongNpcPathFile, "dddd", nil, {
    "Path",
    "Index",
    "PosX",
    "PosY"
  }, nil, nil, false)
  if not tbHusongNpcSetting then
    LogErr("[QianQiongGongMgr::LoadConfig] load failed!", szHusongNpcPathFile)
    return
  end
  local tbObstaclePos = Lib:LoadTabFileEx(szObstaclePos, "sddd", nil, {
    "szTrapName",
    "PosX",
    "PosY",
    "Phase"
  }, nil, nil, false)
  if not tbObstaclePos then
    LogErr("[QianQiongGongMgr::LoadConfig] load failed!", szHusongNpcPathFile)
    return
  end
  local tbTempNpcSetting = {}
  for _, tbValue in pairs(tbSetting) do
    if not tbTempNpcSetting[tbValue.nPhase] then
      tbTempNpcSetting[tbValue.nPhase] = {}
    end
    table.insert(tbTempNpcSetting[tbValue.nPhase], tbValue)
  end
  local tbTemptbGuanQiaSetting = {}
  for _, tbValue in pairs(tbGuanQiaSetting) do
    if not tbTemptbGuanQiaSetting[tbValue.nPhase] then
      tbTemptbGuanQiaSetting[tbValue.nPhase] = {}
    end
    table.insert(tbTemptbGuanQiaSetting[tbValue.nPhase], tbValue)
  end
  local tb = {}
  for nIndex, tbPosInfo in pairs(tbHusongNpcSetting) do
    if tb[tbPosInfo.Index] then
      LogErr("QianQiongGongMgr: LoadEscortPos failed, duplicate nIndex = ", tbPosInfo.Index)
      return false
    end
    tbPosInfo.PosX = tonumber(tbPosInfo.PosX)
    tbPosInfo.PosY = tonumber(tbPosInfo.PosY)
    tb[tbPosInfo.Index] = tbPosInfo
  end
  self.tbObstaclePos = {}
  for _, tbValue in pairs(tbObstaclePos) do
    self.tbObstaclePos[tbValue.szTrapName] = tbValue
  end
  self.tbEscortPosCfg = tb
  self.tbNpcSetting = tbTempNpcSetting
  self.tbGuanQiaNpcSetting = tbTemptbGuanQiaSetting
  self.tbGuanQiaNpcSetting = SetReadOnly(self.tbGuanQiaNpcSetting)
  self.tbNpcSetting = SetReadOnly(self.tbNpcSetting)
end

function QianQiongGongComm:CheckIsLimitMap(nCurMapId)
  local bIsIn = false
  for _, nLimitMapId in pairs(self.Cfg.tbEnterFieldMap) do
    if nCurMapId == nLimitMapId then
      bIsIn = true
      break
    end
  end
  return bIsIn
end

function QianQiongGongComm:GetRandomFieldMapTemplateID()
  local nTemplateId = Lib:RandTable(self.Cfg.tbEnterFieldMap)
  return nTemplateId
end

function QianQiongGongComm:IsOpen()
  return self.IS_SYS_OPEN
end

function QianQiongGongComm:CheckFubenType(nType)
  if not Lib:IsInteger(nType) then
    return false
  end
  for _, n in pairs(QianQiongGongMgr.Def.tbFubenType) do
    if n == nType then
      return true
    end
  end
  return false
end

function QianQiongGongComm:_GetConfigByFubenType(nFuBenType)
  return QianQiongGongMgr.Def.tbFubenType2Cfg[nFuBenType] or {}
end

function QianQiongGongComm:GetEntranceNpcTidByFubenType(nType)
  if not nType then
    return nil
  end
  local tbCfg = self:_GetConfigByFubenType(nType)
  return tbCfg.nNpcTid
end

function QianQiongGongComm:GetMapTidByFubenType(nType)
  if not nType then
    return nil
  end
  local tbCfg = self:_GetConfigByFubenType(nType)
  return tbCfg.nMapTid
end

function QianQiongGongComm:GetDiffCfgByFubenType(nFubenType)
  return self.Def.tbFubenType2Cfg[nFubenType]
end
