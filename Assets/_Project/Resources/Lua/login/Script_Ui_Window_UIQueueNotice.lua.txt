local tbWnd = Ui:CreateWindow("UIQueueNotice")
tbWnd.tbControls = {
  txtMsg = "imgBG/imgMsgBG/txtMsg",
  btnCancel = "imgBG/btnCancel"
}

function tbWnd:CloseByDefault()
  GiveUpWaitQueue()
  self:Close()
end

tbWnd.tbOnClick = {
  btnCancel = function(self)
    self:CloseByDefault()
  end
}

function tbWnd:OnClose()
end

function tbWnd:OnOpen(szText)
  self:SetInfo(szText)
end

function tbWnd:SetInfo(szText)
  self.pPanel:Label_SetText(self.txtMsg, szText)
end
