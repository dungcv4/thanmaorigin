function tbWnd:ToggleAgreement(bToggled)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleDeal, bToggled)

function tbWnd:SwitchAgreementToggle(bOn)

local function OnLoginServerBtnClick(self)
  if Login:IsShowAgreementRequired() and not self.pPanel:Toggle_GetChecked(tbControls.ToggleDeal) then
    Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\232\175\183\229\133\136\229\144\140\230\132\143\231\148\168\230\136\183\229\141\143\232\174\174\229\146\140\233\154\144\231\167\129\230\148\191\231\173\150\239\188\129"), {
      {}
    }, {
      i18n.EasyGet("\231\161\174\232\174\164")
    })
    return
  end
  if self.nConnectServerTimer then
    return
  end
  if not self.nCurServerId then
    return
  end
  Ui:OpenWindow("UILoadingTips", i18n.Get("Logging2Server"))
  Sdk:OnConnectServer(self.nCurServerId)
  Sdk:OnConnectServerV2(self.nCurServerId)
  Client.UpdateModule.Lua2CSValidateVersion(function(bResult)
    if not bResult then
      return
    end
    ConnectServer(self.nCurServerId)
    self.bServerConnectResponse = nil
    self.nConnectServerTimer = Timer:Register(Env.GAME_FPS * 20, function()
      if not self.bServerConnectResponse then
        Ui:CloseWindow("UILoadingTips")
        self.tbSerInfo = nil
        Ui:OpenWindow("UIMessageBoxBig", i18n.EasyGet("\231\153\187\229\189\149\230\156\141\229\138\161\229\153\168\229\164\177\232\180\165"), {
          function()
            self:onClickLogout()
            CloseServerConnect()
          end
        }, {
          i18n.EasyGet("\233\135\141\230\150\176\231\153\187\229\189\149"),
          i18n.szCancel
        })
      end
      self.nConnectServerTimer = nil
    end)
  end)
end

function tbWnd:OnCreate()
  self.pPanel:Sprite_SetNativeSize(tbControls.imgCADPA)
  self.pPanel:SetActive(tbControls.imgCADPA, Client.EnableLoginCADPA)
  self.pPanel:SetActive(tbControls.txtGameAdvice2, Client.EnableLoginCADPA)
  self:CloseConnectServerTimer()
end

function tbWnd:OnOpen()
  local szVersionType = Ui.GameEnv.GetVersionType()
  self.pPanel:SetActive(tbControls.txtAppVersion, false)
  self.pPanel:SetActive(tbControls.btnEwm, false)
  Ui:Toggle_BindEvent(self.pPanel, tbControls.ToggleDeal, function(this, isOn)
    this:SwitchAgreementToggle(isOn)
  end, self)
  Ui:DestroyWindow("UILoginChannelInner")
  Ui:DestroyWindow("UILoginChannelSDK")
  if Login:IsShowAgreementRequired() then
    self.pPanel:SetActive(tbControls.DealRoot, true)
    self.pPanel:SetActive(tbControls.ToggleDeal, true)
    self.pPanel:SetActive(tbControls.btnUserAgreement, true)
    self.pPanel:SetActive(tbControls.szSplitLabel, true)
    self.pPanel:SetActive(tbControls.btnPrivacyPolicy, true)
    self.pPanel:Label_SetText(tbControls.szSplitLabel, i18n.EasyGet("\229\146\140"))
    local nOpenAgreement = Login:GetOpenAgreement()
    if 0 == nOpenAgreement then
      Ui:OpenWindow("UIAgreementNew")
      self:ToggleAgreement(false)
    end
  else
    self.pPanel:SetActive(tbControls.DealRoot, false)
    self.pPanel:SetActive(tbControls.ToggleDeal, false)
    self.pPanel:SetActive(tbControls.btnUserAgreement, false)
    self.pPanel:SetActive(tbControls.szSplitLabel, false)
    self.pPanel:SetActive(tbControls.btnPrivacyPolicy, false)
  end
  Ui:CloseWindow("UILoadingTips")
  RequestServerList()
  RequestAccSerInfo()
end

function tbWnd:Logout()
  if not Ui:WindowVisible("UILoginBG") then
    Ui:OpenWindow("UILoginBG")
    Ui:OpenWindow("UITopEffect")
  end
  Login:SetAutoLogin(false)
  Ui:CloseWindow("UIMessageBoxBig")
  self:Close()
  CloseGateWayConnect()

    Sdk:Logout(true)
    Ui:OpenWindow("UILoginChannelSDK")
  else
    local szAccount = GetAccountName()
    Ui:OpenWindow("UILoginChannelInner", szAccount)
  end
end

function tbWnd:OnServerUnavailable(dwServerId)
  Ui:OpenWindow("UIMessageBoxBig", i18n.Get("ServerMaintenance"))
  if self.nConnectServerTimer then
    Ui:CloseWindow("UILoadingTips")
    Timer:Close(self.nConnectServerTimer)
    self.nConnectServerTimer = nil
  end
  Ui:OpenWindow("UISelectServer", self.tbSerList)
end

function tbWnd:OnServerLoginFail(nRetCode)
  if self.nConnectServerTimer then
    Ui:CloseWindow("UILoadingTips")
    Timer:Close(self.nConnectServerTimer)
    self.nConnectServerTimer = nil
  end
  local szMsg = "LoginFailUnknow"
  if nRetCode >= 1 and nRetCode <= 5 then
    szMsg = i18n.Get("LoginFail" .. nRetCode)
  end
  Ui:OpenWindow("UIMessageBoxBig", szMsg)
end

function tbWnd:OnSyncServerListDone(nCode)
  if 1 == nCode then
    self:__UpdateSerInfo()
    if bAllSameAsLastTime then
      OnLoginServerBtnClick(self)
    end
  else
    Ui:OpenWindow("UIMessageBoxBig", i18n.Get("ServerMaintenance"), {
      function()
        self:Logout()
      end
    }, {
      i18n.szOk
    })
  end
end

function tbWnd:__UpdateSerInfo()
  self.tbSerList = GetServerList()
  local nServerId = Ui.PlayerPrefs.GetInt("LastServerID", 0)
  local tbMyAccSerInfo = Login:GetAccSerInfo()
  local bFind = false
  local tbServerInfo
  for i, v in ipairs(self.tbSerList) do
    if nServerId == v.dwServerId then
      bFind = true
      tbServerInfo = v
      break
    end
  end
  if not bFind then
    local bFind2 = false
    local tbRecommandServerSet = {}
    for i, v in ipairs(self.tbSerList) do
      if v.nType == Login.SERVER_TYPE_RECOMMEND then
        table.insert(tbRecommandServerSet, v)
      end
    end
    if #tbRecommandServerSet > 0 then
      local nIndex = math.random(1, #tbRecommandServerSet)
      local tbServer = tbRecommandServerSet[nIndex]
      nServerId = tbServer.dwServerId
      tbServerInfo = tbServer
      bFind2 = true
    end
    local tbNewServerSet = {}
    for i, v in ipairs(self.tbSerList) do
      if v.nType == Login.SERVER_TYPE_NEW then
        table.insert(tbNewServerSet, v)
      end
    end
    if #tbNewServerSet > 0 then
      local nIndex = math.random(1, #tbNewServerSet)
      local tbServer = tbNewServerSet[nIndex]
      nServerId = tbServer.dwServerId
      tbServerInfo = tbServer
      bFind2 = true
    end
    if not bFind2 then
      for i, v in ipairs(self.tbSerList) do
        if v.nType ~= Login.SERVER_TYPE_OFFLINE then
          nServerId = v.dwServerId
          tbServerInfo = v
        end
      end
    end
  end
  Ui.PlayerPrefs.SetInt("LastServerID", nServerId)
  tbServerInfo = tbServerInfo or self.tbSerList[1]
  if tbServerInfo then
    self.nCurServerId = tbServerInfo.dwServerId
    self:UpdateServerShow(tbServerInfo)
  end
end

function tbWnd:OnQueueNotify(nPos, nGiveUp)
  if 1 == nGiveUp then
    Ui:CloseWindow("UIQueueNotice")
    self:onClickLogout()
    CloseServerConnect()
    return
  end
  if self.nConnectServerTimer then
    Ui:CloseWindow("UILoadingTips")
    Timer:Close(self.nConnectServerTimer)
    self.nConnectServerTimer = nil
  end
  local nLastOrderNumber = GetQueueOrderNumber()
  if 0 ~= nLastOrderNumber and nLastOrderNumber == nPos - 1 then
    nPos = nLastOrderNumber
  end
  SetQueueOrderNumber(nPos)
  local szInfo = i18n.EasyGet("\230\142\146\233\152\159\228\184\173\239\188\140\229\137\141\230\150\185\232\191\152\230\156\137 {0} \228\186\186 ...", nPos)
  local tbWnd = Ui:GetWindow("UIQueueNotice")
  if not tbWnd or not tbWnd:Visible() then
    Ui:OpenWindow("UIQueueNotice", szInfo)
  else
    tbWnd:SetInfo(szInfo)
  end
end

function tbWnd:CloseConnectServerTimer()

    self.nConnectServerTimer = nil
  end
end

function tbWnd:OnFixedUpdate()
  self.nLastRequestServerListTime = self.nLastRequestServerListTime or Ui.Time.time
  if Ui.Time.time - self.nLastRequestServerListTime > 5 then
    self.nLastRequestServerListTime = Ui.Time.time
    RequestServerList()
  end
end
