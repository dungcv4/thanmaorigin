function Login:OnSyncRoleListDone()
  Log(">>> role list done !!")
  print("Login:OnSyncRoleListDone()")
  self.bRoleListDone = true
  local tbRole = GetRoleList()
  self.tbSelRoleInfo = tbRole
  local tbReportRoleList = {}
  for i, v in ipairs(tbRole) do
    table.insert(tbReportRoleList, {
      role_id = v.nRoleID,
      role_name = v.szName
    })
  end
  self:SelRoleInfoSort()
  self.szReportRoleList = Lib:EncodeJson(tbReportRoleList)
  self:OnRoleListDoneAndMapLoaded()
end

function Login:SelRoleInfoSort()
  local function sortFun(a, b)
    local nAZongShiLevel = a.nZongShiLevel or 0
    
    local nBZoneShieLevel = b.nZongShiLevel or 0
    if nAZongShiLevel ~= nBZoneShieLevel then
      return nAZongShiLevel > nBZoneShieLevel
    end
    return a.nLevel > b.nLevel
  end
  
  table.sort(self.tbSelRoleInfo, sortFun)
end

function Login:GetRandomName(nSex)
  local nPreIndx = MathRandom(#self.tbRandomName1)
  local tbRandomMing = self.tbRandomName2
  if nSex and 1 == nSex then
    tbRandomMing = self.tbRandomName3
  end
  local nSuffiIndx = MathRandom(#tbRandomMing)
  local szName = tbRandomMing[nSuffiIndx].Name
  szName = self.tbRandomName1[nPreIndx].Name .. tbRandomMing[nSuffiIndx].Name
  return szName
end

function Login:SetNextLoginTime()
  local nFailCount = LocalData:GetGlobalNumber("LoginFailCount", 0)
  nFailCount = nFailCount + 1
  LocalData:SetGlobalNumber("LoginFailCount", nFailCount)
  local nNextLoginTime = os.time()
  if nFailCount >= 6 then
    nNextLoginTime = nNextLoginTime + 300
  elseif nFailCount >= 3 then
    nNextLoginTime = nNextLoginTime + 10
  end
  LocalData:SetGlobalNumber("NextLoginTime", nNextLoginTime)
end

function Login:ResetNextLoginTime()
  LocalData:SetGlobalNumber("NextLoginTime", 0)
  LocalData:SetGlobalNumber("LoginFailCount", 0)
end

function Login:ResetTryConnectGatewayCounter()
  self.nConnectGatewayTryCounter = 0
end

function Login:RandomLastConnectGatewayIndex()
  local tbList = self:GetZoneList()
  local szAddrs = tbList[self.CurrentSelectZoneID].Addrs
  local tbAddrs = Lib:SplitStr(szAddrs)
  local nTbLen = #tbAddrs
  if nTbLen <= 0 then
    return false
  end
  self.nLastConnectGatewayIndex = math.random(1, nTbLen)
end

function Login:IsNeedRetryConnectGateway()
  local tbList = self:GetZoneList()
  local szAddrs = tbList[self.CurrentSelectZoneID].Addrs
  local tbAddrs = Lib:SplitStr(szAddrs)
  local nTbLen = #tbAddrs
  if nTbLen <= 0 then
    return false
  end
  return self.nConnectGatewayTryCounter <= 2 * nTbLen
end

function Login:GetGatewayAddr()
  self.nLastConnectGatewayIndex = self.nLastConnectGatewayIndex + 1
  local tbList = self:GetZoneList()
  local szAddrs = tbList[self.CurrentSelectZoneID].Addrs
  local tbAddrs = Lib:SplitStr(szAddrs)
  local nTbLen = #tbAddrs
  if nTbLen <= 0 then
    LogErr(string.format("Login:GetGatewayAddr nTbLen <= 0!  self.CurrentSelectZoneID:%d, szAddrs:%s", self.CurrentSelectZoneID, szAddrs))
    return
  end
  if nTbLen < self.nLastConnectGatewayIndex then
    self.nLastConnectGatewayIndex = 1
  end
  local szGatewayAddr = tbAddrs[self.nLastConnectGatewayIndex]
  local tbGatewayAddr = Lib:SplitStr(szGatewayAddr, ":")
  local nTbLen = #tbGatewayAddr
  if 2 ~= nTbLen then
    LogErr(string.format("Login:GetGatewayAddr nTbLen ~= 2!  self.CurrentSelectZoneID:%d, szAddrs:%s, self.nLastConnectGatewayIndex:%d, szGatewayAddr:%s", self.CurrentSelectZoneID, szAddrs, self.nLastConnectGatewayIndex, szGatewayAddr))
    return
  end
  local szGatewayIP = tbGatewayAddr[1]
  local nGatewayPort = tonumber(tbGatewayAddr[2])
  return szGatewayIP, nGatewayPort
end

function Login:RetryOtherGateWayAddr()
  self:ConnectGateWay(self.szAccount, self.szAuthInfo)
end

function Login:ConnectGateWay(szAccount, szAuthInfo)
  local nTestCounter = 0
  local szGatewayIP, nGatewayPort = self:GetGatewayAddr()
  while not (szGatewayIP and nGatewayPort) or "" == szGatewayIP or nGatewayPort <= 0 do
    szGatewayIP, nGatewayPort = self:GetGatewayAddr()
    nTestCounter = nTestCounter + 1
    if nTestCounter >= 100 then
      break
    end
  end
  Log("ConnectGateWay", szGatewayIP, nGatewayPort, szAccount)
  self.nConnectGatewayTryCounter = self.nConnectGatewayTryCounter + 1
  self.szAccount = szAccount
  self.szAuthInfo = szAuthInfo
  ConnectGateway(szGatewayIP, nGatewayPort, szAccount, szAuthInfo)
  Ui:OpenWindow("UILoadingTips", i18n.Get("Connecting2Server"))
end

function Login:GetVersionDesc(nVersion)
  nVersion = nVersion or GAME_VERSION
  local nMajor = math.floor(nVersion / 2.68435456E8)
  local nMinor = math.floor(nVersion % 2.68435456E8 / 4194304.0)
  local nPatch = math.floor(nVersion % 4194304.0 / 65536.0)
  local nBuild = nVersion % 65536.0
  local szDesc = string.format("%d.%d.%d.%d", nMajor, nMinor, nPatch, nBuild)
  return szDesc
end

function Login:GetPrefix(nServerId)
  local szPrefix = ""
  if not (SelectServerPrefix and SelectServerPrefix.PREFIX and SelectServerPrefix.CONSTANTS) or not SelectServerPrefix.CONSTANTS.PREFIX_COUNT then
    return szPrefix
  end
  local nServerIdPrefix = tonumber(string.sub(tostring(nServerId), 1, SelectServerPrefix.CONSTANTS.PREFIX_COUNT))
  if SelectServerPrefix.PREFIX[nServerIdPrefix] then
    szPrefix = i18n.Get(SelectServerPrefix.PREFIX[nServerIdPrefix])
  end
  return szPrefix
end
