Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Event/Event.lua")
local tbActivity = ActivityMgr:GetClass("FirstLoginAward")

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local tbField = {
    "nConfigId",
    "szLogWay",
    "nAwardId",
    "nAwardKey",
    "szMailTitle",
    "szMailMsg",
    "szFrom"
  }
  local szFilepath = "Setting/Activity/FirstLoginAward/FirstLoginAward.tab"
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilepath, "dsddsss", "nConfigId", tbField)
  self.bLoadConfig = true
end

function tbActivity:OnOpen()
  self.nOnFirstLoginEventId = PlayerEvent:RegisterGlobal(PlayerEvent.EventType.OnFirstLogin, self.OnFirstLogin, self)
end

function tbActivity:OnClose()
  if self.nOnFirstLoginEventId then
    PlayerEvent:UnRegisterGlobal(PlayerEvent.EventType.OnFirstLogin, self.nOnFirstLoginEventId)
    self.nOnFirstLoginEventId = nil
  end
end

function tbActivity:OnFirstLogin()
  local pPlayer = me
  if not ActivityMgr:CheckOpening(self.tbScheduleCfg) then
    return
  end
  local tbConfig = self.tbConfig
  if not tbConfig then
    return
  end
  local tbReward = Award:GetAward(tbConfig.nAwardId, tbConfig.nAwardKey)
  local szMailTitle = i18n.Parse(tbConfig.szMailTitle)
  local szMailMsg = i18n.Parse(tbConfig.szMailMsg)
  local szMailFrom = i18n.Parse(tbConfig.szFrom)
  local nLogReason = Env[tbConfig.szLogWay]
  local tbMail = {
    To = pPlayer.dwID,
    Title = szMailTitle,
    Text = szMailMsg,
    From = szMailFrom,
    tbAttach = tbReward,
    nLogReason = nLogReason
  }
  Mail:SendSystemMail(tbMail)
end

function tbActivity:IsServerOnly()
  return true
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end
