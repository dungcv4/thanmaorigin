local tbWnd = Ui:CreateWindow("UIAnnouncement")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UI_MESSAGEBOX_OPEN,
      self.OnMsgBoxOpen,
      self
    },
    {
      EventNotify.emNOTIFY_UI_MESSAGEBOX_CLOSE,
      self.OnMsgBoxClose,
      self
    }
  }
end

tbWnd.tbControls = {
  btnCenter = "node/imgBG/btnCenter",
  Content = "node/imgBG/NormalView/Scroll/Viewport/Content",
  btnTemplate = "node/imgBG/btnList/btnList/paneBG/btnTemplate",
  WebView = "node/imgBG/WebView",
  NormalView = "node/imgBG/NormalView"
}
tbWnd.tbOnClick = {
  btnCenter = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.bulletin = self.pPanel:GetObject(self.Content)
  self.bulletinContent = self.bulletin:GetComponent("BulletinContainer")
  self.tbButtonListScrollView = Ui:NewScrollView(self, self.btnTemplate, function(self, nIndex, tbElement)
    self:OnButtonClick(nIndex)
  end)
end

function tbWnd:OnOpen(tbJsonData)
  self.tbJsonData = tbJsonData
  self.bulletinContent:ClearBulletin()
  self.tbButtonListScrollView:Clear()
  for i, tbInfo in ipairs(tbJsonData) do
    self.tbButtonListScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("Text", tbInfo.title)
    end)
    self.bulletinContent:AddBulletin(tbInfo.title, tbInfo.content)
  end
  self.tbButtonListScrollView:OnClickElement(1)
end

function tbWnd:OnButtonClick(nIndex)
  if self.tbJsonData[nIndex].content_type == "url" then
    self.pPanel:SetActive(self.NormalView, false)
    self.pPanel:SetActive(self.WebView, true)
    self.pPanel:ShowUrl(self.WebView, self.tbJsonData[nIndex].content, 0, 0)
  else
    self.pPanel:SetActive(self.NormalView, true)
    self.pPanel:SetActive(self.WebView, false)
    self.bulletinContent:ShowByIndex(nIndex - 1)
  end
end

function tbWnd:OnMsgBoxOpen()
  self:Close()
end

function tbWnd:OnMsgBoxClose()
  self:Close()
end
