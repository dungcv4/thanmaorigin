local tbWnd = Ui:CreateWindow("UISystemNotice")
local nNoticeMoveSpeed = 6

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SHARE_PHOTO,
      self.Close,
      self
    }
  }
end

tbWnd.tbControls = {
  txtMsg = "Notice/txtMsg",
  msgBG = "bg"
}

function tbWnd:OnCreate()
  self.tbMsg = {}
  self.pPanel:SetActive(self.txtMsg, false)
  self.txtMsgTransform = self.pPanel:GetRectTransform(self.txtMsg)
  self.startPos = self.txtMsgTransform.localPosition
  local NoticeRectTransform = self.pPanel:GetRectTransform("Notice")
  self.nNoticeWidth = NoticeRectTransform.rect.width
  self.nMoveDistance = 0
  self.isShowing = false
end

function tbWnd:PushMsg(szMsg)
  table.insert(self.tbMsg, szMsg)
  if not self.isShowing and 1 == #self.tbMsg then
    self.pPanel:Label_SetText(self.txtMsg, szMsg)
    self.txtMsgTransform.localPosition = self.startPos
    self.pPanel:SetActive(self.txtMsg, true)
  end
end

function tbWnd:OnFixedUpdate()
  if self.isShowing == true then
    self.txtMsgTransform.localPosition = Ui.Vector3(self.txtMsgTransform.localPosition.x - nNoticeMoveSpeed, 0, 0)
    self.nMoveDistance = self.nMoveDistance - nNoticeMoveSpeed
    if self.nMoveDistance <= 0 then
      self.isShowing = false
      self.pPanel:SetActive(self.txtMsg, false)
      self.pPanel:SetActive(self.msgBG, false)
      if #self.tbMsg > 0 then
        self.pPanel:Label_SetText(self.txtMsg, self.tbMsg[1])
      else
        self.pPanel:Label_SetText(self.txtMsg, "")
      end
    end
  elseif #self.tbMsg > 0 then
    self.isShowing = true
    self.pPanel:SetActive(self.txtMsg, true)
    self.pPanel:SetActive(self.msgBG, true)
    self.nMoveDistance = self.txtMsgTransform.rect.width + self.nNoticeWidth
    self.txtMsgTransform.localPosition = self.startPos
    table.remove(self.tbMsg, 1)
  end
end
