function tbWnd:OnCreate()
  self.pPanel:Sprite_SetNativeSize(tbControls.imgCADPA)
  self.pPanel:SetActive(tbControls.imgCADPA, Client.EnableLoginCADPA)
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen()
  EventNotify.OnNotify(EventNotify.emNOTIFY_UILOGINBG_ACTION, "Stop")
  Ui:CloseWindow("UILoadingTips")
  local szVersionType = Ui.GameEnv.GetVersionType()
  self.pPanel:Label_SetText(tbControls.txtAppVersion, i18n.Get("AppVersion") .. Ui.GameEnv.GetAppVersionStr4() .. "." .. Ui.GameEnv.GetUpdateVersion())
  self.pPanel:Button_SetText(tbControls.btnCallChannelLogin, i18n.EasyGet("\231\153\187\229\189\149"), "Text")
  if Ui.bReturnToMain then
    Ui.bReturnToMain = false
    if Login.szAuthInfo then
      Login:InitAndConnectGateWay("", Login.szAuthInfo)
    else
      Sdk:Login()
    end
  elseif Login.bBackToSdkLoginPanel then
    Sdk:OpenUserCenter("")
  else
    Sdk:Login()
  end
  Client.UIModule.CloseStartUI()
  Ui:PlayLoginMusic()
  Ui.SceneCameraController.ClearTarget()
  Ui.SceneCameraController.SetGray(false)
  Client:DeepGC()
  ClientBulletin:GetBulletin(ClientBulletin.BULLETIN_TYPE_LOGIN)
  Sdk:RecordAppsflyerEvent("af_show_login_form")
  Sdk:RecordFirebaseEvent("show_login_form")
  Sdk:RecordCleverTapEvent("show_login_form")
end

function tbWnd:OnLoginSuccess(nCode, szAuthInfo)
  Log("Sdk:Login OnLoginSuccess" .. szAuthInfo)
  Login:InitAndConnectGateWay("", szAuthInfo)
end

function tbWnd:OnLoginFail(nCode, szMsg, szChannelCode)
  Ui:OpenWindow("UIMessageBoxBig", i18n.EasyGet("\231\153\187\229\189\149\229\164\177\232\180\165"), {
    function()
      Sdk:Login()
    end
  }, {
    i18n.szOk
  })
end

function tbWnd:OnLoginCancel(nCode, szMsg)
  Ui:OpenWindow("UIMessageBoxBig", i18n.EasyGet("\231\153\187\229\189\149\229\143\150\230\182\136"), {
    function()
      Sdk:Login()
    end
  }, {
    i18n.szOk
  })
end

function tbWnd:GatewayConnectResult(nResult)
  if 1 ~= nResult then
    if Login:IsNeedRetryConnectGateway() then
      Login:RetryOtherGateWayAddr()
      return
    end
    Ui:OpenWindow("UIMessageBoxBig", i18n.Get("Connect2ServerFailed"), {
      function()
        Sdk:Login()
      end
    }, {
      i18n.szOk
    })
    Ui:CloseWindow("UILoadingTips")
    Sdk:Logout(false)
  end
  Login:SetNextLoginTime()
end

function tbWnd:GatewayHandSuccess(nRetCode, nShowAgreement)
  Login.nShowAgreement = nShowAgreement
  Client.ChannelModule._UID = GetAccountName()
  Log("GatewayHandSuccess UID=", Client.ChannelModule._UID, Login.nShowAgreement)
  local tbArgs = {
    vng_ge_user_id = GetAccountName(),
    login_channel = Client.ChannelModule._ChannelID
  }
  local tbAfArgs = {
    vng_ge_user_id = GetAccountName(),
    login_channel = Client.ChannelModule._ChannelID,
    app_version = Ui.GameEnv.GetAppVersionStr3()
  }
  Sdk:RecordAppsflyerEvent("af_login", tbAfArgs)
  Sdk:RecordFirebaseEvent("login", tbArgs)
  Sdk:RecordCleverTapEvent("login", {
    user_id = GetAccountName()
  })
  if 0 ~= nRetCode then
    if tbLoginErrorMsg[nRetCode] then
      Ui:OpenWindow("UIMessageBoxBig", tbLoginErrorMsg[nRetCode](), {
        function()
          Sdk:Login()
        end
      }, {
        i18n.szOk
      })
    else
      Ui:OpenWindow("UIMessageBoxBig", i18n.Get("GatewayRetCode", nRetCode), {
        function()
          Sdk:Login()
        end
      }, {
        i18n.szOk
      })
    end
    Login:SetNextLoginTime()
    Sdk:Logout(false)
    Ui:CloseWindow("UILoadingTips")
  else
    Ui:CloseWindow("UIMessageBoxBig")
    self:Close()
    Login:ResetNextLoginTime()
    Ui:OpenWindow("UILoginServer")
  end
end

function tbWnd:OnNeedAccountReg(nRetCode)
end

function tbWnd:GatewayConnectLost(nSysErr, nUsrErr)
  if nSysErr > 0 or 2 == nUsrErr then
    Ui:OpenWindow("UIMessageBoxBig", string.format("%s(%d_%d)", i18n.Get("Connect2ServerFailed"), nSysErr, nUsrErr))
  end
  Ui:CloseWindow("UILoadingTips")
end

function tbWnd:OnNeedAccountActive(nRetCode)
  Ui:CloseWindow("UILoadingTips")
  
  local function fnCallBack(szInputCode)
    if "" == szInputCode or string.len(szInputCode) > 15 then
      self:OnNeedAccountActive(1)
      return
    end
    RequestAccountActive(szInputCode)
  end
  
  if 0 == nRetCode then
    Ui:OpenWindow("UIInputBoxCmn", i18n.EasyGet("\232\175\183\232\190\147\229\133\165\230\191\128\230\180\187\231\160\129"), fnCallBack)
  else
    Ui:OpenWindow("UIMessageBoxBig", i18n.EasyGet("\230\191\128\230\180\187\231\160\129\228\184\141\230\173\163\231\161\174"), {
      function()
        Ui:OpenWindow("UIInputBoxCmn", i18n.EasyGet("\232\175\183\232\190\147\229\133\165\230\191\128\230\180\187\231\160\129"), fnCallBack)
      end
    }, {
      i18n.szOk
    })
  end
end

function tbWnd:OnNeedClientUpdate(nGameVersion)
  Ui:CloseWindow("UILoadingTips")
  local szVersionInfo = Login:GetVersionDesc(nGameVersion)
  Ui:OpenWindow("UIMessageBoxBig", i18n.Get("NeedUpdateClient", szVersionInfo), {
    function()
      CS.UnityEngine.Application.Quit()
    end
  }, {
    i18n.Get("QuitGame")
  })
end
