  if nAttrIndex and nAttrIndex > 0 then

    if nAttrCfgIndex then
      local szAttribMsg, _ = FightSkill:GetMagicDescSplit(self.tbSkillAttrCfg[nAttrCfgIndex][nAttrIndex].szAttrType, {
        0,
        0,
        0
      })

    me.CenterMsg(i18n.EasyGet("\229\141\135\231\186\167\230\136\144\229\138\159"))
  end
  if Ui:WindowVisible("UIBag") then
    Ui:GetWindow("UIBag"):NotifyHorseView()
  end
  if Ui:WindowVisible("UIHorseSkillUpgrade") then
    Ui:GetWindow("UIHorseSkillUpgrade"):UpgradeRsp()
  end
end

function HorseSkillMgr:HorseSkillUpgradeReqC(nSkillIndex, nGradeIndex)

  if not bRet then
    me.CenterMsg(szErr)
    return
  end
  RemoteServer.RequestHorseSkill("HorseSkillUpgradeReqS", nSkillIndex, nGradeIndex)
end

function HorseSkillMgr:HorseSkillResetC(nSkillIndex, nGradeIndex)
  local bRet, szErr = self:CanResetHorseSkill(me, nSkillIndex, nGradeIndex)

  end
  RemoteServer.RequestHorseSkill("HorseSkillResetS", nSkillIndex, nGradeIndex)
end

function HorseSkillMgr:GetAttrAddIndex(nSkillIndex, nGrade)
  local nAttrCfgIndex = self:GetSkillAttrCfg(nSkillIndex, nGrade)
  if not nAttrCfgIndex then
    return
  end
  local tbAttrAddIndex = {}
  for nAttrIndex, tbV in pairs(self.tbSkillAttrCfg[nAttrCfgIndex]) do
    tbAttrAddIndex[nAttrIndex] = FightSkill:GetTableValueMainIndex(tbV.szAttrType)
    if not tbAttrAddIndex[nAttrIndex] then
      return
    end
  end
  return tbAttrAddIndex
end

function HorseSkillMgr:GetPerSkillAllAttr(nSkillIndex, pAsyncRole)
  local tbAttrInfo = {}
  local nOpenGrade = self:GetHSOpenGrade()
  for nGrade = 1, nOpenGrade do
    local nSkillLevel = 0
    if pAsyncRole then
      nSkillLevel = self:GetSyncSkillLevel(pAsyncRole, nSkillIndex, nGrade)
    else
      nSkillLevel = self:GetSkillLevel(me, nSkillIndex, nGrade)
    end
    self:GetSkillPerLevelAttr(nSkillIndex, nGrade, nSkillLevel, tbAttrInfo, pAsyncRole)
  end
  local tbRet = {}
  for _, tbV in pairs(tbAttrInfo) do
    tbRet[tbV.nIndex] = tbV
  end
  return tbRet
end

function HorseSkillMgr:GetSkillPerLevelAttr(nSkillIndex, nGrade, nSkillLevel, tbAttrInfo, pAsyncRole)
  local nAttrCfgIndex = self:GetSkillAttrCfg(nSkillIndex, nGrade)
  if not nAttrCfgIndex then
    return
  end
  if not self.tbSkillAttrCfg[nAttrCfgIndex] or not self.tbUpgradeLevelCfg[nAttrCfgIndex] then
    return
  end
  local tbAttrAddIndex = self:GetAttrAddIndex(nSkillIndex, nGrade)
  if not tbAttrAddIndex then
    return
  end
  for nAttrIndex, tbV in ipairs(self.tbSkillAttrCfg[nAttrCfgIndex]) do
    if not tbAttrInfo[tbV.szAttrType] then
      local nCount = Lib:CountTB(tbAttrInfo)
      tbAttrInfo[tbV.szAttrType] = {
        tbValue = {
          0,
          0,
          0
        },
        szAttrType = tbV.szAttrType,
        nIndex = nCount + 1
      }
    end
    if nSkillLevel > 0 and self.tbUpgradeLevelCfg[nAttrCfgIndex][nSkillLevel] then
      local tbValue = self.tbUpgradeLevelCfg[nAttrCfgIndex][nSkillLevel][self.nLevelAttrIndexCfgOffSet + nAttrIndex]
      for nIndex, nValue in pairs(tbValue) do
        if tbAttrAddIndex[nAttrIndex] == nIndex then
          tbAttrInfo[tbV.szAttrType].tbValue[nIndex] = tbAttrInfo[tbV.szAttrType].tbValue[nIndex] + nValue
        end
      end
    end
  end
  for i = self.SKILL_ATTR_MAX_UPGRADE_BEGIN, self.SKILL_ATTR_MAX_UPGRADE_END do
    local nTimes = i - self.SKILL_ATTR_MAX_UPGRADE_BEGIN + 1
    local nAttrIndex
    if pAsyncRole then
      nAttrIndex = self:GetSyncSkillAttrMaxUpIndex(pAsyncRole, nSkillIndex, nGrade, i)
    else
      nAttrIndex = me.GetUserValue(self.tbSaveGroupKey[nSkillIndex], i + self.SKILL_DATA_TYPE_OFFSET * (nGrade - 1))
    end
    if nAttrIndex > 0 and tbAttrAddIndex[nAttrIndex] then
      local tbValue = self.tbSkillQualityAttrCfg[nAttrCfgIndex][nTimes][nAttrIndex]
      local szAttrType = self.tbSkillAttrCfg[nAttrCfgIndex][nAttrIndex].szAttrType
      for nIndex, nValue in pairs(tbValue) do
        if tbAttrAddIndex[nAttrIndex] == nIndex then
          tbAttrInfo[szAttrType].tbValue[nIndex] = tbAttrInfo[szAttrType].tbValue[nIndex] + nValue
        end
      end
    end
  end
end

function HorseSkillMgr:GetSkillAttrQuality(nSkillIndex, nGrade, nAttrIndex, pAsyncRole)
  if not self.tbSaveGroupKey[nSkillIndex] then
    return 0
  end
  local nAttrQuality = 0
  for i = self.SKILL_ATTR_MAX_UPGRADE_BEGIN, self.SKILL_ATTR_MAX_UPGRADE_END do
    local nAttrIndexSave
    if pAsyncRole then
      nAttrIndexSave = self:GetSyncSkillAttrMaxUpIndex(pAsyncRole, nSkillIndex, nGrade, i)
    else
      nAttrIndexSave = me.GetUserValue(self.tbSaveGroupKey[nSkillIndex], i + self.SKILL_DATA_TYPE_OFFSET * (nGrade - 1))
    end
    if nAttrIndex == nAttrIndexSave then
      nAttrQuality = nAttrQuality + 1
    end
  end
  return nAttrQuality or 0
end

function HorseSkillMgr:GetSkillAttrMinQuality(nSkillIndex, nAttrIndex, pAsyncRole)
  if not self.tbSaveGroupKey[nSkillIndex] then
    return 0
  end
  local nOpenGrade = self:GetHSOpenGrade()
  local nQuality = math.maxinteger
  if nOpenGrade <= 0 then
    nQuality = 0
  end
  for i = 1, nOpenGrade do
    local nAttrQuality = self:GetSkillAttrQuality(nSkillIndex, i, nAttrIndex, pAsyncRole)
    nQuality = math.min(nAttrQuality, nQuality)
  end
  return nQuality
end

  self:OnUnEquip(me)
end

function HorseSkillMgr:GetSkillStoneList(nSkillIndex)
  local tbItemList = me.GetItemListInBag("HorseSkillStoneItem")
  if not tbItemList then
    return {}
  end
  local tbRet = {}
  for _, pItem in ipairs(tbItemList) do
    if pItem.nDetailType == nSkillIndex + Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1 - 1 then
      table.insert(tbRet, pItem)
    end
  end
  
  local function fnSort(tbA, tbB)
    if tbA.nLevel ~= tbB.nLevel then
      return tbA.nLevel > tbB.nLevel
    end
    if tbA.nQuality ~= tbB.nQuality then
      return tbA.nQuality > tbB.nQuality
    end
    return tbA.GetSeries() < tbA.GetSeries()
  end
  
  table.sort(tbRet, fnSort)
  return tbRet
end

function HorseSkillMgr:OnUseOrUnuseEquip(bIsUse, nPos, dwEquipId)
  local pEquip = KItem.GetItemObj(dwEquipId)
  if not bIsUse or PlayerEvent.bLogin then
  end
  if nil ~= pEquip and self:IsHorseSkillStone(pEquip) and PlayerEvent.bLogin then
    self:OnCheckSkillStoneRedPoint()
  end
end

function HorseSkillMgr:CheckLevelUpgradeUIHadClickToday()
  local nLocalDay = Lib:GetLocalDay(GetTime() - self.OFFSET)
  local nLastClickOpenTime = LocalData:GetRoleNumber(self.szHorseSkillUIHadOpen, 0) - self.OFFSET
  local nLastClicOpenkDay = Lib:GetLocalDay(nLastClickOpenTime)
  if nLocalDay == nLastClicOpenkDay then
    return true
  else
    return false
  end
end

function HorseSkillMgr:OnCheckRedPoint()
  self:OnCheckSkillLevelRedPoint()
  self:OnCheckSkillStoneRedPoint()
end

function HorseSkillMgr:OnCheckSkillLevelRedPoint()
  local bClickUI = self:CheckLevelUpgradeUIHadClickToday()
  for nSkillIndex = 1, self.nMaxSkillNum do
    for nGrade = 1, self.nMaxGrade do
      local nRedPoint = RedPointMgr.EVENT.HorseSkillLevelBegin + (nSkillIndex - 1) * HorseSkillMgr.nMaxGrade + nGrade - 1
      if self:CanSkillUpgrade(me, nSkillIndex, nGrade) and not bClickUI then
        RedPointMgr:LightRedPoint(nRedPoint)
      else
        RedPointMgr:FadeRedPoint(nRedPoint)
      end
    end
  end
end

function HorseSkillMgr:OnCheckSkillStoneRedPoint()
  for i = 1, self.nMaxSkillStoneNum do
    local nRedPoint = RedPointMgr.EVENT.HorseSkillStoneBegin + i - 1
    local nSlot = Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1 + i - 1
    local pEquip = me.GetEquipByPos(Item.EQUIPPOS_EXTEND, nSlot)
    if pEquip then
      RedPointMgr:FadeRedPoint(nRedPoint)
    else
      local tbItemList = self:GetSkillStoneList(i)
      if #tbItemList > 0 then
        RedPointMgr:LightRedPoint(nRedPoint)
      else
        RedPointMgr:FadeRedPoint(nRedPoint)
      end
    end
  end
end

function HorseSkillMgr:OnPlayerLogin()
  self:OnCheckRedPoint()
end

function HorseSkillMgr:OnSyncItemLimitTime()
  if not PlayerEvent.bLogin then
    return
  end
  self:OnCheckRedPoint()
end

function HorseSkillMgr:OnUseEquip(pEquip, nSlot)
  if not HorseSkillMgr:IsHorseSkillStone(pEquip) then
    return
  end
  if Ui:WindowVisible("UIHorseSkillStone") then
    Ui:GetWindow("UIHorseSkillStone"):UpdateView()
  else
    Ui:OpenWindow("UIHorseSkillStone")
  end
end

function HorseSkillMgr:OnUnUseEquip(pEquip, nSlot)
  if not HorseSkillMgr:IsHorseSkillStone(pEquip) then
    return
  end
  if Ui:WindowVisible("UIHorseSkillStone") then
    Ui:GetWindow("UIHorseSkillStone"):UpdateView()
  end
end
