function tbWnd:OnCreate()
  local obj = self.pPanel:GetObject(tbControls.NpcModel)
  self.NpcModel = obj:GetComponent(Ui.UINpcModel)
  self.tbEquipPath = {
    tbControls.jiangsheng,
    tbControls.majue,
    tbControls.maan,
    tbControls.madeng,
    tbControls.mazhang
  }
  self.nSelectEquip = 1
  self.nSelectSkill = 1
  for nIndex = 1, #self.tbEquipPath do
    local nEquipPos = Horse:GetEquipPosByIndex(nIndex)
    local pEquip = Item.ExtEquip:GetEquipObj(me, nEquipPos)
    if pEquip then
      self.nSelectEquip = nIndex
      break
    end
  end
  self.pPanel:SetActive(self.tbEquipPath[self.nSelectEquip] .. "/imgSelected", true)
  self.tbItemEquip = {}
  for i, szPath in ipairs(self.tbEquipPath) do
    self.tbItemEquip[i] = Ui:NewExistItemGrid(self, szPath .. "/UIItemGrid")
    self.tbItemEquip[i]:SetClickEnable(true, function()
      self.pPanel:SetActive(self.tbEquipPath[self.nSelectEquip] .. "/imgSelected", false)
      self.nSelectEquip = i
      self.pPanel:SetActive(szPath .. "/imgSelected", true)
      self:UpdateHorseEquipInfo()
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUpEquip, {
        RedPointMgr.EVENT.HorseEquipGrade1 + self.nSelectEquip - 1
      })
    end)
    self.tbItemEquip[i]:SetEmpty(true)
  end
  self.tbBtnGroup = Ui:NewButtonGroup(self, {
    [1] = tbControls.btnLevelUp,
    [2] = tbControls.btnHorseEquip,
    [3] = tbControls.btnHorseSkill
  }, true, function(self, nIndex)
    self.nViewType = nIndex
    local tbCfg = self.tbPanelStateDef[self.nViewType]
    if tbCfg[tbWnd.nLevelUpView] then
      self.pPanel:SetActive(tbControls.LevelUp, true)
    else
      self.pPanel:SetActive(tbControls.LevelUp, false)
    end
    if tbCfg[tbWnd.nHorseEquipView] then
      self.pPanel:SetActive(tbControls.HorseEquip, true)
      self.pPanel:SetActive(tbControls.EquipList, true)
    else
      self.pPanel:SetActive(tbControls.HorseEquip, false)
      self.pPanel:SetActive(tbControls.EquipList, false)
    end
    if tbCfg[tbWnd.nHorseSkillView] then
      self.pPanel:SetActive(tbControls.HorseSkill, true)
      self.pPanel:SetActive(tbControls.SkillList, true)
    else
      self.pPanel:SetActive(tbControls.HorseSkill, false)
      self.pPanel:SetActive(tbControls.SkillList, false)
    end
  end)
  self.tbConsumeGridLevel = Ui:NewExistItemGrid(self, tbControls.UIItemGridLevel)
  self.tbConsumeGridLevel:SetClickEnable(true)
  self.tbSelectEquip = Ui:NewExistItemGrid(self, tbControls.UIItemGridEquip)
  self.tbSelectEquip.szItemOpt = "PlayerEquip"
  self.tbSelectEquip:SetClickEnable(true)
  self.tbConsumeGridEquip = Ui:NewExistItemGrid(self, tbControls.UIConsumeItemGridEquip)
  self.tbConsumeGridEquip:SetClickEnable(true)
  self.tbAttrScrollLevel = Ui:NewScrollView(self, tbControls.TemplateLevel, nil, false, true)
  self.tbAttrScrollEquip = Ui:NewScrollView(self, tbControls.TemplateEquip, nil, false, true)
  self.tbSkillAttrScroll = Ui:NewScrollView(self, tbControls.SkillAttrTemplate, nil, false, true)
  for nIndex, szKey in ipairs(tbControls.tbBtnSkillKey) do
    Ui:Button_BindEvent(self.pPanel, szKey, self.OnSkillSelect, self, nIndex)
  end
  self.pPanel:SetActive(tbControls.btnAddEquip, false)
end

function tbWnd:OnOpen(nSelect)
  nSelect = nSelect or 1
  self.pPanel:SetActive(tbControls.btnJinjie, Horse:IsJinJieOpen())
  self.pPanel:SetActive(tbControls.btnHorseEquip, Horse:IsHorseEquipOpen())
  self.pPanel:SetActive(tbControls.btnHorseSkill, HorseSkillMgr:IsHorseSkillOpen(me))
  self.pPanel:SetActive(tbControls.imgArrowLeft, false)
  self.pPanel:SetActive(tbControls.imgArrowRight, false)
  self.nViewType = self.nLevelUpView
  self:ShowModel()
  self.tbBtnGroup:Select(nSelect)
  self:UpdateView()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUp, {
    RedPointMgr.EVENT.HorseLevel
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUpLevel, {
    RedPointMgr.EVENT.HorseLevel
  })
  local tbHorseEquipEventList = {}
  for i = 1, Horse.nHorseEquipCount do
    table.insert(tbHorseEquipEventList, RedPointMgr.EVENT.HorseEquipGrade1 + i - 1)
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.tbEquipPath[i], {
      RedPointMgr.EVENT.HorseEquipGrade1 + i - 1
    })
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnHorseEquip, tbHorseEquipEventList)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUpEquip, {
    RedPointMgr.EVENT.HorseEquipGrade1 + self.nSelectEquip - 1
  })
  local tbAllHorseSkillEventList = {}
  local tbHorseSkillLevelEventList = {}
  for i = 1, HorseSkillMgr.nMaxSkillNum do
    local tbPerSkillRedEvent = {}
    for j = 1, HorseSkillMgr.nMaxGrade do
      local nEventNum = RedPointMgr.EVENT.HorseSkillLevelBegin + (i - 1) * HorseSkillMgr.nMaxGrade + j - 1
      table.insert(tbPerSkillRedEvent, nEventNum)
      table.insert(tbHorseSkillLevelEventList, nEventNum)
      table.insert(tbAllHorseSkillEventList, nEventNum)
    end
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.tbImageSkillKey[i], tbPerSkillRedEvent)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnSkillUpgrade, tbHorseSkillLevelEventList)
  local tbHorseSkillStoneEventList = {}
  for i = RedPointMgr.EVENT.HorseSkillStoneBegin, RedPointMgr.EVENT.HorseSkillStoneEnd do
    table.insert(tbHorseSkillStoneEventList, i)
    table.insert(tbAllHorseSkillEventList, i)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnHorseSkill, tbAllHorseSkillEventList)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnPassiveSkillLook, tbHorseSkillStoneEventList)
end

function tbWnd:UpdateView()
  self:UpdateHorseEquip()
  self:UpdateHorseLevelUpInfo()
  self:UpdateHorseEquipInfo()
  self:UpdateHorseSkillInfo()
  local nfight = Horse:GetHorseAllFightPower(me)
  self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\230\136\152\229\138\155\229\128\188  {0}", nfight))
end

function tbWnd:OnClose()
  self.NpcModel:Release()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUp)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUpLevel)
  for i = 1, Horse.nHorseEquipCount do
    RedPointMgr:DynamicUnRegist(self.pPanel, self.tbEquipPath[i])
  end
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnHorseEquip)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUpEquip)
  for i = 1, HorseSkillMgr.nMaxSkillNum do
    RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.tbImageSkillKey[i])
  end
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnHorseSkill)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnSkillUpgrade)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnPassiveSkillLook)
end

function tbWnd:OnDestroy()
  for nIndex, szKey in ipairs(tbControls.tbBtnSkillKey) do
    Ui:Button_UnBindEvent(self.pPanel, szKey)
  end
end

function tbWnd:ShowModel()
  local nCurGrade = Horse:GetHorseGrade(me)
  local szCurName = i18n.Get(Horse.tbGradeInfoCfg[nCurGrade][1])
  self.pPanel:Label_SetText(tbControls.txtHorse, szCurName)
  local nCurTemplateId = Horse.tbGradeInfoCfg[nCurGrade][8]
  local horseModelId = Horse:GetModel(me, nCurTemplateId)
  self.NpcModel:ShowModel(me.nSex, 1, 0, 0, 0, horseModelId, 0, 0)
  self.NpcModel:SetDir(Npc.ModelDirection.LeftBottom)
end

function tbWnd:UpdateHorseEquip()
  for i = Item.ExtEquip.CFG.SLOT.HORSEREIN, Item.ExtEquip.CFG.SLOT.HORSESHOE do
    local pItem = Item.ExtEquip:GetEquipObj(me, i)
    if pItem then
      self.tbItemEquip[i - Item.ExtEquip.CFG.SLOT.HORSEREIN + 1]:SetItem(pItem.dwId, pItem.dwTemplateId, {bBigIcon = true}, me.nFaction, me.nFactionSect, me.nSex)
    end
  end
end

function tbWnd:UpdateHorseLevelUpInfo()
  local nCurLevel = Horse:GetHorseLevel(me)
  self.pPanel:Label_SetText(tbControls.txtNowLevel, i18n.EasyGet("{0}\231\186\167", nCurLevel))
  local nNextLevel = nCurLevel + 1
  local _, nMaxLevel = Horse:GetMaxCanUpgrade()
  if nNextLevel > nMaxLevel then
    nNextLevel = nMaxLevel
  end
  self.pPanel:Label_SetText(tbControls.txtNextLevel, i18n.EasyGet("{0}\231\186\167", nNextLevel))
  self:UpdateAttrInfoLevel(nCurLevel, nNextLevel)
  self:UpdateConsumeInfoLevel(nCurLevel, nNextLevel)
end

function tbWnd:UpdateAttrInfoLevel(nCurLevel, nNextLevel)
  local tbAttrCur = Horse.tbLevelAttrCfg[nCurLevel][4]
  local tbAttrNext = Horse.tbLevelAttrCfg[nNextLevel][4]
  self.tbAttrScrollLevel:Reset()
  for i = 1, #Horse.tbLevelAttrName do
    self.tbAttrScrollLevel:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbCurAttrNum = tbAttrCur[i]
      local tbNextAttrNum = tbAttrNext[i]
      local szDesc, szValueCur = FightSkill:GetMagicDescSplit(Horse.tbLevelAttrName[i], tbCurAttrNum)
      local _, szValueNext = FightSkill:GetMagicDescSplit(Horse.tbLevelAttrName[i], tbNextAttrNum)
      pPanel:Label_SetText("txtAttrib", szDesc)
      pPanel:Label_SetText("txtValueNow", szValueCur)
      pPanel:Label_SetText("txtValueNext", szValueNext)
      local szSubValue = FightSkill:StringValueSubResult(szValueCur, szValueNext)
      if "" == szSubValue then
        pPanel:Label_SetText("txtValueAdd", "")
      else
        pPanel:Label_SetText("txtValueAdd", szSubValue)
      end
    end)
  end
end

function tbWnd:UpdateConsumeInfoLevel(nCurLevel, nNextLevel)
  if nNextLevel <= nCurLevel then
    self.pPanel:SetActive(tbControls.txtCostLevelWnd, false)
    self.pPanel:SetActive(tbControls.txtMaxLevel, true)
    self.pPanel:SetActive(tbControls.btnLevelUpLevel, false)
    local szTimeFrame = Horse:GetNextLevelTimeframe(2)
    local nNextWorldLevel = Calendar:GetWorldLevelByTimeframe(szTimeFrame)
    if nNextWorldLevel > 0 then
      self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\233\169\175\229\133\187\229\136\176\230\156\128\233\171\152\231\186\167\239\188\140\229\188\128\230\148\190{0}\231\173\137\231\186\167\228\184\138\233\153\144\229\144\142\229\143\175\231\187\167\231\187\173\233\169\175\229\133\187\239\188\129", Lib:GetLevelText(nNextWorldLevel)))
    else
      self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\233\169\175\229\133\187\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
    end
  else
    self.pPanel:SetActive(tbControls.txtCostLevelWnd, true)
    self.pPanel:SetActive(tbControls.btnLevelUpLevel, true)
    self.pPanel:SetActive(tbControls.txtMaxLevel, false)
    local tbCfg = Horse.tbLevelAttrCfg[nNextLevel]
    self.nConsumeTemplateIdLevel = tbCfg[1]
    self.tbConsumeGridLevel:SetItemByTemplate(tbCfg[1])
    local nCurNum = me.GetItemCountByTypeInBag(tbCfg[1])
    local nNeedNum = Horse:GetUpgradeLevelConsumeNum(nNextLevel)
    self.pPanel:Label_SetText(tbControls.txtCostLevel, nCurNum .. "/" .. nNeedNum)
    if nCurNum >= nNeedNum then
      self.pPanel:Label_SetColorByName(tbControls.txtCostLevel, "Enough")
    else
      self.pPanel:Label_SetColorByName(tbControls.txtCostLevel, "NotEnough")
    end
  end
end

function tbWnd:UpdateHorseEquipInfo()
  local nEquipPos = Horse:GetEquipPosByIndex(self.nSelectEquip)
  local pEquip = Item.ExtEquip:GetEquipObj(me, nEquipPos)
  if pEquip and pEquip.dwId > 0 then
    self.tbSelectEquip:SetItem(pEquip.dwId, pEquip.dwTemplateId, {bBigIcon = true}, me.nFaction, me.nFactionSect, me.nSex)
    local nCurLevel = pEquip.nLevel
    self.pPanel:Label_SetText(tbControls.txtNowEquip, i18n.Get(Horse.tbEquipName[nCurLevel]))
    local nNextLevel = nCurLevel + 1
    local _, _, nMaxEquipLevel = Horse:GetMaxCanUpgrade()
    if nNextLevel > nMaxEquipLevel then
      nNextLevel = nMaxEquipLevel
    end
    self.pPanel:Label_SetText(tbControls.txtNextEquip, i18n.Get(Horse.tbEquipName[nNextLevel]))
    self:UpdateAttrInfoEquip(nCurLevel, nNextLevel)
    self:UpdateConsumeInfoEquip(nCurLevel, nNextLevel)
    local szName = Item:GetName(pEquip.dwTemplateId, me)
    self.pPanel:Label_SetText(tbControls.txtEquipName, szName)
  else
    self.tbSelectEquip:SetEmpty(true)
    self.pPanel:SetActive(tbControls.txtEquipName, "")
    self.pPanel:Label_SetText(tbControls.txtNowEquip, "")
    self.pPanel:Label_SetText(tbControls.txtNextEquip, "")
    self.pPanel:SetActive(tbControls.txtCostEquipWns, false)
    self.tbAttrScrollEquip:Reset()
  end
end

function tbWnd:UpdateAttrInfoEquip(nCurLevel, nNextLevel)
  local nTemplateIdCur = Horse.tbEquipAttrCfg[self.nSelectEquip][nCurLevel][4]
  local nTemplateIdNext = Horse.tbEquipAttrCfg[self.nSelectEquip][nNextLevel][4]
  local tbAttrCur = KItem.GetEquipBaseProp(nTemplateIdCur).tbBaseAttrib
  local tbAttrNext = KItem.GetEquipBaseProp(nTemplateIdNext).tbBaseAttrib
  self.tbAttrScrollEquip:Reset()
  for i, tbMA in ipairs(tbAttrCur) do
    self.tbAttrScrollEquip:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local szDesc, szValueCur = FightSkill:GetMagicDescSplit(tbAttrCur[i].szName, tbAttrCur[i].tbValue)
      local _, szValueNext = FightSkill:GetMagicDescSplit(tbAttrNext[i].szName, tbAttrNext[i].tbValue)
      pPanel:Label_SetText("txtAttrib", szDesc)
      pPanel:Label_SetText("txtValueNow", szValueCur)
      pPanel:Label_SetText("txtValueNext", szValueNext)
      local szSubValue = FightSkill:StringValueSubResult(szValueCur, szValueNext)
      if "" == szSubValue then
        pPanel:Label_SetText("txtValueAdd", "")
      else
        pPanel:Label_SetText("txtValueAdd", szSubValue)
      end
    end)
  end
end

function tbWnd:UpdateConsumeInfoEquip(nCurLevel, nNextLevel)
  if nNextLevel <= nCurLevel then
    self.pPanel:SetActive(tbControls.txtCostEquipWns, false)
    self.pPanel:SetActive(tbControls.btnLevelUpEquip, false)
    self.pPanel:SetActive(tbControls.txtMaxEquip, true)
    local szTimeFrame = Horse:GetNextLevelTimeframe(3)
    local nNextWorldLevel = Calendar:GetWorldLevelByTimeframe(szTimeFrame)
    if nNextWorldLevel > 0 then
      self.pPanel:Label_SetText(tbControls.txtMaxEquip, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\230\156\128\233\171\152\233\152\182\239\188\140\229\188\128\230\148\190{0}\231\173\137\231\186\167\228\184\138\233\153\144\229\144\142\229\143\175\231\187\167\231\187\173\229\141\135\233\152\182\239\188\129", Lib:GetLevelText(nNextWorldLevel)))
    else
      self.pPanel:Label_SetText(tbControls.txtMaxEquip, i18n.EasyGet("\229\183\178\229\141\135\233\152\182\229\136\176\228\184\138\233\153\144\231\173\137\233\152\182"))
    end
  else
    self.pPanel:SetActive(tbControls.txtCostEquipWns, true)
    self.pPanel:SetActive(tbControls.btnLevelUpEquip, true)
    self.pPanel:SetActive(tbControls.txtMaxEquip, false)
    local tbCfg = Horse.tbEquipAttrCfg[self.nSelectEquip][nNextLevel]
    self.nConsumeTemplateIdEquip = tbCfg[2]
    self.tbConsumeGridEquip:SetItemByTemplate(self.nConsumeTemplateIdEquip)
    local nCurNum = me.GetItemCountByTypeInBag(self.nConsumeTemplateIdEquip)
    self.pPanel:Label_SetText(tbControls.txtCostEquip, nCurNum .. "/" .. tbCfg[3])
    if nCurNum >= tbCfg[3] then
      self.pPanel:Label_SetColorByName(tbControls.txtCostEquip, "Enough")
    else
      self.pPanel:Label_SetColorByName(tbControls.txtCostEquip, "NotEnough")
    end
    local nNeedCoin = tbCfg[1]
    self.pPanel:Label_SetText(tbControls.txtFeeValueEquip, nNeedCoin)
    if nNeedCoin <= me.GetMoney("Coin") then
      self.pPanel:Label_SetColorByName(tbControls.txtFeeValueEquip, "Enough")
    else
      self.pPanel:Label_SetColorByName(tbControls.txtFeeValueEquip, "NotEnough")
    end
  end
end

function tbWnd:UpdateHorseSkillInfo()
  for nIdx, szImgSkillIconKey in pairs(tbControls.tbImageSkillKey) do
    self.pPanel:Sprite_SetSprite(szImgSkillIconKey, HorseSkillMgr.tbHorseSkillIconPath[nIdx])
    if nIdx == self.nSelectSkill then
      self.pPanel:SetActive(tbControls.tbSkillSelectKey[nIdx], true)
      self.pPanel:Label_SetText(tbControls.txtSkillName, i18n.Get(HorseSkillMgr.tbHorseSkillName[nIdx]))
      self.pPanel:Sprite_SetSprite(tbControls.Img_Skill, HorseSkillMgr.tbHorseSkillIconPath[nIdx])
    else
      self.pPanel:SetActive(tbControls.tbSkillSelectKey[nIdx], false)
    end
  end
  local tbSeriesList = HorseSkillMgr:StatSeries(me)
  local nSkillId = HorseSkillMgr:GetStoneSkillId(tbSeriesList)
  if nSkillId > 0 then
    self.pPanel:SetActive(tbControls.ImgPassiveSkillBG, true)
    self.pPanel:SetActive(tbControls.imgSkillEmpty, false)
    local szIcon = FightSkill:GetSkillIconAtlas(nSkillId)
    self.pPanel:Sprite_SetSprite(tbControls.ImgPassiveSkill, szIcon)
  else
    self.pPanel:SetActive(tbControls.ImgPassiveSkillBG, false)
    self.pPanel:SetActive(tbControls.imgSkillEmpty, true)
  end
  self:UpdateSkillAttrInfo()
end

function tbWnd:UpdateSkillAttrInfo()
  local nSkillIndex = self.nSelectSkill
  local tbAttrValue = HorseSkillMgr:GetPerSkillAllAttr(nSkillIndex)
  self.tbSkillAttrScroll:Reset()
  if not tbAttrValue then
    return
  end
  for nAttrIndex, tbV in pairs(tbAttrValue) do
    self.tbSkillAttrScroll:SetElement(nAttrIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbV.szAttrType, tbV.tbValue)
      pPanel:Label_SetText("txtAttrib", szAttribMsg)
      pPanel:Label_SetText("txtValue", szValue)
      local nQuality = HorseSkillMgr:GetSkillAttrMinQuality(nSkillIndex, nAttrIndex)
      pPanel:Label_SetColorByName("txtAttrib", HorseSkillMgr.tbQuality2Color[nQuality])
      pPanel:Label_SetColorByName("txtValue", HorseSkillMgr.tbQuality2Color[nQuality])
    end)
  end
end

function tbWnd:OnSkillSelect(nIndex)
  self.nSelectSkill = nIndex
  self:UpdateHorseSkillInfo()
end
