local tbHorseSkillStoneSeed = Item:GetClass("HorseSkillStoneSeed")
tbHorseSkillStoneSeed.EXT_PARAM_ITEM_ID = 1
tbHorseSkillStoneSeed.EXT_PARAM_FM_ATTR_COUNT = 2

function tbHorseSkillStoneSeed:OnCreate()
end

function tbHorseSkillStoneSeed:OnInit()
end

function tbHorseSkillStoneSeed:GetUseSetting(nTemplateId, nItemId, szOpt)
  if not nItemId then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local tbRet = {}
  tbRet.szFirstName = i18n.EasyGet("\233\137\180\229\174\154")
  
  function tbRet.fnFirst()
    if pItem.nCount > 1 then
      Ui:OpenWindow("UIItemBatchPanel", nItemId, nTemplateId, "Item")
    else
      Item:ClientUseItem(nItemId)
    end
    Ui:CloseWindow("UIItemTips")
  end
  
  return tbRet
end

function tbHorseSkillStoneSeed:OnUse(pItem)
  local nLeftBag = me.GetFreeBagCount()
  if nLeftBag < 1 then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161\239\188\140\232\175\183\229\133\136\230\184\133\231\144\134\232\131\140\229\140\133"))
    return 0
  end
  if HorseSkillMgr:DoIdentifyHorseSkillStone(me, pItem.dwTemplateId, pItem.dwTradeLimitType, pItem.dwTradeLimitValue) then
    Award:NotifyAwardShow(me.dwID)
    return 1
  end
  return 0
end

function tbHorseSkillStoneSeed:OnUseMany(pItem, nCount)
  local nLeftBag = me.GetFreeBagCount()
  if nCount > nLeftBag then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161\239\188\140\232\175\183\229\133\136\230\184\133\231\144\134\232\131\140\229\140\133"))
    return 0
  end
  local nUseNum = 0
  for i = 1, nCount do
    if HorseSkillMgr:DoIdentifyHorseSkillStone(me, pItem.dwTemplateId, pItem.dwTradeLimitType, pItem.dwTradeLimitValue) then
      nUseNum = nUseNum + 1
    end
  end
  Award:NotifyAwardShow(me.dwID)
  return nUseNum
end
