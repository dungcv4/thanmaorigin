local tbWnd = Ui:CreateClass("UIwndHorsePassiveSkills", "UIPanel")
local tbControls = {
  NotActive = "imgBG/NotActive",
  info = "imgBG/info",
  txtSkillName = "imgBG/info/txtSkillName",
  Img_Skill = "imgBG/info/SkillBG/Img_Skill",
  txtAttrib = "imgBG/info/txtAttrib"
}
tbWnd.tbControls = tbControls

function tbWnd:UpdateView(nPassiveSkillId, nPassiveSkillGrade, nPassSkillShowLevel)
  if nPassiveSkillId and nPassiveSkillId > 0 then
    self.pPanel:SetActive(tbControls.info, true)
    self.pPanel:SetActive(tbControls.NotActive, false)
    local szSkillName = FightSkill:GetSkillName(nPassiveSkillId)
    self.pPanel:Label_SetText(tbControls.txtSkillName, i18n.Parse(szSkillName))
    local szSkillIcon = FightSkill:GetSkillIconAtlas(nPassiveSkillId)
    self.pPanel:Sprite_SetSprite(tbControls.Img_Skill, szSkillIcon)
    local szText = i18n.EasyGet("\230\138\128\232\131\189\231\173\137\231\186\167\239\188\154{0}\n", nPassSkillShowLevel)
    local nRealLevel = HorseSkillMgr.tbPassiveSkillLevelCfg[nPassiveSkillGrade][nPassSkillShowLevel][2]
    local szDesc = FightSkill:GetSkillMagicDesc(nPassiveSkillId, nRealLevel)
    self.pPanel:Label_SetText(tbControls.txtAttrib, szText .. i18n.Parse(szDesc))
  else
    self.pPanel:SetActive(tbControls.info, false)
    self.pPanel:SetActive(tbControls.NotActive, true)
  end
end
