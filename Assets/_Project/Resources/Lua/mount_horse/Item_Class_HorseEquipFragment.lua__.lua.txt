local tbItemClass = Item:GetClass("HorseEquipFragment")

function tbItemClass:OnClientUse(pItem)
  local bRet, szMsg = Horse:IsHorseOpen(me)
  if not bRet then
    me.CenterMsg(szMsg, false)
    return -1
  end
  local nEquipPos = KItem.GetItemExtParam(pItem.dwTemplateId, 1)
  local pEquip = Item.ExtEquip:GetEquipObj(me, nEquipPos)
  if not pEquip or pEquip.dwId <= 0 then
    me.CenterMsg(i18n.EasyGet("\229\176\154\230\156\170\231\169\191\230\136\180\232\163\133\229\164\135"), false)
    return -1
  end
  Ui:OpenWindow("UIBag", 4, 2)
  return 0
end
