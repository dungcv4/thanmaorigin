local tbHorse = Item:GetClass("horse")

function tbHorse:OnCreate(pEquip)
  for i = 0, Item.EQUIP_BASE_ATTRIB_COUNT - 1 do
    pEquip.SetBaseAttribRange(i, MathRandom(0, Item.EQUIP_BASE_ATTRIB_RANGE_ACCURACY))
  end
end

function tbHorse:OnInit(pEquip)
  local nValue = pEquip.nOrgValue
  local nFightPower = pEquip.nBaseFightPower
  pEquip.nFightPower = nFightPower
  pEquip.SetSingleValue(nValue)
end

function tbHorse:GetHorseAttrib(nTemplateId)
  local attribList = {}
  local tbAttrib = KItem.GetEquipBaseProp(nTemplateId).tbBaseAttrib
  if not tbAttrib then
    return
  end
  for i, v in ipairs(tbAttrib) do
    local attrName = v.szName
    local szAttribMsg = FightSkill:GetMagicDesc(attrName, v.tbValue)
    local attribDescData = {}
    attribDescData.attribDesc = szAttribMsg
    table.insert(attribList, attribDescData)
  end
  return attribList
end

function tbHorse:GetTip(pEquip, pPlayer, bIsCompare)
  local tbtips = self:FormatTips(pEquip.dwTemplateId, pEquip)
  return tbtips
end

function tbHorse:GetTipByTemplate(nTemplateId, tbLinkInfo)
  local tbtips = self:FormatTips(nTemplateId, _)
  return tbtips
end

function tbHorse:FormatTips(nTemplateId, pEquip)
  local tbTips = {}
  local tbBaseProp = Item:GetItemBaseProp(nTemplateId)
  if tbBaseProp.szIntro then
    local szIntro = i18n.Parse(tbBaseProp.szIntro)
    local szColor = "NewBlue"
    table.insert(tbTips, {
      nil,
      szColor,
      nil,
      nil,
      szIntro
    })
  end
  table.insert(tbTips, {
    i18n.Get(Item.Equip.tbText.szBaseAttribute)
  })
  local tbAttribSpeed = KItem.GetEquipBaseProp(nTemplateId).tbBaseAttrib
  local szDesc = FightSkill:GetMagicDesc(tbAttribSpeed[1].szName, tbAttribSpeed[1].tbValue)
  table.insert(tbTips, {
    {szDesc}
  })
  local tbAttrib = Horse.tbGradeInfoCfg[tbBaseProp.nLevel][100]
  for k, tbMA in pairs(tbAttrib) do
    local szDesc = FightSkill:GetMagicDesc(tbMA[1], {
      table.unpack(tbMA, 2)
    })
    table.insert(tbTips, {
      {szDesc}
    })
  end
  if tbBaseProp.szSource then
    local szSource = i18n.Parse(tbBaseProp.szSource)
    szSource = i18n.EasyGet("\232\142\183\229\143\150\233\128\148\229\190\132\239\188\154{0}", szSource)
    local szColor = "NewBlue"
    table.insert(tbTips, {
      nil,
      szColor,
      nil,
      nil,
      szSource
    })
  end
  return tbTips
end
