Horse.SAVE_HORSE_GROUP = GetGroupKey("Horse")
Horse.HORSE_GRADE = 1
Horse.HORSE_LEVEL = 2
Horse.HORSE_CURRENT_FEATURE = 3
Horse.nHorseEquipCount = 5

function Horse:GetHorseLevel(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_HORSE_GROUP, self.HORSE_LEVEL) + 1
end

function Horse:GetHorseGrade(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_HORSE_GROUP, self.HORSE_GRADE) + 1
end

function Horse:Init()
  self.tbHorseCfg = Lib:LoadTabFileEx("Setting/Item/Equip/Horse.tab", "dsdddsdss", "TemplateId", {
    "TemplateId",
    "Name",
    "ShowType",
    "Icon",
    "Quality",
    "Attrib1Type",
    "Attrib1Value1",
    "Intro",
    "Source"
  })
  self.tbExchangeCfg = Lib:LoadTabFileEx("Setting/Item/Equip/HorseExchangeCfg.tab", "dddd", "ORIGIN_ITEM_ID", {
    "ORIGIN_ITEM_ID",
    "EXCHANGE_ITEM_ID",
    "EXCHANGE_ITEM_NUM",
    "EXCHANGE_YUANBAO_NUM"
  })
end

function Horse:GetHorseShowType(nTemplateId)
  local tbInfo = Item:GetItemBaseProp(nTemplateId)
  if tbInfo.nItemType == Item.EQUIP_HORSE then
    return self.tbHorseCfg[nTemplateId].ShowType
  else
    return KItem.GetItemExtParam(nTemplateId, 1)
  end
end

function Horse:GetModel(pPlayer, nTemplateId)
  local nShowType = self:GetHorseShowType(nTemplateId)
  local nResId = KItem.GetResIdByShowType(nShowType, pPlayer.nFaction, pPlayer.nFactionSect, pPlayer.nSex)
  return nResId
end

Horse:Init()

function Horse:IsHaveHorseFeature(pPlayer, nTemplateId)
  local tbHorseData = self:GetHorseScriptData(pPlayer)
  if not tbHorseData.tbHorseFeature then
    return false
  end
  if tbHorseData.tbHorseFeature[nTemplateId] then
    return true
  end
  return false
end

function Horse:GetMaxGradeNumCfg()
  return #self.tbGradeInfoCfg
end

function Horse:GetMaxLevelNumCfg()
  return #self.tbLevelAttrCfg
end

function Horse:GetMaxEquipLevel(nIndex)
  return #self.tbEquipAttrCfg[nIndex]
end

function Horse:IsHorseOpen(pPlayer)
  pPlayer = pPlayer or me
  local pHorseItem = pPlayer.GetEquipByPos(Item.EQUIPPOS_HORSE)
  if not pHorseItem then
    return false, i18n.EasyGet("\230\156\170\232\163\133\229\164\135\229\157\144\233\170\145")
  end
  if 1 ~= GetTimeFrameState("OpenLevel49") then
    return false, i18n.EasyGet("\231\179\187\231\187\159\229\176\154\230\156\170\229\188\128\229\144\175")
  end
  return true
end

function Horse:GetMaxCanUpgrade()
  local nMaxLevel, nMaxGrade, nMaxEquipLevel = 0, 0, 0
  for k, v in ipairs(self.tbTimeFrameMaxLevel) do
    if 1 == GetTimeFrameState(v.szTimeFrame) then
      nMaxLevel = v.nLevel
      nMaxGrade = v.nGrade
      nMaxEquipLevel = v.nEquipGrade
    end
  end
  return nMaxGrade, nMaxLevel, nMaxEquipLevel
end

function Horse:GetNextLevelTimeframe(STType)
  local nMaxLevel = 0
  for k, v in ipairs(self.tbTimeFrameMaxLevel) do
    if 1 == GetTimeFrameState(v.szTimeFrame) then
      if 1 == STType then
        nMaxLevel = v.nGrade
      elseif 2 == STType then
        nMaxLevel = v.nLevel
      else
        nMaxLevel = v.nEquipGrade
      end
    elseif 1 == STType then
      if nMaxLevel < v.nGrade then
        return v.szTimeFrame
      end
    elseif 2 == STType then
      if nMaxLevel < v.nLevel then
        return v.szTimeFrame
      end
    elseif nMaxLevel < v.nEquipGrade then
      return v.szTimeFrame
    end
  end
  return ""
end

function Horse:CanUpgradeLevel(pPlayer)
  local bRet, szErr = self:IsHorseOpen(pPlayer)
  if not bRet then
    return bRet, szErr
  end
  local nNextLevel = self:GetHorseLevel(pPlayer) + 1
  local _, nMaxLevel = self:GetMaxCanUpgrade()
  if nNextLevel > nMaxLevel then
    return false, i18n.EasyGet("\229\183\178\232\190\190\228\184\138\233\153\144\231\173\137\231\186\167")
  end
  local tbCfg = self.tbLevelAttrCfg[nNextLevel]
  local nCurNum = pPlayer.GetItemCountByTypeInBag(tbCfg[1])
  local nNeedNum = self:GetUpgradeLevelConsumeNum(nNextLevel)
  if nCurNum < nNeedNum then
    return false, i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179")
  end
  return true
end

function Horse:GetUpgradeLevelConsumeNum(nLevel)
  local nTFIndex = 1
  for i = #self.tbLevelTimeframe, 1, -1 do
    if 1 == GetTimeFrameState(self.tbLevelTimeframe[i]) then
      nTFIndex = i
      break
    end
  end
  local nRet = 0
  for i = #self.tbLevel2ConsumeNum, 1, -1 do
    if nLevel <= self.tbLevel2ConsumeNum[i][1] then
      nRet = self.tbLevel2ConsumeNum[i][1 + nTFIndex]
    end
  end
  return nRet
end

function Horse:CanUpgradeGrade(pPlayer, nCurGrade, nNextGrade)
  local bRet, szErr = self:IsHorseOpen(pPlayer)
  if not bRet then
    return bRet, szErr
  end
  if pPlayer.IsMaskPlayerAttr() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\228\184\141\229\143\175\229\141\135\233\152\182")
  end
  if nNextGrade <= nCurGrade then
    return false, i18n.EasyGet("\232\175\183\233\128\137\230\139\169\229\164\167\228\186\142\229\189\147\229\137\141\231\154\132\233\152\182\230\149\176")
  end
  local tbNeedHorseItem, tbNeedGradeItem, nNeedCoin = self:GetUpgradeGradeConsumeInfo(nCurGrade, nNextGrade)
  if not (tbNeedHorseItem and tbNeedGradeItem) or not nNeedCoin then
    return false
  end
  if nNeedCoin > pPlayer.GetMoney("Coin") then
    return false, i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179")
  end
  local nCurNum = pPlayer.GetItemCountByTypeInBag(tbNeedHorseItem[1])
  if nCurNum < tbNeedHorseItem[2] then
    return false, i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179")
  end
  nCurNum = pPlayer.GetItemCountByTypeInBag(tbNeedGradeItem[1])
  if nCurNum < tbNeedGradeItem[2] then
    return false, i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179")
  end
  local nMaxGrade, _ = self:GetMaxCanUpgrade()
  if nNextGrade > nMaxGrade then
    return false, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\229\189\147\229\137\141\230\156\128\233\171\152\233\152\182")
  end
  local tbCfg = self.tbGradeInfoCfg[nNextGrade]
  if pPlayer.nLevel < tbCfg[2] then
    return false, i18n.EasyGet("\232\167\146\232\137\178\231\173\137\231\186\167\228\184\141\232\182\179\239\188\140\233\156\128\232\190\190\229\136\176{0}\231\186\167", tbCfg[2])
  end
  return true
end

function Horse:GetUpgradeGradeConsumeInfo(nCurGrade, nNextGrade)
  local tbCfg = self.tbGradeInfoCfg[nNextGrade]
  if not tbCfg then
    return
  end
  local tbNeedHorseItem = {
    tbCfg[4],
    tbCfg[5]
  }
  local tbNeedGradeItem = {
    self.HorseCfg.nGradeItemId,
    0
  }
  local nNeedCoin = 0
  for i = nCurGrade + 1, nNextGrade do
    nNeedCoin = nNeedCoin + self.tbGradeInfoCfg[i][3]
    tbNeedGradeItem[2] = tbNeedGradeItem[2] + self.tbGradeInfoCfg[i][7]
    if i < nNextGrade then
      tbNeedGradeItem[2] = tbNeedGradeItem[2] + self.tbGradeInfoCfg[i][6]
    end
  end
  return tbNeedHorseItem, tbNeedGradeItem, nNeedCoin
end

function Horse:GetEquipPosByIndex(nIndex)
  return nIndex + Item.ExtEquip.CFG.SLOT.HORSEREIN - 1
end

function Horse:CanEquipUpgradeLevel(pPlayer, nIndex)
  local bRet, szErr = self:IsHorseOpen(pPlayer)
  if not bRet then
    return bRet, szErr
  end
  if pPlayer.IsMaskPlayerAttr() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\228\184\141\229\143\175\229\141\135\233\152\182")
  end
  local nEquipPos = self:GetEquipPosByIndex(nIndex)
  if nEquipPos < Item.ExtEquip.CFG.SLOT.HORSEREIN or nEquipPos > Item.ExtEquip.CFG.SLOT.HORSESHOE then
    return false
  end
  local pEquip = Item.ExtEquip:GetEquipObj(pPlayer, nEquipPos)
  if not pEquip or pEquip.dwId <= 0 then
    return false, i18n.EasyGet("\229\176\154\230\156\170\231\169\191\230\136\180\232\163\133\229\164\135")
  end
  local _, _, nMaxLevelCfg = self:GetMaxCanUpgrade()
  local nMaxLevel = Horse:GetMaxEquipLevel(nIndex)
  if nMaxLevel <= pEquip.nLevel then
    return false, i18n.EasyGet("\229\183\178\230\187\161\233\152\182")
  end
  if nMaxLevelCfg <= pEquip.nLevel then
    return false, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\229\189\147\229\137\141\230\156\128\233\171\152\233\152\182")
  end
  local tbCfg = Horse.tbEquipAttrCfg[nIndex][pEquip.nLevel + 1]
  if not tbCfg then
    return false
  end
  local nCurNum = pPlayer.GetItemCountByTypeInBag(tbCfg[2])
  if nCurNum < tbCfg[3] then
    return false, i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179")
  end
  if pPlayer.GetMoney("Coin") < tbCfg[1] then
    return false, i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179")
  end
  return true
end

function Horse:ChangeGradeAttr(pNpc, nGrade, bIsApply, nPlayerId)
  if not nGrade or nGrade <= 0 then
    nGrade = 1
  end
  local tbAttr = self.tbGradeInfoCfg[nGrade][100]
  for k, tbV in pairs(tbAttr) do
    if bIsApply then
      pNpc.ApplyMagicAttrib(table.unpack(tbV))
      if nPlayerId then
        AttributeHelp:LogAttrApply(nPlayerId, "Horse", table.unpack(tbV))
      end
    else
      pNpc.RemoveMagicAttrib(table.unpack(tbV))
      if nPlayerId then
        AttributeHelp:LogAttrRemove(nPlayerId, "Horse", table.unpack(tbV))
      end
    end
  end
end

function Horse:RefreshGradeAttrib(pNpc, nOrgGrade, nNewGrade, nPlayerId)
  self:ChangeGradeAttr(pNpc, nOrgGrade, false, nPlayerId)
  self:ChangeGradeAttr(pNpc, nNewGrade, true, nPlayerId)
end

function Horse:ChangeLevelAttr(pNpc, nLevel, bIsApply, nPlayerId)
  if not nLevel or nLevel <= 0 then
    nLevel = 1
  end
  local tbAttr = self.tbLevelAttrCfg[nLevel][4]
  for nIndex, szAttr in pairs(self.tbLevelAttrName) do
    if bIsApply then
      pNpc.ApplyMagicAttrib(szAttr, table.unpack(tbAttr[nIndex]))
      if nPlayerId then
        AttributeHelp:LogAttrApply(nPlayerId, "Horse", szAttr, table.unpack(tbAttr[nIndex]))
      end
    else
      pNpc.RemoveMagicAttrib(szAttr, table.unpack(tbAttr[nIndex]))
      if nPlayerId then
        AttributeHelp:LogAttrRemove(nPlayerId, "Horse", szAttr, table.unpack(tbAttr[nIndex]))
      end
    end
  end
end

function Horse:RefreshLevelAttrib(pPlayer, nOrgLevel, nNewLevel)
  if pPlayer.IsMaskPlayerAttr() then
    return
  end
  self:ChangeLevelAttr(pPlayer.GetNpc(), nOrgLevel, false, pPlayer.dwID)
  self:ChangeLevelAttr(pPlayer.GetNpc(), nNewLevel, true, pPlayer.dwID)
end

function Horse:ApplyHorseAttr(pPlayer, pEquip)
  if pEquip.nEquipPos ~= Item.EQUIPPOS_HORSE then
    return
  end
  local nGrade = self:GetHorseGrade(pPlayer)
  self:ChangeGradeAttr(pPlayer.GetNpc(), nGrade, true, pPlayer.dwID)
  local nLevel = self:GetHorseLevel(pPlayer)
  self:ChangeLevelAttr(pPlayer.GetNpc(), nLevel, true, pPlayer.dwID)
  local nFashionId = Fashion:GetApplyAttrFashionId(pPlayer, Npc.NpcResPartsDef.npc_part_horse)
  self:ApplyHorseFashionAttr(pPlayer.GetNpc(), pPlayer.nRideState, nFashionId, pPlayer.dwID)
end

function Horse:RemoveHorseAttr(pPlayer, pEquip)
  if pEquip.nEquipPos ~= Item.EQUIPPOS_HORSE then
    return
  end
  local nGrade = self:GetHorseGrade(pPlayer)
  self:ChangeGradeAttr(pPlayer.GetNpc(), nGrade, false, pPlayer.dwID)
  local nLevel = self:GetHorseLevel(pPlayer)
  self:ChangeLevelAttr(pPlayer.GetNpc(), nLevel, false, pPlayer.dwID)
  local nFashionId = Fashion:GetApplyAttrFashionId(pPlayer, Npc.NpcResPartsDef.npc_part_horse)
  self:RemoveHorseFashionAttr(pPlayer.GetNpc(), pPlayer.nRideState, nFashionId, true, pPlayer.dwID)
end

function Horse:OnRideStateChange(pPlayer)
  if MODULE_GAMECLIENT then
    pPlayer = me
  end
  local nFashionId = Fashion:GetApplyAttrFashionId(pPlayer, Npc.NpcResPartsDef.npc_part_horse)
  if 1 == pPlayer.nRideState then
    self:ApplyHorseFashionAttr(pPlayer.GetNpc(), pPlayer.nRideState, nFashionId, pPlayer.dwID)
  else
    self:RemoveHorseFashionAttr(pPlayer.GetNpc(), pPlayer.nRideState, nFashionId, false, pPlayer.dwID)
  end
  if MODULE_GAMESERVER then
    pPlayer.CallClientScript("Horse:OnRideStateChange")
  end
end

function Horse:GetHorseAllFightPower(pPlayer)
  local nFPNum = 0
  nFPNum = nFPNum + self:CalcFightPower(pPlayer, "HorseGrade")
  nFPNum = nFPNum + self:CalcFightPower(pPlayer, "HorseLevel")
  nFPNum = nFPNum + HorseEquipMgr:CalcFightPower(pPlayer, "HorseEquip")
  nFPNum = nFPNum + HorseSkillMgr:CalcFightPower(pPlayer, "HorseSkill")
  return nFPNum
end

function Horse:ApplyHorseFashionAttr(pNpc, nRideState, nFashionId, nPlayerId)
  if not Fashion.tbHorseFashionFlag[nFashionId] then
    return
  end
  local tbAttr = Fashion.AttrCfg[nFashionId]
  if not tbAttr then
    return
  end
  if 1 == nRideState then
    pNpc.ApplyMagicAttrib(table.unpack(tbAttr))
    if nPlayerId then
      AttributeHelp:LogAttrApply(nPlayerId, "Horse", table.unpack(tbAttr))
    end
  end
end

function Horse:RemoveHorseFashionAttr(pNpc, nRideState, nFashionId, bIsTakeOff, nPlayerId)
  if not Fashion.tbHorseFashionFlag[nFashionId] then
    return
  end
  local tbAttr = Fashion.AttrCfg[nFashionId]
  if not tbAttr then
    return
  end
  if bIsTakeOff then
    if 1 == nRideState then
      pNpc.RemoveMagicAttrib(table.unpack(tbAttr))
      if nPlayerId then
        AttributeHelp:LogAttrRemove(nPlayerId, "Horse", table.unpack(tbAttr))
      end
    end
  elseif 0 == nRideState then
    pNpc.RemoveMagicAttrib(table.unpack(tbAttr))
    if nPlayerId then
      AttributeHelp:LogAttrRemove(nPlayerId, "Horse", table.unpack(tbAttr))
    end
  end
end

function Horse:OnCreatePlayerRobotNpc(pPlayerAsync, pNpc)
  self:ChangeGradeAttr(pNpc, pPlayerAsync.GetHorseGrade(), true)
  self:ChangeLevelAttr(pNpc, pPlayerAsync.GetHorseLevel(), true)
  local nFashionId = pPlayerAsync.GetFashion(Npc.NpcResPartsDef.npc_part_horse + Fashion.APPLY_ATTR_ID_OFFSET + 1)
  if nFashionId and nFashionId > 0 then
    self:ApplyHorseFashionAttr(pNpc, 1, nFashionId)
  end
end

function Horse:CanJinJieMaterialExchange(pPlayer, nItemId)
  if type(nItemId) ~= "number" then
    return false
  end
  local pItem = pPlayer.GetItemInBag(nItemId)
  if not pItem then
    return false
  end
  if pItem.nCount < 1 then
    return false
  end
  local tbSetting = self.tbExchangeCfg[pItem.dwTemplateId]
  if not tbSetting then
    return false
  end
  return true
end

function Horse:CalcHorseGradeFightPower(pPlayer)
  local nRet = 0
  local pHorseItem = pPlayer.GetEquipByPos(Item.EQUIPPOS_HORSE)
  if pHorseItem then
    nRet = pHorseItem.nFightPower
  end
  return nRet
end

function Horse:CalcHorseLevelFightPower(pPlayer)
  local nRet = 0
  local pHorseItem = pPlayer.GetEquipByPos(Item.EQUIPPOS_HORSE)
  if pHorseItem then
    local nLevel = Horse:GetHorseLevel(pPlayer)
    nRet = Horse.tbLevelAttrCfg[nLevel][3]
  end
  return nRet
end

function Horse:CalcFightPower(pPlayer, szType)
  local strFunc = string.format("Calc%sFightPower", szType)
  if Horse[strFunc] then
    return Horse[strFunc](Horse, pPlayer)
  end
end
