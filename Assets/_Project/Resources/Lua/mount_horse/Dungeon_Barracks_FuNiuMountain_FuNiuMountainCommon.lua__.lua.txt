function FuNiuMountain:CheckCanShow()
  local bEnoughLevel = me.nLevel >= FuNiuMountain.nLevelLimit
  local szOpenTimeFrame = FuNiuMountain.szOpenTimeFrame
  local szCloseTimeFrame = FuNiuMountain.szCloseTimeFrame
  local bIsOpen = true
  if szOpenTimeFrame and "" ~= szOpenTimeFrame then
    bIsOpen = 1 == GetTimeFrameState(szOpenTimeFrame)
  end
  local bNotClose = true
  if szCloseTimeFrame and "" ~= szCloseTimeFrame then
    bNotClose = 1 ~= GetTimeFrameState(szCloseTimeFrame)
  end
  return bEnoughLevel and bIsOpen and bNotClose
end

function FuNiuMountain:LoadNpcData()
  local tbRet = {}
  local tbNpcPosSetting = Lib:LoadTabFileEx(self.fileNpcSetting, "ssdd", nil, {
    "Name",
    "VarName",
    "PosX",
    "PosY"
  }, nil, nil, false)
  if not tbNpcPosSetting then
    error("cannot read " .. self.fileNpcSetting)
  else
    for nIndex, tbNpcInfo in ipairs(tbNpcPosSetting) do
      local szVarName = tbNpcInfo.VarName
      table.insert(tbRet, {
        szName = szVarName,
        nNpcTemplateId = FuNiuMountain.tbNpcConfig[szVarName].nNpcTemplateId,
        nPosX = tbNpcInfo.PosX,
        nPosY = tbNpcInfo.PosY,
        nReviveTime = FuNiuMountain.tbNpcConfig[szVarName].nReviveTime,
        nIndex = nIndex
      })
    end
  end
  tbRet = SetReadOnly(tbRet)
  return tbRet
end

function FuNiuMountain:LoadBatchStartData()
  local tbRet = {}
  for i = self.tbEachStepEnum.Start, self.tbEachStepEnum.End do
    tbRet[i] = {}
  end
  for _, tbEachNpc in ipairs(self.tbNpcSetting) do
    if tbEachNpc.szName == "GrandArtisan" then
      table.insert(tbRet[FuNiuMountain.tbEachStepEnum.DefeatGrandArtisan], tbEachNpc)
    else
      table.insert(tbRet[FuNiuMountain.tbEachStepEnum.Start], tbEachNpc)
    end
  end
  tbRet = SetReadOnly(tbRet)
  return tbRet
end

function FuNiuMountain:LoadTrapPointPos()
  local tbRet = {}
  local tbTrapPoint = Lib:LoadTabFileEx(FuNiuMountain.fileTrapPointPos, "sddd", nil, {
    "Name",
    "Index",
    "PosX",
    "PosY"
  }, nil, nil, false)
  if not tbTrapPoint then
    error("cannot read " .. FuNiuMountain.fileTrapPointPos)
  else
    for _, tbTrapPointInfo in ipairs(tbTrapPoint) do
      tbRet[tbTrapPointInfo.Index] = {
        tbTrapPointInfo.PosX,
        tbTrapPointInfo.PosY
      }
    end
  end
  tbRet = SetReadOnly(tbRet)
  return tbRet
end
