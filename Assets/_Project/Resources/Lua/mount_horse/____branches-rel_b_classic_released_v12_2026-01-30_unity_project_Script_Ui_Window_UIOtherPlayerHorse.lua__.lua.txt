local tbControls = {
  LevelUp = "Right/LevelUp",
  txtNowLevel = "Right/LevelUp/txtNow",
  TemplateLevel = "Right/LevelUp/AttribList/Template",
  HorseEquip = "Right/HorseEquip",
  txtEquipName = "Right/HorseEquip/txtEquipName",
  UIItemGridEquip = "Right/HorseEquip/UIItemGrid",
  txtNowEquip = "Right/HorseEquip/txtNow",
  TemplateEquip = "Right/HorseEquip/AttribList/Template",
  txtHorse = "Left/txtHorse",
  txtFightPower = "Left/txtFightPower",
  NpcModel = "Left/imgModel",
  imgArrowRight = "Left/imgArrow",
  imgArrowLeft = "Left/imgArrow1",
  EquipList = "Left/EquipList",
  jiangsheng = "Left/EquipList/jiangsheng",
  majue = "Left/EquipList/majue",
  maan = "Left/EquipList/maan",
  madeng = "Left/EquipList/madeng",
  mazhang = "Left/EquipList/mazhang",
  btnLevelUp = "Right/btnList/btnLevelUp",
  btnHorseEquip = "Right/btnList/btnHorseEquip",
  HorseSkill = "Right/HorseSkill",
  btnHorseSkill = "Right/btnList/btnHorseSkill",
  SkillList = "Left/SkillList",
  btnLook = "Left/SkillList/passiveSkills/btnLook",
  Img_Skill = "Right/HorseSkill/infoCell/Img_SkillBg/Img_Skill",
  txtSkillName = "Right/HorseSkill/infoCell/txtSkillName",
  ImgPassiveSkillBG = "Left/SkillList/passiveSkills/SkillBG",
  ImgPassiveSkill = "Left/SkillList/passiveSkills/SkillBG/Img_Skill",
  imgSkillEmpty = "Left/SkillList/passiveSkills/imgSkillEmpty",

tbControls.tbBtnSkillKey = {}
tbControls.tbImageSkillKey = {}
tbControls.tbSkillSelectKey = {}

  tbControls.tbBtnSkillKey[nIndex] = string.format("Left/SkillList/Skill%s", nIndex)
  tbControls.tbImageSkillKey[nIndex] = string.format("Left/SkillList/Skill%s/Img_Skill", nIndex)
  tbControls.tbSkillSelectKey[nIndex] = string.format("Left/SkillList/Skill%s/imgSelected", nIndex)
end
tbControls.tbSkillAttrNameKey = {}
tbControls.tbSkillAttrValueKey = {}
for nIndex = 1, HorseSkillMgr.nMaxSkillNum do
  tbControls.tbSkillAttrNameKey[nIndex] = string.format("Right/HorseSkill/infoCell/AttribList/Attr%s/txtAttrib", nIndex)
  tbControls.tbSkillAttrValueKey[nIndex] = string.format("Right/HorseSkill/infoCell/AttribList/Attr%s/txtValue", nIndex)
end
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnLook = function(self)
    self.tbOwner:OpenHorsePassiveSkillActive(self.nPassiveSkillId, self.nPassiveSkillGrade, self.nPassiveSkillShowLevel)
  end
}
tbWnd.nLevelUpView = 1
tbWnd.nHorseEquipView = 2
tbWnd.nHorseSkillView = 3
tbWnd.tbPanelStateDef = {
  [tbWnd.nLevelUpView] = {
    [tbWnd.nLevelUpView] = true,
    [tbWnd.nHorseEquipView] = false,
    [tbWnd.nHorseSkillView] = false
  },
  [tbWnd.nHorseEquipView] = {
    [tbWnd.nLevelUpView] = false,
    [tbWnd.nHorseEquipView] = true,
    [tbWnd.nHorseSkillView] = false
  },
  [tbWnd.nHorseSkillView] = {
    [tbWnd.nLevelUpView] = false,
    [tbWnd.nHorseEquipView] = false,
    [tbWnd.nHorseSkillView] = true
  }
}

function tbWnd:OnCreate()
  local obj = self.pPanel:GetObject(tbControls.NpcModel)
  self.NpcModel = obj:GetComponent(Ui.UINpcModel)
  self.tbEquipPath = {
    tbControls.jiangsheng,
    tbControls.majue,
    tbControls.maan,
    tbControls.madeng,
    tbControls.mazhang
  }
  self.nSelectEquip = 1
  self.nSelectSkill = 1
  self.tbItemEquip = {}
  for i, szPath in ipairs(self.tbEquipPath) do
    self.tbItemEquip[i] = Ui:NewExistItemGrid(self, szPath .. "/UIItemGrid")
    self.tbItemEquip[i]:SetClickEnable(true, function()
      self.pPanel:SetActive(self.tbEquipPath[self.nSelectEquip] .. "/imgSelected", false)
      self.nSelectEquip = i
      self.pPanel:SetActive(szPath .. "/imgSelected", true)
      self:UpdateHorseEquipInfo()
    end)
    self.tbItemEquip[i]:SetEmpty(true)
  end
  self.tbBtnGroup = Ui:NewButtonGroup(self, {
    [1] = tbControls.btnLevelUp,
    [2] = tbControls.btnHorseEquip,
    [3] = tbControls.btnHorseSkill
  }, true, function(self, nIndex)
    self.nViewType = nIndex
    local tbCfg = self.tbPanelStateDef[self.nViewType]
    if tbCfg[tbWnd.nLevelUpView] then
      self.pPanel:SetActive(tbControls.LevelUp, true)
    else
      self.pPanel:SetActive(tbControls.LevelUp, false)
    end
    if tbCfg[tbWnd.nHorseEquipView] then
      self.pPanel:SetActive(tbControls.HorseEquip, true)
      self.pPanel:SetActive(tbControls.EquipList, true)
    else
      self.pPanel:SetActive(tbControls.HorseEquip, false)
      self.pPanel:SetActive(tbControls.EquipList, false)
    end
    if tbCfg[tbWnd.nHorseSkillView] then
      self.pPanel:SetActive(tbControls.HorseSkill, true)
      self.pPanel:SetActive(tbControls.SkillList, true)
    else
      self.pPanel:SetActive(tbControls.HorseSkill, false)
      self.pPanel:SetActive(tbControls.SkillList, false)
    end
  end)
  self.tbSelectEquip = Ui:NewExistItemGrid(self, tbControls.UIItemGridEquip)
  self.tbSelectEquip:SetClickEnable(true)
  self.tbAttrScrollLevel = Ui:NewScrollView(self, tbControls.TemplateLevel, nil, false, true)
  self.tbAttrScrollEquip = Ui:NewScrollView(self, tbControls.TemplateEquip, nil, false, true)
  self.tbSkillAttrScroll = Ui:NewScrollView(self, tbControls.SkillAttrTemplate, nil, false, true)
  for nIndex, szKey in ipairs(tbControls.tbBtnSkillKey) do
    Ui:Button_BindEvent(self.pPanel, szKey, self.OnSkillSelect, self, nIndex)
  end
end

function tbWnd:OnDestroy()
  for nIndex, szKey in ipairs(tbControls.tbBtnSkillKey) do
    Ui:Button_UnBindEvent(self.pPanel, szKey)
  end
end

function tbWnd:OnOpen(nSelect)
  nSelect = nSelect or 1
  self.tbExtEquips = self.tbOwner.tbExtEquips
  for i, szPath in ipairs(self.tbEquipPath) do
    self.tbItemEquip[i]:Clear()
  end
  self.pPanel:SetActive(tbControls.btnHorseEquip, Horse:IsHorseEquipOpen())
  self.pPanel:SetActive(tbControls.btnHorseSkill, HorseSkillMgr:IsHorseSkillOpen(me))
  self.pPanel:SetActive(tbControls.imgArrowLeft, false)
  self.pPanel:SetActive(tbControls.imgArrowRight, false)
  self.nViewType = self.nLevelUpView
  self:ShowModel()
  self.tbBtnGroup:Select(nSelect)
  self:UpdateView()
end

function tbWnd:ShowModel()
  local pAsyncRole = self.tbOwner.pAsyncRole
  local nCurGrade = pAsyncRole.GetHorseGrade()
  if not Horse.tbGradeInfoCfg[nCurGrade] then
    self.pPanel:SetActive(tbControls.txtHorse, false)
    self.NpcModel:Release()
    return
  end
  local szCurName = i18n.Get(Horse.tbGradeInfoCfg[nCurGrade][1])
  self.pPanel:Label_SetText(tbControls.txtHorse, szCurName)
  local nCurTemplateId = Horse.tbGradeInfoCfg[nCurGrade][8]
  local nShowType = Horse:GetHorseShowType(nCurTemplateId)
  local nResId = KItem.GetResIdByShowType(nShowType, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
  self.NpcModel:ShowModel(pAsyncRole.GetSex(), 1, 0, 0, 0, nResId, 0, 0)
  self.NpcModel:SetDir(Npc.ModelDirection.LeftBottom)
end

function tbWnd:GetHorseLevel(pAsyncRole)
  local nLevel = pAsyncRole.GetHorseLevel()
  if 0 == nLevel then
    return 1
  end
  return nLevel
end

function tbWnd:GetHorseAllFightPower()
  local pAsyncRole = self.tbOwner.pAsyncRole
  local nFPNum = 0
  local nLevel = self:GetHorseLevel(pAsyncRole)
  nFPNum = nFPNum + Horse.tbLevelAttrCfg[nLevel][3]
  local nPos = Item.EQUIPPOS_HORSE
  if self.tbOwner.tbEquips[nPos] then
    local pItem = KItem.GetItemObj(self.tbOwner.tbEquips[nPos])
    nFPNum = nFPNum + pItem.nFightPower
  end
  local pEquip, nItemId
  for i = Item.ExtEquip.CFG.SLOT.HORSEREIN, Item.ExtEquip.CFG.SLOT.HORSESHOE do
    nItemId = self.tbOwner.tbExtEquips[i]
    if nItemId and nItemId > 0 then
      pEquip = KItem.GetItemObj(nItemId)
      nFPNum = nFPNum + pEquip.nFightPower
    end
  end
  for nSkillIndex = 1, HorseSkillMgr.nMaxSkillNum do
    for nGrade = 1, HorseSkillMgr.nMaxGrade do
      local nLevel = HorseSkillMgr:GetSyncSkillLevel(pAsyncRole, nSkillIndex, nGrade)
      local nAttrCfgIndex = HorseSkillMgr:GetSkillAttrCfg(nSkillIndex, nGrade)
      if nAttrCfgIndex and nLevel > 0 then
        nFPNum = nFPNum + HorseSkillMgr.tbUpgradeLevelCfg[nAttrCfgIndex][nLevel][2]
      end
    end
  end
  self.tbSeriesList = {}
  local nAllStar = 0
  local nGrade = math.maxinteger
  for nSlot = Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1, Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE5 do
    nItemId = self.tbOwner.tbExtEquips[nSlot]
    local nHole = nSlot - Item.ExtEquip.CFG.SLOT.HORSE_SKILL_STONE1 + 1
    if nItemId and nItemId > 0 then
      pEquip = KItem.GetItemObj(nItemId)
      local nSeries = pEquip.GetSeries()
      self.tbSeriesList[nHole] = nSeries
      nAllStar = nAllStar + pEquip.nQuality - 1
      nGrade = math.min(nGrade, pEquip.nLevel)
    else
      self.tbSeriesList[nHole] = 0
    end
  end
  self.nPassiveSkillGrade = nGrade
  local nSkillId = HorseSkillMgr:GetStoneSkillId(self.tbSeriesList)
  if nSkillId > 0 then
    local nShowLevel = 1
    for i = 2, #HorseSkillMgr.tbPassiveSkillLevelCfg[nGrade] do
      if nAllStar >= HorseSkillMgr.tbPassiveSkillLevelCfg[nGrade][i][1] then
        nShowLevel = i
      end
    end
    self.nPassiveSkillShowLevel = nShowLevel
    nFPNum = nFPNum + HorseSkillMgr.tbPassiveSkillLevelCfg[nGrade][nShowLevel][3]
  end
  return nFPNum
end

function tbWnd:UpdateView()
  self:UpdateHorseEquip()
  self:UpdateHorseLevelUpInfo()
  self:UpdateHorseEquipInfo()
  self:UpdateHorseSkillInfo()
end

function tbWnd:UpdateHorseEquip()
  local pAsyncRole = self.tbOwner.pAsyncRole
  local nItemId, pItem
  for i = Item.ExtEquip.CFG.SLOT.HORSEREIN, Item.ExtEquip.CFG.SLOT.HORSESHOE do
    nItemId = self.tbExtEquips[i]
    if nItemId and nItemId > 0 then
      pItem = KItem.GetItemObj(nItemId)
      if pItem then
        self.tbItemEquip[i]:SetItem(pItem.dwId, pItem.dwTemplateId, {bBigIcon = true}, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
      end
    end
  end
end

function tbWnd:UpdateHorseLevelUpInfo()
  local pAsyncRole = self.tbOwner.pAsyncRole
  local nCurGrade = pAsyncRole.GetHorseGrade()
  if not Horse.tbGradeInfoCfg[nCurGrade] then
    self.pPanel:SetActive(tbControls.txtNowLevel, false)
    self.tbAttrScrollLevel:Reset()
    self.pPanel:SetActive(tbControls.txtFightPower, false)
    return
  end
  local nCurLevel = self:GetHorseLevel(pAsyncRole)
  nCurLevel = nCurLevel > 0 and nCurLevel or 1
  self.pPanel:SetActive(tbControls.txtNowLevel, true)
  self.pPanel:Label_SetText(tbControls.txtNowLevel, i18n.EasyGet("{0}\231\186\167", nCurLevel))
  self:UpdateAttrInfoLevel(nCurLevel)
  local nFightPower = self:GetHorseAllFightPower()
  self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\230\136\152\229\138\155\229\128\188  {0}", nFightPower))
end

function tbWnd:UpdateAttrInfoLevel(nCurLevel)
  local tbAttrCur = Horse.tbLevelAttrCfg[nCurLevel][4]
  self.tbAttrScrollLevel:Reset()
  for i = 1, #Horse.tbLevelAttrName do
    self.tbAttrScrollLevel:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbCurAttrNum = tbAttrCur[i]
      local szDesc, szValueCur = FightSkill:GetMagicDescSplit(Horse.tbLevelAttrName[i], tbCurAttrNum)
      pPanel:Label_SetText("txtAttrib", szDesc)
      pPanel:Label_SetText("txtValueNow", szValueCur)
    end)
  end
end

function tbWnd:UpdateHorseEquipInfo()
  local pAsyncRole = self.tbOwner.pAsyncRole
  local nItemId = self.tbExtEquips[self.nSelectEquip]
  local pEquip = nItemId and nItemId > 0 and KItem.GetItemObj(nItemId) or nil
  if pEquip and pEquip.dwId > 0 then
    self.tbSelectEquip:SetItem(pEquip.dwId, pEquip.dwTemplateId, {bBigIcon = true}, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
    local nCurLevel = pEquip.nLevel
    self:UpdateAttrInfoEquip(nCurLevel)
    local szName = Item:GetName(pEquip.dwTemplateId, me)
    self.pPanel:Label_SetText(tbControls.txtEquipName, szName)
    self.pPanel:Label_SetText(tbControls.txtNowEquip, i18n.EasyGet("{0}\231\186\167", nCurLevel))
  else
    self.tbSelectEquip:SetEmpty(true)
    self.pPanel:SetActive(tbControls.txtEquipName, "")
    self.pPanel:Label_SetText(tbControls.txtNowEquip, "")
    self.tbAttrScrollEquip:Reset()
  end
end

function tbWnd:UpdateAttrInfoEquip(nCurLevel)
  local nTemplateIdCur = Horse.tbEquipAttrCfg[self.nSelectEquip][nCurLevel][4]
  local tbAttrCur = KItem.GetEquipBaseProp(nTemplateIdCur).tbBaseAttrib
  self.tbAttrScrollEquip:Reset()
  for i, tbMA in ipairs(tbAttrCur) do
    self.tbAttrScrollEquip:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local szDesc, szValueCur = FightSkill:GetMagicDescSplit(tbAttrCur[i].szName, tbAttrCur[i].tbValue)
      pPanel:Label_SetText("txtAttrib", szDesc)
      pPanel:Label_SetText("txtValueNow", szValueCur)
    end)
  end
end

function tbWnd:UpdateHorseSkillInfo()
  for nIdx, szImgSkillIconKey in pairs(tbControls.tbImageSkillKey) do
    self.pPanel:Sprite_SetSprite(szImgSkillIconKey, HorseSkillMgr.tbHorseSkillIconPath[nIdx])
    if nIdx == self.nSelectSkill then
      self.pPanel:SetActive(tbControls.tbSkillSelectKey[nIdx], true)
      local nAllLevel = 0
      for nGrade = 1, HorseSkillMgr.nMaxGrade do
        local nLevel = HorseSkillMgr:GetSyncSkillLevel(self.tbOwner.pAsyncRole, nIdx, nGrade)
        if nLevel > 0 then
          nAllLevel = nAllLevel + nLevel
        end
      end
      self.pPanel:Label_SetText(tbControls.txtSkillName, i18n.EasyGet("{0}\239\188\136{1}\231\186\167\239\188\137", i18n.Get(HorseSkillMgr.tbHorseSkillName[nIdx]), nAllLevel))
      self.pPanel:Sprite_SetSprite(tbControls.Img_Skill, HorseSkillMgr.tbHorseSkillIconPath[nIdx])
    else
      self.pPanel:SetActive(tbControls.tbSkillSelectKey[nIdx], false)
    end
  end
  self.nPassiveSkillId = HorseSkillMgr:GetStoneSkillId(self.tbSeriesList)
  if self.nPassiveSkillId > 0 then
    self.pPanel:SetActive(tbControls.ImgPassiveSkillBG, true)
    self.pPanel:SetActive(tbControls.imgSkillEmpty, false)
    local szIcon = FightSkill:GetSkillIconAtlas(self.nPassiveSkillId)
    self.pPanel:Sprite_SetSprite(tbControls.ImgPassiveSkill, szIcon)
  else
    self.pPanel:SetActive(tbControls.ImgPassiveSkillBG, false)
    self.pPanel:SetActive(tbControls.imgSkillEmpty, true)
  end
  self:UpdateSkillAttrInfo()
end

function tbWnd:UpdateSkillAttrInfo()
  local nSkillIndex = self.nSelectSkill
  local pAsyncRole = self.tbOwner.pAsyncRole
  local tbAttrValue = HorseSkillMgr:GetPerSkillAllAttr(nSkillIndex, pAsyncRole)
  self.tbSkillAttrScroll:Reset()
  if not tbAttrValue then
    return
  end
  for nAttrIndex, tbV in pairs(tbAttrValue) do
    self.tbSkillAttrScroll:SetElement(nAttrIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbV.szAttrType, tbV.tbValue)
      pPanel:Label_SetText("txtAttrib", szAttribMsg)
      pPanel:Label_SetText("txtValue", szValue)
      local nQuality = HorseSkillMgr:GetSkillAttrMinQuality(nSkillIndex, nAttrIndex)
      pPanel:Label_SetColorByName("txtAttrib", HorseSkillMgr.tbQuality2Color[nQuality])
      pPanel:Label_SetColorByName("txtValue", HorseSkillMgr.tbQuality2Color[nQuality])
    end)
  end
end

function tbWnd:OnSkillSelect(nIndex)
  self.nSelectSkill = nIndex
  self:UpdateHorseSkillInfo()
end
