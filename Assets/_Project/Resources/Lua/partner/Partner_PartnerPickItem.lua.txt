local tbItem = Item:GetClass("PartnerPickItem")
tbItem.PARAM_ITEM_TYPE = 1

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local nItemType = KItem.GetItemExtParam(nItemTemplateId, self.PARAM_ITEM_TYPE)
  if 1 ~= nItemType and 2 ~= nItemType and 3 ~= nItemType then
    nItemType = 2
  end
  
  local function fnCompose()
    if me.nLevel >= CardPicker.Def.OpenLevel then
      if nItemTemplateId == CardPicker.Def.nPartnerSkillBookPickItemId then
        Ui:OpenWindow("UISkillBookPickingNew")
      else
        Ui:OpenWindow(Partner.szCardPickingUIClassName, nItemType)
      end
    else
      me.CenterMsg(i18n.EasyGet("\229\175\187\232\174\191\232\191\152\230\156\170\229\188\128\230\148\190"))
    end
  end
  
  return {
    szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnFirst = fnCompose
  }
end
