local tbItem = Item:GetClass("PartnerXinWu")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  local tbBaseProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbBaseProp then
    return
  end
  
  local function fnPreview()
    Ui:OpenWindow("UIPartnerGalleryDetail", Partner.tbAllPartnerXinWuInfo[nItemTemplateId].nId)
  end
  
  local function fnCompose()
    if me.nLevel < CardPicker.Def.OpenLevel then
      me.CenterMsg(i18n.EasyGet("{0}\231\186\167\229\188\128\230\148\190\229\144\140\228\188\180\231\179\187\231\187\159\239\188\140\230\149\172\232\175\183\230\156\159\229\190\133", CardPicker.Def.OpenLevel))
      return
    end
    RemoteServer.CallPartnerFunc("ComposePartnerXinWu", nItemTemplateId)
  end
  
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return {
      szFirstName = i18n.EasyGet("\233\162\132\232\167\136"),
      fnFirst = fnPreview
    }
  else
    return {
      szFirstName = i18n.EasyGet("\233\162\132\232\167\136"),
      fnFirst = fnPreview,
      szSecondName = i18n.EasyGet("\229\144\136\230\136\144"),
      fnSecond = fnCompose
    }
  end
end
