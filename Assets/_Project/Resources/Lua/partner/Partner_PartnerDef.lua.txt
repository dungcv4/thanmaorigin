function Partner:LoadMiscSetting()
  local function fnParse(tbSelf, tbFile, szField, fnSetKey, fnSetVal)
    tbSelf[szField] = {}
    
    for szKey, szVal in pairs(tbFile[szField]) do
      local varKey = fnSetKey and fnSetKey(szKey) or szKey
      local varVal = fnSetVal and fnSetVal(szVal) or szVal
      tbSelf[szField][varKey] = varVal
    end
  end
  
  local function fnSplit(szVal)
    return Lib:SplitStr2Num(szVal)
  end
  
  local tbFile = Lib:LoadIniFile("Setting/Partner/PartnerSetting.ini")
  self.nExpItemValue2PartnerExp = tonumber(tbFile.Partner.ExpItemValue2PartnerExp)
  self.nValueToFightPower = tonumber(tbFile.Partner.ValueToFightPower)
  self.nPartnerPotentialItemId = tonumber(tbFile.Partner.PartnerPotentialItemId)
  self.nPartnerExpItemId = tonumber(tbFile.Partner.PartnerExpItemId)
  self.nPartnerRebornItemId = tonumber(tbFile.Partner.PartnerRebornItemId)
  self.nXiSuiItemId = tonumber(tbFile.Partner.XiSuiItemId)
  self.tbXiSuiPotentialDeltaValue = fnSplit(tbFile.Partner.XiSuiPotentialDeltaValue)
  self.nXiSuiCost = tonumber(tbFile.Partner.XiSuiCost)
  self.nPosDistance = tonumber(tbFile.Partner.PosDistance)
  self.PotentialCost = tonumber(tbFile.Partner.PotentialCost)
  self.nPartnerSkillBookFragmentItemId = tonumber(tbFile.Partner.SkillBookFragmentItemId)
  self.nPartnerAwakeningItemId = tonumber(tbFile.Partner.JueXingItemId)
  self.nPartnerAwakeningLevelLimit = tonumber(tbFile.Partner.JueXingLevelLimit)
  self.nPartnerAwakeningStarLimit = tonumber(tbFile.Partner.JueXingStarLimit)
  self.nPartnerAwakeningJueXingItemDunWuValue = tonumber(tbFile.Partner.JueXingItemDunWuValue)
  self.nPartnerAwakeningDunWuDoubleValue = tonumber(tbFile.Partner.DunWuDoubleValue)
  self.nWuYinShuiItemId = tonumber(tbFile.Partner.WuYinShuiItemId)
  fnParse(self, tbFile, "PosNeedLevel", tonumber, tonumber)
  fnParse(self, tbFile, "PosDir", tonumber, tonumber)
  fnParse(self, tbFile, "LevelupGrowthValue", tonumber, tonumber)
  fnParse(self, tbFile, "StarUpPotentialDeltaValue", tonumber, tonumber)
  fnParse(self, tbFile, "StarUpPotentialLimitDeltaValue", tonumber, fnSplit)
  fnParse(self, tbFile, "GradeUpPotentialLimitDeltaValue", tonumber, fnSplit)
  fnParse(self, tbFile, "QualityStarFightValue", tonumber, fnSplit)
  fnParse(self, tbFile, "ServeranceCost", tonumber, tonumber)
  fnParse(self, tbFile, "ProtectSkillLevel", tonumber, fnSplit)
  fnParse(self, tbFile, "PotentialName", tonumber, nil)
  fnParse(self, tbFile, "StarLevelExtSkillNum", tonumber, tonumber)
  fnParse(self, tbFile, "PotentialLevel", tonumber, tonumber)
  fnParse(self, tbFile, "DecomposeReturn", tonumber, fnSplit)
  fnParse(self, tbFile, "Attribute", tonumber, nil)
  fnParse(self, tbFile, "SkillLevelFightValue", tonumber, tonumber)
  fnParse(self, tbFile, "SkillLevelTimeFrame", tonumber, nil)
  fnParse(self, tbFile, "SkillLevelQuality", tonumber, tonumber)
  fnParse(self, tbFile, "StarUpLimitLevel", tonumber, tonumber)
  fnParse(self, tbFile, "JueXingSkillFightPower", tonumber, fnSplit)

function Partner:LoadTemplateSetting()
  local tbField = {
    "nId",
    "szName",
    "nQualityLevel",
    "nGrowthType",
    "nSeries",
    "nValue",
    "nAiType",
    "nSkill2",
    "nNpcTemplateId",
    "nBaseSkillId",
    "szName",
    "nitemid",
    "nProtectSkillId",
    "szTimeFrame",
    "nJueXingSkillId"
  }
  for nId = 1, self.nMaxStarLevel do
    table.insert(tbField, "tbnStarUpCostParIdList" .. nId)
  end
  self.tbAllPartnerInfo = Lib:EasyLoadTabFile("Setting/Partner/PartnerTemplate.tab", {"nId"}, tbField, nil, false)
  self.tbAllPartnerTemplateInfo = Lib:EasyLoadTabFile("Setting/Partner/PartnerTemplate.tab", {
    "nNpcTemplateId"
  }, tbField, nil, false)
  self.tbAllPartnerXinWuInfo = Lib:EasyLoadTabFile("Setting/Partner/PartnerTemplate.tab", {"nitemid"}, tbField, nil, false)
  self.tbPartnerAiTypeSetting = Lib:EasyLoadTabFile("Setting/Partner/PartnerAiType.tab", {"nAiTypeId"}, {
    "nAiTypeId",
    "szInfo",
    "szAiPath",
    "nAiSkillId1",
    "nAiSkillId2"
  })
  self.tbPartnerEnemyNpcCfg = Lib:EasyLoadTabFile("Setting/Partner/PartnerEnemyNpcCfg.tab", {"nPartnerId"}, {
    "nPartnerId",
    "nNpcTemplateId"
  })
  local tbTemp = {}
  for nId = 1, self.nMaxStarLevel do
    local szField = "tbnStarUpCostParIdList" .. nId
    tbTemp[nId], self.tbAllPartnerInfo[szField] = self.tbAllPartnerInfo[szField], tbTemp[nId]
  end
  self.tbAllPartnerInfo.tbnStarUpCostParIdList = tbTemp
  self.tbAllPartnerInfo = SetReadOnly(self.tbAllPartnerInfo)
  self.tbAllPartnerTemplateInfo = SetReadOnly(self.tbAllPartnerTemplateInfo)
  Validator:Register(function()
    for nItemId, _ in pairs(self.tbAllPartnerXinWuInfo) do
      if not Item:IsExist(nItemId) then
        LogErr(string.format("load PartnerTemplate.tab failed, nItemId %d not exist", nItemId))
        return false
      end
    end
    return true
  end)
end

function Partner:LoadPotentialSetting()
  self.tbPotentialToMagic = {}
  self.tbPotentialStartValue = {}
  local tbPotentialSetting = Lib:LoadTabFileEx("Setting/Partner/PotentialSetting.tab", "dddddd", nil, {
    "nQuality",
    "nPotentialVitalityToMagic",
    "nPotentialDexterityToMagic",
    "nPotentialStrengthToMagic",
    "nPotentialEnergyToMagic",
    "nStartValue"
  }, nil, nil, false)
  for _, tbInfo in pairs(tbPotentialSetting) do
    self.tbPotentialToMagic[tbInfo.nQuality] = {}
    self.tbPotentialToMagic[tbInfo.nQuality][self.POTENTIAL_TYPE_VITALITY] = tbInfo.nPotentialVitalityToMagic
    self.tbPotentialToMagic[tbInfo.nQuality][self.POTENTIAL_TYPE_DEXTERITY] = tbInfo.nPotentialDexterityToMagic
    self.tbPotentialToMagic[tbInfo.nQuality][self.POTENTIAL_TYPE_STRENGTH] = tbInfo.nPotentialStrengthToMagic
    self.tbPotentialToMagic[tbInfo.nQuality][self.POTENTIAL_TYPE_ENERGY] = tbInfo.nPotentialEnergyToMagic
    self.tbPotentialStartValue[tbInfo.nQuality] = tbInfo.nStartValue
  end
  self.tbPotentialToMagic = SetReadOnly(self.tbPotentialToMagic)
  self.tbPotentialStartValue = SetReadOnly(self.tbPotentialStartValue)
  self.tbPotentialRandomInfo = {}
  local tbRandomInfo = Lib:LoadTabFileEx("Setting/Partner/PotentialRandom.tab", "ss", nil, {"nTimes", "nRate"}, nil, nil, false)
  local nTotalRate = 0
  for _, tbInfo in pairs(tbRandomInfo) do
    local nTimes = tonumber(tbInfo.nTimes)
    local nRate = tonumber(tbInfo.nRate)
    nTotalRate = nRate + nTotalRate
    table.insert(self.tbPotentialRandomInfo, {nTimes = nTimes, nRate = nTotalRate})
  end
  self.tbPotentialRandomInfo.nTotalRate = nTotalRate
  self.tbPotentialRandomInfo = SetReadOnly(self.tbPotentialRandomInfo)
  self.tbLimitPotentialInfo = {}
  self.nMaxPotentialLimitLevel = 0
  local tbParam = {}
  for _, tbInfo in pairs(self.tbPartnerGrowthInfo) do
    for _, nParam in pairs(tbInfo) do
      if 0 ~= nParam then
        tbParam[nParam] = true
      end
    end
  end
  local tbType = {"d", "d"}
  local tbField = {"nQuality", "nLevel"}
  for nParam, _ in pairs(tbParam) do
    table.insert(tbType, "d")
    table.insert(tbField, tostring(nParam))
  end
  local tbLimitInfo = Lib:LoadTabFileEx("Setting/Partner/PotentialLimit.tab", table.concat(tbType), nil, tbField, nil, nil, false)
  for _, tbInfo in pairs(tbLimitInfo) do
    self.tbLimitPotentialInfo[tbInfo.nQuality] = self.tbLimitPotentialInfo[tbInfo.nQuality] or {}
    self.tbLimitPotentialInfo[tbInfo.nQuality][tbInfo.nLevel] = {}
    local tbLimit = self.tbLimitPotentialInfo[tbInfo.nQuality][tbInfo.nLevel]
    for nParam, _ in pairs(tbParam) do
      tbLimit[nParam] = tbInfo[tostring(nParam)]
    end
    if self.nMaxPotentialLimitLevel < tbInfo.nLevel then
      self.nMaxPotentialLimitLevel = tbInfo.nLevel
    end
  end
  self.tbLimitPotentialInfo = SetReadOnly(self.tbLimitPotentialInfo)
end

function Partner:LoadQualitySetting()
  self.tbPartnerQualityInfo = {}
  self.tbPartnerQualityInfo = Lib:EasyLoadTabFile("Setting/Partner/QualitySetting.tab", {
    "nGrowthType",
    "nQualityLevel"
  }, nil, nil, false)
  for nGrowthType, tbGrowthInfo in pairs(self.tbPartnerQualityInfo) do
    for nQuality, tbQualityInfo in pairs(tbGrowthInfo) do
      for _, szName in pairs(self.tbAllPotentialType) do
        tbQualityInfo[szName] = tonumber(tbQualityInfo[szName])
      end
      tbQualityInfo.tbMagicAttrib = {}
      for nId = 1, self.nMaxQualityAttrNum do
        local szType, nValue1, nValue2, nValue3
        local szField = string.format("szAttrib%dType", nId)
        szType, tbQualityInfo[szField] = tbQualityInfo[szField], szType
        szField = string.format("nAttrib%dValue1", nId)
        nValue1, tbQualityInfo[szField] = tbQualityInfo[szField], nValue1
        szField = string.format("nAttrib%dValue2", nId)
        nValue2, tbQualityInfo[szField] = tbQualityInfo[szField], nValue2
        szField = string.format("nAttrib%dValue3", nId)
        nValue3, tbQualityInfo[szField] = tbQualityInfo[szField], nValue3
        if "" ~= szType then
          table.insert(tbQualityInfo.tbMagicAttrib, {
            szType,
            nValue1 or 0,
            nValue2 or 0,
            nValue3 or 0
          })
        end
      end
    end
  end
  self.tbPartnerQualityInfo = SetReadOnly(self.tbPartnerQualityInfo)
end

function Partner:LoadPartnerSkillBookSetting()
  self.tbSkillBookSetting = {}
  self.tbSkillBookSetting.tbSkillId2BookId = {}
  self.tbSkillBookSetting.tbBookInfo = {}
  local tbSkillBook = Lib:EasyLoadTabFile("Setting/Partner/SkillBook.tab", {
    "nSkillBookId"
  })
  for _, tbInfo in pairs(tbSkillBook) do
    local nSkillBookId = tbInfo.nSkillBookId
    self.tbSkillBookSetting.tbSkillId2BookId[tbInfo.nSkillId] = nSkillBookId
    tbInfo.tbSkillValue = {}
    tbInfo.tbGradeValue = {}
    tbInfo.nRemnantsReturnCount = tbInfo.nRemnantsReturnCount or 20
    for nId = 1, self.nMaxSkillLevel do
      local nVal = tbInfo["nLevelValue" .. nId]
      local nVal2 = tbInfo["nGradeValue" .. nId]
      tbInfo["nLevelValue" .. nId] = nil
      tbInfo["nGradeValue" .. nId] = nil
      if 0 ~= nVal then
        tbInfo.tbSkillValue[nId] = nVal
        tbInfo.nMaxLevel = nId
      end
      if 0 ~= nVal2 then
        tbInfo.tbGradeValue[nId] = nVal2
      end
    end
    self.tbSkillBookSetting.tbBookInfo[nSkillBookId] = tbInfo
  end
  self.tbSkillBookSetting = SetReadOnly(self.tbSkillBookSetting)
end
