local tbWnd = Ui:CreateClass("UIPartnerNew_ZhenFaPreview", "UIPanel")
local tbControls = {
  Info = "info",
  txtZhenFaName = "info/txtZhenFaName",
  txtFightPower = "info/txtFightPower",
  txtAttribute = "info/Attribute/attributeCell%d/txtAttribute",
  txtEnhance = "info/Attribute/attributeCell%d/txtAttribute/txtEnhance",
  txtAttributeTemplate = "info/RandomAttribute/attributeCell/txtAttributeTemplate",
  Activate = "Activate",
  txtActivateCondition = "Activate/txtTitle/txtActivateCondition",
  btnGoActivate = "Activate/txtTitle/btnActivate"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnGoActivate = function(self)
    local tbUIPartnerNew = Ui:GetWindow("UIPartnerNew")
    if tbUIPartnerNew then
      tbUIPartnerNew:MainPanelButtonSelect(5)
    end
  end
}
tbWnd.tbColor = {szRed = "C50505", szPurple = "6D05C5"}

function tbWnd:OnCreate()
  self.tbRandomScrollView = Ui:NewScrollView(self, tbControls.txtAttributeTemplate)
end

function tbWnd:OnOpen()
  self:UpdateView()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGoActivate, {
    RedPointMgr.EVENT.PartnerZFActivate
  })
end

function tbWnd:OnClose()
  self.tbRandomScrollView:Clear()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnGoActivate)
end

function tbWnd:OnZhenFaRefresh(nRefreshType)
  if nRefreshType == PartnerZhenFa.tbUIRefreshType.ACTIVATE_STATE then
    self:UpdateView()
  elseif nRefreshType == PartnerZhenFa.tbUIRefreshType.EYE_CHANGE then
    self:OnChangeZFEye()
  end
end

function tbWnd:UpdateView()
  local bActivate = PartnerZhenFa:IsZFActivate(me)
  if bActivate then
    self.pPanel:SetActive("info", true)
    self.pPanel:SetActive("Activate", false)
    local tbGroupCfg = PartnerZhenFa:GetPosGroupCfg(me)
    if not tbGroupCfg then
      Log("UIPartnerNew_ZhenFaPreview:OnChangeZFState Error GetCurPosGroupCfg is nil")
      return
    end
    self.pPanel:Label_SetText(tbControls.txtZhenFaName, i18n.Parse(tbGroupCfg.szName))
    local nFightPower = PartnerZhenFa:CalcFightPower(me)
    self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\233\152\181\230\179\149\230\136\152\229\138\155\239\188\154{0}", nFightPower))
    for nIndex = 1, PartnerZhenFa.nFixAttrCount do
      local szAttribute = PartnerZhenFa:GetFixAttrWithEyeDesc(nIndex)
      self.pPanel:Label_SetText(string.format(tbControls.txtAttribute, nIndex), szAttribute)
      local nLevel, _, szFinalValue = PartnerZhenFa:GetFixAttrEnhanceDescInfo(nIndex)
      local szEnhance = i18n.EasyGet("\229\188\186\229\140\150\239\188\154Lv.{0}   {1}", nLevel, szFinalValue or 0)
      self.pPanel:Label_SetText(string.format(tbControls.txtEnhance, nIndex), szEnhance)
    end
    self.tbRandomScrollView:Clear()
    local nRandomAttrCount = PartnerZhenFa:GetCurRandomAttrCount(me)
    for nIndex = 1, nRandomAttrCount do
      local szRandomAttr = PartnerZhenFa:GetRandomAttrDesc(nIndex)
      if szRandomAttr then
        self.tbRandomScrollView:SetElement(nIndex, function(tbElement)
          tbElement.pPanel:Label_SetText("Main", szRandomAttr)
        end)
      end
    end
  else
    self.pPanel:SetActive(tbControls.Info, false)
    self.pPanel:SetActive(tbControls.Activate, true)
    local bCanActivate = PartnerZhenFa:CheckCanActivate(me, PartnerZhenFa.nDefaultZFPos)
    self.pPanel:Button_SetEnable(tbControls.btnGoActivate, bCanActivate)
    local nUsefulCount = PartnerZhenFa:GetUsefulPartnerCount(me)
    local nNeedCount = PartnerZhenFa.Cfg.nActivatePartnerCount
    local szQualityLimit = i18n.Get(PartnerZhenFa.Cfg.szPartnerQualityDesc)
    local szActivateCondition = ""
    if nUsefulCount >= nNeedCount then
      szActivateCondition = i18n.EasyGet("\228\184\138\233\152\181{0}/{1}\228\184\170<color=#{2}>{3}</color>\229\143\138\228\187\165\228\184\138\231\154\132\229\144\140\228\188\180\n\232\174\190\231\189\1741\228\184\170<color=#{4}>{5}</color>\230\136\150\230\155\180\233\171\152\229\147\129\232\180\168\231\154\132\229\144\140\228\188\180\228\184\186\233\152\181\231\156\188", nUsefulCount, nNeedCount, self.tbColor.szPurple, szQualityLimit, self.tbColor.szPurple, szQualityLimit)
    else
      szActivateCondition = i18n.EasyGet("\228\184\138\233\152\181<color=#{0}>{1}</color>/{2}\228\184\170<color=#{3}>{4}</color>\229\143\138\228\187\165\228\184\138\231\154\132\229\144\140\228\188\180\n\232\174\190\231\189\1741\228\184\170<color=#{5}>{6}</color>\230\136\150\230\155\180\233\171\152\229\147\129\232\180\168\231\154\132\229\144\140\228\188\180\228\184\186\233\152\181\231\156\188", self.tbColor.szRed, nUsefulCount, nNeedCount, self.tbColor.szPurple, szQualityLimit, self.tbColor.szPurple, szQualityLimit)
    end
    self.pPanel:Label_SetText(tbControls.txtActivateCondition, szActivateCondition)
  end
end

function tbWnd:OnChangeZFEye()
  local bActivate = PartnerZhenFa:IsZFActivate(me)
  if not bActivate then
    local bCanActivate = PartnerZhenFa:CheckCanActivate(me, PartnerZhenFa.nDefaultZFPos)
    self.pPanel:Button_SetEnable(tbControls.btnGoActivate, bCanActivate)
    return
  end
  local nFightPower = PartnerZhenFa:CalcFightPower(me)
  self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\233\152\181\230\179\149\230\136\152\229\138\155\239\188\154{0}", nFightPower))
  for nIndex = 1, PartnerZhenFa.nFixAttrCount do
    local szAttribute = PartnerZhenFa:GetFixAttrWithEyeDesc(nIndex)
    self.pPanel:Label_SetText(string.format(tbControls.txtAttribute, nIndex), szAttribute)
  end
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end
