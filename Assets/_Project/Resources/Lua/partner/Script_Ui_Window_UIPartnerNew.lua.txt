function tbWnd:OnOpen(nPanel, nSelectPartnerTemplateId, nSubPanel)
  self.pPanel:SetActive(tbControls.btnPartnerDecompose, false)
  self.tbReInitUpdateData = {}
  self.bAutoPotentialLevelUp = false
  self.bAutoXiSui = false
  self.pPanel:Toggle_SetChecked(tbControls.toggleAutoLevelUp, false)
  self.pPanel:Toggle_SetChecked(tbControls.toggleAutoXiSui, false)
  self.nLastBattlePartnerCount = self:GetPosPartnerCount()
  self:InitOrderPartnerModel()
  self:InitPartnerOrder(true)
  self:UpdatePartnerList(nSelectPartnerTemplateId)
  if RedPointMgr:CheckEvent(RedPointMgr.EVENT.PartnerCanInPos) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.PartnerCanInPos)
  end
  if nPanel then
    self:MainPanelButtonSelect(nPanel, nSubPanel)
  else
    self:MainPanelButtonSelect(tbMiddlePanelStateIndex.PANEL_PARTNER)
  end
  self:ToggleGroupSelect(tbRightPanelStateIndex.PANEL_INFO)
  if 0 == #self.tbPartnerList then
    Ui:OpenWindow(Partner.szCardPickingUIClassName)
    return 0
  end
  self.pPanel:SetActive(tbControls.wndYuanqiTips, false)
  self:RegistRedPoint()
  self:ClearAllEffect()
  self.pPanel:SetActive(string.format(tbControls.btnPotentialInfo, self.nSelectedPotentialIndex) .. "/imgSelect", true)
  self.pPanel:SetActive(tbControls.wndAwakeningPanelInfo, false)
  self.pPanel:SetActive(tbControls.btnPartnerEquip, 1 == GetTimeFrameState(Item.PartnerEquip.CFG.szOpenLevel))
  self.pPanel:SetActive(tbControls.btnPartnerZhenfa, PartnerZhenFa:CanShowUIEntrance())
end

function tbWnd:UpdatePartnerList(nSelectPartnerTemplateId)
  self.tbPartnerList = self:GetAllPartners()
  self.tbBasePartnerView:SetAllElement(self.tbPartnerList)
  if nSelectPartnerTemplateId then
    for nIndex, pPartner in pairs(self.tbPartnerList) do
      local nNpcTemplateId = Partner.tbAllPartnerInfo[pPartner.nTemplateId].nNpcTemplateId
      if nSelectPartnerTemplateId == nNpcTemplateId then
        self.nCurSelectedPartnerIndex = nIndex
        self.tbBasePartnerView:MovePanelToItemIndex(nIndex)
        self:UpdateCurrentPartnerInfo()
        break
      end
    end
  else
    self.nCurSelectedPartnerIndex = 1
    self.tbBasePartnerView:MovePanelToItemIndex(1)
    self:UpdateCurrentPartnerInfo()
  end
end

function tbWnd:OnClose()
  self:UploadPartnerOrder()
  self:StopAllAutoTimer()
  self:ReleaseOrderPartnerModel()
  self:UnRegistRedPoint()
  self.goPartnerModel:Release()
  self.tbPanelList[4]:SetActive(false)
end

function tbWnd:SetPartnerData(tbElement, pPartner, nIndex)
  local pPanel = tbElement.pPanel
  local nLevel = pPartner.GetLevelInfo()
  local nQuality = Partner.tbAllPartnerInfo[pPartner.nTemplateId].nQualityLevel
  local bIsCurSelected = nIndex == self.nCurSelectedPartnerIndex
  pPanel:SetActive("imgSelected", bIsCurSelected)
  pPanel:Label_SetText("txtPartnerName", i18n.Parse(pPartner.szName))
  pPanel:Label_SetText("FightPower/txtFightPowerValue", pPartner.nFightPower)
  if nLevel >= GameSetting.MAX_PLAYER_LEVEL then
    pPanel:Label_SetText("txtLevel", i18n.EasyGet("\229\174\151\229\184\136{0}", nLevel - GameSetting.MAX_PLAYER_LEVEL))
  else
    pPanel:Label_SetText("txtLevel", i18n.EasyGet("\231\173\137\231\186\167{0}", nLevel))
  end
  local nJueXingLevel = Partner:GetPartnerJueXingLevel(me, pPartner.dwId)
  pPanel:SetActive("ImgAwakening", nJueXingLevel > 0)
  local nCurPartnerStar = pPartner.GetStar()
  local nMoonCount, nStarCount = Partner:CheckPartnerUIStarInfo(nCurPartnerStar)
  local nLightMoonCount = 0
  local nLightStarCount = 0
  local szStarPath = "UI/atlas/common/img_Star01.png"
  local szLightStarPath = "UI/atlas/common/img_Star02.png"
  local szMoonPath = "UI/atlas/common/img_Star03.png"
  local szLightMoonPath = "UI/atlas/common/img_Star04.png"
  local szSunPath = "UI/atlas/common/img_Star05.png"
  local szLightSunPath = "UI/atlas/common/img_Star06.png"
  for i = 1, 8 do
    local nLightType = 0
    if nMoonCount > nLightMoonCount then
      nLightMoonCount = nLightMoonCount + 1
      nLightType = 2
    elseif nStarCount > nLightStarCount then
      nLightStarCount = nLightStarCount + 1
      nLightType = 1
    end
    local bLight = nLightType > 0
    local bOpen = Partner:CheckPartnerUIStarIsOpen(nQuality, nCurPartnerStar, i)
    pPanel:SetActive(string.format("StarList/imgStar%d_0", i), bOpen)
    pPanel:SetActive(string.format("StarList/Star%d", i), bOpen and bLight)
    if 0 == nLightType then
      pPanel:Sprite_SetSprite(string.format("StarList/imgStar%d_0", i), szStarPath)
      pPanel:Sprite_SetSprite(string.format("StarList/Star%d", i), szLightStarPath)
    elseif 1 == nLightType then
      pPanel:Sprite_SetSprite(string.format("StarList/imgStar%d_0", i), szStarPath)
      pPanel:Sprite_SetSprite(string.format("StarList/Star%d", i), szLightStarPath)
    elseif 2 == nLightType then
      pPanel:Sprite_SetSprite(string.format("StarList/imgStar%d_0", i), szMoonPath)
      pPanel:Sprite_SetSprite(string.format("StarList/Star%d", i), szLightMoonPath)
    elseif 3 == nLightType then
      pPanel:Sprite_SetSprite(string.format("StarList/imgStar%d_0", i), szSunPath)
      pPanel:Sprite_SetSprite(string.format("StarList/Star%d", i), szLightSunPath)
    end
  end
  Ui:Toggle_UnBindEvent(pPanel, "ToggleAdd")
  pPanel:Toggle_SetChecked("ToggleAdd", self:IsPosPartner(pPartner))
  Ui:Toggle_BindEvent(pPanel, "ToggleAdd", function(self, isOn)
    local tbNewPartnerPosInfo = self:SetPosPartner(isOn, pPartner.dwId)
    if -1 ~= tbNewPartnerPosInfo then
      self.nCurSelectedPartnerIndex = nIndex
      self.nLastBattlePartnerCount = self:GetPosPartnerCount()
      RemoteServer.CallPartnerFunc("SetPartnerPos", tbNewPartnerPosInfo)
    else
      pPanel:Toggle_SetChecked("ToggleAdd", false)
    end
  end, self)
  tbElement.pPanel:Sprite_SetSprite("imgQualityBg", tbImgQualityBgList[nQuality])
  tbElement.pPanel:SetActive(Partner.tbImgQualityBgEffectList[1], false)
  tbElement.pPanel:SetActive(Partner.tbImgQualityBgEffectList[2], false)
  tbElement.pPanel:SetActive(Partner.tbImgQualityBgEffectList[3], false)
  if nQuality <= 3 then
    tbElement.pPanel:SetActive(Partner.tbImgQualityBgEffectList[nQuality], true)
  end
  local tbRet = Partner:CheckPartnerPensonalRedPoint(pPartner.dwId)
  tbElement.pPanel:SetActive("RedPoint", self:CheckPartnerListRedPoint(tbRet))
  if bIsCurSelected then
    self:LightOnePartnerRedPoint(pPartner.dwId)
  end
  if not self:CheckOrderPartnerMax() and not self:IsPosPartner(pPartner) then
    if nIndex > 1 then
      local pLastPartner = self.tbPartnerList[nIndex - 1]
      if self:IsPosPartner(pLastPartner) then
        tbElement.pPanel:SetActive("RedPoint", true)
      end
    else
      tbElement.pPanel:SetActive("RedPoint", true)
    end
  end
end

function tbWnd:ResetAllElements()
  local nSelectNpcTemplateId = self.tbPartnerList[self.nCurSelectedPartnerIndex].nTemplateId
  self.tbPartnerList = self:GetAllPartners()
  for nIndex, pPartner in pairs(self.tbPartnerList) do
    if nSelectNpcTemplateId == pPartner.nTemplateId then
      self.nCurSelectedPartnerIndex = nIndex
      break
    end
  end
  local nBattlePartnerCount = self:GetPosPartnerCount()
  self.tbBasePartnerView:SetAllElement(self.tbPartnerList)
  if nBattlePartnerCount ~= self.nLastBattlePartnerCount then
    self.tbBasePartnerView:MovePanelToFirstItem()
  else
    self.tbBasePartnerView:MovePanelToItemIndex(self.nCurSelectedPartnerIndex)
  end
end

function tbWnd:UpdateCurrentPartnerInfoWhenItemSync()
  local curPartner = self.tbPartnerList[self.nCurSelectedPartnerIndex]
  if not curPartner then
    return
  end
  local nQualityLevel = Partner.tbAllPartnerInfo[curPartner.nTemplateId].nQualityLevel
  local nLevel, nExp, nLevelupExp = curPartner.GetLevelInfo()
  local nNeedLevelUpItem = self:GetNeedLevelUpItem(nQualityLevel, nLevel, nExp, nLevelupExp)
  local nPartnerExpItemCount = me.GetItemCountInBags(Partner.nPartnerExpItemId)
  self.pPanel:Label_SetText(tbControls.txtLevelUpItem, nPartnerExpItemCount .. " / " .. nNeedLevelUpItem)
  local nPartnerPotentialItemCount = me.GetItemCountInBags(Partner.nPartnerPotentialItemId)
  self.pPanel:Label_SetText(tbControls.txtPotentialLevelUpItem, nPartnerPotentialItemCount .. " / " .. Partner.PotentialCost)
  local nXiSuiItemCost = Partner.nXiSuiCost or 10
  local nXiSuiItemCount = me.GetItemCountInBags(Partner.nXiSuiItemId)
  self.pPanel:Label_SetText(tbControls.txtXiSuiCostNum, nXiSuiItemCount .. " / " .. nXiSuiItemCost)
  self.nCost = Partner.ServeranceCost[nQualityLevel]
  self.pPanel:Label_SetText(tbControls.txtRebornCost, string.format("%d/%d", me.GetItemCountInBags(Partner.nPartnerRebornItemId), Partner.ServeranceCost[nQualityLevel]))
  local nSkillLevel = Partner:GetPartnerJueXingLevel(me, curPartner.dwId)
  local nItemValue = me.GetItemCountInBags(Partner.nPartnerAwakeningItemId)
  local nTargetLevelTimeFrame = Partner:GetPartnerJueXingSkillNextTimeFrame(nQualityLevel, nSkillLevel + 1)
  if -999 == nTargetLevelTimeFrame or nSkillLevel >= Partner.JueXingSkillLevelLimit[nQualityLevel] or 1 ~= GetTimeFrameState(nTargetLevelTimeFrame) then
    self.pPanel:Label_SetText(tbControls.txtAwakeningCost2Cost, i18n.EasyGet("{0}", nItemValue))
  else
    self.pPanel:Label_SetText(tbControls.txtAwakeningCost2Cost, i18n.EasyGet("{0} / {1}", nItemValue, Partner.JueXingSkillJueXingItemNeeded[nQualityLevel][nSkillLevel + 1]))
  end
end

function tbWnd:UpdateCurPartnerRightPotentialView()

    local potentialValue = curPartner.GetPotential(Partner.tbPotentialList[i])

    self.pPanel:Sprite_SetFill(string.format(tbControls.imgPartnerAttribProgressBar, i), potentialProgress)
    self.pPanel:Label_SetText(string.format(tbControls.txtPartnerAttribQualityText, i), i18n.Parse(Partner.PotentialName[Partner.tbPotentialList[i]]))

function tbWnd:UpdateCurPartnerRightXiSuiView()
  local curPartner = self.tbPartnerList[self.nCurSelectedPartnerIndex]
  if not curPartner then
    return
  end
  self.tbXiSuiLimitPotential = {}
  for nType, _ in pairs(Partner.tbAllPotentialType) do
    local nValue = Partner:GetXiSuiPotentialByObj(curPartner, nType)
    self.tbXiSuiLimitPotential[nType] = nValue
  end
  local nDuraTime = 0
  for nType, _ in pairs(Partner.tbAllPotentialType) do
    local nLimitValue = Partner:GetLimitPotentialByObj(curPartner, nType)
    local nMaxLimitValue = self.tbXiSuiLimitPotential[nType]
    if self.tbReInitUpdateData and self.tbReInitUpdateData[nType] and self.tbReInitUpdateData[nType] > 0 then
      self.pPanel:Label_SetText(string.format(tbControls.txtPartnerXiSuiAttribQualityPercent, Partner.tbAllPotentialTypeToIndex[nType]), nLimitValue .. "<color=#2A8247>(+" .. self.tbReInitUpdateData[nType] .. ")</color>/" .. nMaxLimitValue)
    else
      self.pPanel:Label_SetText(string.format(tbControls.txtPartnerXiSuiAttribQualityPercent, Partner.tbAllPotentialTypeToIndex[nType]), nLimitValue .. "/" .. nMaxLimitValue)
    end
    self.pPanel:Sprite_SetFillAmount(string.format(tbControls.imgPartnerXiSuiQualityBar, Partner.tbAllPotentialTypeToIndex[nType]), 0, nLimitValue / nMaxLimitValue, nDuraTime, true)
    self.pPanel:Label_SetText(string.format(tbControls.txtPartnerXiSuiAttribQuality, Partner.tbAllPotentialTypeToIndex[nType]), Partner:CheckPotentialLevel(nLimitValue, nMaxLimitValue))
  end
  local nXiSuiItemCost = Partner.nXiSuiCost or 10
  local nXiSuiItemCount = me.GetItemCountInBags(Partner.nXiSuiItemId)
  self.pPanel:Label_SetText(tbControls.txtXiSuiCostNum, nXiSuiItemCount .. " / " .. nXiSuiItemCost)
  for i = 1, 4 do
    self.pPanel:Label_SetText(string.format(tbControls.txtPartnerXiSuiAttribQualityText, i), i18n.Parse(Partner.PotentialName[Partner.tbPotentialList[i]]))

function tbWnd:UpdateCurPartnerRightStarUpView()
  local curPartner = self.tbPartnerList[self.nCurSelectedPartnerIndex]
  if not curPartner then
    return
  end
  local nPartnerId = curPartner.dwId
  local nPartnerTemplateId = curPartner.nTemplateId
  local nQualityLevel = Partner.tbAllPartnerInfo[nPartnerTemplateId].nQualityLevel
  local nGrowthType = Partner.tbAllPartnerInfo[nPartnerTemplateId].nGrowthType
  local nLevel = curPartner.nLevel
  local tbSkillInfo = {}
  local tbPotentialInfo = {}
  local nMainPartnerRedundanceNum = Partner:GetMainPartnerRedundanceNumById(me, nPartnerId)
  local nStarLevel = curPartner.GetStar()
  local tbGrowth = {
    [1] = Partner.tbPartnerGrowthInfo[nGrowthType].Strength / 10,
    [2] = Partner.tbPartnerGrowthInfo[nGrowthType].Vitality / 10,
    [3] = Partner.tbPartnerGrowthInfo[nGrowthType].Energy / 10,
    [4] = Partner.tbPartnerGrowthInfo[nGrowthType].Dexterity / 10
  }
  local nQuality = Partner.tbAllPartnerInfo[nPartnerTemplateId].nQualityLevel
  self.pPanel:Label_SetText(tbControls.txtStarUpNowFightPower, curPartner.nFightPower)
  for i = 1, #Partner.tbPotentialList do
    local potentialValueLimit = Partner:GetLimitPotentialByObj(curPartner, Partner.tbPotentialList[i])
    local potentialValueXiSui = Partner:GetXiSuiPotentialByObj(curPartner, Partner.tbPotentialList[i])
    self.pPanel:Label_SetText(string.format(tbControls.txtStarUpAttributeText, i), i18n.Parse(Partner.PotentialName[Partner.tbPotentialList[i]]))
    if nStarLevel >= Partner.nMaxStarLevel or nStarLevel >= Partner.StarUpLimitLevel[nQuality] then
      self.pPanel:Label_SetText(string.format(tbControls.txtStarUpAttributeQuality, i), self:CheckPotentialLevel(potentialValueLimit, potentialValueXiSui))
      self.pPanel:Label_SetText(string.format(tbControls.txtStarUpProgress, i), potentialValueLimit .. "/" .. potentialValueXiSui)
      self.pPanel:Sprite_SetFill(string.format(tbControls.imgStarUpProgress, i), potentialValueLimit / potentialValueXiSui)
    else
      local thenPotentialValueLimit = potentialValueLimit
      local nQualityLevel = Partner.tbAllPartnerInfo[nPartnerTemplateId].nQualityLevel
      local thenPotentialValueXiSui = math.floor(potentialValueXiSui + Partner.StarUpPotentialLimitDeltaValue[nQualityLevel][curPartner.GetStar() + 1] * tbGrowth[i])
      self.pPanel:Label_SetText(string.format(tbControls.txtStarUpAttributeQuality, i), self:CheckPotentialLevel(thenPotentialValueLimit, thenPotentialValueXiSui))
      self.pPanel:Label_SetText(string.format(tbControls.txtStarUpProgress, i), thenPotentialValueLimit .. "/" .. thenPotentialValueXiSui .. "<color=#2A8247>(+" .. thenPotentialValueXiSui - potentialValueXiSui .. ")</color>")
      self.pPanel:Sprite_SetFill(string.format(tbControls.imgStarUpProgress, i), thenPotentialValueLimit / thenPotentialValueXiSui)
    end
    tbPotentialInfo[i] = curPartner.GetPotential(i)
    for nId = 1, Partner.nMaxSkillNum do
      local nSkillId, nSkillLevel = curPartner.GetSkillInfo(nId)
      if nSkillId > 0 then
        tbSkillInfo[nSkillId] = math.max(nSkillLevel, 1)
      end
    end
  end
  if nStarLevel < Partner.nMaxStarLevel and nStarLevel < Partner.StarUpLimitLevel[nQuality] then
    self.pPanel:SetActive(tbControls.btnStarLevelUp, true)
    local thenFightPower = Partner:GetFightPower(nPartnerTemplateId, nQualityLevel, nGrowthType, nLevel, tbPotentialInfo, tbSkillInfo, nStarLevel + 1)
    self.pPanel:Label_SetText(tbControls.txtStarUpNextFightPower, thenFightPower)
    self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartner, i18n.Parse(curPartner.szName))
    if nMainPartnerRedundanceNum < Partner.tbStarLevelUpInfo[nQualityLevel][nStarLevel + 1].nCostParNum then
      self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartnerNumber, "(<color=#c50505>" .. nMainPartnerRedundanceNum .. "</color>/" .. Partner.tbStarLevelUpInfo[nQualityLevel][nStarLevel + 1].nCostParNum .. ")")
    else
      self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartnerNumber, "(" .. nMainPartnerRedundanceNum .. "/" .. Partner.tbStarLevelUpInfo[nQualityLevel][nStarLevel + 1].nCostParNum .. ")")
    end
    if nLevel < Partner.tbStarLevelUpInfo[nQualityLevel][nStarLevel + 1].nNeedLevel then
      self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartnerLevel, i18n.EasyGet("\229\144\140\228\188\180\231\173\137\231\186\167\239\188\154<color=#c50505>{0}</color>\231\186\167", Lib:GetLevelText(Partner.tbStarLevelUpInfo[nQualityLevel][nStarLevel + 1].nNeedLevel)))
    else
      self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartnerLevel, i18n.EasyGet("\229\144\140\228\188\180\231\173\137\231\186\167\239\188\154{0}\231\186\167", Lib:GetLevelText(Partner.tbStarLevelUpInfo[nQualityLevel][nStarLevel + 1].nNeedLevel)))
    end
  else
    self.pPanel:Label_SetText(tbControls.txtStarUpNextFightPower, curPartner.nFightPower)
    self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartner, i18n.Parse(curPartner.szName))
    self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartnerNumber, "(" .. nMainPartnerRedundanceNum .. ")")
    self.pPanel:Label_SetText(tbControls.txtStarUpNeedPartnerLevel, i18n.EasyGet("\229\183\178\232\190\190\229\189\147\229\137\141\230\156\128\229\164\167\230\152\159\231\186\167"))
    self.pPanel:SetActive(tbControls.btnStarLevelUp, false)
  end
end

function tbWnd:UpdateCurPartnerRightRebornView()
  local curPartner = self.tbPartnerList[self.nCurSelectedPartnerIndex]
  if not curPartner then
    return
  end
  local tbAttribNow = {
    txtLevel = 0,
    txtStar = 0,
    txtPotential = 0,
    txtSkillBook = 0,
    txtSkill = 0,
    txtFriend = 0
  }
  local tbAttribReborn = {
    txtLevel = 1,
    txtStar = 0,
    txtPotential = 0,
    txtSkillBook = 0,
    txtSkill = 0,
    txtFriend = 0
  }
  local nPartnerId = curPartner.dwId
  local nPartnerTemplateId = curPartner.nTemplateId
  local nQualityLevel = Partner.tbAllPartnerInfo[nPartnerTemplateId].nQualityLevel
  local nGrowthType = Partner.tbAllPartnerInfo[nPartnerTemplateId].nGrowthType
  local tbRebornSkillInfo = {}
  local nLevel = curPartner.GetLevelInfo()
  tbAttribNow.txtLevel = Lib:GetLevelShortText(nLevel)
  local nStar = curPartner.GetStar()
  local nMaxSkillNum = Partner.StarLevelExtSkillNum[nStar] or Partner.nMaxSkillNum
  if nMaxSkillNum > Partner.nMaxSkillNum then
    nMaxSkillNum = Partner.nMaxSkillNum
  end
  tbAttribNow.txtStar = nStar
  local nVitality, nDexterity, nStrength, nEnergy = Partner:RandomPotential(nGrowthType, nQualityLevel)
  local tbRebornPotentialList = {
    nStrength = nStrength,
    nEnergy = nEnergy,
    nVitality = nVitality,
    nDexterity = nDexterity
  }
  for _, value in pairs(tbRebornPotentialList) do
    tbAttribReborn.txtPotential = tbAttribReborn.txtPotential + value
  end
  for i = 1, 4 do
    tbAttribNow.txtPotential = tbAttribNow.txtPotential + curPartner.GetPotential(Partner.tbPotentialList[i])
  end
  for i = 1, nMaxSkillNum do
    local nSkillId, nSkillLevel = curPartner.GetSkillInfo(i)
    if nSkillId > 0 then
      tbAttribNow.txtSkill = tbAttribNow.txtSkill + 1
      tbAttribNow.txtSkillBook = tbAttribNow.txtSkillBook + nSkillLevel
    end
  end
  local nNowFightPower = curPartner.nFightPower
  local nRebornFightPower = Partner:GetFightPower(nPartnerTemplateId, nQualityLevel, nGrowthType, tbAttribReborn.txtLevel, tbRebornPotentialList, tbRebornSkillInfo, tbAttribReborn.txtStar)
  local nFetterId = Partner:GetFetterByTemplatedId(Partner.tbAllPartnerInfo[nPartnerTemplateId].nNpcTemplateId)
  local tbPartnerList = {}
  for i = 1, 4 do
    local nPartnerId = 0
    if 1 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nFetterId].nPartnerId1
    elseif 2 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nFetterId].nPartnerId2
    elseif 3 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nFetterId].nPartnerId3
    elseif 4 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nFetterId].nPartnerId4
    end
    self.tbAllPartnerList = self:GetAllPartners()
    local pCurPartner
    if nPartnerId > 0 then
      for _, pPartner in pairs(self.tbAllPartnerList) do
        local nNpcTemplateId = Partner.tbAllPartnerInfo[pPartner.nTemplateId].nNpcTemplateId
        if nNpcTemplateId == nPartnerId then
          pCurPartner = pPartner
          break
        end
      end
    end
    if pCurPartner then
      table.insert(tbPartnerList, pCurPartner)
    end
  end
  local nNowFriendLevel, nRebornFriendLevel = self:GetFriendLevel(tbPartnerList, nFetterId, nPartnerTemplateId)
  tbAttribNow.txtFriend = nNowFriendLevel
  tbAttribReborn.txtFriend = nRebornFriendLevel
  local tbReturnItem = Partner:GetRebornReturnItem(me, nPartnerId)
  self.pPanel:Label_SetText(tbControls.txtRebornFightPowerNow, nNowFightPower)
  self.pPanel:Label_SetText(tbControls.txtRebornFightPowerThen, nRebornFightPower)
  for attribute, value in pairs(tbAttribNow) do

    self.pPanel:Label_SetText(string.format(tbControls.txtRebornAttributeThen, attribute), tbAttribReborn[attribute])
  end
  local i = 1
  self.tbReturnItemScrollView:Clear()
  for nType, nCount in pairs(tbReturnItem) do
    if nCount > 0 then
      self.tbReturnItemScrollView:SetElement(i, function(tbElement)
        local nItemId = Partner:GetItemIdByType(nType, nPartnerTemplateId)
        local potentialLevelUpGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        potentialLevelUpGrid:SetItemByTemplate(nItemId, nCount)
        potentialLevelUpGrid:SetClickEnable(true)
      end)
      i = i + 1
    end
  end
  self.tbReturnItemScrollView:ResetFrom(i)
  self.nCost = Partner.ServeranceCost[nQualityLevel]
  self.pPanel:Label_SetText(tbControls.txtRebornCost, string.format("%d/%d", me.GetItemCountInBags(Partner.nPartnerRebornItemId), Partner.ServeranceCost[nQualityLevel]))
end

function tbWnd:UpdateCurPartnerRightAwakeningView()

    local szAttribName = tbAwakeningSkillInfo["Attrib" .. i .. "Type"]

    local tbAttribSection = Lib:SplitStr2Num(tbAwakeningSkillInfo["szLevelupSection" .. i], ",")

        tbAttribValue[nValue] = tbAttribValue[nValue] * nSkillLevel

      table.insert(tbNowSkillDescName, szDescNow)
      table.insert(tbNowSkillMagicDesc, szValueNow)
    end
  end

    local tbAttribValue = Lib:SplitStr2Num(tbAwakeningSkillInfo["Attrib" .. i .. "Value1"], ",")
    local tbAttribSection = Lib:SplitStr2Num(tbAwakeningSkillInfo["szLevelupSection" .. i], ",")
    if "" ~= szAttribName then
      for _, nValue in pairs(tbAttribSection) do
        tbAttribValue[nValue] = tbAttribValue[nValue] * (nSkillLevel + 1)
      end
      local _, szValueNow = FightSkill:GetMagicDescSplit(szAttribName, tbAttribValue)
      table.insert(tbNextSkillMagicDesc, szValueNow)
    end
  end
  for i = 1, 2 do

      self.pPanel:Label_SetText(tbControls.txtAwakeningSkill1Now, "0")

      self.pPanel:Label_SetText(tbControls.txtAwakeningSkill1Then, tbNextSkillMagicDesc[i])

  self.pPanel:Button_SetEnable(tbControls.btnAwakening, true)

function tbWnd:InitPartnerOrder(bUpdate)
  self.tbPosInfo = me.GetPartnerPosInfo()
  self.tbPartnerOrder = {}
  local tempPartnerOrder = {}
  local tempPartnerPos = {}
  for i = 1, Partner.nMaxPos do
    tempPartnerPos[self.tbPosInfo[i]] = i
  end
  local playerFlag = false
  for i = 1, Partner.nMaxOrder do
    local nPartnerId = me.GetUserValue(GetGroupKey("PartnerOrder"), i)
    self.tbPartnerOrder[i] = nPartnerId
    if nPartnerId > 0 and nil == tempPartnerPos[nPartnerId] then
      self.tbPartnerOrder[i] = 0
    elseif -1 == nPartnerId then
      playerFlag = true
    end
    tempPartnerOrder[nPartnerId] = i
  end
  local tbInsertPartner = {}
  if false == playerFlag then
    playerFlag = true
    table.insert(tbInsertPartner, -1)
  end
  for i = 1, Partner.nMaxPos do
    if self.tbPosInfo[i] > 0 and not tempPartnerOrder[self.tbPosInfo[i]] then
      table.insert(tbInsertPartner, self.tbPosInfo[i])
    end
  end
  for i = 1, #tbInsertPartner do
    for nIndex = 1, Partner.nMaxOrder do
      if 0 == self.tbPartnerOrder[nIndex] then
        self.tbPartnerOrder[nIndex] = tbInsertPartner[i]
        break
      end
    end
  end
  if bUpdate then
    self:UploadPartnerOrder()
  end
end

function tbWnd:GetFriendLevel(tbPartnerList, nFetterId, nPartnerTemplateId)
  local nFriendFightPower = 0
  local nFriendLevel = 0
  local nFriendRebornFightPower = 0
  local nFriendRebornLevel = 0
  for _, pPartner in pairs(tbPartnerList) do
    nFriendFightPower = nFriendFightPower + pPartner.nFightPower
    if pPartner.nTemplateId ~= nPartnerTemplateId then
      nFriendRebornFightPower = nFriendRebornFightPower + pPartner.nFightPower
    end
  end
  local nRequireFightPower = Partner.tbPartnerFetterSetting[nFetterId].nRequireFightPower
  nFriendLevel = math.modf(nFriendFightPower / nRequireFightPower) + 1
  nFriendRebornLevel = math.modf(nFriendRebornFightPower / nRequireFightPower) + 1
  return nFriendLevel, nFriendRebornLevel
end

function tbWnd:GetAllPartners()
  local function funcSort(a, b)
    return a.nFightPower > b.nFightPower
  end
  
  local tbPartnerList = {}
  local tbAllPartner = me.GetAllPartner()
  local tbTempBattlePartnerId = {}
  local tbBattlerPartnerList = {}
  local tbQualityPartnerList = {
    [1] = {},
    [2] = {},
    [3] = {},
    [4] = {},
    [5] = {}
  }
  local tbBattlePos = me.GetPartnerPosInfo()
  for _, nPartnerId in pairs(tbBattlePos) do
    if 0 ~= nPartnerId then
      local pPartner = me.GetPartnerObj(nPartnerId)
      table.insert(tbBattlerPartnerList, pPartner)
      tbTempBattlePartnerId[nPartnerId] = true
    end
  end
  table.sort(tbBattlerPartnerList, funcSort)
  for nPartnerId in pairs(tbAllPartner) do
    if nil == tbTempBattlePartnerId[nPartnerId] then
      local pPartner = me.GetPartnerObj(nPartnerId)
      local nQuality = Partner.tbAllPartnerInfo[pPartner.nTemplateId].nQualityLevel
      table.insert(tbQualityPartnerList[nQuality], pPartner)
    end
  end
  for i = 1, #tbQualityPartnerList do
    table.sort(tbQualityPartnerList[i], funcSort)
  end
  local count = 1
  for _, pPartner in pairs(tbBattlerPartnerList) do
    if nil ~= pPartner then
      pPartner.nSortId = count
      tbPartnerList[count] = pPartner
      count = count + 1
    end
  end
  for i = 1, #tbQualityPartnerList do
    for j = 1, #tbQualityPartnerList[i] do
      local pPartner = tbQualityPartnerList[i][j]
      if 1 == pPartner.GetMainDevFlag() then
        pPartner.nSortId = count
        tbPartnerList[count] = pPartner
        count = count + 1
      end
    end
  end
  return tbPartnerList
end

  local tbPosInfo = me.GetPartnerPosInfo()
  if bSetup then
    for _, nOtherPartnerId in pairs(tbPosInfo) do
      if nOtherPartnerId == nPartnerId then
        return -1
      end
    end
    for i = 1, Partner.nMaxPos do
      if 0 == tbPosInfo[i] then
        tbPosInfo[i] = nPartnerId
        break
      end
    end
    local tbPartnerIdMap = {}
    for nIdx, nPartnerId in pairs(tbPosInfo) do
      if 0 ~= nPartnerId then
        tbPartnerIdMap[nPartnerId] = true
      end
    end
    local nPartnerNum = Lib:CountTB(tbPartnerIdMap)
    if nPartnerNum == Partner.nMaxPos and not tbPartnerIdMap[nPartnerId] then
      me.CenterMsg(i18n.EasyGet("\229\144\140\228\188\180\228\184\138\233\152\181\230\149\176\233\135\143\229\183\178\232\190\190\233\153\144\229\136\182"))
      return -1
    end
    if 0 ~= nPartnerNum and (not Partner.PosNeedLevel[nPartnerNum] or Partner.PosNeedLevel[nPartnerNum] > me.nLevel or nPartnerNum > Partner.nMaxPos) then
      me.CenterMsg(i18n.EasyGet("\229\144\140\228\188\180\228\184\138\233\152\181\230\149\176\233\135\143\229\183\178\232\190\190\233\153\144\229\136\182"))
      return -1
    end
  else
    local nFoundIndex = false
    for nOtherIndex, nOtherPartnerId in pairs(tbPosInfo) do
      if nOtherPartnerId == nPartnerId then
        nFoundIndex = nOtherIndex
        break
      end
    end
    if nFoundIndex then
      tbPosInfo[nFoundIndex] = 0
    else
      return -1
    end
  end
  return tbPosInfo
end

function tbWnd:CheckPartnerInfoViewRedPoint(nKey, tbRedPointInfo)

  if nKey == Partner.emREDPOINT.StartUp then
    self.pPanel:SetActive(tbControls.toggleStarsUP .. "/Text/RedPoint", bLight)
    self.pPanel:SetActive(tbControls.btnStarLevelUp .. "/RedPoint", bLight)
  elseif nKey == Partner.emREDPOINT.GradeUp then
    self.pPanel:SetActive(tbControls.toggleInfo .. "/Text/RedPoint", bLight)
    self.pPanel:SetActive(tbControls.btnPartnerLevelUp .. "/RedPoint", bLight)
  elseif nKey == Partner.emREDPOINT.JueXing then
    self.pPanel:SetActive(tbControls.toggleAwakening .. "/Text/RedPoint", bLight)
  end
end

function tbWnd:CheckPartnerSKillRedPoint(nPartnerId, tbRedPointInfo)
  local bBtnLight = false
  local tbSkillRedPoint = tbRedPointInfo[Partner.emREDPOINT.Skill]
  local nMaxSkillCount = Partner.StarLevelExtSkillNum[me.GetPartnerObj(nPartnerId).GetStar()] or Partner.nMaxSkillNum
  nMaxSkillCount = nMaxSkillCount > Partner.nMaxSkillNum and Partner.nMaxSkillNum or nMaxSkillCount
  local nSkillCountBound = self.pPanel:GetObject(tbControls.goSkillPanel .. "/SkillList/Viewport/Content").transform.childCount - 1
  for i = 1, Partner.nMaxSkillNum do
    if i > nSkillCountBound then
      break
    end
    self.pPanel:SetActive(string.format(tbControls.goPartnerSkillElement, i) .. "/RedPoint", false)
  end
  for nIndex, bLight in pairs(tbSkillRedPoint) do
    if nSkillCountBound < nIndex then
      break
    end
    if bLight then
      bBtnLight = true
    end
    if nMaxSkillCount >= nIndex then
      self.pPanel:SetActive(string.format(tbControls.goPartnerSkillElement, nIndex) .. "/RedPoint", bLight)
    end
  end
  self.pPanel:SetActive(tbControls.toggleSkill .. "/Text/RedPoint", bBtnLight)
end
