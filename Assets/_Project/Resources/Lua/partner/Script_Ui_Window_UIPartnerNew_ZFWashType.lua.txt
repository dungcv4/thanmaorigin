local tbWnd = Ui:CreateClass("UIPartnerNew_ZFWashType", "UIPanel")
local tbControls = {
  btnSave = "btnSave",
  btnReset = "btnAttributeReset",
  btnAdd = "costCell/btnAdd",
  txtCost = "costCell/txtPotentialUpCost",
  UIItemGrid = "costCell/UIItemGrid",
  BeforeTemplate = "AttributeBefore/attributeCell/txtAttributeTemplate",
  AfterTemplate = "AttributeAfter/attributeCell/txtAttributeTemplate",
  AfterAttr = "AttributeAfter/attributeCell",
  imgEmpty = "AttributeAfter/imgEmpty"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnSave = function(self)
    PartnerZhenFa:SendClientReq(PartnerZhenFa.Protocol.ReqReplaceRandomAttrType)
  end,
  btnReset = function(self)
    local bRet = PartnerZhenFa:SendClientReq(PartnerZhenFa.Protocol.ReqWashRandomAttrType)
    if bRet then
      self.pPanel:SetActive(tbControls.btnReset, false)
      if self.nDelayClickResetTimer then
        Timer:Close(self.nDelayClickResetTimer)
        self.nDelayClickResetTimer = nil
      end
      self.nDelayClickResetTimer = Timer:Register(Env.GAME_FPS, function()
        self.pPanel:SetActive(tbControls.btnReset, true)
      end)
    end
  end
}

function tbWnd:OnCreate()
  self.tbAfterScrollView = Ui:NewScrollView(self, tbControls.AfterTemplate)
  self.tbBeforeScrollView = Ui:NewScrollView(self, tbControls.BeforeTemplate)
end

function tbWnd:OnOpen()
  self:UpdateView()
end

function tbWnd:OnClose()
  self.tbAfterScrollView:Clear()
  self.tbBeforeScrollView:Clear()
  if self.nDelayClickResetTimer then
    Timer:Close(self.nDelayClickResetTimer)
    self.nDelayClickResetTimer = nil
  end
end

function tbWnd:UpdateView()
  self.tbBeforeScrollView:Clear()
  local nCurAttrCount = PartnerZhenFa:GetCurRandomAttrCount(me)
  for nIndex = 1, nCurAttrCount do
    self.tbBeforeScrollView:SetElement(nIndex, function(tbElement)
      local szAttrDesc = PartnerZhenFa:GetRandomAttrDesc(nIndex)
      tbElement.pPanel:Label_SetText("Main", szAttrDesc)
    end)
  end
  local nAfterWashCount = PartnerZhenFa:GetCurWashRandomAttrCount(me)
  if nAfterWashCount == nCurAttrCount then
    self.tbAfterScrollView:Clear()
    for nIndex = 1, nCurAttrCount do
      self.tbAfterScrollView:SetElement(nIndex, function(tbElement)
        local szAttrDesc = PartnerZhenFa:GetRandomAttrDesc(nIndex, nil, true)
        tbElement.pPanel:Label_SetText("Main", szAttrDesc)
      end)
    end
    self.pPanel:SetActive(tbControls.AfterAttr, true)
    self.pPanel:SetActive(tbControls.btnSave, true)
    self.pPanel:SetActive(tbControls.imgEmpty, false)
  else
    self.pPanel:SetActive(tbControls.AfterAttr, false)
    self.pPanel:SetActive(tbControls.btnSave, false)
    self.pPanel:SetActive(tbControls.imgEmpty, true)
  end
  local tbZFLevelCfg = PartnerZhenFa:GetZFLevelCfg()
  local nHasCount = me.GetItemCountByTypeInBag(tbZFLevelCfg.nWashTypeCostId)
  local nNeedCount = tbZFLevelCfg.nWashTypeCostCount
  if nHasCount < nNeedCount then
    self.pPanel:Label_SetText(tbControls.txtCost, i18n.EasyGet("<color=#c50505>{0}/{1}</color>", nHasCount, nNeedCount))
  else
    self.pPanel:Label_SetText(tbControls.txtCost, i18n.EasyGet("{0}/{1}", nHasCount, nNeedCount))
  end
  local pGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  pGrid:SetItemByTemplate(tbZFLevelCfg.nWashTypeCostId, 0)
  pGrid:SetClickEnable(true)
  Ui:Button_BindEvent(self.pPanel, tbControls.btnAdd, function()
    local tbWindow = Ui:GetWindow("UIPartnerNew")
    if tbWindow then
      tbWindow:UpdateJumpToWnd(tbZFLevelCfg.nWashTypeCostId)
    end
  end)
end

function tbWnd:OnZhenFaRefresh()
  self:UpdateView()
end
