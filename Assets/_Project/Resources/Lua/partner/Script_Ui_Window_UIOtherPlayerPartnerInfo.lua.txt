local tbWnd = Ui:CreateClass("UIOtherPlayerPartnerInfo", "UIPanel")
local FiveElements = {
  [1] = "/Metal",
  [2] = "/Wood",
  [3] = "/Water",
  [4] = "/Fire",
  [5] = "/Earth"
}
local tbControls = {
  txtFightPowerValue = "Middle/FightPower/txtFightPowerValue",
  imgTalentSkill = "Middle/TalentSkill/imgSkill",
  btnTalentSkillClick = "Middle/TalentSkill/btnClick",
  imgTalentSkillMask = "Middle/TalentSkill/imgMask",
  imgProtectSkill = "Middle/ProtectSkill/imgSkill",
  btnProtectSkillClick = "Middle/ProtectSkill/btnClick",
  imgProtectSkillMask = "Middle/ProtectSkill/imgMask",
  goAwakeningSkill = "Middle/AwakeningSkill",
  imgAwakeningSkill = "Middle/AwakeningSkill/imgSkill",
  btnAwakeningSkillClick = "Middle/AwakeningSkill/btnClick",
  imgAwakeningSkillMask = "Middle/AwakeningSkill/imgMask",
  imgAwakeningLvBg = "Middle/PartnerName/txtName/imgLVBg",
  txtAwakeningLvBg = "Middle/PartnerName/txtName/imgLVBg/txtLVName",
  imgModel = "Middle/imgModel",
  Stars = "Middle/StarsUp/Stars",
  imgCurrentPartnerStars = "Middle/StarsUp/Stars/imgStar%d_1",
  imgCurrentPartnerStarsBg = "Middle/StarsUp/Stars/imgStar%d_0",
  txtName = "Middle/PartnerName/txtName",
  imgFiveElements = "Middle/PartnerName/Series",
  partnerTemplate = "Middle/PartnerOrderShow/Partner1",
  btnInfo = "Right/btnList/btnInfo",
  btnPotential = "Right/btnList/btnPotential",
  btnSkill = "Right/btnList/btnSkill",
  panelInfo = "Right/panelInfo",
  PotentialPanel = "Right/PotentialPanel",
  SkillPanel = "Right/SkillPanel",
  partnerAttribTemplate = "Right/panelInfo/Attribute/AttribList/Viewport/Content/Template",
  txtStoryDetails = "Right/panelInfo/PartnerStory/PartnerStoryList/Viewport/Content/txtStoryDetails",
  txtPartnerAttribQuality = "Right/PotentialPanel/Potential%d/txtQuality",
  txtPartnerAttribQualityPercent = "Right/PotentialPanel/Potential%d/txtQualityPercent",
  imgPartnerAttribProgressBar = "Right/PotentialPanel/Potential%d/imgProgressBar",
  txtPartnerAttribQualityText = "Right/PotentialPanel/Potential%d/txtAttribute",
  partnerSkillTemplate = "Right/SkillPanel/SkillList/Viewport/Content/Template",
  partnerSkillElement = "Right/SkillPanel/SkillList/Viewport/Content/Element%d",
  btnPotentialInfo = "Right/PotentialPanel/Potential%d"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnTalentSkillClick = function(self)
    local nSkill1 = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nBaseSkillId
    local nPassiveSkill = 3
    Ui:OpenWindow("UIPartnerSkillShow", nPassiveSkill, self.curPartner, nSkill1, true, self.pAsyncRole)
  end,
  btnProtectSkillClick = function(self)
    local nSkill2 = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nProtectSkillId
    local nProtectSKill = 4
    Ui:OpenWindow("UIPartnerSkillShow", nProtectSKill, self.curPartner, nSkill2, true, self.pAsyncRole)
  end,
  btnAwakeningSkillClick = function(self)
    if not self.curPartner then
      return
    end
    local nSkill2 = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nJueXingSkillId
    local nAwakeningSKill = 6
    Ui:OpenWindow("UIPartnerSkillShow", nAwakeningSKill, self.curPartner, nSkill2, true, self.pAsyncRole)
  end
}

function tbWnd:OnCreate()
  self.curSelectedSkillPath = nil
  local pObj = self.pPanel:GetObject(tbControls.imgModel)
  self.pModel = pObj:GetComponent(Ui.UINpcModel)
  self.tbAttribScrollView = Ui:NewScrollView(self, tbControls.partnerAttribTemplate, nil, false, false)
  self.tbPartnerSkillScrollView = Ui:NewScrollView(self, tbControls.partnerSkillTemplate, function(tbOwner, nIndex, element)
    local nSkillId, nSkillLevel, nSkillExp = self.curPartner.GetSkillInfo(nIndex)
    if 0 == nSkillId then
    else
      Ui:OpenWindow("UIPartnerSkillShow", 1, self.curPartner, nIndex, true)
    end
  end, false, true)
  self.tbPartnerBtnScrollView = Ui:NewScrollView(self, tbControls.partnerTemplate, nil, false, true)
  self.tbCurrentPartnerInfoGroup = Ui:NewButtonGroup(self, {
    tbControls.btnInfo,
    tbControls.btnPotential,
    tbControls.btnSkill
  }, true, function(self, nIndex)
    self.nCurSelectedInfoIndex = nIndex
    self:UpdateCurrentPartnerInfo()
  end)
  self.tbPotentialInfoGroup = Ui:NewButtonGroup(self, {
    [1] = string.format(tbControls.btnPotentialInfo, 1),
    [2] = string.format(tbControls.btnPotentialInfo, 2),
    [3] = string.format(tbControls.btnPotentialInfo, 3),
    [4] = string.format(tbControls.btnPotentialInfo, 4)
  }, true, function(self, nIndex)
    for i = 1, 4 do
      if i == nIndex then
        self.pPanel:SetActive(string.format(self.btnPotentialInfo, i) .. "/imgSelect", true)
      else
        self.pPanel:SetActive(string.format(self.btnPotentialInfo, i) .. "/imgSelect", false)
      end
    end
    self.nSelectedPotentialIndex = nIndex
  end)
end

function tbWnd:OnOpen()
  self.pAsyncRole = self.tbOwner.pAsyncRole
  self.tbPartners = self.tbOwner.tbPartners
  self.nSelectPartnerIndex = nil
  for nIndex, value in pairs(self.tbPartners) do
    self.nSelectPartnerIndex = nIndex
    break
  end
  if not self.nSelectPartnerIndex then
    return
  end
  self.nCurSelectedInfoIndex = 1
  self.curPartner = self.tbPartners[self.nSelectPartnerIndex]
  self:PartnerBtnScrollView()
  self:UpdateView()
  self.tbCurrentPartnerInfoGroup:Select(1)
end

function tbWnd:OnClose()
  self.pModel:Release()
end

function tbWnd:UpdateView()
  if not self.curPartner then
    return
  end
  self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(self.curPartner.szName))
  local nSeries = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nSeries
  self:SetFiveElements(self.pPanel, tbControls.imgFiveElements, nSeries)
  self.pPanel:Label_SetText(tbControls.txtFightPowerValue, self.curPartner.nFightPower)
  local nNpcTemplateId = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nNpcTemplateId
  local nNpcResId = KNpc.GetNpcTemplateInfo(nNpcTemplateId).nNpcResID
  self.pModel:ShowNpc(nNpcResId)
  if Partner.tbPartnerScaleInfo[nNpcResId].Scale ~= "" then
    self.pModel.fModelScale = 125 * Partner.tbPartnerScaleInfo[nNpcResId].Scale
  else
    self.pModel.fModelScale = 125
  end
  local nQuality = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nQualityLevel
  local nCurPartnerStar = self.curPartner.GetStar()
  local nMoonCount, nStarCount = Partner:CheckPartnerUIStarInfo(nCurPartnerStar)
  local nLightMoonCount = 0
  local nLightStarCount = 0
  local szStarPath = "UI/atlas/common/img_Star01.png"
  local szLightStarPath = "UI/atlas/common/img_Star02.png"
  local szMoonPath = "UI/atlas/common/img_Star03.png"
  local szLightMoonPath = "UI/atlas/common/img_Star04.png"
  local szSunPath = "UI/atlas/common/img_Star05.png"
  local szLightSunPath = "UI/atlas/common/img_Star06.png"
  for i = 1, 8 do
    local nLightType = 0
    if nMoonCount > nLightMoonCount then
      nLightMoonCount = nLightMoonCount + 1
      nLightType = 2
    elseif nStarCount > nLightStarCount then
      nLightStarCount = nLightStarCount + 1
      nLightType = 1
    end
    local bLight = nLightType > 0
    local bOpen = Partner:CheckPartnerUIStarIsOpen(nQuality, nCurPartnerStar, i)
    self.pPanel:SetActive(string.format(tbControls.imgCurrentPartnerStarsBg, i), bOpen)
    self.pPanel:SetActive(string.format(tbControls.imgCurrentPartnerStars, i), bOpen and bLight)
    if 0 == nLightType then
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStarsBg, i), szStarPath)
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStars, i), szLightStarPath)
    elseif 1 == nLightType then
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStarsBg, i), szStarPath)
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStars, i), szLightStarPath)
    elseif 2 == nLightType then
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStarsBg, i), szMoonPath)
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStars, i), szLightMoonPath)
    elseif 3 == nLightType then
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStarsBg, i), szSunPath)
      self.pPanel:Sprite_SetSprite(string.format(tbControls.imgCurrentPartnerStars, i), szLightSunPath)
    end
  end
  local nBaseSkillId = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nBaseSkillId
  local nProtectSkillId = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nProtectSkillId
  local nAwakeningSkillId = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nJueXingSkillId
  local _, szProtectSkillIcon = Partner:GetPartnerProtectSkillNameAndIconPath(nil, nProtectSkillId)
  local _, szAwakeningSkillIcon = Partner:GetPartnerAwakeningSkillNameAndIconPath(nil, nAwakeningSkillId)
  local nAwakeningSKillLevel = 0
  for i = 1, 6 do
    if self.curPartner.dwId == self.pAsyncRole.GetPartnerOrder(i) then
      nAwakeningSKillLevel = self.pAsyncRole.GetPartnerAwakeningSkill(i)
      break
    end
  end
  self.pPanel:Sprite_SetSprite(tbControls.imgTalentSkill, FightSkill:GetSkillIconAtlas(nBaseSkillId))
  self.pPanel:Sprite_SetSprite(tbControls.imgProtectSkill, szProtectSkillIcon)
  self.pPanel:Sprite_SetSprite(tbControls.imgAwakeningSkill, szAwakeningSkillIcon)
  local nQualityLevel = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nQualityLevel
  local nTargetLevelTimeFrame = Partner:GetPartnerJueXingSkillNextTimeFrame(nQualityLevel, 1)
  if -999 == nTargetLevelTimeFrame or 1 ~= GetTimeFrameState(nTargetLevelTimeFrame) then
    self.pPanel:SetActive(tbControls.goAwakeningSkill, false)
  elseif nAwakeningSkillId <= 0 or nAwakeningSKillLevel <= 0 then
    self.pPanel:SetActive(tbControls.goAwakeningSkill, false)
  else
    self.pPanel:SetActive(tbControls.goAwakeningSkill, true)
  end
  self.pPanel:SetActive(tbControls.imgTalentSkillMask, false)
  self.pPanel:SetActive(tbControls.imgProtectSkillMask, false)
  self.pPanel:SetActive(tbControls.imgAwakeningSkillMask, false)
  if nAwakeningSKillLevel <= 0 then
    self.pPanel:SetActive(tbControls.imgAwakeningLvBg, false)
    self.pPanel:Label_SetText(tbControls.txtAwakeningLvBg, 0)
  else
    self.pPanel:SetActive(tbControls.imgAwakeningLvBg, true)
    self.pPanel:Label_SetText(tbControls.txtAwakeningLvBg, nAwakeningSKillLevel)
  end
  self:UpdateCurrentPartnerInfo()
end

function tbWnd:UpdateCurrentPartnerBasicInfo()
  if not self.curPartner then
    return
  end
  local nAiType = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nAiType
  self.pPanel:Label_SetText(tbControls.txtStoryDetails, Partner:GetAiTypeByIndex(nAiType))
  for i = 1, #Partner.Attribute do
    self.tbAttribScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local attribute = self:GetAttributeByIndex(self.curPartner, Partner.Attribute[i])
      local typeName = Partner:GetAttributeNameByIndex(Partner.Attribute[i])
      pPanel:Label_SetText("Text", string.format("%s %d", typeName, attribute))
    end)
  end
end

function tbWnd:SetFiveElements(pPanel, path, nSeries)
  for i = 1, #FiveElements do
    pPanel:SetActive(path .. FiveElements[i], false)
  end
  pPanel:SetActive(path .. FiveElements[nSeries], true)
end

function tbWnd:GetAttributeByIndex(curPartner, nIndex)
  if "nStrength" == nIndex then
    return curPartner.GetPotential(Partner.POTENTIAL_TYPE_STRENGTH)
  elseif "nEnergy" == nIndex then
    return curPartner.GetPotential(Partner.POTENTIAL_TYPE_ENERGY)
  elseif "nVitality" == nIndex then
    return curPartner.GetPotential(Partner.POTENTIAL_TYPE_VITALITY)
  elseif "nDexterity" == nIndex then
    return curPartner.GetPotential(Partner.POTENTIAL_TYPE_DEXTERITY)
  end
  return 0
end

function tbWnd:UpdateCurrentPartnerInfo()
  if not self.curPartner then
    return
  end
  if 1 == self.nCurSelectedInfoIndex then
    self.pPanel:SetActive(tbControls.panelInfo, true)
    self.pPanel:SetActive(tbControls.PotentialPanel, false)
    self.pPanel:SetActive(tbControls.SkillPanel, false)
    self:UpdateCurrentPartnerBasicInfo()
  elseif 2 == self.nCurSelectedInfoIndex then
    self.pPanel:SetActive(tbControls.panelInfo, false)
    self.pPanel:SetActive(tbControls.PotentialPanel, true)
    self.pPanel:SetActive(tbControls.SkillPanel, false)
    self:UpdateCurrentPartnerPotentialoInfo()
  elseif 3 == self.nCurSelectedInfoIndex then
    self.pPanel:SetActive(tbControls.panelInfo, false)
    self.pPanel:SetActive(tbControls.PotentialPanel, false)
    self.pPanel:SetActive(tbControls.SkillPanel, true)
    self:UpdateCurrentPartnerSkillInfo()
  end
end

function tbWnd:UpdateCurrentPartnerPotentialoInfo()
  for i = 1, 4 do
    local potentialValue = self.curPartner.GetPotential(Partner.tbPotentialList[i])
    local potentialValueLimit = Partner:GetLimitPotentialByObj(self.curPartner, Partner.tbPotentialList[i])
    local npotentialMaxLimitValue = Partner:GetXiSuiPotentialByObj(self.curPartner, Partner.tbPotentialList[i])
    local potentialProgress = potentialValue / potentialValueLimit
    self.pPanel:Label_SetText(string.format(tbControls.txtPartnerAttribQuality, i), Partner:CheckPotentialLevel(potentialValueLimit, npotentialMaxLimitValue))
    self.pPanel:Label_SetText(string.format(tbControls.txtPartnerAttribQualityPercent, i), potentialValue .. "/" .. potentialValueLimit)
    self.pPanel:Sprite_SetFill(string.format(tbControls.imgPartnerAttribProgressBar, i), potentialProgress)
    self.pPanel:Label_SetText(string.format(tbControls.txtPartnerAttribQualityText, i), i18n.Parse(Partner.PotentialName[Partner.tbPotentialList[i]]))
  end
end

function tbWnd:UpdateCurrentPartnerSkillInfo()
  local nQuality = Partner.tbAllPartnerInfo[self.curPartner.nTemplateId].nQualityLevel
  local nMaxSkillNum = Partner.nMaxSkillNum
  for i = 1, nMaxSkillNum do
    local bOpen = Partner:CheckStarIsOpen(nQuality, i - 1)
    if bOpen then
      local nSkillId, nSkillLevel, _ = self.curPartner.GetSkillInfo(i)
      self.tbPartnerSkillScrollView:SetElement(i, function(tbElement)
        self:SetPartnerSkillInfo(tbElement.pPanel, nSkillId, nSkillLevel, i)
      end)
    end
  end
end

function tbWnd:SetPartnerSkillInfo(pPanel, nSkillId, nSkillLevel, nPos)
  pPanel:SetActive("Lock", false)
  pPanel:SetActive("Empty", false)
  if 0 == nSkillId then
    pPanel:SetActive("txtSkillName", false)
    pPanel:SetActive("txtSkillLevel", false)
    pPanel:SetActive("Skill", false)
    pPanel:SetActive("Skill/imgMask", true)
  else
    pPanel:SetActive("txtSkillName", true)
    pPanel:SetActive("txtSkillLevel", true)
    pPanel:SetActive("Skill", true)
    pPanel:SetActive("Skill/imgMask", false)
    pPanel:Label_SetText("txtSkillName", i18n.Parse(Partner.tbPartnerSkillSetting[nSkillId].szName))
    pPanel:Label_SetText("txtSkillLevel", i18n.EasyGet("\231\173\137\231\186\167{0}", nSkillLevel))
    pPanel:Sprite_SetSprite("Skill/imgSkill", Partner.tbPartnerSkillSetting[nSkillId].szIconPath)
  end
  local nMaxSkillNum = Partner.StarLevelExtSkillNum[self.curPartner.GetStar()] or Partner.nMaxSkillNum
  if nMaxSkillNum > Partner.nMaxSkillNum then
    nMaxSkillNum = Partner.nMaxSkillNum
  end
  if nPos > nMaxSkillNum then
    pPanel:SetActive("txtSkillName", false)
    pPanel:SetActive("txtSkillLevel", false)
    pPanel:SetActive("Skill", false)
    pPanel:SetActive("Skill/imgMask", false)
  end
end

function tbWnd:PartnerBtnScrollView()
  self.nPartnerLimit = 0
  local nLevel = self.pAsyncRole.GetLevel()
  for i = #Partner.PosNeedLevel, 1, -1 do
    if nLevel >= Partner.PosNeedLevel[i] then
      self.nPartnerLimit = i
      break
    end
  end
  if self.nPartnerLimit <= 0 then
    Lib:LogErrFormat("UIOtherPlayerPartnerInfo:PartnerBtnScrollView() nPartnerLimit %d is invalid, nLevel = %d", self.nPartnerLimit, nLevel)
    return
  end
  self.tbPartnerBtnScrollView:Clear()
  for nIndex = 1, self.nPartnerLimit do
    self.tbPartnerBtnScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      local pPartner = self.tbPartners[nIndex]
      for i = 1, 3 do
        pPanel:SetActive(Partner.tbImgQualityBgEffectList[i], false)
      end
      if pPartner then
        pPanel:SetActive("imgEmpty", false)
        pPanel:SetActive("imgNameBg", true)
        pPanel:Label_SetText("txtName", i18n.Parse(pPartner.szName))
        local nQuality = Partner.tbAllPartnerInfo[pPartner.nTemplateId].nQualityLevel
        if nQuality and Partner.tbImgQualityBgEffectList[nQuality] then
          pPanel:SetActive(Partner.tbImgQualityBgEffectList[nQuality], true)
        end
        pPanel:Sprite_SetSprite("imgNameBg", Partner.tbPartnerNameBg[nQuality])
        pPanel:SetActive("imgSelected", self.nSelectPartnerIndex == nIndex)
        Ui:Button_BindEvent(pPanel, "Main", self.OnPartnerBtnClick, self, nIndex)
      else
        pPanel:SetActive("imgEmpty", true)
        pPanel:SetActive("imgNameBg", false)
        pPanel:SetActive("imgSelected", false)
        pPanel:Label_SetText("txtName", "")
      end
    end)
  end
end

function tbWnd:OnPartnerBtnClick(nIndex)
  if not self.tbPartners[nIndex] then
    return
  end
  self.nSelectPartnerIndex = nIndex
  self.curPartner = self.tbPartners[nIndex]
  for i = 1, self.nPartnerLimit do
    local tbElement = self.tbPartnerBtnScrollView:GetElement(i)
    if not tbElement then
      return
    end
    tbElement.pPanel:SetActive("imgSelected", i == nIndex)
  end
  self:UpdateView()
end
