local tbWnd = Ui:CreateClass("UIPartnerNew_ZFInfo", "UIPanel")
local tbControls = {
  btnOrder = "btnOrder",
  btnSwitchZF = "btnZhenFa",
  txtZhenFaName = "txtZhenFaName",
  txtFightPower = "txtFightPower",
  txtFixAttr = "Attribute/attributeCell%d/txtAttribute",
  txtFixEnhance = "Attribute/attributeCell%d/txtAttribute/txtEnhance",
  RandomAttrTemplate = "RandomAttribute/attributeCell/txtAttributeTemplate"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnOrder = function(self)
    local tbUIPartnerNew = Ui:GetWindow("UIPartnerNew")
    if tbUIPartnerNew then
      tbUIPartnerNew:MainPanelButtonSelect(2)
    end
  end,
  btnSwitchZF = function(self)
    local tbWindow = Ui:GetWindow("UIPartnerNew")
    if tbWindow then
      tbWindow:UpdateZFReplaceWnd()
    end
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.RandomAttrTemplate)
end

function tbWnd:OnOpen()
  self:UpdateView()
end

function tbWnd:OnClose()
  self.tbScrollView:Clear()
end

function tbWnd:UpdateView()
  local tbGroupCfg = PartnerZhenFa:GetPosGroupCfg(me)
  if not tbGroupCfg then
    LogErr("UIPartnerNew_ZFInfo:UpdateView Error GetCurPosGroupCfg is nil")
    return
  end
  self.pPanel:Label_SetText(tbControls.txtZhenFaName, i18n.Parse(tbGroupCfg.szName))
  local nFightPower = PartnerZhenFa:CalcFightPower(me)
  self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\233\152\181\230\179\149\230\136\152\229\138\155\239\188\154{0}", nFightPower))
  for nIndex = 1, PartnerZhenFa.nFixAttrCount do
    local szAttribute = PartnerZhenFa:GetFixAttrWithEyeDesc(nIndex)
    self.pPanel:Label_SetText(string.format(tbControls.txtFixAttr, nIndex), szAttribute)
    local nLevel, _, szFinalValue = PartnerZhenFa:GetFixAttrEnhanceDescInfo(nIndex)
    local szEnhance = i18n.EasyGet("\229\188\186\229\140\150\239\188\154Lv.{0}   {1}", nLevel, szFinalValue or 0)
    self.pPanel:Label_SetText(string.format(tbControls.txtFixEnhance, nIndex), szEnhance)
  end
  self.tbScrollView:Clear()
  local nRandomAttrCount = PartnerZhenFa:GetCurRandomAttrCount(me)
  for nIndex = 1, nRandomAttrCount do
    local szRandomAttr = PartnerZhenFa:GetRandomAttrDesc(nIndex)
    if szRandomAttr then
      self.tbScrollView:SetElement(nIndex, function(tbElement)
        tbElement.pPanel:Label_SetText("Main", szRandomAttr)
      end)
    end
  end
end

function tbWnd:OnZhenFaRefresh()
  self:UpdateView()
end
