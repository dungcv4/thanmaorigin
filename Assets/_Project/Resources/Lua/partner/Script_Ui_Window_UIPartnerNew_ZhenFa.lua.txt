local tbWnd = Ui:CreateClass("UIPartnerNew_ZhenFa", "UIPanel")
local tbControls = {
  Activate = "Activate",
  txtActivateCondition = "Activate/txtTitle/txtActivateCondition",
  btnOrder = "Activate/txtTitle/btnOrder",
  btnActivate = "Activate/txtTitle/btnActivate",
  Right = "Right",
  panelInfo = "Right/panelInfo",
  panelEnhance = "Right/panelEnhance",
  panelLevelUp = "Right/panelLevelUp",
  panelWashType = "Right/panelAttributeReset",
  panelWashVaule = "Right/panelAttributeResetNum",
  ToggleInfo = "Right/ToggleList/ToggleInfo",
  ToggleEnhance = "Right/ToggleList/ToggleEnhance",
  ToggleLevelUp = "Right/ToggleList/ToggleLevelUp",
  ToggleWashType = "Right/ToggleList/ToggleAttributeReset",
  ToggleWashVaule = "Right/ToggleList/ToggleAttributeResetNum"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnOrder = function(self)
    local tbUIPartnerNew = Ui:GetWindow("UIPartnerNew")
    if tbUIPartnerNew then
      tbUIPartnerNew:MainPanelButtonSelect(2)
    end
  end,
  btnActivate = function(self)
    if PartnerZhenFa:IsZFActivate(me) then
      return
    end
    PartnerZhenFa:SendClientReq(PartnerZhenFa.Protocol.ReqActivateZF, {
      PartnerZhenFa.nDefaultZFPos
    })
  end
}
tbWnd.PANEL_ZF_INFO = 1
tbWnd.PANEL_ZF_ENHANCE = 2
tbWnd.PANEL_ZF_LEVELUP = 3
tbWnd.PANEL_ZF_WASHTYPE = 4
tbWnd.PANEL_ZF_WASHVAL = 5
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_ZF_INFO] = {
    CLASS = "UIPartnerNew_ZFInfo",
    OBJ = tbControls.panelInfo,
    BTN = tbControls.ToggleInfo,
    OpenFunc = function()
      return PartnerZhenFa:IsZFActivate(me)
    end
  },
  [tbWnd.PANEL_ZF_ENHANCE] = {
    CLASS = "UIPartnerNew_ZFEnhance",
    OBJ = tbControls.panelEnhance,
    BTN = tbControls.ToggleEnhance,
    OpenFunc = function()
      return PartnerZhenFa:IsZFActivate(me)
    end
  },
  [tbWnd.PANEL_ZF_LEVELUP] = {
    CLASS = "UIPartnerNew_ZFLevelUp",
    OBJ = tbControls.panelLevelUp,
    BTN = tbControls.ToggleLevelUp,
    OpenFunc = function()
      return PartnerZhenFa:IsZFActivate(me)
    end
  },
  [tbWnd.PANEL_ZF_WASHTYPE] = {
    CLASS = "UIPartnerNew_ZFWashType",
    OBJ = tbControls.panelWashType,
    BTN = tbControls.ToggleWashType,
    OpenFunc = function()
      return PartnerZhenFa:IsZFActivate(me)
    end
  },
  [tbWnd.PANEL_ZF_WASHVAL] = {
    CLASS = "UIPartnerNew_ZFWashVal",
    OBJ = tbControls.panelWashVaule,
    BTN = tbControls.ToggleWashVaule,
    OpenFunc = function()
      return PartnerZhenFa:IsZFActivate(me)
    end
  }
}
tbWnd.tbColor = {szRed = "C50505", szPurple = "6D05C5"}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
end

function tbWnd:OnOpen(nSelectPanel)
  self:RegRedPoint()
  self:UpdateView()
  if PartnerZhenFa:IsZFActivate(me) then
    if not self.tbTabControl:OnOpen(nSelectPanel or self.PANEL_ZF_INFO) then
      return
    end
    self.tbTabControl:UpdateButtonGroup()
  end
  PartnerZhenFa:TryFadeDailyRedPoint()
end

function tbWnd:OnClose()
  self:UnRegRedPoint()
  self.tbTabControl:ClearSelected()
end

function tbWnd:RegRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnActivate, {
    RedPointMgr.EVENT.PartnerZFActivate
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.ToggleEnhance, {
    RedPointMgr.EVENT.PartnerZFEnhance
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.ToggleLevelUp, {
    RedPointMgr.EVENT.PartnerZFLevelUp
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.ToggleWashVaule, {
    RedPointMgr.EVENT.PartnerZFWashRandomVal
  })
end

function tbWnd:UnRegRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnActivate)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.ToggleEnhance)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.ToggleLevelUp)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.ToggleWashVaule)
end

function tbWnd:UpdateView()
  if PartnerZhenFa:IsZFActivate(me) then
    self.pPanel:SetActive(tbControls.Right, true)
    self.pPanel:SetActive(tbControls.Activate, false)
  else
    self.pPanel:SetActive(tbControls.Right, false)
    self.pPanel:SetActive(tbControls.Activate, true)
    local bCanActivate = PartnerZhenFa:CheckCanActivate(me, PartnerZhenFa.nDefaultZFPos)
    self.pPanel:Button_SetEnable(tbControls.btnActivate, bCanActivate)
    local nUsefulCount = PartnerZhenFa:GetUsefulPartnerCount(me)
    local nNeedCount = PartnerZhenFa.Cfg.nActivatePartnerCount
    local szQualityLimit = i18n.Get(PartnerZhenFa.Cfg.szPartnerQualityDesc)
    local szActivateCondition = ""
    if nUsefulCount >= nNeedCount then
      szActivateCondition = i18n.EasyGet("\228\184\138\233\152\181{0}/{1}\228\184\170<color=#{2}>{3}</color>\229\143\138\228\187\165\228\184\138\231\154\132\229\144\140\228\188\180\n\232\174\190\231\189\1741\228\184\170<color=#{4}>{5}</color>\230\136\150\230\155\180\233\171\152\229\147\129\232\180\168\231\154\132\229\144\140\228\188\180\228\184\186\233\152\181\231\156\188", nUsefulCount, nNeedCount, self.tbColor.szPurple, szQualityLimit, self.tbColor.szPurple, szQualityLimit)
    else
      szActivateCondition = i18n.EasyGet("\228\184\138\233\152\181<color=#{0}>{1}</color>/{2}\228\184\170<color=#{3}>{4}</color>\229\143\138\228\187\165\228\184\138\231\154\132\229\144\140\228\188\180\n\232\174\190\231\189\1741\228\184\170<color=#{5}>{6}</color>\230\136\150\230\155\180\233\171\152\229\147\129\232\180\168\231\154\132\229\144\140\228\188\180\228\184\186\233\152\181\231\156\188", self.tbColor.szRed, nUsefulCount, nNeedCount, self.tbColor.szPurple, szQualityLimit, self.tbColor.szPurple, szQualityLimit)
    end
    self.pPanel:Label_SetText(tbControls.txtActivateCondition, szActivateCondition)
  end
end

function tbWnd:OnZhenFaRefresh(nRefreshType, ...)
  self:UpdateView()
  if PartnerZhenFa:IsZFActivate(me) then
    local nPanelType = self.tbTabControl:GetCurPanelType()
    if not nPanelType then
      self.tbTabControl:OnOpen(self.PANEL_ZF_INFO)
      nPanelType = self.tbTabControl:GetCurPanelType()
    end
    local pPanel = self.tbTabControl:GetSubPanel(nPanelType)
    if pPanel.OnZhenFaRefresh then
      pPanel:OnZhenFaRefresh(nRefreshType, ...)
    end
    self.tbTabControl:UpdateButtonGroup()
  end
end

function tbWnd:SetActive(bActive, nPanel)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    if not self.tbTabControl:GetCurPanelType() then
      self:OnOpen(nPanel)
    end
  else
    self:OnClose()
  end
end
