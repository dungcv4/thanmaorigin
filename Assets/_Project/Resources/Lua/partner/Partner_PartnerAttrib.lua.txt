Require("CommonScript/AttributeModule/AttributeModule.lua")

function Partner:OnEquip(pPlayer)
  self:InitAllPartnerFetterMagicAttrib(pPlayer)
  self:InitPartnerSkillBookMagicAttrib(pPlayer)
  self:InitPosPartnerProtectSkill(pPlayer)
  self:InitPosPartnerJueXingSkill(pPlayer)
end

function Partner:OnUnEquip(pPlayer)
  self:ClearPlayerFetterMagicAttrib(pPlayer)
  self:ClearPlayerSkillBookMagicAttrib(pPlayer)
  self:ClearProtectSkillMagicAttrib(pPlayer)
  self:ClearJueXingSkillMagicAttrib(pPlayer)
end

AttributeModule:Register("Partner", Partner)

function Partner:ClearPlayerFetterMagicAttrib(pPlayer)
  pPlayer = pPlayer or me
  for nFetterId, _ in pairs(self.tbPartnerFetterSetting) do
    local nCurFetterLevel = self:GetPartnerFetterLevel(pPlayer, nFetterId)
    if nCurFetterLevel >= 1 then
      self:ActiveFetterMagicAttribById(pPlayer.GetNpc(), nFetterId, nCurFetterLevel, false, pPlayer.dwID)
    end
  end
end

function Partner:ClearPlayerSkillBookMagicAttrib(pPlayer)
  pPlayer = pPlayer or me
  local tbCurPartnerPosInfo = pPlayer.GetPartnerPosInfo()
  for nCount, nPartnerId in pairs(tbCurPartnerPosInfo) do
    if nPartnerId > 0 then
      local pPartner = pPlayer.GetPartnerObj(nPartnerId)
      for i = 1, self.nMaxSkillNum do
        local nSkillId, nSkillLevel, nSkillExp = pPartner.GetSkillInfo(i)
        if nSkillId > 0 then
          self:ApplyPartnerSkillBookMagicAttrib(pPlayer.GetNpc(), nSkillId, nSkillLevel, false, false, pPlayer.dwID)
        end
      end
    end
  end
end

function Partner:ClearProtectSkillMagicAttrib(pPlayer)
  local tbPartnerPos = pPlayer.GetPartnerPosInfo()
  for nIndex, nPartnerId in pairs(tbPartnerPos) do
    if 0 ~= nPartnerId then
      local pPartner = pPlayer.GetPartnerObj(nPartnerId)
      if pPartner then
        local nProtectSkillId = self.tbAllPartnerInfo[pPartner.nTemplateId].nProtectSkillId
        local nFightPower = pPartner.nFightPower
        local nQualityLevel = self.tbAllPartnerInfo[pPartner.nTemplateId].nQualityLevel
        local nSkillLevel = self:GetProtectSkillLevel(nQualityLevel, nFightPower)
        nSkillLevel = nSkillLevel + pPlayer.nPartnerProtectExtraLevel
        self:ApplyPartnerProtectSkill(pPlayer.GetNpc(), nProtectSkillId, nSkillLevel, false, pPlayer.dwID)
      end
    end
  end
end

function Partner:ClearJueXingSkillMagicAttrib(pPlayer)
  local tbPartnerPos = pPlayer.GetPartnerPosInfo()
  for nIndex, nPartnerId in pairs(tbPartnerPos) do
    if 0 ~= nPartnerId then
      local pPartner = pPlayer.GetPartnerObj(nPartnerId)
      if pPartner then
        local nJueXingSkillId = self.tbAllPartnerInfo[pPartner.nTemplateId].nJueXingSkillId
        local nSkillLevel = self:GetPartnerJueXingLevel(pPlayer, nPartnerId)
        self:ApplyPartnerJueXingSkill(pPlayer.GetNpc(), nJueXingSkillId, nSkillLevel, false, pPlayer.dwID)
      end
    end
  end
end
