function Partner:GetPartnerDirAndLenInfo(bUseAsyncData, pPlayer, nPlayerId)
  local NormalLength = Partner.nPosDistance
  
  local nPlayerPos = 0
  local tbPartnerPos = {}
  local tbPartnerDirInfo = {}
  local tbPartnerLenInfo = {}
  if not bUseAsyncData then
    for i = 1, 6 do
      local nPartnerId = pPlayer.GetUserValue(GetGroupKey("PartnerOrder"), i)
      if -1 == nPartnerId then
        nPlayerPos = i
      elseif nPartnerId > 0 then
        tbPartnerPos[i] = nPartnerId
      end
    end
  else
    local pAsyncData = KPlayer.GetAsyncData(nPlayerId)
    for i = 1, 6 do
      local nPartnerId = pAsyncData.GetPartnerOrder(i)
      if -1 == nPartnerId then
        nPlayerPos = i
      elseif nPartnerId > 0 then
        tbPartnerPos[i] = nPartnerId
      end
    end
  end
  for nPosId, nPartnerId in pairs(tbPartnerPos) do
    if nPosId <= 3 and nPlayerPos <= 3 or nPosId >= 4 and nPlayerPos >= 4 then
      if 1 == nPlayerPos - nPosId then
        tbPartnerDirInfo[nPartnerId] = 64
        tbPartnerLenInfo[nPartnerId] = NormalLength
      elseif 2 == nPlayerPos - nPosId then
        tbPartnerDirInfo[nPartnerId] = 64
        tbPartnerLenInfo[nPartnerId] = NormalLength * 2
      elseif 1 == nPosId - nPlayerPos then
        tbPartnerDirInfo[nPartnerId] = 192
        tbPartnerLenInfo[nPartnerId] = NormalLength
      elseif 2 == nPosId - nPlayerPos then
        tbPartnerDirInfo[nPartnerId] = 192
        tbPartnerLenInfo[nPartnerId] = NormalLength * 2
      end
    elseif 3 == nPosId - nPlayerPos or 3 == nPlayerPos - nPosId then
      if nPosId < nPlayerPos then
        tbPartnerDirInfo[nPartnerId] = 0
        tbPartnerLenInfo[nPartnerId] = NormalLength
      elseif nPosId > nPlayerPos then
        tbPartnerDirInfo[nPartnerId] = 128
        tbPartnerLenInfo[nPartnerId] = NormalLength
      end
    elseif nPosId >= 4 and nPlayerPos <= 3 then
      if 1 == nPlayerPos and 5 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(1)) / 90 * 64 + 128
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      elseif 1 == nPlayerPos and 6 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(2)) / 90 * 64 + 128
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 5)
      elseif 2 == nPlayerPos and 4 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(1)) / 90 * 64 + 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      elseif 2 == nPlayerPos and 6 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(1)) / 90 * 64 + 128
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      elseif 3 == nPlayerPos and 4 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(2)) / 90 * 64 + 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 5)
      elseif 3 == nPlayerPos and 5 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(1)) / 90 * 64 + 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      end
    elseif nPosId <= 3 and nPlayerPos >= 4 then
      if 4 == nPlayerPos and 2 == nPosId then
        tbPartnerDirInfo[nPartnerId] = 256 - math.deg(math.atan(1)) / 90 * 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      elseif 4 == nPlayerPos and 3 == nPosId then
        tbPartnerDirInfo[nPartnerId] = 256 - math.deg(math.atan(2)) / 90 * 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 5)
      elseif 5 == nPlayerPos and 1 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(1)) / 90 * 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      elseif 5 == nPlayerPos and 3 == nPosId then
        tbPartnerDirInfo[nPartnerId] = 256 - math.deg(math.atan(1)) / 90 * 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      elseif 6 == nPlayerPos and 1 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(2)) / 90 * 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 5)
      elseif 6 == nPlayerPos and 2 == nPosId then
        tbPartnerDirInfo[nPartnerId] = math.deg(math.atan(1)) / 90 * 64
        tbPartnerLenInfo[nPartnerId] = math.sqrt(NormalLength * NormalLength * 2)
      end
    end
  end
  return tbPartnerDirInfo, tbPartnerLenInfo
end
