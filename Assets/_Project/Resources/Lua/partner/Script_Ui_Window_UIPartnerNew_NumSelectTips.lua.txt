local tbWnd = Ui:CreateWindow("UIPartnerNew_NumSelectTips")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtName = "node/imgBG/Content/PartnerName/txtName",
  imgNameBg = "node/imgBG/Content/PartnerName/imgNameBg",
  sliderNum = "node/imgBG/Content/SliderNum",
  txtHave = "node/imgBG/Content/txtHave",
  txtMin = "node/imgBG/Content/txtMin",
  txtMax = "node/imgBG/Content/txtMax",
  btnOK = "node/imgBG/Content/btnOK",
  txtDunWuNum = "node/imgBG/Content/txtNumber",
  txtDunWuX = "node/imgBG/Content/txtDouble",
  txtFullTips = "node/imgBG/Content/txtTips",
  btnReduce = "node/imgBG/Content/btnReduce",
  btnAdd = "node/imgBG/Content/btnAdd"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    Partner:SyncSelectedDunWuDecomposePartnerList(self.nTemplateId, self.nHaveCount)
    self:Close()
  end,
  btnReduce = function(self)
    if self.nHaveCount > 0 then
      self.pPanel:Slider_SetValue(self.sliderNum, self.nHaveCount - 1)
    end
  end,
  btnAdd = function(self)
    if self.nHaveCount < self.nCount then
      self.pPanel:Slider_SetValue(self.sliderNum, self.nHaveCount + 1)
    end
  end
}
tbWnd.tbOnSliderClick = {
  sliderNum = function(self, nValue)
    self.nHaveCount = nValue
    local nQualityLevel = Partner.tbAllPartnerInfo[self.nTemplateId].nQualityLevel
    local nDunWuValue = Partner.PartnerJueXingExpValue[nQualityLevel]
    self.pPanel:Label_SetText(self.txtHave, i18n.EasyGet("\229\183\178\233\128\137\230\139\169") .. " " .. self.nHaveCount)
    local nNowAllValue = 0
    if self.TargetPartnerTemplateId == self.nTemplateId then
      nNowAllValue = self.nHaveCount * nDunWuValue * Partner.nPartnerAwakeningDunWuDoubleValue
    else
      nNowAllValue = self.nHaveCount * nDunWuValue
    end
    if self.nPartnerJueXingExp > 0 then
      self.pPanel:Label_SetText(self.txtDunWuNum, i18n.EasyGet("\233\161\191\230\130\159\229\128\188\239\188\154{0}(+{1})/{2}", nNowAllValue + self.nTotalSelect + self.nPartnerJueXingExp, self.nPartnerJueXingExp, self.nNextLevelNeedDunWuValue))
    else
      self.pPanel:Label_SetText(self.txtDunWuNum, i18n.EasyGet("\233\161\191\230\130\159\229\128\188\239\188\154{0}/{1}", nNowAllValue + self.nTotalSelect + self.nPartnerJueXingExp, self.nNextLevelNeedDunWuValue))
    end
    if nNowAllValue > self.nLeftNeeded then
      self.pPanel:SetActive(self.txtFullTips, true)
      self.pPanel:Label_SetText(self.txtFullTips, i18n.EasyGet("\230\156\172\230\172\161\233\161\191\230\130\159\229\128\188\229\183\178\230\186\162\229\135\186{0}\239\188\140\230\186\162\229\135\186\233\131\168\229\136\134\229\143\175\231\148\168\228\189\156\228\184\139\230\172\161\229\141\135\231\186\167", nNowAllValue - self.nLeftNeeded))
    else
      self.pPanel:SetActive(self.txtFullTips, false)
    end
  end
}

function tbWnd:OnOpen(nTemplateId, nCount, nHaveCount, nTargetPartnerTemplateId, nLeftNeeded, nPartnerJueXingExp, nPartnerJueXingLevel, nNextLevelNeedDunWuValue)
  self.nTemplateId = nTemplateId
  self.TargetPartnerTemplateId = nTargetPartnerTemplateId
  self.nCount = nCount
  self.nHaveCount = nHaveCount
  self.nLeftNeeded = 0
  self.nPartnerJueXingExp = nPartnerJueXingExp
  self.nPartnerJueXingLevel = nPartnerJueXingLevel
  self.nNextLevelNeedDunWuValue = nNextLevelNeedDunWuValue
  local nQualityLevel = Partner.tbAllPartnerInfo[self.nTemplateId].nQualityLevel
  local nQualityLevel = Partner.tbAllPartnerInfo[self.nTemplateId].nQualityLevel
  local nDunWuValue = Partner.PartnerJueXingExpValue[nQualityLevel]
  local nNowDunWuValue = 0
  if self.TargetPartnerTemplateId == self.nTemplateId then
    self.nLeftNeeded = nLeftNeeded + self.nHaveCount * nDunWuValue * Partner.nPartnerAwakeningDunWuDoubleValue
    nNowDunWuValue = self.nHaveCount * nDunWuValue * Partner.nPartnerAwakeningDunWuDoubleValue
  else
    self.nLeftNeeded = nLeftNeeded + self.nHaveCount * nDunWuValue
    nNowDunWuValue = self.nHaveCount * nDunWuValue
  end
  self.nTotalSelect = nNextLevelNeedDunWuValue - nLeftNeeded - self.nPartnerJueXingExp - nNowDunWuValue
  self.pPanel:Label_SetText(self.txtName, i18n.Parse(Partner.tbAllPartnerInfo[nTemplateId].szName))
  self.pPanel:Label_SetText(self.txtMin, 0)
  self.pPanel:Label_SetText(self.txtMax, nCount)
  self.pPanel:Slider_SetRange(self.sliderNum, 0, nCount)
  self.pPanel:Slider_SetValue(self.sliderNum, nHaveCount)
  self.pPanel:Label_SetText(self.txtHave, i18n.EasyGet("\229\183\178\233\128\137\230\139\169") .. " " .. nHaveCount)
  local nQualityLevel = Partner.tbAllPartnerInfo[nTemplateId].nQualityLevel
  self.pPanel:Sprite_SetSprite(self.imgNameBg, Partner.tbPartnerNameBg[nQualityLevel])
  if self.nPartnerJueXingExp > 0 then
    self.pPanel:Label_SetText(self.txtDunWuNum, i18n.EasyGet("\233\161\191\230\130\159\229\128\188\239\188\154{0}(+{1})/{2}", nNowDunWuValue + self.nTotalSelect + self.nPartnerJueXingExp, self.nPartnerJueXingExp, nNextLevelNeedDunWuValue))
  else
    self.pPanel:Label_SetText(self.txtDunWuNum, i18n.EasyGet("\233\161\191\230\130\159\229\128\188\239\188\154{0}/{1}", nNowDunWuValue + self.nTotalSelect + self.nPartnerJueXingExp, nNextLevelNeedDunWuValue))
  end
  self.pPanel:SetActive(self.txtFullTips, false)
  if nTargetPartnerTemplateId == nTemplateId then
    self.pPanel:Label_SetText(self.txtDunWuX, "")
  else
    self.pPanel:Label_SetText(self.txtDunWuX, "")
  end
end
