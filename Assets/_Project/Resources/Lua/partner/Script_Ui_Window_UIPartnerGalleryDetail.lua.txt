local tbWnd = Ui:CreateWindow("UIPartnerGalleryDetail")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtPotentialCharacter = "node/imgBG/Content/Potential/Potential%d/txtCharacter",
  txtPotentialQualityPercent = "node/imgBG/Content/Potential/Potential%d/txtQualityPercent",
  imgQualityBar = "node/imgBG/Content/Potential/Potential%d/imgQualityBar",
  txtAttrib = "node/imgBG/Content/Potential/Potential%d/txtAttribute",
  txtPartnerName = "node/imgBG/Content/txtPartnerName",
  txtName = "node/imgBG/Content/Info/txtName",
  txtQuality = "node/imgBG/Content/Info/txtQuality",
  imgFiveElements = "node/imgBG/Content/Info/FiveElements",
  txtDesc = "node/imgBG/Content/Story/PartnerStoryList/Viewport/Content/txtStoryDetails",
  txtTalentSkill = "node/imgBG/Content/Info/txtTalentSkill",
  txtProtectSkill = "node/imgBG/Content/Info/txtProtectSkill",
  txtAwakeningSkill = "node/imgBG/Content/Attribute/AwakeningSkill/imgMark/txtMark",
  imgTalentSkill = "node/imgBG/Content/Info/TalentSkill/imgSkill",
  imgProtectSKill = "node/imgBG/Content/Info/ProtectSkill/imgSkill",
  imgAwakeningSkill = "node/imgBG/Content/Attribute/AwakeningSkill/imgSkill",
  btnTalentSkillClick = "node/imgBG/Content/Info/TalentSkill/btnClick",
  btnProtectSkillClick = "node/imgBG/Content/Info/ProtectSkill/btnClick",
  btnAwakeningSkillClick = "node/imgBG/Content/Attribute/AwakeningSkill/btnClick",
  imgTalentSkillMask = "node/imgBG/Content/Info/TalentSkill/imgMask",
  imgProtectSkillMask = "node/imgBG/Content/Info/ProtectSkill/imgMask",
  imgAwakeningSkillMask = "node/imgBG/Content/Attribute/AwakeningSkill/imgMask",
  goAwakeningSkill = "node/imgBG/Content/Attribute/AwakeningSkill",
  txtFriend = "node/imgBG/Content/Friend/txtFriend",
  imgModel = "node/imgBG/Content/imgModel"
}
local template = {
  partnerAttribTemplate = "node/imgBG/Content/Attribute/AttribList/Viewport/Content/Template"
}
local tbAtrributeSetting = Lib:EasyLoadTabFile("Setting/Partner/QualitySetting.tab", {
  "nGrowthType",
  "nQualityLevel"
}, {
  "nGrowthType",
  "nQualityLevel",
  "nAttrib1Value1"
})
local tbTemplateIdToSzNameSetting = Lib:EasyLoadTabFile("Setting/Partner/PartnerTemplate.tab", {
  "nNpcTemplateId"
}, {
  "nNpcTemplateId",
  "szName"
})
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnTalentSkillClick = function(self)
    local nSkill1 = Partner.tbAllPartnerInfo[self.nTemplateId].nBaseSkillId
    Ui:OpenWindow("UIPartnerSkillShow", 3, nil, nSkill1)
  end,
  btnProtectSkillClick = function(self)
    local nSkill2 = Partner.tbAllPartnerInfo[self.nTemplateId].nProtectSkillId
    Ui:OpenWindow("UIPartnerSkillShow", 5, nil, nSkill2)
  end,
  btnAwakeningSkillClick = function(self)
    local nSkill = Partner.tbAllPartnerInfo[self.nTemplateId].nJueXingSkillId
    Ui:OpenWindow("UIPartnerSkillShow", 6, nil, nSkill)
  end
}

function tbWnd:OnCreate()
  local pObj = self.pPanel:GetObject(self.imgModel)
  local pModel = pObj:GetComponent(Ui.UINpcModel)
  self.npcModel = pModel
  self.tbAttribScrollView = Ui:NewScrollView(self, template.partnerAttribTemplate, nil, false, false)
end

function tbWnd:OnOpen(nTemplateId)
  self.nTemplateId = nTemplateId
  self.nFetterId = Partner:GetFetterByTemplatedId(Partner.tbAllPartnerInfo[self.nTemplateId].nNpcTemplateId)
  self:UpdatePartnerBasicInfo()
  self:UpdatePartnerModelInfo()
  for i = 1, 4 do
    self.pPanel:Label_SetText(string.format(self.txtAttrib, i), i18n.Parse(Partner.PotentialName[Partner.tbPotentialList[i]]))
  end
end

function tbWnd:OnClose()
end

function tbWnd:UpdatePartnerBasicInfo()
  local szName = Partner.tbAllPartnerInfo[self.nTemplateId].szName
  self.nQualityLevel = Partner.tbAllPartnerInfo[self.nTemplateId].nQualityLevel
  local nSeries = Partner.tbAllPartnerInfo[self.nTemplateId].nSeries
  local nAiType = Partner.tbAllPartnerInfo[self.nTemplateId].nAiType
  local nBaseSkillId = Partner.tbAllPartnerInfo[self.nTemplateId].nBaseSkillId
  local nSkill2 = Partner.tbAllPartnerInfo[self.nTemplateId].nProtectSkillId
  local nAwakeningSkillId = Partner.tbAllPartnerInfo[self.nTemplateId].nJueXingSkillId
  self.nGrowthType = Partner.tbAllPartnerInfo[self.nTemplateId].nGrowthType
  self.pPanel:Label_SetText(self.txtPartnerName, i18n.Parse(szName))
  self.pPanel:Label_SetText(self.txtName, i18n.Parse(szName))
  self.pPanel:Label_SetText(self.txtQuality, Partner:GetQualityText(self.nQualityLevel))
  self:SetFiveElements(self.pPanel, self.imgFiveElements, nSeries)
  self.pPanel:Label_SetText(self.txtDesc, Partner:GetAiTypeByIndex(nAiType))
  local szProtectSkillName, szProtectSkillIcon = Partner:GetPartnerProtectSkillNameAndIconPath(nil, nSkill2)
  local _, szAwakeningSkillIcon = Partner:GetPartnerAwakeningSkillNameAndIconPath(me, nAwakeningSkillId)
  self.pPanel:Label_SetText(self.txtTalentSkill, i18n.EasyGet("\229\164\169\232\181\139") .. " " .. i18n.Parse(FightSkill:GetSkillName(nBaseSkillId)))
  self.pPanel:Label_SetText(self.txtProtectSkill, i18n.EasyGet("\230\138\164\228\184\187") .. " " .. i18n.Parse(szProtectSkillName))
  self.pPanel:Label_SetText(self.txtAwakeningSkill, i18n.EasyGet("\232\167\137\233\134\146"))
  self.pPanel:Sprite_SetSprite(self.imgTalentSkill, FightSkill:GetSkillIconAtlas(nBaseSkillId))
  self.pPanel:Sprite_SetSprite(self.imgProtectSKill, szProtectSkillIcon)
  self.pPanel:Sprite_SetSprite(self.imgAwakeningSkill, szAwakeningSkillIcon)
  self.pPanel:SetActive(self.imgTalentSkillMask, false)
  self.pPanel:SetActive(self.imgProtectSkillMask, false)
  self.pPanel:SetActive(self.imgAwakeningSkillMask, false)
  self.pPanel:SetActive(self.goAwakeningSkill, nAwakeningSkillId > 0)
  local nVitality, nDexterity, nStrength, nEnergy = Partner:RandomPotential(self.nGrowthType, self.nQualityLevel)
  local nHealth = tbAtrributeSetting[self.nGrowthType][self.nQualityLevel].nAttrib1Value1
  self.tbAttributeList = {
    nStrength = nStrength,
    nEnergy = nEnergy,
    nVitality = nVitality,
    nDexterity = nDexterity,
    nHealth = nHealth
  }
  self:UpdateInitAttributeInfo()
  local nLimitVitality, nLimitDexterity, nLimitStrength, nLimitEnergy = Partner:RandomLimitPotential(self.nGrowthType, self.nQualityLevel)
  self.tbPotentialList = {
    [1] = nStrength,
    [2] = nVitality,
    [3] = nEnergy,
    [4] = nDexterity
  }
  self.tbPotentialLimitList = {
    [1] = nLimitStrength,
    [2] = nLimitVitality,
    [3] = nLimitEnergy,
    [4] = nLimitDexterity
  }
  self:UpdateInitPotentialInfo()
  self.pPanel:Label_SetText(self.txtFriend, self:GetFetterDescInfo())
end

function tbWnd:UpdateInitAttributeInfo()
  for i = 1, #Partner.Attribute do
    self.tbAttribScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local attribute = self:GetAttributeByIndex(Partner.Attribute[i])
      local typeName = Partner:GetAttributeNameByIndex(Partner.Attribute[i])
      pPanel:Label_SetText("Text", string.format("%s %d", typeName, attribute))
    end)
  end
end

function tbWnd:GetAttributeByIndex(nIndex)
  if "nStrength" == nIndex then
    return self.tbAttributeList.nStrength
  elseif "nEnergy" == nIndex then
    return self.tbAttributeList.nEnergy
  elseif "nVitality" == nIndex then
    return self.tbAttributeList.nVitality
  elseif "nDexterity" == nIndex then
    return self.tbAttributeList.nDexterity
  end
  return 0
end

function tbWnd:UpdateInitPotentialInfo()
  for i = 1, #Partner.tbPotentialList do
    local potentialValue = self.tbPotentialList[i]
    local potentialValueLimit = self.tbPotentialLimitList[i]
    local potentialProgress = potentialValue / potentialValueLimit
    self.pPanel:Label_SetText(string.format(self.txtPotentialCharacter, i), Partner:CheckPotentialLevel(potentialValue, potentialValueLimit))
    self.pPanel:Label_SetText(string.format(self.txtPotentialQualityPercent, i), potentialValue .. "/" .. potentialValueLimit)
    self.pPanel:Sprite_SetFill(string.format(self.imgQualityBar, i), potentialProgress)
  end
end

function tbWnd:GetFetterDescInfo()
  local nameDesc = i18n.EasyGet("\228\190\160\231\188\152\231\187\132\229\144\136\239\188\154")
  for i = 1, 4 do
    if 0 ~= Partner.tbPartnerFetterSetting[self.nFetterId]["nPartnerId" .. i] then
      if 1 == i then
        nameDesc = nameDesc .. i18n.Parse(tbTemplateIdToSzNameSetting[Partner.tbPartnerFetterSetting[self.nFetterId]["nPartnerId" .. i]].szName)
      else
        nameDesc = nameDesc .. "\227\128\129" .. i18n.Parse(tbTemplateIdToSzNameSetting[Partner.tbPartnerFetterSetting[self.nFetterId]["nPartnerId" .. i]].szName)
      end
    end
  end
  return nameDesc
end

function tbWnd:UpdatePartnerModelInfo()
  local nNpcTemplateId = Partner.tbAllPartnerInfo[self.nTemplateId].nNpcTemplateId
  local nNpcResId = KNpc.GetNpcTemplateInfo(nNpcTemplateId).nNpcResID
  self.npcModel:ShowNpc(nNpcResId)
  if Partner.tbPartnerScaleInfo[nNpcResId].Scale ~= "" then
    self.npcModel.fModelScale = 125 * Partner.tbPartnerScaleInfo[nNpcResId].Scale
  else
    self.npcModel.fModelScale = 125
  end
end

function Partner:CreateTemplateIdToFetterList()
  self.tbTemplateIdToFetterList = {}
  for i = 1, 4 do
    local index = "nPartnerId" .. i
    for j = 1, #Partner.tbPartnerFetterSetting do
      if Partner.tbPartnerFetterSetting[j][index] ~= nil then
        self.tbTemplateIdToFetterList[Partner.tbPartnerFetterSetting[j][index]] = j
      end
    end
  end
end

function Partner:GetFetterByTemplatedId(nTemplateId)
  if self.tbTemplateIdToFetterList == nil then
    Partner:CreateTemplateIdToFetterList()
  end
  if self.tbTemplateIdToFetterList[nTemplateId] ~= nil then
    return self.tbTemplateIdToFetterList[nTemplateId]
  end
end

local FiveElements = {
  [1] = "/txtGold",
  [2] = "/txtWood",
  [3] = "/txtWater",
  [4] = "/txtFire",
  [5] = "/txtEarth"
}

function tbWnd:GetFiveElementsByIndex(nSeries)
  return FiveElements[nSeries]
end

function tbWnd:ResetFiveElements(pPanel, path)
  for i = 1, #FiveElements do
    pPanel:SetActive(path .. FiveElements[i], false)
  end
end

function tbWnd:SetFiveElements(pPanel, path, nSeries)
  self:ResetFiveElements(pPanel, path)
  pPanel:SetActive(path .. self:GetFiveElementsByIndex(nSeries), true)
end
