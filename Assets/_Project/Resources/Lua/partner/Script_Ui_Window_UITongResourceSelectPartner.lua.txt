local tbWnd = Ui:CreateClass("UITongResourceSelectPartner", "UIPanel")
tbWnd.tbControls = {
  goEmpty4Raycast = "Empty4Raycast",
  svTemplate = "imgBg/Scroll View/Viewport/Content/Template",
  scrollView = "imgBg/Scroll View",
  txtTitle = "imgBg/txtTitle"
}
tbWnd.tbOnClick = {
  goEmpty4Raycast = function(self)
    self:Hide()
  end
}

function tbWnd:OnCreate()
  self.sv = Ui:NewScrollView(self, self.svTemplate, nil, false, false)
  self.pScrollRect = self.pPanel:GetScrollRect(self.scrollView)
  self.pScrollRect.horizontal = false
end

function tbWnd:Show(panelResource, nCurMiner)
  self.panelResource = panelResource
  self.nCurMiner = nCurMiner
  self.pPanel:SetActive("Main", true)
  self:Update()
end

function tbWnd:Hide(bAbnormal)
  if not bAbnormal and self.panelResource then
    self.panelResource:OnPartnerSelectEnd()
    self.panelResource = nil
  end
  self.nCurMiner = nil
  self.pPanel:SetActive("Main", false)
  self.sv:Reset()
end

function tbWnd:IsShow()
  return self.pPanel:IsActive("Main")
end

function tbWnd:Update()
  if not self:IsShow() then
    return
  end
  local tbShow = ResourceBattleMining:GetPlayerPartnerListForSelect()
  for nIndex, v in ipairs(tbShow) do
    self.sv:SetElement(nIndex, function(tbElement, nIndex)
      tbElement.tbPartnerInfos = v
      self:SetElementInfo(tbElement, nIndex)
    end)
  end
  if self.nCurMiner and 0 ~= self.nCurMiner then
    self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\230\155\191\230\141\162\229\144\140\228\188\180\239\188\136\231\130\185\229\135\187\229\133\182\228\187\150\229\140\186\229\159\159\229\133\179\233\151\173\239\188\137"))
  else
    self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\233\128\137\230\139\169\229\144\140\228\188\180\239\188\136\231\130\185\229\135\187\229\133\182\228\187\150\229\140\186\229\159\159\229\133\179\233\151\173\239\188\137"))
  end
end

tbWnd.Wuxing2Txt = {
  [Npc.emSERIES_TYPE.SERIES_METAL] = "txtMetal",
  [Npc.emSERIES_TYPE.SERIES_WOOD] = "txtWood",
  [Npc.emSERIES_TYPE.SERIES_WATER] = "txtWater",
  [Npc.emSERIES_TYPE.SERIES_FIRE] = "txtFire",
  [Npc.emSERIES_TYPE.SERIES_EARTH] = "txtEarth"
}

function tbWnd:SetElementInfo(tbElement, nIndex)
  local nQuality = tbElement.tbPartnerInfos[1]
  local tbPartners = tbElement.tbPartnerInfos[2]
  local tbMinerCxt = self.panelResource:GetMinerCxt()
  local nPreferWuxing = self.panelResource:GetPreferWuxing(tbMinerCxt.nSelectMine)
  local setSelect = {}
  for nMineIndex, v in ipairs(tbMinerCxt.tbPos) do
    for nMinerPosIndex, nPartnerID in ipairs(v) do
      if 0 ~= nPartnerID then
        setSelect[nPartnerID] = true
      end
    end
  end
  tbElement.pPanel:Label_SetText("txtTittle", Partner.tbQualityText[nQuality]())
  tbElement.svPartner = tbElement.svPartner or Ui:NewScrollView(tbElement, "Template/PartnerName", nil, false, false)
  tbElement.svPartner:Reset()
  for i, v in ipairs(tbPartners) do
    tbElement.svPartner:SetElement(i, function(tbPartnerElement, i)
      local nTemplateId = v[1]
      local nSeries = v[2]
      local tbPartnerInfo = Partner:GetPartnerInfo(nTemplateId)
      tbPartnerElement.pPanel:Label_SetText("txtName", i18n.Parse(tbPartnerInfo.szName))
      if not setSelect[nTemplateId] then
        tbPartnerElement.pPanel:SetActive("Gray", false)
        for nWuxing, szWuxingTxt in pairs(self.Wuxing2Txt) do
          if nWuxing == nSeries then
            tbPartnerElement.pPanel:SetActive("btnSeriesText/" .. szWuxingTxt, true)
          else
            tbPartnerElement.pPanel:SetActive("btnSeriesText/" .. szWuxingTxt, false)
          end
        end
        if nPreferWuxing == nSeries then
          tbPartnerElement.pPanel:SetActive("imgArrows", true)
        else
          tbPartnerElement.pPanel:SetActive("imgArrows", false)
        end
        tbPartnerElement.pPanel:Sprite_SetSprite("imgNameBg", Partner.tbPartnerNameBg[tbPartnerInfo.nQualityLevel])
        tbPartnerElement.pPanel:Button_BindEvent("imgNameBg", function()
          self.panelResource:OnClickPartner(nTemplateId)
          self:Update()
        end)
      else
        tbPartnerElement.pPanel:SetActive("Gray", true)
        tbPartnerElement.pPanel:Label_SetText("Gray/btnSeriesText/txtMetal", Npc.Series[nSeries]())
        if nPreferWuxing == nSeries then
          tbPartnerElement.pPanel:SetActive("Gray/imgArrows", true)
        else
          tbPartnerElement.pPanel:SetActive("Gray/imgArrows", false)
        end
      end
    end)
  end
end
