local tbItem = Item:GetClass("PartnerJueXingDan")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function fnCompose()
    Ui:OpenWindowWithCallback(Partner.szPartnerUIClassName, function(_, tbWnd)
      local curPartner = tbWnd.tbPartnerList[tbWnd.nCurSelectedPartnerIndex]
      if not curPartner then
        return
      end
      local nQualityLevel = Partner.tbAllPartnerInfo[curPartner.nTemplateId].nQualityLevel
      local nTargetLevelTimeFrame = Partner:GetPartnerJueXingSkillNextTimeFrame(nQualityLevel, 1)
      if me.nLevel < Partner.nPartnerAwakeningLevelLimit or -999 == nTargetLevelTimeFrame or 1 ~= GetTimeFrameState(nTargetLevelTimeFrame) then
        me.CenterMsg(i18n.EasyGet("\230\173\164\229\144\140\228\188\180\228\184\141\230\187\161\232\182\179\232\167\137\233\134\146\230\157\161\228\187\182\239\188\140\232\175\183\232\135\170\232\161\140\233\128\137\230\139\169\232\166\129\232\167\137\233\134\146\231\154\132\229\144\140\228\188\180"))
      else
        tbWnd:ToggleGroupSelect(7)
        tbWnd:UpdateCurrentPartnerInfo()
      end
    end, nil, 1)
  end
  
  return {
    szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnFirst = fnCompose
  }
end
