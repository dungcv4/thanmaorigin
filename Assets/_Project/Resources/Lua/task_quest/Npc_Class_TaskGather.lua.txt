local tbNpc = Npc:GetClass("TaskGather")
tbNpc.tbMsgType = {
  [1] = function()
    return i18n.EasyGet("\233\135\135\233\155\134\228\184\173")
  end,
  [2] = function()
    return i18n.EasyGet("\229\188\128\229\144\175\228\184\173")
  end
}

function tbNpc:OnCreate(szParam)
end

function tbNpc:OnDialog(szParam)
  local bIsGahterNpc, nDelete, nMsgType = Task:IsGatherNpc(me, him)
  if not bIsGahterNpc then
    return
  end
  local szMsg
  if nMsgType and 0 ~= nMsgType then
    szMsg = tbNpc.tbMsgType[nMsgType]()
  else
    szMsg = tbNpc.tbMsgType[1]()
  end
  Progress:BeginProgress(me, szMsg, GameSetting.PROGRESS_TIME_TASK_GATHER, self.EndProcess, {
    self,
    me.dwID,
    him.nId,
    nDelete
  })
end

function tbNpc:EndProcess(nPlayerId, nNpcId, nDelete)
  local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
  if not pPlayer then
    return
  end
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc or pNpc.IsDelayDelete() then
    pPlayer.CenterMsg(i18n.EasyGet("\229\183\178\232\162\171\229\133\182\228\187\150\228\186\186\230\138\162\229\133\136\230\148\182\232\181\176\229\149\166"))
    return
  end
  Task:OnGatherFinish(pPlayer, pNpc)
  if 1 == nDelete then
    pNpc.Delete()
  end
end
