local tbWnd = Ui:CreateWindow("UICommerceTaskSubmit")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  ItemTemplate = "node/imgBG/Content/SubmitScroll/maskView/Content/ItemTemplate"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbItemScrollView = Ui:NewItemGridScrollView(self, self.ItemTemplate, 6, 1, "CommerceTaskSubmit")
  self.tbItemScrollView:Reset()
end

function tbWnd:OnOpen()
  local nTaskId = CommerceTask:GetCurTaskId()
  if nTaskId <= 0 then
    return
  end
  local tbInfo = CommerceTask:GetTaskCfg(nTaskId)
  if not tbInfo then
    return
  end
  if tbInfo.nItemTemplateId > 0 then
    return
  end
  local tbItemList = {}
  local tbItem = me.GetItemListInBag()
  for _, pItem in ipairs(tbItem) do
    if pItem.nItemType == tbInfo.nItemType and pItem.nLevel == tbInfo.nItemLevel then
      table.insert(tbItemList, {
        nItemId = pItem.dwId,
        bEnableClick = true
      })
    end
  end
  self.tbItemScrollView:UpdateItemList(tbItemList)
end

function tbWnd:OnSelectItem(nItemId)
  self:Close()
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_COMMERCETASK_SELECT_ITEM,
      self.OnSelectItem
    }
  }
end
