function KinFreightTask:OnCheckRedPoint()
  if not (KinFreightTask:IsKFTOpen(me) and me.dwKinId) or me.dwKinId <= 0 then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.KinFreightTask)
    return
  end
  self.tbKinFreightTaskData.nAcceptCount = self.tbKinFreightTaskData.nAcceptCount or 0
  local tbTask = self:GetKinFreightTask()
  if self.tbKinFreightTaskData.nAcceptCount < self.KinFreightTaskCfg.nDayAcceptCount and 0 == Lib:CountTB(tbTask) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.KinFreightTask)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.KinFreightTask)
  end
end

function KinFreightTask:IsDoingTask()
  local bRet, szErr = self:IsKFTOpen(me)
  if not bRet then
    return false
  end
  self.tbKinFreightTaskData.nAcceptCount = self.tbKinFreightTaskData.nAcceptCount or 0
  local nMax = self.KinFreightTaskCfg.nDayAcceptCount
  local nDegree = nMax - self.tbKinFreightTaskData.nAcceptCount
  if nDegree > 0 then
    return true
  end
  local tbTask = KinFreightTask:GetKinFreightTask()
  if 0 == Lib:CountTB(tbTask) then
    return false
  end
  return true
end

end

function KinFreightTask:GetShowTarget()
  local nFillingCount = KinFreightTask:GetFillingTaskCount(me)
  if nFillingCount < KinFreightTask.KinFreightTaskCfg.nFreightMinCount then
    nFillingCount = KinFreightTask.KinFreightTaskCfg.nFreightMinCount
  end
  local tbAwards = Award:GetAward(KinFreightTask.nAwardId, nFillingCount)
  return tbAwards
end
