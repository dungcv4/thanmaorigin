local tbWnd = Ui:CreateWindow("UIRequestPermission")
tbWnd.tbControls = {
  btnConfirm = "PanelMain/btnConfirm",
  btnClose = "PanelMain/btnClose",
  descTxt = "PanelMain/Text"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnConfirm = function(self)
    if self.szPermissionType == "Voice" then
      local nRet = CS.VoiceModule.CheckMicPermission()
      if -1 == nRet then
        CS.VoiceModule.JumpToPermissionSettingApp()
      end
    elseif self.szPermissionType == "Notification" then
      CS.NotificationModule.JumpToNotificationPermissionSetting()
    elseif self.szPermissionType == "WriteSDCard" then
      local nRet = Ui.SceneCameraController.GetExternalWriteSDCardPermission()
      if -1 == nRet then
        CS.VoiceModule.JumpToPermissionSettingApp()
      end
    end
    self:Close()
  end
}

function tbWnd:OnCreate()
  self:SetGroup(2)
  self.szPermissionType = ""
end

function tbWnd:OnOpenEnd(szDesc, szPermissionType)
  self.szPermissionType = szPermissionType
  self.pPanel:Label_SetText(self.descTxt, szDesc)
end
