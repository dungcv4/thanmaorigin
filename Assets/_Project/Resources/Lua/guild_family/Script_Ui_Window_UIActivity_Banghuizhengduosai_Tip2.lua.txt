local tbWnd = Ui:CreateWindow("UIActivity_Banghuizhengduosai_Tip2")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_TONGBATTLE_GUESS_DATA,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    }
  }
end

tbWnd.tbControls = tbControls

tbWnd.tbPageBtnIndex = {
  [tbWnd.PANEL_TYPE.CHAMPION] = tbControls.btnChampion,
  [tbWnd.PANEL_TYPE.COMPETITION] = tbControls.btnCompetition
}
tbWnd.DEFAULT_TYPE = tbWnd.PANEL_TYPE.btnCompetition
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnAdd = function(self)
    self:AddItem()
  end
}

function tbWnd:OnCreate()
  self.tbPageGroup = Ui:NewButtonGroup(self, self.tbPageBtnIndex, true, function(self, nIndex)
    self:ChangePanel(nIndex)
  end)
  self.tbScrollViewInfo = Ui:NewScrollView(self, tbControls.TemplateInfo)
  self.tbScrollViewTemplate = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  TongBattle:ApllySyncGuessData()
  self.tbPageGroup:Select(self.DEFAULT_TYPE)
  self.pPanel:SetActive(tbControls.UIMoneyPanel, false)
end

function tbWnd:OnClose()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:UpdatePanel()
  self.tbPageGroup:Select(self.nType)
end

function tbWnd:ChangePanel(nType)
  self.nType = nType
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nNum1, nNum2 = 0, 0
  if nType == self.PANEL_TYPE.CHAMPION then
    self.pPanel:SetActive(tbControls.txtGongAndDrum01, true)
    self.pPanel:SetActive(tbControls.txtHorn01, false)
    self.pPanel:SetActive(tbControls.txtGongAndDrum02, true)
    self.pPanel:SetActive(tbControls.txtHorn02, false)
    self.pPanel:SetActive(tbControls.txtInfo_L06, true)
    self.pPanel:SetActive(tbControls.cont_txtInfo_R, true)
    self.pPanel:SetActive(tbControls.Cont_txtInfo_L, false)
    self.pPanel:SetActive(tbControls.ScrollViewCompetition, false)
    self.pPanel:SetActive(tbControls.ScrollViewChampion, true)
    self.pPanel:SetActive(tbControls.imgIcon01, false)
    self.pPanel:SetActive(tbControls.imgIcon02, false)
    self.pPanel:SetActive(tbControls.imgIcon03, true)
    self.pPanel:SetActive(tbControls.imgIcon04, true)
    nNum1 = 0
    if TongBattle.tbMyChampionGuessData then
      for nStage, tbData in pairs(TongBattle.tbMyChampionGuessData) do
        if not tbData.bSendAward then
          nNum1 = tbData.nGuessNum
          break
        end
      end
    end
    nNum2 = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLuoGuItemID)
    self:UpdateChampionScroll()
  elseif nType == self.PANEL_TYPE.COMPETITION then
    self.pPanel:SetActive(tbControls.txtGongAndDrum01, false)
    self.pPanel:SetActive(tbControls.txtHorn01, true)
    self.pPanel:SetActive(tbControls.txtGongAndDrum02, false)
    self.pPanel:SetActive(tbControls.txtHorn02, true)
    self.pPanel:SetActive(tbControls.txtInfo_L06, false)
    self.pPanel:SetActive(tbControls.cont_txtInfo_R, false)
    self.pPanel:SetActive(tbControls.Cont_txtInfo_L, true)
    self.pPanel:ContentSizeFitter_Refresh(tbControls.Cont_txtInfo_L .. "/GameObject")
    self.pPanel:SetActive(tbControls.ScrollViewCompetition, true)
    self.pPanel:SetActive(tbControls.ScrollViewChampion, false)
    self.pPanel:SetActive(tbControls.imgIcon01, true)
    self.pPanel:SetActive(tbControls.imgIcon02, true)
    self.pPanel:SetActive(tbControls.imgIcon03, false)
    self.pPanel:SetActive(tbControls.imgIcon04, false)
    nNum1 = 0
    if TongBattle.tbMyMatchGuessData then
      for nGuessMatch, tbData in pairs(TongBattle.tbMyMatchGuessData) do
        nNum1 = nNum1 + tbData.nGuessNum
      end
    end
    nNum2 = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLaBaItemID)
    self:UpdateMatchScroll()
  end
  self.pPanel:Label_SetText(tbControls.txtNum01, nNum1)
  self.pPanel:Label_SetText(tbControls.txtNum02, nNum2)
end

function tbWnd:AddItem()
  if self.nType == self.PANEL_TYPE.CHAMPION then
    Shop:OpenShopBuyWidDefault(TongBattle.GuessCfg.nLuoGuItemID)
  elseif self.nType == self.PANEL_TYPE.COMPETITION then
    Shop:OpenShopBuyWidDefault(TongBattle.GuessCfg.nLaBaItemID)
  end
end

function tbWnd:UpdateChampionScroll()
  if not TongBattle.tbFinalsData then
    return
  end
  local nNowStage = TongBattle:GetNowFinalsSatge(TongBattle.tbFinalsData)
  local nGuessEndTime = TongBattle:GetStageGuessEndTime(TongBattle.nWarZone, nNowStage)
  if not nGuessEndTime then
    return
  end
  local bCanGuess = true
  if nGuessEndTime <= GetTime() then
    bCanGuess = false
  end
  if not TongBattle.ChampionGuessMultiple[nNowStage] then
    bCanGuess = false
  end
  local tbChampionGuessData
  if TongBattle.tbMyChampionGuessData then
    for nStage, tbData in pairs(TongBattle.tbMyChampionGuessData) do
      if not tbData.bSendAward then
        tbChampionGuessData = tbData
        break
      end
    end
  end
  if tbChampionGuessData and tbChampionGuessData.nStage ~= nNowStage then
    bCanGuess = false
  end
  self.tbScrollViewTemplate:Reset()
  local nIndex = 0
  for i = TongBattle.FinalsStageCfg[nNowStage].nStartMatch, TongBattle.FinalsStageCfg[nNowStage].nEndMatch do
    if TongBattle.tbFinalsData[i] then
      for j = 1, 2 do
        if TongBattle.tbFinalsData[i].tbTongList[j] then
          nIndex = nIndex + 1
          self.tbScrollViewTemplate:SetElement(nIndex, function(tbElementMatch)
            local pPanel = tbElementMatch.pPanel
            local nTongID = TongBattle.tbFinalsData[i].tbTongList[j]
            local tbTong = TongBattle.tbTongData[nTongID]
            if tbTong then
              pPanel:Label_SetText("txtPlayerName01", tbTong.szTongName)
              pPanel:Label_SetText("txtServer", GetServerName(tbTong.nServerId))
              pPanel:Label_SetText("txtFightingNum02", tbTong.nFightPower)
            end
            pPanel:Label_SetText("btnCheer_R/Text", i18n.EasyGet("\231\171\158\231\140\156"))
            pPanel:Label_SetText("btnFinish/Text", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
            if tbChampionGuessData then
              if tbChampionGuessData.nTongID == nTongID then
                pPanel:SetActive("btnCheer_R", true)
                pPanel:SetActive("btnFinish", false)
                pPanel:Button_SetEnable("btnCheer_R", bCanGuess)
                pPanel:SetActive("txtFinishNum", true)
                pPanel:Label_SetText("txtFinishNum", i18n.EasyGet("\229\183\178\231\171\158\231\140\156\239\188\154{0}", tbChampionGuessData.nGuessNum))
                Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
                  local nItemID = TongBattle.GuessCfg.nLuoGuItemID
                  local szItemName = Item:GetName(nItemID, me)
                  local nNum = me.GetItemCountByTypeInBag(nItemID)
                  if nNum <= 0 then
                    me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
                    return
                  end
                  local tbTongData = TongBattle.tbTongData[nTongID]
                  local tbData = {
                    {
                      nId = nTongID,
                      szName = tbTongData.szTongName
                    },
                    i18n.EasyGet("\230\136\152\229\138\155\239\188\154{0}", tbTongData.nFightPower),
                    i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\231\165\168\230\149\176\239\188\154"),
                    nil
                  }
                  local tbCallBack = {
                    TongBattle.RequestGuessChampion,
                    TongBattle,
                    nTongID
                  }
                  Ui:OpenWindow("UIActivity_Banghuizhengduosai_Tip", tbData, nItemID, nil, tbCallBack)
                end)
              else
                pPanel:SetActive("btnCheer_R", true)
                pPanel:SetActive("btnFinish", false)
                pPanel:SetActive("txtFinishNum", false)
                pPanel:Button_SetEnable("btnCheer_R", false)
              end
            else
              pPanel:SetActive("btnCheer_R", true)
              pPanel:SetActive("btnFinish", false)
              pPanel:Button_SetEnable("btnCheer_R", bCanGuess)
              pPanel:SetActive("txtFinishNum", false)
              Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
                local nItemID = TongBattle.GuessCfg.nLuoGuItemID
                local szItemName = Item:GetName(nItemID, me)
                local nNum = me.GetItemCountByTypeInBag(nItemID)
                if nNum <= 0 then
                  me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
                  return
                end
                local tbTongData = TongBattle.tbTongData[nTongID]
                local tbData = {
                  {
                    nId = nTongID,
                    szName = tbTongData.szTongName
                  },
                  i18n.EasyGet("\230\136\152\229\138\155\239\188\154{0}", tbTongData.nFightPower),
                  i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\231\165\168\230\149\176\239\188\154"),
                  nil
                }
                local tbCallBack = {
                  TongBattle.RequestGuessChampion,
                  TongBattle,
                  nTongID
                }
                Ui:OpenWindow("UIActivity_Banghuizhengduosai_Tip", tbData, nItemID, nil, tbCallBack)
              end)
            end
          end)
        end
      end
    end
  end
  local nResult = 0
  if TongBattle.tbMyChampionGuessData then
    for nStage, tbData in pairs(TongBattle.tbMyChampionGuessData) do
      if not tbData.bSendAward then
        local nGuessStage = tbData.nStage
        local nRightMultiple = TongBattle.ChampionGuessMultiple[nGuessStage].nRightMultiple
        nResult = nRightMultiple * tbData.nGuessNum * TongBattle.GuessCfg.nLuoGuYuanBao
        break
      end
    end
  end
  self.pPanel:Label_SetText(tbControls.cont_txtInfo_R .. "/txtYuanBaoValue", nResult)
end

function tbWnd:UpdateMatchScroll()
  self.tbScrollViewInfo:Reset()
  local nNowStage = TongBattle:GetNowFinalsSatge(TongBattle.tbFinalsData)
  local nGuessEndTime = TongBattle:GetStageGuessEndTime(TongBattle.nWarZone, nNowStage)
  if not nGuessEndTime then
    return
  end
  local bCanGuess = true
  if nGuessEndTime <= GetTime() then
    bCanGuess = false
  end
  local nIndexStage = 0
  for i = nNowStage, 1, -1 do
    if TongBattle.FinalsStageCfg[i] then
      nIndexStage = nIndexStage + 1
      self.tbScrollViewInfo:SetElement(nIndexStage, function(tbElement)
        if not tbElement.tbScrollViewTong then
          tbElement.tbScrollViewTong = Ui:NewScrollView(tbElement, "Template")
        end
        if nNowStage == i and bCanGuess then
          tbElement.pPanel:SetActive("imgBG/txtTimeTittle", true)
          tbElement.pPanel:SetActive("imgBG/txtTime", true)
          tbElement.pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc5(nGuessEndTime - GetTime()))
          self:UpdateLeftTime(tbElement.pPanel)
        else
          tbElement.pPanel:SetActive("imgBG/txtTimeTittle", false)
          tbElement.pPanel:SetActive("imgBG/txtTime", false)
        end
        tbElement.pPanel:Label_SetText("imgBG/txtActivityTittle", i18n.Get(TongBattle.FinalsStageCfg[i].szStage))
        tbElement.tbScrollViewTong:Reset()
        local nIndex = 0
        for j = TongBattle.FinalsStageCfg[i].nStartMatch, TongBattle.FinalsStageCfg[i].nEndMatch do
          if TongBattle.tbFinalsData[j] then
            nIndex = nIndex + 1
            tbElement.tbScrollViewTong:SetElement(nIndex, function(tbElementMatch)
              local pPanel = tbElementMatch.pPanel
              local nTong1 = TongBattle.tbFinalsData[j].tbTongList[1]
              local nTong2 = TongBattle.tbFinalsData[j].tbTongList[2]
              local tbTong1 = TongBattle.tbTongData[nTong1]

                pPanel:Label_SetText("txtPlayerName01", tbTong1.szTongName)
                pPanel:Label_SetText("txtFightingNum02", tbTong1.nFightPower)
                pPanel:SetActive("txtFighting02", true)
              else
                pPanel:Label_SetText("txtPlayerName01", "")
                pPanel:Label_SetText("txtFightingNum02", "")
                pPanel:SetActive("txtFighting02", false)
              end
              local tbTong2 = TongBattle.tbTongData[nTong2]
              if tbTong2 then
                pPanel:Label_SetText("txtPlayerName02", tbTong2.szTongName)
                pPanel:Label_SetText("txtFightingNum01", tbTong2.nFightPower)
                pPanel:SetActive("txtFighting01", true)
              else
                pPanel:Label_SetText("txtPlayerName02", "")
                pPanel:Label_SetText("txtFightingNum01", "")
                pPanel:SetActive("txtFighting01", false)
              end
              pPanel:Label_SetText("btnCheer_L/Text", i18n.EasyGet("\231\171\158\231\140\156"))
              pPanel:Label_SetText("btnCheer_R/Text", i18n.EasyGet("\231\171\158\231\140\156"))
              pPanel:Label_SetText("Cont_Finish_L/txtPlayerName02 (3)", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
              pPanel:Label_SetText("Cont_Finish_R/txtPlayerName02 (3)", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
              if TongBattle.tbMyMatchGuessData and TongBattle.tbMyMatchGuessData[j] then
                if TongBattle.tbMyMatchGuessData[j].nTongID == nTong1 then
                  pPanel:SetActive("Cont_Finish_L", true)
                  pPanel:SetActive("btnCheer_L", false)
                  pPanel:SetActive("Cont_Finish_R", false)
                  pPanel:SetActive("btnCheer_R", false)
                  pPanel:Label_SetText("Cont_Finish_L/txtPlayerName02 (4)", TongBattle.tbMyMatchGuessData[j].nGuessNum)
                elseif TongBattle.tbMyMatchGuessData[j].nTongID == nTong2 then
                  pPanel:SetActive("Cont_Finish_L", false)
                  pPanel:SetActive("btnCheer_L", false)

                end
              else

                pPanel:SetActive("btnCheer_R", true)
                if -1 == TongBattle.tbFinalsData[j].nWinner then

                        me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))

function tbWnd:UpdateLeftTime(pPanel)
  local nNowStage = TongBattle:GetNowFinalsSatge(TongBattle.tbFinalsData)
  local nGuessEndTime = TongBattle:GetStageGuessEndTime(TongBattle.nWarZone, nNowStage)
  if not nGuessEndTime then
    return false
  end
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nTimeRemaining = nGuessEndTime - GetTime()
  if nTimeRemaining > 0 then
    pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc2(nGuessEndTime - GetTime()))
  else
    self:UpdateMatchScroll()
    return false
  end
  local nCurTime = tonumber(os.date("%Y%m%d%H%M", GetTime()))
  local nNextTime = 1
  if nTimeRemaining > 3600 then
    nNextTime = Lib:GetDate2Time(nCurTime + 1) - GetTime() + 1
  end
  if nTimeRemaining > 0 then
    self.nTimer = Timer:Register(nNextTime * Env.GAME_FPS, self.UpdateLeftTime, self, pPanel)
  end
  return false
end

function tbWnd:OnSyncItem(dwItemId, bNewItem, nAddCount)
  if dwItemId then
    local pItem = KItem.GetItemObj(dwItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId == TongBattle.GuessCfg.nLuoGuItemID or pItem.dwTemplateId == TongBattle.GuessCfg.nLaBaItemID then
      self:UpdatePanel()
    end
  end
end
