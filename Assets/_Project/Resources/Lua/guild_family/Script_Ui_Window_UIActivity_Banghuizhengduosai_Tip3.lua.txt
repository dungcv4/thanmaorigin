local tbWnd = Ui:CreateWindow("UIActivity_Banghuizhengduosai_Tip3")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_TONGBATTLE_GUESS_DATA,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnTipClose",
  node = "node/imgBG/node",
  ScrollViewCompetition = "node/imgBG/node/Scroll View",
  UIMoneyPanel = "node/imgBG/node/imgBG/UIMoneyPanel",
  TemplateInfo = "node/imgBG/node/Scroll View/Viewport/Content/Info",
  txtHorn01 = "node/imgBG/node/txtHorn01",
  txtHornNum01 = "node/imgBG/node/txtHorn01/txtNum01",
  txtHornNum02 = "node/imgBG/node/txtHorn01/txtNum02",
  txtHorn02 = "node/imgBG/node/txtHorn02",
  txtNum03 = "node/imgBG/node/txtHorn02/txtNum03",
  txtNum04 = "node/imgBG/node/txtHorn02/txtNum04",
  imgIcon03 = "node/imgBG/node/imgIcon03",
  imgIcon04 = "node/imgBG/node/imgIcon04",
  btnAdd01 = "node/imgBG/node/txtHorn02/btnAdd01",
  btnAdd02 = "node/imgBG/node/txtHorn02/btnAdd02",
  Cont_txtInfo_L = "node/imgBG/node/Cont_txtInfo_L",
  txtInfo_L06 = "node/imgBG/node/txtInfo_L06",
  cont_txtInfo_R = "node/imgBG/node/cont_txtInfo_R"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnAdd01 = function(self)
    self:AddItem(TongBattle.GuessCfg.nLaBaItemID)
  end,
  btnAdd02 = function(self)
    self:AddItem(TongBattle.GuessCfg.nLuoGuItemID)
  end
}

function tbWnd:OnCreate()
  self.tbScrollViewInfo = Ui:NewScrollView(self, tbControls.TemplateInfo)
end

function tbWnd:OnOpen()
  TongBattle:ApllySyncGuessData()
  self.pPanel:SetActive(tbControls.UIMoneyPanel, false)
end

function tbWnd:OnClose()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:UpdatePanel()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.pPanel:SetActive(tbControls.txtHorn01, true)
  self.pPanel:SetActive(tbControls.txtHorn02, true)
  self.pPanel:SetActive(tbControls.txtInfo_L06, false)
  self.pPanel:SetActive(tbControls.cont_txtInfo_R, false)
  self.pPanel:SetActive(tbControls.Cont_txtInfo_L, true)
  self.pPanel:ContentSizeFitter_Refresh(tbControls.Cont_txtInfo_L .. "/GameObject")
  self.pPanel:SetActive(tbControls.ScrollViewCompetition, true)
  self.pPanel:SetActive(tbControls.imgIcon03, false)
  self.pPanel:SetActive(tbControls.imgIcon04, false)
  local nGuessedLaBaNum = 0
  local nGuessedLuoGuNum = 0
  if TongBattle.tbMyMatchGuessData then
    for _, tbData in pairs(TongBattle.tbMyMatchGuessData) do
      nGuessedLaBaNum = nGuessedLaBaNum + (tbData[TongBattle.GuessCfg.nLaBaItemID] and tbData[TongBattle.GuessCfg.nLaBaItemID].nGuessNum or 0)
      nGuessedLuoGuNum = nGuessedLuoGuNum + (tbData[TongBattle.GuessCfg.nLuoGuItemID] and tbData[TongBattle.GuessCfg.nLuoGuItemID].nGuessNum or 0)
    end
  end
  self:UpdateMatchScroll()
  self.pPanel:Label_SetText(tbControls.txtHorn01, i18n.EasyGet("\230\156\172\232\189\174\229\183\178\230\182\136\232\128\151\239\188\154"))
  self.pPanel:Label_SetText(tbControls.txtHornNum01, nGuessedLaBaNum)
  self.pPanel:Label_SetText(tbControls.txtHornNum02, nGuessedLuoGuNum)
  local nOwnLaBaNum = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLaBaItemID)
  local nOwnLuoGuNum = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLuoGuItemID)
  self.pPanel:Label_SetText(tbControls.txtNum03, nOwnLaBaNum)
  self.pPanel:Label_SetText(tbControls.txtNum04, nOwnLuoGuNum)
end

function tbWnd:AddItem(nItemID)
  Shop:OpenShopBuyWidDefault(nItemID)
end

function tbWnd:UpdateMatchScroll()
  self.tbScrollViewInfo:Reset()
  local nNowStage = TongBattle:GetNowFinalsSatge(TongBattle.tbFinalsData)
  local nGuessEndStage = math.min(nNowStage, TongBattle.GuessCfg.nEndStage)
  local nGuessStartStage = math.max(nNowStage, TongBattle.GuessCfg.nStartStage)
  local nGuessEndTime = TongBattle:GetStageGuessEndTime(TongBattle.nWarZone, nGuessEndStage)
  if not nGuessEndTime then
    return
  end
  local bCanGuess = true
  if nGuessEndTime <= GetTime() then
    bCanGuess = false
  end
  local nIndexStage = 0
  for i = nGuessEndStage, nGuessStartStage, -1 do
    if TongBattle.FinalsStageCfg[i] then
      nIndexStage = nIndexStage + 1
      self.tbScrollViewInfo:SetElement(nIndexStage, function(tbElement)
        tbElement.nIdx = i
      end)
    end
  end
  local tbAllElement = self.tbScrollViewInfo:GetAllElement()
  for _, tbElement in ipairs(tbAllElement) do
    self:SetSubElement(tbElement, tbElement.nIdx, nNowStage, bCanGuess, nGuessEndTime)
  end
end

function tbWnd:SetSubElement(tbElement, nIdx, nNowStage, bCanGuess, nGuessEndTime)
  if not tbElement.tbScrollViewTong then
    tbElement.tbScrollViewTong = Ui:NewScrollView(tbElement, "Template")
  end
  if nNowStage == nIdx and bCanGuess then
    tbElement.pPanel:SetActive("imgBG/txtTimeTittle", true)
    tbElement.pPanel:SetActive("imgBG/txtTime", true)
    tbElement.pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc5(nGuessEndTime - GetTime()))
    self:UpdateLeftTime(tbElement.pPanel)
  else
    tbElement.pPanel:SetActive("imgBG/txtTimeTittle", false)
    tbElement.pPanel:SetActive("imgBG/txtTime", false)
  end
  tbElement.pPanel:Label_SetText("imgBG/txtActivityTittle", i18n.Get(TongBattle.FinalsStageCfg[nIdx].szStage))
  tbElement.tbScrollViewTong:Reset()
  local nIndex = 0
  for j = TongBattle.FinalsStageCfg[nIdx].nStartMatch, TongBattle.FinalsStageCfg[nIdx].nEndMatch do
    if TongBattle.tbFinalsData[j] then
      nIndex = nIndex + 1
      tbElement.tbScrollViewTong:SetElement(nIndex, function(tbElementMatch)
        local pPanel = tbElementMatch.pPanel
        local nTong1 = TongBattle.tbFinalsData[j].tbTongList[1]
        local nTong2 = TongBattle.tbFinalsData[j].tbTongList[2]
        local tbTong1 = TongBattle.tbTongData[nTong1]
        if tbTong1 then
          pPanel:Label_SetText("txtPlayerName01", tbTong1.szTongName)
          pPanel:Label_SetText("txtFightingNum02", tbTong1.nFightPower)
          pPanel:Label_SetText("txtServerName01", GetServerName(tbTong1.nServerId))
          pPanel:SetActive("txtFighting02", true)
        else
          pPanel:Label_SetText("txtPlayerName01", "")
          pPanel:Label_SetText("txtFightingNum02", "")
          pPanel:Label_SetText("txtServerName01", "")
          pPanel:SetActive("txtFighting02", false)
        end
        local tbTong2 = TongBattle.tbTongData[nTong2]
        if tbTong2 then
          pPanel:Label_SetText("txtPlayerName02", tbTong2.szTongName)
          pPanel:Label_SetText("txtFightingNum01", tbTong2.nFightPower)
          pPanel:Label_SetText("txtServerName02", GetServerName(tbTong2.nServerId))
          pPanel:SetActive("txtFighting01", true)
        else
          pPanel:Label_SetText("txtPlayerName02", "")
          pPanel:Label_SetText("txtFightingNum01", "")
          pPanel:Label_SetText("txtServerName02", "")
          pPanel:SetActive("txtFighting01", false)
        end
        pPanel:Label_SetText("btnCheer_L/Text", i18n.EasyGet("\231\171\158\231\140\156"))
        pPanel:Label_SetText("btnCheer_R/Text", i18n.EasyGet("\231\171\158\231\140\156"))
        pPanel:Label_SetText("Cont_Finish_L/txtTittle", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
        pPanel:Label_SetText("Cont_Finish_R/txtTittle", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
        if TongBattle.tbMyMatchGuessData and TongBattle.tbMyMatchGuessData[j] then
          local _, tbGuessData = next(TongBattle.tbMyMatchGuessData[j])
          local nTongID = tbGuessData and tbGuessData.nTongID or 0
          if nTongID == nTong1 then
            pPanel:SetActive("Cont_Finish_L", true)
            pPanel:SetActive("btnCheer_L", false)
            pPanel:SetActive("Cont_Finish_R", false)
            pPanel:SetActive("btnCheer_R", false)
            pPanel:Label_SetText("Cont_Finish_L/txtNum01", TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLuoGuItemID] and TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLuoGuItemID].nGuessNum or 0)
            pPanel:Label_SetText("Cont_Finish_L/txttxtNum02", TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLaBaItemID] and TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLaBaItemID].nGuessNum or 0)
          elseif nTongID == nTong2 then
            pPanel:SetActive("Cont_Finish_L", false)
            pPanel:SetActive("btnCheer_L", false)
            pPanel:SetActive("Cont_Finish_R", true)
            pPanel:SetActive("btnCheer_R", false)
            pPanel:Label_SetText("Cont_Finish_R/txtNum01", TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLuoGuItemID] and TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLuoGuItemID].nGuessNum or 0)
            pPanel:Label_SetText("Cont_Finish_R/txtNum02", TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLaBaItemID] and TongBattle.tbMyMatchGuessData[j][TongBattle.GuessCfg.nLaBaItemID].nGuessNum or 0)
          end
        else
          pPanel:SetActive("Cont_Finish_L", false)
          pPanel:SetActive("btnCheer_L", true)
          pPanel:SetActive("Cont_Finish_R", false)
          pPanel:SetActive("btnCheer_R", true)
          if -1 == TongBattle.tbFinalsData[j].nWinner then
            if tbTong1 then
              pPanel:Button_SetEnable("btnCheer_L", bCanGuess)
              Ui:Button_BindEvent(pPanel, "btnCheer_L", function()
                local nLaBaNum = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLaBaItemID)
                local nLuoGuNum = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLuoGuItemID)
                if nLaBaNum <= 0 and nLuoGuNum <= 0 then
                  me.CenterMsg(i18n.EasyGet("\229\138\169\229\168\129\233\129\147\229\133\183\230\149\176\233\135\143\228\184\141\232\182\179"))
                  return
                end
                Ui:OpenWindow("UIPKCompetitionGuess_Tip1", nIdx, j, nTong1)
              end)
            else
              pPanel:Button_SetEnable("btnCheer_L", false)
            end
            if tbTong2 then
              pPanel:Button_SetEnable("btnCheer_R", bCanGuess)
              Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
                local nLaBaNum = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLaBaItemID)
                local nLuoGuNum = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLuoGuItemID)
                if nLaBaNum <= 0 and nLuoGuNum <= 0 then
                  me.CenterMsg(i18n.EasyGet("\229\138\169\229\168\129\233\129\147\229\133\183\230\149\176\233\135\143\228\184\141\232\182\179"))
                  return
                end
                Ui:OpenWindow("UIPKCompetitionGuess_Tip1", nIdx, j, nTong2)
              end)
            else
              pPanel:Button_SetEnable("btnCheer_R", false)
            end
          else
            pPanel:Button_SetEnable("btnCheer_L", false)
            pPanel:Button_SetEnable("btnCheer_R", false)
          end
        end
      end)
    end
  end
end

function tbWnd:UpdateLeftTime(pPanel)
  local nNowStage = TongBattle:GetNowFinalsSatge(TongBattle.tbFinalsData)
  local nGuessEndTime = TongBattle:GetStageGuessEndTime(TongBattle.nWarZone, nNowStage)
  if not nGuessEndTime then
    return false
  end
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nTimeRemaining = nGuessEndTime - GetTime()
  if nTimeRemaining > 0 then
    pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc2(nGuessEndTime - GetTime()))
  else
    self:UpdateMatchScroll()
    return false
  end
  local nCurTime = tonumber(os.date("%Y%m%d%H%M", GetTime()))
  local nNextTime = 1
  if nTimeRemaining > 3600 then
    nNextTime = Lib:GetDate2Time(nCurTime + 1) - GetTime() + 1
  end
  if nTimeRemaining > 0 then
    self.nTimer = Timer:Register(nNextTime * Env.GAME_FPS, self.UpdateLeftTime, self, pPanel)
  end
  return false
end

function tbWnd:OnSyncItem(dwItemId, bNewItem, nAddCount)
  if dwItemId then
    local pItem = KItem.GetItemObj(dwItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId == TongBattle.GuessCfg.nLuoGuItemID or pItem.dwTemplateId == TongBattle.GuessCfg.nLaBaItemID then
      self:UpdatePanel()
    end
  end
end
