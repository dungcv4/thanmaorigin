local tbWnd = Ui:CreateClass("UIBookWedding_Clothes", "UIPanel")
local tbControls = {
  Template = "Scroll View/Viewport/Content/Template",
  toggleGender = "node/imgBG/Clothes/Scroll View/Viewport/Content/Template/toggleGender"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  Fashion.bHorse = false
  self.nLastSelect = Marry.nSelectClothingLevel
  self.ScrollView:Clear()
  for nIndex, tbInfo in ipairs(Marry.Cfg.WeddingClothing) do
    self.ScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtNameNormal", Item:GetName(tbInfo.nId, me))
      pPanel:SetActive("Cost", false)
      pPanel:SetActive("CostNow", false)
      local tbMoney = Lib:SplitStrTry2Num(tbInfo.szPrice, "|")
      local szIcon = Currency:GetIcon(tbMoney[1])
      if not Marry.tbHaveClothes[nIndex].bMeHaveClothing or not Marry.tbHaveClothes[nIndex].bLoverHaveClothing then
        pPanel:SetActive("CostNow", true)
        pPanel:Sprite_SetSprite("CostNow/imgMoney", szIcon)
        pPanel:Label_SetText("CostNow/txtCost", tbMoney[2])
      else
        pPanel:SetActive("Cost", true)
        pPanel:SetActive("Toggle", true)
        pPanel:SetActive("toggleGender", true)
        pPanel:Sprite_SetSprite("Cost/imgMoney", szIcon)
        pPanel:Sprite_SetSprite("Cost/CostDiscounts/imgMoney", szIcon)
        pPanel:Label_SetText("Cost/CostDiscounts/txtCost", tbMoney[2])
        if Marry.tbHaveClothes[nIndex].bMeHaveClothing and Marry.tbHaveClothes[nIndex].bLoverHaveClothing then
          pPanel:Label_SetText("Cost/txtCost", 0)
        else
          pPanel:Label_SetText("Cost/txtCost", tbMoney[2] / 2)
        end
      end
      local nMaxClothes = Marry.Cfg.Limit[Marry.nSelectMapLevel].nMaxClothes
      if nMaxClothes < nIndex then
        pPanel:SetActive("Toggle", false)
      end
      Ui:Toggle_BindEvent(pPanel, "Toggle", function(this, isOn)
        self:OnSelect(nIndex, isOn)
      end, self)
      pPanel:Toggle_SetChecked("Toggle", nIndex == Marry.nSelectClothingLevel)
      Ui:Toggle_BindEvent(pPanel, "toggleGender", function(this, isOn)
        self:OnChangeClothingSex(tbElement, nIndex, isOn)
      end, self)
      pPanel:Toggle_SetChecked("toggleGender", 1 == me.nSex)
      tbElement.objResPreview = tbElement.pPanel:GetObject("NpcModel")
      Fashion:Preview_Init(tbElement.objResPreview)
      Fashion.nTemporarySex = me.nSex
      tbElement.objResPreview = tbElement.pPanel:GetObject("NpcModel")
      Fashion:Preview_Init(tbElement.objResPreview)
      local tbSetting1 = Fashion:GetFashionSetting(me, Marry.Cfg.WeddingClothing[nIndex].nFashionId1)
      local tbSetting2 = Fashion:GetFashionSetting(me, Marry.Cfg.WeddingClothing[nIndex].nFashionId2)
      if tbSetting1 and tbSetting2 then
        Fashion:Preview_TryOnAFashion(tbSetting1.nPart, Marry.Cfg.WeddingClothing[nIndex].nFashionId1)
        Fashion:Preview_TryOnAFashion(tbSetting2.nPart, Marry.Cfg.WeddingClothing[nIndex].nFashionId2)
        Fashion:Preview_RefreshPreview()
      end
    end)
  end
  Marry:RequestSynSelect2Lover(3)
end

function tbWnd:OnSelect(nIndex, bSelect)
  if not bSelect then
    if nIndex == self.nLastSelect then
      Marry.nSelectClothingLevel = 0
    end
    return
  end
  if 0 ~= self.nLastSelect and self.nLastSelect ~= nIndex then
    local pPanel = self.ScrollView:GetElement(self.nLastSelect).pPanel
    pPanel:Toggle_SetChecked("Toggle", false)
  end
  Marry.nSelectClothingLevel = nIndex
  self.nLastSelect = nIndex
end

function tbWnd:OnChangeClothingSex(tbElement, nIndex, isOn)
  if isOn then
    Fashion.nTemporarySex = 1
  else
    Fashion.nTemporarySex = 0
  end
  tbElement.objResPreview = tbElement.pPanel:GetObject("NpcModel")
  Fashion:Preview_Init(tbElement.objResPreview)
  local tbSetting1 = Fashion:GetFashionSetting(me, Marry.Cfg.WeddingClothing[nIndex].nFashionId1)
  local tbSetting2 = Fashion:GetFashionSetting(me, Marry.Cfg.WeddingClothing[nIndex].nFashionId2)
  if tbSetting1 and tbSetting2 then
    Fashion:Preview_TryOnAFashion(tbSetting1.nPart, Marry.Cfg.WeddingClothing[nIndex].nFashionId1)
    Fashion:Preview_TryOnAFashion(tbSetting2.nPart, Marry.Cfg.WeddingClothing[nIndex].nFashionId2)
    Fashion:Preview_RefreshPreview()
  end
end

function tbWnd:OnClose()
  Fashion:Preview_Destroy()
end
