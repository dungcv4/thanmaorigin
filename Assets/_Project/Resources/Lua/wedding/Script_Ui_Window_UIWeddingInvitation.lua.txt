local tbWnd = Ui:CreateWindow("UIWeddingInvitation")
local tbControls = {
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  txtTitle = "node/imgBG/txtTitle",
  btnClose = "node/imgBG/btnClose",
  btnSend = "node/imgBG/Content/btnSend",
  ToggleAll = "node/imgBG/Content/grid/ToggleColthes",
  ToggleOnline = "node/imgBG/Content/grid/ToggleNow",
  btnFriend = "node/imgBG/btnList/btnFriend",
  btnTong = "node/imgBG/btnList/btnInfo",
  panelFriend = "node/imgBG/Content/panelFriend",
  panelTong = "node/imgBG/Content/TongMemberPanel",
  txtTotalInvite = "node/imgBG/Content/txtFriendTips",
  txtCanInvite = "node/imgBG/Content/txtFriendInvitable",
  txtAlreadyInvite = "node/imgBG/Content/txtFriendNum"
}
tbWnd.tbControls = tbControls

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNoTIFY_SYNC_FRIEND_DATA,
      self.RefreshFriend
    },
    {
      EventNotify.emNOTIFY_SYNC_TONG_MEMBERLIST,
      self.RefreshTong
    },
    {
      EventNotify.emNOTIFY_SYNC_KIN_DATA,
      self.RefreshTong
    }
  }
end

tbWnd.PANEL_TYPE_FRIEND = 1
tbWnd.PANEL_TYPE_TONG = 2
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_TYPE_FRIEND] = {
    CLASS = "UIWeddingInvite_Friend",
    OBJ = tbControls.panelFriend,
    BTN = tbControls.btnFriend,
    OpenFunc = function()
      return true
    end
  },
  [tbWnd.PANEL_TYPE_TONG] = {
    CLASS = "UIWeddingInvite_Tong",
    OBJ = tbControls.panelTong,
    BTN = tbControls.btnTong,
    OpenFunc = function()
      return 0 ~= me.dwKinId or 0 ~= me.dwTongId
    end
  }
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("Wedding2")
  end,
  btnSend = function(self)
    Marry:TrySendInvitation()
  end
}
tbWnd.tbOnToggleClick = {
  ToggleOnline = function(self, isOn)
    local nType = self.tbTabControl.nPanelType or self.tbTabControl:GetOpenPanel()
    self.tbTabControl:UpdateSubPanel(nType, "SelectOnline", isOn)
  end
}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
end

function tbWnd:OnOpen()
  self.tbTabControl:OnOpen(self.PANEL_TYPE_FRIEND)
  self:InitInfo(Marry.nInviteTimes, Marry.nInviteTimes, 0)
end

function tbWnd:InitInfo(nTotalInvite, nCanInvite, nAlreadyInvite)
  self.pPanel:Label_SetText(tbControls.txtTotalInvite, i18n.EasyGet("\230\173\164\233\130\128\232\175\183\229\135\189\229\143\175\233\130\128\232\175\183{0}\228\186\186", nTotalInvite))
  self.pPanel:Label_SetText(tbControls.txtCanInvite, i18n.EasyGet("\229\143\175\233\130\128\232\175\183: {0}\228\186\186", nCanInvite))
  self.pPanel:Label_SetText(tbControls.txtAlreadyInvite, i18n.EasyGet("\229\183\178\233\128\137\230\139\169: {0}\228\186\186", nAlreadyInvite))
  if 0 == me.dwTongId then
    self.pPanel:Button_SetText(tbControls.btnTong, i18n.EasyGet("\229\174\182\230\151\143"), "Text")
  else
    self.pPanel:Button_SetText(tbControls.btnTong, i18n.EasyGet("\229\184\174\228\188\154"), "Text")
  end
end

function tbWnd:Refresh()
  self:InitInfo(Marry.nInviteTimes, Marry.nInviteTimes, 0)
  local nType = self.tbTabControl.nPanelType or self.tbTabControl:GetOpenPanel()
  self.tbTabControl:UpdateSubPanel(nType, "Refresh")
end

function tbWnd:RefreshTong()
  local nType = self.tbTabControl.nPanelType or self.tbTabControl:GetOpenPanel()
  if nType == self.PANEL_TYPE_TONG then
    self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_TONG, "Refresh")
  end
  self:InitInfo(Marry.nInviteTimes, Marry.nInviteTimes, 0)
end

function tbWnd:RefreshFriend()
  local nType = self.tbTabControl.nPanelType or self.tbTabControl:GetOpenPanel()
  if nType == self.PANEL_TYPE_FRIEND then
    self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_FRIEND, "Refresh")
  end
  self:InitInfo(Marry.nInviteTimes, Marry.nInviteTimes, 0)
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
end
