local tbWnd = Ui:CreateClass("UIBookWedding_Ring", "UIPanel")
local tbControls = {
  Template = "Scroll View/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  self.nLastSelect = Marry.nSelectRingLevel
  self.ScrollView:Reset()
  for nIndex, tbInfo in ipairs(Marry.Cfg.Ring) do
    self.ScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtTittle", Item:GetName(tbInfo.nId, me))
      local tbMoney = Lib:SplitStrTry2Num(tbInfo.szPrice, "|")
      local szIcon = Currency:GetIcon(tbMoney[1])
      pPanel:Sprite_SetSprite("ImgGold", szIcon)
      pPanel:Label_SetText("ImgGold/txtNum", tbMoney[2])
      local nMaxRing = Marry.Cfg.Limit[Marry.nSelectMapLevel].nMaxRing
      if nMaxRing < nIndex then
        pPanel:SetActive("Toggle", false)
      else
        pPanel:SetActive("Toggle", true)
      end
      Ui:Toggle_BindEvent(pPanel, "Toggle", function(this, isOn)
        self:OnSelect(nIndex, isOn)
      end, self)
      pPanel:Toggle_SetChecked("Toggle", nIndex == Marry.nSelectRingLevel)
      Ui:Button_BindEvent(pPanel, "btnMore", function()
        self:ShowTips(nIndex)
      end, self)
      local ItemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      ItemGrid:SetItemByTemplate(tbInfo.nId)
    end)
  end
  Marry:RequestSynSelect2Lover(4)
end

function tbWnd:ShowTips(nIndex)
  local tbInfo = Marry.Cfg.Ring[nIndex]
  local tbElement = self.ScrollView:GetElement(2)
  local tbPosition = tbElement.pPanel:GetSelfPosition()
  local szDesc = i18n.Get(tbInfo.szDesc)
  self.tbOwner:OnShowTips(Item:GetName(tbInfo.nId, me), szDesc, tbPosition)
end

function tbWnd:OnSelect(nIndex, bSelect)
  if not bSelect then
    if nIndex == self.nLastSelect then
      Marry.nSelectRingLevel = 0
    end
    return
  end
  if 0 ~= self.nLastSelect and self.nLastSelect ~= nIndex then
    local pPanel = self.ScrollView:GetElement(self.nLastSelect).pPanel
    pPanel:Toggle_SetChecked("Toggle", false)
  end
  Marry.nSelectRingLevel = nIndex
  self.nLastSelect = nIndex
end
