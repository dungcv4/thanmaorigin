local tbWnd = Ui:CreateClass("UIBookWedding_Balance", "UIPanel")
local tbControls = {
  txtTime = "txtTittle2",
  ImgGold = "ImgGold",
  txtGoldNum = "ImgGold/txtNum",
  ImgYuanBao = "ImgGold/ImgMoney",
  txtYuanBaoNum = "ImgGold/ImgMoney/txtNum",
  txtMapName = "Scroll View/Viewport/Content/Template1/txtTittle",
  MapImage = "Scroll View/Viewport/Content/Template1/Image",
  ImgMapMoney = "Scroll View/Viewport/Content/Template1/ImgGold",
  txtMapMoneyNum = "Scroll View/Viewport/Content/Template1/ImgGold/txtNum",
  txtClothingName = "Scroll View/Viewport/Content/Template2/txtTittle",
  ClothingGrid = "Scroll View/Viewport/Content/Template2/UIItemGrid",
  ImgClothingMoney = "Scroll View/Viewport/Content/Template2/ImgGold",
  txtClothingMoneyNum = "Scroll View/Viewport/Content/Template2/ImgGold/txtNum",
  txtRingName = "Scroll View/Viewport/Content/Template3/txtTittle",
  RingGrid = "Scroll View/Viewport/Content/Template3/UIItemGrid",
  ImgRingMoney = "Scroll View/Viewport/Content/Template3/ImgGold",
  txtRingMoneyNum = "Scroll View/Viewport/Content/Template3/ImgGold/txtNum",
  btnOk = "btnOk"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnOk = function(self)
    if Marry.bReBook then
      Marry:TryReBook()
    else
      Marry:TryBookWedding()
    end
  end
}
local tbCurrencyType2Path = {
  [Currency.TYPE.Gold] = {
    Img = tbControls.ImgGold,
    Num = tbControls.txtGoldNum
  },
  [Currency.TYPE.YuanBao] = {
    Img = tbControls.ImgYuanBao,
    Num = tbControls.txtYuanBaoNum
  }
}

function tbWnd:OnOpen()
  local tbAllPrice = {}
  local tbOnePrice = {}
  if Marry.nSelectMapLevel > 0 then
    local tbInfo = Marry.Cfg.WeddingMapCfg[Marry.nSelectMapLevel]
    tbOnePrice = Lib:SplitStrTry2Num(tbInfo.szPrice, "|")
    tbAllPrice[tbOnePrice[1]] = (tbAllPrice[tbOnePrice[1]] or 0) + tbOnePrice[2]
    self.pPanel:Label_SetText(tbControls.txtMapName, i18n.Get(tbInfo.szName))
    local tbMiniMapSize, szMiniMap, tbMiniMapRawInfo = Map:GetMiniMapInfo(tbInfo.nMapId)
    self.pPanel:Sprite_SetSprite(tbControls.MapImage, szMiniMap)
    self.pPanel:Sprite_SetSprite(tbControls.ImgMapMoney, Currency:GetIcon(tbOnePrice[1]))
    self.pPanel:Label_SetText(tbControls.txtMapMoneyNum, tbOnePrice[2])
  end
  if Marry.nSelectYear > 0 and Marry.nSelectMonth > 0 and Marry.nSelectDay > 0 and Marry.nSelectTimePoint > 0 then
    local szWeddingTime = Marry.Cfg.TimePoint[Marry.nSelectTimePoint].szWeedingTime
    local tbHour = Lib:SplitStrTry2Num(szWeddingTime, ":")
    local tbParam = {
      year = Marry.nSelectYear,
      month = Marry.nSelectMonth,
      day = Marry.nSelectDay,
      hour = tbHour[1],
      min = tbHour[2],
      sec = tbHour[3]
    }
    local nTime = os.time(tbParam) - GetZoneTimeSecDiff()
    self.pPanel:Label_SetText(tbControls.txtTime, Lib:TimeDesc11(nTime))
  end
  if Marry.nSelectRingLevel > 0 then
    local tbInfo = Marry.Cfg.Ring[Marry.nSelectRingLevel]
    tbOnePrice = Lib:SplitStrTry2Num(tbInfo.szPrice, "|")
    tbAllPrice[tbOnePrice[1]] = (tbAllPrice[tbOnePrice[1]] or 0) + tbOnePrice[2]
    self.pPanel:Label_SetText(tbControls.txtRingName, Item:GetName(tbInfo.nId, me))
    local ItemGrid = Ui:NewExistItemGrid(self, tbControls.RingGrid)
    ItemGrid:SetItemByTemplate(tbInfo.nId)
    self.pPanel:Sprite_SetSprite(tbControls.ImgRingMoney, Currency:GetIcon(tbOnePrice[1]))
    self.pPanel:Label_SetText(tbControls.txtRingMoneyNum, tbOnePrice[2])
  end
  if Marry.nSelectClothingLevel > 0 then
    local tbInfo = Marry.Cfg.WeddingClothing[Marry.nSelectClothingLevel]
    tbOnePrice = Lib:SplitStrTry2Num(tbInfo.szPrice, "|")
    local nIndex = Marry.nSelectClothingLevel
    tbOnePrice = Lib:SplitStrTry2Num(tbInfo.szPrice, "|")
    if Marry.tbHaveClothes[nIndex].bMeHaveClothing and Marry.tbHaveClothes[nIndex].bLoverHaveClothing then
      tbOnePrice[2] = 0
    elseif Marry.tbHaveClothes[nIndex].bMeHaveClothing or Marry.tbHaveClothes[nIndex].bLoverHaveClothing then
      tbOnePrice[2] = tbOnePrice[2] / 2
    end
    tbAllPrice[tbOnePrice[1]] = (tbAllPrice[tbOnePrice[1]] or 0) + math.floor(tbOnePrice[2])
    self.pPanel:Label_SetText(tbControls.txtClothingName, Item:GetName(tbInfo.nId, me))
    local ItemGrid = Ui:NewExistItemGrid(self, tbControls.ClothingGrid)
    ItemGrid:SetItemByTemplate(tbInfo.nId)
    self.pPanel:Sprite_SetSprite(tbControls.ImgClothingMoney, Currency:GetIcon(tbOnePrice[1]))
    self.pPanel:Label_SetText(tbControls.txtClothingMoneyNum, tbOnePrice[2])
  end
  for szType, tbPath in pairs(tbCurrencyType2Path) do
    self.pPanel:Sprite_SetSprite(tbPath.Img, Currency:GetIcon(szType))
    self.pPanel:Label_SetText(tbPath.Num, tostring(tbAllPrice[szType] or 0))
  end
  Marry:RequestSynSelect2Lover(5)
end
