function Marry:LoadConfig()
  local szSettingFile = "Setting/Marry/MarryBuff.tab"
  self.tbImitityLevel = Lib:LoadTabFileEx(szSettingFile, "dddd", "nLevel", {
    "nLevel",
    "nSkillLevel",
    "nMin",
    "nImitity"
  })
end

Marry:LoadConfig()

function Marry:CalImitityLevel(Imitity)
  local nImitityLevel = 1
  local nSkillLevel = 1
  for nLevel, tbInfo in ipairs(self.tbImitityLevel) do
    if Imitity < tbInfo.nImitity then
      nImitityLevel = nLevel
      nSkillLevel = tbInfo.nSkillLevel
      break
    end
  end
  return nImitityLevel, nSkillLevel
end

function Marry:GetWeddingTime(nDay, nTimePoint)
  local tbWeddingTime = Lib:SplitStrTry2Num(Marry.Cfg.TimePoint[nTimePoint].szWeedingTime, ":")
  local nOriginTime = nDay * 3600 * 24
  local tbTime = os.date("*t", nOriginTime)
  local tbParam = {
    year = tbTime.year,
    month = tbTime.month,
    day = tbTime.day,
    hour = tbWeddingTime[1],
    min = tbWeddingTime[2],
    sec = tbWeddingTime[3]
  }
  return os.time(tbParam)
end
