local function LoadPos(szFilePath)
  local tbFileData = Lib:LoadTabFileEx(szFilePath, "dd", nil, {"PosX", "PosY"})
  if not tbFileData then
    LogErr("[Wedding] fail to load file:", szPath)
    return nil
  end
  return tbFileData
end

local function LoadWeddingDefaultPos()
  local tbFilePath = {
    [334] = "Setting/Map/jiehun_1_haitan/info/default_pos_info.txt",
    [335] = "Setting/Map/jiehun_2_pingmin/info/default_pos_info.txt",
    [336] = "Setting/Map/jiehun_3_guizu/info/default_pos_info.txt",
    [337] = "Setting/Map/jiehun_4_huanggong/info/default_pos_info.txt"
  }
  local tbMapDefaultPos = {}
  for key, szPath in pairs(tbFilePath) do
    tbMapDefaultPos[key] = LoadPos(szPath)
  end
  return tbMapDefaultPos
end

Wedding.tbMapDefaultPos = LoadWeddingDefaultPos()

function Wedding:GetRandomDefaultPos(nMapTID)
  if not nMapTID or not self.tbMapDefaultPos[nMapTID] then
    return
  end
  local tbDefaultPos = self.tbMapDefaultPos[nMapTID]
  local nRandom = math.random(1, #tbDefaultPos)
  return tbDefaultPos[nRandom]
end

    [337] = "Setting/Map/jiehun_4_huanggong/info/JiuZhuo_pos_info.txt"
  }
  local tbNpcInfo = {}
  for key, szPath in pairs(tbNpcPath) do
    local tbFileData = Lib:LoadTabFileEx(szPath, "ddd", nil, {
      "NpcTemplateId",
      "PosX",
      "PosY"
    })
    if not tbFileData then
      LogErr("[Wedding] fail to load file:", szPath)
    else
      tbNpcInfo[key] = tbFileData
    end
  end
  return tbNpcInfo
end
