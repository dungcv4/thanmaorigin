local tbWnd = Ui:CreateClass("UIBookWedding_Time", "UIPanel")
local tbControls = {
  YearTemplate = "ScrollViewYear/Viewport/Content/Template",
  MonthTemplate = "ScrollViewMonth/Viewport/Content/Template",
  DayTemplate = "ScrollViewDay/Viewport/Content/Template",
  TimePointTemplate = "ScrollViewTime/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbYearSpuerList = Ui:NewSuperListView(self, tbControls.YearTemplate, self.SetData, nil, nil, nil, self.OnYearChange)
  self.tbMonthSpuerList = Ui:NewSuperListView(self, tbControls.MonthTemplate, self.SetData, nil, nil, nil, self.OnMonthChange)
  self.tbDaySpuerList = Ui:NewSuperListView(self, tbControls.DayTemplate, self.SetData, nil, nil, nil, self.OnDayChange)
  self.tbTimePointSpuerList = Ui:NewSuperListView(self, tbControls.TimePointTemplate, self.SetTimePoint, nil, nil, nil, self.OnTimePointChange)
end

function tbWnd:OnOpen()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_UI_BOOWEDDINGTIME_UPDATE, self.InitTime, self)
  if Marry.tbBookTime[Marry.nSelectMapLevel] and 0 ~= Lib:CountTB(Marry.tbBookTime[Marry.nSelectMapLevel]) then
    self:InitTime()
  else
    Marry.nSelectYear = 0
    Marry.nSelectMonth = 0
    Marry.nSelectDay = 0
    Marry.nSelectTimePoint = 0
    me.CenterMsg(i18n.EasyGet("\232\175\165\231\173\137\231\186\167\229\169\154\231\164\188\229\183\178\232\162\171\229\133\168\233\131\168\233\162\132\232\174\162\229\174\140"))
  end
  Marry:RequestSynSelect2Lover(2)
end

function tbWnd:InitTime()
  self:OnSuperDataChange(self.tbYearSpuerList, Marry.tbBookTime[Marry.nSelectMapLevel], Marry.nSelectYear, self.OnYearChange)
end

function tbWnd:SetData(tbElement, nData)
  local pPanel = tbElement.pPanel
  pPanel:Label_SetText("txtTime", tostring(nData))
end

function tbWnd:SetTimePoint(tbElement, nTimePoint)
  local pPanel = tbElement.pPanel
  local szTimePoint = Marry.Cfg.TimePoint[nTimePoint].szWeedingTime
  pPanel:Label_SetText("txtTime", szTimePoint)
end

function tbWnd:OnYearChange(nOldIndex, nNewIndex)
  local nNewYear = self.tbYearSpuerList:GetElementData()[nNewIndex]
  self.nCurYear = nNewYear
  Marry.nSelectYear = nNewYear
  self:OnSuperDataChange(self.tbMonthSpuerList, Marry.tbBookTime[Marry.nSelectMapLevel][self.nCurYear], Marry.nSelectMonth, self.OnMonthChange)
end

function tbWnd:OnMonthChange(nOldIndex, nNewIndex)
  local nNewMonth = self.tbMonthSpuerList:GetElementData()[nNewIndex]
  self.nCurMonth = nNewMonth
  Marry.nSelectMonth = nNewMonth
  self:OnSuperDataChange(self.tbDaySpuerList, Marry.tbBookTime[Marry.nSelectMapLevel][self.nCurYear][self.nCurMonth], Marry.nSelectDay, self.OnDayChange)
end

function tbWnd:OnDayChange(nOldIndex, nNewIndex)
  local nNewDay = self.tbDaySpuerList:GetElementData()[nNewIndex]
  self.nCurDay = nNewDay
  Marry.nSelectDay = nNewDay
  self:OnSuperDataChange(self.tbTimePointSpuerList, Marry.tbBookTime[Marry.nSelectMapLevel][self.nCurYear][self.nCurMonth][self.nCurDay], Marry.nSelectTimePoint, self.OnTimePointChange)
end

function tbWnd:OnSuperDataChange(SuperList, tableData, nSnapData, FnNext)
  local tbData = {}
  for nData in pairs(tableData) do
    table.insert(tbData, nData)
  end
  table.sort(tbData)
  local nIndex = 0
  SuperList:SetAllElement(tbData)
  for i, nData in pairs(tbData) do
    if nData == nSnapData then
      nIndex = i
      break
    end
  end
  if nIndex <= 0 then
    if 1 == SuperList:GetCurSnapTargetIndex() then
      FnNext(self, 1, 1)
    else
      SuperList:SetSnapTargetItemIndex(1)
    end
  elseif SuperList:GetCurSnapTargetIndex() == nIndex then
    FnNext(self, nIndex, nIndex)
  else
    SuperList:SetSnapTargetItemIndex(nIndex)
  end
end

function tbWnd:OnTimePointChange(nOldIndex, nNewIndex)
  local nSelectTimePoint = self.tbTimePointSpuerList:GetElementData()[nNewIndex]
  Marry.nSelectYear = self.nCurYear
  Marry.nSelectMonth = self.nCurMonth
  Marry.nSelectDay = self.nCurDay
  Marry.nSelectTimePoint = nSelectTimePoint
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_UI_BOOWEDDINGTIME_UPDATE, self)
end
