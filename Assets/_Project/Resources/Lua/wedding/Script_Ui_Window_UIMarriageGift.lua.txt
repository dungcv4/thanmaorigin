local tbWnd = Ui:CreateWindow("UIMarriageGift")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_CHANGE_MONEY,
      self.OnChangeMoney,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  Content = "node/imgBG/Content/Scroll View/Viewport/Content",
  Template1 = "node/imgBG/Content/Scroll View/Viewport/Content/Template1",
  Template2 = "node/imgBG/Content/Scroll View/Viewport/Content/Template2",
  Template3 = "node/imgBG/Content/Scroll View/Viewport/Content/Template3",
  Toggle1 = "node/imgBG/Content/Scroll View/Viewport/Content/Template1/Toggle",
  Toggle2 = "node/imgBG/Content/Scroll View/Viewport/Content/Template2/Toggle",
  Toggle3 = "node/imgBG/Content/Scroll View/Viewport/Content/Template3/Toggle",
  btnMore1 = "node/imgBG/Content/Scroll View/Viewport/Content/Template1/btnMore",
  btnMore2 = "node/imgBG/Content/Scroll View/Viewport/Content/Template2/btnMore",
  btnMore3 = "node/imgBG/Content/Scroll View/Viewport/Content/Template3/btnMore",
  btnOk = "node/imgBG/Content/btnOk",
  UIMoneyPanel = "node/imgBG/UIMoneyPanel",
  wndTips = "wndSubAttribute",
  txtHelpTitle = "wndSubAttribute/imgBg/txtSubAttribute",
  txtHelpDesc = "wndSubAttribute/imgBg/ScrollView/Viewport/Content/Template/txtAttributeName",
  Empty4Raycast = "wndSubAttribute/Empty4Raycast"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOk = function(self)
    self:SendCash()
  end,
  Empty4Raycast = function(self)
    self.pPanel:SetActive(tbControls.wndTips, false)
  end,
  btnMore1 = function(self)
    self:OnShowTips(1)
  end,
  btnMore2 = function(self)
    self:OnShowTips(2)
  end,
  btnMore3 = function(self)
    self:OnShowTips(3)
  end
}
tbWnd.tbOnToggleClick = {
  Toggle1 = function(self, isOn)
    if isOn then
      self:UpdateCashToggle(1)
    end
  end,
  Toggle2 = function(self, isOn)
    if isOn then
      self:UpdateCashToggle(2)
    end
  end,
  Toggle3 = function(self, isOn)
    if isOn then
      self:UpdateCashToggle(3)
    end
  end
}

function tbWnd:OnCreate()
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, self.UIMoneyPanel)
  
  local function fnSetModel(szObjPath, nNpcTemplateId)
    local NpcModelObj = self.pPanel:GetObject(szObjPath)
    local modelComp = NpcModelObj:GetComponent(Ui.UINpcModel)
    local nNpcResId = KNpc.GetNpcTemplateInfo(nNpcTemplateId).nNpcResID
    modelComp:ShowNpc(nNpcResId)
    modelComp.fModelScale = 55
  end
  
  fnSetModel(tbControls.Template1 .. "/NpcModel", Wedding.Cfg.tbCashNpcLevel[1].nNpcTID)
  fnSetModel(tbControls.Template2 .. "/NpcModel", Wedding.Cfg.tbCashNpcLevel[2].nNpcTID)
  fnSetModel(tbControls.Template3 .. "/NpcModel", Wedding.Cfg.tbCashNpcLevel[3].nNpcTID)
  self.pPanel:Label_SetText(tbControls.Template1 .. "/txtTittle", i18n.Parse(KNpc.GetNameByTemplateId(Wedding.Cfg.tbCashNpcLevel[1].nNpcTID)))
  self.pPanel:Label_SetText(tbControls.Template2 .. "/txtTittle", i18n.Parse(KNpc.GetNameByTemplateId(Wedding.Cfg.tbCashNpcLevel[2].nNpcTID)))
  self.pPanel:Label_SetText(tbControls.Template3 .. "/txtTittle", i18n.Parse(KNpc.GetNameByTemplateId(Wedding.Cfg.tbCashNpcLevel[3].nNpcTID)))
  self.pPanel:Label_SetText(tbControls.Template1 .. "/ImgGold/txtNum", tostring(Wedding.Cfg.tbCashNpcLevel[1].nCash))
  self.pPanel:Label_SetText(tbControls.Template2 .. "/ImgGold/txtNum", tostring(Wedding.Cfg.tbCashNpcLevel[2].nCash))
  self.pPanel:Label_SetText(tbControls.Template3 .. "/ImgGold/txtNum", tostring(Wedding.Cfg.tbCashNpcLevel[3].nCash))
end

function tbWnd:OnOpen()
  self.tbMoneyPanel:SetMoneyValue()
  self:UpdateCashToggle(1)
end

function tbWnd:UpdateCashToggle(nIndex)
  self.pPanel:Toggle_SetChecked(self.Toggle1, 1 == nIndex)
  self.pPanel:Toggle_SetChecked(self.Toggle2, 2 == nIndex)
  self.pPanel:Toggle_SetChecked(self.Toggle3, 3 == nIndex)
end

function tbWnd:GetCashIndex()
  if self.pPanel:Toggle_GetChecked(self.Toggle1) then
    return 1
  elseif self.pPanel:Toggle_GetChecked(self.Toggle2) then
    return 2
  elseif self.pPanel:Toggle_GetChecked(self.Toggle3) then
    return 3
  else
    return 0
  end
end

function tbWnd:SendCash()
  local nIndex = self:GetCashIndex()
  if nIndex <= 0 or nIndex > #Wedding.Cfg.tbCashNpcLevel then
    me.CenterMsg(i18n.EasyGet("\230\151\160\230\179\149\232\181\160\233\128\129\231\164\188\233\135\145"))
    return
  end
  local nSendGoldNum = Wedding.Cfg.tbCashNpcLevel[nIndex].nCash
  if nSendGoldNum > me.GetMoney("Gold") then
    me.CenterMsg(i18n.EasyGet("\233\135\145\229\184\129\228\184\141\232\182\179"))
    return
  end
  
  local function fnOK()
    RemoteServer.OnWeddingRequest("SendCashGift", nIndex)
    self:Close()
  end
  
  Ui:OpenWindow("UIMessageBox", i18n.Get(Wedding.Cfg.szCashGiftConfirm, nSendGoldNum), {
    {fnOK},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function tbWnd:OnChangeMoney()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:OnShowTips(nIndex)
  self.pPanel:SetActive(tbControls.wndTips, true)
  local szMsg = i18n.Get(Wedding.Cfg.tbCashNpcLevel[nIndex].szTip)
  local szName = i18n.Parse(KNpc.GetNameByTemplateId(Wedding.Cfg.tbCashNpcLevel[nIndex].nNpcTID))
  local tbPosition = self.pPanel:Object_GetPosition(tbControls["btnMore" .. tostring(nIndex)])
  local tbPos = {
    [1] = {x = 2.6, y = -0.7},
    [2] = {x = 2.6, y = -0.7},
    [3] = {x = -7.1, y = -0.7}
  }
  self.pPanel:Label_SetText(tbControls.txtHelpTitle, szName)
  self.pPanel:Label_SetText(tbControls.txtHelpDesc, szMsg)
  self.pPanel:Object_SetPosition(tbControls.wndTips, Ui.Vector3(tbPosition.x + tbPos[nIndex].x, tbPosition.y + tbPos[nIndex].y, tbPosition.z))
end
