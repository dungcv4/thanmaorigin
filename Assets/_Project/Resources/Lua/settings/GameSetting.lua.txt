GameSetting.MAX_COUNT_IN_BAG = 100
GameSetting.MAX_COUNT_IN_BOX = 40
GameSetting.MAX_PLAYER_LEVEL = 200
GameSetting.MAX_PLAYER_TOTAL_LEVEL = 400
GameSetting.PROGRESS_TIME_TASK_GATHER = 3
GameSetting.PROGRESS_TIME_FLY = 2
GameSetting.tbGolbalObjStack = {}

function GameSetting:SetGlobalObj(pPlayer, pNpc, pItem)
  self.tbGolbalObjStack[#self.tbGolbalObjStack + 1] = {
    pPlayer = me,
    pNpc = him,
    pItem = it
  }
  me = pPlayer or me
  him = pNpc or him
  it = pItem or it
end

function GameSetting:RestoreGlobalObj()
  local tb = self.tbGolbalObjStack[#self.tbGolbalObjStack]
  if not tb then
    assert(false)
  end
  me = tb.pPlayer or me
  him = tb.pNpc or him
  it = tb.pItem or it
  self.tbGolbalObjStack[#self.tbGolbalObjStack] = nil
end
