local tbRelationTypeToTextField = {
  [ChuanGong.tbRelationType.Recieve] = Ui.MenuText.szGetChuanGong,
  [ChuanGong.tbRelationType.Send] = Ui.MenuText.szSendChuanGong
}

local function CheckCanChuanGong_KinOrTong(tbData, emRelation)
  local tbChuanGong = tbData.tbChuanGong
  if nil == tbChuanGong then
    return false
  else
    local tbStatus = ChuanGong:CheckStatusFromOtherData(tbData.nTotalLevel, tbData.nVipLevel, tbChuanGong.nRecieveTimes > 0, tbChuanGong.nSendTimes > 0, tbChuanGong.nLastSendTime)
    return ChuanGong:StatusCanChuanGong(tbStatus, emRelation)
  end
end

local function MakeChuanGongMenuItem(tbClickGoToCloseWindow, emRelation)
  return {
    fnName = function(tbData)
      return i18n.Get(tbRelationTypeToTextField[emRelation])
    end,
    fnOnClick = function(tbMenuWnd)
      local tbData = tbMenuWnd.tbData
      local nDiffLevel = MasterPrentice:GetMinLevelDiff2(me.GetVipLevel(), tbData.nVipLevel or 0)
      local nId = tbData.nMemberId or tbData.dwID
      if ChuanGong:StatusCanChuanGong_MP(tbData, emRelation, tbData.nTotalLevel, nDiffLevel) then
        ChuanGong:RequestChuanGong_MP(nId, emRelation, tbClickGoToCloseWindow)
      else
        ChuanGong:RequestChuanGongC_Kin(nId, tbData.nTotalLevel, tbData.nVipLevel or 0, tbClickGoToCloseWindow, emRelation)
      end
      tbMenuWnd:Close()
    end,
    fnAvaliable = function(tbData)
      local nDiffLevel = MasterPrentice:GetMinLevelDiff2(me.GetVipLevel(), tbData.nVipLevel or 0)
      local bAvailable = ChuanGong:StatusCanChuanGong_MP(tbData, emRelation, tbData.nTotalLevel, nDiffLevel)
      if (me.dwKinId > 0 and me.dwKinId == tbData.dwKinId or me.dwTongId > 0 and me.dwTongId == tbData.dwTongId) and not bAvailable then
        bAvailable = CheckCanChuanGong_KinOrTong(tbData, emRelation)
      end
      return bAvailable
    end
  }
end

function Ui:LoadMenuSetting()
  local tbMenuSetting = {}
  for szMenuType, tbInfo in pairs(self.MenuSequence) do
    local tbBtnSetting = {}
    for nIndex, szBtnName in ipairs(tbInfo) do
      table.insert(tbBtnSetting, self.MenuItem[szBtnName])
    end
    tbMenuSetting[szMenuType] = tbBtnSetting
  end
  self.tbRightPopUpMenuSetting = tbMenuSetting
end

function Ui:GetMenuSetting()
  return self.tbRightPopUpMenuSetting or {}
end

Ui:LoadMenuSetting()
