AFKMedicineSetting.DefaultMedicineTId = Item:GetClass("MedicineItem").nDefaultMedicineTId

function AFKMedicineSetting:Init()
  self.nReplyMedicineTemplateId = self.DefaultMedicineTId
  self.bSettingLoaded = false
  self.bOpenProtect = true
  self.bOpenReplyMedicine = false
  self.bOpenUseItem = false
  self.bOpenUseJiu = true
  self.bOpenBuyJiuNotice = true
  self.bOpenUseChangXiao = false
  self.bOpenReplyRemindMsg = false
  self.bOpenTownPortalProtected = false
  self.tbCurRecommendMap = {}
  self.tbLifeProtect = {
    [1] = {nThresholdValue = 60, nMedicineCount = 1}
  }
  self.tbManaProtect = {
    [1] = {nThresholdValue = 30, nMedicineCount = 1}
  }
end

function AFKMedicineSetting:SetReplyMedicineTemplateId(nMedicineTemplateId)
  self.nReplyMedicineTemplateId = nMedicineTemplateId or self.DefaultMedicineTId
end

function AFKMedicineSetting:GetReplyMedicineTemplateId()
  self.nReplyMedicineTemplateId = self.nReplyMedicineTemplateId or self.DefaultMedicineTId
  return self.nReplyMedicineTemplateId
end

function AFKMedicineSetting:_LoadMedicineSetting()
  if self.bSettingLoaded then
    return
  end
  self.tbLifeProtect[1].nThresholdValue = LocalData:GetRoleNumber("AutoMedicine_ThresholdValue1", 60)
  self.tbLifeProtect[1].nMedicineCount = LocalData:GetRoleNumber("AutoMedicine_MedicineCount1", 1)
  self.tbManaProtect[1].nThresholdValue = LocalData:GetRoleNumber("AutoMedicine_ThresholdValue3", 30)
  self.tbManaProtect[1].nMedicineCount = LocalData:GetRoleNumber("AutoMedicine_MedicineCount3", 1)
  self.bOpenProtect = LocalData:GetRoleBoolean("AutoMedicine_OpenProtectFlag", true)
  self.bOpenReplyMedicine = LocalData:GetRoleBoolean("AutoMedicine_OpenReplyMedicine", true)
  self.bOpenUseJiu = LocalData:GetRoleBoolean("AutoMedicine_OpenUseJiu", true)
  self.bOpenBuyJiuNotice = LocalData:GetRoleBoolean("AutoMedicine_OpenBuyJiuNotice", true)
  self.bOpenUseChangXiao = LocalData:GetRoleBoolean("AutoMedicine_OpenUseChangXiao", false)
  self.bOpenReplyRemindMsg = LocalData:GetRoleBoolean("AutoMedicine_OpenReplyRemindMsg", false)
  self.bSettingLoaded = true
end

function AFKMedicineSetting:_SaveMedicineSetting()
  LocalData:SetRoleNumber("AutoMedicine_ThresholdValue1", self.tbLifeProtect[1].nThresholdValue)
  LocalData:SetRoleNumber("AutoMedicine_MedicineCount1", self.tbLifeProtect[1].nMedicineCount)
  LocalData:SetRoleNumber("AutoMedicine_ThresholdValue3", self.tbManaProtect[1].nThresholdValue)
  LocalData:SetRoleNumber("AutoMedicine_MedicineCount3", self.tbManaProtect[1].nMedicineCount)
  LocalData:SetRoleBoolean("AutoMedicine_OpenProtectFlag", self.bOpenProtect)
  LocalData:SetRoleBoolean("AutoMedicine_OpenReplyMedicine", self.bOpenReplyMedicine)
  LocalData:SetRoleBoolean("AutoMedicine_OpenUseJiu", self.bOpenUseJiu)
  LocalData:SetRoleBoolean("AutoMedicine_OpenBuyJiuNotice", self.bOpenBuyJiuNotice)
  LocalData:SetRoleBoolean("AutoMedicine_OpenUseChangXiao", self.bOpenUseChangXiao)
  LocalData:SetRoleBoolean("AutoMedicine_OpenReplyRemindMsg", self.bOpenReplyRemindMsg)
end

function AFKFightSetting:Init()
  self.bSettingLoaded = false
  self.nMode = self.ModeType.BaseOnFightPoint
  self.nSearchCoefficient = 1
  self.nFollowCoefficient = 1
  self.nFollowType = self.FollowType.Near
  self.nSelectType = self.SelectType.AnyOne
  self.nMaxFollowRadius = self.MaxFollowRadiusConfigFar
  self.nSearchRadius = self.NormalSearchRadius
  self.bLeftRouletteMedicineSetting = false
  self.nGoToFightMapType = self.GoToFightMapType.CurMap
  self.tbLastFightMapPos = {}
  self.tbLastAFKMapPos = {}
  self.bOpenAttackRadius = false
  self.nFrameCtrlOfQualitySetting = self.FrameCtrlOfQualitySettingTB.Middle
end

function AFKFightSetting:SetOpenAttackRadius(bStatus)
  self.bOpenAttackRadius = bStatus
end

function AFKFightSetting:SetGoToFightMapType(nType)
  self.nGoToFightMapType = nType or self.GoToFightMapType.CurMap
end

function AFKFightSetting:GetMapRecommendLevel(nMapId)
  local nLevel = 1
  for _, v in ipairs(self.RecommendMapTB) do
    for _, tbMapSet in ipairs(v[2]) do
      if tbMapSet[1] == nMapId then
        nLevel = v[1]
        return nLevel
      end
    end
  end
  return nLevel
end

function AFKFightSetting:ChooseGoToMap()
  local nMeMapId, nMeMapX, nMeMapY = me.GetWorldPos()
  local tbMap
  local nLevel = self:GetMapRecommendLevel(nMeMapId)
  local tbCurMap = {
    nMapId = nMeMapId,
    x = nMeMapX,
    y = nMeMapY,
    nLevel = nLevel
  }
  local tbLastFightMap = self.tbLastAFKMapPos
  local tbRecommendMap = self.tbCurRecommendMap
  if self.nGoToFightMapType == self.GoToFightMapType.CurMap then
    tbMap = tbCurMap
  elseif self.nGoToFightMapType == self.GoToFightMapType.LastFightMap then
    tbMap = tbLastFightMap
  elseif self.nGoToFightMapType == self.GoToFightMapType.RecommendMap then
    tbMap = tbRecommendMap
  end
  local bMapValid = self:IsMapValid(tbMap)
  if bMapValid then
    return tbMap
  end
  tbMap = tbCurMap
  bMapValid = self:IsMapValid(tbMap)
  if bMapValid then
    return tbMap
  end
  tbMap = tbLastFightMap
  bMapValid = self:IsMapValid(tbMap)
  if bMapValid then
    return tbMap
  end
  tbMap = tbRecommendMap
  bMapValid = self:IsMapValid(tbMap)
  if bMapValid then
    return tbMap
  end
end

function AFKFightSetting:IsMapValid(tbMap)
  if tbMap and tbMap.nMapId and tbMap.nMapId > 0 and tbMap.x and tbMap.x > 0 and tbMap.y and tbMap.y > 0 then
    return true
  end
  return false
end

function AFKFightSetting:ChooseRecommendMap()
  local nLevel = me.nLevel or 1
  for i = 1, #self.RecommendMapTB do
    if i == #self.RecommendMapTB then
      return {
        nMapId = -1,
        x = -1,
        y = -1,
        nLevel = 1
      }
    end
    if i == #self.RecommendMapTB or nLevel >= self.RecommendMapTB[i][1] and nLevel < self.RecommendMapTB[i + 1][1] then
      local tbPosSet = self.RecommendMapTB[i][2]
      local nChosenPos = math.random(1, #tbPosSet)
      self.tbCurRecommendMap = {
        nMapId = tbPosSet[nChosenPos][1],
        x = tbPosSet[nChosenPos][2],
        y = tbPosSet[nChosenPos][3],
        nLevel = self.RecommendMapTB[i][1]
      }
      return self.tbCurRecommendMap
    end
  end
end

function AFKFightSetting:SetFightRadius()
  self.nMaxFollowRadius = self.MaxFollowRadiusConfigFar
  if self.nFollowType == self.FollowType.Near then
    self.nMaxFollowRadius = self.MaxFollowRadiusConfigNear
  end
  self.nMaxFollowRadius = self.nMaxFollowRadius * self.nFollowCoefficient
  self.nSearchRadius = self.NormalSearchRadius * self.nSearchCoefficient
end

function AFKFightSetting:_LoadFightSetting()
  if self.bSettingLoaded then
    return
  end
  local nFollowType = LocalData:GetRoleNumber("AutoFight_FollowType", self.FollowType.Near)
  nFollowType = math.max(nFollowType, self.FollowType.Far)
  nFollowType = math.min(nFollowType, self.FollowType.Near)
  self.nFollowType = nFollowType
  local nSelectType = LocalData:GetRoleNumber("AutoFight_SelectType", self.SelectType.AnyOne)
  nSelectType = math.max(nSelectType, self.SelectType.AnyOne)
  nSelectType = math.min(nSelectType, self.SelectType.Boss)
  self.nSelectType = nSelectType
  local nGoToFightMapType = LocalData:GetRoleNumber("AutoFight_GoToFightMapType", self.GoToFightMapType.CurMap)
  nGoToFightMapType = math.max(nGoToFightMapType, self.GoToFightMapType.CurMap)
  nGoToFightMapType = math.min(nGoToFightMapType, self.GoToFightMapType.RecommendMap)
  self.nGoToFightMapType = nGoToFightMapType
  local nLastFightMapId = LocalData:GetRoleNumber("AutoFight_LastFightMapId", -1)
  local nLastFightMapX = LocalData:GetRoleNumber("AutoFight_LastFightMapX", -1)
  local nLastFightMapY = LocalData:GetRoleNumber("AutoFight_LastFightMapY", -1)
  self.tbLastAFKMapPos = {
    nMapId = nLastFightMapId,
    x = nLastFightMapX,
    y = nLastFightMapY
  }
  self.bLeftRouletteMedicineSetting = LocalData:GetRoleBoolean("AutoFight_LeftRouletteMedicineSetting", false)
  self:UpdateLeftRouletteMedicineSetting()
  self.bOpenAttackRadius = LocalData:GetRoleBoolean("AutoFight_OpenAttackRadius", false)
  self:SetFightRadius()
  self.bSettingLoaded = true
end

function AFKFightSetting:SetAutoFightRange(bForceShowRange)
  if 1 ~= me.nFightMode then
    return
  end
  if not Map:IsAllowPk(me.nMapTemplateId) then
    return
  end
  if not bForceShowRange then
    local tbFightPos = self:GetFightPos()
    self.nAutoFightRangeX = self.nAutoFightRangeX or 0
    self.nAutoFightRangeY = self.nAutoFightRangeY or 0
    if self.nAutoFightRangeX == tbFightPos.x and self.nAutoFightRangeY == tbFightPos.y then
      return
    end
    self.nAutoFightRangeX = tbFightPos.x
    self.nAutoFightRangeY = tbFightPos.y
  end
  if self.nCloseAutoFightRangeTimer then
    Timer:Close(self.nCloseAutoFightRangeTimer)
    self.nCloseAutoFightRangeTimer = nil
  end
  self.nCloseAutoFightRangeTimer = Timer:Register(Env.GAME_FPS * self.AutoFightRangeTimeCount, self.CloseAutoFightRange, self, true)
end

function AFKFightSetting:CloseAutoFightRange(IsTimerClose)
  if not IsTimerClose and self.nCloseAutoFightRangeTimer then
    Timer:Close(self.nCloseAutoFightRangeTimer)
    self.nCloseAutoFightRangeTimer = nil
  end
  Client.RepresentModule.SetAutoFightRangeEffect(false)
  self.nCloseAutoFightRangeTimer = nil
  return false
end

function AFKFightSetting:ResetAutoFightCoefficient()
  local nSearchCoefficient, nFollowCoefficient = Map:GetMapAutoFightCoefficient(me.nMapTemplateId)
  self.nSearchCoefficient = nSearchCoefficient
  self.nFollowCoefficient = nFollowCoefficient
end

function AFKFightSetting:SetLeftRouletteMedicineSetting(bShow)
  self.bLeftRouletteMedicineSetting = bShow
  self:UpdateLeftRouletteMedicineSetting()
end

function AFKFightSetting:UpdateLeftRouletteMedicineSetting()
  local bShow = self.bLeftRouletteMedicineSetting
  EventNotify.OnNotify(EventNotify.emNOTIFY_UPDATE_LEFT_ROULETTE_MEDICINE_SETTING, bShow)
end

function AFKFightSetting:UpdateAutoFightRange()
  self:SetFightRadius()
end

function AFKFightSetting:SetFollowType(nType)
  self.nFollowType = nType
  self:UpdateAutoFightRange()
end

function AFKFightSetting:SetFightModeType(nType)
  local nOldMode = self.nMode
  local bIsFuben = Map:IsFuben(me.nMapTemplateId)
  self.nMode = nType
  if bIsFuben and Dungeon:GetAutoRunDungeonState() then
    local nGuidX, nGuidY = Dungeon:GetTargetPos()
    if nGuidX and nGuidX > 0 and nGuidY and nGuidY > 0 then
      self.nMode = self.ModeType.BaseOnPlayer
    end
  end
  if nOldMode ~= self.nMode then
    self:ResetFightPos()
  end
  self:UpdateAutoFightRange()
end

function AFKFightSetting:SetSelectType(nType)
  self.nSelectType = nType
  Operation.nLastTargetId = 0
end

function AFKFightSetting:ResetFightPos()
  local nCurMapId, x, y = me.GetWorldPos()
  self.tbLastFightMapPos = {
    nMapId = nCurMapId,
    x = x,
    y = y
  }
  local bIsFuben = Map:IsFuben(me.nMapTemplateId)
  if bIsFuben and Dungeon:GetAutoRunDungeonState() then
    local nGuidX, nGuidY = Dungeon:GetTargetPos()
    if nGuidX and nGuidX > 0 and nGuidY and nGuidY > 0 then
      self.tbLastFightMapPos = {
        nMapId = nCurMapId,
        x = nGuidX,
        y = nGuidY
      }
    end
  end
  TeamMgr:SyncTeamMemberFightPos()
  if AFK:IsInLeadFightComplexMode() then
    self.tbLastAFKMapPos = self.tbLastFightMapPos
  end
  self:_UpdateRepresentFightPointPos()
  return self.tbLastFightMapPos
end

function AFKFightSetting:SetFightPos(nCurMapId, nX, nY)
  local bIsFuben = Map:IsFuben(me.nMapTemplateId)
  self.tbLastFightMapPos = {
    nMapId = nCurMapId,
    x = nX,
    y = nY
  }
  if bIsFuben and Dungeon:GetAutoRunDungeonState() then
    local nGuidX, nGuidY = Dungeon:GetTargetPos()
    if nGuidX and nGuidX > 0 and nGuidY and nGuidY > 0 then
      self.tbLastFightMapPos = {
        nMapId = nCurMapId,
        x = nGuidX,
        y = nGuidY
      }
    end
  end
  TeamMgr:SyncTeamMemberFightPos()
  if AFK:IsInLeadFightComplexMode() then
    self.tbLastAFKMapPos = self.tbLastFightMapPos
  end
  self:_UpdateRepresentFightPointPos()
  return self.tbLastFightMapPos
end

function AFKFightSetting:ClearFightPos()
  self.tbLastFightMapPos = {
    nMapId = -1,
    x = -1,
    y = -1
  }
end

function AFKFightSetting:GetFightPos()
  if not (self.tbLastFightMapPos and self.tbLastFightMapPos.nMapId and not (self.tbLastFightMapPos.nMapId < 0) and self.tbLastFightMapPos.x and not (self.tbLastFightMapPos.x < 0) and self.tbLastFightMapPos.y) or self.tbLastFightMapPos.y < 0 then
    self:ResetFightPos()
  end
  return self.tbLastFightMapPos
end

function AFKFightSetting:UpdateLeaderRepresentFightPointPos(nMapId, nX, nY)
  local nCurMapId = me.GetWorldPos()
  if nCurMapId ~= nMapId then
    return
  end
  Client.RepresentModule.SetAutoFightPointPos(nX, nY)
end

function AFKFightSetting:_UpdateRepresentFightPointPos()
  Client.RepresentModule.SetAutoFightPointPos(self.tbLastFightMapPos.x, self.tbLastFightMapPos.y)
end

function AFKFightSetting:_SaveFightSetting()
  LocalData:SetRoleBoolean("AutoFight_LeftRouletteMedicineSetting", self.bLeftRouletteMedicineSetting)
  LocalData:SetRoleNumber("AutoFight_FollowType", self.nFollowType)
  LocalData:SetRoleNumber("AutoFight_SelectType", self.nSelectType)
  LocalData:SetRoleNumber("AutoFight_GoToFightMapType", self.nGoToFightMapType)
  LocalData:SetRoleNumber("AutoFight_LastFightMapId", self.tbLastAFKMapPos.nMapId)
  LocalData:SetRoleNumber("AutoFight_LastFightMapX", self.tbLastAFKMapPos.x)
  LocalData:SetRoleNumber("AutoFight_LastFightMapY", self.tbLastAFKMapPos.y)
  LocalData:SetRoleBoolean("AutoFight_OpenAttackRadius", self.bOpenAttackRadius)
end

function AFKFightSetting:_OnQualitySettingChanged()
  if Client.QualityModule._CurrentQuality == Client.QualityModule.Quality.VeryLow then
    self.nFrameCtrlOfQualitySetting = self.FrameCtrlOfQualitySettingTB.VeryLow
    return
  end
  if Client.QualityModule._CurrentQuality == Client.QualityModule.Quality.Low then
    self.nFrameCtrlOfQualitySetting = self.FrameCtrlOfQualitySettingTB.Low
    return
  end
  if Client.QualityModule._CurrentQuality == Client.QualityModule.Quality.Middle then
    self.nFrameCtrlOfQualitySetting = self.FrameCtrlOfQualitySettingTB.Middle
    return
  end
  if Client.QualityModule._CurrentQuality == Client.QualityModule.Quality.High then
    self.nFrameCtrlOfQualitySetting = self.FrameCtrlOfQualitySettingTB.High
    return
  end
  self.nFrameCtrlOfQualitySetting = self.FrameCtrlOfQualitySettingTB.Middle
  return
end
