local tbWnd = Ui:CreateClass("UISetting_CustomHotkeyPanel", "UIPanel")
local tbControls = {
  btnSet = "panelPCBtnSetBattle/setList/Viewport/Content/btnSet",
  btnSave = "panelPCBtnSetBattle/btnSave",
  btnRecover = "panelPCBtnSetBattle/btnRecover",
  btnSet1 = "panelPCBtnSetFunction/setList/Viewport/Content/btnSet",
  btnSave1 = "panelPCBtnSetFunction/btnSave",
  btnRecover1 = "panelPCBtnSetFunction/btnRecover",
  panelPCBtnSetBattle = "panelPCBtnSetBattle",
  panelPCBtnSetFunction = "panelPCBtnSetFunction",
  Skill1Image = "panelPCBtnSetBattle/skillbg/Skill1Cell/skillImage",
  Skill2Image = "panelPCBtnSetBattle/skillbg/Skill2Cell/skillImage",
  Skill3Image = "panelPCBtnSetBattle/skillbg/Skill3Cell/skillImage",
  Skill4Image = "panelPCBtnSetBattle/skillbg/Skill4Cell/skillImage",
  Skill5Image = "panelPCBtnSetBattle/skillbg/Skill5Cell/skillImage",
  MedicineImage = "panelPCBtnSetBattle/skillbg/MedicineCell/skillImage"
}
tbWnd.tbControls = tbControls
local tbIconDict = {
  HOTKEY_BASIC_ATTACK = "panelPCBtnSetBattle/skillbg/Skill1Cell",
  HOTKEY_SKILL_1 = "panelPCBtnSetBattle/skillbg/Skill2Cell",
  HOTKEY_SKILL_2 = "panelPCBtnSetBattle/skillbg/Skill3Cell",
  HOTKEY_SKILL_3 = "panelPCBtnSetBattle/skillbg/Skill4Cell",
  HOTKEY_SKILL_4 = "panelPCBtnSetBattle/skillbg/Skill5Cell",
  HOTKEY_AUTO_FIGHT = "panelPCBtnSetBattle/skillbg/ManualCell",
  HOTKEY_USE_MEDICINE = "panelPCBtnSetBattle/skillbg/MedicineCell",
  HOTKEY_QINGKUNG = "panelPCBtnSetBattle/skillbg/JumpCell",
  HOTKEY_RIDE = "panelPCBtnSetBattle/skillbg/RideCell",
  HOTKEY_LOCK_TARGET = "panelPCBtnSetBattle/skillbg/SwitchTargetCell"
}
tbWnd.tbOnClick = {
  btnSave = function(self)
    CustomHotkey:ApplyEditBattleHotkey()
    CustomHotkey:ApplyEditCommonHotkey()
    self:NeedSaved(false)
    me.CenterMsg(i18n.EasyGet("\229\191\171\230\141\183\233\148\174\229\183\178\228\191\157\229\173\152"))
  end,
  btnRecover = function(self)
    CustomHotkey:RestoreEditBattleHotKey()
    self:RefreshBattleHotkeyDisplay()
    self:RefreshCommonHotkeyDisplay()
    self:NeedSaved(CustomHotkey:CheckEditChanged())
    me.CenterMsg(i18n.EasyGet("\233\135\141\231\189\174\230\136\152\230\150\151\229\191\171\230\141\183\233\148\174"))
  end,
  btnSave1 = function(self)
    CustomHotkey:ApplyEditBattleHotkey()
    CustomHotkey:ApplyEditCommonHotkey()
    self:NeedSaved(false)
    me.CenterMsg(i18n.EasyGet("\229\191\171\230\141\183\233\148\174\229\183\178\228\191\157\229\173\152"))
  end,
  btnRecover1 = function(self)
    CustomHotkey:RestoreEditCommonHotKey()
    self:RefreshBattleHotkeyDisplay()
    self:RefreshCommonHotkeyDisplay()
    self:NeedSaved(CustomHotkey:CheckEditChanged())
    me.CenterMsg(i18n.EasyGet("\233\135\141\231\189\174\229\184\184\232\167\132\229\191\171\230\141\183\233\148\174"))
  end
}
tbWnd.curEditInfo = {nPanelIndex = 0, keyName = nil}
tbWnd.isEdit = false

function tbWnd:OnCreate()
  CustomHotkey:InitEditHotkey()
  self.tbBattleBtnScrollView = Ui:NewScrollView(self, self.btnSet, function(_, nIndex, tbElement)
    tbWnd.curEditInfo.keyName = CustomHotkey.EditBattleHotkey[nIndex].KeyName
    for i = 1, #self.tbBattleBtnScrollView.tbElementList do
      local element = self.tbBattleBtnScrollView.tbElementList[i]
      element.pPanel:SetActive("imgSelect", i == nIndex)
    end
    self:SetIconHighLight(tbWnd.curEditInfo.keyName)
  end)
  for i = 1, #CustomHotkey.EditBattleHotkey do
    self.tbBattleBtnScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local btnInfo = CustomHotkey.EditBattleHotkey[i]
      pPanel:Label_SetText("txtSetContent", btnInfo.DisplayName())
      pPanel:Label_SetText("txtPcBtnBg", btnInfo.KeyCode)
    end)
  end
  self.tbCommonBtnScrollView = Ui:NewScrollView(self, self.btnSet1, function(_, nIndex, tbElement)
    tbWnd.curEditInfo.keyName = CustomHotkey.EditCommonHotkey[nIndex].KeyName
    for i = 1, #self.tbCommonBtnScrollView.tbElementList do
      local element = self.tbCommonBtnScrollView.tbElementList[i]
      element.pPanel:SetActive("imgSelect", i == nIndex)
    end
  end)
  for i = 1, #CustomHotkey.EditCommonHotkey do
    self.tbCommonBtnScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local btnInfo = CustomHotkey.EditCommonHotkey[i]
      pPanel:Label_SetText("txtSetContent", btnInfo.DisplayName())
      pPanel:Label_SetText("txtPcBtnBg", btnInfo.KeyCode)
    end)
  end
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end

function tbWnd:OnOpen()
  self:RefreshBattleHotkeyDisplay()
  self:RefreshCommonHotkeyDisplay()
  self.pPanel:SetActive(tbControls.panelPCBtnSetBattle, true)
  self.pPanel:SetActive(tbControls.panelPCBtnSetFunction, false)
  self.curEditInfo.keyName = nil
  self.curEditInfo.nIndex = 0
  self.curEditInfo.nPanelIndex = 0
  self:ResetSelect(1)
  self:NeedSaved(CustomHotkey:CheckEditChanged())
  self.isEdit = true
end

function tbWnd:OnClose()
  self.curEditInfo.keyName = nil
  self.curEditInfo.nIndex = 0
  self.curEditInfo.nPanelIndex = 0
  self.isEdit = false
end

function tbWnd:RefreshBattleHotkeyDisplay()
  for i = 1, #CustomHotkey.EditBattleHotkey do
    local element = self.tbBattleBtnScrollView.tbElementList[i]
    local keyCode = CustomHotkey.EditBattleHotkey[i].KeyCode
    local keyCodeSettingName = ""
    if CustomHotkey.KeyUIDisplayName[keyCode] then
      keyCodeSettingName = CustomHotkey.KeyUIDisplayName[keyCode].keyCodeSettingDisplayName()
    end
    if not keyCodeSettingName or "" == keyCodeSettingName then
      keyCodeSettingName = keyCode
    end
    element.pPanel:Label_SetText("txtPcBtnBg", keyCodeSettingName)
  end
end

function tbWnd:RefreshCommonHotkeyDisplay()
  for i = 1, #CustomHotkey.EditCommonHotkey do
    local element = self.tbCommonBtnScrollView.tbElementList[i]
    local keyCode = CustomHotkey.EditCommonHotkey[i].KeyCode
    local keyCodeSettingName = ""
    if CustomHotkey.KeyUIDisplayName[keyCode] then
      keyCodeSettingName = CustomHotkey.KeyUIDisplayName[keyCode].keyCodeSettingDisplayName()
    end
    if not keyCodeSettingName or "" == keyCodeSettingName then
      keyCodeSettingName = keyCode
    end
    element.pPanel:Label_SetText("txtPcBtnBg", keyCodeSettingName)
  end
end

function tbWnd:NeedSaved(bChanged)
  self.pPanel:Button_SetEnable(tbControls.btnSave, bChanged)
  self.pPanel:Button_SetEnable(tbControls.btnSave1, bChanged)
end

function tbWnd:CurEditKeyName()
  return self.curEditInfo.keyName
end

function tbWnd:IsEdit()
  return self.isEdit
end

function tbWnd:ResetSelect(panelIndex)
  if panelIndex == self.curEditInfo.nPanelIndex then
    return
  end
  self.curEditInfo.nPanelIndex = panelIndex
  self.curEditInfo.nIndex = 0
  self.curEditInfo.keyName = ""
  self.tbBattleBtnScrollView:ClearSelected()
  for i = 1, #self.tbBattleBtnScrollView.tbElementList do
    local element = self.tbBattleBtnScrollView.tbElementList[i]
    element.pPanel:SetActive("imgSelect", false)
  end
  self.tbCommonBtnScrollView:ClearSelected()
  for i = 1, #self.tbCommonBtnScrollView.tbElementList do
    local element = self.tbCommonBtnScrollView.tbElementList[i]
    element.pPanel:SetActive("imgSelect", false)
  end
  self:SetIconHighLight()
end

function tbWnd:RefreshIcon()
  local szIcon = MedicineMgr:GetCurUsedMedicineIcon()
  self.pPanel:Sprite_SetSprite(tbControls.MedicineImage, szIcon)
  self.tbSkills = {}
  self.tbUseSkill = {}
  local tbShowSkill = FightSkill:GetFactionSkill(me.nFaction, me.nFactionSect)
  if tbShowSkill then
    for k, skill in pairs(tbShowSkill) do
      local v = Lib:CopyTB(skill)
      self.tbUseSkill[v.SkillId] = v
      self.tbSkills[v.SortInUI] = v
      if 1 == v.SortInUI then
        self.pPanel:Sprite_SetSprite(tbControls.Skill1Image, v.IconAltlas)
      elseif 2 == v.SortInUI then
        self.pPanel:Sprite_SetSprite(tbControls.Skill2Image, v.IconAltlas)
      elseif 3 == v.SortInUI then
        self.pPanel:Sprite_SetSprite(tbControls.Skill3Image, v.IconAltlas)
      elseif 4 == v.SortInUI then
        self.pPanel:Sprite_SetSprite(tbControls.Skill4Image, v.IconAltlas)
      elseif 5 == v.SortInUI then
        self.pPanel:Sprite_SetSprite(tbControls.Skill5Image, v.IconAltlas)
      end
    end
  end
end

function tbWnd:SetIconHighLight(keyName)
  for key, iconPath in pairs(tbIconDict) do
    self.pPanel:SetActive(iconPath .. "/imgSelect", key == keyName)
  end
end
