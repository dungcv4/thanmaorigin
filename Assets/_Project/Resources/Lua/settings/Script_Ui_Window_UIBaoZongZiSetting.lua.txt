local tbWnd = Ui:CreateWindow("UIBaoZongZiSetting")
tbWnd.nMaxInputCount = 150
tbWnd.nMinInputCount = 1
tbWnd.tbControls = {
  txtMax = "node/imgBG/Content/txtMax",
  btnOK = "node/imgBG/Content/btnOK",
  SliderNum = "node/imgBG/Content/SliderNum",
  txtMin = "node/imgBG/Content/txtMin",
  btnClose = "node/imgBG/btnClose",
  txtSetNumTip = "node/imgBG/Content/SetNum/txtSetNumTip",
  txtInputNumValue = "node/imgBG/Content/SetNum/txtSetNumValue"
}
tbWnd.ZongziType = {ShangdengZongzi = 1, LvZongzi = 2}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    if self.nValue > 0 then
      self:MakeZongzi(self.nZongziType, self.nValue)
    end
    self:Close()
  end
}
tbWnd.tbOnSliderClick = {
  SliderNum = function(self, nValue)
    self.nValue = nValue
    self.pPanel:Label_SetText(self.txtInputNumValue, nValue)
  end
}

function tbWnd:NotifyEvents()
  return {}
end

function tbWnd:OnCreate()
end

function tbWnd:MakeZongzi(nZongziType, nValue)
  if not self.pInstance then
    return
  end
  if nZongziType == self.ZongziType.ShangdengZongzi then
    self.pInstance:MakeShangdengZongzi(nValue)
  elseif nZongziType == self.ZongziType.LvZongzi then
    self.pInstance:MakeLvZongzi(nValue)
  end
end

function tbWnd:OnOpen(nActiveID, nZongziType, nMaxNum, nTargetNum)
  if 0 == nMaxNum then
    me.CenterMsg(i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179\239\188\140\232\175\183\229\137\141\229\142\187\232\142\183\229\143\150"))
    self:Close()
    return
  end
  self.pInstance = ActivityMgr:GetInstanceById(nActiveID)
  self.nZongziType = nZongziType
  self:SetShowInfo(nZongziType)
  self.nTargetNum = nTargetNum
  self.nValue = 1
  self.pPanel:Slider_SetRange(self.SliderNum, 1, nMaxNum)
  self.pPanel:Slider_SetValue(self.SliderNum, nMaxNum)
  self.pPanel:Label_SetText(self.txtMin, 1)
  self.pPanel:Label_SetText(self.txtMax, nMaxNum)
end

function tbWnd:SetShowInfo(nZongziType)
  local szShowInfo = ""
  if nZongziType == self.ZongziType.ShangdengZongzi then
    szShowInfo = i18n.EasyGet("\233\128\137\230\139\169\229\136\182\228\189\156<color=#c56d05>\228\184\138\231\173\137\231\178\189\229\173\144</color>\231\154\132\230\149\176\233\135\143")
  elseif nZongziType == self.ZongziType.LvZongzi then
    szShowInfo = i18n.EasyGet("\233\128\137\230\139\169\229\136\182\228\189\156<color=#c56d05>\231\187\191\232\177\134\231\178\189\229\173\144</color>\231\154\132\230\149\176\233\135\143")
  end
  self.pPanel:Label_SetText(self.txtSetNumTip, szShowInfo)
end
