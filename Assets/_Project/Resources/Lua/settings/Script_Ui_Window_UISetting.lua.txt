local tbWnd = Ui:CreateWindow("UISetting")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnBasic = "node/imgBG/Content/btnList/btnBasic",
  btnSkill = "node/imgBG/Content/btnList/btnSkill",
  btnSafeLock = "node/imgBG/Content/btnList/btnLock",
  btnService = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/btnService",
  btnAgreement = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/btnAgreement",
  btnTakePhoto = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/btnTakePhoto",
  btnCode = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/btnCode",
  btnLock = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/lock/btnLock",
  btnUnLock = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/lock/btnUnLock",
  lock = "node/imgBG/Content/panelBasic/panelSystemSetting/ButtonGroup/lock",
  btnPCBtn = "node/imgBG/Content/btnList/btnPCBtn",
  btnPCBtnSetFunction = "node/imgBG/Content/btnList/btnPCBtnSetFunction",
  btnPCBtnSetBattle = "node/imgBG/Content/btnList/btnPCBtnSetBattle",
  panelPCBtnSetBattle = "node/imgBG/Content/panelPCBtn/panelPCBtnSetBattle",
  panelPCBtnSetFunction = "node/imgBG/Content/panelPCBtn/panelPCBtnSetFunction"
}
tbWnd.tbControls = tbControls
local hotkeyBtnIndex = 3
tbWnd.tbPanelListRoot = {
  {
    name = "node/imgBG/Content/panelBasic",
    panel = "UISetting_BasicPanel"
  },
  {
    name = "node/imgBG/Content/panelSkill",
    panel = "UISetting_FightPanel",
    cb = function(self)
      self.SkillPanelScroll.verticalNormalizedPosition = 1
    end
  },
  {
    name = "node/imgBG/Content/panelPCBtn",
    panel = "UISetting_CustomHotkeyPanel"
  },
  {
    name = "node/imgBG/Content/panelLock",
    panel = "UISetting_SafeLockPanel"
  }
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnService = function(self)
    Sdk:showCustomerSupport()
  end,
  btnAgreement = function(self)
    Ui:OpenWindow("UIAgreementReminderBig")
  end,
  btnTakePhoto = function(self)
    Ui:OpenWindow("UITakePhoto")
    self:Close()
  end,
  btnCode = function(self)
    Ui:OpenWindow("UIActivationCode")
  end,
  btnLock = function(self)
    SafeLock:TryUnLock()
  end,
  btnUnLock = function(self)
    SafeLock:TrySetPassWord(SafeLock.OperationTypeDef.SetPassWord)
  end,
  btnPCBtnSetBattle = function(self)
    self.pPanel:SetActive(tbControls.panelPCBtnSetBattle, true)
    self.pPanel:SetActive(tbControls.btnPCBtnSetBattle .. "/imgSmallSelect", true)
    self.pPanel:SetActive(tbControls.panelPCBtnSetFunction, false)
    self.pPanel:SetActive(tbControls.btnPCBtnSetFunction .. "/imgSmallSelect", false)
    self:GetUISetting_CustomHotkeyPanel():ResetSelect(1)
  end,
  btnPCBtnSetFunction = function(self)
    self.pPanel:SetActive(tbControls.panelPCBtnSetFunction, true)
    self.pPanel:SetActive(tbControls.btnPCBtnSetFunction .. "/imgSmallSelect", true)
    self.pPanel:SetActive(tbControls.panelPCBtnSetBattle, false)
    self.pPanel:SetActive(tbControls.btnPCBtnSetBattle .. "/imgSmallSelect", false)
    self:GetUISetting_CustomHotkeyPanel():ResetSelect(2)
  end
}

function tbWnd:OnCreate()
  self:SetClickEmptyToClose()
  self.SkillPanelScroll = self.pPanel:GetScrollRect("node/imgBG/Content/panelSkill")
  if not CustomHotkey:CheckPlatform() then
    self.pPanel:SetActive(tbControls.btnPCBtn, false)
    self.pPanel:SetActive(tbControls.btnPCBtnSetBattle, false)
    self.pPanel:SetActive(tbControls.btnPCBtnSetFunction, false)
  end
  self.tbPanelList = {}
  self.nCurPage = self.nCurPage or 1
  for i = 1, #self.tbPanelListRoot do
    self.tbPanelList[i] = Ui:CreatePanel(self.tbPanelListRoot[i].panel, self.tbPanelListRoot[i].name, self)
  end
  local tbBtnList = {
    tbControls.btnBasic,
    tbControls.btnSkill,
    tbControls.btnPCBtn,
    tbControls.btnSafeLock
  }
  self.tbButtonGroup = Ui:NewButtonGroup(self, tbBtnList, true, function(self, nIndex)
    self.nCurPage = nIndex
    if self.tbPanelListRoot[nIndex].cb ~= nil then
      self.tbPanelListRoot[nIndex].cb(self)
    end
    for i = 1, #self.tbPanelListRoot do
      self.tbPanelList[i]:SetActive(i == nIndex)
    end
    self.pPanel:SetActive(tbControls.btnPCBtnSetBattle, nIndex == hotkeyBtnIndex)
    self.pPanel:SetActive(tbControls.btnPCBtnSetBattle .. "/imgSmallSelect", nIndex == hotkeyBtnIndex)
    self.pPanel:SetActive(tbControls.btnPCBtnSetFunction, nIndex == hotkeyBtnIndex)
    self.pPanel:SetActive(tbControls.btnPCBtnSetFunction .. "/imgSmallSelect", false)
  end)
end

function tbWnd:SetPage(nPage)
  if nPage == self.nCurPage then
    return
  end
  self.nCurPage = nPage or 1
  self.tbButtonGroup:Select(self.nCurPage)
end

function tbWnd:OnOpen()
  RemoteServer.RequestSafeLock("RequestSyncData")
  self:OnChangeKeFu()
  self:OnChangeGiftCode()
  self:OnChangeLock()
  AFK:GetSetting():Load()
  CustomHotkey:InitEditHotkey()
  self.nCurPage = 1
  self.tbButtonGroup:Select(self.nCurPage)
  if Ui:WindowVisible("UILoginBG") then
    self.pPanel:SetActive(tbControls.btnTakePhoto, false)
  else
    self.pPanel:SetActive(tbControls.btnTakePhoto, true)
  end
  self:GetUISetting_CustomHotkeyPanel():RefreshIcon()
  self.pPanel:SetActive(tbControls.btnAgreement, Client.ShowAgreement)
end

function tbWnd:OnClose()
  if 0 ~= me.dwID then
    AFK:GetSetting():Save()
  end
  for i = 1, #self.tbPanelList do
    if self.tbPanelList[i].OnClose then
      self.tbPanelList[i]:OnClose()
    end
  end
  CustomHotkey:OnUISettingClose()
end

function tbWnd:OnDestroy()
end

function tbWnd:OnClickServiceBtn()
  Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\233\130\174\231\174\177\239\188\154jxsjqy@kingsoft.com"), {
    {},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function tbWnd:OnChangeKeFu()
  self.pPanel:SetActive(tbControls.btnService, OnOffController:Check(OnOffController.KEY.KeFu))
end

function tbWnd:OnChangeGiftCode()
  local bIsShow = OnOffController:Check(OnOffController.KEY.GiftCode)
  self.pPanel:SetActive(tbControls.btnCode, bIsShow and not IOS and not Ui:WindowVisible("UILoginBG"))
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_KEFU_ONOFF,
      self.OnChangeKeFu
    },
    {
      EventNotify.emNOTIFY_GIFTCODE_ONOFF,
      self.OnChangeGiftCode
    },
    {
      EventNotify.emNOTIFY_SAFELOCK_UNLOCK,
      self.OnChangeLock
    },
    {
      EventNotify.emNOTIFY_SAFELOCK_SYNDATA,
      self.OnChangeLock
    }
  }
end

function tbWnd:OnChangeLock()
  self.pPanel:SetActive(tbControls.lock, true)
  local bUnLock = not SafeLock:IsLock(me)
  local bSetPassWord = false
  if SafeLock.tbSafeLockData and SafeLock.tbSafeLockData.szPassword and not Lib:IsEmptyStr(SafeLock.tbSafeLockData.szPassword) then
    bSetPassWord = true
  end
  self.pPanel:SetActive(tbControls.btnLock, bSetPassWord and not bUnLock)
  self.pPanel:SetActive(tbControls.btnUnLock, not bSetPassWord or bUnLock)
  self.pPanel:SetActive(tbControls.btnSafeLock, bSetPassWord)
end

function tbWnd:GetUISetting_CustomHotkeyPanel()
  if self.tbPanelList then
    return self.tbPanelList[hotkeyBtnIndex]
  end
  return nil
end
