function tbWnd:UpdateKeyBoardTipSetting()
  if Sdk:GetOsType() == Sdk.eOSType_Windows then
    local nActive = Ui:GetKeyBoardTipActiveSetting()
    self.pPanel:SetActive(tbControls.toggleKeyboard, true)
    if 0 == nActive then
      self:KeyboardTipsSetting(false)
    else
      self:KeyboardTipsSetting(true)
    end
    return
  end
  self.pPanel:SetActive(self.tbControls.toggleKeyboard, false)
  self:KeyboardTipsSetting(false)
end

function tbWnd:KeyboardTipsSetting(bActive)
  if bActive then
    self.pPanel:Toggle_SetChecked(tbControls.toggleKeyboard, true)
    Ui:SetKeyBoardTipActiveSetting(1)
    EventNotify.OnNotify(EventNotify.emNOTIFY_UISetting_ACTIVE_KEYBOARD_TIPS)
  else
    self.pPanel:Toggle_SetChecked(tbControls.toggleKeyboard, false)
    Ui:SetKeyBoardTipActiveSetting(0)
    EventNotify.OnNotify(EventNotify.emNOTIFY_UISetting_ACTIVE_KEYBOARD_TIPS)
  end
end

function tbWnd:UpdateIsAutoBackFightPosAfterRevive()
  local bActive = Ui:GetIsAutoBackFightPosAfterRevive()
  self:SetIsAutoBackFightPosAfterRevive(bActive, true)
end

function tbWnd:SetIsAutoBackFightPosAfterRevive(bActive, bOnlyShow)
  self.pPanel:Toggle_SetChecked(tbControls.toggleReturn, bActive)
  if not bOnlyShow then
    Ui:SetIsAutoBackFightPosAfterRevive(bActive)
  end
end

function tbWnd:OnCreate()
  self.tbAttackTargetPriorGroup = Ui:NewButtonGroup(self, self.tbAttackTargetPriorGroupKey, true, function(self, nIndex)
    for i = 1, #self.tbAttackTargetPriorGroupKey do
      self.pPanel:SetActive(self.tbAttackTargetPriorGroupKey[i] .. "/imgChoose", i == nIndex)
    end
    local nSelectType = self.tbAttackTargetPriorGroup2Type[nIndex]
    AFKFightSetting:SetSelectType(nSelectType)
  end)
  self.tbAttackPursueSettingGroup = Ui:NewButtonGroup(self, self.tbAttackPursueSettingGroupKey, true, function(self, nIndex)
    for i = 1, #self.tbAttackPursueSettingGroupKey do
      self.pPanel:SetActive(self.tbAttackPursueSettingGroupKey[i] .. "/imgChoose", i == nIndex)
    end
    AFKFightSetting:SetFollowType(nIndex)
  end)
  self.tbLifeProtectSettingGroup = Ui:NewButtonGroup(self, self.tbLifeProtectSettingGroupKey, false, function(self, nIndex)
    self:OnLifeProtectInputClick(nIndex)
  end)
  self.tbChooseSkillCastTypeGroup = Ui:NewButtonGroup(self, self.tbChooseSkillCastTypeGroupKey, true, function(self, nIndex)
    for i = 1, #self.tbChooseSkillCastTypeGroupKey do
      self.pPanel:SetActive(self.tbChooseSkillCastTypeGroupKey[i] .. "/imgChoose", i == nIndex)
    end
    self:OnChooseSkillCastType(nIndex)
  end)
  for i, v in pairs(self.tbMedicineSlidersGroupKey) do
    self.pPanel:Slider_BindEvent(v, function(nValue)
      self.pPanel:Label_SetText(self.tbLifeProtectSettingGroupKey[i] .. "/txtInput", tostring(nValue))
      AFK:GetAutoMedicine():SetThresholdValue(i, nValue)
    end)
  end
  self.pPanel:DragInit(tbControls.SkillListPanel, {
    tbControls.SkillTemplate1,
    tbControls.SkillTemplate2,
    tbControls.SkillTemplate3,
    tbControls.SkillTemplate4
  }, tbControls.SkillTemplate)
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end

function tbWnd:UpdateSkillCastTypeState()
  local nIdx = Ui.PlayerPrefs.GetInt("SkillCastType", 1)
  self.tbChooseSkillCastTypeGroup:Select(nIdx)
end

function tbWnd:OnOpen()
  self.tbAttackTargetPriorGroup:Select(self.tbType2AttackTargetPriorGroup[AFKFightSetting.nSelectType])
  self.tbAttackPursueSettingGroup:Select(AFKFightSetting.nFollowType)
  if me.HasFaction() then
    self.pPanel:SetActive(tbControls.SkillList, true)
    self:UpdateAutoSkillSetting(false)
  else
    self.pPanel:SetActive(tbControls.SkillList, false)
  end
  self:UpdateSkillCastTypeState()
  self:InitMedicineSetting()
  self:InitToggleSetting()
  self:UpdateKeyBoardTipSetting()
  self:UpdateIsAutoBackFightPosAfterRevive()
end

function tbWnd:OnClose()
  self.pPanel:SetActive(tbControls.SkillTemplate, false)
end

function tbWnd:OnChooseSkillCastType(nIdx)
  Ui.PlayerPrefs.SetInt("SkillCastType", nIdx)
  EventNotify.OnNotify(EventNotify.emNOTIFY_SWITCH_SKILL_CAST_TYPE, nIdx)
end

function tbWnd:OnLifeProtectInputClick(nIdx)
  if self.fnInputFunc then
    self.fnInputFunc(nIdx)
  end
  Ui:OpenWindow("UINumKeyboard", nil, function(nValue)
    return self:LifeProtectInputValueChanged(nIdx, nValue)
  end, self.tbLifeProtectPercentInputPanelPos[nIdx])
end

function tbWnd:LifeProtectInputValueChanged(nIdx, nValue)
  if nValue < 0 then
    nValue = 0
  end
  if nValue > 100 then
    nValue = 100
  end
  self.pPanel:Label_SetText(self.tbLifeProtectSettingGroupKey[nIdx] .. "/txtInput", tostring(nValue))
  self.pPanel:Slider_SetValue(self.tbMedicineSlidersGroupKey[nIdx], nValue)
  AFK:GetAutoMedicine():SetThresholdValue(nIdx, nValue)
  return nValue
end

function tbWnd:UpdateAutoSkillSetting(bReSet)
  local tbSetting = AFKSkillMgr:GetSetting()
  if not tbSetting then
    return
  end
  self.nAutoSkillCount = Lib:CountTB(tbSetting)
  if self.nAutoSkillCount > self.nMaxSkillNum then
    self.nAutoSkillCount = self.nMaxSkillNum
  end
  local nEnd = self.nAutoSkillCount
  local tbAutoFightSkillOrder = self:GetSkillOrder(bReSet)
  self.tbSkillState = {}
  for i = 1, self.nAutoSkillCount do
    local tbSkill = tbSetting[i]
    local tbFactionSkillInfo = FightSkill:GetFactionSkillInfo(me.nFaction, me.nFactionSect, tbSkill.nSkillId)
    if tbFactionSkillInfo and 1 == tbFactionSkillInfo.SortInUI then
      tbSkill.bActive = false
      tbSkill.bShow = false
      self.tbSkillState["Skill" .. nEnd] = tbSkill
      nEnd = nEnd - 1
    else
      for j = 1, self.nAutoSkillCount - 1 do
        if tbAutoFightSkillOrder[j].nOrderInUI == tbFactionSkillInfo.SortInUI then
          tbSkill.bShow = true
          self.tbSkillState["Skill" .. j] = tbSkill
          tbSkill.bActive = 1 == tbAutoFightSkillOrder[j].nActive and true or false
        end
      end
    end
  end
  if bReSet then
    self:SetSkillOrder()
  else
    self:UpdateAutoSkillList()
  end
end

function tbWnd:UpdateAutoSkillList()
  for i = 1, self.nAutoSkillCount do
    local tbSkillState = self.tbSkillState["Skill" .. i]
    if not (tbSkillState and tbSkillState.bShow) then
      break
    end
    local tbSkillInfo = FightSkill:GetFactionSkillInfo(me.nFaction, me.nFactionSect, tbSkillState.nSkillId)
    if tbSkillInfo then
      local skillTemplate = self.tbControls["SkillTemplate" .. i]
      self.pPanel:Sprite_SetSprite(skillTemplate .. "/skill/framebg/imgSkill", tbSkillInfo.IconAltlas)
      self.pPanel:SetActive(skillTemplate .. "/btnSwitch/imgChoose", tbSkillState.bActive)
      Ui:Button_BindEvent(self.pPanel, skillTemplate .. "/btnSwitch", function()
        tbSkillState.bActive = not tbSkillState.bActive
        self.bAutoSkillSettingUpdate = true
        self:UpdateAutoSkillList()
        self:SetSkillOrder()
        self.bAutoSkillSettingUpdate = nil
      end)
    end
  end
  local tbSetting = AFKSkillMgr:GetSetting()
  for i = 1, self.nAutoSkillCount do
    tbSetting[i] = self.tbSkillState["Skill" .. i]
  end
  if self.bAutoSkillSettingUpdate then
    PlayerEvent:OnEvent(PlayerEvent.EventType.AutoSkillSet)
  end
end

function tbWnd:SetSkillTemplateImg(nIdx)
  local tbSkillInfo = FightSkill:GetFactionSkillInfo(me.nFaction, me.nFactionSect, self.tbSkillState["Skill" .. nIdx + 1].nSkillId)
  self.pPanel:Sprite_SetSprite(tbControls.SkillTemplate .. "/imgSkill", tbSkillInfo.IconAltlas)
end

function tbWnd:OnSkillObjectDrag(nIdx, nExchangeIdx)
  nIdx = nIdx + 1
  nExchangeIdx = nExchangeIdx + 1
  if 0 == nExchangeIdx or nIdx == nExchangeIdx then
    return
  end
  local tbTemp = self.tbSkillState["Skill" .. nIdx]
  self.tbSkillState["Skill" .. nIdx] = self.tbSkillState["Skill" .. nExchangeIdx]
  self.tbSkillState["Skill" .. nExchangeIdx] = tbTemp
  self.bAutoSkillSettingUpdate = true
  self:UpdateAutoSkillList()
  self:SetSkillOrder()
  self.bAutoSkillSettingUpdate = nil
end

function tbWnd:GetSkillOrder(bReSet)
  local nValue = me.GetUserValue(FightSkill.nAutoFightSkillGroup, 1)
  local tbAutoFightSkillOrder = {}
  if nValue > 0 and not bReSet then
    local nTemp = 15
    for i = 1, FightSkill.NORMAL_SKILL_COUNT - 1 do
      local nBitMove = (i - 1) * 5
      local nValueMoved = nValue >> nBitMove
      tbAutoFightSkillOrder[i] = {
        nOrderInUI = nValueMoved & nTemp,
        nActive = nValueMoved >> 4 & 1
      }
    end
  else
    for i = 1, FightSkill.NORMAL_SKILL_COUNT - 1 do
      tbAutoFightSkillOrder[i] = {
        nOrderInUI = i + 1,
        nActive = 1
      }
    end
  end
  return tbAutoFightSkillOrder
end

function tbWnd:SetSkillOrder()
  local nValue = 0
  for i = 1, FightSkill.NORMAL_SKILL_COUNT - 1 do
    local tbSkill = self.tbSkillState["Skill" .. i]
    local tbFactionSkillInfo = FightSkill:GetFactionSkillInfo(me.nFaction, me.nFactionSect, tbSkill.nSkillId)
    local nActive = tbSkill.bActive and 1 or 0
    local nTempValue = tbFactionSkillInfo.SortInUI
    nTempValue = nTempValue | nActive << 4
    local nBitMove = (i - 1) * 5
    local nValueMoved = nTempValue << nBitMove
    nValue = nValue | nValueMoved
  end
  RemoteServer.SetAutoFightSkillTask(nValue)
end

function tbWnd:ApplyFightSetting()
  local bHideName = 0 == Ui.PlayerPrefs.GetInt("ShowAllPlayerName", 1)
  local bHideBloodBar = 0 == Ui.PlayerPrefs.GetInt("ShowAllPlayerBloodBar", 1)
  local bHideOthers = 0 == Ui.PlayerPrefs.GetInt("ShowOtherPlayer", 1)
  local bHideNPC = 0 == Ui.PlayerPrefs.GetInt("ShowNPC", 1)
  Client.RepresentModule.SetNpcHeadTextVisable(not bHideName)
  Client.RepresentModule.SetPlayerHeadTextVisable(not bHideName)
  Client.RepresentModule.SetMeHeadTextVisable(not bHideName)
  Client.RepresentModule.SetNpcBloodBarVisable(not bHideBloodBar)
  Client.RepresentModule.SetPlayerBloodBarVisable(not bHideBloodBar)
  Client.RepresentModule.SetMeBloodBarVisable(not bHideBloodBar)
  Client.RepresentModule.SetPlayerVisable(not bHideOthers)
  Client.RepresentModule.SetPlayerShadowVisable(not bHideOthers)
  Client.RepresentModule.SetNpcVisable(not bHideNPC)
  Client.RepresentModule.SetNpcShadowVisable(not bHideNPC)
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, tbWnd.ApplyFightSetting, tbWnd)
