local tbWnd = Ui:CreateClass("UIClosetShowSetting", "UIPanel")
local tbControls = {
  Template = "imgBG/List/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbShowInfoConfig = {
  [1] = {
    pCallBackFunc = function(bHide)
      Fashion:SetFashionHide(Npc.NpcResPartsDef.npc_part_mantle, bHide)
      Fashion:HidePartFashionPreview(Npc.NpcResPartsDef.npc_part_mantle, bHide)
    end,
    pGetStateFunc = function()
      return Fashion:IsFeatureHide(me, Npc.NpcResPartsDef.npc_part_mantle)
    end,
    pGetNameFunc = function()
      return i18n.EasyGet("\233\154\144\232\151\143\230\138\171\233\163\142/\230\140\130\228\187\182")
    end
  }
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:UpdateView()
  self.tbScrollView:Clear()
  for nIndex, tbInfo in ipairs(tbWnd.tbShowInfoConfig) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtMapName", tbInfo.pGetNameFunc())
      pPanel:Toggle_SetChecked("Toggle", tbInfo.pGetStateFunc())
      Ui:Toggle_BindEvent(pPanel, "Toggle", function(self, bIsOn)
        tbInfo.pCallBackFunc(bIsOn)
      end)
    end)
  end
end
