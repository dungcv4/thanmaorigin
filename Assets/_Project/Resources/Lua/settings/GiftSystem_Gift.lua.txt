function Gift:CheckSex(nFaction)
  if Gift.tbGirls[nFaction] then
    return Gift.Sex.Girl
  elseif Gift.tbBoys[nFaction] then
    return Gift.Sex.Boy
  end
end

function Gift:CheckCommond(pPlayer, nAcceptId, nCount, nGiftType)
  if not nCount or nCount < 1 then
    return false, i18n.EasyGet("\232\175\183\233\128\137\230\139\169\232\166\129\232\181\160\233\128\129\231\154\132\231\137\169\229\147\129")
  end
  local pAcceptPlayer = KPlayer.GetPlayerObjById(nAcceptId)
  if not pAcceptPlayer then
    return false, i18n.EasyGet("\229\175\185\230\150\185\228\184\141\229\156\168\231\186\191")
  end
  local nSex = Player:GetPlayerSex(pAcceptPlayer)
  if not nSex then
    return false, i18n.EasyGet("\229\175\185\230\150\185\230\152\175\231\148\183?\230\152\175\229\165\179?")
  end
  local bIsFriend = FriendShip:IsFriend(pPlayer.dwID, nAcceptId)
  if not bIsFriend then
    return false, i18n.EasyGet("\229\175\185\230\150\185\228\184\141\230\152\175\228\189\160\231\154\132\229\165\189\229\143\139")
  end
  local nItemId = Gift.AllItem[nGiftType] and Gift.AllItem[nGiftType][nSex]
  if not nItemId then
    return false, i18n.EasyGet("\230\137\190\228\184\141\229\136\176\232\166\129\232\181\160\233\128\129\231\154\132\231\137\169\229\147\129")
  end
  local nOrgCount = pPlayer.GetItemCountByTypeInBag(nItemId)
  local szItemName = KItem.GetItemShowInfo(nItemId, pPlayer.nFaction, pPlayer.nFactionSect)
  if nCount > nOrgCount then
    return false, i18n.EasyGet("\230\130\168\231\154\132{0}\228\184\141\229\164\159", szItemName)
  end
  return true, "", pAcceptPlayer, nSex, nItemId, szItemName
end

function Gift:ItemDes(nGiftType, nSex)
  local szUnit = ""
  local tbItems = Gift.AllItem[nGiftType]
  if not tbItems then
    return
  end
  local nItemTId = tbItems[nSex]
  if not nItemTId then
    return
  end
  local tbBase = Item:GetItemBaseProp(nItemTId)
  if not tbBase then
    return
  end
  return tbBase.szName
end
