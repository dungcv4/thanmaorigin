if EDITOR then
  local record = {}
  local mt = {}
  setmetatable(EventNotify, {
    __index = mt,
    __newindex = function(t, k, v)
      if rawget(mt, k) ~= nil then
        error(string.format("EventNotify.%s Redefine ", k))
      end
      if type(v) == "number" then
        if record[v] then
          error(string.format("EventNotify.%s == EventNotify.%s value is %d", k, record[v], v))
        else
          record[v] = k
        end
      end
      rawset(mt, k, v)
    end
  })
end

function EventNotify:Init()
  self.tbCallback = {}
end

if not EventNotify.tbCallback then
  EventNotify:Init()
end

function EventNotify:RegistNotify(nEvent, fnProc, tbTable)
  assert(tbTable, string.format("nEvent:%s,", tostring(nEvent)))
  assert(nEvent, string.format("UiGroup:%s, nEvent:%s,", tostring(tbTable.ClassName), tostring(nEvent)))
  assert(fnProc, string.format("UiGroup:%s, nEvent:%s,fnProc is nil", tostring(tbTable.ClassName), tostring(nEvent)))
  local tbCallback = self.tbCallback[nEvent]
  if not tbCallback then
    tbCallback = {}
    self.tbCallback[nEvent] = tbCallback
  elseif tbCallback[tostring(tbTable)] then
    LogErr("Repeat Regist\239\188\154", nEvent)
    LogErr(debug.traceback())
    return
  end
  if type(tbTable) == "table" then
    tbCallback[tostring(tbTable)] = {tbSelf = tbTable, fnProc = fnProc}
  end
end

function EventNotify:UnRegistNotify(nEvent, tbTable)
  local tbCallback = self.tbCallback[nEvent]
  if tbCallback and tbCallback[tostring(tbTable)] then
    tbCallback[tostring(tbTable)] = nil
  end
end

function EventNotify.OnNotify(nEvent, ...)
  if not EventNotify.tbCallback[nEvent] then
    return
  end
  for _, tb in pairs(EventNotify.tbCallback[nEvent]) do
    local szUiName = tb.tbSelf.ClassName
    Lib:CallBack({
      tb.fnProc,
      tb.tbSelf,
      ...
    })
  end
end

local metatb = getmetatable(EventNotify)
local mt
if metatb then
  mt = metatb.__index
end
setmetatable(EventNotify, {
  __index = function(t, k)
    local tb = mt or t
    local v = rawget(tb, k)
    if nil == v and EDITOR then
      error(string.format("EventNotify.%s == nil", k))
    end
    return v
  end,
  __newindex = function(t, k, v)
    if EDITOR then
      error(string.format("EventNotify is readonly", k))
    end
  end
})
