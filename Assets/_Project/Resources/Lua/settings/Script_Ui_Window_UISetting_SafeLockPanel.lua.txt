local tbWnd = Ui:CreateClass("UISetting_SafeLockPanel", "UIPanel")
local tbControls = {
  btnChange = "btnChange",
  btnCancel = "btnCancel",
  btnForget = "btnForget",
  btnSave = "btnSave",
  Cancel = "btnCancel/txtCell",
  txtCancelTime = "btnCancel/txtCell/txtTime",
  Forget = "btnForget/txtCell",
  txtForgetTime = "btnForget/txtCell/txtTime",
  ToggleDevelop = "panelProtect/ToggleDevelop",
  ToggleSocial = "panelProtect/ToggleSocial",
  ToggleCurrency = "panelProtect/ToggleCurrency",
  ToggleRole = "panelProtect/ToggleRole"
}
tbWnd.tbControls = tbControls
local tbModule2Toggle = {
  [SafeLock.ModuleDef.Develop] = tbControls.ToggleDevelop,
  [SafeLock.ModuleDef.Social] = tbControls.ToggleSocial,
  [SafeLock.ModuleDef.Currency] = tbControls.ToggleCurrency,
  [SafeLock.ModuleDef.Player] = tbControls.ToggleRole
}
tbWnd.tbOnClick = {
  btnChange = function(self)
    SafeLock:TryChangePassWord(false)
  end,
  btnCancel = function(self)
    Ui:OpenWindow("UILock", SafeLock.OperationTypeDef.CancelPassWord, true)
  end,
  btnForget = function(self)
    RemoteServer.RequestSafeLock("RequestSetForgetPassWord")
  end,
  btnSave = function(self)
    if SafeLock:IsLock(me) then
      Ui:OpenWindow("UILock", SafeLock.OperationTypeDef.UnLock, true)
      return
    end
    local tbState = {}
    for _, nModule in pairs(SafeLock.ModuleDef) do
      if self.tbNewModuleState[nModule] ~= SafeLock.tbSafeLockData.tbModuleOnOff[nModule] then
        tbState[nModule] = self.tbNewModuleState[nModule]
      end
    end
    if 0 ~= Lib:CountTB(tbState) then
      SafeLock:TrySetModuleOnOff(tbState)
    end
  end
}
tbWnd.tbOnToggleClick = {
  ToggleDevelop = function(self, isOn)
    self:OnToggleClick(SafeLock.ModuleDef.Develop, isOn)
  end,
  ToggleSocial = function(self, isOn)
    self:OnToggleClick(SafeLock.ModuleDef.Social, isOn)
  end,
  ToggleCurrency = function(self, isOn)
    self:OnToggleClick(SafeLock.ModuleDef.Currency, isOn)
  end,
  ToggleRole = function(self, isOn)
    self:OnToggleClick(SafeLock.ModuleDef.Player, isOn)
  end
}

function tbWnd:OnCreate()
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end

function tbWnd:OnOpen()
  if not SafeLock.tbSafeLockData then
    return
  end
  self.tbNewModuleState = {}
  self:InitToggole()
  self:InitForget()
  self:InitCancel()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SAFELOCK_SYNDATA, self.UpDatePanel, self)
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SAFELOCK_SYNDATA, self)
end

function tbWnd:InitToggole()
  if SafeLock.tbSafeLockData and SafeLock.tbSafeLockData.tbModuleOnOff then
    for nModule, bOpen in pairs(SafeLock.tbSafeLockData.tbModuleOnOff) do
      local szPath = tbModule2Toggle[nModule]
      self.pPanel:Toggle_SetChecked(szPath, bOpen)
      self.tbNewModuleState[nModule] = bOpen
    end
  end
end

function tbWnd:InitForget()
  local bInForget = true
  if not SafeLock.tbSafeLockData.nForgetEndTime or SafeLock.tbSafeLockData.nForgetEndTime < GetTime() then
    bInForget = false
  end
  self.pPanel:SetActive(tbControls.Forget, bInForget)
  if bInForget then
    local nDiff = SafeLock.tbSafeLockData.nForgetEndTime - GetTime()
    local szDesc = i18n.EasyGet("\229\174\137\229\133\168\233\148\129\229\176\134\228\186\142{0}\229\144\142\229\164\177\230\149\136", Lib:TimeDesc12(nDiff))
    self.pPanel:Label_SetText(tbControls.txtForgetTime, szDesc)
    self.pPanel:Label_SetText(tbControls.btnForget .. "/txt", i18n.EasyGet("\231\187\136\230\173\162\229\191\152\232\174\176\229\175\134\231\160\129"), "Text")
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnForget, true)
  else
    self.pPanel:Label_SetText(tbControls.btnForget .. "/txt", i18n.EasyGet("\229\191\152\232\174\176\229\175\134\231\160\129"), "Text")
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnForget, false)
  end
end

function tbWnd:InitCancel()
  local bInCancel = true
  if not SafeLock.tbSafeLockData.nCancelEndTime or SafeLock.tbSafeLockData.nCancelEndTime < GetTime() then
    bInCancel = false
  end
  self.pPanel:SetActive(tbControls.Cancel, bInCancel)
  if bInCancel then
    local nDiff = SafeLock.tbSafeLockData.nCancelEndTime - GetTime()
    local szDesc = i18n.EasyGet("\229\174\137\229\133\168\233\148\129\229\176\134\228\186\142{0}\229\144\142\229\164\177\230\149\136", Lib:TimeDesc12(nDiff))
    self.pPanel:Label_SetText(tbControls.txtCancelTime, szDesc)
    self.pPanel:Label_SetText(tbControls.btnCancel .. "/txt", i18n.EasyGet("\231\187\136\230\173\162\229\143\150\230\182\136\229\175\134\231\160\129"), "Text")
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnCancel, true)
  else
    self.pPanel:Label_SetText(tbControls.btnCancel .. "/txt", i18n.EasyGet("\229\143\150\230\182\136\229\175\134\231\160\129"), "Text")
    self.pPanel:ButtonGroup_SetSelect(tbControls.btnCancel, false)
  end
end

function tbWnd:UpDatePanel()
  self:InitToggole()
  self:InitForget()
  self:InitCancel()
end

function tbWnd:OnToggleClick(nType, isOn)
  if not self.tbNewModuleState[nType] and not isOn then
    return
  end
  if self.tbNewModuleState[nType] == isOn then
    return
  end
  if SafeLock:IsLock(me) then
    Ui:OpenWindow("UILock", SafeLock.OperationTypeDef.UnLock, true)
    self:InitToggole()
    return
  end
  self.tbNewModuleState[nType] = isOn
end
