tbWnd.tbOnSliderClick = {
  sliderMusic = function(self, nValue)
    ClientUtil:SetMusicVolume(nValue, true)
    self:UpdateMusicIcon(0 == nValue)
  end,
  sliderSound = function(self, nValue)
    ClientUtil:SetSoundVolume(nValue, true)
    self:UpdateSoundIcon(0 == nValue)
  end,
  sliderShowPlayerCount = function(self, nValue)
    local nCount = Lib:MathRound(nValue * Client.QualityModule._Setting.MaxSyncPlayerCount / 100)
    self.pPanel:Label_SetText(self.txtShowPlayerCount, nCount)
    Client:Quality_SetMaxShowPlayerCount(nCount, true)
  end,
  SliderSight = function(self, nValue)
    Ui:SetCameraDistance(nValue, true)
  end
}
tbWnd.tbOnToggleClick = {
  ToggleQualityVeryLow = function(self, isOn)
    if isOn then
      self:OnClickQualityToggle(0)
    end
  end,
  ToggleQualityLow = function(self, isOn)
    if isOn then
      self:OnClickQualityToggle(1)
    end
  end,
  ToggleQualityMiddle = function(self, isOn)
    if isOn then
      self:OnClickQualityToggle(2)
    end
  end,
  ToggleQualityHigh = function(self, isOn)
    if isOn then
      self:OnClickQualityToggle(3)
    end
  end,
  ToggleShowPlayerFactionAndLevel = function(self, isOn)
    Client.QualityModule.SetShowPlayerFactionAndLevel(isOn)
    Client.NpcManager.UpdateAllPlayerShowName()
  end,
  ToggleForbidWeather = function(self, isOn)
    if isOn then
      Ui:SetFbrbidWeatherSetting(1)
      Ui:StopRaining()
    else
      Ui:SetFbrbidWeatherSetting(0)
      Weather:CheckWeather()
    end
  end,
  ToggleHideHeadText = function(self, isOn)
    if isOn then
      Client:SetToggleHideHeadText(1)
    else
      Client:SetToggleHideHeadText(0)
    end
  end,
  ToggleHideBloodBar = function(self, isOn)
    if isOn then
      Client:SetToggleHideBloodBar(1)
    else
      Client:SetToggleHideBloodBar(0)
    end
  end,
  TogglePowerSave = function(self, isOn)
    self:ModifySavePowerSetting(isOn)
  end
}

function tbWnd:OnCreate()
  self.pPanel:SetActive(tbControls.btnAgreement, Client.AgreementUrl ~= nil)
  self.pPanel:SetActive(tbControls.btnPrivacy, nil ~= Client.PrivacyUrl)
  self.pPanel:SetActive(tbControls.Dropdown, false)
end

function tbWnd:SetActive(bActive)
  self.pPanel:SetActive("Main", bActive)
  if bActive then
    self:OnOpen()
  else
    self:OnClose()
  end
end

function tbWnd:OnOpen()
  self:UpdateSoundSetting()
  self:UpdateCameraViewSetting()
  self:UpdateQualitySetting()
  self:UpdateShowPlayerCount()
  self:UpdateSavePowerSetting()
  self:UpdateCurrencyTradeLimit()
  self.pPanel:SetActive(tbControls.btnTuoGuan, false)
  self.pPanel:SetActive(tbControls.btnShare, false)
  self.pPanel:SetActive(tbControls.btnCustomSafeArea, false)
  self.pPanel:SetActive(tbControls.btnUserAgreement, false)
  if 1 == Login.nShowAgreement then
    self.pPanel:SetActive(tbControls.btnAgreement, false)
    self.pPanel:SetActive(tbControls.btnPrivacy, false)
  end
  self.pPanel:SetActive(tbControls.ToggleShowVipIcon, Client.EnableToggleVip == true)
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSoundSetting()
  local nMusicValue = Ui.PlayerPrefs.GetFloat("MusicVolume", 100)
  self.pPanel:Slider_SetValue(tbControls.sliderMusic, nMusicValue)
  self:UpdateMusicIcon(0 == nMusicValue)
  ClientUtil:SetMusicVolume(nMusicValue, false)
  local nSoundValue = Ui.PlayerPrefs.GetFloat("SoundVolume", 100)
  self.pPanel:Slider_SetValue(tbControls.sliderSound, nSoundValue)
  self:UpdateSoundIcon(0 == nSoundValue)
  ClientUtil:SetSoundVolume(nSoundValue, false)
  self.pPanel:SetActive(tbControls.txtVoice, Client.ShowVoiceSetting)
  self.pPanel:SetActive(tbControls.sliderVoice, Client.ShowVoiceSetting)
  self.pPanel:SetActive(tbControls.imgVoice, Client.ShowVoiceSetting)
end

function tbWnd:UpdateCameraViewSetting()
  local nDistance = Ui:GetCameraDistance()
  self.pPanel:Slider_SetValue(tbControls.SliderSight, nDistance)
  local nShowPlayerFactionAndLevel = Ui.PlayerPrefs.GetInt("ShowPlayerFactionAndLevel", 0)
  if 1 == nShowPlayerFactionAndLevel then
    self.pPanel:Toggle_SetChecked(tbControls.ToggleShowPlayerFactionAndLevel, true)
  else
    self.pPanel:Toggle_SetChecked(tbControls.ToggleShowPlayerFactionAndLevel, false)
  end
  local nForbidWeather = Ui:GetFbrbidWeatherSetting()
  if 1 == nForbidWeather then
    self.pPanel:Toggle_SetChecked(tbControls.ToggleForbidWeather, true)
  else
    self.pPanel:Toggle_SetChecked(tbControls.ToggleForbidWeather, false)
  end
  self.pPanel:Toggle_SetChecked(tbControls.ToggleHideHeadText, 1 == Client.nToggleHideHeadText)
  self.pPanel:Toggle_SetChecked(tbControls.ToggleHideBloodBar, 1 == Client.nToggleHideBloodBar)
end

function tbWnd:UpdateCurrencyTradeLimit()
  if not Client.EnableCurrencyTradeLimit or not CurrencyTradeLimit then
    self.pPanel:SetActive(self.objCurrencyTradeLimit, false)
    return
  end
  self.pPanel:SetActive(self.objCurrencyTradeLimit, CurrencyTradeLimit:CheckIsLimitOpen())
  local nLimit = CurrencyTradeLimit:GetAllCurrencyTradeLimit(me)
  if CurrencyTradeLimit:CheckIsLimitOpen() then
    self.pPanel:Label_SetText(self.txtLimit, i18n.EasyGet("\229\137\169\228\189\153\233\135\145\229\184\129\228\186\164\230\152\147\233\162\157\229\186\166\239\188\154{0}", nLimit))
  end
end

function tbWnd:UpdateQualitySetting()
  local nRecommandQualitySetting = Client.QualityModule.GetRecommandQuality()
  local nQualitySetting = Ui.PlayerPrefs.GetInt("QualitySetting", nRecommandQualitySetting)
  if 0 == nQualitySetting then
    self.pPanel:Toggle_SetChecked(tbControls.ToggleQualityVeryLow, true)
  elseif 1 == nQualitySetting then
    self.pPanel:Toggle_SetChecked(tbControls.ToggleQualityLow, true)
  elseif 2 == nQualitySetting then
    self.pPanel:Toggle_SetChecked(tbControls.ToggleQualityMiddle, true)
  elseif 3 == nQualitySetting then
    self.pPanel:Toggle_SetChecked(tbControls.ToggleQualityHigh, true)
  end
  if 0 == nRecommandQualitySetting then
    self.pPanel:SetActive(tbControls.RecommandVeryLow, true)
    self.pPanel:SetActive(tbControls.RecommandLow, false)
    self.pPanel:SetActive(tbControls.RecommandMiddle, false)
    self.pPanel:SetActive(tbControls.RecommandHigh, false)
  elseif 1 == nRecommandQualitySetting then
    self.pPanel:SetActive(tbControls.RecommandVeryLow, false)
    self.pPanel:SetActive(tbControls.RecommandLow, true)
    self.pPanel:SetActive(tbControls.RecommandMiddle, false)
    self.pPanel:SetActive(tbControls.RecommandHigh, false)
  elseif 2 == nRecommandQualitySetting then
    self.pPanel:SetActive(tbControls.RecommandVeryLow, false)
    self.pPanel:SetActive(tbControls.RecommandLow, false)
    self.pPanel:SetActive(tbControls.RecommandMiddle, true)
    self.pPanel:SetActive(tbControls.RecommandHigh, false)
  elseif 3 == nRecommandQualitySetting then
    self.pPanel:SetActive(tbControls.RecommandVeryLow, false)
    self.pPanel:SetActive(tbControls.RecommandLow, false)
    self.pPanel:SetActive(tbControls.RecommandMiddle, false)
    self.pPanel:SetActive(tbControls.RecommandHigh, true)
  end
end

function tbWnd:UpdateShowPlayerCount()
  local nQualitySetting = Ui.PlayerPrefs.GetInt("QualitySetting", 0)
  if 0 == nQualitySetting then
    self.pPanel:Slider_SetEnable(tbControls.sliderShowPlayerCount, false)
  elseif 1 == nQualitySetting then
    self.pPanel:Slider_SetEnable(tbControls.sliderShowPlayerCount, true)
  elseif 2 == nQualitySetting then
    self.pPanel:Slider_SetEnable(tbControls.sliderShowPlayerCount, true)
  elseif 3 == nQualitySetting then
    self.pPanel:Slider_SetEnable(tbControls.sliderShowPlayerCount, true)
  end
  local nPlayerCount = Client.QualityModule.GetShowPlayerCount()
  self.pPanel:Label_SetText(tbControls.txtShowPlayerCount, nPlayerCount)
  self.pPanel:Label_SetText(tbControls.txtShowPlayerCountMin, 0)
  self.pPanel:Label_SetText(tbControls.txtShowPlayerCountMax, Client.QualityModule._Setting.MaxSyncPlayerCount)
  self.pPanel:Slider_SetValue(tbControls.sliderShowPlayerCount, nPlayerCount * 100 / Client.QualityModule._Setting.MaxSyncPlayerCount)
end

function tbWnd:UpdateMusicIcon(bValueIsZero)
  self.pPanel:SetActive(tbControls.imgMusic, not bValueIsZero)
  self.pPanel:SetActive(tbControls.imgNoMusic, bValueIsZero)
end

function tbWnd:UpdateSoundIcon(bValueIsZero)
  self.pPanel:SetActive(tbControls.imgSound, not bValueIsZero)
  self.pPanel:SetActive(tbControls.imgNoSound, bValueIsZero)
end

function tbWnd:UpdateSavePowerSetting()
  local nInterval = Ui:GetSavePowerModeInterval()
  if nInterval <= 0 then
    self.pPanel:Toggle_SetChecked(tbControls.TogglePowerSave, false)
    return
  end
  self.pPanel:Toggle_SetChecked(tbControls.TogglePowerSave, true)
end

function tbWnd:ModifySavePowerSetting(isOn)
  if not isOn then
    Ui:SetSavePowerModeInterval(0)
    return
  end
  Ui:SetSavePowerModeInterval()
end

function tbWnd:SetSpeakerVoice(nVol)
  self.pPanel:Slider_SetValue(tbControls.sliderVoice, nVol)
  self.pPanel:SetActive(tbControls.imgVoice, 0 ~= nVol and Client.ShowVoiceSetting)
  self.pPanel:SetActive(tbControls.imgNoVoice, 0 == nVol and Client.ShowVoiceSetting)
end

function tbWnd:OnClickQualityToggle(nQuality)
  if Ui.PlayerPrefs.GetInt("QualitySetting", -1) == nQuality then
    return
  end
  Client:QualityChanged(nQuality)
  Client:Quality_SetMaxSyncPlayerCount()
  self:UpdateShowPlayerCount()
end
