function PlayerLevelSetting:LoadSetting()
  self.tbSetting = Lib:LoadTabFileEx("Setting/Player/PlayerLevel.tab", "ddd", "Level", {
    "Level",
    
    "ExpUpGrade",
    "BaseAwardExp",
    "FightPower",
    "AttributePower"
  })
end

function PlayerLevelSetting:GetLevelBasicExp(nLevel)
  local tbSetting = self.tbSetting[nLevel]
  return tbSetting and tbSetting.BaseAwardExp
end

PlayerLevelSetting:LoadSetting()
