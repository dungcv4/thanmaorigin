local tbWnd = Ui:CreateWindow("UITeamAroundTeam")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_NEARBY_TEAMS,
      self.UpdateList
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  TemplateNode = "node/imgBG/Content/List/Template",
  NotInfo = "node/imgBG/Content/NotInfo"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, self.TemplateNode, nil, false, true)
  self.tbTeams = {}
end

function tbWnd:OnOpen()
  self:UpdateList({})
  TeamMgr:SyncNearbyTeams()
end

function tbWnd:UpdateList(tbTeams)
  self.tbTeams = tbTeams
  self.tbScrollView:Reset()
  self.pPanel:SetActive(self.NotInfo, false)
  if not self.tbTeams or 0 == #self.tbTeams then
    self.pPanel:SetActive(self.NotInfo, true)
    return
  end
  for i, tbTeam in ipairs(self.tbTeams) do
    local tbTeam = self.tbTeams[i]
    self.tbScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtName", i18n.EasyGet("{0}\231\154\132\233\152\159\228\188\141", tbTeam.tbCaptainInfo.szName))
      pPanel:Label_SetText("txtMemberNum", tbTeam.nMemberCount)
      local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
      tbPortraitObjPanel:SetRoleIcon(tbTeam.tbCaptainInfo.nPortrait)
      tbPortraitObjPanel:SetFactionIcon(tbTeam.tbCaptainInfo.nFaction)
      tbPortraitObjPanel:SetLevel(tbTeam.tbCaptainInfo.nLevel)
      Ui:Button_BindEvent(pPanel, "btnApply", self.onClickApplyAddTeamReq, self, i)
    end)
  end
end

function tbWnd:onClickApplyAddTeamReq(idx)
  if idx < 1 or idx > #self.tbTeams then
    return
  end
  local tbTeam = self.tbTeams[idx]
  if not tbTeam then
    return
  end
  TeamMgr:Apply(tbTeam.tbCaptainInfo.nPlayerID)
  self:Close()
end
