local tbWnd = Ui:CreateWindow("UICreateBattleTeam")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTips = "node/imgBG/Content/txtTips",
  txtInput = "node/imgBG/Content/inputTeamName",
  btnOK = "node/imgBG/Content/btnOK"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    self:CreateTeam()
  end
}

function tbWnd:OnCreate()
  self.tbSetting = {
    MultiPvP = {
      nMin = MultiPvP.NAME_LEN_MIN,
      nMax = MultiPvP.NAME_LEN_MAX
    },
    WuLinDaHui = {
      nMin = WuLinDaHui.TeamCfg.NAME_LEN_MIN,
      nMax = WuLinDaHui.TeamCfg.NAME_LEN_MAX
    },
    DingFengPK = {
      nMin = DingFengPK.TeamCfg.NAME_LEN_MIN,
      nMax = DingFengPK.TeamCfg.NAME_LEN_MAX
    }
  }
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen(szMoudle)
  self.szModule = szMoudle
  local tbSetting = self.tbSetting[self.szModule]
  if not tbSetting then
    return
  end
  self.pPanel:Input_SetText(tbControls.txtInput, "")
  local nMin, nMax = tbSetting.nMin, tbSetting.nMax
  if i18n.CurrentLanguageCode() == "zh-CN" or i18n.CurrentLanguageCode() == "zh-TW" then
    nMin = math.floor(nMin / 2)
    nMax = math.floor(nMax / 2)
  end
  self.pPanel:Label_SetText(tbControls.txtTips, i18n.EasyGet("\230\136\152\233\152\159\229\144\141\231\167\176\233\156\128\232\166\129\229\156\168{0}~{1}\228\184\170\229\173\151\228\185\139\229\134\133", nMin, nMax))
end

function tbWnd:CreateTeam()
  local tbSetting = self.tbSetting[self.szModule]
  if not tbSetting then
    return
  end
  local szName = self.pPanel:Input_GetText(tbControls.txtInput) or ""
  if not szName or "" == szName then
    me.CenterMsg(i18n.EasyGet("\232\175\183\232\190\147\229\133\165\233\152\159\228\188\141\229\144\141"))
    return
  end
  local bOk, nMin, nMax = Lib:CheckLength(szName, i18n.CurrentLanguageCode(), tbSetting.nMin, tbSetting.nMax)
  if not bOk then
    me.CenterMsg(i18n.EasyGet("\229\144\141\229\173\151\233\149\191\229\186\166\233\156\128\232\166\129\229\156\168{0}~{1}\228\184\170\229\173\151\229\134\133", nMin, nMax))
    return
  end
  if tonumber(szName) ~= nil then
    me.CenterMsg(i18n.EasyGet("\229\144\141\229\173\151\228\184\141\232\131\189\228\189\191\231\148\168\231\186\175\230\149\176\229\173\151"))
    return
  end
  if CheckNameAvailable(szName) then
    self:SendRequest(szName)
  else
    me.CenterMsg(i18n.EasyGet("\233\152\159\228\188\141\229\144\141\231\167\176\228\184\141\229\144\136\230\179\149"))
  end
  self:Close()
end

function tbWnd:SendRequest(szName)
  if self.szModule == "MultiPvP" then
    RemoteServer.RequestMultiPvP("ChkTeamName", szName)
  elseif self.szModule == "WuLinDaHui" then
    RemoteServer.RequestWuLinDaHui("ChkTeamName", szName)
  elseif self.szModule == "DingFengPK" then
    RemoteServer.RequestDingFengPK("ChkTeamName", szName)
  end
end
