function MPTeam:GetPreliminaryRank(tbTeam)
  return tbTeam[MPTeam.RANK]
end

function MPTeam:ChkPlayerInMapEx(pPlayer)
  if not pPlayer then
    return
  end
  if pPlayer.nMapTemplateId == MultiPvP.Cfg.PRE_MAP_ID or pPlayer.nMapTemplateId == MultiPvP:GetSeasonSettingValue("MapId") then
    return true
  end
  return false
end

function MPTeam:GetWinningRate(tbTeam)
  local nWinCount = MPTeam:GetWinCount(tbTeam)
  local nWinningRate = math.floor(nWinCount / #tbTeam[MPTeam.WIN_COUNT] * 100)
  if #tbTeam[MPTeam.WIN_COUNT] <= 0 then
    nWinningRate = 50
  end
  return nWinningRate
end

function MPTeam:GetWinCount(tbTeam)
  local nWinCount = 0
  for _, nWin in pairs(tbTeam[MPTeam.WIN_COUNT]) do
    if nWin == MPTeam.Win then
      nWinCount = nWinCount + 1
    end
  end
  return nWinCount
end

function MPTeam:GetFailCount(tbTeam)
  local nFailCount = 0
  for _, nFail in pairs(tbTeam[MPTeam.WIN_COUNT]) do
    if nFail == MPTeam.Fail then
      nFailCount = nFailCount + 1
    end
  end
  return nFailCount
end

function MPTeam:GetDrawCount(tbTeam)
  local nDrawCount = 0
  for _, nDraw in pairs(tbTeam[MPTeam.WIN_COUNT]) do
    if nDraw == MPTeam.Draw then
      nDrawCount = nDrawCount + 1
    end
  end
  return nDrawCount
end

function MPTeam:IsCaptain(tbTeam)
  if not me then
    return false
  end
  if not tbTeam then
    return false
  end
  if tbTeam[MPTeam.TEAM_ID] ~= MultiPvP:GetBattleTeamId(me.dwID) then
    return false
  end
  return tbTeam[MPTeam.CAPTAIN] == me.dwID
end

function MPTeam:GetCaptain(tbTeam)
  if not tbTeam then
    return false
  end
  return tbTeam[MPTeam.CAPTAIN]
end
