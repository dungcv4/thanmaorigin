local tbWnd = Ui:CreateWindow("UITeamInvitedMsg")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnCleanAll = "node/imgBG/Content/btnCleanAll",
  imgInviteList = "node/imgBG/Content/imgApplyList",
  InviteTemplate = "node/imgBG/Content/imgApplyList/ApplyList/ApplyTemplate",
  NoInfo = "node/imgBG/Content/NotInfo"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnCleanAll = function(self)
    TeamMgr:ClearInviteList()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, self.InviteTemplate, nil, false, true)
end

function tbWnd:OnOpen()
  self:UpdateInviteList()
end

function tbWnd:UpdateInviteList()
  self.tbScrollView:Reset()
  if TeamMgr:HasTeam() then
    self.pPanel:SetActive(self.imgInviteList, false)
    self.pPanel:SetActive(self.NoInfo, true)
    return
  end
  local tbInviteList = TeamMgr:GetInviteList()
  if 0 == #tbInviteList then
    self.pPanel:SetActive(self.imgInviteList, false)
    self.pPanel:SetActive(self.NoInfo, true)
    return
  end
  self.pPanel:SetActive(self.imgInviteList, true)
  self.pPanel:SetActive(self.NoInfo, false)
  for nId, tbInvite in ipairs(tbInviteList) do
    self.tbScrollView:SetElement(nId, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
      tbPortraitObjPanel:SetRoleIcon(tbInvite.nPortrait)
      tbPortraitObjPanel:SetFactionIcon(tbInvite.nFaction)
      tbPortraitObjPanel:SetLevel(tbInvite.nLevel)
      pPanel:Label_SetText("txtName", tbInvite.szName)
      if tbInvite.szTongName ~= "" then
        pPanel:Label_SetText("txtKin", i18n.EasyGet("\229\184\174\228\188\154:") .. tbInvite.szTongName)
      elseif "" ~= tbInvite.szKinName then
        pPanel:Label_SetText("txtKin", i18n.EasyGet("\229\174\182\230\151\143:") .. tbInvite.szKinName)
      else
        pPanel:Label_SetText("txtKin", i18n.EasyGet("\229\174\182\230\151\143:\230\151\160"))
      end
      pPanel:Label_SetText("txtTarget/txtTargetValue", tbInvite.szTarget and i18n.Parse(tbInvite.szTarget) or i18n.Get(TeamMgr.Terms.ActivityDefault))
      Ui:Button_BindEvent(pPanel, "btnYes", self.OnClickRespond, self, nId, true)
      Ui:Button_BindEvent(pPanel, "btnNo", self.OnClickRespond, self, nId, false)
    end)
  end
end

function tbWnd:OnClickRespond(nId, bAgree)
  local tbInviteList = TeamMgr:GetInviteList()
  if not tbInviteList[nId] then
    return
  end
  TeamMgr:InviteRespond(tbInviteList[nId].nTeamId, bAgree)
end

function tbWnd:OnTeamUpdate(szParam)
  if szParam and "NewInvite" ~= szParam and "new" ~= szParam then
    return
  end
  self:UpdateInviteList()
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_TEAM_UPDATE,
      self.OnTeamUpdate
    },
    {
      EventNotify.emNOTIFY_TEAM_REMOVE_INVITE,
      self.OnTeamUpdate
    },
    {
      EventNotify.emNOTIFY_TEAM_CLEAR_INVITE,
      self.OnTeamUpdate
    }
  }
end
