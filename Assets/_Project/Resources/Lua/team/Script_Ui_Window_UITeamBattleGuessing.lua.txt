local tbWnd = Ui:CreateWindow("UITeamBattleGuessing")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/txtTitle",
  txtName = "node/imgBG/Content/TitleList/txtName",
  txtMember = "node/imgBG/Content/TitleList/txtMember",
  txtFightPower = "node/imgBG/Content/TitleList/txtFightPower",
  btnGuess = "node/imgBG/Content/btnGuess",
  Template = "node/imgBG/Content/TeamList/Viewport/Content/Template",
  imgSelected = "node/imgBG/Content/TeamList/Viewport/Content/Element%d/imgSelected"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnGuess = function(self)
    self:ApplyGuess()
  end
}

function tbWnd:OnCreate()
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template, function(self, nIndex, tbElement)
    self:SwichByIndex(nIndex)
  end)
  self.tbMemberListGroup = {}
end

function tbWnd:OnClose()
  if self.nSelect then
    local szPath = string.format(tbControls.imgSelected, self.nSelect)
    self.pPanel:SetActive(szPath, false)
  end
  self.nSelect = nil
  for _, tbMemberList in pairs(self.tbMemberListGroup) do
    tbMemberList:Clear()
  end
end

function tbWnd:OnOpen(tbSetting, tbDataList, tbGuessCallBack, nId)
  self.tbSetting = tbSetting or {}
  self.tbDataList = tbDataList or {}
  self.tbGuessCallBack = tbGuessCallBack
  self:UpdateTitle()
  self:SetTemplate()
  self:SwichByID(nId)
  self.pPanel:SetActive(tbControls.btnGuess, true)
end

function tbWnd:UpdateTitle()
  self.pPanel:Label_SetText(tbControls.txtTitle, self.tbSetting.szTitle)
  self.pPanel:Label_SetText(tbControls.txtName, self.tbSetting.szTitle1)
  self.pPanel:Label_SetText(tbControls.txtMember, self.tbSetting.szTitle2)
  self.pPanel:Label_SetText(tbControls.txtFightPower, self.tbSetting.szTitle3)
end

function tbWnd:SetTemplate()
  self.ScrollView:Reset()
  local i = 0
  for nId, tbData in pairs(self.tbDataList) do
    i = i + 1
    self.ScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtName", i18n.Parse(tbData.szName))
      local tbMemberList = self.tbMemberListGroup[i] or Ui:NewScrollView(self, tbElement.szKey .. "/MemberList/MemberTemplate")
      tbMemberList:Reset()
      local nAllFightPower = 0
      for j, tbMemberData in ipairs(tbData.tbMember) do
        nAllFightPower = nAllFightPower + tbMemberData.nFight
        tbMemberList:SetElement(j, function(tbMemberElement)
          tbMemberElement.pPanel:Label_SetText("txtMemberName", i18n.Parse(tbMemberData.szPlayerName))
          tbMemberElement.pPanel:Sprite_SetSprite("PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbMemberData.nPortrait))
          PlayerPortraitFrame:SetPortraitFrame(tbMemberElement.pPanel, "PortraitObj/PortraitFrame", tbMemberData.nPortraitFrame)
          tbMemberElement.pPanel:Sprite_SetSprite("PortraitObj/imgFactionIcon", Faction:GetIcon(tbMemberData.nFaction))
          tbMemberElement.pPanel:Label_SetText("PortraitObj/Level/Text", tbMemberData.nLevel)
        end)
      end
      tbElement.pPanel:Label_SetText("txtFightPower", nAllFightPower)
      self.tbMemberListGroup[i] = tbMemberList
    end)
  end
end

function tbWnd:SwichByIndex(nIndex)
  if not nIndex or nIndex <= 0 or nIndex > 8 then
    return
  end
  self:UpdateChoice(nIndex)
end

function tbWnd:SwichByID(nId)
  if not nId or nId < 0 then
    return
  end
  local nIndex = self:IsInList(nId)
  if not nIndex then
    return
  end
  self.ScrollView:ScrollToIndex(nIndex, true)
end

function tbWnd:IsInList(nId)
  for nIndex, tbData in ipairs(self.tbDataList) do
    if tbData.nId == nId then
      return nIndex
    end
  end
end

function tbWnd:UpdateChoice(nIndex)
  local nOldSelect = self.nSelect
  self.nSelect = nIndex
  if nOldSelect then
    local szPath = string.format(tbControls.imgSelected, nOldSelect)
    self.pPanel:SetActive(szPath, false)
  end
  local szPath = string.format(tbControls.imgSelected, self.nSelect)
  self.pPanel:SetActive(szPath, true)
end

function tbWnd:ApplyGuess()
  if not self.nSelect or self.nSelect <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\175\183\233\128\137\230\139\169\231\171\158\231\140\156\231\154\132\229\175\185\232\177\161"))
    return
  end
  local nId = self.tbDataList[self.nSelect] and self.tbDataList[self.nSelect].nId
  if not nId then
    me.CenterMsg(i18n.EasyGet("\232\175\183\233\128\137\230\139\169\231\171\158\231\140\156\231\154\132\229\175\185\232\177\161"))
    return
  end
  if self.tbGuessCallBack then
    local tbCallback = {
      table.unpack(self.tbGuessCallBack)
    }
    table.insert(tbCallback, nId)
    Lib:CallBack(tbCallback)
    self:Close()
  end
end
