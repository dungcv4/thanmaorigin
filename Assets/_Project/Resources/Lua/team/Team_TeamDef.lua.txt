TeamMgr.forbidden_operation_team_in_client = 2

function TeamMgr:CanTeam(nMapTemplateId, szOptType)
  local tbMapSetting = Map:GetMapSetting(nMapTemplateId)
  local bRet = tbMapSetting and tbMapSetting.TeamForbidden ~= self.forbidden_make_and_operation_team
  if not bRet then
    return false
  end
  if self.tbTeamChecker and self.tbTeamChecker[nMapTemplateId] and self.tbTeamChecker[nMapTemplateId].CanTeam and not self.tbTeamChecker[nMapTemplateId]:CanTeam(szOptType) then
    return false
  end
  return true
end

function TeamMgr:IsCanTeam(pPlayer, pTarget)
  if pPlayer.nState == Player.emPLAYER_STATE_ALONE then
    return false, i18n.Get(self.Terms.MapLimitWhenJoin)
  end
  if pTarget.nState == Player.emPLAYER_STATE_ALONE then
    return false, i18n.Get(self.Terms.MapLimitWhenJoin)
  end
  local bPlayerInZone = pPlayer.nState == Player.emPLAYER_STATE_ZONE or pPlayer.nState == Player.emPLAYER_STATE_ZONE_OFFLINE
  local bTargetInZone = pTarget.nState == pTarget.emPLAYER_STATE_ZONE or pTarget.nState == pTarget.emPLAYER_STATE_ZONE_OFFLINE
  if bPlayerInZone ~= bTargetInZone then
    return false, i18n.Get(self.Terms.StateLimitWhenInvite)
  end
  if not TeamMgr:CanTeam(pPlayer.nMapTemplateId) then
    return false, i18n.Get(self.Terms.MapLimitWhenJoin)
  end
  if not TeamMgr:CanTeam(pTarget.nMapTemplateId) then
    return false, i18n.Get(self.Terms.MapLimitWhenJoin)
  end
  if pPlayer.nMapId ~= pTarget.nMapId then
    if Map:IsFubenMap(pPlayer.nMapTemplateId) and not Map:IsIgnoreTeamLimitInFuben(pPlayer.nMapTemplateId) then
      return false, i18n.Get(self.Terms.MapLimitWhenJoin)
    end
    if Map:IsFubenMap(pTarget.nMapTemplateId) and not Map:IsIgnoreTeamLimitInFuben(pTarget.nMapTemplateId) then
      return false, i18n.Get(self.Terms.TargetMapLimitWhenJoin)
    end
  end
  if GlobalHangUp:IsGlobalHangUpMap(pPlayer.nMapTemplateId) then
    local nServerIdx = pPlayer.GetWorldServerIdx()
    local nCaptainServerIdx = pTarget.GetWorldServerIdx()
    if not (nServerIdx and nCaptainServerIdx) or nServerIdx ~= nCaptainServerIdx then
      return false, i18n.Get(self.Terms.NotSameServer)
    end
    if (Darting:IsTeamDarting(pPlayer.dwTeamID) or Darting:IsTeamDarting(pTarget.dwTeamID)) and pPlayer.dwTongId ~= pTarget.dwTongId then
      return false, i18n.EasyGet("\233\152\159\228\188\141\232\191\144\233\149\150\228\184\173\239\188\140\228\184\141\229\144\140\229\184\174\228\188\154\230\136\144\229\145\152\228\184\141\232\131\189\229\138\160\229\133\165")
    end
  end
  return true
end

function TeamMgr:GetCanShowActivityTreeData()
  local tbRet = Lib:CopyTB(self.ACTIVITY_TREE_DATA)
  local tbNeedRemoveBigIndex = {}
  for nBigIndex, tbBigData in ipairs(tbRet) do
    local bNeedAdd = false
    if tbBigData.nActivityId ~= TeamMgr.ACTIVITYID_NO_TARGET then
      local status, bRet = Lib:CallBack(tbBigData.tbCheckShow)
      if status and bRet then
        bNeedAdd = true
      end
    end
    if tbBigData.tbLeaves ~= nil then
      local tbNeedRemoveSmallIndex = {}
      for nSmallIndex, tbSmallData in ipairs(tbBigData.tbLeaves) do
        if tbSmallData.nActivityId == TeamMgr.ACTIVITYID_NO_TARGET then
          table.insert(tbNeedRemoveSmallIndex, nSmallIndex)
        else
          local status, bRet = Lib:CallBack(tbSmallData.tbCheckShow)
          if status and bRet then
            bNeedAdd = true
          else
            table.insert(tbNeedRemoveSmallIndex, nSmallIndex)
          end
        end
      end
      if #tbNeedRemoveSmallIndex >= #tbBigData.tbLeaves then
        tbBigData.tbLeaves = nil
      else
        for i = #tbNeedRemoveSmallIndex, 1, -1 do
          table.remove(tbBigData.tbLeaves, tbNeedRemoveSmallIndex[i])
        end
      end
    end
    if not bNeedAdd then
      table.insert(tbNeedRemoveBigIndex, nBigIndex)
    end
  end
  for i = #tbNeedRemoveBigIndex, 1, -1 do
    table.remove(tbRet, tbNeedRemoveBigIndex[i])
  end
  return tbRet
end
