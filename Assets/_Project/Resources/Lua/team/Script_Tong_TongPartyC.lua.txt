function TongParty:DoJoinTongParty()
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  local bLeadFight = AFK:IsLeadFightRunning()
  if bLeadFight then
    AFK:Stop()
  end
  TongParty:OnJoinTongParty()
end

function TongParty:OnJoinTongParty()
  AFK:Stop()
  local nPosX, nPosY = Map:GetDefaultPos(Tong.Def.nMainMapTemplateId)
  AutoPath:GotoAndCallEx(Tong.Def.nMainMapTemplateId, nPosX, nPosY)
end

function TongParty:SCIsInPartyArea(bIsInPartyArea)
  self.bIsInPartyArea = bIsInPartyArea
  if self.bIsInPartyArea then
    if self.nPartyStatus ~= self.tbPartyStatus.Close then
      Ui:OpenWindow("UIKinFlagDisplay")
      self:CheckFireWorkItem(true)
    else
      self:CheckFireWorkItem(false)
    end
  else
    Ui:CloseWindow("UIKinFlagDisplay")
    Ui:CloseWindow("UIKinFlagAnswer")
  end
end

function TongParty:SCQuestionInfo(tbQuestion)
  local nIndex = tbQuestion.nIndex
  self.nCurQuiz = nIndex
  self.tbQuestions[nIndex] = tbQuestion
  self.nQuestionTime = GetTime() + self.nQuizInterval
  if self.bIsInPartyArea then
    if Ui:WindowVisible("UIKinFlagAnswer") then
      Ui:GetWindow("UIKinFlagAnswer"):NextQuiz()
    elseif self.nPartyStatus == self.tbPartyStatus.Question then
      Ui:OpenWindow("UIKinFlagAnswer")
    end
    if Ui:WindowVisible("UIKinFlagDisplay") then
      Ui:GetWindow("UIKinFlagDisplay"):UpdateQuizTimer()
    end
  end
end

function TongParty:SCPartyStatus(nPartyStatus)
  self.nPartyStatus = nPartyStatus
  if self.nPartyStatus == self.tbPartyStatus.Divination then
    if self:GetAnswerRightNum() > 0 then
      Ui:OpenWindow("UIKinFlagDivination")
    end
  elseif self.nPartyStatus == self.tbPartyStatus.Normal and Ui:WindowVisible("UIKinFlagDivination") then
    Ui:CloseWindow("UIKinFlagDivination")
  end
  if self.nPartyStatus ~= self.tbPartyStatus.Close then
    if Map:IsTongMap(me.nMapTemplateId) and self.bIsInPartyArea then
      if not Ui:WindowVisible("UIKinFlagDisplay") then
        Ui:OpenWindow("UIKinFlagDisplay")
      end
      self:CheckFireWorkItem(true)
    end
  else
    self:CheckFireWorkItem(false)
  end
  if Ui:WindowVisible("UIKinFlagDisplay") then
    Ui:GetWindow("UIKinFlagDisplay"):UpdateView()
  end
  if nPartyStatus ~= self.tbPartyStatus.Question then
    Ui:CloseWindow("UIKinFlagAnswer")
  end
end

function TongParty:CheckFireWorkItem(bStatus)
  if bStatus then
    Event:OnEvent(Event.emEVENT_OBFIREWORK_FIREWORK_QUICKUSING, bStatus)
  else
    local szQuickUseItemWindowName = "UIQuickUseItemOB"
    if Ui:GetWindow(szQuickUseItemWindowName) then
      Ui:CloseWindow(szQuickUseItemWindowName)
    end
  end
end

function TongParty:CSQuestionReq(nSelectIndex)
  local tbQuestion = self.tbQuestions[self.nCurQuiz]
  if not tbQuestion then
    return
  end
  if self.tbDoneAnswer[self.nCurQuiz] then
    return
  end
  RemoteServer.OnTongPartyRequest("OnQuestionReq", self.nCurQuiz, nSelectIndex)
end

  if self.nPartyStatus ~= self.tbPartyStatus.Question then
    me.CenterMsg(i18n.EasyGet("\231\173\148\233\162\152\229\183\178\231\187\147\230\157\159"))
    return

    me.CenterMsg(i18n.EasyGet("\232\175\183\229\155\158\229\136\176\229\184\174\228\188\154\233\162\134\229\156\176\230\136\152\230\151\151\230\151\129\229\143\130\228\184\142\231\173\148\233\162\152!"))

  end

end

function TongParty:GetAnswerRightNum()
  if not self.tbDoneAnswer then
    return 0
  end
  local nCount = 0
  for i, v in pairs(self.tbDoneAnswer) do
    if 1 == v[2] then
      nCount = nCount + 1
    end
  end
  return nCount
end

function TongParty:GetCurQuestion()
  return self.tbQuestions[self.nCurQuiz]
end

  self.nRemainTime = 0
  self.nJoinPlayerCount = 0
  self.nCheerCount = 0
  self.nRealBagAnswer = 0
  self.nDivinationTime = 0
  self.nQuestionTime = 0
  self.bIsInPartyArea = false
  self.tbSelectCount = {}
end
