local tbWnd = Ui:CreateWindow("UITeamBattleResult")
tbWnd.tbControls = {
  MyTemplate = "imgBG/MyGroup/MyTemplate",
  EnemyTemplate = "imgBG/EnemyGroup/EnemyTemplate",
  Victory = "imgBG/Victory",
  Defeat = "imgBG/Defeat",
  Draw = "imgBG/Draw",
  Ready = "imgBG/Ready",
  btnOk = "imgBG/btnOk",
  MyTeam = "imgBG/txtMyTeam",
  EnemyTeam = "imgBG/txtEnemyTeam"
}
tbWnd.tbOnClick = {
  btnOk = function(self)
    self:CloseView()
  end
}

function tbWnd:OnCreate()
  self.pSelfTeamScorllView = Ui:NewScrollView(self, self.MyTemplate, function(_, nIndex, tbElement)
  end, false, false)
  self.pEnemyTeamScorllView = Ui:NewScrollView(self, self.EnemyTemplate, function(_, nIndex, tbElement)
  end, false, false)
end

function tbWnd:OnOpen(tbTeamDataLeft, tbTeamDataRight, nState, nCloseTime)
  self.nState = nState
  local bIsReady = nState == Ui.TeamBattleState.Ready
  local bIsWin = nState == Ui.TeamBattleState.Win
  local bIsLose = nState == Ui.TeamBattleState.Lose
  local bIsDraw = nState == Ui.TeamBattleState.Draw
  self.pPanel:SetActive(self.Ready, bIsReady)
  self.pPanel:SetActive(self.Victory, bIsWin)
  self.pPanel:SetActive(self.Defeat, bIsLose)
  self.pPanel:SetActive(self.Draw, bIsDraw)
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
  end
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * nCloseTime, function()
    self:CloseView()
    return false
  end)
  if tbTeamDataLeft.szTeamName and tbTeamDataLeft.szTeamName ~= "" then
    self.pPanel:SetActive(self.MyTeam, true)
    self.pPanel:Label_SetText(self.MyTeam, i18n.EasyGet("{0} \230\136\152\233\152\159", tbTeamDataLeft.szTeamName))
  else
    self.pPanel:SetActive(self.MyTeam, false)
  end
  if tbTeamDataRight.szTeamName and tbTeamDataRight.szTeamName ~= "" then
    self.pPanel:SetActive(self.EnemyTeam, true)
    self.pPanel:Label_SetText(self.EnemyTeam, i18n.EasyGet("{0} \230\136\152\233\152\159", tbTeamDataRight.szTeamName))
  else
    self.pPanel:SetActive(self.EnemyTeam, false)
  end
  self:InitSelfTeamScorllView(tbTeamDataLeft.tbTeamPlayerInfoDataList)
  self:InitEnemyTeamScorllView(tbTeamDataRight.tbTeamPlayerInfoDataList)
end

function tbWnd:OnDestroyed()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
  end
end

function tbWnd:InitSelfTeamScorllView(tbTeamPlayerInfoDataList)
  self.pSelfTeamScorllView:Clear()
  for i, playerInfoData in pairs(tbTeamPlayerInfoDataList) do
    self.pSelfTeamScorllView:SetElement(i, function(tbElement)
      self:SetPlayerInfo(tbElement, playerInfoData)
    end)
  end
  self.pSelfTeamScorllView:PlayItemGridAnimation()
end

function tbWnd:InitEnemyTeamScorllView(tbTeamPlayerInfoDataList)
  self.pEnemyTeamScorllView:Clear()
  for i, playerInfoData in pairs(tbTeamPlayerInfoDataList) do
    self.pEnemyTeamScorllView:SetElement(i, function(tbElement)
      self:SetPlayerInfo(tbElement, playerInfoData)
    end)
  end
  self.pEnemyTeamScorllView:PlayItemGridAnimation()
end

function tbWnd:SetPlayerInfo(tbElement, playerInfoData)
  tbElement.pPanel:Sprite_SetSprite("PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(playerInfoData.nPortrait))
  PlayerPortraitFrame:SetPortraitFrame(tbElement.pPanel, "PortraitObj/PortraitFrame", playerInfoData.nPortraitFrame)
  tbElement.pPanel:Sprite_SetSprite("PortraitObj/imgFactionIcon", Faction:GetIcon(playerInfoData.nFaction))
  tbElement.pPanel:Label_SetText("PortraitObj/Level/Text", playerInfoData.nLevel)
  tbElement.pPanel:Label_SetText("txtName", playerInfoData.szPlayerName)
  tbElement.pPanel:Animator_SetEnable("Main", true)
  if playerInfoData.nFactionSect then
    tbElement.pPanel:SetActive("txtSect", true)
    tbElement.pPanel:Label_SetText("txtSect", Faction:GetSectName(playerInfoData.nFaction, playerInfoData.nFactionSect))
    tbElement.pPanel:Sprite_SetSprite("txtSect/imgFactionIcon", Faction:GetIcon(playerInfoData.nFaction))
  else
    tbElement.pPanel:SetActive("txtSect", false)
  end
  if self.nState == Ui.TeamBattleState.Ready or self.nState == Ui.TeamBattleState.ReadyNoTitle then
    tbElement.pPanel:SetActive("Ready", true)
    tbElement.pPanel:SetActive("Result", false)
    if playerInfoData.szKinOrTongName and not Lib:IsEmptyStr(playerInfoData.szKinOrTongName) then
      tbElement.pPanel:SetActive("Ready/txtKin", true)
      tbElement.pPanel:Label_SetText("Ready/txtKin", i18n.EasyGet("\229\174\182\230\151\143 {0}", playerInfoData.szKinOrTongName))
    else
      tbElement.pPanel:SetActive("Ready/txtKin", false)
    end
    tbElement.pPanel:Label_SetText("Ready/txtFightPower", i18n.EasyGet("\230\136\152\229\138\155 {0}", playerInfoData.nFight))
  else
    tbElement.pPanel:SetActive("Ready", false)
    tbElement.pPanel:SetActive("Result", true)
    tbElement.pPanel:Label_SetText("Result/txtKillCount", i18n.EasyGet("\229\135\187\232\180\165\230\149\176\239\188\154{0}", playerInfoData.nKill))
    tbElement.pPanel:Label_SetText("Result/txtDamage", i18n.EasyGet("\228\188\164\229\174\179\233\135\143\239\188\154{0}", playerInfoData.nDamage))
  end
end

function tbWnd:CloseView()
  self:Close()
  Ui:DestroyWindow("UITeamBattleResult")
end
