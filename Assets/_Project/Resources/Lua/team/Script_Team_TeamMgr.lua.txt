function TeamMgr:IsCaptain(nPlayerID)
  nPlayerID = nPlayerID or me.dwID
  if type(nPlayerID) ~= "number" then
    nPlayerID = nPlayerID.dwID
  end
  return nPlayerID == _TeamDataCache().nCaptainID
end

function TeamMgr:GetTeamMatesPos()
  local tbPos = {}
  for _, tbMemberData in ipairs(_TeamDataCache().tbTeamMember) do
    if tbMemberData.nMapId == me.nMapId then
      table.insert(tbPos, {
        tbMemberData.nPosX,
        tbMemberData.nPosY
      })
    end
  end
  return tbPos
end

function TeamMgr:GetActivityInfo(nActivityId)
  local tbActivitys = TeamMgr:GetActivityList()
  for _, tbActivity in pairs(tbActivitys) do
    if nActivityId == tbActivity.nActivityId then
      local szOriginName = TeamMgr.TEAM_ACTIVITY_NAME[tbActivity.subtype]()
      local szName = szOriginName
      local szType = tbActivity.szType
      if TeamMgr.TEAM_ACTIVITY_NAME[tbActivity.szType] and tbActivity.szType ~= tbActivity.subtype then
        szName = TeamMgr.TEAM_ACTIVITY_NAME[tbActivity.szType]() .. "\194\183" .. szName
        szType = tbActivity.subtype
      end
      return szName, tbActivity.nMinOpenMember, szOriginName, szType
    end
  end
end

function TeamMgr:OnLogin()
  if not TeamMgr:HasTeam() then
    return
  end
  local nFollowId = self:GetMeFollowTeammate() or AFK:GetFollowTeammateIdCache()
  AFK:SetFollowTeammateIdCache(nil)
  if not nFollowId then
    return
  end
  local tbMemberData = TeamMgr:GetMember(nFollowId)
  if not tbMemberData then
    self:StopFollowMember()
    return
  end
  self:StartFollowMember(nFollowId, tbMemberData.nNpcID)
end

function TeamMgr:OnSyncTeamMemberFightPos(nNpcId, nMapId, nX, nY)
  print("TeamMgr:OnSyncTeamMemberFightPos is invoked!", nNpcId, nMapId, nX, nY)
  if not nNpcId or nNpcId <= 0 then
    return
  end
  if not (nMapId and not (nMapId <= 0) and nX and not (nX <= 0) and nY) or nY <= 0 then
    return
  end
  local tbMember = TeamMgr:GetMemberData(nNpcId)
  if not tbMember then
    return
  end
  local bIsFollow = self:IsFollowTeammate()
  if not bIsFollow then
    return
  end

  if not nFollowId or nFollowId ~= tbMember.nPlayerID then
    return
  end
  AFKFightSetting:UpdateLeaderRepresentFightPointPos(nMapId, nX, nY)
end

function TeamMgr:OnInvited(nTeamID, nInviterID, szInvitorName, nLevel, nFaction, nPortrait, nHonorLevel, szKinName, szTongName, szTarget)
  if not TeamMgr:CanTeam(me.nMapTemplateId) then
    return false
  end
  for idx, tbTeamInfo in ipairs(_TeamDataCache().tbInviterList) do
    if tbTeamInfo.nTeamId == nTeamID or tbTeamInfo.nPlayerId == nInviterID then
      table.remove(_TeamDataCache().tbInviterList, idx)
    end
  end
  local tbInviteData = {
    nTeamId = nTeamID,
    nPlayerId = nInviterID,
    szName = szInvitorName,
    nLevel = nLevel,
    nFaction = nFaction,
    nPortrait = nPortrait,
    nHonorLevel = nHonorLevel,
    szTarget = szTarget,
    nTime = GetTime(),
    szKinName = szKinName,
    szTongName = szTongName
  }
  table.insert(_TeamDataCache().tbInviterList, tbInviteData)
  local szTips = i18n.Get(self.Terms.InviteHint, szInvitorName)
  ChatMgr:OnChannelMessage(ChatMgr.ChannelType.System, 0, ChatMgr.SystemMsgType.Team, "", 0, 0, 0, szTips)
  EventNotify.OnNotify(EventNotify.emNOTIFY_TEAM_UPDATE, "NewInvite")
end

function TeamMgr:InviteRespond(nTeamID, bAgree)
  for i, tbTeamInfo in ipairs(_TeamDataCache().tbInviterList) do
    if nTeamID == tbTeamInfo.nTeamId then
      local nNotifyMsgIndex = tbTeamInfo.nNotifyMsgIndex or 0
      Ui:CancelNotifyMsg(nNotifyMsgIndex)
      RemoteServer.OnTeamRequest("AcceptInvitation", tbTeamInfo.nTeamId, tbTeamInfo.nPlayerId, bAgree)
      table.remove(_TeamDataCache().tbInviterList, i)
      EventNotify.OnNotify(EventNotify.emNOTIFY_TEAM_REMOVE_INVITE)
      break
    end
  end
end

function TeamMgr:RequestLeaderFightPos()

  if not bHasTeam then
    return
  end
  local bIsFollow = self:IsFollowTeammate()

    return
  end

  RemoteServer.RequestLeaderFightPos(nFollowId)
end

function TeamMgr:SyncTeamMemberFightPos(bCheckFightStep)
  local bHasTeam = self:HasTeam()
  if not bHasTeam then
    return
  end
  local tbMeBeFollowed = TeamMgr:GetMeBeFollowed()
  if not tbMeBeFollowed then
    return
  end
  if not AFKFightSetting or not AFKFightSetting.tbLastFightMapPos then
    return
  end
  local bInLeadRunning = AFK:IsLeadFightRunning()
  if not bInLeadRunning then
    return
  end
  if bCheckFightStep then
    local bIsInFightStep = AFK:IsInFightStep()
    if not bIsInFightStep then
      return
    end
  end
  local nMapId = AFKFightSetting.tbLastFightMapPos.nMapId
  local nX = AFKFightSetting.tbLastFightMapPos.x
  local nY = AFKFightSetting.tbLastFightMapPos.y
  if not nMapId or nMapId <= 0 then
    return
  end
  if not nX or nX <= 0 then
    return
  end
  if not nY or nY <= 0 then
    return
  end
  SyncTeamMemberFightPos(nMapId, nX, nY)
end

function TeamMgr:Apply(nTargetPlayerID, bNoFeedback)
  if nTargetPlayerID == me.dwID then
    me.CenterMsg(i18n.Get(self.Terms.ApplyingSelf))
    return
  end
  if me.nLevel < TeamMgr.Cfg.OPEN_LEVEL then
    me.CenterMsg(i18n.Get(self.Terms.MyLevelLimit, TeamMgr.Cfg.OPEN_LEVEL))
    return
  end
  if TeamMgr:HasTeam() then
    me.CenterMsg(i18n.Get(self.Terms.MeInTeam))
    return
  end
  if not TeamMgr:CanTeam(me.nMapTemplateId) then
    me.CenterMsg(i18n.Get(self.Terms.MapLimitWhenJoin))
    return
  end
  RemoteServer.OnTeamRequest("Apply", nTargetPlayerID, bNoFeedback)
end

function TeamMgr:ReqBecomeCaptain()
  if not self:HasTeam() or self:IsCaptain() then
    return
  end
  local tbCaptainData = self:GetCaptainData()
  if not tbCaptainData then
    me.CenterMsg(i18n.Get(self.Terms.NoCaptain))
    return
  end
  if me.dwID == tbCaptainData.nPlayerID then
    return true
  end
  if self:IsNewbie() then
    me.CenterMsg(i18n.Get(self.Terms.NewbieLimit))
    return
  end
  local pNpc = me.GetNpc()
  if pNpc and Map:IsFuben(pNpc.nMapTemplateId) then
    me.CenterMsg(i18n.Get(self.Terms.MapLimitWhenCaptainRequest))
    return
  end
  local tbInfo = _TeamDataCache().tbBecomeCaptainInfo
  if tbInfo then
    local nNow = GetTime()
    if nNow - tbInfo[2] < self.Cfg.nBecomeCaptainTimeSpan then
      me.CenterMsg(i18n.Get(self.Terms.CaptainRequestCD, math.floor(self.Cfg.nBecomeCaptainTimeSpan / 60)))
      return
    end
  end
  local nImityLevel = FriendShip:GetFriendImityLevel(me.dwID, tbCaptainData.nPlayerID)
  if not nImityLevel or nImityLevel < self.Cfg.nBecomeCaptainImityLevel then
    me.CenterMsg(i18n.Get(self.Terms.FriendImityLimit))
    return
  end
  RemoteServer.OnTeamRequest("ReqBecomeCaptain")
  return true
end

function TeamMgr:OnReqBecomeCaptain(nTargetId, nTargetNpcId)
  if not self:HasTeam() or not self:IsCaptain() then
    return
  end
  if not nTargetNpcId then
    return
  end
  local tbMember = self:GetMemberData(nTargetNpcId)
  if not tbMember then
    return
  end
  _TeamDataCache().tbBecomeCaptainInfo = {
    nTargetId,
    GetTime()
  }
  
  local function fnAgree()
    self:ResBecomeCaptain(nTargetId, true)
  end
  
  local function fnDisagree()
    self:ResBecomeCaptain(nTargetId, false)
  end
  
  local function fnTimeOut()
    fnAgree()
    Ui:CloseWindow("UIMessageBox")
  end
  
  local szMsg = i18n.Get(self.Terms.CaptainRequestHint, tbMember.szName)
  me.MsgBox(szMsg, {
    {
      i18n.Get(self.Terms.Agree),
      fnAgree
    },
    {
      i18n.Get(self.Terms.Disagree),
      fnDisagree
    }
  }, nil, self.Cfg.nBecomeCaptainTimerTime, fnTimeOut)
end

end

function TeamMgr:Recall()
  if not self:HasTeam() then
    me.CenterMsg(i18n.Get(self.Terms.NoTeam))
    return
  end
  if not self:IsCaptain() then
    me.CenterMsg(i18n.Get(self.Terms.PermissionDenied))

end

function TeamMgr:OnRecall(nCaptainNpcId, nMemberId)
  local nCaptainID = _TeamDataCache().nCaptainID
  
  local function fnAgree()
    if nCaptainID ~= _TeamDataCache().nCaptainID then
      return
    end
    self:StartFollowMember(nMemberId, nCaptainNpcId)
    RemoteServer.OnTeamRequest("ResponseRecall", nMemberId, 1)
  end
  
  local function fnDisagree()
    RemoteServer.OnTeamRequest("ResponseRecall", nMemberId)
  end
  
  if Client.SceneModule.IsLoading() then
    fnDisagree()
    return
  end
  
  local function fnTimeOut()
    fnAgree()
    Ui:CloseWindow("UIMessageBox")
  end
  
  local tbMember = self:GetMemberData(nCaptainNpcId)
  local szName = tbMember and tbMember.szName or ""
  local szMsg = i18n.Get(self.Terms.RecallHintMember, szName)
  me.MsgBox(szMsg, {
    {
      i18n.Get(self.Terms.Agree),
      fnAgree
    },
    {
      i18n.Get(self.Terms.Disagree),
      fnDisagree
    }
  }, nil, self.Cfg.nRecallTime, fnTimeOut)
end

function TeamMgr:StopFollowMember(bNotSend)
  if not self:HasTeam() then
    return
  end
  local tbTeamData = _TeamDataCache()
  tbTeamData.tbFollow = tbTeamData.tbFollow or {}
  local nMemberId = self:GetMeFollowTeammate()
  if not tbTeamData.tbFollow[me.dwID] then
    return
  end
  tbTeamData.tbFollow[me.dwID] = nil
  AFK:RawStop()
  if me.nFightMode == Player.emFIGHT_STATE.FIGHTING then
    AFK:Start()
  end
  if not bNotSend then
    self:SyncFollowMemberRequest(nMemberId, false)
  end
  EventNotify.OnNotify(EventNotify.emNOTIYF_FOLLOW_STATE_UPDATE)
end

  self:StopFollowMember(true)
end

function TeamMgr:StopFollowMemberAndSyncFollow(nMemberId, bFollow)
  self:StopFollowMember(false)
  self:OnSyncFollow(nMemberId, bFollow)
end

function TeamMgr:StartFollowMember(nMemberId, nNpcId, bNotSync)
  if not self:HasTeam() then
    return
  end
  local tbTeamData = _TeamDataCache()
  tbTeamData.tbFollow = tbTeamData.tbFollow or {}
  if self:IsFollowMe(nMemberId) then
    me.CenterMsg(i18n.Get(self.Terms.FollowLoop))
    return
  end
  AFK:RawStop()
  tbTeamData.tbFollow[me.dwID] = nMemberId
  AFK:Start(AFK.FightTypeTB.emFollowFight, nNpcId)
  if not bNotSync then
    self:SyncFollowMemberRequest(nMemberId, true)
  end
  self:RequestLeaderFightPos()
  EventNotify.OnNotify(EventNotify.emNOTIYF_FOLLOW_STATE_UPDATE)
end

function TeamMgr:SyncFollowMemberRequest(nMemberId, bFollow)
  if not nMemberId then
    return
  end
  RemoteServer.OnOtherTeamRequest("FollowMember", nMemberId, bFollow)
end

end

function TeamMgr:IsFollowMe(nMemberId)
  if 0 == me.dwTeamID then
    return
  end
  local tbMember = self:GetMember(nMemberId)
  if not tbMember then
    return
  end
  local tbTeamData = _TeamDataCache()
  return tbTeamData.tbFollow and tbTeamData.tbFollow[nMemberId] == me.dwID
end

function TeamMgr:OnSyncFollow(nMemberId, bFollow)
  local tbTeamData = _TeamDataCache()
  if bFollow then
    tbTeamData.tbFollow = tbTeamData.tbFollow or {}
    tbTeamData.tbFollow[nMemberId] = me.dwID
  elseif tbTeamData.tbFollow then
    tbTeamData.tbFollow[nMemberId] = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIYF_FOLLOW_STATE_UPDATE, nMemberId, bFollow)
end

function TeamMgr:OnCancelFollow(nMemberId)
  local nTargetId = self:GetFollowTarget(me.dwID)
  if nTargetId == nMemberId then
    local tbTeamData = _TeamDataCache()
    tbTeamData.tbFollow[me.dwID] = nil
    AFK:RawStop()
    AFK:Start()
  end
end

function TeamMgr:GetFollowTarget(nMemberId)
  local tbTeamData = _TeamDataCache()
  return tbTeamData.tbFollow and tbTeamData.tbFollow[nMemberId]
end

function TeamMgr:GetMeBeFollowed()
  local tbFollower = {}
  local tbTeamData = _TeamDataCache()
  for nFollowId, dwId in pairs(tbTeamData.tbFollow or {}) do
    if dwId == me.dwID then
      tbFollower[nFollowId] = 1

function TeamMgr:CheckJoinActivity()
  if not self:IsCaptain() then
    return
  end
  local nActivityId = self:GetCurActivityId()
  if not nActivityId then
    return
  end
  if self:GetMembersCount() < TeamMgr.Cfg.MAX_MEMBER_COUNT then
    return
  end
  local tbActivityList = self.TEAM_ACTIVITY_DATA
  if not tbActivityList then
    return
  end
  local szType, subtype
  for _, tbActivity in pairs(tbActivityList) do
    if tbActivity.nActivityId == nActivityId then
      szType = tbActivity.szType
      subtype = tbActivity.subtype
      break
    end
  end
  if not szType or not subtype then
    return
  end
  local nLeftTimes = TeamMgr:GetActivityLeftTimes(me, szType, subtype)
  if nLeftTimes and nLeftTimes < 1 then
    return
  end
  if not TeamMgr:IsTeamMatchingAct() then
    return
  end
  local nCountDownTime = 15
  local szMsg = i18n.Get(self.Terms.ActivityJoinConfirm)
  
  local function fnAgree()
    self:EnterActivity()
  end
  
  local function fnDisagree()
  end
  
  local function fnClose()
    if nActivityId ~= self:GetCurActivityId() then
      return
    end
    if not self:IsCaptain() then
      return
    end
    if self:GetMembersCount() < TeamMgr.Cfg.MAX_MEMBER_COUNT then
      return
    end
    self:EnterActivity()
  end
  
  me.MsgBox(szMsg, {
    {
      i18n.Get(self.Terms.Agree),
      fnAgree
    },
    {
      i18n.Get(self.Terms.Disagree),
      fnDisagree
    }
  }, nil, nCountDownTime, fnClose)
end

function TeamMgr:OnSyncTeamMemberInfo(...)

function TeamMgr:EnterActivity()
  if not TeamMgr:IsCaptain() then
    me.CenterMsg(i18n.Get(self.Terms.PermissionDenied))
    return false
  end
  if not TeamMgr:HasTeam() then
    me.CenterMsg(i18n.Get(self.Terms.NoTeam))
    return false
  end
  local nActivityId = TeamMgr:GetCurActivityId()
  if not nActivityId then
    me.CenterMsg(i18n.Get(self.Terms.NoActivity))
    return false
  end
  local _, _, _, szActivityType = TeamMgr:GetCurActivityInfo()
  if self:IsGuaJiActivity(szActivityType) then
    self:OpenFieldPracticeWindow()
    return true
  end
  
  local function fnEnter()
    RemoteServer.OnTeamUpRequest("EnterActivity", nActivityId)
    TeamMgr:CancelTeamQuickMatch()
  end
  
  local function fnAgree()
    Ui:CloseWindow("UIMessageBox")
    local bRet, szMsg = ActivityCheck:CheckCanEnterActivityEx(me, nil, fnEnter, false, false, true, nil, {"TeamPanel", "UITeam"}, false)
    if not bRet then
      me.CenterMsg(szMsg)
      return true
    end
    return true
  end

function TeamMgr:GetActivityTargetTreeData()
  local tbTree = {}
  local tbItems = TeamMgr:GetActivityList()
  local tbActivityClass = {}
  local tbSubTypes = {}
  table.insert(tbTree, {
    tbData = {
      szName = i18n.Get(self.Terms.ActivityDefault)
    }
  })
  local nPos = 0
  for _, tbItem in ipairs(tbItems) do
    if tbItem.szType == tbItem.subtype then
      table.insert(tbTree, {tbData = tbItem})
      nPos = nPos + 1
    else
      if not tbActivityClass[tbItem.szType] then
        table.insert(tbTree, {
          tbData = {
            szType = tbItem.szType,
            subtype = tbItem.szType
          },
          bHasLeaves = true
        })
        nPos = nPos + 1
        tbActivityClass[tbItem.szType] = {
          tbLeaves = {},
          nPos = nPos + 1
        }
        table.insert(tbSubTypes, tbActivityClass[tbItem.szType])
      end
      if tbItem.nActivityId == self.nCurActivityId then
        tbActivityClass[tbItem.szType].bDown = true
      end
      table.insert(tbActivityClass[tbItem.szType].tbLeaves, tbItem)
    end
  end
  table.sort(tbSubTypes, function(a, b)
    return a.nPos > b.nPos
  end)
  for _, subTypes in ipairs(tbSubTypes) do
    for k, tbItem in ipairs(subTypes.tbLeaves) do
      table.insert(tbTree, subTypes.nPos + k, tbItem)
    end
  end
  return tbTree
end

function TeamMgr:_CheckSpecialFubenLeave(nMapTemplateID, nLastMapTemplateId)
  if not me.szFubenType then
    return
  end
  if Map:IsFuben(nMapTemplateID) then
    return
  end
  if not Map:IsFuben(nLastMapTemplateId) then
    return
  end
  local szFubenType = me.szFubenType
  me.szFubenType = nil
  if not self:HasTeam() then

function TeamMgr:OnSpecialFubenLeave(bHasTimes, szFubenType)
  if bHasTimes then
    local nCountDownTime = 5
    local szMsg = i18n.Get(self.Terms.ActivityJoinConfirm3, nCountDownTime)
    
    local function fnAgree()
      self:EnterActivity()
    end
    
    local function fnDisagree()
    end
    
    local function fnClose()
      self:EnterActivity()
    end
    
    me.MsgBox(szMsg, {
      {
        i18n.Get(self.Terms.Agree),
        fnAgree
      },
      {
        i18n.Get(self.Terms.Disagree),
        fnDisagree
      }
    }, nil, nCountDownTime, fnClose)
  else
    local szMsg
    local tbItem = self:_CheckActivityHasLeftTimes(szFubenType)
    if tbItem then
      local szTypeName = tbItem.szName or TeamMgr.TEAM_ACTIVITY_NAME[tbItem.szType]() or tbItem.szType
      szMsg = i18n.Get(self.Terms.ActivityChangedConfirm, szTypeName)
      
      local function fnAgree()
        if self:SetTeamActivity(tbItem.nActivityId) then
          RemoteServer.OnTeamUpRequest("EnterActivity", tbItem.nActivityId)
        end
      end
      
      local function fnDisagree()
      end
      
      me.MsgBox(szMsg, {
        {
          i18n.Get(self.Terms.Agree),
          fnAgree
        },
        {
          i18n.Get(self.Terms.Disagree),
          fnDisagree
        }
      })
    else
      szMsg = i18n.Get(self.Terms.ActivityChangedConfirm2)
      
      local function fnAgree()
        local data = self:GetActivityTargetTreeData()
        Ui:OpenWindow("TeamTargetPanel", data)
      end
      
      local function fnDisagree()
      end
      
      me.MsgBox(szMsg, {
        {
          i18n.Get(self.Terms.Agree),
          fnAgree
        },
        {
          i18n.Get(self.Terms.Disagree),
          fnDisagree
        }
      })
    end
  end
end

function TeamMgr:_CheckActivityHasLeftTimes(szOrgType)
  local tbTarget = {}
  local tbNoNeedCheck = {}
  local bHas
  local tbCanJoin = {}
  local tbItems = TeamMgr:GetActivityList()
  for _, szType in pairs(tbTarget) do
    if bHas then
      break
    end
    if szType ~= szOrgType then
      for _, tbItem in ipairs(tbItems) do
        if tbItem.szType == szType then
          local nLeftTimes = TeamMgr:GetActivityLeftTimes(me, szType)
          if nLeftTimes and nLeftTimes > 0 then
            bHas = true
            if tbNoNeedCheck[szType] then
              return tbItem
            end
            table.insert(tbCanJoin, tbItem)
          end
        end
      end
    end
  end
  if not next(tbCanJoin) then
    return
  end
  return tbCanJoin[#tbCanJoin]
end

function TeamMgr:ChangeTimerState(szState)
  if not self.nTimer then
    self.nTimer = Timer:Register(Env.GAME_FPS, self.TimerCheck, self)
  end
  self.nRecordTime = GetTime()
  self.szState = szState
end

function TeamMgr:StopTimer()
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
  self:TimerClear()
end

function TeamMgr:TimerClear()
  self.nTimer = nil
  self.nRecordTime = nil
  self.szState = nil
end

function TeamMgr:TimerCheck()
  if self.szState and self.tbTimerCallBackList[self.szState] then
    if self.nRecordTime < GetTime() - self.tbTimerCallBackList[self.szState][1] then
      local bContinue = self[self.tbTimerCallBackList[self.szState][2]](self)
      if not bContinue then
        self:TimerClear()
      end
      return bContinue
    end
  else
    self:TimerClear()
    return false
  end
  return true
end

function TeamMgr:QuickMatchTimeOut()
  if self:IsMatchingAct() and not self:HasTeam() then
    local tb = TeamMgr:GetMatchingActivityIds()
    assert(tb)
    local nActivityIds, _ = next(tb)
    TeamMgr:CreateOnePersonTeam(nActivityIds, true)
    return false
  else
    return false
  end
end

function TeamMgr:DungeonTimeOut()
  if self:HasTeam() and self:IsCaptain() and self:GetCurActivityId() then
    local _, _, _, szActivityType = TeamMgr:GetCurActivityInfo()
    if TeamMgr.CAN_ENTER_WITH_ASSIST_NPC[szActivityType] then
      self:EnterActivity()
    end
  end
  return false
end

function TeamMgr:StartCheckTargetTimer()
  if self.nCheckTargetTimer then
    Timer:Close(self.nCheckTargetTimer)
    self.nCheckTargetTimer = nil
  end
  if self:GetCurActivityId() == self.ACTIVITYID_NO_TARGET then
    return
  end
  self.nCheckTargetTimer = Timer:Register(Env.GAME_FPS * self.nCheckTargetCD, self.OnTimerCheckTarget, self)
  self.nRecordStateTime = GetTime()
  self.nCurPlayerExp = me.GetExp()
end

function TeamMgr:OnTimerCheckTarget()
  if not self.nRecordStateTime then
    self.nCheckTargetTimer = nil
    return false
  end
  if self:GetCurActivityId() == self.ACTIVITY_ID.XiuLian and me.GetExp() ~= self.nCurPlayerExp then
    self.nRecordStateTime = GetTime()
  end
  if GetTime() - self.nRecordStateTime > self.nChangeTargetCD then
    self:SetTeamActivity(self.ACTIVITYID_NO_TARGET)
    self.nCheckTargetTimer = nil
    return false
  end
  return true
end

function TeamMgr:EnterActivitySuccess(nActivityId)
  if self.nCheckTargetTimer then
    Timer:Close(self.nCheckTargetTimer)
  end
  self.nCheckTargetTimer = nil
end
