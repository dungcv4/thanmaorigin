local bCannotChangeToggleNotify = true
tbWnd.tbOnToggleClick = {
  toggleAutoApply = function(self, isOn)
    local bIsCaptain = TeamMgr:IsCaptain()
    if not bIsCaptain then
      if bCannotChangeToggleNotify then
        me.CenterMsg(i18n.Get(TeamMgr.Terms.PermissionDenied))
        bCannotChangeToggleNotify = false
      end
      self.pPanel:Toggle_SetChecked(self.toggleAutoApply, TeamMgr:IsAutoAgree())
      bCannotChangeToggleNotify = true
      return
    end
    TeamMgr:SetAutoAgree(isOn)
  end
}

function tbWnd:OnOpen(nActivityId)
  TeamMgr:Ask4AllKindTeams()
  self.nCurActivityId = nActivityId or self.ACTIVITYID_DEFAULT
  self:UpdateApplyList()
  self:UpdateInviteList()
  self:UpdateActivityList()
  self:Update(true)
  local nAutoSelectBtnIndex = 0 == me.dwTeamID and 1 or 2
  if nActivityId then
    for nIndex, tbValue in ipairs(self.uiTree.tbBigData) do
      if nActivityId == tbValue.tbInfo.nActivityId then
        nAutoSelectBtnIndex = nIndex
      end
    end
  end
  self.uiTree:SelectNode(nAutoSelectBtnIndex, 0)
  self.nTimer_RefreshAllTeam = Timer:Register(Env.GAME_FPS * 60, self.__timer_RefreshAllTeam, self)
  self.pPanel:SetActive(self.NotApply, false)
  self.pPanel:SetActive(self.NotInvite, false)
  self:UpdateTeamQuickMatchStatus()
end

function tbWnd:OnTeamUpdate(szParam)
  if not Ui:WindowVisible("UITeam") then
    return
  end
  self.nCurActivityId = self.nCurActivityId or self.ACTIVITYID_DEFAULT
  self.nCurActivityId = "new" == szParam and self.ACTIVITYID_DEFAULT or self.nCurActivityId
  self:UpdateApplyList()
  self:UpdateInviteList()
  self:Update(false)
  if "new" == szParam then
    self.uiTree:SelectNode(2, 0)
  end
  if "quite" == szParam and self.nCurActivityId > 0 then
    self:UpdateActivityTeamList(self.nCurActivityId)
  end
  if "MemberChanged" == szParam or "quite" == szParam or "new" ~= szParam then
  end
  self:UpdateTeamQuickMatchStatus()
end

function tbWnd:OnLateUpdate()
  if self.bNeedQuickTeamUpdate then
    self:DoQuickTeamUpdate(self.szQuickTeamParam)
    self.bNeedQuickTeamUpdate = false
  end
  if self.bNeedUpdateAllTeamList then
    self:DoUpdateAllTeamList(self.bForceUpdateAllTeam)
    self.bNeedUpdateAllTeamList = false
  end
  if self.bNeedUpdate then
    self:DoUpdate(self.bForceUpdate)
    self.bNeedUpdate = false
  end
end

function tbWnd:OnQuickTeamUpdate(szParam)
  self.szQuickTeamParam = szParam
  self.bNeedQuickTeamUpdate = true
end

function tbWnd:DoQuickTeamUpdate(szParam)
  if not Ui:WindowVisible("UITeam") then
    return
  end
  self.nCurActivityId = self.nCurActivityId or self.ACTIVITYID_DEFAULT
  self:UpdateApplyList()
  self:UpdateInviteList()
  if "ActivityList" == szParam then
    self:UpdateActivityList()
  elseif "AllTeams" == szParam then
    self:UpdateAllTeamList()
  end
  self:Update(false)
end

function tbWnd:DoUpdateAllTeamList(bForce)

  self.pPanel:SetActive(self.panelTeamList, false)
  self.pPanel:SetActive(self.panelAllTeamList, true)
  local bMatching = TeamMgr:IsMatchingAct()
  local szBtnName = bMatching and i18n.Get(TeamMgr.Terms.MatchStop) or i18n.Get(TeamMgr.Terms.MatchStart)

function tbWnd:GetMatchTargetTreeData(bAutoMatch)
  local tbTreeData = TeamMgr:GetCanShowActivityTreeData()
  local nCurActivityId = TeamMgr:GetCurActivityId() or self.ACTIVITYID_NONE
  
  local function fnGetInsertData(tbInnerTreeData)
    local nLeftTimes = TeamMgr:GetActivityLeftTimes(me, tbInnerTreeData.szType, tbInnerTreeData.subtype)
    local szExternMsg = nLeftTimes and i18n.Get(TeamMgr.Terms.TicketLeft, nLeftTimes) or ""
    local szTypeText = tbInnerTreeData.subtype or tbInnerTreeData.szType
    local tbBigData = {
      key = tbInnerTreeData.nActivityId,
      szText = TeamMgr.TEAM_ACTIVITY_NAME[szTypeText](),
      szExternMsg = szExternMsg,
      bCheck = nCurActivityId == tbInnerTreeData.nActivityId
    }
    return tbBigData
  end
  
  local tbData = {}
  for _, tbBigTreeData in ipairs(tbTreeData) do
    local tbEachBigData = fnGetInsertData(tbBigTreeData)
    local tbSmallData = {}
    if tbBigTreeData.tbLeaves and Lib:CountTB(tbBigTreeData.tbLeaves) > 0 then
      for _, tbSmallTreeData in ipairs(tbBigTreeData.tbLeaves) do
        local tbEachSmallData = fnGetInsertData(tbSmallTreeData)
        table.insert(tbSmallData, tbEachSmallData)
      end
      tbEachBigData.tbLeaves = tbSmallData
    end
    table.insert(tbData, tbEachBigData)
  end
  if not bAutoMatch then
    table.insert(tbData, 1, {
      key = self.ACTIVITYID_NONE,
      szText = i18n.Get(TeamMgr.Terms.ActivityDefault),
      szExternMsg = "",
      bCheck = nCurActivityId == self.ACTIVITYID_NONE
    })
  end
  return tbData
end

function tbWnd:GetActivityTreeData()
  local tbTreeData = TeamMgr:GetCanShowActivityTreeData()
  local tbBigData = {
    {
      nType = "allTeam",
      szText = i18n.Get(TeamMgr.Terms.AllTeam),
      tbInfo = {
        nActivityId = self.ACTIVITYID_ALLTEAM
      }
    },
    {
      nType = "myTeam",
      szText = i18n.Get(TeamMgr.Terms.MyTeam),
      tbInfo = {
        nActivityId = self.ACTIVITYID_DEFAULT
      }
    }
  }
  local tbSmallData = {}
  for _, tbBigInfo in ipairs(tbTreeData) do
    local szBigType = tbBigInfo.szType
    table.insert(tbBigData, {
      nType = szBigType,
      szText = TeamMgr.TEAM_ACTIVITY_NAME[tbBigInfo.szType](),
      tbInfo = {
        nActivityId = tbBigInfo.nActivityId,
        szType = szBigType
      }
    })
    if tbBigInfo.tbLeaves then
      for _, tbSmallInfo in ipairs(tbBigInfo.tbLeaves) do
        table.insert(tbSmallData, {
          nType = szBigType,
          szText = TeamMgr.TEAM_ACTIVITY_NAME[tbSmallInfo.subtype](),
          tbInfo = {
            nActivityId = tbSmallInfo.nActivityId,
            szType = szBigType,
            subtype = tbSmallInfo.subtype
          }
        })
      end
    end
  end
  return tbBigData, tbSmallData
end

function tbWnd:UpdateCurState()

  local bCaptain = bHasTeam and TeamMgr:IsCaptain(me.dwID) or false
  local nActivityId = TeamMgr:GetCurActivityId()
  local bHasTarget = nActivityId and nActivityId > 0
  if not bHasTeam then
    self.nCurState = self.State_NoTeam_NoTarget
  elseif bHasTarget then
    if bCaptain then
      self.nCurState = self.State_Team_Target_Captain
    else
      self.nCurState = self.State_Team_Target_Member
    end
  elseif bCaptain then
    self.nCurState = self.State_Team_NoTarget_Captain
  else
    self.nCurState = self.State_Team_NoTarget_Member
  end
end

function tbWnd:ChangeActivity()
  if not self.nCurActivityId then
    me.CenterMsg(i18n.Get(TeamMgr.Terms.NoActivity))
    return
  end
  if self.nCurActivityId == TeamMgr:GetCurActivityId() then
    self:EnterActivity()
    return
  end
  if not TeamMgr:HasTeam() then
    local function fnAgree()
      TeamMgr:CreateOnePersonTeam(self.nCurActivityId)
    end
    
    local function fnCancel()
    end
    
    Ui:OpenWindow("UIMessageBoxBig", i18n.Get(TeamMgr.Terms.CreateConfirm), {fnAgree, fnCancel})
    return
  end
  
  local function fnAgree()
    TeamMgr:SetTeamActivity(self.nCurActivityId)
  end
  
  local function fnCancel()
  end
  
  local szCurActivityName = TeamMgr:GetCurActivityInfo()
  if not szCurActivityName then
    fnAgree()
    return
  end
  local szTargetActivityName = i18n.Parse(TeamMgr:GetActivityInfo(self.nCurActivityId))
  local szMsg = i18n.Get(TeamMgr.Terms.ActivityChangeConfirm, szTargetActivityName)
  Ui:OpenWindow("UIMessageBoxBig", szMsg, {fnAgree, fnCancel})
end

function tbWnd:SelectBigAndSmallButton(szType, szSubType)
  local nSelectButtonIndex
  for nIndex, tbValue in ipairs(self.uiTree.tbBigData) do
    if tbValue.tbInfo.szType == szType then
      nSelectButtonIndex = nIndex
      break
    end
  end
  if nil ~= nSelectButtonIndex then
    local nSelectSubButtonIndex
    local nSubIndex = 1
    for _, tbSubValue in ipairs(self.uiTree.tbSmallData) do
      if tbSubValue.tbInfo.szType == szType then
        if tbSubValue.tbInfo.subtype == szSubType then
          nSelectSubButtonIndex = nSubIndex
          break
        end
        nSubIndex = nSubIndex + 1
      end
    end
    if nil ~= nSelectSubButtonIndex then
      self.uiTree:SelectNode(nSelectButtonIndex, nSelectSubButtonIndex)
      return true
    else
      return false
    end
  else
    return false
  end
end

function tbWnd:OnClickStartTeamMatch()
  TeamMgr:TeamQuickMatch()
end

function tbWnd:OnClickStopTeamMatch()
  TeamMgr:CancelTeamQuickMatch()
end

function tbWnd:OnTeamQuickMatch()
  self:UpdateTeamQuickMatchStatus()
end

function tbWnd:UpdateTeamQuickMatchStatus()
  local bIsCaptain = TeamMgr:HasTeam() and TeamMgr:IsCaptain(me.dwID)
  local nCurActivityId = TeamMgr:GetCurActivityId()
  if not bIsCaptain or self.nCurActivityId ~= nCurActivityId and self.nCurActivityId ~= self.ACTIVITYID_DEFAULT or nCurActivityId == TeamMgr.ACTIVITYID_NO_TARGET or TeamMgr.NOT_TEAM_MATCH[nCurActivityId] then
    bIsCaptain = false
    self.pPanel:SetActive(self.BtnStartMatch_Detail, bIsCaptain)
    self.pPanel:SetActive(self.BtnStopMatch_Detail, bIsCaptain)
    self.pPanel:SetActive(self.BtnStartMatch_TeamList, bIsCaptain)
    self.pPanel:SetActive(self.BtnStopMatch_TeamList, bIsCaptain)
  else
    local bMatching = TeamMgr:IsTeamMatchingAct()
    self.pPanel:SetActive(self.BtnStartMatch_Detail, not bMatching)
    self.pPanel:SetActive(self.BtnStopMatch_Detail, bMatching)
    self.pPanel:SetActive(self.BtnStartMatch_TeamList, not bMatching)
    self.pPanel:SetActive(self.BtnStopMatch_TeamList, bMatching)
  end
  self:UpdateTeamMatchTxtLabelStatus()
end

function tbWnd:UpdateTeamMatchTxtLabelStatus()
  local nActivityId = TeamMgr:GetCurActivityId()
  local bMatching = TeamMgr:IsTeamMatchingAct()
  local tbBigKeyList, tbSmallKeyList = {}, {}
  for _, tbInfo in ipairs(self.uiTree.tbButtonList) do
    if 1 == tbInfo[2] then
      tbBigKeyList[tbInfo[3]] = tbInfo[1] .. "/txtMatch"
    elseif 2 == tbInfo[2] then
      tbSmallKeyList[tbInfo[4]] = tbInfo[1] .. "/txtMatch"
    end
  end
  for nIndex, tbData in ipairs(self.uiTree.tbBigData) do
    if tbBigKeyList[nIndex] then
      self.uiTree.pPanel:SetActive(tbBigKeyList[nIndex], tbData.tbInfo.nActivityId == nActivityId and bMatching)
    end
  end
  local szBigType
  for nIndex, tbData in ipairs(self.uiTree.tbSmallData) do
    if tbSmallKeyList[nIndex] then
      self.uiTree.pPanel:SetActive(tbSmallKeyList[nIndex], tbData.tbInfo.nActivityId == nActivityId and bMatching)
      if tbData.tbInfo.nActivityId == nActivityId then
        szBigType = tbData.tbInfo.szType
      end
    end
  end
  local bStatus
  for nIndex, tbData in ipairs(self.uiTree.tbBigData) do
    if tbBigKeyList[nIndex] then
      bStatus = false
      if bMatching and (tbData.tbInfo.nActivityId == nActivityId or szBigType == tbData.nType) then
        bStatus = true
      end
      self.uiTree.pPanel:SetActive(tbBigKeyList[nIndex], bStatus)
    end
  end
end

function tbWnd:OnChangeActivityTarget()
  self:UpdateTeamQuickMatchStatus()
end

function tbWnd:OnQuitTeam()
  self:UpdateTeamQuickMatchStatus()
end
