DingFengPK.DingFengTeam = DingFengPK.DingFengTeam or {}
local DingFengTeam = DingFengPK.DingFengTeam

function DingFengTeam:GetWinningRate(tbTeamData)
  local nWinCount = DingFengTeam:GetWinCount(tbTeamData)
  local nWinningRate = math.floor(nWinCount / #tbTeamData[DingFengPK.WIN_COUNT] * 100)
  if #tbTeamData[DingFengPK.WIN_COUNT] <= 0 then
    nWinningRate = 50
  end
  return nWinningRate
end

function DingFengTeam:GetWinCount(tbTeamData)
  local nWinCount = 0
  for _, nWin in pairs(tbTeamData[DingFengPK.WIN_COUNT]) do
    if nWin == DingFengPK.Win then
      nWinCount = nWinCount + 1
    end
  end
  return nWinCount
end

function DingFengTeam:GetFailCount(tbTeamData)
  local nFailCount = 0
  for _, nFail in pairs(tbTeamData[DingFengPK.WIN_COUNT]) do
    if nFail == DingFengPK.Fail then
      nFailCount = nFailCount + 1
    end
  end
  return nFailCount
end

function DingFengTeam:GetDrawCount(tbTeamData)
  local nDrawCount = 0
  for _, nDraw in pairs(tbTeamData[DingFengPK.WIN_COUNT]) do
    if nDraw == DingFengPK.Draw then
      nDrawCount = nDrawCount + 1
    end
  end
  return nDrawCount
end

function DingFengTeam:BattleResultStat(tbTeamData)
  local nWin = 0
  local nLose = 0
  local nDraw = 0
  for _, nResult in pairs(tbTeamData[DingFengPK.WIN_COUNT]) do
    if nResult == DingFengPK.Win then
      nWin = nWin + 1
    elseif nResult == DingFengPK.Fail then
      nLose = nLose + 1
    elseif nResult == DingFengPK.Draw then
      nDraw = nDraw + 1
    end
  end
  return nWin, nLose, nDraw, tbTeamData[DingFengPK.TOTAL_JOIN]
end

function DingFengTeam:IsWeekOverCount(tbTeamData)
  return tbTeamData[DingFengPK.WEEK_JOIN] >= DingFengPK.PreCfg.WEEK_COUNT
end

function DingFengTeam:IsTotalJoinEnough(tbTeamData)
  return tbTeamData[DingFengPK.TOTAL_JOIN] >= DingFengPK.PreCfg.TOTAL_COUNT
end

function DingFengTeam:IsPreTotalJoinEnough(tbTeamData)
  return tbTeamData[DingFengPK.TOTAL_JOIN] >= DingFengPK.PreCfg.PRE_TOTAL_COUNT
end

function DingFengTeam:IsDailyJoinEnough(tbTeamData)
  return tbTeamData[DingFengPK.DAY_JOIN] >= DingFengPK.PreCfg.DAY_COUNT
end

function DingFengTeam:GetName(tbTeamData)
  return tbTeamData[DingFengPK.TEAM_NAME]
end
