local tbWnd = Ui:CreateWindow("UITeamBattleInfo")
tbWnd.tbControls = {
  MyKill = "imgBG/Top/MyScore/MyKill",
  MyOutputDamage = "imgBG/Top/MyScore/MyOutputDamage",
  EnemyKill = "imgBG/Top/EnemyScore/EnemyKill",
  EnemyOutputDamage = "imgBG/Top/EnemyScore/EnemyOutputDamage",
  MyResult = "imgBG/Top/MyResult",
  EnemyResult = "imgBG/Top/EnemyResult"
}
tbWnd.ROUND_COUNT = 3
tbWnd.tbResult = {
  [0] = "Empty",
  [1] = "Win",
  [2] = "Lose"
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(tbTeamResultTable, nSelfTeamId)
  self:RefreshView(tbTeamResultTable, nSelfTeamId)
end

function tbWnd:RefreshView(tbTeamResultTable, nSelfTeamId)
  local tbSelfTeamResultData = tbTeamResultTable[nSelfTeamId]
  self.pPanel:Label_SetText(self.MyKill, i18n.EasyGet("\229\135\187\232\180\165\230\149\176 {0}", tbSelfTeamResultData.nKillCount))
  self.pPanel:Label_SetText(self.MyOutputDamage, i18n.EasyGet("\228\188\164\229\174\179\233\135\143 {0}", tbSelfTeamResultData.nDamageCount))
  if tbSelfTeamResultData.tbGameRecord then
    self.pPanel:SetActive(self.MyResult, true)
    for nRound, nValue in ipairs(tbSelfTeamResultData.tbGameRecord) do
      for nResult, szResult in pairs(self.tbResult) do
        self.pPanel:SetActive(string.format("%s/Round%d/%s", self.MyResult, nRound, szResult), nValue == nResult)
      end
    end
  else
    self.pPanel:SetActive(self.MyResult, false)
  end
  for nTeamId, tbTeamResultData in pairs(tbTeamResultTable) do
    if nSelfTeamId ~= nTeamId then
      self.pPanel:Label_SetText(self.EnemyKill, i18n.EasyGet("\229\135\187\232\180\165\230\149\176 {0}", tbTeamResultData.nKillCount))
      self.pPanel:Label_SetText(self.EnemyOutputDamage, i18n.EasyGet("\228\188\164\229\174\179\233\135\143 {0}", tbTeamResultData.nDamageCount))
      if tbTeamResultData.tbGameRecord then
        self.pPanel:SetActive(self.EnemyResult, true)
        for nRound, nValue in ipairs(tbTeamResultData.tbGameRecord) do
          for nResult, szResult in pairs(self.tbResult) do
            self.pPanel:SetActive(string.format("%s/Round%d/%s", self.EnemyResult, nRound, szResult), nValue == nResult)
          end
        end
      else
        self.pPanel:SetActive(self.EnemyResult, false)
      end
    end
  end
end

function tbWnd:OnDestroyed()
end
