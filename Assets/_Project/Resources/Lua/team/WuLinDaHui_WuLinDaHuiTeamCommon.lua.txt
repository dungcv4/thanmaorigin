function WldhTeam:ChkPlayerInMapEx(pPlayer)
  if not pPlayer then
    return
  end
  if pPlayer.nMapTemplateId == WuLinDaHui.Cfg.READY_MAP_ID or pPlayer.nMapTemplateId == WuLinDaHui.Cfg.FIGHT_MAP_ID then
    return true
  end
  return false
end

function WldhTeam:GetWinningRate(tbTeam)
  local nWinCount = WldhTeam:GetWinCount(tbTeam)
  local nWinningRate = math.floor(nWinCount / #tbTeam[WldhTeam.WIN_COUNT] * 100)
  if #tbTeam[WldhTeam.WIN_COUNT] <= 0 then
    nWinningRate = 50
  end
  return nWinningRate
end

function WldhTeam:GetWinCount(tbTeam)
  local nWinCount = 0
  for _, nWin in pairs(tbTeam[WldhTeam.WIN_COUNT]) do
    if nWin == WldhTeam.Win then
      nWinCount = nWinCount + 1
    end
  end
  return nWinCount
end

function WldhTeam:GetFailCount(tbTeam)
  local nFailCount = 0
  for _, nFail in pairs(tbTeam[WldhTeam.WIN_COUNT]) do
    if nFail == WldhTeam.Fail then
      nFailCount = nFailCount + 1
    end
  end
  return nFailCount
end

function WldhTeam:GetDrawCount(tbTeam)
  local nDrawCount = 0
  for _, nDraw in pairs(tbTeam[WldhTeam.WIN_COUNT]) do
    if nDraw == WldhTeam.Draw then
      nDrawCount = nDrawCount + 1
    end
  end
  return nDrawCount
end

function WldhTeam:BattleResultStat(tbTeam)
  local nWin = 0
  local nLose = 0
  local nDraw = 0
  for _, nResult in pairs(tbTeam[WldhTeam.WIN_COUNT]) do
    if nResult == WldhTeam.Win then
      nWin = nWin + 1
    elseif nResult == WldhTeam.Fail then
      nLose = nLose + 1
    elseif nResult == WldhTeam.Draw then
      nDraw = nDraw + 1
    end
  end
  return nWin, nLose, nDraw, tbTeam[WldhTeam.TOTAL_JOIN]
end

function WldhTeam:IsWeekOverCount(tbTeam)
  return tbTeam[WldhTeam.WEEK_JOIN] >= WuLinDaHui.PreCfg.WEEK_COUNT
end

function WldhTeam:IsTotalJoinEnough(tbTeam)
  return tbTeam[WldhTeam.TOTAL_JOIN] >= WuLinDaHui.PreCfg.TOTAL_COUNT
end

function WldhTeam:IsDailyJoinEnough(tbTeam)
  return tbTeam[WldhTeam.DAY_JOIN] >= WuLinDaHui.PreCfg.DAY_COUNT
end

function WldhTeam:GetName(tbTeam)
  return tbTeam[WldhTeam.TEAM_NAME]
end

if MODULE_GAMESERVER then
  function WldhTeamMgr:AddTeam(nTeamId, tbTeam)
    self.tbNameList[tbTeam[WldhTeam.TEAM_NAME]] = 1
    
    if self.tbAllTeam[nTeamId] then
      LogErr("WldhTeamMgr:AddTeam team repeted ", nTeamId)
      return
    end
    WldhTeamMgr:ModTeamData(nTeamId, tbTeam)
  end
  
  function WldhTeamMgr:DelTeamData(nTeamId)
    local tbTeamInfo = self.tbAllTeam[nTeamId]
    self.tbNameList[tbTeamInfo[WldhTeam.TEAM_NAME]] = nil
    self.tbAllTeam[nTeamId] = nil
    ScriptData:Del(WuLinDaHui.SCRIPT_SAVE_KEY_TEAM, nTeamId)
  end
  
  function WldhTeamMgr:ModTeamData(nTeamId, tbData)
    self.tbAllTeam[nTeamId] = tbData
    ScriptData:SaveValue(WuLinDaHui.SCRIPT_SAVE_KEY_TEAM, nTeamId, tbData)
  end
  
  function WldhTeamMgr:ModifyTeamData(nTeamId)
    ScriptData:MarkModify(WuLinDaHui.SCRIPT_SAVE_KEY_TEAM, nTeamId)
  end
  
  function WldhTeamMgr:GetTeamValue(nTeamId, nKey)
    if not nTeamId or not nKey then
      return
    end
    if self.tbAllTeam[nTeamId] and self.tbAllTeam[nTeamId][nKey] then
      return self.tbAllTeam[nTeamId][nKey]
    end
  end
  
  function WldhTeamMgr:OpearateTeam(nTeamId, func, ...)
    local tbTeam = self.tbAllTeam[nTeamId]
    if tbTeam then
      return func(WldhTeam, tbTeam, ...)
    end
  end
  
  function WldhTeamMgr:OpearateTeamByPlayer(nPlayerId, func, ...)
    local tbTeam = self:GetTeamDataByPlayerId(nPlayerId)
    if tbTeam then
      return func(WldhTeam, tbTeam, ...)
    end
  end
  
  function WldhTeamMgr:GetTeamData(nTeamId)
    return self.tbAllTeam[nTeamId]
  end
  
  function WldhTeamMgr:GetTeamDataByPlayerId(nPlayerId)
    local nPvPTeamId = WuLinDaHui:GetBattleTeamId(nPlayerId)
    if nPlayerId <= 0 or nPvPTeamId <= 0 then
      return
    end
    if self.tbAllTeam[nPvPTeamId] then
      return self.tbAllTeam[nPvPTeamId]
    end
  end
end
