Require("CommonScript/Activity/ChildrenActivity/PreTask/LimitAuction.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.LimitAuction)

function tbActivity:OpenLimitAuctionUI()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ChildrenLimitAuction)
  Ui:OpenWindowWithCallback("UIAuction", function(_, tbWnd)
    tbWnd.tbAuctionTypeBtnGroup:Select(3)
  end)
end

function tbActivity:GetRedPointEventId()
  return {
    RedPointMgr.EVENT.ChildrenLimitAuction
  }
end
