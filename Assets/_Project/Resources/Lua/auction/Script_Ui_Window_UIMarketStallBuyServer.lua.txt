local tbWnd = Ui:CreateWindow("UIMarketStallBuyServer")
local tbControls = {
  btnClose = "imgBG/btnClose",
  uiItemGrid = "imgBG/ItemInfo/UIItemGrid",
  txtItemName = "imgBG/ItemInfo/txtGoodName",
  txtItemNum = "imgBG/ItemInfo/txtGoodNum",
  txtItemDetail = "imgBG/ItemInfo/panelDetail/txtDetail",
  imgPriceMoney = "imgBG/ItemBuy/imgPriceMoney",
  txtPriceValue = "imgBG/ItemBuy/txtPriceValue",
  imgCostMoney = "imgBG/ItemBuy/imgCostMoney",
  txtCostValue = "imgBG/ItemBuy/txtTotalCostValue",
  uiAddAndSubGroup = "imgBG/ItemBuy/UIAddAndSubGroup",
  btnBuy = "imgBG/ItemBuy/btnBuy"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnBuy = function(self)
    TreasureShop:BuyReq(self.nWareId, self.nCurNum, self.nPrice)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbItemGrid = Ui:NewExistItemGrid(self, self.uiItemGrid)
  self.tbItemGrid:SetClickEnable(true)
  self.tbAddAndSubGroup = Ui:NewExistAddAndSubGroup(self, self.uiAddAndSubGroup)
  self.tbAddAndSubGroup:BindEvent(function(nValue)
    return self:OnUpdateNumInput(nValue)
  end)
end

function tbWnd:OnOpen(nWareId, nPrice, nMaxNum)
  local tbWareSetting = TreasureShop:GetWareSettingByWareId(nWareId)
  if not tbWareSetting then
    return 0
  end
  self.nWareId = nWareId
  self.nPrice = nPrice
  self.nMaxNum = nMaxNum
  self.nCurNum = 1
  local nItemTid = tbWareSetting.nItemTid
  self.tbItemGrid:SetItemByTemplate(nItemTid, self.nMaxNum)
  self.pPanel:Label_SetText(tbControls.txtItemName, Item:GetName(nItemTid, me))
  self.pPanel:Label_SetText(tbControls.txtItemNum, i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\239\188\154{0}", me.GetItemCountInBags(nItemTid)))
  self.pPanel:Label_SetText(tbControls.txtItemDetail, Item:GetIntro(nItemTid))
  local szIcon = Currency:GetIcon(tbWareSetting.szCurrency)
  self.pPanel:Sprite_SetSprite(tbControls.imgPriceMoney, szIcon)
  self.pPanel:Sprite_SetSprite(tbControls.imgCostMoney, szIcon)
  self.pPanel:Label_SetText(tbControls.txtPriceValue, self.nPrice)
  self.pPanel:Label_SetText(tbControls.txtCostValue, self.nPrice * self.nCurNum)
  self.tbAddAndSubGroup:SetNumRange(1, self.nMaxNum)
  self.tbAddAndSubGroup:SetInputValue(self.nCurNum)
  self:OnUpdateNumInput(self.nCurNum)
end

function tbWnd:OnUpdateNumInput(nValue)
  local nResult = nValue
  if nResult < 1 then
    nResult = 1
  end
  if nResult > self.nMaxNum then
    nResult = self.nMaxNum
  end
  self.nCurNum = nResult
  local tbWareSetting = TreasureShop:GetWareSettingByWareId(self.nWareId)
  if self.nPrice * self.nCurNum <= me.GetMoney(tbWareSetting.szCurrency) then
    self.pPanel:Label_SetColorByName(tbControls.txtCostValue, "NewGold")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtCostValue, "NewRed")
  end
  self.pPanel:Label_SetText(tbControls.txtCostValue, self.nPrice * self.nCurNum)
  return nResult
end
