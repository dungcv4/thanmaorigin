local tbWnd = Ui:CreateWindow("UIMarketStallBuy")
tbWnd.tbControls = {
  txtGoodName = "imgBG/ItemInfo/txtGoodName",
  UIItemGrid = "imgBG/ItemInfo/UIItemGrid",
  UIAddAndSubGroup = "imgBG/ItemBuy/UIAddAndSubGroup",
  imgPriceMoney = "imgBG/ItemBuy/imgPriceMoney",
  imgCostMoney = "imgBG/ItemBuy/imgCostMoney",
  txtTotalCostValue = "imgBG/ItemBuy/txtTotalCostValue",
  txtPriceValue = "imgBG/ItemBuy/txtPriceValue",
  btnBuy = "imgBG/ItemBuy/btnBuy",
  txtDetail = "imgBG/ItemInfo/panelDetail/txtDetail",
  btnClose = "imgBG/btnClose",
  txtGoodNum = "imgBG/ItemInfo/txtGoodNum"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnBuy = function(self)
    self:DoBuy()
  end
}

function tbWnd:NotifyEvents()
  return {}
end

function tbWnd:OnCreate()
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, self.UIAddAndSubGroup)
  self.AddAndSubGroup:BindEvent(function(nValue)
    self.nbuyCount = self:UpdateNumberInput(nValue)
    return self.nbuyCount
  end)
  self.AddAndSubGroup:SetNumPadPos(-148, 12)
  self.AddAndSubGroup:SetEnable(true)
  self.ItemGrid = Ui:NewExistItemGrid(self, self.UIItemGrid)
end

function tbWnd:OnOpen(tbData)
  if not tbData then
    return true
  end
  self.dwItemId = tbData.dwItemId
  self.nStallType = tbData.nStallType
  self.nSeTypeId = tbData.nSeTypeId
  self.nStallItemTempId = tbData.tbItem.nStallItemTempId
  self.tbItem = tbData.tbItem
  self.nPerPrice = self.tbItem.nPrice
  self.nItemCount = self.tbItem.nCount or 1
  self.nMaxInputCount = self.nItemCount
  self.nbuyCount = 1
  self.nMinInputCount = 1
  local nHaveNum = me.GetItemCountInBags(self.nStallItemTempId)
  self.pPanel:Label_SetText(self.txtGoodNum, i18n.Get(Shop.Terms.HaveNum, nHaveNum))
  self.pPanel:Label_SetText(self.txtPriceValue, self.nPerPrice)
  self.AddAndSubGroup:SetNumRange(self.nMinInputCount, self.nMaxInputCount)
  self:UpdateNumberInput(self.nbuyCount)
  self.AddAndSubGroup:SetInputValue(self.nbuyCount)
  if self.dwItemId and self.dwItemId > 0 then
    self.ItemGrid:SetItem(self.dwItemId, self.nStallItemTempId)
    local MarketStallClass = MarketStallMgr:GetMarketStallObj(self.nStallType)
    MarketStallClass:SetStallItemGrid(self.ItemGrid, self.tbItem, MarketStallBase.tbReqItemDataCbType.BUY_INFO_PANEL)
  else
    self.ItemGrid:SetItemByTemplate(self.nStallItemTempId, self.nItemCount)
    self.ItemGrid:SetClickEnable(true)
  end
  local szTitle = KItem.GetItemShowInfo(self.nStallItemTempId)
  self.pPanel:Label_SetText(self.txtGoodName, i18n.Parse(szTitle))
  local szIcon = Currency:GetIcon(MarketStallBase.tbMoneyType[self.nStallType])
  self.pPanel:Sprite_SetSprite(self.imgPriceMoney, szIcon)
  self.pPanel:Sprite_SetSprite(self.imgCostMoney, szIcon)
  self.pPanel:Label_SetText(self.txtDetail, Item:GetIntro(self.nStallItemTempId) or "")
end

function tbWnd:DoBuy()
  local MarketStallClass = MarketStallMgr:GetMarketStallObj(self.nStallType)
  if MarketStallClass:TryBuyStallItem(self.nSeTypeId, self.nStallItemTempId, self.tbItem, self.nbuyCount) then
    self:Close()
  end
end

function tbWnd:UpdateNumberInput(nNum)
  if nNum < self.nMinInputCount then
    nNum = self.nMinInputCount
    return nNum
  end
  if nNum > self.nMaxInputCount then
    nNum = self.nMaxInputCount
    return nNum
  end
  if self.nPerPrice * nNum <= me.GetMoney(MarketStallBase.tbMoneyType[self.nStallType]) then
    self.pPanel:Label_SetColorByID(self.txtTotalCostValue, 602)
  else
    self.pPanel:Label_SetColorByName(self.txtTotalCostValue, "Red")
  end
  self.pPanel:Label_SetText(self.txtTotalCostValue, self.nPerPrice * nNum)
  return nNum
end
