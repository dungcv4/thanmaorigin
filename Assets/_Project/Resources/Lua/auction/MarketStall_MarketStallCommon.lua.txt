function MarketStallBase:Load()
  local szType = "dsdddddssdd"
  local tbParam = {
    "nStallItemTempId",
    "szMoneyType",
    "nSeType",
    "nDetailType",
    "nIndex",
    "nMinPercent",
    "nMaxPercent",
    "szOpenTimeFrame",
    "szEndTimeFrame",
    "nPrice",
    "nShieldStudio"
  }
  for i = 1, self.nMaxTimeFrameCount do
    szType = szType .. "sd"
    table.insert(tbParam, "szTimeFrame" .. i)
    table.insert(tbParam, "nPrice" .. i)
  end
  self.tbYuanBaoTypeSetting = Lib:LoadTabFileEx("Setting/MarketStall/MarketYuanBaoTypeDef.tab", "ddddsssddd", "nDetailTypeId", {
    "nDetailTypeId",
    "nSeTypeId",
    "nTypeId",
    "nSeTypeSort",
    "szTypeName",
    "szSeTypeName",
    "szDetailTypeName",
    "nIcon",
    "nShowType",
    "nSort"
  })
  self.tbGoldTypeSetting = Lib:LoadTabFileEx("Setting/MarketStall/MarketGoldTypeDef.tab", "ddddsssdddd", "nDetailTypeId", {
    "nDetailTypeId",
    "nSeTypeId",
    "nTypeId",
    "nSeTypeSort",
    "szTypeName",
    "szSeTypeName",
    "szDetailTypeName",
    "nIcon",
    "nShowType",
    "nItemType",
    "nSort"
  })
  self.tbYuanBaoSeTypeSettings = {}
  for _, tbStallTypeSetting in pairs(self.tbYuanBaoTypeSetting) do
    self.tbYuanBaoSeTypeSettings[tbStallTypeSetting.nSeTypeId] = self.tbYuanBaoSeTypeSettings[tbStallTypeSetting.nSeTypeId] or {}
    table.insert(self.tbYuanBaoSeTypeSettings[tbStallTypeSetting.nSeTypeId], tbStallTypeSetting)
  end
  self.tbGoldSeTypeSettings = {}
  for _, tbStallTypeSetting in pairs(self.tbGoldTypeSetting) do
    self.tbGoldSeTypeSettings[tbStallTypeSetting.nSeTypeId] = self.tbGoldSeTypeSettings[tbStallTypeSetting.nSeTypeId] or {}
    table.insert(self.tbGoldSeTypeSettings[tbStallTypeSetting.nSeTypeId], tbStallTypeSetting)
  end
  self.tbYuanBaoDetailTypeList = {}
  for nSeType, tbStallSettingInfo in pairs(self.tbYuanBaoSeTypeSettings) do
    self.tbYuanBaoDetailTypeList[nSeType] = {}
    for _, tbStallDetailTypeInfo in pairs(tbStallSettingInfo) do
      self.tbYuanBaoDetailTypeList[nSeType][tbStallDetailTypeInfo.nDetailTypeId] = true
    end
  end
  self.tbGoldDetailTypeList = {}
  for nSeType, tbStallSettingInfo in pairs(self.tbGoldSeTypeSettings) do
    self.tbGoldDetailTypeList[nSeType] = {}
    for _, tbStallDetailTypeInfo in pairs(tbStallSettingInfo) do
      self.tbGoldDetailTypeList[nSeType][tbStallDetailTypeInfo.nDetailTypeId] = true
    end
  end
  local tbStallItemSettings = Lib:LoadTabFileEx("Setting/MarketStall/MarketStallItem.tab", szType, nil, tbParam)
  self.tbYuanBaoStallItemSettings = {}
  self.tbGoldStallItemSettings = {}
  self.tbYuanBaoDetailType2TempId = {}
  self.tbGoldDetailType2TempId = {}
  for _, tbInfo in pairs(tbStallItemSettings) do
    tbInfo.tbPriceInfo = {}
    for i = 1, self.nMaxTimeFrameCount do
      local szTimeFrame = tbInfo["szTimeFrame" .. i]
      local nPrice = tbInfo["nPrice" .. i]
      if szTimeFrame and "" ~= szTimeFrame then
        assert(nPrice > 0, string.format("Setting/MarketStall/MarketStallItem.tab price error !!! nPrice = %d", nPrice))
        table.insert(tbInfo.tbPriceInfo, {szTimeFrame = szTimeFrame, nPrice = nPrice})
      end
    end
    self:UpdatePrice(tbInfo)
    if tbInfo.szMoneyType == "YuanBao" then
      assert(not self.tbYuanBaoStallItemSettings[tbInfo.nStallItemTempId], string.format("Setting/MarketStall/MarketStallItem.tab item repeat !!! %d", tbInfo.nStallItemTempId))
      self.tbYuanBaoStallItemSettings[tbInfo.nStallItemTempId] = tbInfo
      self.tbYuanBaoDetailType2TempId[tbInfo.nDetailType] = tbInfo.nStallItemTempId
    elseif tbInfo.szMoneyType == "Gold" then
      assert(not self.tbGoldStallItemSettings[tbInfo.nStallItemTempId], string.format("Setting/MarketStall/MarketStallItem.tab item repeat !!! %d", tbInfo.nStallItemTempId))
      self.tbGoldStallItemSettings[tbInfo.nStallItemTempId] = tbInfo
      self.tbGoldDetailType2TempId[tbInfo.nDetailType] = tbInfo.nStallItemTempId
    end
  end
  local tbStallYuanBaoLimitSetting = Lib:LoadTabFileEx("Setting/MarketStall/MarketYuanBaoLimitSetting.tab", "ddddd", nil, {
    "nItemTempId",
    "nSellCountLimit",
    "nSellMoneyLimit",
    "nBuyCountLimit",
    "nBuyMoneyLimit"
  })
  self.tbStallYuanBaoLimitSetting = {}
  local nSaveDataIndex = 0
  for _, tbSetting in ipairs(tbStallYuanBaoLimitSetting) do
    if nSaveDataIndex >= MarketStallBase.nMarketLimitMaxIndex then
      LogErr("MarketYuanBaoLimitSetting.tab setting error! nSaveDataIndex out of range! size:", MarketStallBase.nMarketLimitMaxIndex)
      break
    end
    if not self.tbStallYuanBaoLimitSetting[tbSetting.nItemTempId] then
      self.tbStallYuanBaoLimitSetting[tbSetting.nItemTempId] = {
        nSaveDataIndex = nSaveDataIndex,
        nSellCountLimit = tbSetting.nSellCountLimit,
        nSellMoneyLimit = tbSetting.nSellMoneyLimit,
        nBuyCountLimit = tbSetting.nBuyCountLimit,
        nBuyMoneyLimit = tbSetting.nBuyMoneyLimit
      }
      nSaveDataIndex = nSaveDataIndex + 1
    else
      LogErr("MarketYuanBaoLimitSetting.tab setting error! Id conflict!,Id:", tbSetting.nItemTempId)
    end
  end
  self.tbYuanBaoSeTypeSettings = SetReadOnly(self.tbYuanBaoSeTypeSettings)
  self.tbGoldSeTypeSettings = SetReadOnly(self.tbGoldSeTypeSettings)
  self.tbYuanBaoDetailTypeList = SetReadOnly(self.tbYuanBaoDetailTypeList)
  self.tbGoldDetailTypeList = SetReadOnly(self.tbGoldDetailTypeList)
  self.tbYuanBaoStallItemSettings = SetReadOnly(self.tbYuanBaoStallItemSettings)
  self.tbGoldStallItemSettings = SetReadOnly(self.tbGoldStallItemSettings)
  self.tbYuanBaoDetailType2TempId = SetReadOnly(self.tbYuanBaoDetailType2TempId)
  self.tbGoldDetailType2TempId = SetReadOnly(self.tbGoldDetailType2TempId)
end

function MarketStallBase:UpdatePrice(tbInfo)
  for _, tbPriceInfo in ipairs(tbInfo.tbPriceInfo) do
    local nPrice = tbPriceInfo.nPrice
    local szTimeFrame = tbPriceInfo.szTimeFrame
    if 1 == GetTimeFrameState(szTimeFrame) then
      tbInfo.nPrice = nPrice
    else
      tbInfo.szNextTimeFrame = szTimeFrame
      break
    end
  end
  if tbInfo.szMoneyType == "YuanBao" then
    self:UpdateAllowPrice(tbInfo)
  end
end

function MarketStallBase:UpdateAllowPrice(tbInfo)
  tbInfo.tbAllowPrice = {}
  for nCurPercent = 100, tbInfo.nMaxPercent, self.nPriceChangePercent do
    local nCurPrice
    if 100 == nCurPercent then
      nCurPrice = tbInfo.nPrice
    else
      nCurPrice = math.floor(tbInfo.nPrice * nCurPercent / 100)
    end
    tbInfo.tbAllowPrice[nCurPrice] = 1
  end
  for nCurPercent = 100, tbInfo.nMinPercent, -self.nPriceChangePercent do
    local nCurPrice
    if 100 == nCurPercent then
      nCurPrice = tbInfo.nPrice
    else
      nCurPrice = math.floor(tbInfo.nPrice * nCurPercent / 100)
    end
    tbInfo.tbAllowPrice[nCurPrice] = 1
  end
end

  end
  if not Item:IsInBag(nPos) then
    return false, i18n.EasyGet("\233\129\147\229\133\183\228\184\141\229\156\168\232\131\140\229\140\133\228\184\173\239\188\140\228\184\141\229\143\175\229\135\186\229\148\174")
  end
  local bCanSell = Item:CheckCanSell(pItem)
  if not bCanSell then
    return false, i18n.EasyGet("\230\173\164\231\137\169\229\147\129\228\184\141\229\143\175\229\135\186\229\148\174")
  end
  nCurCount = pItem.nCount

function MarketStallBase:CheckCanNewSellItem(pPlayer, dwItemId, nCount, nPrice, bIgnorCheckCost)
  if self:IsMyStoreFull(pPlayer) then
    return false, i18n.EasyGet("\228\184\141\232\131\189\228\184\138\230\158\182\230\155\180\229\164\154\231\137\169\229\147\129")
  end
  local pItem = KItem.GetItemObj(dwItemId)
  if not pItem then
    return false, i18n.EasyGet("\233\129\147\229\133\183\228\184\141\229\173\152\229\156\168")
  end
  local bRet, szMsg, nCurCount, tbPriceInfo = self:CheckCanSellItem(pPlayer, dwItemId)
  if not bRet then
    return false, szMsg
  end
  if not bIgnorCheckCost and nCount > nCurCount then
    return false, i18n.EasyGet("\229\176\145\228\190\160\230\178\161\230\156\137\232\191\153\228\185\136\229\164\154\231\137\169\229\147\129\229\145\128\239\188\129")
  end
  local nCost, szCurrencyType = self:GetSellCost(nCount * nPrice)
  local nHasMoney = pPlayer.GetMoney(szCurrencyType)
  if not bIgnorCheckCost and nCost > nHasMoney then
    return false, i18n.EasyGet("\229\176\145\228\190\160\230\137\139\231\187\173\232\180\185\228\184\141\232\182\179\239\188\129")
  end
  return true, "", nCost, szCurrencyType
end

function MarketStallBase:IsCanNextPage(pPlayer, szType)
  if not szType then
    return false
  end
  self.tbPlayerPageTime[pPlayer.dwID] = self.tbPlayerPageTime[pPlayer.dwID] or {}
  self.tbPlayerPageTime[pPlayer.dwID][szType] = self.tbPlayerPageTime[pPlayer.dwID][szType] or 0
  local nLastTime = self.tbPlayerPageTime[pPlayer.dwID][szType]
  local nCurTime = GetFrame()
  local nGapTime = nCurTime - nLastTime
  if nGapTime <= self.nWaitForNextPage then
    return false
  end
  self.tbPlayerPageTime[pPlayer.dwID][szType] = nCurTime
  return true
end

function MarketStallBase:CheckTotalSellOutOfLimit(pPlayer, nMoney)
  local nSellMoneyLimitKey = self.tbMapSellMoneyLimit[self.nStallType]
  local nTotalSellMoney = self.tbMapTotalSellMoneyNum[self.nStallType]
  if not nSellMoneyLimitKey or not nTotalSellMoney then
    return false
  end
  local nCurrTotalSellMoney = pPlayer.GetUserValue(self.nMarketLimitGroup, nSellMoneyLimitKey)
  if nTotalSellMoney < nCurrTotalSellMoney + nMoney then
    return true
  end
  return false

function MarketStallBase:CheckSellItemOutOfLimit(pPlayer, nItemTemplateId, nCount, nMoney)
  if not self.tbStallLimitSetting then
    return false
  end
  local tbSetting = self.tbStallLimitSetting[nItemTemplateId]
  if not tbSetting then
    return false
  end
  local nSaveDataIndex = tbSetting.nSaveDataIndex
  local nKeySellCount = self.nMarketLimitKeyBegin + nSaveDataIndex * self.nPerLimitKeyNeed + self.tbDataType2IndexMap.nSellCountLimitKey
  local nKeySellMoney = self.nMarketLimitKeyBegin + nSaveDataIndex * self.nPerLimitKeyNeed + self.tbDataType2IndexMap.nSellMoneyLimitKey
  local nCurrSellCount = pPlayer.GetUserValue(self.nMarketLimitGroup, nKeySellCount)
  local nCurrSellMoney = pPlayer.GetUserValue(self.nMarketLimitGroup, nKeySellMoney)
  if nCurrSellCount + nCount > tbSetting.nSellCountLimit or nCurrSellMoney + nMoney > tbSetting.nSellMoneyLimit then
    return true
  end
  return false
end

function MarketStallBase:CheckMarketSellItemTypeSettings(pItem)
  local nErrCheck = 0
  local bCanMarketStall = false
  local nMarketStallType
  if not pItem then
    return false, nil
  end
  if pItem.IsInTradeLimit(Item.eTRADELIMITWAY.MARKETSTALL) then
    return false, nil
  end
  if MarketStallBase.tbYuanBaoStallItemSettings[pItem.dwTemplateId] then
    local szOpenTimeFrame = MarketStallBase.tbYuanBaoStallItemSettings[pItem.dwTemplateId].szOpenTimeFrame
    local szEndTimeFrame = MarketStallBase.tbYuanBaoStallItemSettings[pItem.dwTemplateId].szEndTimeFrame
    if (Lib:IsEmptyStr(szOpenTimeFrame) or 1 == GetTimeFrameState(szOpenTimeFrame)) and (Lib:IsEmptyStr(szEndTimeFrame) or 1 ~= GetTimeFrameState(szEndTimeFrame)) then
      bCanMarketStall = true
      nMarketStallType = MarketStallBase.tbStallType.YuanBao
      nErrCheck = nErrCheck + 1
    end
  end
  if MarketStallBase.tbGoldStallItemSettings[pItem.dwTemplateId] then
    local szOpenTimeFrame = MarketStallBase.tbGoldStallItemSettings[pItem.dwTemplateId].szOpenTimeFrame
    local szEndTimeFrame = MarketStallBase.tbGoldStallItemSettings[pItem.dwTemplateId].szEndTimeFrame
    if (Lib:IsEmptyStr(szOpenTimeFrame) or 1 == GetTimeFrameState(szOpenTimeFrame)) and (Lib:IsEmptyStr(szEndTimeFrame) or 1 ~= GetTimeFrameState(szEndTimeFrame)) then
      bCanMarketStall = true
      nMarketStallType = MarketStallBase.tbStallType.Gold
      nErrCheck = nErrCheck + 1
    end
  end
  if nErrCheck > 1 then
    LogErr("MarketStallBase:CheckMarketSellItemTypeSettings error detected! ItemTemplateId:", pItem.dwTemplateId)
    return false, nil
  end
