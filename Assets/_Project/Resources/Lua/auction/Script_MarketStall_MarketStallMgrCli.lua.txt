Require("Script/MarketStall/MarketStallYuanBao.lua")
Require("Script/MarketStall/MarketStallGold.lua")
Require("Script/Ui/Common/RedPointMgr.lua")
MarketStallMgr.tbMarketStallClass = {
  [MarketStallBase.tbStallType.YuanBao] = MarketStallYuanBao,
  [MarketStallBase.tbStallType.Gold] = MarketStallGold
}
MarketStallMgr.tbStallType2RedPoint = {
  [MarketStallBase.tbStallType.YuanBao] = RedPointMgr.EVENT.MarketStallYuanBaoTimeoutKeep,
  [MarketStallBase.tbStallType.Gold] = RedPointMgr.EVENT.MarketStallGoldTimeoutKeep
}

function MarketStallMgr:init()
  for nType, tbClass in pairs(self.tbMarketStallClass) do
    tbClass:init()
  end
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_MARKETSTALLITEM, self.OnSyncItemData, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_MARKETSTALLITEMLIST, self.OnSyncItemDataList, self)
end

function MarketStallMgr:OnEvent(nMoneyType, nEvent, ...)
  if 0 == nMoneyType or nEvent <= 0 then
    print("MarketStallMgr:OnEvent Error ", nMoneyType, nEvent)
    return
  end
  local tbClass = self.tbMarketStallClass[nMoneyType]
  tbClass:OnEvent(nEvent, ...)
end

function MarketStallMgr:OnSyncItemData(nStallId, dwItemId, nCbType)
  for nType, tbClass in pairs(self.tbMarketStallClass) do
    tbClass:OnSyncItemData(nStallId, dwItemId, nCbType)
  end
end

function MarketStallMgr:OnSyncItemDataList(nCbType)
  for nType, tbClass in pairs(self.tbMarketStallClass) do
    tbClass:OnSyncItemDataList(nCbType)
  end
end

function MarketStallMgr:GetMarketStallObj(nStallType)
  return self.tbMarketStallClass[nStallType]
end

MarketStallMgr:init()
