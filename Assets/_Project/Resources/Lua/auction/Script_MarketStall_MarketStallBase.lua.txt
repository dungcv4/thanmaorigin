function MarketStallBase:BuidTypeTreeData()
  local tbExistMap = {}
  local tbAvaliableSeType = {}
  self.tbTypeTreeData = {}
  local tbStallItemSettings
  if self.nStallType == self.tbStallType.YuanBao then
    tbStallItemSettings = self.tbYuanBaoStallItemSettings
  elseif self.nStallType == self.tbStallType.Gold then
    tbStallItemSettings = self.tbGoldStallItemSettings
  else
    return
  end
  for nItemId, tbInfo in pairs(tbStallItemSettings) do
    if not tbAvaliableSeType[tbInfo.nSeType] and 1 == GetTimeFrameState(tbInfo.szOpenTimeFrame) and (Lib:IsEmptyStr(tbInfo.szEndTimeFrame) or 1 ~= GetTimeFrameState(tbInfo.szEndTimeFrame)) then
      tbAvaliableSeType[tbInfo.nSeType] = 1
    end
  end
  for _, tbInfo in pairs(self.tbTypeDefSettings) do
    if tbAvaliableSeType[tbInfo.nSeTypeId] then
      if not tbExistMap[tbInfo.nTypeId] then
        table.insert(self.tbTypeTreeData, {
          nTypeId = tbInfo.nTypeId,
          szTypeName = tbInfo.szTypeName,
          tbKind = {}
        })
        tbExistMap[tbInfo.nTypeId] = {
          nIndex = #self.tbTypeTreeData,
          tbSub = {}
        }
      end
      if not tbExistMap[tbInfo.nTypeId].tbSub[tbInfo.nSeTypeId] then
        table.insert(self.tbTypeTreeData[tbExistMap[tbInfo.nTypeId].nIndex].tbKind, {
          nSeTypeId = tbInfo.nSeTypeId,
          nTypeId = tbInfo.nTypeId,
          szSeTypeName = tbInfo.szSeTypeName,
          nSeTypeSort = tbInfo.nSeTypeSort
        })
        tbExistMap[tbInfo.nTypeId].tbSub[tbInfo.nSeTypeId] = 1
      end
    end
  end
  for nIndex, tbInfo in ipairs(self.tbTypeTreeData) do
    table.sort(tbInfo.tbKind, function(a, b)
      if a.nSeTypeSort ~= b.nSeTypeSort then
        return a.nSeTypeSort > b.nSeTypeSort
      end
      return a.nSeTypeId < b.nSeTypeId
    end)
  end
  table.sort(self.tbTypeTreeData, function(a, b)
    return a.nTypeId < b.nTypeId
  end)
end

function MarketStallBase:GetTypeTreeData()
  local nToday = Lib:GetLocalDay(GetTime() - 14400)
  if not self.nRefreshTypeTreeTime or self.nRefreshTypeTreeTime ~= nToday then
    self:BuidTypeTreeData()
    self.nRefreshTypeTreeTime = nToday
  end
  return self.tbTypeTreeData
end

end

function MarketStallBase:TryBuyStallItemBase(nSeType, nStallItemTempId, tbData, nCount)
  if nSeType == self.nStarType then
    nSeType = self:GetItemType(nStallItemTempId)
  elseif self.nSeType ~= nSeType then
    return false
  end
  if not nSeType then
    return false
  end
  if nCount <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\180\173\228\185\176\230\149\176\233\135\143\228\184\141\232\131\189\228\184\1860"))
    return false
  end
  if nCount > tbData.nCount then
    me.CenterMsg(i18n.EasyGet("\229\149\134\229\147\129\230\149\176\233\135\143\229\183\178\229\143\152\229\138\168\239\188\140\232\175\183\233\135\141\230\150\176\231\161\174\232\174\164"))
    return false
  end
  local cost = nCount * tbData.nPrice
  if cost > me.GetMoney(self.tbMoneyType[self.nStallType]) then
    GainCurrency:ReadyForJump(self.tbMoneyType[self.nStallType])
    return false
  end
  local nCanAddCount = me.GetCanAddItemCount(nStallItemTempId)
  if nCount > nCanAddCount then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\233\129\147\229\133\183\230\149\176\233\135\143\232\191\135\229\164\154\239\188\140\232\175\183\230\149\180\231\144\134\228\184\128\228\184\139\239\188\129"))
    return false
  end
  self.tbMgr.DelStallItem(tbData.nStallId)
  if self.nStallType == self.tbStallType.YuanBao then
    self.rpc(self.nStallType, "RPC_BuyStallItem", nSeType, nStallItemTempId, nCount, tbData.nPrice)
  else
    self.rpc(self.nStallType, "RPC_BuyStallItem", nSeType, tbData.nStallId, nCount, tbData.nPrice)
  end
  return true
end

function MarketStallBase:NewSell(nShowType, nIndex, nPrice, nCount)
  local tbData = self.tbAvaliableItems[nShowType] and self.tbAvaliableItems[nShowType][nIndex]
  if not tbData then
    return
  end
  local nStallItemTempId = tbData.nStallItemTempId
  local can, err = self:CheckCanNewSellItem(me, tbData.dwItemId, nCount, nPrice)
  local tbItemBase = Item:GetItemBaseProp(nStallItemTempId)
  local nMaxCount = tbItemBase.nStackMax
  if nMaxCount < 1 then
    nMaxCount = 1
  end
  if nCount > nMaxCount then
    local szMsg = i18n.EasyGet("\229\141\149\230\172\161\230\156\128\229\164\154\229\143\170\229\143\175\228\184\138\230\158\182{0}\228\184\170", nMaxCount)
    can, err = false, szMsg
  end
  if not can then
    me.CenterMsg(err)
    return false
  end
  self:SetSavedPrice(nStallItemTempId, nPrice)
  self.rpc(self.nStallType, "RPC_NewSellItem", tbData.dwItemId, nCount, nPrice)
  return true
end

function MarketStallBase:GetDetailTypeList(tbItems)
  local tbList = {}
  local tbRet = {}
  for nIndex = 1, #tbItems do
    local tbItem = tbItems[nIndex]
    tbRet[tbItem.nDetailTypeId] = tbRet[tbItem.nDetailTypeId] or {}
    tbRet[tbItem.nDetailTypeId].nCount = 0
    tbRet[tbItem.nDetailTypeId].tbItems = {}
  end
  for nIndex = 1, #tbItems do
    local tbItem = tbItems[nIndex]
    local nDetailTypeId = tbItem.nDetailTypeId
    local tbDetailInfo = self.tbTypeDefSettings[nDetailTypeId]
    local nCurCount = tbRet[nDetailTypeId].nCount + tbItem.nCount
    for n = 1, #tbItem.tbItems do
      table.insert(tbRet[nDetailTypeId].tbItems, tbItem.tbItems[n])
    end
    local tbData = {
      nDetailTypeId = nDetailTypeId,
      szDetailTypeName = tbDetailInfo.szDetailTypeName,
      nIcon = tbDetailInfo.nIcon,
      szShowType = tbDetailInfo.szShowType,
      nCount = nCurCount,
      tbItems = tbRet[nDetailTypeId].tbItems,
      nSort = tbDetailInfo.nSort
    }
    tbRet[nDetailTypeId] = tbData
  end
  for _, tb in pairs(tbRet) do
    table.insert(tbList, tb)
  end
  table.sort(tbList, function(a, b)
    if a.nSort ~= b.nSort then
      return a.nSort > b.nSort
    end
    return a.nDetailTypeId < b.nDetailTypeId
  end)
  return tbList
end

function MarketStallBase:SortMyItems()
  local tbItems = self.tbMyItems
  table.sort(tbItems, function(a, b)
    local sa = self:GetItemState(a)
    local sb = self:GetItemState(b)
    return sa < sb or sa == sb and a.nTime > b.nTime
  end)
  self.tbMyItems = tbItems
end

function MarketStallBase:OpenWindowByItemTempId(nStallType, nTemplateId)
  local tbStallInfo, tbTypeSetting
  if nStallType == MarketStallBase.tbStallType.YuanBao then
    tbStallInfo = self.tbYuanBaoStallItemSettings
    tbTypeSetting = self.tbYuanBaoTypeSetting
  else
    tbStallInfo = self.tbGoldStallItemSettings
    tbTypeSetting = self.tbGoldTypeSetting
  end
  local tbInfo = tbStallInfo[nTemplateId]
  if not tbInfo then
    return false
  end
  local MarketStallClass = MarketStallMgr:GetMarketStallObj(nStallType)
  local nTypeId = tbTypeSetting[tbInfo.nDetailType] and tbTypeSetting[tbInfo.nDetailType].nTypeId
  local nSeTypeId = tbTypeSetting[tbInfo.nDetailType] and tbTypeSetting[tbInfo.nDetailType].nSeTypeId
  local tbTypeTreeData = MarketStallClass:GetTypeTreeData()
  local nBigIndex = 0
  local nSmallIndex = 0
  local bFind = false
  if tbTypeTreeData then
    for nIndex, tbInfo in ipairs(tbTypeTreeData) do
      if tbInfo.nTypeId == nTypeId then
        nBigIndex = nIndex
        break
      elseif #tbInfo.tbKind > 1 then
        nSmallIndex = nSmallIndex + #tbInfo.tbKind
      end
    end
    if nBigIndex > 0 then
      if #tbTypeTreeData[nBigIndex].tbKind > 1 then
        for nIndex, tbInfo in ipairs(tbTypeTreeData[nBigIndex].tbKind) do
          if tbInfo.nSeTypeId == nSeTypeId then
            bFind = true
            nSmallIndex = nSmallIndex + nIndex
            break
          end
        end
      else
        bFind = true
        nSmallIndex = 0
      end
    end
  end
  if not bFind then
    return false
  end
  Ui:OpenWindow("UIMarketStall", nStallType, Ui.UIMarketStall.tbSelectTypeIndex.SELECT_TYPE_BUY, nBigIndex + 1, nSmallIndex)
  return true
end

  local tbAvaliableItems = MarketStallClass:GetAvaliableItems()
  local tbItems = tbAvaliableItems[Ui.tbToggle.TogAll]
  local nIndex = 0
  local tbItem
  if tbItems and #tbItems > 0 then
    for index, tbItemInfo in ipairs(tbItems) do
      if tbItemInfo.dwItemId == nItemId then
        nIndex = index
        tbItem = tbItemInfo
        break
      end
    end
  end
  if nIndex > 0 then
    local tbWndData = {
      dwItemId = tbItem.dwItemId,
      nItemTemplateId = tbItem.nStallItemTempId,
      szWndType = MarketStallBase.tbUIName[MarketStallBase.tbReqItemDataCbType.SELL_INFO_PANEL],
      nShowType = Ui.tbToggle.TogAll,
      index = nIndex,
      bBag = true,
      nStallType = nMarketType,
      nCount = tbItem.nCount,
      nPrice = tbItem.nPrice,
      tbItem = tbItem
    }
    Ui:OpenWindow(MarketStallBase.tbUIName[MarketStallBase.tbReqItemDataCbType.SELL_INFO_PANEL], tbWndData)
  end
end

function MarketStallBase:CheckCanOpenUi()
  if 0 == GetTimeFrameState(self.szCheckTimeFrame) then
    return true
  end
  local tbInfo = Calendar:GetPreviewInfo()
  if tbInfo and me.nLevel >= tbInfo.nCurMaxLevel - self.nMaxLevelDiff then
    return true
  end
  return false, tbInfo.nCurMaxLevel - self.nMaxLevelDiff
end
