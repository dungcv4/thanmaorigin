Auction.tbAuctionData = Auction.tbAuctionData or {}
Auction.tbAuctionDealList = Auction.tbAuctionDealList or {}
Auction.tbNotifyIndex = Auction.tbNotifyIndex or {}

function Auction:OnAuctionSyncData(szType, tbAuction, bSyncFinish)
  if szType then
    self.tbAuctionData[szType] = tbAuction
  end
  if bSyncFinish then
    EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_AUCTION_DATA, "Auction", szType)
  end
end

function Auction:GetAuctionData(szType)
  local nNow = GetTime()
  local szGlobalType = self:GetGlobalType()
  if not self.tbAuctionData[szGlobalType] then
    self.tbAuctionData[szGlobalType] = {
      szType = szGlobalType,
      tbItems = {},
      nEndTime = nNow * 2
    }
  end
  local szShoperType = self:GetShoperType()
  if not self.tbAuctionData[szShoperType] then
    self.tbAuctionData[szShoperType] = {
      szType = szShoperType,
      tbItems = {},
      nEndTime = nNow * 2
    }
  end
  if szType then
    return self.tbAuctionData[szType]
  end
  return self.tbAuctionData
end

function Auction:Ask4AllAuctionDataAndMonitorUI(bMonitor)
  RemoteServer.OnAuctionRequest("OnSyncAllAuctionsAndMonitorUI", bMonitor)
end

function Auction:ClearAuctionCache()
  local nNowTime = GetTime()
  local nEndTime
  for szType, tbAuctionData in pairs(self.tbAuctionData or {}) do
    if self:IsGlobal(szType) or self:IsShoper(szType) or self.Def.tbTypeToName[szType] then
      nEndTime = tbAuctionData.nEndTime
      for _, tbItemData in pairs(tbAuctionData.tbItems or {}) do
        if nEndTime < tbItemData.nTimeOut then
          nEndTime = tbItemData.nTimeOut
        end
      end
      if nNowTime > nEndTime then
        self.tbAuctionData[szType] = nil
      end
    end
  end
end

function Auction:AuctionOpenNew(szType)
  RedPointMgr:LightRedPoint(RedPointMgr.EVENT.AuctionOpen)
  if self:IsShoper(szType) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ShoperAuctionOpen)
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ChildrenLimitAuction)
  end
  if not self:IsGlobal(szType) then
    local nIndex = self.tbNotifyIndex[szType]
    if nIndex then
      Ui:CancelNotifyMsg(nIndex)
    end
    local bShoper = self:IsShoper(szType)
    local nNow = GetTime()
    local nTimeOut
    if bShoper then
      nTimeOut = nNow + self.Def.nShoperLastingTime
    else
      nTimeOut = nNow + self.Def.nAuctionPrepareTime + self.Def.nKinOrTongAuctionLastingTime + self.Def.nBidOverDelayTime
    end
    self.tbNotifyIndex[szType] = Ui:SynNotifyMsg({
      szType = "AuctionOpenNew",
      nTimeOut = nTimeOut,
      bShoper = bShoper
    })
  end
end

function Auction:AuctionFinish(szType)
  if not self:IsGlobal(szType) or not self:IsShoper(szType) then
    local nIndex = self.tbNotifyIndex[szType]
    if nIndex then
      self.tbNotifyIndex[szType] = nil
      Ui:CancelNotifyMsg(nIndex)
    end
  end
end

function Auction:SyncOneItem(szType, tbSyncData, nSyncId)
  local tbItems = self.tbAuctionData[szType] and self.tbAuctionData[szType].tbItems or nil
  if tbItems then
    for nId, _ in pairs(tbItems) do
      if nSyncId == tbItems[nId].nId then
        if tbSyncData then
          tbItems[nId] = tbSyncData
          if tbSyncData.bSold then
            tbSyncData.bKeepSoldOut = true
          end
          break
        end
        tbItems[nId].bKeepSoldOut = true
        break
      end
    end
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_AUCTION_DATA, "SyncOneItem", szType, tbSyncData, nSyncId)
end

function Auction:OnNotifyExpectEachGold(szType, nEachGold)
  local tbAuctionData = self.tbAuctionData[szType]
  if not tbAuctionData then
    return
  end
  tbAuctionData.nExpectEachGold = nEachGold
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_AUCTION_DATA, "ExpectEachGold", szType, nEachGold)
end

function Auction:SaveMyAuctionInfo(szType, tbItemData)
  local nNow = GetTime()
  if not self.tbMyAuctionData then
    self.tbMyAuctionData = {}
  end
  if not self.tbMyAuctionData[szType] then
    self.tbMyAuctionData[szType] = {}
  end
  self.tbMyAuctionData[szType][tbItemData.nId] = tbItemData
end

function Auction:IsMyAuctionInfo(szType, nId)
  local nNow = GetTime()
  if not self.tbMyAuctionData then
    return false
  end
  if not self.tbMyAuctionData[szType] then
    return false
  end
  if not self.tbMyAuctionData[szType][nId] then
    return false
  end
  if nNow > self.tbMyAuctionData[szType][nId].nTimeOut then
    self.tbMyAuctionData[szType][nId] = nil
    return false
  end
  return true
end

function Auction:OnLogout()
  self.tbAuctionData = {}
  self.tbAuctionDealList = {}
  self.tbNotifyIndex = {}
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, Auction)
EventNotify:RegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, Auction.OnLogout, Auction)
