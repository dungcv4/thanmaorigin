function Trade:CheckSelfCommon()
  if not Map:CanTrade(me.nMapTemplateId) then
    return false, Trade.ERR.CANNOT_TRADE_THIS_MAP
  end
  if 0 ~= me.nFightMode then
    return false, Trade.ERR.SELF_IN_FIGHT_MODE
  end
  return true
end

function Trade:CheckDistance(pNpc)
  if me.nMapTemplateId ~= pNpc.nMapTemplateId then
    return false, Trade.ERR.NOT_IN_SAME_MAP
  end
  local nDistance = me.GetNpc().GetDistance(pNpc.nId)
  if nDistance < 0 or nDistance >= Trade.nDistance then
    return false, Trade.ERR.OTHER_SIDE_NOT_ONLINE
  end
  return true
end

function Trade:CheckMoneyEnough(pPlayer, dwNewMoney)
  local nMoney = pPlayer.GetMoney(Trade.CurrencyType)
  return dwNewMoney <= nMoney and dwNewMoney >= 0
end

function Trade:CheckIsTradding(pPlayer)
  return pPlayer.nIsP2PTrading > 0
end

function Trade:CheckTradeStateDecision(processor, nToState)
  local dwPlayerId = processor.GetOtherPlayerId(me.dwID)
  if nToState == Trade.STATE.LOCK and Trade.CheckLock(processor, me.dwID) then
    return false, Trade.ERR.IN_LOCK
  end
  if nToState == Trade.STATE.LOCK and Trade.CheckConfirm(processor, me.dwID) then
    return false, Trade.ERR.IN_CONFIRM
  end
  if nToState == Trade.STATE.CONFIRM and Trade.CheckConfirm(processor, me.dwID) then
    return false, Trade.ERR.IN_CONFIRM
  end
  if nToState == Trade.STATE.CONFIRM and Trade.CheckExchanging(processor, dwPlayerId) then
    return false, Trade.ERR.OTHER_IN_UNLOCK
  end
  if nToState == Trade.STATE.EXCHANGING and Trade.CheckConfirm(processor, me.dwID) then
    return false, Trade.ERR.IN_CONFIRM
  end
  if nToState == Trade.STATE.EXCHANGING and not Trade.CheckLock(processor, me.dwID) then
    return false, Trade.ERR.SELF_IN_UNLOCK
  end
  return true
end

function Trade:CheckOneItemAdd(pPlayer, dwItemId, nCount)
  local pItem = pPlayer.GetItemInAllSpace(dwItemId)
  if not pItem then
    return false
  end
  return nCount <= pItem.nCount
end

function Trade:CheckItemCanTrade(dwItemId)
  local pItem = dwItemId and KItem.GetItemObj(dwItemId)
  return pItem and not pItem.IsInTradeLimit(Item.eTRADELIMITWAY.P2P)
end

function Trade.CheckLock(processor, dwPlayerId)
  return processor.GetPlayerTradeState(dwPlayerId) == Trade.STATE.LOCK
end

function Trade.CheckConfirm(processor, dwPlayerId)
  return processor.GetPlayerTradeState(dwPlayerId) == Trade.STATE.CONFIRM
end

function Trade.CheckExchanging(processor, dwPlayerId)
  return processor.GetPlayerTradeState(dwPlayerId) == Trade.STATE.EXCHANGING
end

function Trade:LoadTradeItem()
  local tbField = {
    "TemplateId",
    "TradeLimitType",
    "TradeLimitValue",
    "TradeWay"
  }
  local tbTemp = Lib:LoadTabFileEx("Setting/Trade/TradeItem.tab", "ddds", nil, tbField)
  self.tbTemplateId2TradeItem = {}
  for _, tbData in pairs(tbTemp) do
    local tbParamInfo = Lib:SplitStr(tbData.TradeWay, "|")
    local tbItem = {
      nTradeLimitType = tbData.TradeLimitType,
      nTradeLimitValue = tbData.TradeLimitValue
    }
    tbItem.nTradeWay = 0
    for i, v in ipairs(tbParamInfo) do
      tbItem.nTradeWay = tbItem.nTradeWay | 1 << v - 1
    end
    self.tbTemplateId2TradeItem[tbData.TemplateId] = tbItem
  end
  Validator:Register(self.CheckTradeItem, self)
end

function Trade:CheckTradeItem()
  for TemplateId, v in pairs(self.tbTemplateId2TradeItem) do
    if not Item:IsExist(TemplateId) then
      LogErr(string.format("load TradeItem.tab failed, TemplateId %d not exist", TemplateId))
      return false
    end
  end
  return true
end

Trade:LoadTradeItem()

function Trade:GetTradeLimitWay(nTemplateId)
  if not nTemplateId or not self.tbTemplateId2TradeItem[nTemplateId] then
    return 0
  end
  return self.tbTemplateId2TradeItem[nTemplateId].nTradeWay or 0
end
