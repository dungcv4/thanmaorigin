local tbWnd = Ui:CreateWindow("UIMarketStallServer")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  uiMoneyPanel = "node/UIMoneyPanel",
  btnAllStall = "node/imgBG/btnList/btnAllStall",
  btnMyStall = "node/imgBG/btnList/btnMyStall",
  btnAttach = "node/imgBG/btnList/btnAttachment",
  goAllStall = "node/imgBG/Content/AllStall",
  goMyStall = "node/imgBG/Content/MyStall",
  goAttach = "node/imgBG/Content/Attachment"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("TreasureShopHelp")
  end
}
tbWnd.PANEL_TYPE_PUB = 1
tbWnd.PANEL_TYPE_MY = 2
tbWnd.PANEL_TYPE_ATTACH = 3
tbWnd.PANEL_SETTING = {
  [tbWnd.PANEL_TYPE_PUB] = {
    CLASS = "UIMarketStallServerPubPanel",
    OBJ = tbControls.goAllStall,
    BTN = tbControls.btnAllStall
  },
  [tbWnd.PANEL_TYPE_MY] = {
    CLASS = "UIMarketStallServerMyPanel",
    OBJ = tbControls.goMyStall,
    BTN = tbControls.btnMyStall
  },
  [tbWnd.PANEL_TYPE_ATTACH] = {
    CLASS = "UIMarketStallServerAttachPanel",
    OBJ = tbControls.goAttach,
    BTN = tbControls.btnAttach
  }
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_CHANGE_MONEY,
      self.OnChangeMoney
    },
    {
      EventNotify.emNOTIFY_TREASURESHOP_SYNC_MY_STALL,
      self.OnUpdateMyStall
    },
    {
      EventNotify.emNOTIFY_TREASURESHOP_SYNC_PUB_STALL,
      self.OnUpdatePubStall
    },
    {
      EventNotify.emNOTIFY_TREASURESHOP_SYNC_MAILBOX,
      self.OnUpdateMailbox
    },
    {
      EventNotify.emNOTIFY_TREASURESHOP_UPDATE_BAG_WARE,
      self.OnUpdateBagWare
    }
  }
end

function tbWnd:OnCreate()
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, self.uiMoneyPanel)
  self.tbTabCtrl = Ui:NewTabControl(self, self.PANEL_SETTING)
end

function tbWnd:OnOpen()
  self:RegistRedPoint()
  self.tbTabCtrl:OnOpen()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:OnClose()
  self:UnRegistRedPoint()
  self.tbTabCtrl:OnClose()
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnMyStall, {
    RedPointMgr.EVENT.TreasureShopStallTimeout
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnAttach, {
    RedPointMgr.EVENT.TreasureShopAttachTypeGiveback,
    RedPointMgr.EVENT.TreasureShopAttachTypeBuy,
    RedPointMgr.EVENT.TreasureShopAttachTypeRequire
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnMyStall)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnAttach)
end

function tbWnd:OnChangeMoney()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:OnUpdateMyStall()
  self.tbTabCtrl:UpdateSubPanel(self.PANEL_TYPE_MY, "OnUpdateStall")
end

function tbWnd:OnUpdateBagWare()
  self.tbTabCtrl:UpdateSubPanel(self.PANEL_TYPE_MY, "OnUpdateBagWare")
end

function tbWnd:OnUpdatePubStall()
  self.tbTabCtrl:UpdateSubPanel(self.PANEL_TYPE_PUB, "OnUpdateStall")
end

function tbWnd:OnUpdateMailbox()
  self.tbTabCtrl:UpdateSubPanel(self.PANEL_TYPE_PUB, "OnUpdateMailbox")
  self.tbTabCtrl:UpdateSubPanel(self.PANEL_TYPE_ATTACH, "OnUpdateMailbox")
end
