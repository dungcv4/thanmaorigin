  end,
  btnFire = function(self)

  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_BLESSING_CHANGE,
      self.UpdateUI,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_BLESSING_RESULT,
      self.OnSyncBlessingResult,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.pPanel:SetActive(self.txtSoil, false)
  self.pPanel:SetActive(self.txtFire, false)
  self.pPanel:SetActive(self.txtWood, false)
  self.pPanel:SetActive(self.txtGold, false)
  self.pPanel:SetActive(self.txtWater, false)
  self.tbAttrScrollView1 = Ui:NewScrollView(self, self.attrTemplate1)
  self.tbAttrScrollView2 = Ui:NewScrollView(self, self.attrTemplate2)
  self.tbAttrScrollView3 = Ui:NewScrollView(self, self.attrTemplate3)
end

function tbWnd:OnOpen()
  self:UpdateUI()
end

function tbWnd:UpdateUI()
  self:UpdateTimeTxt()
  self:ClearBlessingElement()
  local bIsOn = 1 == Ui.PlayerPrefs.GetInt("ToggleSkipBlessingAnimation")
  self.pPanel:SetActive(self.toggleSkip, Blessing:HasEverBlessed(me))
  self.pPanel:Toggle_SetChecked(self.toggleSkip, bIsOn)
  for nSeries = Npc.emSERIES_TYPE.SERIES_METAL, Npc.emSERIES_TYPE.SERIES_EARTH do
    self.pPanel:SetActive(self["imgSuitSeriesSelect" .. nSeries], false)
  end
  if Blessing:CheckFreeBlessing(me) then
    self.pPanel:Label_SetText(string.format("%s%s", self.btnChange, "/Text"), i18n.EasyGet("\231\165\136\231\166\143"))
  else
    self.pPanel:Label_SetText(string.format("%s%s", self.btnChange, "/Text"), i18n.EasyGet("\233\128\134\229\164\169\230\148\185\229\145\189"))
  end
  local nBlessingId = Blessing:GetBlessingId(me)
  if 0 == nBlessingId or Blessing:CheckBlessingTimeOut(me) then
    self.pPanel:SetActive(self.Left, false)
    self.pPanel:SetActive(self.Right, false)
    self.pPanel:SetActive(self.wndInfoTips, false)
    for i = 1, 5 do
      local szPath = string.format(self.imgSpotCenter, i)
      self.pPanel:Sprite_SetSprite(szPath, tbSeries2SpotImg[0])
      self.pPanel:SetActive(string.format("%s%s", szPath, "/Effect"), false)
    end
    return
  end
  self.pPanel:SetActive(self.Left, true)
  self.pPanel:SetActive(self.Right, true)
  self.pPanel:SetActive(self.wndInfoTips, false)
  self.tbBlessingSeries = Blessing:ParseBlessingId2Table(nBlessingId)
  self.tbBlessingElementKey = {}
  for k, id in ipairs(self.tbBlessingSeries) do
    local szTemplateKey = self[tbSeries2Txt[id]]
    local szKey = self.pPanel:CloneObject(szTemplateKey, string.format("%s%d", "Element", k))
    table.insert(self.tbBlessingElementKey, szKey)
  end
  if #self.tbBlessingSeries - 1 >= 2 then
    self.nSuitSeries = self.tbBlessingSeries[#self.tbBlessingSeries - 1]
  else
    self.nSuitSeries = nil
  end
  local bShowEffect = false
  if self.tbBlessingSeries[1] == self.tbBlessingSeries[#self.tbBlessingSeries] then
    bShowEffect = true
  end
  self.pPanel:SetActive(self.eftBlessingType, bShowEffect)
  self.pPanel:SetActive(self.eftZhuanPan, bShowEffect)
  local tbConfig = Blessing:GetBlessingConfig(nBlessingId)
  self.pPanel:Label_SetText(self.txtBlessingType, i18n.Parse(tbConfig.szBlessingType))
  self.pPanel:Label_SetText(self.txtDesc, i18n.Parse(tbConfig.szDesc))
  self.pPanel:Label_SetText(self.txtToDo, i18n.Parse(tbConfig.szToDo))
  for nSeries = Npc.emSERIES_TYPE.SERIES_METAL, Npc.emSERIES_TYPE.SERIES_EARTH do
    local txtSuitSeries = string.format("%s%d", "txtSuitSeries", nSeries)
    local txtSuitSeriesGray = string.format("%s%d", "txtSuitSeriesGray", nSeries)
    local bActive = nSeries == self.nSuitSeries
    self.pPanel:SetActive(self[txtSuitSeries], bActive)
    self.pPanel:SetActive(self[txtSuitSeriesGray], not bActive)
  end
  for i = 1, 5 do
    local szPath = string.format(self.imgSpotCenter, i)
    if i <= #self.tbBlessingSeries then
      self.pPanel:Sprite_SetSprite(szPath, tbSeries2SpotImg[self.tbBlessingSeries[i]])
      self.pPanel:SetActive(string.format("%s%s", szPath, "/Effect"), true)
    else
      self.pPanel:Sprite_SetSprite(szPath, tbSeries2SpotImg[0])
      self.pPanel:SetActive(string.format("%s%s", szPath, "/Effect"), false)
    end
  end
  self:UpdateAttr(nBlessingId)
end

function tbWnd:OnSyncBlessingResult(nBlessingId)
  if 1 == Ui.PlayerPrefs.GetInt("ToggleSkipBlessingAnimation") and Blessing:HasEverBlessed(me) then
    if 0 ~= Blessing:GetTempBlessingId(me) then
      Ui:OpenWindow("UIPraySelect")
    else
      self:UpdateUI()
    end
    return
  end
  for i = 1, 5 do
    local szPath = string.format(self.imgSpotCenter, i)
    self.pPanel:Sprite_SetSprite(szPath, tbSeries2SpotImg[0])
    self.pPanel:SetActive(string.format("%s%s", szPath, "/Effect"), false)
  end
  self.tbAnimationBlessingSeries = Blessing:ParseBlessingId2Table(nBlessingId)
  self.nAnimationIndex = 1
  local Animator = self.pPanel:GetAnimator(self.animZhuanPan)
  self.pPanel:Animator_SetEnable(self.animZhuanPan, true)
  if Animator then
    Animator:Play(tbSeries2ZhuanPanAnimation[self.tbAnimationBlessingSeries[self.nAnimationIndex]], 0, 0)
  end
end

function tbWnd:OnAnimationEnd()
  local szPath = string.format(self.imgSpotCenter, self.nAnimationIndex)
  self.pPanel:Sprite_SetSprite(szPath, tbSeries2SpotImg[self.tbAnimationBlessingSeries[self.nAnimationIndex]])
  self.pPanel:SetActive(string.format("%s%s", szPath, "/Effect"), true)
  self.nAnimationIndex = self.nAnimationIndex + 1
  if self.nAnimationIndex > #self.tbAnimationBlessingSeries then
    local Animator = self.pPanel:GetAnimator(self.animZhuanPan)
    if Animator then
      Animator:Play(tbSeries2ZhuanPanAnimation[self.tbAnimationBlessingSeries[1]], 0, 0)
      Animator:Update(0)
    end
    self.pPanel:Animator_SetEnable(self.animZhuanPan, false)
    if 0 ~= Blessing:GetTempBlessingId(me) then
      Ui:OpenWindow("UIPraySelect")
    else
      self:UpdateUI()
    end
    return
  end
  local Animator = self.pPanel:GetAnimator(self.animZhuanPan)
  if Animator then
    Animator:Play(tbSeries2ZhuanPanAnimation[self.tbAnimationBlessingSeries[self.nAnimationIndex]], 0, 0)
  end
end

function tbWnd:OnClickSeries(nSeriesType)
  self.pPanel:SetActive(self.wndInfoTips, true)
  self.pPanel:SetActive(self["imgSuitSeriesSelect" .. nSeriesType], true)
  self:UpdateWndInfoTips(nSeriesType)
end

function tbWnd:UpdateTimeTxt()
  local nBlessingRemainTime = Blessing:GetBlessingEndTime(me) - GetTime()
  if nBlessingRemainTime > 0 then
    self.pPanel:Label_SetText(self.txtTimeRight, i18n.EasyGet("\229\137\169\228\189\153{0}", Lib:TimeDesc3(nBlessingRemainTime)))
  end
  local nNextBlessTime = Blessing:GetNextDayBlessTime()
  if self.nTimer then
    Timer:Close(self.nTimer)
  end
  self.nTimer = Timer:Register(Env.GAME_FPS, function()
    if nBlessingRemainTime > 0 then
      nBlessingRemainTime = nBlessingRemainTime - 1
      self.pPanel:Label_SetText(self.txtTimeRight, i18n.EasyGet("\229\137\169\228\189\153{0}", Lib:TimeDesc3(nBlessingRemainTime)))
    else
      local nBlessingId = Blessing:GetBlessingId(me)
      if 0 == nBlessingId or Blessing:CheckBlessingTimeOut(me) then
        self.pPanel:SetActive(self.Left, false)
        self.pPanel:SetActive(self.Right, false)
        self.pPanel:SetActive(self.wndInfoTips, false)
      end
    end
    if nNextBlessTime > 0 then
      nNextBlessTime = nNextBlessTime - 1
    end
    return nBlessingRemainTime > 0 or nNextBlessTime > 0
  end)
end

function tbWnd:UpdateAttr(nBlessingId)
  local tbAttr, tbSuitAttr = Blessing:GetBlessingAttrib(nBlessingId, me.nLevel)
  self.tbAttrScrollView1:Clear()
  self.tbAttrScrollView2:Clear()
  for k, v in ipairs(tbAttr) do
    self.tbAttrScrollView1:SetElement(k, function(tbElement)
      local szName, nValue = FightSkill:GetMagicDescSplit(v.szAttribType, v.tbValue)
      tbElement.pPanel:Label_SetText("Main", szName)
      tbElement.pPanel:Label_SetText("TxtNum", nValue)
    end)
  end
  for k, v in ipairs(tbSuitAttr) do
    self.tbAttrScrollView2:SetElement(k, function(tbElement)
      local szName, nValue = FightSkill:GetMagicDescSplit(v.szAttribType, v.tbValue)
      tbElement.pPanel:Label_SetText("Main", szName)
      tbElement.pPanel:Label_SetText("TxtNum", nValue)
    end)
  end
end

function tbWnd:UpdateWndInfoTips(nSeriesType)
  self.pPanel:Label_SetText(self.txtSeriesTitle, Npc.Series[nSeriesType])
  local nCount = 0
  for index, nSeries in ipairs(self.tbBlessingSeries) do
    if nSeries == nSeriesType then
      nCount = nCount + 1
    end
  end
  for i = 1, 5 do
    if i <= nCount then
      self.pPanel:Sprite_SetSprite(string.format(self.imgSpotWnd, i), tbSeries2SpotImg[nSeriesType])
    else
      self.pPanel:Sprite_SetSprite(string.format(self.imgSpotWnd, i), tbSeries2SpotImg[0])
    end
  end
  local nBlessingId = nSeriesType * 11111
  self.tbAttrScrollView3:Clear()
  local _, tbSuitAttr = Blessing:GetBlessingAttrib(nBlessingId, me.nLevel)
  for k, v in ipairs(tbSuitAttr) do
    self.tbAttrScrollView3:SetElement(k, function(tbElement)
      local szName, nValue = FightSkill:GetMagicDescSplit(v.szAttribType, v.tbValue)
      local bGrey = false
      local szPrefix = ""
      if 1 == k then
        szPrefix = i18n.EasyGet("{0}{1}", 2, Npc.Series[nSeriesType]())
        szName = string.format("%s%s%s", szPrefix, " ", szName)
        if nCount < 2 then
          bGrey = true
        end
      elseif 2 == k then
        szPrefix = i18n.EasyGet("{0}{1}", 5, Npc.Series[nSeriesType]())
        szName = string.format("%s%s%s", szPrefix, " ", szName)
        if nCount < 5 then
          bGrey = true
        end
      end
      if bGrey then
        szName = string.format("<color=#7e7e7e>%s</color>", szName)
        nValue = string.format("<color=#7e7e7e>%s</color>", nValue)
      end
      tbElement.pPanel:Label_SetText("Main", szName)
      tbElement.pPanel:Label_SetText("TxtNum", nValue)
    end)
  end
end

function tbWnd:ClearBlessingElement()
  if self.tbBlessingElementKey then
    for _, szKey in ipairs(self.tbBlessingElementKey) do
      self.pPanel:DeleteObject(szKey)
    end
  end
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.tbAppliedSeries = nil
  self:ClearBlessingElement()
  self.tbAttrScrollView1:Clear()
  self.tbAttrScrollView2:Clear()
  self.tbAttrScrollView3:Clear()
end
