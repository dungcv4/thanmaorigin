tbWnd.tbOnClick = {
  btnClose = function(self)
    self:SelectBlessing(false, true)
  end,
  btnApplyOrigin = function(self)
    self:SelectBlessing(false)
  end,
  btnApplyNew = function(self)
    self:SelectBlessing(true)
  end,
  btnCancel = function(self)
    self.pPanel:SetActive(self.wndConfirm, false)
  end,
  btnConfirm = function(self)

local tbSeries2Txt = {
  [Npc.emSERIES_TYPE.SERIES_METAL] = "txtGold",
  [Npc.emSERIES_TYPE.SERIES_WOOD] = "txtWood",
  [Npc.emSERIES_TYPE.SERIES_WATER] = "txtWater",
  [Npc.emSERIES_TYPE.SERIES_FIRE] = "txtFire",
  [Npc.emSERIES_TYPE.SERIES_EARTH] = "txtSoil"
}

function tbWnd:OnCreate()
  self.pPanel:SetActive(self.txtSoilLeft, false)
  self.pPanel:SetActive(self.txtFireLeft, false)
  self.pPanel:SetActive(self.txtWoodLeft, false)
  self.pPanel:SetActive(self.txtGoldLeft, false)
  self.pPanel:SetActive(self.txtWaterLeft, false)
  self.pPanel:SetActive(self.txtSoilRight, false)
  self.pPanel:SetActive(self.txtFireRight, false)
  self.pPanel:SetActive(self.txtWoodRight, false)
  self.pPanel:SetActive(self.txtGoldRight, false)
  self.pPanel:SetActive(self.txtWaterRight, false)
  self.pPanel:SetActive(self.txtTitle, false)
  self.tbAttrScrollViewLeft = Ui:NewScrollView(self, self.attrTemplateLeft)
  self.tbAttrSuitScrollViewLeft = Ui:NewScrollView(self, self.attrSuitTemplateLeft)
  self.tbAttrScrollViewRight = Ui:NewScrollView(self, self.attrTemplateRight)
  self.tbAttrSuitScrollViewRight = Ui:NewScrollView(self, self.attrSuitTemplateRight)
end

function tbWnd:OnOpen()
  self.nOriBlessingId = Blessing:GetBlessingId(me)
  self.nNewBlessingId = Blessing:GetTempBlessingId(me)
  self.pPanel:SetActive(self.wndConfirm, false)
  self:UpdateUI()
end

function tbWnd:UpdateUI()
  self.tbBlessingSeriesLeft = Blessing:ParseBlessingId2Table(self.nOriBlessingId)
  self.tbBlessingElementKeyLeft = {}
  for k, id in ipairs(self.tbBlessingSeriesLeft) do
    local szTemplateKey = self[string.format("%s%s", tbSeries2Txt[id], "Left")]
    local szKey = self.pPanel:CloneObject(szTemplateKey, string.format("%s%d", "Element", k))
    table.insert(self.tbBlessingElementKeyLeft, szKey)
  end
  self.tbBlessingSeriesRight = Blessing:ParseBlessingId2Table(self.nNewBlessingId)
  self.tbBlessingElementKeyRight = {}
  for k, id in ipairs(self.tbBlessingSeriesRight) do
    local szTemplateKey = self[string.format("%s%s", tbSeries2Txt[id], "Right")]
    local szKey = self.pPanel:CloneObject(szTemplateKey, string.format("%s%d", "Element", k))
    table.insert(self.tbBlessingElementKeyRight, szKey)
  end
  local nSuitSeriesLeft, nSuitSeriesRight
  if #self.tbBlessingSeriesLeft - 1 >= 2 then
    nSuitSeriesLeft = self.tbBlessingSeriesLeft[#self.tbBlessingSeriesLeft - 1]
  end
  if #self.tbBlessingSeriesRight - 1 >= 2 then
    nSuitSeriesRight = self.tbBlessingSeriesRight[#self.tbBlessingSeriesRight - 1]
  end
  for nSeries = Npc.emSERIES_TYPE.SERIES_METAL, Npc.emSERIES_TYPE.SERIES_EARTH do
    local txtSuitSeriesLeft = string.format("%s%d", "txtSuitSeriesLeft", nSeries)
    local txtSuitSeriesGrayLeft = string.format("%s%d", "txtSuitSeriesGrayLeft", nSeries)
    local bActive = nSeries == nSuitSeriesLeft
    self.pPanel:SetActive(self[txtSuitSeriesLeft], bActive)
    self.pPanel:SetActive(self[txtSuitSeriesGrayLeft], not bActive)
    local txtSuitSeriesRight = string.format("%s%d", "txtSuitSeriesRight", nSeries)
    local txtSuitSeriesGrayRight = string.format("%s%d", "txtSuitSeriesGrayRight", nSeries)
    local bActive = nSeries == nSuitSeriesRight
    self.pPanel:SetActive(self[txtSuitSeriesRight], bActive)
    self.pPanel:SetActive(self[txtSuitSeriesGrayRight], not bActive)
  end
  self:UpdateAttr()
end

function tbWnd:UpdateAttr()
  local tbAttrLeft, tbSuitAttrLeft = Blessing:GetBlessingAttrib(self.nOriBlessingId, me.nLevel)
  self.tbAttrScrollViewLeft:Clear()
  self.tbAttrSuitScrollViewLeft:Clear()
  for k, v in ipairs(tbAttrLeft) do
    self.tbAttrScrollViewLeft:SetElement(k, function(tbElement)
      self:SetAttr(tbElement, v)
    end)
  end
  for k, v in ipairs(tbSuitAttrLeft) do
    self.tbAttrSuitScrollViewLeft:SetElement(k, function(tbElement)
      self:SetAttr(tbElement, v)
    end)
  end
  local tbAttrRight, tbSuitAttrRight = Blessing:GetBlessingAttrib(self.nNewBlessingId, me.nLevel)
  self.tbAttrScrollViewRight:Clear()
  self.tbAttrSuitScrollViewRight:Clear()
  for k, v in ipairs(tbAttrRight) do
    self.tbAttrScrollViewRight:SetElement(k, function(tbElement)
      self:SetAttr(tbElement, v)
    end)
  end
  for k, v in ipairs(tbSuitAttrRight) do
    self.tbAttrSuitScrollViewRight:SetElement(k, function(tbElement)
      self:SetAttr(tbElement, v)
    end)
  end
end

function tbWnd:SetAttr(tbElement, tbAttr)
  local szName, nValue = FightSkill:GetMagicDescSplit(tbAttr.szAttribType, tbAttr.tbValue)
  tbElement.pPanel:Label_SetText("txtAttrib", szName)
  tbElement.pPanel:Label_SetText("txtValueAll", nValue)
end

function tbWnd:ClearBlessingElement()
  if self.tbBlessingElementKeyLeft then
    for _, szKey in ipairs(self.tbBlessingElementKeyLeft) do
      self.pPanel:DeleteObject(szKey)
    end
  end
  if self.tbBlessingElementKeyRight then
    for _, szKey in ipairs(self.tbBlessingElementKeyRight) do
      self.pPanel:DeleteObject(szKey)
    end
  end
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_BLESSING_CHANGE)
  self:ClearBlessingElement()
  self.tbAttrScrollViewLeft:Clear()
  self.tbAttrSuitScrollViewLeft:Clear()
  self.tbAttrScrollViewRight:Clear()
  self.tbAttrSuitScrollViewRight:Clear()
end
