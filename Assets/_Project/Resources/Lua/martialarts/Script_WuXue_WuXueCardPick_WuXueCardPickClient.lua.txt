WuXueCardPicker = WuXueCardPicker or {}

function WuXueCardPicker:OnLogin()
end

function WuXueCardPicker:RequestCardPick(szType)
  if not WuXueCardPicker:CheckCardPickMoney(szType) then
    return
  end
  RemoteServer.OnWuXueCardPickerRequest(szType)
end

function WuXueCardPicker:CheckCardPickMoney(szType)
  if "LimitedTenPick" == szType then
    if me.GetMoney("YuanBao") < WuXueCardPicker.Def.nGoldTenCost then
      self:NoMoney("YuanBao")
      return false
    end
  elseif "LimitedPick" == szType and me.GetMoney("YuanBao") < WuXueCardPicker.Def.nGoldCost then
    self:NoMoney("YuanBao")
    return false
  end
  return true
end

function WuXueCardPicker:OnPickResult(tbItem, szType)
  Ui:CloseWindow("UIMartialArts")
  Ui:OpenWindowAsync("UIMartialArtsPickingResult", tbItem, szType)
end

function WuXueCardPicker:OnTenPickResult(tbItems, szType)
  Ui:CloseWindow("UIMartialArts")
  Ui:OpenWindowAsync("UIMartialArtsPickingResult", tbItems, szType)
end

function WuXueCardPicker:NoMoney(szType)
  local function funcGoldOK()
    GainCurrency:DirectJump("YuanBao")
  end
  
  local function funcCancel()
  end
  
  Ui:OpenWindowAsync("UIMessageBox", i18n.EasyGet("\229\133\131\229\174\157\228\184\141\232\182\179\239\188\140\230\152\175\229\144\166\229\137\141\229\190\128\232\142\183\229\143\150\239\188\159"), {
    {funcGoldOK},
    {funcCancel}
  }, {
    i18n.szOk,
    i18n.szCancel
  })
end

function WuXueCardPicker:SetLimitedPoolActivityState(bOpenActivity, tbUpPartnerInfo, nEndTime)
  WuXueCardPicker.bIsInLimitedPoolActivity = bOpenActivity
  WuXueCardPicker.tbLimitedPoolUpInfo = tbUpPartnerInfo
  WuXueCardPicker.nLimitedPoolActivityEndTime = nEndTime
  Ui:CloseWindow("UIMartialArtsPickingResult")
end

function WuXueCardPicker:CheckRedPoint()
end

function WuXueCardPicker:ResetRedPointCheckTimer()
end

WuXueCardPicker.LimitedPickData = WuXueCardPicker.LimitedPickData or {}

function WuXueCardPicker.LimitedPickData:SetNotShowValue(bNotShowToday)
  if bNotShowToday then
    LocalData:SetRoleNumber("WuXuePartner_CloseCardPickerNoticeTime", GetTime())
  end
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, WuXueCardPicker.OnLogin, WuXueCardPicker)
