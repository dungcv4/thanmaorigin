function tbWnd:OnCreate()
  self:SetClickEmptyToClose(tbControls.goWndTips)
  self.nCurSelectSkillIndex = 1
  self.nCurSelectWuXueId = 0
  self.tbInfoSkillListScrollView = Ui:NewScrollView(self, tbControls.goLeftToggleInfo, function(tbOwner, nIndex, element)
    self.nCurSelectSkillIndex = nIndex
    self:UpdateSkillInfo(nIndex)
    self:UpdateHuShiInfo(nIndex)
  end, false, false)
  self.tbCenterStarGridView = Ui:NewScrollView(self, tbControls.goStarGrid)
  self.tbCenterStarBgGridView = Ui:NewScrollView(self, tbControls.goStarGridBg)
  self.tbRightStarGridView = Ui:NewScrollView(self, tbControls.goRightStarGrid)
end

function tbWnd:OnOpen()
  for i = 1, 4 do
    self.tbInfoSkillListScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      if 1 == i then
        pPanel:Label_SetText("Text", i18n.EasyGet("\228\184\187\229\138\168\230\138\128\232\131\189"))
        pPanel:Label_SetText("Selected/Text", i18n.EasyGet("\228\184\187\229\138\168\230\138\128\232\131\189"))
      else
        pPanel:Label_SetText("Text", i18n.EasyGet("\232\162\171\229\138\168\230\138\128\232\131\189"))
        pPanel:Label_SetText("Selected/Text", i18n.EasyGet("\232\162\171\229\138\168\230\138\128\232\131\189"))
      end
    end)
  end
  self.tbInfoSkillListScrollView:SelectElement(1)
  self.pPanel:Label_SetText(tbControls.txtStar, i18n.EasyGet("\230\152\159\231\186\167"))
end

function tbWnd:UpdateSkillInfo(nIndex)
  local tbWuXueSkillInfo = WuXue:GetAllWuXueInstalled(me)
  local nWuXueId = tbWuXueSkillInfo[nIndex]
  self.tbCenterStarGridView:Clear()
  self.tbCenterStarBgGridView:Clear()
  self.tbRightStarGridView:Clear()
  for i = 1, 6 do
    self.tbCenterStarBgGridView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
    end)
  end
  if nWuXueId and nWuXueId > 0 then
    self.nCurSelectWuXueId = nWuXueId
    local nSkillId = 0
    local nLearnState = WuXue:GetWuXueSkillLearnState(me, nWuXueId)
    local nActiveSkillId, nPassiveSkillId = WuXue:GetSkillIdByWuXueId(nWuXueId)
    if 1 == nIndex then
      nSkillId = nActiveSkillId
    else
      nSkillId = nPassiveSkillId
    end
    local szWuXueName = WuXue.tbWuXueSkillInfo[nWuXueId].szWuXueName
    local szSkillName = FightSkill:GetSkillName(nSkillId)
    local szSkillIcon = FightSkill:GetSkillIconAtlas(nSkillId)
    self.pPanel:SetActive(tbControls.goSkillBG, true)
    self.pPanel:SetActive(tbControls.btnSkillCenter, true)
    self.pPanel:SetActive(tbControls.btnSkillCenterNoSkill, false)
    self.pPanel:Sprite_SetSprite(tbControls.btnSkillCenter, szSkillIcon)
    self.pPanel:Label_SetText(tbControls.txtName, i18n.Parse(szWuXueName))
    self:SetSkillDetail(true, nWuXueId, nSkillId)
    if nLearnState > 1 then
      for i = 2, nLearnState do
        self.tbCenterStarGridView:SetElement(i, function(tbElement)
          local pPanel = tbElement.pPanel
        end)
        self.tbRightStarGridView:SetElement(i, function(tbElement)
          local pPanel = tbElement.pPanel
        end)
      end
    end
  else
    self.nCurSelectWuXueId = 0
    self.pPanel:SetActive(tbControls.goSkillBG, false)
    self.pPanel:SetActive(tbControls.btnSkillCenter, false)
    self.pPanel:SetActive(tbControls.btnSkillCenterNoSkill, true)
    self.pPanel:Label_SetText(tbControls.txtName, i18n.EasyGet("\230\156\170\229\173\166\228\185\160\228\187\187\228\189\149\230\138\128\232\131\189"))
    self:SetSkillDetail(false, 0, 0)
  end
end

function tbWnd:UpdateHuShiInfo(nIndex)
  for i = 1, 5 do
    local tbItemId = WXHuShi:GetWXHuShiData(me, nIndex, i)
    if tbItemId[1] <= 0 then
      self.pPanel:SetActive(tbControls["goEmpty0" .. i], false)
    else
      local szIconAtlas, szIconSprite = self:GetItemIconPath(tbItemId[1])
      self.pPanel:SetActive(tbControls["goEmpty0" .. i], true)
      self.pPanel:Sprite_SetSprite(tbControls["goEmpty0" .. i] .. "/btnSkill", szIconSprite)
    end
  end
end

function tbWnd:SetSkillDetail(bShow, nWuXueId, nActiveSkillId)
  if not bShow then
    self.pPanel:SetActive(tbControls.goRightImgEmpty, true)
    self.pPanel:SetActive(tbControls.goRightSkillInfo, false)
    return
  else
    local nWuXueActiveSkillId, nWuXuePassiveSkillId = WuXue:GetSkillIdByWuXueId(nWuXueId)
    local szWuXuename = WuXue.tbWuXueSkillInfo[nWuXueId].szWuXueName
    local szSkillIcon = FightSkill:GetSkillIconAtlas(nActiveSkillId)
    local nLearnState = WuXue:GetWuXueSkillLearnState(me, nWuXueId)
    self.pPanel:SetActive(tbControls.goRightImgEmpty, false)
    self.pPanel:SetActive(tbControls.goRightSkillInfo, true)
    self.pPanel:Label_SetText(tbControls.txtRightSkillName, i18n.Parse(szWuXuename))
    self.pPanel:Sprite_SetSprite(tbControls.imgRightSkillIcon, szSkillIcon)
    local tbActiveSkillInfo = WuXue:GetPlayerOneSkillInfoWithoutConfig(nWuXueActiveSkillId, nLearnState)
    local tbPassiveSkillInfo = WuXue:GetPlayerOneSkillInfoWithoutConfig(nWuXuePassiveSkillId, nLearnState)
    local szDescText1, szDescText2
    if nWuXuePassiveSkillId == nActiveSkillId then
      szDescText1 = WuXue:GenerateSkillDescText(tbPassiveSkillInfo, true, true, nWuXuePassiveSkillId == nActiveSkillId)
      szDescText2 = WuXue:GenerateSkillDescText(tbActiveSkillInfo, true, true, nWuXueActiveSkillId == nActiveSkillId)
    else
      szDescText1 = WuXue:GenerateSkillDescText(tbActiveSkillInfo, true, true, nWuXueActiveSkillId == nActiveSkillId)
      szDescText2 = WuXue:GenerateSkillDescText(tbPassiveSkillInfo, true, true, nWuXuePassiveSkillId == nActiveSkillId)
    end
    local szDescText = szDescText1 .. [[


]] .. szDescText2
    self.pPanel:Label_SetText(tbControls.txtSkillDesc, szDescText)
  end
end

function tbWnd:GetItemIconPath(nItemTemplateId)
  local tbBaseProp = Item:GetItemBaseProp(nItemTemplateId)
  if not tbBaseProp then
    return nil, nil
  end
  local szName, nIcon, nView, nQuality = KItem.GetItemShowInfo(nItemTemplateId, me.nFaction)
  if not nIcon then
    return nil, nil
  end
  local szIconAtlas, szIconSprite = Item:GetIcon(nIcon)
  return szIconAtlas, szIconSprite
end
