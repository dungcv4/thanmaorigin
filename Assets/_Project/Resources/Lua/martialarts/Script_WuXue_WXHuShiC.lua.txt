function WXHuShi:GetHuShiList(nSeries)
  local tbItemList = me.GetItemListInBag("WXHuShiItem") or {}
  local tbRet = {}
  for _, pItem in pairs(tbItemList) do
    local nItemSeries = self:GetHuShiSeries(pItem.dwTemplateId)
    if nItemSeries == nSeries then
      table.insert(tbRet, pItem)
    end
  end
  
  local function pFuncSort(pItemA, pItemB)
    local nEnhanceLevelA = Item:GetClass("WXHuShiItem"):GetEnhanceLevel(pItemA)
    local nEnhanceLevelB = Item:GetClass("WXHuShiItem"):GetEnhanceLevel(pItemB)
    if nEnhanceLevelA ~= nEnhanceLevelB then
      return nEnhanceLevelA > nEnhanceLevelB
    end
    if pItemA.nLevel ~= pItemB.nLevel then
      return pItemA.nLevel > pItemB.nLevel
    end
    if pItemA.nQuality ~= pItemB.nQuality then
      return pItemA.nQuality > pItemB.nQuality
    end
    if pItemA.dwTemplateId ~= pItemB.dwTemplateId then
      return pItemA.dwTemplateId < pItemB.dwTemplateId
    end
    return pItemA.dwId < pItemB.dwId
  end
  
  table.sort(tbRet, pFuncSort)
  return tbRet
end

function WXHuShi:HSIdentifyReqC(nItemId, nCount)
  if not Lib:IsInteger(nItemId) then
    return
  end
  RemoteServer.OnWXHuShiRequest("HSIdentifyReqS", nItemId, nCount)
end

end
