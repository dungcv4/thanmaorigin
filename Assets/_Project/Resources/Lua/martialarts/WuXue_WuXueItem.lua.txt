local tbItem = Item:GetClass("WuXueItem")
tbItem.PARAM_ITEM_TYPE = 1

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function fnCompose()
    local nWuXueId = WuXue.tbWuXueItemTemplateId2WuXueId[nItemTemplateId]
    local nLearnState = WuXue:GetWuXueSkillLearnState(me, nWuXueId)
    if 0 == nLearnState then
      WuXue:RequestLearnWuXue(nWuXueId)
    else
      Ui:OpenWindowAsync("UIMartialArtsSkill")
    end
  end
  
  return {
    szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnFirst = fnCompose
  }
end
