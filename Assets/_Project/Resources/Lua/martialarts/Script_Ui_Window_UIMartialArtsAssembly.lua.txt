local tbWnd = Ui:CreateWindow("UIMartialArtsAssembly")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  goList = "node/imgBG/Content/Scroll View",
  goTemplate = "node/imgBG/Content/Scroll View/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbSkillListScrollView = Ui:NewScrollView(self, tbControls.goTemplate, nil, false, false)
end

function tbWnd:OnOpen(nIndex)
  local tbWuXueLearned = WuXue:GetWuXueLearnedCount(me)
  local i = 0
  for _, tbInfo in pairs(tbWuXueLearned) do
    i = i + 1
    self.tbSkillListScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local nWuXueId = tbInfo.nWuXueId
      local nStar = tbInfo.nStar - 1
      local szWuXueName = WuXue.tbWuXueSkillInfo[nWuXueId].szWuXueName
      local nItemTemplateId = WuXue.tbWuXueSkillInfo[nWuXueId].nItemTemplateId
      pPanel:Label_SetText("txtName", i18n.Parse(szWuXueName))
      local nInstallState, nType = WuXue:CheckWuXueInstalled(me, nWuXueId)
      if nInstallState then
        if 1 == nType then
          pPanel:Label_SetText("txtState", i18n.EasyGet("\229\183\178\229\186\148\231\148\168\228\184\187\229\138\168\230\138\128\232\131\189"))
        else
          pPanel:Label_SetText("txtState", i18n.EasyGet("\229\183\178\229\186\148\231\148\168\232\162\171\229\138\168\230\138\128\232\131\189"))
        end
        pPanel:SetActive("imgMark", true)
        pPanel:SetActive("ImgSelect", true)
      else
        pPanel:Label_SetText("txtState", "")
        pPanel:SetActive("imgMark", false)
        pPanel:SetActive("ImgSelect", false)
      end
      for i = 1, 6 do
        if nStar >= i then
          pPanel:SetActive(string.format("StarGrid/Start0%d/imgLight", i), true)
        end
      end
      pPanel:Button_BindEvent("btnOK", function()
        if 1 == nIndex then
          WuXue:RequestInstallWuXueToActiveSkill(nWuXueId)
        else
          WuXue:RequestInstallWuXueToPassiveSkill(nWuXueId, nIndex - 1)
        end
      end)
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(nItemTemplateId, 1)
      itemGrid:SetClickEnable(false)
    end)
  end
end
