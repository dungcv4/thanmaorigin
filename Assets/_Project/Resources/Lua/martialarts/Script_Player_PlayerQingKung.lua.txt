function PlayerQingKung:Init()
  self.nQingKungRequestDownHorseTime = 0
end

function PlayerQingKung:CanDoQingKung()
  local nDoing = me.GetDoing()
  if nDoing ~= Npc.Doing.stand and nDoing ~= Npc.Doing.run and nDoing ~= Npc.Doing.sit and nDoing ~= Npc.Doing.common and nDoing ~= Npc.Doing.skill then
    return false, PlayerQingKung.tbDiscirble.sForbinUse
  end
  if me.nLevel < PlayerQingKung.nNeedLevel then
    return false, PlayerQingKung.tbDiscirble.sLevelLimit
  end
  local qr = me.GetNpc().CanDoQingKung()
  if qr ~= PlayerQingKung.CanQingKungResult.qrSuccess then
    return false, self:GetQingKungErrMsg(qr)
  end
  if Map:IsForbidQingKung(me.nMapTemplateId) then
    return false, i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\230\151\160\230\179\149\228\189\191\231\148\168\232\189\187\229\138\159")
  end
  if me.GetNpc().HaveRefFlag(Npc.RefFlagType.npc_forbid_move) then
    return false, nil
  end
  if Player:IsForbinQingkungIcon() then
    return false, nil
  end
  self.nQingKungRequestDownHorseTime = self.nQingKungRequestDownHorseTime or 0
  if 1 == me.nRideState then
    local nCurTime = GetTime()
    if nCurTime >= self.nQingKungRequestDownHorseTime + Env.GAME_FPS * 2 then
      self.nQingKungRequestDownHorseTime = nCurTime
      SetRideState(0)
    end
    return false, nil
  end
  return true, ""
end

function PlayerQingKung:DoQingKung(bCastByCommand)
  local bCanQK, sMsg = self:CanDoQingKung()
  if not bCanQK then
    if sMsg and "" ~= sMsg then
      me.CenterMsg(sMsg)
    end
    if not bCastByCommand then
      me.GetNpc().SetQingKungCommandCache()
    end
    return false
  end
  RemoteServer.CheckQingKung()
  return true
end

function PlayerQingKung:DoQingKunResult(bResult)
  local pMeNpc = me.GetNpc()
  if not pMeNpc then
    return
  end
  if not bResult then
    local nCommandType = pMeNpc.GetLastActionCommandType()
    if nCommandType and nCommandType == Npc.NpcActionCommandType.emNpcActionCommand_UseQingKung then
      pMeNpc.RollBackLastActionCommand()
    end
    return
  end
  local bCanQK, sMsg = self:CanDoQingKung()
  if not bCanQK then
    if sMsg and "" ~= sMsg then
      me.CenterMsg(sMsg)
    end
    return false
  end
  me.GetNpc().DoQingKung()
  return true
end

function PlayerQingKung:ChangeEquipMaskState()
  self.nQingKungRequestDownHorseTime = 0
end

function PlayerQingKung:UpdateQingKungCD(nCoolTime)
  EventNotify.OnNotify(EventNotify.emNOTIFY_UPDATE_QINGKUNG_CD, nCoolTime)
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_EQUIP_MASK_STATE, PlayerQingKung)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_EQUIP_MASK_STATE, PlayerQingKung.ChangeEquipMaskState, PlayerQingKung)
PlayerQingKung:Init()
