Require("CommonScript/WuXue/WuXueCfg.lua")
WuXue = WuXue or {}
WuXue.Def = WuXue.Def or {}
WuXue.Def.nSkillGroupKey = GetGroupKey("WuXueSkill")
WuXue.Def.tbSkillGroup = {
  CurActiveSkill = 1,
  CurPassiveSkill1 = 2,
  CurPassiveSkill2 = 3,
  CurPassiveSkill3 = 4
}
WuXue.Def.nDataGroupKey = GetGroupKey("WuXueData")
WuXue.Def.nMaxStar = 6
WuXue.Def.tbPassiveSkillIndex = {
  [1] = WuXue.Def.tbSkillGroup.CurPassiveSkill1,
  [2] = WuXue.Def.tbSkillGroup.CurPassiveSkill2,
  [3] = WuXue.Def.tbSkillGroup.CurPassiveSkill3
}

function WuXue:LoadWuXueSkillSetting()
  local tbWuXueSkillInfo = Lib:EasyLoadTabFile("Setting/WuXue/WuXueSkill.tab", {"nId"})
  self.tbWuXueSkillInfo = {}
  self.tbWuXueItemTemplateId2WuXueId = {}
  for nId, tbInfo in pairs(tbWuXueSkillInfo) do
    self.tbWuXueSkillInfo[nId] = tbInfo
    self.tbWuXueItemTemplateId2WuXueId[tbInfo.nItemTemplateId] = nId
  end
  self.tbWuXueSkillInfo = SetReadOnly(self.tbWuXueSkillInfo)
  self.tbWuXueItemTemplateId2WuXueId = SetReadOnly(self.tbWuXueItemTemplateId2WuXueId)
end

function WuXue:LoadWuXueExtraPassiveSkillList()
  if not self.Cfg.WuXueStar2SkillExtraPassiveSkill then
    return
  end
  if not self.Cfg.WuXueStar2SkillLevelDesc then
    return
  end
  self.tbWuXueExtraPassiveSkillList = {}
  self.tbWuXueSkillLevelDescList = {}
  for nSkillId, szPassiveSkillData in pairs(self.Cfg.WuXueStar2SkillExtraPassiveSkill) do
    self.tbWuXueExtraPassiveSkillList[nSkillId] = {}
    local tbPassiveSkillList = Lib:SplitStr2Num(szPassiveSkillData, ",")
    for nWuXueLearnState, nPassiveSkillId in pairs(tbPassiveSkillList) do
      self.tbWuXueExtraPassiveSkillList[nSkillId][nWuXueLearnState] = nPassiveSkillId
    end
  end
  for nWuXueId, szDesc in pairs(self.Cfg.WuXueStar2SkillLevelDesc) do
    self.tbWuXueSkillLevelDescList[nWuXueId] = {}
    local tbDescList = Lib:SplitStr(szDesc, "|")
    for nWuXueLearnState, szDesc in pairs(tbDescList) do
      self.tbWuXueSkillLevelDescList[nWuXueId][nWuXueLearnState] = szDesc
    end
  end
  self.tbWuXueExtraPassiveSkillList = SetReadOnly(self.tbWuXueExtraPassiveSkillList)
  self.tbWuXueSkillLevelDescList = SetReadOnly(self.tbWuXueSkillLevelDescList)
end

WuXue:LoadWuXueSkillSetting()
WuXue:LoadWuXueExtraPassiveSkillList()
