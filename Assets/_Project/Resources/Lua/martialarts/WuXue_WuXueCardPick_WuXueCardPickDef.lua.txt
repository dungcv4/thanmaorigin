WuXueCardPicker = WuXueCardPicker or {}
WuXueCardPicker.Def = {
  OpenLevel = 25,
  nGoldCost = 200,
  nGoldTenCost = 1888,
  nCardPickMustSSSCount = 200,
  CARD_SAVE_GROUP = GetGroupKey("WuXueCardPick"),
  SAVE_CALC_SUBSIDY_SSS_COUNT = 1,
  SAVE_FREE_LIMITED_PICK_TIME = 2,
  SAVE_LIMITED_POOL_START_TIME = 3
}

function WuXueCardPicker:InitTab()
  self.tbWuXueCardPickerTab = {}
  self.tbWuXueCardPickerTab = Lib:LoadTabFileEx("Setting/WuXue/WuXueCardPicker.tab", "sddd", nil, {
    "PickerType",
    "ItemTemplateId",
    "ItemNum",
    "Prob"
  })
  self.tbWuXueCardPickerTab = SetReadOnly(self.tbWuXueCardPickerTab)
  self.tbWuXueCardPickerItemList = {}
  for _, tbInfo in pairs(self.tbWuXueCardPickerTab) do
    local szPickerType = tbInfo.PickerType
    self.tbWuXueCardPickerItemList[szPickerType] = self.tbWuXueCardPickerItemList[szPickerType] or {}
    table.insert(self.tbWuXueCardPickerItemList[szPickerType], {
      tbInfo.ItemTemplateId,
      tbInfo.ItemNum
    })
  end
  self.tbWuXueCardPickerItemList = SetReadOnly(self.tbWuXueCardPickerItemList)
end

function WuXueCardPicker:InitLimitedPoolData()
  WuXueCardPicker.Def.tbLimitedPoolData = {}
  for nId, szData in pairs(WuXueCardPicker.Cfg.LimitedPool) do
    local tbData = Lib:SplitStr(szData, "|")
    local szPoolName = tbData[1]
    local nUpPartnerNum = tonumber(tbData[2])
    local nUpItemId = tonumber(tbData[3])
    local nUpItemCount = tonumber(tbData[4])
    local nSubUpItemId = tonumber(tbData[5])
    local nSubUpItemCount = tonumber(tbData[6])
    WuXueCardPicker.Def.tbLimitedPoolData[nId] = {
      szPoolName = szPoolName,
      nUpPartnerNum = nUpPartnerNum,
      nUpItemId = nUpItemId,
      nUpItemCount = nUpItemCount,
      nSubUpItemId = nSubUpItemId,
      nSubUpItemCount = nSubUpItemCount
    }
  end
end

WuXueCardPicker:InitTab()
WuXueCardPicker:InitLimitedPoolData()

function WuXueCardPicker:GetSecondsToNextWednesday4AM(timestamp)
  timestamp = timestamp or GetTime()
  local current = os.date("*t", timestamp)
  local currentWeekday = current.wday
  local targetWeekday = 4
  local daysDiff
  if currentWeekday < targetWeekday then
    daysDiff = targetWeekday - currentWeekday
  else
    daysDiff = 7 - currentWeekday + targetWeekday
  end
  if 0 == daysDiff then
    if current.hour < 4 or 4 == current.hour and current.min < 0 then
      daysDiff = 0
    else
      daysDiff = 7
    end
  elseif 0 == daysDiff and current.hour >= 4 then
    daysDiff = 7
  end
  local targetTime = {
    year = current.year,
    month = current.month,
    day = current.day + daysDiff,
    hour = 4,
    min = 0,
    sec = 0
  }
  local targetTimestamp = os.time(targetTime)
  return targetTimestamp - timestamp
end
