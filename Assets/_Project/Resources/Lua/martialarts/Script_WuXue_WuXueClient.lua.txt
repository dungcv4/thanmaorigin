WuXue = WuXue or {}

function WuXue:UseActiveSkill()
  if me.nFightMode == Npc.FIGHT_MODE.emFightMode_None then
    me.CenterMsg(i18n.EasyGet("\229\174\137\229\133\168\229\140\186\229\134\133\228\184\141\232\131\189\228\189\191\231\148\168\232\175\165\230\138\128\232\131\189"))
    return
  end
  local nWuXueId = me.GetUserValue(self.Def.nSkillGroupKey, self.Def.tbSkillGroup.CurActiveSkill)
  if nWuXueId <= 0 then
    me.CenterMsg("\230\178\161\230\156\137\228\184\187\229\138\168\230\138\128\232\131\189")
    return
  end
  local nActiveSkillId = WuXue:GetSkillIdByWuXueId(nWuXueId)
  if not nActiveSkillId or nActiveSkillId <= 0 then
    return
  end
  Operation:Attack(nActiveSkillId, false)
end

function WuXue:RequestInstallWuXueToActiveSkill(nWuXueId)
  RemoteServer.CallWuXueFunc("TryInstallWuXueToActiveSkill", nWuXueId)
end

function WuXue:RequestInstallWuXueToPassiveSkill(nWuXueId, nIndex)
  RemoteServer.CallWuXueFunc("TryInstallWuXueToPassiveSkill", nWuXueId, nIndex)
end

function WuXue:RequestLearnWuXue(nWuXueId)
  if nWuXueId <= 0 or not WuXue.tbWuXueSkillInfo[nWuXueId] then
    return
  end
  local nItemTemplateId = WuXue.tbWuXueSkillInfo[nWuXueId].nItemTemplateId
  local nItemCount = me.GetItemCountInBags(nItemTemplateId)
  if nItemCount <= 0 then
    me.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\228\184\141\232\182\179"))
    return
  end
  RemoteServer.CallWuXueFunc("TryLearnWuXue", nItemTemplateId)
end

function WuXue:RefreshWuXueUI(nType)
  if 1 == nType then
    Ui:OpenUpgradeEffect(i18n.EasyGet("\229\173\166\228\185\160\230\136\144\229\138\159"))
  elseif 2 == nType then
    Ui:OpenUpgradeEffect(i18n.EasyGet("\229\141\135\230\152\159\230\136\144\229\138\159"))
  end
  local tbWnd = Ui:GetWindow("UIMartialArtsAssembly")
  if tbWnd then
    tbWnd:Close()
  end
  local tbWnd2 = Ui:GetWindow("UIMartialArtsMain")
  if tbWnd2 then
    tbWnd2:UpdateSkillInfo(tbWnd2.nCurSelectSkillIndex)
  end
  local tbWnd3 = Ui:GetWindow("UIMartialArtsSkill")
  if tbWnd3 then
    tbWnd3:UpdateUI()
  end
end
