function Map:InitSetting()
  for _, tbMap in pairs(self.tbMapList) do
    if type(tbMap.TeamLimit) == "string" and tbMap.TeamLimit == "" then
      tbMap.TeamLimit = {}
    elseif type(tbMap.TeamLimit) == "string" then
      tbMap.TeamLimit = Lib:SplitStr(tbMap.TeamLimit, "|")
    end
  end
  for nTemplateId, tbMap in pairs(self.tbMapList) do
    if not self.tbMapListEx[nTemplateId] then
      self.tbMapListEx[nTemplateId] = {}
    end
    local szMiniMapInfoSettingPath = string.format("Setting/Map/%s/info/minimap_info.ini", tbMap.InfoFilePath)
    local tbMiniMapInfoSetting = Lib:LoadIniFile(szMiniMapInfoSettingPath).MiniMapInfo
    if tbMiniMapInfoSetting then
      self.tbMapListEx[nTemplateId].RawWidth = tonumber(tbMiniMapInfoSetting.RawWidth)
      self.tbMapListEx[nTemplateId].RawHeight = tonumber(tbMiniMapInfoSetting.RawHeight)
      self.tbMapListEx[nTemplateId].RawScale = tonumber(tbMiniMapInfoSetting.RawScale)
      self.tbMapListEx[nTemplateId].PadWidth = tonumber(tbMiniMapInfoSetting.PadWidth)
      self.tbMapListEx[nTemplateId].PadHeight = tonumber(tbMiniMapInfoSetting.PadHeight)
    end
    self.tbMapListEx[nTemplateId].tbDefaultPos = Lib:LoadTabFileEx(string.format("Setting/Map/%s/info/default_pos_info.txt", tbMap.LogicFilePath), "dd", nil, {"PosX", "PosY"})
  end
end

function Map:InitWorldMapSetting()
  local tbKey = {
    TemplateId = 1,
    MapType = 1,
    MapLevel = 1,
    EnterLevel = 1,
    bDynamicLevel = 1,
    LevelSettingKey
  }
  local tbFile = Lib:LoadTabFile("Setting/Map/worldmap.tab", tbKey)
  assert(tbFile)
  local tbSwitchMap = {}
  local tbUnOpenMap = {}
  for _, v in pairs(tbFile) do
    if v.TemplateId > 0 then
      local tbMap = {
        nTemplateId = v.TemplateId,
        nMapLevel = v.MapLevel,
        nEnterLevel = v.EnterLevel,
        szShowTimeFrame = v.ShowTimeFrame,
        szOpenTimeFrame = v.OpenTimeFrame,
        bDynamicLevel = v.bDynamicLevel,
        szLevelSettingKey = v.LevelSettingKey
      }
      if MODULE_GAMECLIENT then
        tbMap.szUIKey = v.UIKey
        tbMap.nMapType = v.MapType
      end
      tbSwitchMap[v.TemplateId] = tbMap
    else
      table.insert(tbUnOpenMap, v.UIKey)
    end
  end
  Map.tbSwitchMap = SetReadOnly(tbSwitchMap)
  Map.tbUnOpenMap = SetReadOnly(tbUnOpenMap)
end

function Map:InitDynamicLevelSetting()
  local tbLevelList = Lib:EasyLoadTabFile("Setting/Map/MapDynamicLevel.tab")
  self.tbDynamicLevelList = {}
  self.tbDynamicLevelExpList = {}
  for _, tbInfo in ipairs(tbLevelList) do
    self.tbDynamicLevelList[tbInfo.szKey] = self.tbDynamicLevelList[tbInfo.szKey] or {}
    self.tbDynamicLevelExpList[tbInfo.szKey] = self.tbDynamicLevelExpList[tbInfo.szKey] or {}
    self.tbDynamicLevelExpList[tbInfo.szKey][tbInfo.nNpcLevel] = {
      tbInfo.nNpcExp,
      tbInfo.nBossNpcExp
    }
    table.insert(self.tbDynamicLevelList[tbInfo.szKey], tbInfo)
  end
  
  local function compareFunc(tbFirst, tbSecond)
    local nFirstTime = Lib:IsEmptyStr(tbFirst.szTimeFrame) and 0 or Lib:CalcTimeFrameOpenTime(tbFirst.szTimeFrame)
    local nSecondTime = Lib:IsEmptyStr(tbSecond.szTimeFrame) and 0 or Lib:CalcTimeFrameOpenTime(tbSecond.szTimeFrame)
    return nFirstTime > nSecondTime
  end
  
  for _, tbList in pairs(self.tbDynamicLevelList) do
    table.sort(tbList, compareFunc)
  end
  self.tbDynamicLevelList = SetReadOnly(self.tbDynamicLevelList, true)
  self.tbDynamicLevelExpList = SetReadOnly(self.tbDynamicLevelExpList, true)
end

function Map:GetEnterLevel(nTemplateId)
  local tbSwitchMapSetting = Map.tbSwitchMap[nTemplateId]
  if tbSwitchMapSetting and tbSwitchMapSetting.bDynamicLevel > 0 then
    local nCurMaxLevel = 0
    if MODULE_GAMESERVER then
      nCurMaxLevel = GetMaxTotalLevel()
    end
    if MODULE_GAMECLIENT then
      nCurMaxLevel = Client.nServerMaxTotalLevel
    end
    if nCurMaxLevel <= tbSwitchMapSetting.nMapLevel then
      return tbSwitchMapSetting.nEnterLevel
    end
    return nCurMaxLevel - nCurMaxLevel % 10
  end
  return tbSwitchMapSetting and tbSwitchMapSetting.nEnterLevel or 0
end

function Map:GetMapLevel(nTemplateId)
  local tbSwitchMapSetting = Map.tbSwitchMap[nTemplateId]
  if tbSwitchMapSetting and tbSwitchMapSetting.bDynamicLevel > 0 then
    local nCurMaxLevel = 0
    if MODULE_GAMESERVER then
      nCurMaxLevel = GetMaxTotalLevel()
    end
    if MODULE_GAMECLIENT then
      nCurMaxLevel = Client.nServerMaxTotalLevel
    end
    if nCurMaxLevel <= tbSwitchMapSetting.nMapLevel then
      return tbSwitchMapSetting.nMapLevel
    end
    return nCurMaxLevel - nCurMaxLevel % 10
  end
  local nLevel = tbSwitchMapSetting and tbSwitchMapSetting.nMapLevel or 0
  if tbSwitchMapSetting and not Lib:IsEmptyStr(tbSwitchMapSetting.szLevelSettingKey) then
    return self:GetDynamicLevel(tbSwitchMapSetting.szLevelSettingKey, nLevel)
  end
  return nLevel
end

function Map:GetMiniMapInfo(nTemplateId)
  local tbMapSetting = Map:GetMapSetting(nTemplateId)
  local tbMapSettingEx = Map:GetMapSettingEx(nTemplateId)
  local szMiniMap = tbMapSetting.MiniMap or ""
  local tbMapInfo = Map:GetMapSetting(nTemplateId)
  local szResInfoPath = tbMapInfo and "Setting/Map/" .. tbMapInfo.InfoFilePath .. "/3d/info.txt"
  local nRealRegionX = 0
  local nRealRegionY = 0
  if Client.RepresentModule.IsResourceExist(szResInfoPath) then
    local tbLogicMapSetting = Lib:LoadIniFile(szResInfoPath).BaseInfo
    nRealRegionX = tbLogicMapSetting.RegionCountX
    nRealRegionY = tbLogicMapSetting.RegionCountY
  else
    local nRegionX, nRegionY = GetMapBaseInfo(nTemplateId)
    if nil ~= nRegionX and nil ~= nRegionY then
      nRealRegionX = nRegionX
      nRealRegionY = nRegionY
    end
  end
  local tbMiniMapSetting = {}
  tbMiniMapSetting.SizeX = nRealRegionX * 8 * 800 / 25
  tbMiniMapSetting.SizeY = nRealRegionY * 4 * 800 / 25
  return tbMiniMapSetting, szMiniMap, tbMapSettingEx
end

function Map:CheckCanLeave(nTemplateId)
  if not self.tbSafeMap then
    self.tbSafeMap = {
      [Battle.READY_MAP_ID] = 1,
      [TeamBattle.PRE_MAP_ID] = 1,
      [Fuben.FenglingduFuben.OUTSIDE_MAPID] = 1,
      [608] = 1,
      [FactionBattle.Cfg.PREPARE_MAP_TAMPLATE_ID] = 1,
      [MultiPvP.Cfg.PRE_MAP_ID] = 1,
      [BattleRoyaleDef.MATCH_MAP_TEMPLATE_ID] = 1,
      [BattleRoyaleDef.ROUND_MAP_TEMPLATE_ID] = 1
    }
    for nDomainMapTemplate = 501, 513 do
      self.tbSafeMap[nDomainMapTemplate] = 1
    end
  end
  if not self.tbIsMapCanLeave then
    self.tbIsMapCanLeave = {
      [TeamBattle.PRE_MAP_ID] = function()
        return TeamBattle:IsPreMapCanLeave()
      end
    }
  end
  return nTemplateId and self.tbSafeMap[nTemplateId]
end

function Map:GetMapNpcDlgPos(nMapTemplateId, nNpcTemplateId, nXPos, nYPos)
  local nDlgX = 0
  local nDlgY = 0
  if not nMapTemplateId or nMapTemplateId < 0 then
    return nDlgX, nDlgY
  end
  if not nNpcTemplateId or nNpcTemplateId < 0 then
    return nDlgX, nDlgY
  end
  local tbMapNpcInfo = Map:GetMapNpcInfoByNpcTemplate(nMapTemplateId, nNpcTemplateId)
  if not tbMapNpcInfo then
    return nDlgX, nDlgY
  end
  for _, info in ipairs(tbMapNpcInfo) do
    if nXPos == info.XPos and nYPos == info.YPos then
      nDlgX = tbMapNpcInfo[1].DlgXPos or 0
      nDlgY = tbMapNpcInfo[1].DlgYPos or 0
      break
    end
  end
  return nDlgX, nDlgY
end

local tbNormalMapType = {
  fight = true,
  city = true,

}

function Map:IsInActivity(pPlayer)
  if not pPlayer then
    return false
  end
  local nMapTemplateId = pPlayer.nMapTemplateId
  local szMapType = Map:GetClassDesc(nMapTemplateId)
  return not tbNormalMapType[szMapType]
end

function Map:CanSwitchMap(nPlayerLevel, nMapTemplateId)
  local tbSwitchMapSetting = Map.tbSwitchMap[nMapTemplateId]
  if not tbSwitchMapSetting then
    return true
  end
  local nEnterLevel = Map:GetEnterLevel(nMapTemplateId) or 0
  if nPlayerLevel < nEnterLevel then
    local szMsg
    if ZongShiSkill:IsZongShiByLevel(nEnterLevel) then
      szMsg = i18n.EasyGet("\229\137\141\230\150\185\229\141\129\229\136\134\229\135\182\233\153\169\239\188\140\229\176\145\228\190\160\233\156\128\229\133\136\229\141\135\229\136\176\229\174\151\229\184\136{0}\231\186\167\229\134\141\230\157\165\229\176\157\232\175\149", ZongShiSkill:ToZongShiLevel(nEnterLevel))
    else
      szMsg = i18n.EasyGet("\229\137\141\230\150\185\229\141\129\229\136\134\229\135\182\233\153\169\239\188\140\229\176\145\228\190\160\233\156\128\229\133\136\229\141\135\229\136\176{0}\231\186\167\229\134\141\230\157\165\229\176\157\232\175\149", nEnterLevel)
    end
    return false, szMsg
  end
  local szOpenTimeFrame = tbSwitchMapSetting.szOpenTimeFrame
  if not szOpenTimeFrame or "" == szOpenTimeFrame then
    return true
  end
  if 1 ~= GetTimeFrameState(szOpenTimeFrame) then
    return false, i18n.EasyGet("\229\189\147\229\137\141\229\156\176\229\155\190\229\176\154\230\156\170\229\188\128\230\148\190")
  end
  return true
end

function Map:CheckNpcLevelConfig(tbNpcInfoList, tbMapInfo)
  local tbNpcInfoMap = {}
  local tbMapLevelNpcMap = {}
  local bRet = true
  for _, tbNpcInfo in ipairs(tbNpcInfoList) do
    local nMapTemplateId = tbNpcInfo[1]
    local nNpcId = tbNpcInfo[2]
    tbMapLevelNpcMap[nNpcId] = 1
    local szNpcFile = Map:GetMapPosFile(nMapTemplateId, "npc_info.txt")
    local tbNpcInfoSetting = Lib:LoadTabFileEx(szNpcFile, "ds", nil, {
      "NpcTemplateId",
      "LevelSettingKey"
    }, nil, nil, false)
    if not tbNpcInfoSetting then
      LogErr("[Map:CheckNpcLevelConfig] load failed!", szNpcFile)
      return false
    end
    local bError = true
    for _, tbV2 in pairs(tbNpcInfoSetting) do
      if tbV2.NpcTemplateId == nNpcId then
        bError = false
        if tbV2.LevelSettingKey ~= tbMapInfo.szKey then
          LogErr("[Map:CheckNpcLevelConfig] tbV2.LevelSettingKey ~= tbV.szKey", tbV2.LevelSettingKey, tbMapInfo.szKey, nNpcId, szNpcFile)
          bRet = false
        end
      end
      if tbV2.LevelSettingKey and not Lib:IsEmptyStr(tbV2.LevelSettingKey) then
        tbNpcInfoMap[tbV2.NpcTemplateId] = nMapTemplateId
      end
    end
    if bError then
      LogErr("[Map:CheckNpcLevelConfig] not find nNpcId", szNpcFile, nNpcId, szNpcFile)
      bRet = false
    end
  end
  for NpcTemplateId, nMapTemplateId in pairs(tbNpcInfoMap) do
    if not tbMapLevelNpcMap[NpcTemplateId] then
      local szNpcFile = Map:GetMapPosFile(nMapTemplateId, "npc_info.txt")
      LogErr("[Map:CheckNpcLevelConfig]  not tbMapLevelNpcMap[NpcTemplateId]", szNpcFile, NpcTemplateId)
      bRet = false
    end
  end
  return bRet
end

function Map:CheckMapLevelConfig()
  local bRet = true
  for _, tbInfo in pairs(self.tbDynamicLevelList) do
    for _, tbMapInfo in ipairs(tbInfo) do
      if tbMapInfo.szNpcInfo and not Lib:IsEmptyStr(tbMapInfo.szNpcInfo) then
        local tbszNpcInfo = Lib:SplitStr(tbMapInfo.szNpcInfo, ";")
        local tbNpcInfoList = {}
        for _, tbszValue in ipairs(tbszNpcInfo) do
          local tbNpcInfo = Lib:SplitStr2Num(tbszValue, "|")
          table.insert(tbNpcInfoList, tbNpcInfo)
        end
        if not self:CheckNpcLevelConfig(tbNpcInfoList, tbMapInfo) then
          bRet = false
        end
      end
    end
  end
  Log("-------------Map:CheckNpcLevelConfig() bRet = ", bRet)
  return bRet
end
