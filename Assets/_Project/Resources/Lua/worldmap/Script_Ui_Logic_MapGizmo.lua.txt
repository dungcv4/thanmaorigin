MapGizmo.Gizmos = MapGizmo.Gizmos or {}
MapGizmo.LineStyle = {DefaultLine = 0}
MapGizmo.CircleStyle = {SafeCircle = 1, PoisonCircle = 3}
MapGizmo.PointStyle = {DefaultPoint = 2}
MapGizmo.styleCheck = {
  line = {},
  circle = {},
  point = {}
}

function MapGizmo:Init()
  local tbStyleIDMap = {}
  for k, v in pairs(MapGizmo.LineStyle) do
    if tbStyleIDMap[v] then
      LogErr(string.format("[MapGizmo] style ID %d is duplicated!", v))
    end
    tbStyleIDMap[v] = true
    MapGizmo.styleCheck.line[v] = true
  end
  for k, v in pairs(MapGizmo.CircleStyle) do
    if tbStyleIDMap[v] then
      LogErr(string.format("[MapGizmo] style ID %d is duplicated!", v))
    end
    tbStyleIDMap[v] = true
    MapGizmo.styleCheck.circle[v] = true
  end
  for k, v in pairs(MapGizmo.PointStyle) do
    if tbStyleIDMap[v] then
      LogErr(string.format("[MapGizmo] style ID %d is duplicated!", v))
    end
    tbStyleIDMap[v] = true
    MapGizmo.styleCheck.point[v] = true
  end
end

function MapGizmo:SetMapData(_miniMapSizeX, _miniMapSizeY, _mapSizeX, _mapSizeY, _scaleFactor, _rawX, _rawY)
  Ui.MapGizmoManager.SetMapData(_miniMapSizeX, _miniMapSizeY, _mapSizeX, _mapSizeY, _scaleFactor, _rawX, _rawY)
end

function MapGizmo:DrawLine(beginX, beginY, endX, endY, style)
  if not MapGizmo.styleCheck.line[style] then
    LogErr(string.format("[MapGizmo] style %d is not a line!", style))
    return 0
  end
  return Ui.MapGizmoManager.DrawLine(beginX, beginY, endX, endY, style)
end

function MapGizmo:SetLine(gizmoId, beginX, beginY, endX, endY)
  Ui.MapGizmoManager.SetLine(gizmoId, beginX, beginY, endX, endY)
end

function MapGizmo:DrawCircle(srcX, srcY, radius, style)
  if not MapGizmo.styleCheck.circle[style] then
    LogErr(string.format("[MapGizmo] style %d is not a circle!", style))
    return 0
  end
  return Ui.MapGizmoManager.DrawCircle(srcX, srcY, radius, style)
end

function MapGizmo:SetCircle(gizmoId, srcX, srcY, radius)
  Ui.MapGizmoManager.SetCircle(gizmoId, srcX, srcY, radius)
end

function MapGizmo:DrawPoint(srcX, srcY, style)
  if not MapGizmo.styleCheck.point[style] then
    LogErr(string.format("[MapGizmo] style %d is not a point!", style))
    return 0
  end
  return Ui.MapGizmoManager.DrawPoint(srcX, srcY, style)
end

function MapGizmo:SetPointPos(gizmoId, srcX, srcY)
  Ui.MapGizmoManager.SetPointPos(gizmoId, srcX, srcY)
end

function MapGizmo:RemoveGizmo(gizmoId)
  Ui.MapGizmoManager.RemoveGizmo(gizmoId)
end

function MapGizmo:RemoveAll()
  Ui.MapGizmoManager.RemoveAll()
end

MapGizmo:Init()
