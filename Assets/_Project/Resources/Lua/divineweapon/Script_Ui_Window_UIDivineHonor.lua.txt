Require("Script/ShenBingPu/ShenBingPuC.lua")
local tbWnd = Ui:CreateClass("UIDivineHonor", "UIPanel")
local tbControls = {
  HonorTemplate = "imgBG/List/Viewport/Content/Template",
  ShenBing_Effect = "imgBG/List/Viewport/Content/Template/ShenBing_Effect",
  txtInfo = "imgBG/txtInfo"
}
tbWnd.tbControls = tbControls
local ShenBingPuC = Item.ShenBingPu

function tbWnd:UpdateView(pAsyncData)
  local nCurHonorLevel, nAllEnhanceLevel = ShenBingPuC:GetHornorLevelAsync(pAsyncData)
  if nCurHonorLevel and nAllEnhanceLevel then
    self:ShowHonorSmallPanel(nCurHonorLevel, nAllEnhanceLevel)
  end
end

function tbWnd:ShowHonorSmallPanel(nCurHonorLevel, nAllEnhanceLevel)
  if not self.tbHonorPanelView then
    self.tbHonorPanelView = Ui:NewScrollView(self, tbControls.HonorTemplate)
  end
  if #self.tbHonorPanelView:GetAllElement() <= 0 then
    for nHonorLevel, tbInfo in ipairs(Item.ShenBingPu.CFG.HONOR) do
      local szHonorTitle = i18n.Get(tbInfo[4])
      local szHonorDesc = i18n.Get(tbInfo[5])
      self.tbHonorPanelView:SetElement(nHonorLevel, function(tbElement)
        tbElement.pPanel:Label_SetText("txtName", szHonorTitle)
        tbElement.pPanel:Label_SetText("txtDesc", szHonorDesc)
        local nIndex = 0
        local szKey = string.format("ShenBing%02d", nIndex)
        while tbElement.pPanel:IsObjectExist("ShenBing_Effect/" .. szKey) do
          tbElement.pPanel:SetActive("ShenBing_Effect/" .. szKey, szKey == tbInfo[3])
          nIndex = nIndex + 1
          szKey = string.format("ShenBing%02d", nIndex)
        end
      end)
    end
  end
  local tbAllElement = self.tbHonorPanelView:GetAllElement()
  for nHonorLevel, tbElement in pairs(tbAllElement) do
    if nHonorLevel <= nCurHonorLevel then
      tbElement.pPanel:Label_SetText("txtUndown", i18n.EasyGet("<color=#138446>\229\183\178\232\190\190\230\136\144</color>"))
    else
      tbElement.pPanel:Label_SetText("txtUndown", i18n.EasyGet("\230\156\170\232\190\190\230\136\144"))
    end
  end
  self.pPanel:Label_SetText(tbControls.txtInfo, i18n.EasyGet("\229\189\147\229\137\141\229\133\187\230\136\144\239\188\154{0}", nAllEnhanceLevel))
end
