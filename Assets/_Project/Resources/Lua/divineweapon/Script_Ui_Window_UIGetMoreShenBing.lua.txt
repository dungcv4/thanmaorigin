local tbWnd = Ui:CreateClass("UIGetMoreShenBing", "UIPanel")
local tbControls = {
  Template = "imgBG/List/Viewport/Content/Template"
}
local ShenBingC = Item.ShenBingPu
tbWnd.tbControls = tbControls

function tbWnd:UpdateView()
  if not self.tbScrollView then
    self.tbScrollView = Ui:NewScrollView(self, tbControls.Template, self.OnClickJumpTo, false, true)
  end
  for nIndex, tbInfo in ipairs(ShenBingC.tbEquipGetWayCfg) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      self:SetElement(tbElement, tbInfo)
    end)
  end
end

function tbWnd:SetElement(tbElement, tbWayInfo)
  tbElement.pPanel:Label_SetText("txtName", i18n.Parse(tbWayInfo.szTitle))
  tbElement.pPanel:Label_SetText("txtDesc", i18n.Parse(tbWayInfo.szDesc))
end

function tbWnd:OnClickJumpTo(nIndex)
  if ShenBingC.tbEquipGetWayCfg[nIndex] then
    if ShenBingC.tbEquipGetWayCfg[nIndex].szJumpType == "OpenWindow" then
      Ui:OpenWindow(table.unpack(Lib:SplitStrTry2Num(ShenBingC.tbEquipGetWayCfg[nIndex].szJumpParam, "|")))
    elseif ShenBingC.tbEquipGetWayCfg[nIndex].szJumpType == "OpenLimitShop" then
      Shop:OpenLimitShop(table.unpack(Lib:SplitStrTry2Num(ShenBingC.tbEquipGetWayCfg[nIndex].szJumpParam, "|")))
    end
  end
end
