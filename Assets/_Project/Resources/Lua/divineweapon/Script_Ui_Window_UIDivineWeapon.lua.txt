tbWnd.LEFT_BTN_GROUP = {
  [1] = tbControls.Btn01,
  [2] = tbControls.Btn02,
  [3] = tbControls.Btn03
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OpenWindow("UIGeneralHelp", Item.ShenBingPu.CFG.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbLeftBtnGroup = Ui:NewButtonGroup(self, tbWnd.LEFT_BTN_GROUP, nil, self.OnLeftBtnClick)
  self.tbSuitPanel = Ui:CreatePanel("UIDivineSuit", tbControls.wndSuit, self)
  self.tbHonorPanel = Ui:CreatePanel("UIDivineHonor", tbControls.wndHonor, self)
  self.tbGetMorePanel = Ui:CreatePanel("UIGetMoreShenBing", tbControls.wndGetMore, self)
  self:SetClickEmptyToClose(tbControls.wndHonor)
  self:SetClickEmptyToClose(tbControls.wndSuit)
  self:SetClickEmptyToClose(tbControls.wndGetMore)
  self:RegistRedPoint()
end

function tbWnd:OnOpen(nPanelType)
  self.tbTabControl:UpdateButtonGroup()
  self:UpdateLeftBtnGroup()
  self:UpdateFightPowerInfo()
  self:UpdateSuitAttrBtn()
  self:UpdateHonorBtn()
  self:RefreshRightToggle()
  self:RefreshTabRedPoint()
  if nPanelType then
    self:DoChangePanel(nPanelType)
  else
    self:DoChangePanel(self.PANEL_TYPE_STRENGTHEN)
  end
end

function tbWnd:OnClose()
  self.tbTabControl:ClearSelected()
end

function tbWnd:UpdateLeftBtnGroup()
  for nIndex, szBtnPath in pairs(self.LEFT_BTN_GROUP) do
    local nSlot = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + nIndex - 1
    local szName = ShenBingPuC:GetFactionWeaponName(me.nFaction, me.nFactionSect, nSlot)
    self.pPanel:Label_SetText(szBtnPath .. "/TxtNormal", szName)
    self.pPanel:Label_SetText(szBtnPath .. "/OrderSelected/Text", szName)
  end
  self:UpdateLeftBtnLV()
  if self.tbLeftBtnGroup.nSelectIndex then
    self.tbLeftBtnGroup:Select(self.tbLeftBtnGroup.nSelectIndex)
  else
    self.tbLeftBtnGroup.nSelectIndex = 1
    self.tbLeftBtnGroup:Select(1)
  end
end

function tbWnd:UpdateLeftBtnLV()
  local tbAllWeapon = ShenBingPuC:GetAllUsingShenBingPu2(me)
  for nIndex, szBtnPath in pairs(self.LEFT_BTN_GROUP) do
    local nSlot = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + nIndex - 1
    if tbAllWeapon[nSlot] then
      self.pPanel:Label_SetText(szBtnPath .. "/ImgLV/TxtLV", ShenBingPuC:GetStarLevel(tbAllWeapon[nSlot]))
      self.pPanel:SetActive(szBtnPath .. "/ImgLV", true)
    else
      self.pPanel:SetActive(szBtnPath .. "/ImgLV", false)
    end
  end
end

function tbWnd:OnLeftBtnClick(nBtnIndex)
  if self.tbLeftBtnGroup.nSelectIndex and self.tbLeftBtnGroup.nSelectIndex ~= nBtnIndex then
    self.pPanel:SetActive(self.LEFT_BTN_GROUP[self.tbLeftBtnGroup.nSelectIndex] .. "/TxtNormal", true)
    self.pPanel:SetActive(self.LEFT_BTN_GROUP[self.tbLeftBtnGroup.nSelectIndex] .. "/OrderSelected", false)
  end
  self.pPanel:SetActive(self.LEFT_BTN_GROUP[nBtnIndex] .. "/TxtNormal", false)
  self.pPanel:SetActive(self.LEFT_BTN_GROUP[nBtnIndex] .. "/OrderSelected", true)
  self.tbLeftBtnGroup.nSelectIndex = nBtnIndex
  if self.nSubPanelType then
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "OnSelectWeaponChange", self.tbLeftBtnGroup.nSelectIndex)
  end
end

function tbWnd:UpdateFightPowerInfo()
  local nTotalFightPower = ShenBingPuC:CalcFightPower(me)
  self.pPanel:Label_SetText(tbControls.txtFight, i18n.EasyGet("\231\165\158\229\133\181\230\128\187\230\136\152\229\138\155\239\188\154{0}", nTotalFightPower))
end

function tbWnd:ShowGetMorePanel()
  self.tbGetMorePanel:UpdateView()
  self.pPanel:SetActive(tbControls.wndGetMore, true)
end

function tbWnd:UpdateSuitAttrBtn()
  local _, nSuitAttrLevel, szSuitAttrName, nAllStarLevel = ShenBingPuC:GetPlayerAllSuitAttr(me)
  if not Lib:IsEmptyStr(szSuitAttrName) then
    self.pPanel:Label_SetText(tbControls.btnSuit, string.format("%s\227\128\139", i18n.Get(szSuitAttrName)))
  else
    self.pPanel:Label_SetText(tbControls.btnSuit, i18n.EasyGet("\230\156\170\230\191\128\230\180\187\227\128\139"))
  end
  Ui:Button_BindEvent(self.pPanel, tbControls.btnSuit, self.ShowSuitAttrSmallPanel, self, nSuitAttrLevel, nAllStarLevel)
end

function tbWnd:ShowSuitAttrSmallPanel(nSuitAttrLevel, nAllStarLevel)
  self.tbSuitPanel:ShowSuitAttrSmallPanel(nSuitAttrLevel, nAllStarLevel)
  self.pPanel:SetActive(tbControls.wndSuit, true)
end

function tbWnd:UpdateHonorBtn()
  Ui:Button_BindEvent(self.pPanel, tbControls.btnHonor, self.ShowHonorSmallPanel, self)
end

function tbWnd:ShowHonorSmallPanel()
  local nCurHonorLevel, nAllEnhanceLevel = ShenBingPuC:GetHornorLevel(me)
  if nCurHonorLevel and nAllEnhanceLevel then
    self.tbHonorPanel:ShowHonorSmallPanel(nCurHonorLevel, nAllEnhanceLevel)
    self.pPanel:SetActive(tbControls.wndHonor, true)
  else
    self.pPanel:SetActive(tbControls.wndHonor, false)
  end
end

function tbWnd:RefreshRightToggle()
  for nType, tbInfo in pairs(self.PANEL_SETTING) do
    self.pPanel:SetActive(tbInfo.EFFECT, false)
  end
end

function tbWnd:OnPanelChange(nPanelType)
  for nType, tbInfo in pairs(self.PANEL_SETTING) do
    self.pPanel:SetActive(tbInfo.EFFECT, nPanelType == nType)
  end
  self.nSubPanelType = nPanelType
  self.tbTabControl:UpdateSubPanel(nPanelType, "OnPanelOpen", self.tbLeftBtnGroup.nSelectIndex)
  self.pPanel:SetActive(tbControls.txtTittle, nPanelType ~= self.PANEL_TYPE_COMPOSE)
  self.pPanel:SetActive(tbControls.btnSuit, nPanelType ~= self.PANEL_TYPE_COMPOSE)
  self.pPanel:SetActive(tbControls.btnHonor, nPanelType ~= self.PANEL_TYPE_COMPOSE)
end

function tbWnd:OnActivateSucc(nSlot)
  if self.nSubPanelType == self.PANEL_TYPE_STRENGTHEN then
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "UpdateMiddleAndRight", nSlot - Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + 1)
    self:UpdateFightPowerInfo()
  end
  self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint", nSlot)
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnEnhanceSucc(nSlot, nMainAttrIndex, nHistoryLevel, nAddLevel)
  if self.nSubPanelType == self.PANEL_TYPE_STRENGTHEN then
    local pUsingWeapon = ShenBingPuC:GetUsingShenBingPuBySlot(me, nSlot)
    if pUsingWeapon then
      local tbCurEnhanceAttrInfo = ShenBingPuC:GetAttrInfoByHoleIndex(me, pUsingWeapon, nMainAttrIndex, 2)
      local nPrevEnahcneVal1, nPrevEnahcneVal2, nPrevEnahcneVal3 = ShenBingPuC:GetAttrEnhanceValByLevel(nHistoryLevel, nSlot, nMainAttrIndex)
      local tbPrevEnhanceAttrInfo = {
        nAttrId = tbCurEnhanceAttrInfo.nAttrId,
        nAttrValue1 = nPrevEnahcneVal1,
        nAttrValue2 = nPrevEnahcneVal2,
        nAttrValue3 = nPrevEnahcneVal3
      }
      local _, nPrevEnhanceValue, nCurEnhanceValue, _ = ShenBingPuC:GetWeaponAttrDiff(tbPrevEnhanceAttrInfo, tbCurEnhanceAttrInfo)
      self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "OnEnhanceSucc", pUsingWeapon, nMainAttrIndex, nHistoryLevel, nAddLevel, nPrevEnhanceValue, nCurEnhanceValue)
      self:UpdateFightPowerInfo()
    end
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshActivateBtnRedPoint", nSlot)
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshStrengthBtnRedPoint", nSlot)
  end
  self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint", nSlot)
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnRecycleSucc(tbTemplateList)
  if self.nSubPanelType == self.PANEL_TYPE_BREAK_DOWN or self.nSubPanelType == self.PANEL_TYPE_REPLACE then
    local nTemplateId = tbTemplateList[1]
    if nTemplateId then
      local tbInfo = KItem.GetItemBaseProp(nTemplateId)
      local nSlot = 0
      if tbInfo.szClass == "ShenBingPuItem" then
        nSlot = tbInfo.nDetailType
      elseif tbInfo.szClass == "ShenBingPuSeed" then
        local tbClass = Item:GetClass("ShenBingPuSeed")
        nSlot = tbClass:GetSlot(nTemplateId) or 0
      end
      self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "UpdateMiddleAndRight", nSlot - Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + 1)
    end
  end
  self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint")
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnIdentifySucc(nSeedTemplateId, nNewWeaponId)
  if self.nSubPanelType == self.PANEL_TYPE_REPLACE then
    local tbClass = Item:GetClass("ShenBingPuSeed")
    local nSlot = tbClass:GetSlot(nSeedTemplateId) or 0
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "UpdateMiddleAndRight", nSlot - Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + 1)
  end
  self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint")
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnDelItem(dwItemId, dwTemplateId)
  if self.nSubPanelType == self.PANEL_TYPE_REPLACE then
    local tbInfo = KItem.GetItemBaseProp(dwTemplateId)
    local nSlot = 0
    if tbInfo.szClass == "ShenBingPuItem" then
      nSlot = tbInfo.nDetailType
    elseif tbInfo.szClass == "ShenBingPuSeed" then
      local tbClass = Item:GetClass("ShenBingPuSeed")
      nSlot = tbClass:GetSlot(dwTemplateId) or 0
    end
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "UpdateMiddleAndRight", nSlot - Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + 1)
  elseif self.nSubPanelType == self.PANEL_TYPE_COMPOSE and Item.ShenBingPu.tbEquipComposeCfg[dwTemplateId] then
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "OnDelItem", dwTemplateId)
  end
  self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint")
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnComposeDone(nResult, nNewWeaponId, nNewTemplateId, nMainItemId)
  if 0 == nResult then
    self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_COMPOSE, "OnComposeFail", nMainItemId)
  elseif 1 == nResult then
    self.tbTabControl:UpdateSubPanel(self.PANEL_TYPE_COMPOSE, "OnComposeSucc", nNewWeaponId)
  end
end

function tbWnd:OnSyncItem(dwItemId)
  local pItem = me.GetItemInBag(dwItemId)
  if not pItem then
    return
  end
  if not Item.ShenBingPu.tbEquipComposeCfg[pItem.dwTemplateId] then
    return
  end
  if self.nSubPanelType == self.PANEL_TYPE_COMPOSE then
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "OnSyncItem", dwItemId, pItem.dwTemplateId)
  end
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnSyncItemTimeLimit()
  if self.nSubPanelType == self.PANEL_TYPE_STRENGTHEN then
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "OnSyncItem")
  end
  self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint")
  self:RefreshTabRedPoint()
  self:RefreshBagEntranceRedPoint()
end

function tbWnd:OnUseWeapon(pEquip, nSlot)
  if ShenBingPuC:CheckIsValidSlot(nSlot) then
    if self.nSubPanelType == self.PANEL_TYPE_REPLACE then
      self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "UpdateMiddleInfo", nSlot)
      self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "UpdateRightInfo", nSlot)
      self:UpdateFightPowerInfo()
    end
    self:UpdateSuitAttrBtn()
    self:UpdateLeftBtnLV()
    self.tbTabControl:UpdateSubPanel(self.nSubPanelType, "RefreshSlotRedPoint")
    self:RefreshTabRedPoint()
    self:RefreshBagEntranceRedPoint()
  end
end

function tbWnd:DoChangePanel(nPanelType)
  if self.PANEL_SETTING[nPanelType] then
    self.tbTabControl:SelectPanel(nPanelType)
  end
  for nType, tbInfo in pairs(self.PANEL_SETTING) do
    self.pPanel:SetActive(tbInfo.EFFECT, nPanelType == nType)
  end
end

function tbWnd:OnDestroy()
  self:UnRegistRedPoint()
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.Btn01, {
    RedPointMgr.EVENT.ShenBingStrengthForSlot1,
    RedPointMgr.EVENT.ShenBingReplaceForSlot1,
    RedPointMgr.EVENT.ShenBingActivateForSlot1
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.Btn02, {
    RedPointMgr.EVENT.ShenBingStrengthForSlot2,
    RedPointMgr.EVENT.ShenBingReplaceForSlot2,
    RedPointMgr.EVENT.ShenBingActivateForSlot2
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.Btn03, {
    RedPointMgr.EVENT.ShenBingStrengthForSlot3,
    RedPointMgr.EVENT.ShenBingReplaceForSlot3,
    RedPointMgr.EVENT.ShenBingActivateForSlot3
  })
  RedPointMgr:ChangeRedPointPositionLeft(self.pPanel, tbControls.Btn01 .. "/RedPoint")
  RedPointMgr:ChangeRedPointPositionLeft(self.pPanel, tbControls.Btn02 .. "/RedPoint")
  RedPointMgr:ChangeRedPointPositionLeft(self.pPanel, tbControls.Btn03 .. "/RedPoint")
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.ToggleStrengthen, {
    RedPointMgr.EVENT.ShenBingStrengthForTab
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.ToggleReplace, {
    RedPointMgr.EVENT.ShenBingReplaceForTab
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.Btn01)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.Btn02)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.Btn03)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.ToggleStrengthen)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.ToggleReplace)
end

function tbWnd:RefreshTabRedPoint()
  for nPanelType, tbPanelCfg in pairs(self.PANEL_SETTING) do
    local pSubPanel = self.tbTabControl:GetSubPanel(nPanelType)
    if pSubPanel and pSubPanel.CheckTabRedPoint then
      local bRet = pSubPanel:CheckTabRedPoint()
      if bRet and not RedPointMgr:CheckEvent(tbPanelCfg.RedPoint) then
        RedPointMgr:LightRedPoint(tbPanelCfg.RedPoint)
      elseif not bRet and RedPointMgr:CheckEvent(tbPanelCfg.RedPoint) then
        RedPointMgr:FadeRedPoint(tbPanelCfg.RedPoint)
      end
    end
  end
end

function tbWnd:CheckHasRedPointInside()
  for _, tbPanelCfg in pairs(self.PANEL_SETTING) do
    local tbWndClass = Ui:GetClass(tbPanelCfg.CLASS)
    if tbWndClass and tbWndClass.CheckTabRedPoint then
      local bRet = tbWndClass:CheckTabRedPoint()
      if bRet then
        return true
      end
    end
  end
  return false
end

function tbWnd:RefreshBagEntranceRedPoint()
  local tbClass = Ui:GetClass("UIBagPanel")
  if tbClass and tbClass.RefreshShenBingEntranceRedPoint then
    tbClass:RefreshShenBingEntranceRedPoint()
  end
end
