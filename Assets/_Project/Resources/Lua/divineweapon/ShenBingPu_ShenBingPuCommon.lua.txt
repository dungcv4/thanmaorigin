function ShenBingPu:InitCfg()
  if self.bLoadConfig then
    Log("ShenBingPuCommon: Cfg Has been Loaded, do you want to reload?")
    return true
  end
  local szFilepath = "Setting/ShenBingPu/ShenBingPuAttrCfg.tab"
  self.tbAttrCfg = Lib:EasyLoadTabFile(szFilepath, {"nAttrId"})
  if not self.tbAttrCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  szFilepath = "Setting/ShenBingPu/ShenBingPuRandAttrGroupCfg.tab"
  local tbAttrGroupCfg = Lib:EasyLoadTabFile(szFilepath)
  if not tbAttrGroupCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  self.tbAttrGroupCfg = {}
  local tbAttrGroupWeightForCheck = {}
  for _, tbInfo in pairs(tbAttrGroupCfg) do
    self.tbAttrGroupCfg[tbInfo.nGroupId] = self.tbAttrGroupCfg[tbInfo.nGroupId] or {}
    tbAttrGroupWeightForCheck[tbInfo.nGroupId] = tbAttrGroupWeightForCheck[tbInfo.nGroupId] or 0
    if not self.tbAttrGroupCfg[tbInfo.nGroupId][tbInfo.nAttrId] then
      self.tbAttrGroupCfg[tbInfo.nGroupId][tbInfo.nAttrId] = tbInfo
      tbAttrGroupWeightForCheck[tbInfo.nGroupId] = tbAttrGroupWeightForCheck[tbInfo.nGroupId] + tbInfo.nWeight
    else
      Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, duplicate attr id in the same group, nGroupId = %d, nAttrId = %d", tbInfo.nGroupId, tbInfo.nAttrId)
      return false
    end
    if not self.tbAttrCfg[tbInfo.nAttrId] then
      Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, nAttrId exist in ShenBingPuRandAttrGroupCfg.tab not found in ShenBingPuAttrCfg.tab, nAttrId = %d", tbInfo.nAttrId)
      return false
    end
  end
  for nGroupId, nSumWeight in pairs(tbAttrGroupWeightForCheck) do
    if nSumWeight > self.GROUP_MAX_WEIGHT then
      Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, sum weight over than %d, nGroupId = %d", self.GROUP_MAX_WEIGHT, nGroupId)
      return false
    end
  end
  szFilepath = "Setting/ShenBingPu/ShenBingPuItemCfg.tab"
  self.tbItemCfg = Lib:EasyLoadTabFile(szFilepath, {
    "nTemplateId"
  })
  if not self.tbItemCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  szFilepath = "Setting/ShenBingPu/ShenBingPuFaction.tab"
  self.tbFactionCfg = Lib:EasyLoadTabFile(szFilepath, {"nFaction", "nSect"})
  if not self.tbFactionCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  szFilepath = "Setting/ShenBingPu/ShenBingPuGetWay.tab"
  self.tbEquipGetWayCfg = Lib:EasyLoadTabFile(szFilepath, nil)
  if not self.tbEquipGetWayCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  szFilepath = "Setting/ShenBingPu/ShenBingPuRandAttrNumCfg.tab"
  local tbRandAttrNumCfg = Lib:EasyLoadTabFile(szFilepath, nil)
  if not tbRandAttrNumCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  self.tbRandAttrNumCfg = {}
  for _, tbCfg in pairs(tbRandAttrNumCfg) do
    self.tbRandAttrNumCfg[tbCfg.nStarLevel] = self.tbRandAttrNumCfg[tbCfg.nStarLevel] or {}
    if not self.tbRandAttrNumCfg[tbCfg.nStarLevel][tbCfg.nQuality] then
      self.tbRandAttrNumCfg[tbCfg.nStarLevel][tbCfg.nQuality] = {}
      local nSumWeight = 0
      for i = 0, self.RAND_ATTR_MAX_NUM do
        local nProbility = tbCfg["nProbilityNum" .. i]
        local nFightPower = tbCfg["nFightPower" .. i]
        if nProbility and nFightPower then
          nSumWeight = nSumWeight + nProbility
          table.insert(self.tbRandAttrNumCfg[tbCfg.nStarLevel][tbCfg.nQuality], {
            nNum = i,
            nProbility = nProbility,
            nFightPower = nFightPower
          })
        else
          break
        end
      end
      if nSumWeight ~= self.RAND_ATTR_NUM_MAX_WEIGHT then
        Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, sum probility of rand attr num cfg is not equal to max probility, nStarLevel = %d, nQuality = %d, nSumWeight = %d", tbCfg.nStarLevel, tbCfg.nQuality, nSumWeight)
        return false
      end
    else
      Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, duplicate quality in the same star level, nStarLevel = %d, nQuality = %d", tbCfg.nStarLevel, tbCfg.nQuality)
      return false
    end
  end
  szFilepath = "Setting/ShenBingPu/ShenBingPuComposeCfg.tab"
  local tbEquipComposeCfg = Lib:EasyLoadTabFile(szFilepath, nil)
  if not tbEquipComposeCfg then
    Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, %s not found", szFilepath)
    return false
  end
  self.tbEquipComposeCfg = {}
  for _, tbCfg in pairs(tbEquipComposeCfg) do
    self.tbEquipComposeCfg[tbCfg.nEquipTemplateId1] = self.tbEquipComposeCfg[tbCfg.nEquipTemplateId1] or {}
    if not self.tbEquipComposeCfg[tbCfg.nEquipTemplateId1][tbCfg.nEquipTemplateId2] then
      local nSumWeight = 0
      for i = 1, self.COMPOSE_MAX_RESULT_NUM do
        local nComposeTemplateId = tbCfg["nComposeTemplateId" .. i]
        local nProbability = tbCfg["nProbability" .. i]
        if 0 == nComposeTemplateId and 0 == nProbability then
        elseif nComposeTemplateId > 0 and nProbability > 0 then
        else
          Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, invalid compose template id or probability, nTemplateId = %d, nIndex = %d", tbCfg.nEquipTemplateId, i)
          return false
        end
        nSumWeight = nSumWeight + nProbability
      end
      if nSumWeight > self.COMPOSE_MAX_WEIGHT then
        Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, sum weight of compose cfg is not equal to max weight, nTemplateId = %d, nSumWeight = %d", tbCfg.nEquipTemplateId, nSumWeight)
        return false
      end
      self.tbEquipComposeCfg[tbCfg.nEquipTemplateId1][tbCfg.nEquipTemplateId2] = tbCfg
    else
      Lib:LogErrFormat("ShenBingPuCommon: InitCfg failed, duplicate equip template id in the same compose cfg, nEquipTemplateId1 = %d, nEquipTemplateId2 = %d", tbCfg.nEquipTemplateId1, tbCfg.nEquipTemplateId2)
      return false
    end
  end
  self.tbSortIterationList = Lib:SetToList(ShenBingPuCFG.OpenTimeFrame)
  table.sort(self.tbSortIterationList, function(szIter1, szIter2)
    return Lib:CalcTimeFrameOpenTime(ShenBingPuCFG.OpenTimeFrame[szIter1]) < Lib:CalcTimeFrameOpenTime(ShenBingPuCFG.OpenTimeFrame[szIter2])
  end)
  self.tbSortIterationSet = {}
  for nIndex, szIter in ipairs(self.tbSortIterationList) do
    self.tbSortIterationSet[szIter] = nIndex
  end
  table.sort(ShenBingPuCFG.SUIT_ATTR, function(tbA, tbB)
    return tbA[1] < tbB[1]
  end)
  table.sort(ShenBingPuCFG.HONOR, function(tbA, tbB)
    return tbA[1] + tbA[2] < tbB[1] + tbB[2]
  end)
  self.bLoadConfig = true
  return true
end

ShenBingPu.bLoadConfig = false

function ShenBingPu:CheckCfg()
  return self.bInitRet

function ShenBingPu:CheckIsShenBingPu(pEquip)
  if not pEquip then
    return
  end
  local nSlot = Item:GetExtEquipSlot(pEquip)
  return self:CheckIsValidSlot(nSlot)
end

function ShenBingPu:CheckIsValidSlot(nSlot)
  if not Lib:IsInteger(nSlot) then
    return false
  end
  if nSlot < Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 or nSlot > Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON3 then
    return false
  else
    return true
  end
end

function ShenBingPu:CheckCanEnhance(pPlayer, nSlot, nMaterialId)
  if not self:CheckHasActivate(pPlayer, nSlot) then
    return false, i18n.EasyGet("\232\163\133\229\164\135\229\176\154\230\156\170\229\188\128\229\136\131")
  end
  local szMaterialIteration = self:GetMaterialIteration(nMaterialId)
  local szCurIteration = self:GetCurIteration()
  if Lib:IsEmptyStr(szMaterialIteration) then
    return false, ""
  elseif self.tbSortIterationSet[szCurIteration] > self.tbSortIterationSet[szMaterialIteration] then
    return false, i18n.EasyGet("\232\175\165\230\157\144\230\150\153\230\151\160\230\179\149\229\188\186\229\140\150\232\175\165\232\163\133\229\164\135")
  end
  local bFullMax = self:IsAllFixedAttrFullMax(pPlayer, nSlot)
  if bFullMax then
    return false, i18n.EasyGet("\232\175\165\232\163\133\229\164\135\229\188\186\229\140\150\229\183\178\232\190\190\228\184\138\233\153\144")
  else
    return true, ""
  end
end

function ShenBingPu:CheckHasActivate(pPlayer, nSlot)
  if not pPlayer then
    return
  end
  if not self:CheckIsValidSlot(nSlot) then
    return
  end
  local szActivateSaveId = string.format("WEAPON_%d_ACTIVATE", nSlot - Item.ShenBingPu.WEAPON_1 + 1)
  if self.SAVE_ID[szActivateSaveId] then
    return pPlayer.GetUserValue(self.SAVE_GROUP, self.SAVE_ID[szActivateSaveId]) == self.eSTATUS.ACTIVATED
  end
end

function ShenBingPu:CheckHasActivateAsync(pPlayerAsync, nSlot)
  if not self:CheckIsValidSlot(nSlot) then
    return
  end
  return pPlayerAsync.GetShenBingPuActivate(nSlot - Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + 1) == self.eSTATUS.ACTIVATED
end

function ShenBingPu:CheckEnoughMaterial(pPlayer, nMaterialId, nNeedNum)
  if not pPlayer or not Lib:IsInteger(nNeedNum) then
    return
  end
  local nHasMaterialNum = pPlayer.GetItemCountInBags(nMaterialId)
  if nNeedNum > nHasMaterialNum then
    return false, i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179")
  else
    return true, ""
  end
end

function ShenBingPu:CheckHasDailySaleCount(pPlayer)
  if not pPlayer then
    return
  end
  local nRemainSaleCount = DegreeCtrl:GetDegree(pPlayer, "ShenBingPu")
  return nRemainSaleCount > 0
end

function ShenBingPu:CalcRandAttrNum(nStarLevel, nQuality)
  local tbRandAttrCfg = self.tbRandAttrNumCfg[nStarLevel] and self.tbRandAttrNumCfg[nStarLevel][nQuality] or nil
  local nCurWeight = 0
  if tbRandAttrCfg then
    local nRand = MathRandom(1, self.RAND_ATTR_NUM_MAX_WEIGHT)
    for _, tbInfo in ipairs(tbRandAttrCfg) do
      nCurWeight = nCurWeight + tbInfo.nProbility
      if nRand <= nCurWeight then
        return tbInfo.nNum
      end
    end
  end
end

function ShenBingPu:ModifyDailySaleCount(pPlayer, nAddCount)
  if not pPlayer or not Lib:IsInteger(nAddCount) then
    return
  end
  DegreeCtrl:AddDegree(pPlayer, "ShenBingPu", nAddCount)
end

function ShenBingPu:GetFactionWeaponName(nFaction, nSect, nSlot)
  local szSlotName = ""
  local szWeaponName = ""
  if ShenBingPuCFG.SlotName[nSlot] then
    szSlotName = i18n.Get(ShenBingPuCFG.SlotName[nSlot])
  end
  if self.tbFactionCfg[nFaction] and self.tbFactionCfg[nFaction][nSect] then
    szWeaponName = i18n.Parse(self.tbFactionCfg[nFaction][nSect].szWeaponName)
  end
  local szLanguage = i18n.CurrentLanguageCode()
  if "zh-CN" == szLanguage or "zh-TW" == szLanguage then
    return szSlotName .. szWeaponName
  else
    return szSlotName .. " " .. szWeaponName
  end
end

function ShenBingPu:GetIterationSuitName(szIteration)
  local szIterSuitName = "unknow"
  if ShenBingPuCFG.IterationName[szIteration] then
    szIterSuitName = ShenBingPuCFG.IterationName[szIteration]
  end
  return i18n.Get(szIterSuitName)
end

function ShenBingPu:GetUsingShenBingPuBySlot(pPlayer, nSlot)
  if nSlot < Item.ShenBingPu.WEAPON_1 or nSlot > Item.ShenBingPu.WEAPON_3 then
    return
  end
  return Item.ExtEquip:GetEquipObj(pPlayer, nSlot)
end

function ShenBingPu:GetAllUsingShenBingPu(pPlayer)
  local tbWeaponList = {}
  for i = Item.ShenBingPu.WEAPON_1, Item.ShenBingPu.WEAPON_3 do
    local pUsingWeapon = self:GetUsingShenBingPuBySlot(pPlayer, i)
    if pUsingWeapon then
      table.insert(tbWeaponList, pUsingWeapon)
    end
  end
  return tbWeaponList
end

function ShenBingPu:GetAllUsingShenBingPu2(pPlayer)
  local tbWeapon = {}
  for i = Item.ShenBingPu.WEAPON_1, Item.ShenBingPu.WEAPON_3 do
    local pUsingWeapon = self:GetUsingShenBingPuBySlot(pPlayer, i)
    if pUsingWeapon then
      tbWeapon[i] = pUsingWeapon
    end
  end
  return tbWeapon
end

function ShenBingPu:GetAllShenBingPuItemInBagBySlot(pPlayer, nSlot, bNotSort)
  if not self:CheckIsValidSlot(nSlot) then
    return {}
  end
  local tbAllItem = {}
  local tbItemList = pPlayer.GetItemListInBag("ShenBingPuItem") or {}
  for _, pWeapon in ipairs(tbItemList) do
    if Item:GetExtEquipSlot(pWeapon) == nSlot then
      table.insert(tbAllItem, pWeapon)
    end
  end
  if not bNotSort then
    table.sort(tbAllItem, function(pWeaponA, pWeaponB)
      local nStarLevelA = self:GetStarLevel(pWeaponA)
      local nStarLevelB = self:GetStarLevel(pWeaponB)
      return nStarLevelA > nStarLevelB
    end)
  end
  return tbAllItem
end

function ShenBingPu:GetAllShenBingPuSeedInBagBySlot(pPlayer, nSlot, bNotSort)
  if not self:CheckIsValidSlot(nSlot) then
    return {}
  end
  local tbAllItem = {}
  local tbSeedClass = Item:GetClass("ShenBingPuSeed")
  local tbItemList = pPlayer.GetItemListInBag("ShenBingPuSeed") or {}
  for _, pSeed in ipairs(tbItemList) do
    if tbSeedClass:GetSlot(pSeed.dwTemplateId) == nSlot then
      table.insert(tbAllItem, pSeed)
    end
  end
  if not bNotSort then
    table.sort(tbAllItem, function(pWeaponA, pWeaponB)
      local nStarLevelA = self:GetStarLevelForSeed(pWeaponA)
      local nStarLevelB = self:GetStarLevelForSeed(pWeaponB)
      return nStarLevelA > nStarLevelB
    end)
  end
  return tbAllItem
end

function ShenBingPu:GetAllCanComposeShenBingPuInBagBySlot(pPlayer, nSlot, bNotSort)
  local tbAllItem = self:GetAllShenBingPuInBagBySlot(pPlayer, nSlot, bNotSort)
  local tbCanComposeItem = {}
  for _, pItem in ipairs(tbAllItem) do
    if self.tbEquipComposeCfg[pItem.dwTemplateId] then
      local szOpenTimeFrame = ""
      if self:CheckIsShenBingPu(pItem) then
        local szIteration = self:GetIteration(pItem)
        szOpenTimeFrame = ShenBingPuCFG.ComposeOpenTimeFrame[szIteration]
      elseif pItem.szClass == "ShenBingPuSeed" then
        local szIteration = self:GetIterationForSeed(pItem)
        szOpenTimeFrame = ShenBingPuCFG.ComposeOpenTimeFrame[szIteration]
      end
      if not Lib:IsEmptyStr(szOpenTimeFrame) and 1 == GetTimeFrameState(szOpenTimeFrame) then
        table.insert(tbCanComposeItem, pItem)
      end
    end
  end
  return tbCanComposeItem
end

function ShenBingPu:GetAllShenBingPuInBagBySlot(pPlayer, nSlot, bNotSort)
  if not self:CheckIsValidSlot(nSlot) then
    return {}
  end
  local tbAllItem = {}
  Lib:MergeTable(tbAllItem, self:GetAllShenBingPuItemInBagBySlot(pPlayer, nSlot, false))
  Lib:MergeTable(tbAllItem, self:GetAllShenBingPuSeedInBagBySlot(pPlayer, nSlot, false))
  if not bNotSort then
    table.sort(tbAllItem, function(pWeaponA, pWeaponB)
      local nIsSeedA = pWeaponA.szClass == "ShenBingPuSeed" and 1 or 0
      local nIsSeedB = pWeaponB.szClass == "ShenBingPuSeed" and 1 or 0
      if nIsSeedA ~= nIsSeedB then
        return nIsSeedA < nIsSeedB
      end
      local nStarLevelA = pWeaponA.GetIntValue(self.WEAPON_LEVEL)
      local nStarLevelB = pWeaponB.GetIntValue(self.WEAPON_LEVEL)
      if nStarLevelA ~= nStarLevelB then
        return nStarLevelA > nStarLevelB
      end
      return pWeaponA.nQuality > pWeaponA.nQuality
    end)
  end
  return tbAllItem
end

function ShenBingPu:GetIdentifyCost(nSeedTemplateId)
  local tbCostCfg = ShenBingPuCFG.IDENTIFY_COST[nSeedTemplateId]
  if tbCostCfg then
    return tbCostCfg[1], tbCostCfg[2]
  end
end

function ShenBingPu:GetStarLevel(pEquip)
  if not self:CheckIsShenBingPu(pEquip) then
    return
  end
  return pEquip.GetIntValue(self.WEAPON_LEVEL)
end

function ShenBingPu:GetStarLevelForSeed(pSeed)
  if pSeed.szClass ~= "ShenBingPuSeed" then
    return
  end
  return pSeed.GetIntValue(self.WEAPON_LEVEL)
end

function ShenBingPu:GetIteration(pEquip)
  if not self:CheckIsShenBingPu(pEquip) then
    return
  end
  return self.tbItemCfg[pEquip.dwTemplateId] and self.tbItemCfg[pEquip.dwTemplateId].szIteration or nil
end

function ShenBingPu:GetIterationForSeed(pSeed)
  if pSeed.szClass ~= "ShenBingPuSeed" then
    return
  end
  local tbSeedItem = Item:GetClass("ShenBingPuSeed")
  local nItemTemplate = KItem.GetItemExtParam(pSeed.dwTemplateId, tbSeedItem.EXTPARAM_IDENTTEMPLATE)
  return self.tbItemCfg[nItemTemplate] and self.tbItemCfg[nItemTemplate].szIteration or nil
end

function ShenBingPu:GetStarLevelByTemplateId(nTemplateId)
  return self.tbItemCfg[nTemplateId] and self.tbItemCfg[nTemplateId].nStarLevel or 0
end

function ShenBingPu:GetAllStarLevel(pPlayer)
  if not pPlayer then
    return 0
  end
  local tbUsingWeaponList = self:GetAllUsingShenBingPu(pPlayer)
  local nAllStarLevel = 0
  for _, pUsingWeapon in ipairs(tbUsingWeaponList) do
    nAllStarLevel = nAllStarLevel + self:GetStarLevel(pUsingWeapon)
  end
  return nAllStarLevel
end

function ShenBingPu:GetAllStarLevelAsync(pPlayerAsync)
  if not pPlayerAsync then
    return 0
  end
  return pPlayerAsync.GetShenBingPuAllStarLevel()
end

function ShenBingPu:UpdateAllStarLevelAsync(pPlayer)
  local nAllStarLevel = self:GetAllStarLevel(pPlayer)
  local pPlayerAsync = KPlayer.GetAsyncData(pPlayer.dwID)
  if pPlayerAsync then
    pPlayerAsync.SetShenBingPuAllStarLevel(nAllStarLevel)
  end
end

function ShenBingPu:SetStarLevel(pEquip, nStarLevel)
  if not self:CheckIsShenBingPu(pEquip) then
    return
  end
  pEquip.SetIntValue(self.WEAPON_LEVEL, nStarLevel)
end

function ShenBingPu:SetStarLevelForSeed(pSeed, nStarLevel)
  if pSeed.szClass ~= "ShenBingPuSeed" then
    return
  end
  pSeed.SetIntValue(self.WEAPON_LEVEL, nStarLevel)
end

function ShenBingPu:GetRandAttrValRange(nWeaponTemplate, nAttrId, bAllowStrVal)
  local nRandGroupId = self.tbItemCfg[nWeaponTemplate] and self.tbItemCfg[nWeaponTemplate].nRandAttrGroupId or -1
  if not self.tbAttrGroupCfg[nRandGroupId] or not self.tbAttrGroupCfg[nRandGroupId][nAttrId] then
    return 0, 0
  end
  local tbRandAttrCfg = self.tbAttrGroupCfg[nRandGroupId][nAttrId]
  local tbMinAttrInfo = {
    nAttrId = nAttrId,
    nAttrValue1 = tbRandAttrCfg.nAttrValueMin1,
    nAttrValue2 = tbRandAttrCfg.nAttrValueMin2,
    nAttrValue3 = tbRandAttrCfg.nAttrValueMin3
  }
  local tbMaxAttrInfo = {
    nAttrId = nAttrId,
    nAttrValue1 = tbRandAttrCfg.nAttrValueMax1,
    nAttrValue2 = tbRandAttrCfg.nAttrValueMax2,
    nAttrValue3 = tbRandAttrCfg.nAttrValueMax3
  }
  local _, nAttrMinValue, nAttrMaxValue = self:GetWeaponAttrDiff(tbMinAttrInfo, tbMaxAttrInfo, bAllowStrVal)
  if bAllowStrVal then
    return tostring(nAttrMinValue) or "", tostring(nAttrMaxValue) or ""
  else
    return tonumber(nAttrMinValue) or 0, tonumber(nAttrMaxValue) or 0
  end
end

function ShenBingPu:GetAllFixedAttrValRange(nWeaponTemplateId)
  local tbItemCfg = self.tbItemCfg[nWeaponTemplateId]
  local tbMinAttrInfoList = {}
  local tbMaxAttrInfoList = {}
  local tbRangeValList = {}
  if tbItemCfg then
    for i = 1, Item.ShenBingPu.CFG.nMainAttrNum do

          nAttrId = nFixedAttrId,

          nAttrValue2 = tbItemCfg["nFixedAttr" .. i .. "MaxValue2"],

        }

function ShenBingPu:GetAttrFromRandomGroup(nGroupId, tbExceptAttrList)
  if not self.tbAttrGroupCfg[nGroupId] then
    Lib:LogErrFormat("ShenBingPuCommon: GetAttrFromRandomGroup failed, unexist nGroupId = %d", nGroupId)
    return nil
  end
  local tbExceptAttrIdList = {}
  if not tbExceptAttrList then
    tbExceptAttrList = {}
  else
    for _, tbAttr in pairs(tbExceptAttrList) do
      tbExceptAttrIdList[tbAttr.nAttrId] = true
    end
  end
  local tbAttrGroup = Lib:CopyTB(self.tbAttrGroupCfg[nGroupId])
  tbAttrGroup = Lib:GetSubTable2(tbAttrGroup, tbExceptAttrIdList)
  local nSumWeight = 0
  local nPrevWeight = 0
  for nAttrId, tbInfo in pairs(tbAttrGroup) do
    nSumWeight = nSumWeight + tbInfo.nWeight
    tbAttrGroup[nAttrId] = nPrevWeight + self.tbAttrGroupCfg[nGroupId][nAttrId].nWeight
    nPrevWeight = tbAttrGroup[nAttrId]
  end
  if nSumWeight > 0 then
    local nRand = MathRandom(1, nSumWeight)
    for nAttrId, nDealWeight in pairs(tbAttrGroup) do
      if nDealWeight >= nRand then
        return self:CalcNewRandAttr(nAttrId, nGroupId)
      end
    end
    return nil
  else
    Lib:LogErrFormat("ShenBingPuCommon: GetAttrFromRandomGroup failed, SumWeight less than zero, nGroupId = %d", nGroupId)
    return nil
  end
end

function ShenBingPu:CalcNewRandAttr(nAttrId, nRandGroupId)
  local tbAttr = self.tbAttrCfg[nAttrId]
  if not tbAttr then
    Lib:LogErrFormat("ShenBingPuCommon: CalcNewAttrById failed, Attr not exist in ShenBingPuAttrCfg, nAttrId = %d", nAttrId)
    return nil
  end
  if not self.tbAttrGroupCfg[nRandGroupId] or not self.tbAttrGroupCfg[nRandGroupId][nAttrId] then
    Lib:LogErrFormat("ShenBingPuCommon: CalcNewAttrById failed, groupId or Attr not exist in ShenBingPuAttrCfg, nGroupId = %d, nAttrId = %d", nRandGroupId, nAttrId)
  end
  local tbRandAttrCfg = self.tbAttrGroupCfg[nRandGroupId][nAttrId]
  local nAttrValue1 = MathRandom(tbRandAttrCfg.nAttrValueMin1, tbRandAttrCfg.nAttrValueMax1)
  local nAttrValue2 = MathRandom(tbRandAttrCfg.nAttrValueMin2, tbRandAttrCfg.nAttrValueMax2)
  local nAttrValue3 = 0
  if tbAttr.szAttrType == "basic_damage_v" then
    nAttrValue3 = nAttrValue1
  else
    nAttrValue3 = MathRandom(tbRandAttrCfg.nAttrValueMin3, tbRandAttrCfg.nAttrValueMin3)
  end
  return {
    nAttrId = nAttrId,
    szAttrType = tbAttr.szAttrType,
    nAttrValue1 = nAttrValue1,
    nAttrValue2 = nAttrValue2,
    nAttrValue3 = nAttrValue3
  }
end

function ShenBingPu:GetAttrInfoByHoleIndex(pPlayer, pEquip, nHoleIndex, nAttrChoice)
  if not Lib:IsInteger(nHoleIndex) then
    return nil
  end
  if nHoleIndex < 1 or nHoleIndex > self.ATTR_HOLE_NUM then
    return nil
  end
  local nOffset = 5 * (nHoleIndex - 1) + self.ATTR_OFFSET
  local nAttrId = pEquip.GetIntValue(nOffset + 1)
  if nAttrId <= 0 then
    return nil
  end
  local nAttrFightPower = 0
  local nAttrValue1 = 0
  local nAttrValue2 = 0
  local nAttrValue3 = 0
  local szAttrType = self:GetAttrTypeStrByAttrId(nAttrId)
  local nSlot = 0
  if pEquip.szClass == "ShenBingPuSeed" then
    local nItemTemplate = KItem.GetItemExtParam(pEquip.dwTemplateId, Item:GetClass("ShenBingPuSeed").EXTPARAM_IDENTTEMPLATE)
    local tbInfo = KItem.GetItemBaseProp(nItemTemplate)
    nSlot = tbInfo.nDetailType
  else
    nSlot = Item:GetExtEquipSlot(pEquip)
  end
  local nAttrEnhanceLevel = self:GetAttrEnhanceLevel(pPlayer, nSlot, nHoleIndex)
  nAttrChoice = nAttrChoice or self.eATTRCHOICE.ALL
  if nAttrChoice == self.eATTRCHOICE.ONLYBASE then
    nAttrValue1 = pEquip.GetIntValue(nOffset + 2)
    nAttrValue2 = pEquip.GetIntValue(nOffset + 3)
    nAttrValue3 = pEquip.GetIntValue(nOffset + 4)
  elseif nAttrChoice == self.eATTRCHOICE.ONLYENHANCE then
    local nVal1, nVal2, nVal3
    if nHoleIndex >= self.FIXED_ATTR_MIN and nHoleIndex <= self.FIXED_ATTR_MAX then
      nVal1, nVal2, nVal3 = self:GetAttrEnhanceValByHole(pPlayer, nSlot, nHoleIndex)
      local tbEnhanceValCfg = ShenBingPuCFG.ENHANCE_ATTR_VAL[nSlot]
      for i = 1, nAttrEnhanceLevel do
        if tbEnhanceValCfg[i] then
          nAttrFightPower = nAttrFightPower + (tbEnhanceValCfg[i][5] or 0)
        end
      end
    end
    nAttrValue1 = nVal1 or 0
    nAttrValue2 = nVal2 or 0
    nAttrValue3 = nVal3 or 0
  else
    local nVal1, nVal2, nVal3
    if nHoleIndex >= self.FIXED_ATTR_MIN and nHoleIndex <= self.FIXED_ATTR_MAX then
      nVal1, nVal2, nVal3 = self:GetAttrEnhanceValByHole(pPlayer, nSlot, nHoleIndex)
      local tbEnhanceValCfg = ShenBingPuCFG.ENHANCE_ATTR_VAL[nSlot]
      for i = 1, nAttrEnhanceLevel do
        if tbEnhanceValCfg[i] then
          nAttrFightPower = nAttrFightPower + (tbEnhanceValCfg[i][5] or 0)
        end
      end
    end
    nAttrValue1 = pEquip.GetIntValue(nOffset + 2) + (nVal1 or 0)
    nAttrValue2 = pEquip.GetIntValue(nOffset + 3) + (nVal2 or 0)
    nAttrValue3 = pEquip.GetIntValue(nOffset + 4) + (nVal3 or 0)
  end
  return {
    szAttrType = szAttrType,
    nAttrId = nAttrId,
    nAttrValue1 = nAttrValue1,
    nAttrValue2 = nAttrValue2,
    nAttrValue3 = nAttrValue3,
    nFightPower = nAttrFightPower
  }
end

    if nAllStarLevel >= ShenBingPuCFG.SUIT_ATTR[i][1] then
      local tbSuitAttrCfg = ShenBingPuCFG.SUIT_ATTR[i]
      szSuitAttrName = ShenBingPuCFG.SUIT_ATTR[i][6]
      nSuitAttrLevel = i
      for j = 1, 3 do
        if not Lib:IsEmptyStr(tbSuitAttrCfg[1 + j][1]) then
          local tbSingleAttrCfg = tbSuitAttrCfg[1 + j]
          table.insert(tbAllSuitAttr, {
            nAttrId = 0,
            szAttrType = tbSingleAttrCfg[1],
            nAttrValue1 = tbSingleAttrCfg[2],
            nAttrValue2 = tbSingleAttrCfg[3],
            nAttrValue3 = tbSingleAttrCfg[4],
            nFightPower = 0
          })
        end
      end
      if #tbAllSuitAttr > 0 then
        tbAllSuitAttr[#tbAllSuitAttr].nFightPower = tbSuitAttrCfg[5]
      end
      break
    end
  end
  return tbAllSuitAttr, nSuitAttrLevel, szSuitAttrName

function ShenBingPu:GetEnhanceLevelIter(nEnhanceLevel)
  local szLevelIteration = "T" .. Lib:CountTB(ShenBingPuCFG.OpenTimeFrame)
  for szIter, nLevelLimit in pairs(ShenBingPuCFG.ATTR_LEVEL_LIMIT) do
    if nEnhanceLevel < nLevelLimit and self.tbSortIterationSet[szLevelIteration] > self.tbSortIterationSet[szIter] then
      szLevelIteration = szIter
    end
  end
  return szLevelIteration
end

function ShenBingPu:GetEnhanceLevelInfo(pPlayer, nEquipSlot)
  local tbEnhanceLevel = {}
  if not self:CheckIsValidSlot(nEquipSlot) then
    return tbEnhanceLevel
  end
  for i = self.FIXED_ATTR_MIN, self.FIXED_ATTR_MAX do
    local nEnhanceLevel = self:GetAttrEnhanceLevel(pPlayer, nEquipSlot, i)
    if nEnhanceLevel then
      local szLevelIter = self:GetEnhanceLevelIter(nEnhanceLevel)
      tbEnhanceLevel[i] = {nEnhanceLevel = nEnhanceLevel, szLevelIter = szLevelIter}
    end
  end
  return tbEnhanceLevel
end

function ShenBingPu:GetMinEnhanceLevel(pPlayer, nEquipSlot)

  for i = self.FIXED_ATTR_MIN, self.FIXED_ATTR_MAX do

    if nEnhanceLevel and nMinEnhanceLevel > nEnhanceLevel then

function ShenBingPu:GetCurIteration()
  local szCurIteration = ""
  local nCurTimeFrameTime = -1
  for szIteration, szTimeFrame in pairs(ShenBingPuCFG.OpenTimeFrame) do
    if 1 == GetTimeFrameState(szTimeFrame) then
      local nOpenTime = Lib:CalcTimeFrameOpenTime(szTimeFrame)
      if nCurTimeFrameTime < nOpenTime then
        nCurTimeFrameTime = nOpenTime
        szCurIteration = szIteration
      end
    end
  end
  return szCurIteration
end

function ShenBingPu:GetCurMaxEnhanceLevel()
  local szCurIteration = self:GetCurIteration()
  local nCurMaxEnhanceLevel = ShenBingPuCFG.ATTR_LEVEL_LIMIT[szCurIteration]
  return nCurMaxEnhanceLevel
end

function ShenBingPu:GetCurMaxStarLevel()
  local szCurIteration = self:GetCurIteration()
  local nCurMaxStarLevel = 0
  for _, tbItemInfo in pairs(self.tbItemCfg) do
    if self.tbSortIterationSet[tbItemInfo.szIteration] <= self.tbSortIterationSet[szCurIteration] and nCurMaxStarLevel < tbItemInfo.nStarLevel then
      nCurMaxStarLevel = tbItemInfo.nStarLevel
    end
  end
  return nCurMaxStarLevel
end

function ShenBingPu:CalcEnhancePlan(pPlayer, nSlot, nMainAttrIndex, nMaterialId)
  local nSingleVal = self:GetMaterialVal(nMaterialId)
  local nCurEnhanceLevel = self:GetAttrEnhanceLevel(pPlayer, nSlot, nMainAttrIndex)
  local nTargetCostVal = self:GetEnhanceCostValByIter(nCurEnhanceLevel)
  local nAddLevel = 0
  local nNeedCoin = 0
  local nNextCostVal = self:GetEnhanceCostValByLevel(nCurEnhanceLevel + nAddLevel)
  local nTotalCostVal = 0
  if nTargetCostVal >= nNextCostVal then
    local nRestVal = nTargetCostVal
    while nRestVal > 0 and nNextCostVal <= nRestVal do
      nTotalCostVal = nTotalCostVal + nNextCostVal
      nNeedCoin = nNeedCoin + self:GetEnhanceCostCoinByLevel(nCurEnhanceLevel + nAddLevel)
      nAddLevel = nAddLevel + 1
      goto lbl_38
      do break end
      ::lbl_38::
      nRestVal = nRestVal - nNextCostVal
      nNextCostVal = self:GetEnhanceCostValByLevel(nCurEnhanceLevel + nAddLevel)
    end
  else
    nTotalCostVal = nNextCostVal
    nNeedCoin = self:GetEnhanceCostCoinByLevel(nCurEnhanceLevel)
    nAddLevel = 1
  end
  return nAddLevel, math.ceil(nTotalCostVal / nSingleVal), nNeedCoin
end

function ShenBingPu:GetHornorLevel(pPlayer)

    local nCurMinEnhanceLevel = Env.INT_MAX
    for j = 1, self.FIXED_ATTR_MAX do
      local nEnhanceLevel = self:GetAttrEnhanceLevel(pPlayer, i, j)
      if nEnhanceLevel and nCurMinEnhanceLevel > nEnhanceLevel then
        nCurMinEnhanceLevel = nEnhanceLevel

  for i = #ShenBingPuCFG.HONOR, 1, -1 do

      nHonorLevel = i
      break
    end
  end

function ShenBingPu:IsExistAttrInHole(pEquip, nHoleIndex)
  if not Lib:IsInteger(nHoleIndex) then
    return
  end
  if nHoleIndex < 1 or nHoleIndex > self.ATTR_HOLE_NUM then
    return
  end
  local nOffset = 5 * (nHoleIndex - 1) + self.ATTR_OFFSET
  local nAttrId = pEquip.GetIntValue(nOffset + 1)
  return nAttrId > 0
end

function ShenBingPu:RemoveAttrInfoByHoleIndex(pEquip, nHoleIndex)
  local tbEmptyAttr = {
    nAttrId = 0,
    nAttrValue1 = 0,

  self:SetAttrInfoByHoleIndex(pEquip, nHoleIndex, tbEmptyAttr, true)

function ShenBingPu:SetAttrInfoByHoleIndex(pEquip, nHoleIndex, tbAttrInfo, bForce)
  if not Lib:IsInteger(nHoleIndex) then
    return
  end
  if nHoleIndex < 1 or nHoleIndex > self.ATTR_HOLE_NUM then
    return
  end
  if self:IsExistAttrInHole(pEquip, nHoleIndex) and not bForce then
    return
  end
  local nOffset = 5 * (nHoleIndex - 1) + self.ATTR_OFFSET
  pEquip.SetIntValue(nOffset + 1, tbAttrInfo.nAttrId)
  pEquip.SetIntValue(nOffset + 2, tbAttrInfo.nAttrValue1)
  pEquip.SetIntValue(nOffset + 3, tbAttrInfo.nAttrValue2)
  pEquip.SetIntValue(nOffset + 4, tbAttrInfo.nAttrValue3)
end

function ShenBingPu:GetEnhanceCostValByLevel(nCurLevel)

function ShenBingPu:GetEnhanceCostValByIter(nAttrLevel)

  local szLevelIteration = self:GetEnhanceLevelIter(nAttrLevel)
  return math.min(ShenBingPuCFG.ENHANCE_COST_VAL[szCurIteration], ShenBingPuCFG.ENHANCE_COST_VAL[szLevelIteration])

function ShenBingPu:GetEnhanceCostCoinByLevel(nCurLevel)

      return ShenBingPuCFG.ENHANCE_COST_COIN[szIter]
    end
  end
  return Env.INT_MAX
end

function ShenBingPu:GetEnhanceCostCoinByIter(pPlayer, nEquipSlot)

    return Env.INT_MAX

  local _, nMinLevel = self:GetMinEnhanceLevel(pPlayer, nEquipSlot)
  local szLevelIteration = self:GetEnhanceLevelIter(nMinLevel)
  return math.min(ShenBingPuCFG.ENHANCE_COST_COIN[szCurIteration], ShenBingPuCFG.ENHANCE_COST_COIN[szLevelIteration])
end

function ShenBingPu:GetEnhanceCostNumByIter(pPlayer, nEquipSlot)
  if not self:CheckIsValidSlot(nEquipSlot) then
    return Env.INT_MAX
  end

  if nMaterialId then
    local _, nMinLevel = self:GetMinEnhanceLevel(pPlayer, nEquipSlot)
    local nCurIterCostVal = self:GetEnhanceCostValByIter(nMinLevel)
    local nCurMaterialVal = self:GetMaterialVal(nMaterialId)

    return Env.INT_MAX
  end
end

function ShenBingPu:GetWeaponAttrDiff(tbAttrInfo1, tbAttrInfo2, bAllowStrVal)
  if Lib:IsEmptyStr(tbAttrInfo1.szAttrType) and tbAttrInfo1.nAttrId > 0 then
    tbAttrInfo1.szAttrType = self:GetAttrTypeStrByAttrId(tbAttrInfo1.nAttrId)
  end
  if tbAttrInfo2 and Lib:IsEmptyStr(tbAttrInfo2.szAttrType) and tbAttrInfo1.nAttrId > 0 then
    tbAttrInfo2.szAttrType = self:GetAttrTypeStrByAttrId(tbAttrInfo2.nAttrId)
  end
  if tbAttrInfo2 and tbAttrInfo1.szAttrType ~= tbAttrInfo2.szAttrType then
    return
  end
  local szAttr1Desc, szAttr1Value = FightSkill:GetMagicDescSplit(tbAttrInfo1.szAttrType, {
    tbAttrInfo1.nAttrValue1,
    tbAttrInfo1.nAttrValue2,
    tbAttrInfo1.nAttrValue3
  })
  if tbAttrInfo2 then
    local _, szAttr2Value = FightSkill:GetMagicDescSplit(tbAttrInfo2.szAttrType, {
      tbAttrInfo2.nAttrValue1,
      tbAttrInfo2.nAttrValue2,
      tbAttrInfo2.nAttrValue3
    })
    if bAllowStrVal then
      return szAttr1Desc, tonumber(szAttr1Value) or szAttr1Value, tonumber(szAttr2Value) or szAttr2Value
    else
      return szAttr1Desc, tonumber(szAttr1Value), tonumber(szAttr2Value)
    end
  end
  if bAllowStrVal then
    return szAttr1Desc, tonumber(szAttr1Value) or szAttr1Value, nil
  else
    return szAttr1Desc, tonumber(szAttr1Value), nil
  end
end

function ShenBingPu:CalcComposeResult(nEquipTemplateId1, nEquipTemplateId2)
  local nRandomNum = MathRandom(self.COMPOSE_MAX_WEIGHT)
  local nSumWeight = 0
  local tbEquipComposeCfg = self.tbEquipComposeCfg[nEquipTemplateId1] and self.tbEquipComposeCfg[nEquipTemplateId1][nEquipTemplateId2] or nil
  if not tbEquipComposeCfg then
    LogErr("ShenBingPu: CalcComposeResult failed, unexist compose cfg, nEquipTemplateId1 = ", nEquipTemplateId1, ", nEquipTemplateId2 = ", nEquipTemplateId2)
    return
  end
  for i = 1, self.COMPOSE_MAX_RESULT_NUM do
    nSumWeight = nSumWeight + tbEquipComposeCfg["nProbability" .. i]
    if nRandomNum <= nSumWeight then
      return tbEquipComposeCfg["nComposeTemplateId" .. i]
    end
  end
  return 0
end

  local tbInfo = KItem.GetItemBaseProp(nItemTemplate)
  local nSumFightPower = pSeed.nFightPower
  nSumFightPower = nSumFightPower + ShenBingPuCFG.ENHANCE_ATTR_VAL[tbInfo.nDetailType][0][5]
  if tbAllFixedAttrInfo then
    for _, tbAttr in pairs(tbAllFixedAttrInfo) do
      nSumFightPower = nSumFightPower + tbAttr.nFightPower
    end
  end
  local nRandAttrNum = #tbAllRandAttrInfo
  local nStarLevel = self:GetStarLevelForSeed(pSeed)
  nSumFightPower = nSumFightPower + self.tbRandAttrNumCfg[nStarLevel][pSeed.nQuality][nRandAttrNum + 1].nFightPower
  return nSumFightPower
end

function ShenBingPu:GetWeaponFightPower(pPlayer, pWeapon)
  local tbAllRandAttrInfo = self:GetAllRandAttr(pPlayer, pWeapon)
  local tbAllFixedAttrInfo = self:GetAllFixedAttr(pPlayer, pWeapon)
  local nSlot = Item:GetExtEquipSlot(pWeapon)
  local nSumFightPower = pWeapon.nFightPower
  nSumFightPower = nSumFightPower + ShenBingPuCFG.ENHANCE_ATTR_VAL[nSlot][0][5]
  if tbAllFixedAttrInfo then
    for _, tbAttr in pairs(tbAllFixedAttrInfo) do
      nSumFightPower = nSumFightPower + tbAttr.nFightPower
    end
  end
  local nRandAttrNum = #tbAllRandAttrInfo
  local nStarLevel = self:GetStarLevel(pWeapon)
  nSumFightPower = nSumFightPower + self.tbRandAttrNumCfg[nStarLevel][pWeapon.nQuality][nRandAttrNum + 1].nFightPower
  return nSumFightPower
end

function ShenBingPu:CalcFightPower(pPlayer)
  local nTotalPower = 0
  for nEquipPos = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1, Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON3 do
    local pEquip = Item.ExtEquip:GetEquipObj(pPlayer, nEquipPos)
    if pEquip then
      nTotalPower = nTotalPower + self:GetWeaponFightPower(pPlayer, pEquip)
    end
  end
  return nTotalPower
end

function ShenBingPu:GetWeaponFightPowerAsync(pPlayerAsync, pWeapon)
  if not self:CheckIsShenBingPu(pWeapon) then
    return 0
  end
  local tbAllRandAttrInfo = self:GetAllRandAttrAsync(pPlayerAsync, pWeapon)
  local tbAllFixedAttrInfo = self:GetAllFixedAttrAsync(pPlayerAsync, pWeapon)
  local nSlot = Item:GetExtEquipSlot(pWeapon)
  local nSumFightPower = pWeapon.nFightPower
  nSumFightPower = nSumFightPower + ShenBingPuCFG.ENHANCE_ATTR_VAL[nSlot][0][5]
  if tbAllFixedAttrInfo then
    for _, tbAttr in pairs(tbAllFixedAttrInfo) do
      nSumFightPower = nSumFightPower + tbAttr.nFightPower
    end
  end
  local nRandAttrNum = #tbAllRandAttrInfo
  local nStarLevel = self:GetStarLevel(pWeapon)
  nSumFightPower = nSumFightPower + self.tbRandAttrNumCfg[nStarLevel][pWeapon.nQuality][nRandAttrNum + 1].nFightPower
  return nSumFightPower
end

function ShenBingPu:CalcFightPowerAsync(pPlayerAsync, tbUsingWeapon)
  local nTotalPower = 0
  for _, pUsingWeapon in pairs(tbUsingWeapon) do
    if pUsingWeapon then
      nTotalPower = nTotalPower + self:GetWeaponFightPowerAsync(pPlayerAsync, pUsingWeapon)
    end
  end
  return nTotalPower
end

function ShenBingPu:UpdateFightPower(tbTypes, dwPlayerId)
  local pPlayer = KPlayer.GetPlayerObjById(dwPlayerId)
  if not pPlayer then
    return
  end
  FightPower:ChangePower(tbTypes, pPlayer)
end

function ShenBingPu:onFightPowerUpdateEvent(tbTypes)
  Event:RegisterEvent(Event.emEVENT_SHENBINGPU_ENHANCE, ShenBingPu, "UpdateFightPower", tbTypes)
  Event:RegisterEvent(Event.emEVENT_SHENBINGPU_ACTIVATE, ShenBingPu, "UpdateFightPower", tbTypes)
end

function ShenBingPu:OnEventFightPowerChange(dwID, szType, nFightPower)
  Event:OnEvent(Event.emEVENT_SHENBINGPU_FIGHTPOWER_CHANGE, dwID, nFightPower)
end

FightPower:RegisterFPSystemModule({"ShenBingPu"}, ShenBingPu)
