Require("Script/ShenBingPu/ShenBingPuC.lua")
local tbWnd = Ui:CreateClass("UIDivineSuit", "UIPanel")
local tbControls = {
  SuitTemplate = "imgBG/List/Viewport/Content/Template",
  TxtLV = "imgBG/txtInfo/ImgLV/TxtLV"
}
tbWnd.tbControls = tbControls
local ShenBingPuC = Item.ShenBingPu

function tbWnd:UpdateView(pAsyncData)
  local _, nSuitAttrLevel, _, nAllStarLevel = ShenBingPuC:GetPlayerAllSuitAttrAsync(pAsyncData)
  self:ShowSuitAttrSmallPanel(nSuitAttrLevel, nAllStarLevel)
end

function tbWnd:ShowSuitAttrSmallPanel(nSuitAttrLevel, nAllStarLevel)
  if not self.tbSuitPanelView then
    self.tbSuitPanelView = Ui:NewScrollView(self, tbControls.SuitTemplate)
  end
  if #self.tbSuitPanelView:GetAllElement() <= 0 then
    for nSuitLevel, _ in ipairs(Item.ShenBingPu.CFG.SUIT_ATTR) do
      local nStarRequire, tbDesc = ShenBingPuC:GetSuitAttrDescByLevel(nSuitLevel)
      local szDesc = table.concat(tbDesc, "\n")
      self.tbSuitPanelView:SetElement(nSuitLevel, function(tbElement)
        tbElement.pPanel:Label_SetText("txtName/ImgLV/TxtLV", nStarRequire)
        tbElement.pPanel:Label_SetText("txtDesc", szDesc)
      end)
    end
  end
  local tbAllElement = self.tbSuitPanelView:GetAllElement()
  for nSuitLevel, tbElement in pairs(tbAllElement) do
    tbElement.pPanel:SetActive("txtNow", nSuitAttrLevel == nSuitLevel)
  end
  self.pPanel:SetActive("Main", true)
  self.pPanel:Label_SetText(tbControls.TxtLV, nAllStarLevel)
end
