function AvenueOfStars:GenShortDate(szName)
  local tbDate = self[szName]
  if type(tbDate[1]) == "string" then
    local nStartYear, nStartMonth, nStartDay = string.match(tbDate[1], "(%d+).(%d+).(%d+)")
    local nEndYear, nEndMonth, nEndDay = string.match(tbDate[2], "(%d+).(%d+).(%d+)")
    self[szName .. "_YMD"] = {
      string.format("%04d.%02d.%02d", nStartYear, nStartMonth, nStartDay),
      string.format("%04d.%02d.%02d", nEndYear, nEndMonth, nEndDay)
    }
    self[szName .. "_MD"] = {
      string.format("%02d.%02d", nStartMonth, nStartDay),
      string.format("%02d.%02d", nEndMonth, nEndDay)
    }
  else
    local szNameYMD = szName .. "_YMD"
    local szNameMD = szName .. "_MD"
    self[szNameYMD] = {}
    self[szNameMD] = {}
    for nIndex, tbSubDate in ipairs(tbDate) do
      local nStartYear, nStartMonth, nStartDay = string.match(tbSubDate[1], "(%d+).(%d+).(%d+)")
      local nEndYear, nEndMonth, nEndDay = string.match(tbSubDate[2], "(%d+).(%d+).(%d+)")
      self[szNameYMD][nIndex] = {
        string.format("%04d.%02d.%02d", nStartYear, nStartMonth, nStartDay),
        string.format("%04d.%02d.%02d", nEndYear, nEndMonth, nEndDay)
      }
      self[szNameMD][nIndex] = {
        string.format("%02d.%02d", nStartMonth, nStartDay),
        string.format("%02d.%02d", nEndMonth, nEndDay)
      }
    end
  end
end
