local tbWnd = Ui:CreateClass("UINewInformation_NewServerNoviceGift", "UIPanel")
local tbControls = {
  List = "imgList/List",
  Template = "imgList/List/Template"
}
tbWnd.tbControls = tbControls
tbWnd.szClassName = "UINewInformation_NewServerNoviceGift"

function tbWnd:OnCreate()
  self.tbActivityObj = ActivityMgr:GetNewServerNoviceGiftInstance()
  if not self.tbActivityObj then
    return
  end
  self.tbActivityObj:LoadConfig()
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.tbItemListGroup = {}
end

function tbWnd:OnOpen()
  self:UpdateView()
end

function tbWnd:UpdateView()
  local nLoginDays = me.GetUserValue(GetGroupKey(self.tbActivityObj.tbConfig.szSaveKey), self.tbActivityObj.tbConfig.nSaveLoginDays)
  self.ScrollView:Reset()
  for i, tbAward in ipairs(self.tbActivityObj.tbAward) do
    self.ScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtDays", i18n.EasyGet("\231\172\172{0}\229\164\169", tbAward.nDays))
      local tbAwards = Award:GetAward(tbAward.nAwardId, tbAward.nAwardKey)
      local tbItemList = self.tbItemListGroup[i] or Ui:NewItemGridView(self, tbElement.szKey .. "/AwardList/Viewport/Content/ItemTemplate")
      self.tbItemListGroup[i] = tbItemList
      for j = 1, #tbAwards do
        tbItemList:SetItem(j, tbAwards[j], true)
      end
      tbItemList:ResetFrom(#tbAwards + 1)
      self.pPanel:SetActive(tbElement.szKey .. "/AwardList/Viewport/Content/ItemTemplate", false)
      local nReceived = me.GetUserValue(GetGroupKey(self.tbActivityObj.tbConfig.szSaveKey), tbAward.nSaveID)
      if 1 == nReceived then
        tbElement.pPanel:SetActive("btnGet", false)
        tbElement.pPanel:SetActive("imgRecieved", true)
      elseif nLoginDays >= tbAward.nDays then
        tbElement.pPanel:SetActive("btnGet", true)
        tbElement.pPanel:SetActive("imgRecieved", false)
        tbElement.pPanel:Button_SetEnable("btnGet", true)
        Ui:Button_BindEvent(tbElement.pPanel, "btnGet", function()
          ActivityMgr:CallServerScript(self.tbActivityObj.nId, ActivityMgr.NewServerNoviceGift.Protocol.GetAward, tbAward.nDays)
        end)
      else
        tbElement.pPanel:SetActive("btnGet", true)
        tbElement.pPanel:SetActive("imgRecieved", false)
        tbElement.pPanel:Button_SetEnable("btnGet", false)
      end
    end)
  end
end

function tbWnd:OnClose()
  for i, tbItemList in pairs(self.tbItemListGroup) do
    tbItemList:Clear()
  end
  self.ScrollView:Clear()
end
