local tbWnd = Ui:CreateClass("UIOtherZhenFaInfo", "UIPanel")
local tbControls = {
  txtZhenFaName = "Left/txtZhenFaName",
  txtFightPower = "Left/txtFightPower",
  UIItemGrid = "Left/UIItemGrid",
  txtZhenFaLV = "Left/txtZhenFaLV",
  Empty = "Empty",
  Left = "Left",
  Right = "Right",
  AttrTemp = "Right/Scroll View/Viewport/template/AttrTemp",
  ZhenFaAttrtemplate = "Left/Scroll View/Viewport/Content/template"
}
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.tbZhenJueSV = Ui:NewScrollView(self, tbControls.AttrTemp)
  self.tbZhenFaAttrSV = Ui:NewScrollView(self, tbControls.ZhenFaAttrtemplate)
  self.tbZFItemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  self.tbZFItemGrid:SetClickEnable(false)
end

function tbWnd:OnOpen()
  self.pAsyncRole = self.tbOwner.pAsyncRole
  self.nZhenFaLevel = self.pAsyncRole.GetZhenFaLevel()
  if 0 == self.nZhenFaLevel then
    self.nZhenFaLevel = 1
  end
  self.nZhenFaPos = self.pAsyncRole.GetZhenFaPos()
  self.nZhenFaSubPos = self.pAsyncRole.GetZhenFaSubPos()
  if self.nZhenFaPos <= 0 or self.nZhenFaSubPos <= 0 then
    self.pPanel:SetActive(tbControls.Empty, true)
    self.pPanel:SetActive(tbControls.Left, false)
    self.pPanel:SetActive(tbControls.Right, false)
  else
    self.pPanel:SetActive(tbControls.Empty, false)
    self.pPanel:SetActive(tbControls.Left, true)
    self.pPanel:SetActive(tbControls.Right, true)
    self:UpdateView()
  end
end

function tbWnd:GetFightPower()
  local nFightPower = 0
  nFightPower = nFightPower + ZhenFa.ZhenFaLevelInfo[self.nZhenFaLevel][2 + self.nZhenFaSubPos]
  local tbAttrIdList = {}
  local tbAttrIdLevel = {}
  for i = 1, ZhenFa.nMaxZhenJue do
    local nAttrId = self.pAsyncRole.GetZFZhenJueAttrId(i)
    if nAttrId > 0 then
      table.insert(tbAttrIdList, nAttrId)
      local nAttrIdLevel = self.pAsyncRole.GetZFZhenJueLevel(i)
      if 0 == nAttrIdLevel then
        nAttrIdLevel = 1
      end
      table.insert(tbAttrIdLevel, nAttrIdLevel)
    end
  end
  for nIndex, nAttrId in ipairs(tbAttrIdList) do
    if nAttrId > 0 then
      local nLevel = tbAttrIdLevel[nIndex]
      if nLevel > 0 then
        nFightPower = nFightPower + ZhenFa.ZhenJueLevelInfo[nLevel][13]
      end
    end
  end
  return nFightPower
end

function tbWnd:UpdateView()
  self.pPanel:Label_SetText(tbControls.txtZhenFaName, i18n.Get(ZhenFa.ZhenFaInfo[self.nZhenFaPos][self.nZhenFaSubPos][1]))
  self.pPanel:Label_SetText(tbControls.txtZhenFaLV, i18n.EasyGet("\233\152\181\230\179\149\231\173\137\231\186\167 {0}\231\186\167", self.nZhenFaLevel))
  self.pPanel:Label_SetText(tbControls.txtFightPower, self:GetFightPower())
  self.tbZFItemGrid:SetItemByTemplate(ZhenFa.ZhenFaInfo[self.nZhenFaPos][self.nZhenFaSubPos][2])
  self:UpdateZhenJueSV()
  self:UpdateZFAttrSV()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateZhenJueSV()
  local tbAttrIdList = {}
  local tbAttrIdLevel = {}
  for i = 1, ZhenFa.nMaxZhenJue do
    local nAttrId = self.pAsyncRole.GetZFZhenJueAttrId(i)
    if nAttrId > 0 then
      table.insert(tbAttrIdList, nAttrId)
      local nAttrIdLevel = self.pAsyncRole.GetZFZhenJueLevel(i)
      if 0 == nAttrIdLevel then
        nAttrIdLevel = 1
      end
      table.insert(tbAttrIdLevel, nAttrIdLevel)
    end
  end
  local tbRet = {}
  for nHole, nAttrId in ipairs(tbAttrIdList) do
    local nLevel = tbAttrIdLevel[nHole]
    local nItemId = ZhenFa.ZhenJueAttrInfo[nAttrId][nLevel][2]
    tbRet[nHole] = {
      nItemId,
      nLevel,
      nAttrId
    }
  end
  self.tbZhenJueSV:Reset()
  for i = 1, #tbRet do
    self.tbZhenJueSV:SetElement(i, function(tbElement, nIndex)
      local pPanel = tbElement.pPanel
      local nItemId = tbRet[nIndex][1]
      local nLevel = tbRet[nIndex][2]
      local nAttrId = tbRet[nIndex][3]
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(nItemId)
      local szName = Item:GetName(nItemId, me)
      pPanel:Label_SetText("txtName", szName)
      pPanel:Label_SetText("txtLV", nLevel)
      if ZhenFa.tbAttactZJCfg[nAttrId] then
        pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeAttact)
      else
        pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeDefance)
      end
      local tbAttrCur = ZhenFa.ZhenJueAttrInfo[nAttrId][nLevel][1]
      local szAttr = FightSkill:GetMagicDesc(tbAttrCur[1], {
        table.unpack(tbAttrCur, 2)
      })
      pPanel:Label_SetText("txtAttr", szAttr)
    end)
  end
end

function tbWnd:UpdateZFAttrSV()
  local nAttrId = ZhenFa.ZhenFaInfo[self.nZhenFaPos][self.nZhenFaSubPos][4]
  PushString("ZFAttrInfo")
  AppendString(tostring(nAttrId))
  local szAttr = PopString()
  local tbZFAttr = ZhenFa[szAttr]
  local tbAttrCur = tbZFAttr[1][self.nZhenFaLevel]
  self.tbZhenFaAttrSV:Reset()
  for i = 1, #tbAttrCur do
    self.tbZhenFaAttrSV:SetElement(i, function(tbElement, nIndex)
      local pPanel = tbElement.pPanel
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCur[i][1], {
        table.unpack(tbAttrCur[i], 2)
      })
      pPanel:Label_SetText("txtAttrValueNow", szAttribMsg)
      pPanel:Label_SetText("txtPlayerNum01", szValue)
    end)
  end
end
