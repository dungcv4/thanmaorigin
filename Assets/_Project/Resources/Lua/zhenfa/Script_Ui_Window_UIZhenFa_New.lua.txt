function tbWnd:OnCreate()

  self.tbZhenFaReplace = Ui:CreatePanel("ZhenFaReplace", tbControls.ZhenFaReplace, self)
  self:InitAttrObj()
  self.tbZhenJueSV = Ui:NewScrollView(self, tbControls.ZhenJuetemplate)
  self.tbAttrTitle = {
    [1] = i18n.EasyGet("\230\151\160\233\152\159\229\143\139\230\191\128\230\180\187"),
    [2] = i18n.EasyGet("1\229\144\141\233\152\159\229\143\139\230\191\128\230\180\187"),
    [3] = i18n.EasyGet("2\229\144\141\233\152\159\229\143\139\230\191\128\230\180\187"),
    [4] = i18n.EasyGet("3\229\144\141\233\152\159\229\143\139\230\191\128\230\180\187")
  }
  self.tbZFItemGrid = Ui:NewExistItemGrid(self, tbControls.ZhenFaItemGrid)
  self.tbCostItemGrid = Ui:NewExistItemGrid(self, tbControls.CostItemGrid)
  self.tbCostItemGrid:SetClickEnable(true)
  self.pPanel:SetActive(tbControls.ZhenFaReplace, false)
  self.pPanel:SetActive(tbControls.ZhenFa, true)
end

function tbWnd:OnOpen(nPos)
  self.tbZhenFaData = ZhenFa:GetHadActiveZhenFaList(me)
  if #self.tbZhenFaData <= 0 then
    local function fnOk()
      local nNpcId = 4
      
      local nMapId = 1
      local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(nNpcId, nMapId)
      
      local function fnCallback()
        local function __timer_Callback()
          local nNpcId = KNpc.GetNpcIdByTemplateId(nNpcId)
          
          if nNpcId then
            Operation:SimpleTap(nNpcId)
          end
        end
        
        Timer:Register(5, __timer_Callback)
      end
      
      AutoPath:GotoAndCallEx(nMapId, nPosX, nPosY, fnCallback, nWalkNearLength)
      Ui:CloseWindow("UIBag")
    end
    
    Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\154\130\230\156\170\232\163\133\229\164\135\228\187\187\228\189\149\233\152\181\230\179\149\239\188\140\232\175\183\229\137\141\229\190\128\231\153\189\231\167\139\231\144\179\229\164\132\230\142\165\229\143\150\228\187\187\229\138\161\226\128\156\233\152\181\230\179\149\231\167\152\230\156\175\226\128\157"), {
      {fnOk},
      {}
    })
    return 0
  end
  local tbEventList = {}
  for nRedPoint = RedPointMgr.EVENT.ZhenFaZJLevelUpBegin, RedPointMgr.EVENT.ZhenFaZJLevelUpEnd do
    table.insert(tbEventList, nRedPoint)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLVup, tbEventList)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnLevelUp, {
    RedPointMgr.EVENT.ZhenFaLevelUp
  })
  self:UpdateView()
  if nPos then
    self:OpenZFChangePanel(nPos)
  end
end

function tbWnd:OnClose()
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLevelUp)
  RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnLVup)
end

function tbWnd:UpdateZhenFaSV()
  self.tbZhenFaReplace:UpdateView()
end

function tbWnd:UpdateView(szParam)
  if "PutOn" == szParam then
    self:CloseZFChangePanel()
  end
  self:UpdateAttrInfo()
  self:UpdateDetailInfo()
  local nFightPower = ZhenFa:CalcFightPower(me)
  self.pPanel:Label_SetText(tbControls.txtFightPower, nFightPower)
end

function tbWnd:UpdateDetailInfo()
  local nPos, nSubPos = ZhenFa:GetZhenFaCurPutOnPos(me)
  self.pPanel:Label_SetText(tbControls.txtZhenFaName, i18n.Get(ZhenFa.ZhenFaInfo[nPos][nSubPos][1]))
  self.tbZFItemGrid:SetItemByTemplate(ZhenFa.ZhenFaInfo[nPos][nSubPos][2])
  local nZhenFaLevel = ZhenFa:GetZhenFaLevel(me)
  self.pPanel:Label_SetText(tbControls.txtZhenFaLV, i18n.EasyGet("\233\152\181\230\179\149\231\173\137\231\186\167  {0}\231\186\167", nZhenFaLevel))
  local nMaxLevel, nMaxTimeFrameLevel, nMaxQualityLevel = ZhenFa:GetZhenFaMaxLevel()
  local nZhenFaNextLevel
  if nZhenFaLevel >= nMaxLevel then
    nZhenFaNextLevel = nZhenFaLevel
    self.pPanel:SetActive(tbControls.txtMaxLevel, true)
    self.pPanel:SetActive(tbControls.btnLevelUp, false)
    if nMaxTimeFrameLevel <= nMaxLevel then
      local szTimeFrame = ZhenFa:GetNextLevelTimeframe(1, nMaxLevel)
      local nNextWorldLevel = Calendar:GetWorldLevelByTimeframe(szTimeFrame)
      if nNextWorldLevel > 0 then
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\230\156\128\233\171\152\231\173\137\231\186\167\239\188\140\229\188\128\230\148\190{0}\231\173\137\231\186\167\228\184\138\233\153\144\229\144\142\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\239\188\129", Lib:GetLevelText(nNextWorldLevel)))
      else
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\231\186\167\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
      end
    elseif nMaxQualityLevel <= nMaxLevel then
      local nTipGrade = ZhenFa:GetNextLevelTipsGrade(1, nMaxLevel)
      if nTipGrade > 0 then
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\232\135\179\230\156\128\233\171\152\231\173\137\231\186\167\239\188\140\232\163\133\229\164\135\228\187\187\230\132\143\231\137\185\231\186\167{0}\233\152\182\233\152\181\230\179\149\229\144\142\229\143\175\231\187\167\231\187\173\229\141\135\231\186\167\239\188\129", nTipGrade))
      else
        self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\231\186\167\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
      end
    else
      self.pPanel:Label_SetText(tbControls.txtMaxLevel, i18n.EasyGet("\229\183\178\229\141\135\231\186\167\229\136\176\228\184\138\233\153\144\231\173\137\231\186\167"))
    end
  else
    self.pPanel:SetActive(tbControls.txtMaxLevel, false)
    self.pPanel:SetActive(tbControls.btnLevelUp, true)
    self.pPanel:Button_SetText(tbControls.btnLevelUp, i18n.EasyGet("\233\152\181\230\179\149\229\141\135\231\186\167"), "Text")
    nZhenFaNextLevel = nZhenFaLevel + 1
  end
  self:UpdateCostInfo(nZhenFaNextLevel)
  self.pPanel:Label_SetText(tbControls.txtLevelNow, i18n.EasyGet("{0}\231\186\167", nZhenFaLevel))
  self.pPanel:Label_SetText(tbControls.txtLevelNext, i18n.EasyGet("{0}\231\186\167", nZhenFaNextLevel))
  self:UpdateZhenJueSV()
end

function tbWnd:UpdateZhenJueSV()
  local nPos, _ = ZhenFa:GetZhenFaCurPutOnPos(me)
  local tbShowZJId = ZhenFa:GetZhenJueAttrShowItemIdList(me, nPos)
  self.tbZhenJueSV:Reset()
  for i = 1, #tbShowZJId do
    self.tbZhenJueSV:SetElement(i, function(tbElement, nIndex)
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(tbShowZJId[nIndex])
      local nLevel = ZhenFa:GetZhenJueHoleLevel(me, nIndex)
      itemGrid:SetLabelSuffixText("+" .. tostring(nLevel))
      itemGrid:SetClickEnable(true, function()
        Item:ShowItemDetail({
          nTemplate = tbShowZJId[nIndex],
          tbItemExtParams = {nPos = nPos, nHole = nIndex}
        })
      end)
    end)
  end
end

function tbWnd:UpdateCostInfo(nZhenFaNextLevel)
  local nCostId, nCostNum = ZhenFa:GetZhenFaUpgradeConsumeInfo(nZhenFaNextLevel)
  self.tbCostItemGrid:SetItemByTemplate(nCostId)
  local nCurNum = me.GetItemCountByTypeInBag(nCostId)
  self.pPanel:Label_SetText(tbControls.txtCost, string.format("%d/%d", nCurNum, nCostNum))
  if nCostNum <= nCurNum then
    self.pPanel:Label_SetColorByName(tbControls.txtCost, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtCost, "NotEnough")
  end
end

function tbWnd:OnSyncItemEvent(nItemId, bUpdateAll)
  local nZhenFaLevel = ZhenFa:GetZhenFaLevel(me)
  local nMaxLevel = ZhenFa:GetZhenFaMaxLevel()
  local nZhenFaNextLevel
  if nZhenFaLevel >= nMaxLevel then
    nZhenFaNextLevel = nZhenFaLevel
  else
    nZhenFaNextLevel = nZhenFaLevel + 1
  end
  self:UpdateCostInfo(nZhenFaNextLevel)
end

function tbWnd:UpdateAttrInfo()
  local nPos, nSubPos = ZhenFa:GetZhenFaCurPutOnPos(me)
  local nAttrId = ZhenFa.ZhenFaInfo[nPos][nSubPos][4]
  PushString("ZFAttrInfo")
  AppendString(tostring(nAttrId))
  local szAttr = PopString()
  local tbZFAttr = ZhenFa[szAttr]
  local nZhenFaLevel = ZhenFa:GetZhenFaLevel(me)
  local nZhenFaNextLevel = nZhenFaLevel + 1
  if nZhenFaNextLevel > ZhenFa:GetZhenFaMaxLevel() then
    nZhenFaNextLevel = nZhenFaLevel
  end
  local nTeamMemberNum = TeamMgr:GetMembersCount()
  nTeamMemberNum = math.max(1, nTeamMemberNum)
  for i = 1, ZhenFa.nMaxTeamMember do
    self.pPanel:Label_SetText(self.tbAttrObj[i] .. "/txtAttTittle", self.tbAttrTitle[i])
    if nTeamMemberNum == i then
      self.pPanel:SetActive(self.tbAttrObj[i] .. "/txtAttState", true)
    else
      self.pPanel:SetActive(self.tbAttrObj[i] .. "/txtAttState", false)
    end
    local tbAttrCur = tbZFAttr[i][nZhenFaLevel]
    local tbAttrNext = tbZFAttr[i][nZhenFaNextLevel]
    for j = 1, #tbAttrCur do
      PushString(self.tbAttrObj[i])
      AppendString("/Grid/Item0")
      AppendString(tostring(j))
      local szAttrPath = PopString()
      self.pPanel:SetActive(szAttrPath, true)
      local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCur[j][1], {
        table.unpack(tbAttrCur[j], 2)
      })
      local _, szValueNext = FightSkill:GetMagicDescSplit(tbAttrNext[j][1], {
        table.unpack(tbAttrNext[j], 2)
      })
      self.pPanel:Label_SetText(string.format("%s%s", szAttrPath, "/txtAttrValueNow"), szAttribMsg)
      self.pPanel:Label_SetText(string.format("%s%s", szAttrPath, "/txtAttrValueNext"), szValue)
      self.pPanel:Label_SetText(string.format("%s%s", szAttrPath, "/txtAttrValueUp"), szValueNext)
      self.pPanel:SetActive(string.format("%s%s", szAttrPath, "/imgArrow2"), true)
    end
    for j = #tbAttrCur + 1, 3 do
      PushString(self.tbAttrObj[i])
      AppendString("/Grid/Item0")
      AppendString(tostring(j))
      local szAttrPath = PopString()
      self.pPanel:SetActive(szAttrPath, false)
    end
  end
end

function tbWnd:InitAttrObj()
  self.tbAttrObj = {}
  for i = 2, 3 do
    local szKey = self.pPanel:CloneObject(tbControls.Item01, "Item0" .. i)
  end
  for i = 1, ZhenFa.nMaxTeamMember do
    self.tbAttrObj[i] = self.pPanel:CloneObject(tbControls.AttrTemp, "AttrTemp" .. i)
  end
  self.pPanel:SetActive(tbControls.AttrTemp, false)
end

function tbWnd:CloseZFChangePanel()
  self.pPanel:SetActive(tbControls.ZhenFaReplace, false)
  self.pPanel:SetActive(tbControls.ZhenFa, true)
  self.tbZhenFaReplace:OnClose()
end

function tbWnd:OpenZFChangePanel(nPos)
  self.pPanel:SetActive(tbControls.ZhenFaReplace, true)
  self.pPanel:SetActive(tbControls.ZhenFa, false)
  self.tbZhenFaReplace:OnOpen(nPos)
end

function tbWnd:UpdateAttrIdLibSV(nPos, nSubPos)
  local tbCfg = ZhenFa.ZhenFaInfo[nPos][nSubPos]
  local tbAttackZJLib = ZhenFa.ZhenJueLibrary[tbCfg[6]]
  local tbDefenseZJLib = ZhenFa.ZhenJueLibrary[tbCfg[8]]
  local tbZJAttrId = {}
  Lib:MergeTable(tbZJAttrId, tbAttackZJLib)
  Lib:MergeTable(tbZJAttrId, tbDefenseZJLib)
  local tbShowZJId = {}
  for nHole, nAttrId in ipairs(tbZJAttrId) do
    local nItemId = ZhenFa.ZhenJueAttrInfo[nAttrId][1][2]
    tbShowZJId[nHole] = nItemId
  end
  self.tbAttrIdSVLib:Reset()
  for i = 1, #tbZJAttrId do
    if tbShowZJId[i] then
      self.tbAttrIdSVLib:SetElement(i, function(tbElement, nIndex)
        local pPanel = tbElement.pPanel
        local szName = Item:GetName(tbShowZJId[nIndex], me)
        pPanel:Label_SetText("txtName", szName)
        local nLevel = ZhenFa:GetZhenJueHoleLevel(me, nIndex)
        local tbAttrCur = ZhenFa.ZhenJueAttrInfo[tbZJAttrId[nIndex]][nLevel][1]
        local szAttribMsg, szValue = FightSkill:GetMagicDescSplit(tbAttrCur[1], {
          table.unpack(tbAttrCur, 2)
        })
        pPanel:Label_SetText("txtAttr", szAttribMsg)
        if ZhenFa.tbAttactZJCfg[tbZJAttrId[i]] then
          pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeAttact)
        else
          pPanel:Sprite_SetSprite("imgType", ZhenFa.szTypeDefance)
        end
        local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        itemGrid:SetItemByTemplate(tbShowZJId[nIndex])
      end)
    end
  end
end
