Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.HuaHaiMiYuan)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/WomenDayActivity/HuaHaiMiYuanCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

tbActivity.FindTypeIndex = {OnceFind = 1, BatchFind = 2}
tbActivity.FindType = {
  [tbActivity.FindTypeIndex.OnceFind] = {nFindCount = 1, nConsumeCount = 1},
  [tbActivity.FindTypeIndex.BatchFind] = {nFindCount = 10, nConsumeCount = 9}
}
tbActivity.SaveKey = {
  nLastJoinTime = 1,
  nFindCount = 2,
  nTotalScore = 3
}
tbActivity.Protocol = {
  ReqFindYuan = 1,
  ReqFindYuanDataSync = 2,
  FindYuanDataSyncResp = 3
}

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.szTitle = i18n.Encode(self.tbConfig.szTitle)
  self.tbConfig.szHaveCostTxt = i18n.Encode(self.tbConfig.szHaveCostTxt)
  self.tbConfig.szLeftTopTitle = i18n.Encode(self.tbConfig.szLeftTopTitle)
  self.tbConfig.txtSecondTittle = i18n.Encode(self.tbConfig.txtSecondTittle)
  self.tbConfig.tbnGoldWareId = Lib:SplitStrTry2Num(self.tbConfig.tbnGoldWareId, "|")
  self.tbConfig.szLotteryTerm = i18n.Encode(self.tbConfig.szLotteryTerm)
  if self.tbConfig.szBtnGoText then
    self.tbConfig.szBtnGoText = i18n.Encode(self.tbConfig.szBtnGoText)
  end
  self.tbConfig.tbFindType = {}
  for nIndex, tbFindType in ipairs(tbActCfg.tbFindType) do
    self.tbConfig.tbFindType[nIndex] = tbFindType
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
  return true
end

function tbActivity:CheckConfigValid()
  local tbConfig = self:GetConfig()
  if Lib:IsEmptyStr(tbConfig.szLogWay) then
    LogErr("[HuaHaiMiYuan:CheckConfigValid] bad logway! please check it! szLogWay=", tbConfig.szLogWay, self:GetId())
    return false
  end
  return true
end

function tbActivity:GetScriptDataFieldName()
  return "szScriptDataKey"
end
