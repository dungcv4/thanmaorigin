Require("CommonScript/Activity/ActivityMgr.lua")
ActivityMgr.szKinRankingAwardsClassName = "KinRankingAwards"
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szKinRankingAwardsClassName)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/KinRankingAwards/KinRankingAwards.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(szFilepath .. "not find")
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
    self.nConfigId = nConfigId
  end
  szFilepath = "Setting/Activity/KinRankingAwards/Awards.tab"
  local tbActiveAward = Lib:EasyLoadTabFile(szFilepath, nil)
  self.ACTIVITY_CONFIG[self.nConfigId].tbActiveAward = {}
  for nAwardIndex, tbInfo in ipairs(tbActiveAward or {}) do
    self.ACTIVITY_CONFIG[self.nConfigId].tbActiveAward[nAwardIndex] = tbInfo
  end
  szFilepath = "Setting/Activity/KinRankingAwards/AwardsTitle.tab"
  local tbActiveAwardTitle = Lib:EasyLoadTabFile(szFilepath, nil)
  self.ACTIVITY_CONFIG[self.nConfigId].tbActiveAwardTitle = {}
  if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[self.nConfigId].szLogWay) then
    self.ACTIVITY_CONFIG[self.nConfigId].szLogWay = "LogWay_KinRankingAward"
  end
  for nAwardIndex, tbInfo in ipairs(tbActiveAwardTitle or {}) do
    self.ACTIVITY_CONFIG[self.nConfigId].tbActiveAwardTitle[nAwardIndex] = tbInfo
  end
  self.bLoadConfig = true
  return true
end

function tbActivity:GetAwardConfig()
  return self.ACTIVITY_CONFIG[self.nConfigId].tbActiveAward
end

function tbActivity:OnInit(tbConfig)
  if not self:CheckConfig(tbConfig.nConfigId) then
    return false
  end
  self:OnModify(tbConfig)
  return true
end

function ActivityMgr:GetKinRankingActivityInstance()
  for _, pInstance in pairs(self.tbInstance) do
    if pInstance.szClassName == ActivityMgr.szKinRankingAwardsClassName and pInstance:IsOpen() then
      return pInstance
    end
  end
end

function tbActivity:IsAwardTime()
  local nAwardTime = Lib:CalcTimeFrameOpenTime(self.tbConfig.szAwardDay, GetServerCreateTime())
  return nAwardTime < GetTime() + 5
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end
