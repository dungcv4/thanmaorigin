ActivityMgr.Def.tbCheckFieldTyep = {
  string = "string",
  number = "number",
  award = "award"
}
ActivityMgr.Def.tbCheckFieldFunc = {
  [ActivityMgr.Def.tbCheckFieldTyep.string] = function(tbConfig, szField)
    if Lib:IsEmptyStr(tbConfig[szField]) then
      LogErr(string.format("Activity config ERROR Please check %s", szField))
      return false
    end
    return true
  end,
  [ActivityMgr.Def.tbCheckFieldTyep.number] = function(tbConfig, szField)
    if not Lib:IsInteger(tbConfig[szField]) then
      LogErr(string.format("Activity config ERROR Please check %s ", szField))
      return false
    end
    return true
  end,
  [ActivityMgr.Def.tbCheckFieldTyep.award] = function(tbConfig, szAwardId, szAwardKey)
    local nAwardId = tbConfig[szAwardId]
    if not Lib:IsInteger(nAwardId) then
      LogErr(string.format("Activity config ERROR Please check %s", szAwardId))
      return false
    end
    local nAwardKey = tbConfig[szAwardKey]
    if not Lib:IsInteger(nAwardKey) then
      LogErr(string.format("Activity config ERROR Please check %s", szAwardKey))
      return false
    end
    if Award.tbAllAwards[nAwardId] == nil then
      LogErr(string.format("award is not exsit. %s , %s", szAwardId, szAwardKey))
      return false
    end
    local tbRandAwards = Award.tbAllAwards[nAwardId].tbRandAwards and Award.tbAllAwards[nAwardId].tbRandAwards[nAwardKey]
    local tbFixAwards = Award.tbAllAwards[nAwardId].tbFixAwards and Award.tbAllAwards[nAwardId].tbFixAwards[nAwardKey]
    if nil == tbRandAwards and nil == tbFixAwards then
      LogErr(string.format("award is not exsit. %s: %d, tbRandAwards: %d, tbFixAwards:%d", szAwardId, nAwardId, tbRandAwards, tbFixAwards))
      return false
    end
    return true
  end
}
