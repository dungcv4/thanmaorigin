Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.WuLinZhengFeng)
tbActivity.Protocol = {AwardPlayerListReq = 1, OnAwardPlayerListResp = 2}

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/WuLinZhengFeng/WuLinZhengFengCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    tbConfig.szAwardTime = Lib:TimeDesc16(Lib:CalcTimeFrameOpenTime(tbConfig.szAwardTimeFrame))
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[nConfigId].szLogWay) then
      self.ACTIVITY_CONFIG[nConfigId].szLogWay = "LogWay_WuLinZhengFeng"
    end
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[nConfigId].szSaveKey) then
      self.ACTIVITY_CONFIG[nConfigId].szSaveKey = ActivityMgr.Def.ClassNameList.WuLinZhengFeng
    end
  end
  szFilepath = "Setting/Activity/WuLinZhengFeng/PersonalAward.tab"
  local tbPersonalAward = Lib:EasyLoadTabFile(szFilepath)
  if not tbPersonalAward then
    LogErr("%s not find", szFilepath)
    return false
  end
  local nChecker = 0
  for _, tbInfo in pairs(tbPersonalAward) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("WuLinZhengFeng: Load PersonalAward Config failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPersonalAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPersonalAward or {}
    if not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPersonalAward[tbInfo.nRank] then
      if nChecker + 1 == tbInfo.nRank then
        table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPersonalAward, tbInfo.nAwardKey)
        nChecker = tbInfo.nRank
      else
        LogErr("WuLinZhengFeng: PersonalAward table's nRank sequence seems chaos")
        return false
      end
    else
      LogErr("WuLinZhengFeng: Duplicate nRank in PersonalAward, nRank=%d", tbInfo.nRank)
      return false
    end
  end
  szFilepath = "Setting/Activity/WuLinZhengFeng/FactionAward.tab"
  local tbFactionAward = Lib:EasyLoadTabFile(szFilepath)
  if not tbFactionAward then
    LogErr("%s not find", szFilepath)
    return false
  end
  for _, tbInfo in pairs(tbFactionAward) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("WuLinZhengFeng: Load FactionAward Config failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFactionAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFactionAward or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFactionAward[tbInfo.nFactionId] = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFactionAward[tbInfo.nFactionId] or {}
    if not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFactionAward[tbInfo.nFactionId][tbInfo.nRank] then
      table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFactionAward[tbInfo.nFactionId], tbInfo.nRank, tbInfo.nAwardKey)
    else
      LogErr("WuLinZhengFeng: Duplicate nRank in FactionAward in the same Faction, nFactionId=%d, nRank=%d", tbInfo.nFactionId, tbInfo.nRank)
      return false
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end
