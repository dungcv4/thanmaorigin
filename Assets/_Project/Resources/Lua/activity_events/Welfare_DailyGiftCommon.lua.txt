DailyGift.SAVE_GROUP = GetGroupKey("DailyGift")
DailyGift.AccumulateCount = 9
DailyGift.DailyGiftOne = 1
DailyGift.DailyGiftSix = 2
DailyGift.DailyGiftTwelve = 3
DailyGift.DailyGiftCount = 3
DailyGift.DailyGiftFree = 4
DailyGift.nAccumulateCountMax = 7
DailyGift.nNotBuy = 1
DailyGift.nNotGet = 2
DailyGift.nHadGet = 3
DailyGift.tbIndex2ProductId = {
  [1] = 19,
  [2] = 20,
  [3] = 21,
  [4] = 22
}
DailyGift.tbProductId2Index = {
  [19] = 1,
  [20] = 2,
  [21] = 3,
  [22] = 4,
  [69] = 2,
  [70] = 2,
  [71] = 3,
  [72] = 3
}
DailyGift.nOpenLevel = 11

function DailyGift:GetDayGiftRemainTime(pPlayer, nGiftType)
  return pPlayer.GetUserValue(self.SAVE_GROUP, (nGiftType << 1) - 1)
end

function DailyGift:GetDayGiftGetAwardTimestamp(pPlayer, nGiftType)
  return pPlayer.GetUserValue(self.SAVE_GROUP, nGiftType << 1)
end

function DailyGift:GetAccumulateCount(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.AccumulateCount)
end

function DailyGift:GetDayGiftGetState(pPlayer, nGiftType)
  local nCurLocalDay = Lib:GetLocalDay(GetTime() - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  local nGetLocalDay = Lib:GetLocalDay(self:GetDayGiftGetAwardTimestamp(pPlayer, nGiftType) - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  if nCurLocalDay == nGetLocalDay then
    return self.nHadGet
  end
  if nGiftType == self.DailyGiftFree then
    return self:GetAccumulateCount(pPlayer) <= 0 and self.nHadGet or self.nNotGet
  end
  local nRemainCount = self:GetDayGiftRemainTime(pPlayer, nGiftType)
  return nRemainCount <= 0 and self.nNotBuy or self.nNotGet
end

function DailyGift:GetShowPriceByProductIndex(nProductIndex)
  local nProductId = DailyGift.tbIndex2ProductId[nProductIndex]
  local tbProductSetting = Recharge:GetProductSettingById(nProductId)
  local szShowPrice = Recharge:GetShowPrice(tbProductSetting)
  return szShowPrice
end
