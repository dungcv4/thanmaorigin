local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.OperSwitch)
tbActivity.SWITCH_HANDLER = {
  ShoperAuction = "SwitchShoperAuction"
}

function tbActivity:LoadConfig()
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile("Setting/Activity/OperSwitch/OperSwitch.tab", {"nConfigId"})
end

function tbActivity:OnOpen()
  self:Switch(true)
end

function tbActivity:OnClose()
  self:Switch(false)
end

function tbActivity:Switch(bOpen)
  local szHandler = self.SWITCH_HANDLER[self.tbConfig.szType]
  if not szHandler then
    LogErr("Activity.OperSwitch", "Switch", "config:" .. self.tbConfig.nConfigId, "type:" .. self.tbConfig.szType, "errNoHandler")
    return
  end
  self[szHandler](self, bOpen)
end

function tbActivity:SwitchShoperAuction(bOpen)
  Auction:SetShoperOpenByActivity(bOpen)
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end

function tbActivity:GetLogWayFieldName()
  return ActivityMgr.NoLogWay
end
