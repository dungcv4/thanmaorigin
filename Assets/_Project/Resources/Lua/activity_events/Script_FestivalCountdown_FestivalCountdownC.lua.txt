local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.FestivalCountdown)
local Protocol = {
  [tbActivity.Protocol.ReqStartCountDown] = "ReqStartCountDown"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    me.CenterMsg(i18n.EasyGet("\230\156\141\229\138\161\231\171\175\232\190\147\229\133\165\233\148\153\232\175\175"))
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:ReqStartCountDown(szParam)
  if Map:IsFubenMap(me.nMapTemplateId) then
    return
  end
  szParam = i18n.Parse(szParam)
  local tbParam = Lib:SplitStrTry2Num(szParam, "|")
  local tbReady = Lib:SplitStrTry2Num(tbParam[1], ":")
  local szCountDownUI = tbReady[1]
  local nCountDownTime = tbReady[2]
  local szStart = i18n.Parse(tbReady[3])
  local tbEffect = Lib:SplitStrTry2Num(tbParam[2], ":")
  local szUIEffect = tbEffect[1]
  local nEffectTime = tbEffect[2]
  Ui:OpenWindow(szCountDownUI, nCountDownTime, szStart)
  self.nTimer = Timer:Register(nCountDownTime * Env.GAME_FPS, self.ShowEffect, self, szUIEffect, nEffectTime)
end

function tbActivity:ShowEffect(szUIEffect, nEffectTime)
  Ui:OpenWindow(szUIEffect, nEffectTime)
end
