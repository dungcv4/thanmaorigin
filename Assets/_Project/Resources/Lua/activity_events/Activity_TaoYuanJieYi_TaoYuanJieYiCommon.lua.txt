Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.TaoYuanJieYi)
tbActivity.Protocol = {GetJieYiBoxAward = 1}
tbActivity.nSaveGroupID = GetGroupKey("TaoYuanJieYi")
tbActivity.nHasGetBoxSaveID = 1
tbActivity.nSaveLastStartTime = 2
tbActivity.bHasJieYiSaveID = 3

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/EmeiActivity/TaoYuanJieYi/TaoYuanJieYi.tab"
  local tbField = {
    "nConfigId",
    "szMailTitle",
    "szMailFrom",
    "szAlertTime",
    "szAlertMFrom",
    "szAlertMTitle",
    "szAlertMText",
    "szLogWay",
    "szSaveKey",
    "szDesc",
    "szName",
    "szShowTime",
    "nLevel",
    "nShowAwardId",
    "nShowAwardKey",
    "nAwardId",
    "nAwardKey"
  }
  local tbActivityConfig = Lib:LoadTabFileEx(szFilepath, "dsssssssssssddddd", "nConfigId", tbField)
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig or {}) do
    tbConfig.nTimeOut = not Lib:IsEmptyStr(tbConfig.szTimeOut) and Lib:ParseDateTime(tbConfig.szTimeOut) or 0
    tbConfig.nAlertTime = Lib:ParseDateTime(tbConfig.szAlertTime)
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  self.bLoadConfig = true
end

function tbActivity:PlayerLevelEnough(pPlayer)
  if not pPlayer then
    LogErr("IsJieYi player nil")
    return
  end
  local tbCfg = self.ACTIVITY_CONFIG[self.nConfigId]
  return pPlayer.nLevel >= tbCfg.nLevel
end

function tbActivity:IsJieYi(pPlayer)
  if not pPlayer then
    LogErr("IsJieYi player nil")
    return
  end
  return pPlayer.GetUserValue(self.nSaveGroupID, self.bHasJieYiSaveID) > 0
end

function tbActivity:HasGetAwardBox(pPlayer)
  if not pPlayer then
    LogErr("IsJieYi player nil")
    return
  end
  return pPlayer.GetUserValue(self.nSaveGroupID, self.nHasGetBoxSaveID) > 0
end
