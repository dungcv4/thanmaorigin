function ActivityRank:Init()
  local szFilePath = "Setting/Activity/Common/FirstLayer.tab"
  
  local tbField = {
    "nFirstLayerId",
    "szTitle",
    "szSecendLayerId",
    "nActivityIndex",
    "nButtonIndex",
    "szExtraParam"
  }
  self.tbFirstLayerCfg = Lib:LoadTabFileEx(szFilePath, "dssdds", "nFirstLayerId", tbField)
  local szFilePath = "Setting/Activity/Common/SecendLayer.tab"
  local tbField = {
    "nSecendLayerId",
    "szTitle",
    "nActivityIndex",
    "nButtonIndex",
    "szExtraParam"
  }
  self.tbSecendLayerCfg = Lib:LoadTabFileEx(szFilePath, "dsdds", "nSecendLayerId", tbField)
end

function ActivityRank:GetTreeData(tbFirstLayerList)
  local tbUITreeBigDataList = {}
  local tbUITreeSmallDataList = {}
  for i, nFirstLayerId in ipairs(tbFirstLayerList) do
    local tbFirstLayerData = self.tbFirstLayerCfg[nFirstLayerId]
    if tbFirstLayerData then
      local tbUITreeBigData = {}
      tbUITreeBigData.nType = tbFirstLayerData.nFirstLayerId
      tbUITreeBigData.szText = i18n.Parse(tbFirstLayerData.szTitle)
      tbUITreeBigData.tbInfo = {
        nActivityIndex = tbFirstLayerData.nActivityIndex,
        nOnClickIndex = tbFirstLayerData.nButtonIndex,
        nType = tbFirstLayerData.nFirstLayerId,
        szExtraParam = tbFirstLayerData.szExtraParam
      }
      table.insert(tbUITreeBigDataList, tbUITreeBigData)
      local tbSecendLayerList = {}
      if tbFirstLayerData.szSecendLayerId ~= "" and 0 ~= tbFirstLayerData.szSecendLayerId then
        tbSecendLayerList = Lib:SplitStrTry2Num(tbFirstLayerData.szSecendLayerId, "|")
      end
      for i, nSecendLayerId in ipairs(tbSecendLayerList) do
        local tbSecendLayerData = self.tbSecendLayerCfg[nSecendLayerId]
        if tbSecendLayerData then
          local tbUITreeSmallData = {}
          tbUITreeSmallData.nType = tbFirstLayerData.nFirstLayerId
          tbUITreeSmallData.szText = i18n.Parse(tbSecendLayerData.szTitle)
          tbUITreeSmallData.tbInfo = {
            nActivityIndex = tbSecendLayerData.nActivityIndex,
            nOnClickIndex = tbSecendLayerData.nButtonIndex,
            szExtraParam = tbSecendLayerData.szExtraParam,
            nType = tbFirstLayerData.nFirstLayerId,
            nSecendLayerId = nSecendLayerId
          }
          table.insert(tbUITreeSmallDataList, tbUITreeSmallData)
        end
      end
    end
  end
  return tbUITreeBigDataList, tbUITreeSmallDataList
end

ActivityRank:Init()
