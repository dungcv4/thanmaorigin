local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.CollectPoint)
tbActivity.SaveID = {
  LastJoinTime = 1,
  nPoint = 2,
  nDailyAwardTime = 3,
  nTotalExchangeNum = 4,
  IsFixData = 5
}
tbActivity.Protocol = {
  OnExchangeReq = 1,
  OnPointChange = 2,
  OnFixBranch = 3,
  OnExchangeOpen = 4
}
tbActivity.DailyAwardMsg = {
  NoConfig = 1,
  OccurError = 2,
  NotUseExchangeItem = 3,
  HasGetAwardToday = 4,
  Success = 5
}

function tbActivity:LoadConfig()
  local szFilepath = "Setting/Activity/CollectPoint/CollectPointCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  szFilepath = "Setting/Activity/CollectPoint/PointExchangeCfg.tab"
  local tbExchangeConfig = Lib:EasyLoadTabFile(szFilepath, {
    "nConfigId",
    "nExchangeId"
  })
  if not tbExchangeConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    assert(tbExchangeConfig[nConfigId])
    assert(tbExchangeConfig[nConfigId][tbConfig.nDefaultExchangeId])
    tbConfig.tbExchange = tbExchangeConfig[nConfigId]
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  Validator:Register(self.CheckAllConfig, self)
end

function tbActivity:CheckAllConfig()
  for nConfigId, tbConfigs in pairs(self.ACTIVITY_CONFIG) do
    if not (tbConfigs.nDefaultExchangeId and tbConfigs.szMailFrom and tbConfigs.szMailTitle) or not tbConfigs.szMailText then
      LogErr("CollectPoint Lost Config", nConfigId)
      return false
    end
    if not tbConfigs.tbExchange or not tbConfigs.tbExchange[tbConfigs.nDefaultExchangeId] then
      LogErr("CollectPoint not Exchange Info", nConfigId)
      return false
    end
    if not Env[tbConfigs.szLogWay] then
      LogErr("CollectPoint szLogWay not Exsit", nConfigId)
      return false
    end
    if not Event[tbConfigs.szAddPointEvent] then
      LogErr("CollectPoint szAddPointEvent not Exsit", nConfigId)
      return false
    end
    for nExchangeId, tbExchangeInfo in pairs(tbConfigs.tbExchange) do
      local tbAward = Award:GetAward(tbExchangeInfo.nAwardId, tbExchangeInfo.nAwardKey)
      if not tbAward or 0 == #tbAward then
        LogErr("CollectPoint Award not Exsit", nConfigId, nExchangeId)
        return false
      end
    end
  end
  return true
end

function tbActivity:CheckCanGetDailyAward(pPlayer)
  if not pPlayer then
    return tbActivity.DailyAwardMsg.OccurError
  end
  local nLastGetTime = pPlayer.GetUserValue(GetGroupKey(self.tbConfig.szSaveKey), self.SaveID.nDailyAwardTime)
  local nNow = GetTime()
  if Lib:IsSameDay(14400, nLastGetTime, nNow) then
    return tbActivity.DailyAwardMsg.HasGetAwardToday
  end
  local nCount = self:GetTotalExchangeNum(pPlayer)
  if nCount > 0 then
    if self.tbConfig.nDailyAwardId > 0 and self.tbConfig.nDailyAwardKey > 0 then
      return tbActivity.DailyAwardMsg.Success
    else
      return tbActivity.DailyAwardMsg.NoConfig
    end
  else
    return tbActivity.DailyAwardMsg.NotUseExchangeItem
  end
end

function tbActivity:GetTotalExchangeNum(pPlayer)
  local SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
  return pPlayer.GetUserValue(SaveKey, self.SaveID.nTotalExchangeNum)
end

function tbActivity:GetPoint(pPlayer)
  local SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
  return pPlayer.GetUserValue(SaveKey, self.SaveID.nPoint)
end

function tbActivity:GetStartExchangeTime()
  return Lib:GetDate2Time(self.tbConfig.nStartExchangeTime)
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szMailFrom = i18n.Encode(self.tbConfig.szMailFrom)
  self.tbConfig.szMailTitle = i18n.Encode(self.tbConfig.szMailTitle)
  self.tbConfig.szMailText = i18n.Encode(self.tbConfig.szMailText)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szDetailDesc = i18n.Encode(self.tbConfig.szDetailDesc)
  self.tbConfig.tbExchange = tbActCfg.PointExchangeCfg
  self.tbConfig = SetReadOnly(self.tbConfig, true)
  return true
end

function tbActivity:CheckConfigValid(tbConfig)
  return true
end
