local tbWnd = Ui:CreateWindow("UIFestival_PasswordRedPacket")
local tbControls = {
  txtTitle = "Right/txtTitle",
  txtTime01 = "Right/txtTime01",
  txtLV = "Right/txtLV",
  txtSecondTittle02 = "Right/txtSecondTittle02",
  Template = "Right/List/Viewport/Content/Template01",
  btnHelp = "Right/txtTitle/btnHelp"
}
tbWnd.tbControls = tbControls
tbWnd.tbShowAwardPath = {
  [1] = {
    szTitle = "Left/Cont_Tittle01/txtTittle01",
    szAward = "Left/txtTittle01"
  },
  [2] = {
    szTitle = "Left/Cont_Tittle03/txtTittle01",
    szAward = "Left/txtTittle03"
  },
  [3] = {
    szTitle = "Left/Cont_Tittle04/txtTittle01",
    szAward = "Left/txtTittle04"
  },
  [4] = {
    szTitle = "Left/Cont_Tittle02/txtTittle01",
    szAward = "Left/txtTittle02"
  }
}
tbWnd.ElementPath = {
  btnGo = "ImgAll/btnGo",
  btnOfficialAccount = "ImgAll/btnOfficialAccount",
  txtInfo = "ImgAll/Cont_Info/txtInfo01",
  txtOpenTime = "ImgAll/Cont_Info/txtOpenTime"
}
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  self.nIndex = nIndex
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.szHelpKey = tbInfo.szHelpKey
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.Get(tbInfo.szName))
  self.pPanel:Label_SetText(tbControls.txtTime01, i18n.Get(tbInfo.szShowTime))
  local tbAward = ActivityMgr.CommonEntryLinkData[tbInfo.nId].tbAward
  self.pPanel:Label_SetText(tbControls.txtLV, i18n.EasyGet("\231\173\137\231\186\167\233\153\144\229\136\182\239\188\154{0}", tbInfo.nLevel))
  for index, tbPath in ipairs(self.tbShowAwardPath) do
    self.pPanel:SetActive(tbPath.szTitle, false)
    self.pPanel:SetActive(tbPath.szAward, false)
  end
  for index, tbAwardInfo in ipairs(tbAward) do
    self.pPanel:SetActive(self.tbShowAwardPath[index].szTitle, true)
    self.pPanel:SetActive(self.tbShowAwardPath[index].szAward, true)
    self.pPanel:Label_SetText(self.tbShowAwardPath[index].szTitle, i18n.Get(tbAwardInfo.szTitle))
    self.pPanel:Label_SetText(self.tbShowAwardPath[index].szAward, i18n.Get(tbAwardInfo.szAward))
  end
  self.pPanel:Label_SetText(tbControls.txtSecondTittle02, i18n.Get(ActivityMgr.CommonEntryLinkData[tbInfo.nId].szDesc))
  self:UpdateViewList()
end

function tbWnd:UpdateViewList()
  self.tbScrollView:Reset()
  for index, tbInfo in ipairs(ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].tbSetp) do
    self.tbScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:SetActive(self.ElementPath.btnGo, true)
      tbElement.pPanel:SetActive(self.ElementPath.btnOfficialAccount, false)
      tbElement.pPanel:Label_SetText(self.ElementPath.txtInfo, i18n.Get(tbInfo.szInfo))
      tbElement.pPanel:Label_SetText(self.ElementPath.txtOpenTime, i18n.Get(tbInfo.szTime))
      RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, self.ElementPath.btnGo, {
        RedPointMgr.EVENT.PassWordRedPacket
      })
      Ui:Button_BindEvent(tbElement.pPanel, self.ElementPath.btnGo, self.OnClickGo, self, tbInfo.szCallBack, tbInfo.szParam)
    end)
  end
end

function tbWnd:OnClickGo(szCallBack, szParam)
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam, "|")
  fn(self.tbActivityList[self.nIndex].nId, table.unpack(tbParam))
end

function tbWnd:OnClose()
  self.tbScrollView:Reset()
end
