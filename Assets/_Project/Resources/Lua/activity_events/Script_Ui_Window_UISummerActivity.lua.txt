local tbWnd = Ui:CreateClass("UISummerActivity", "UICommonActWndBase")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MAP_SWITCH,
      self.CloseByDefault
    },
    {
      EventNotify.emNOTIFY_WND_CLOSED,
      self.OnWndClose,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBg/btnClose",
  wndActivityiTips = "wndActivityiTips",
  wndJumpTo = "wndJumpTo",
  btnTemplate = "node/imgBg/btnList/Viewport/Content/btnTemplate",
  ActivityInfo = "wndActivityiTips/imgBg/ActivityInfo",
  btnJoin = "wndActivityiTips/imgBg/btnJoin",
  ItemTemplate = "wndActivityiTips/imgBg/ActivityInfo/AwardList/Viewport/Content/ItemTemplate",
  timeWaringRed = "wndJumpTo/imgBG/List/timeWaringRed",
  timeWaringGreen = "wndJumpTo/imgBG/List/timeWaringGreen",
  MapTemplate = "wndJumpTo/imgBG/List/Viewport/Content/Template",
  txtTitle = "wndJumpTo/imgBG/txtTitle",
  txtSmallTitle = "node/imgBg/txtTime",
  btnRanking = "node/imgBg/btnTemplate",
  btnHelp = "node/imgBg/Title/btnHelp"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:CloseByDefault()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("Fishing")
  end
}

function tbWnd:OnCreate()
  self.tbBtnScrollView = Ui:NewScrollView(self, tbControls.btnTemplate, self.OnClickActivityHelper, nil, false)
  self.tbShowAwardScrollView = Ui:NewItemGridScrollView(self, tbControls.ItemTemplate, 1, 1)
  self:SetClickEmptyCallback(tbControls.wndActivityiTips)
  self.tbMapScrollView = Ui:NewScrollView(self, tbControls.MapTemplate, self.OnClickGoToMap)
end

function tbWnd:OnOpen(nEntranceIndex)
  self.nEntranceIndex = nEntranceIndex
  self:UpdateActivityList()
  self.pPanel:SetActive(tbControls.wndActivityiTips, false)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  self.pPanel:Label_SetText(tbControls.txtSmallTitle, i18n.EasyGet("\230\180\187\229\138\168\230\151\182\233\151\180\239\188\154{0}", i18n.Parse(ActivityMgr.EntranceSchedule:GetSmallTitle(self.nEntranceIndex))))
  self.pPanel:Button_BindEvent(tbControls.btnRanking, function()
    local pFishingInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.SummerFishing)
    if pFishingInstance then
      Ui:OpenWindow("UIRankingSummer", pFishingInstance.tbConfig.nRankingAwardActivityId)
    end
  end)
end

function tbWnd:OnClickActivityHelper(nId)
  self:OnClickActivity(nId)
end

function tbWnd:OnClickActivity(nId)
  local tbInfo
  for _, tbTmpInfo in ipairs(self.tbActivityList) do
    if tbTmpInfo.nId == nId then
      tbInfo = tbTmpInfo
      break
    end
  end
  if not tbInfo then
    return
  end
  if nId == self.nCurActivityId then
    self.pPanel:SetActive(tbControls.wndActivityiTips, false)
    self.pPanel:SetActive(tbControls.wndJumpTo, false)
    self:UpdateActivedStatus(self.nCurActivityId, false)
    self.nCurActivityId = nil
    return
  end
  self.nCurActivityId = nId
  self:DealActivityDetail(tbInfo)
  self:DealJumpTo(tbInfo)
end

function tbWnd:UpdateActivedStatus(nActivityId, bStatus)
  local tbElement = self.tbBtnScrollView:GetElement(nActivityId)
  tbElement.pPanel:SetActive("txtNormal", not bStatus)
  tbElement.pPanel:SetActive("txtSelected", bStatus)
  local obj = tbElement.pPanel:GetObject("Main")
  local trans = obj:GetComponent(Ui.BGroupTrans)
  if trans then
    if bStatus then
      trans:OnSelected()
    else
      tbElement.pPanel:Button_SetEnable("Main", true)
      trans:OnUnSelected()
    end
  end
end

function tbWnd:UpdateActivityList()
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(self.nEntranceIndex)
  self.nCurActivityId = nil
  self.tbBtnScrollView:Reset()
  for _, tbInfo in ipairs(self.tbActivityList) do
    self.tbBtnScrollView:SetElement(tbInfo.nId, function(tbElement)
      self:UpdateBtnInfo(tbElement, tbInfo.nId, tbInfo)
    end)
  end
end

function tbWnd:CloseByDefault()
  self.pPanel:SetActive(tbControls.wndActivityiTips, false)
  self:Close()
end

function tbWnd:UpdateBtnInfo(tbElement, nId, tbInfo)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  local bOpening = pInstance and pInstance:IsOpen()
  tbElement.pPanel:SetActive("txtNormal", bOpening)
  tbElement.pPanel:SetActive("imgLock", false)
  tbElement.pPanel:SetActive("txtSelected", false)
  tbElement.pPanel:SetActive("txtLock", not bOpening)
  tbElement.pPanel:Label_SetText("txtNormal", tbInfo.szName)
  tbElement.pPanel:Label_SetText("txtSelected", tbInfo.szName)
  tbElement.pPanel:Label_SetText("txtLock", tbInfo.szName)
  RedPointMgr:DynamicUnRegist(tbElement.pPanel, "imgRedPoint")
  tbElement.pPanel:SetActive("imgRedPoint", false)
  if tbInfo.tbRedPointEventId then
    tbElement.pPanel:SetActive("imgRedPoint", true)
    RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "imgRedPoint", tbInfo.tbRedPointEventId)
  end
end

function tbWnd:DealActivityDetail(tbInfo)
  self.pPanel:SetActive(tbControls.wndActivityiTips, true)
  self.pPanel:SetActive(tbControls.wndActivityiTips .. "/imgBg", true)
  self.pPanel:SetActive(tbControls.ActivityInfo .. "/txtActDegreeTip", true)
  self.pPanel:SetActive(tbControls.ActivityInfo .. "/txtActDegree", true)
  self.pPanel:Label_SetText(tbControls.ActivityInfo .. "/txtLevelLimit", tbInfo.nLevel)
  self.pPanel:Label_SetText(tbControls.ActivityInfo .. "/txtLevelLimit", tbInfo.nLevel)
  self.pPanel:Label_SetText(tbControls.wndActivityiTips .. "/imgBg/txtActivityName", tbInfo.szName)
  self.pPanel:Label_SetText(tbControls.ActivityInfo .. "/ActivityDescScroll/Viewport/Content/txtActivityDesc", tbInfo.szDesc)
  self.pPanel:Label_SetText(tbControls.ActivityInfo .. "/txtActDegreeTip", i18n.EasyGet("\230\180\187\229\138\168\230\151\182\233\151\180"))
  self.pPanel:Label_SetText(tbControls.ActivityInfo .. "/txtActDegree", tbInfo.szShowTime)
  self.tbShowAwardScrollView:UpdateGenericItem(tbInfo.tbAward, true)
  self.tbShowAwardScrollView:ResetParent()
  if not tbInfo.pFunc then
    self.pPanel:SetActive(tbControls.btnJoin, false)
  else
    self.pPanel:SetActive(tbControls.btnJoin, true)
    if tbInfo.szClassName == ActivityMgr.Def.ClassNameList.SummerFishing then
      self.pPanel:Button_SetText(tbControls.btnJoin, i18n.EasyGet("\229\137\141\229\190\128\233\177\188\231\175\147"), "Text")
    else
      self.pPanel:Button_SetText(tbControls.btnJoin, i18n.EasyGet("\229\137\141\229\190\128"), "Text")
    end
    Ui:Button_BindEvent(self.pPanel, tbControls.btnJoin, self.OnClickJoinBtn, self, tbInfo.pFunc, tbInfo.nId, table.unpack(Lib:SplitStrTry2Num(tbInfo.szJumpParam, "|")))
    RedPointMgr:DynamicUnRegist(self.pPanel, tbControls.btnJoin)
    if tbInfo.tbRedPointEventId then
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnJoin, tbInfo.tbRedPointEventId)
    end
  end
end

function tbWnd:DealJumpTo(tbInfo)
  if tbInfo.szClassName == ActivityMgr.Def.ClassNameList.SummerFishing then
    local pInstance = ActivityMgr:GetInstanceById(tbInfo.nId)
    if pInstance then
      self:ShowJumpToWnd(pInstance)
    end
  else
    self.pPanel:SetActive(tbControls.wndJumpTo, false)
  end
end

function tbWnd:OnClickJoinBtn(func, ...)
  self:OnClickEmptyClose()
  func(...)
end

function tbWnd:OnClickEmptyClose()
  if self.nCurActivityId then
    self:OnClickActivity(self.nCurActivityId)
  end
  self.pPanel:SetActive(tbControls.wndActivityiTips, false)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
end

function tbWnd:OnWndClose(szUiName)
end

function tbWnd:SetMapInfoList(tbElement, tbInfo)
  tbElement.pPanel:Label_SetText("txtMapName", tbInfo.szName)
end

function tbWnd:ShowJumpToWnd(pInstance)
  self.tbMapInfoList = pInstance:GetMapInfoList()
  if self.pPanel:IsActive(tbControls.wndJumpTo) then
    self.pPanel:SetActive(tbControls.wndJumpTo, false)
    return
  end
  self.pPanel:SetActive(tbControls.timeWaringGreen, pInstance.bDailyOpenFishing == true)
  self.pPanel:SetActive(tbControls.timeWaringRed, pInstance.bDailyOpenFishing ~= true)
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\229\135\186\231\142\176\229\156\176\229\155\190"))
  self.pPanel:SetActive(tbControls.wndJumpTo, true)
  for nIndex, tbInfo in ipairs(self.tbMapInfoList) do
    self.tbMapScrollView:SetElement(nIndex, function(tbElement)
      self:SetMapInfoList(tbElement, tbInfo)
    end)
  end
end

function tbWnd:OnClickGoToMap(nIndex)
  ActivityMgr:GotoDynamicNpc(self.tbMapInfoList[nIndex].nMapTemplateId, self.tbMapInfoList[nIndex].nPosX, self.tbMapInfoList[nIndex].nPosY, function()
  end)
  self:Close()
end
