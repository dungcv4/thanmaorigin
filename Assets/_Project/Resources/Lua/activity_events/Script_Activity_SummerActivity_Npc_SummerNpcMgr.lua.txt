local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.SummerFishing)

function tbActivity:InitFishBallNpc()
  self.tbFishBallNpcId = {}
  self:DelFishBallNPC()
end

function tbActivity:UnInitFishBallNpc()
  self:DelFishBallNPC()
  self.tbFishBallNpcId = nil
end

function tbActivity:GetFishBallNPC()
  if not self.tbFishBallNpcId then
    Log("SummerNpcMgr: GetFishBallNPC() table nil")
    return
  end
  local nIndex, _ = next(self.tbFishBallNpcId)
  if not nIndex then
    return
  end
  local pNpc = KNpc.GetById(self.tbFishBallNpcId[nIndex])
  return pNpc
end

function tbActivity:DelFishBallNPC()
  if not self.tbFishBallNpcId then
    Log("SummerNpcMgr: DelFishBallNPC() table nil")
    return
  end
  local pNpc = self:GetFishBallNPC()
  if pNpc then
    self.tbFishBallNpcId[pNpc.nId] = nil
    pNpc.Delete()
  else
    Log("SummerNpcMgr: DelFishBallNPC failed")
  end
end

function tbActivity:AddFishBallNPC(nMapId, nLocX, nLocY)
  if not self.tbFishBallNpcId then
    Log("SummerNpcMgr: tbFishBallNpcId table nil")
    return
  end
  if not (nMapId and nLocX) or not nLocY then
    return
  end
  if self:GetFishBallNPC() then
    self:DelFishBallNPC()
  end
  if not self.tbConfig or not self.tbConfig.nFishBallNpcId then
    LogErr("SummerNpcMgr: AddBoxNPC failed, config is nil")
    return
  end
  local pNpc = KNpc.Add(self.tbConfig.nFishBallNpcId, 1, -1, nMapId, nLocX, nLocY, 0, 0, 0)
  if pNpc then
    self.tbFishBallNpcId[pNpc.nId] = pNpc.nId
    return pNpc
  else
    Log("SummerNpcMgr: AddFishBallNPC failed, nMapId=%d, nLocX=%d, nLocY=%d", nMapId, nLocX, nLocY)
  end
end

function tbActivity:FishBallNpcCastSkill(nStatusId, nMapId, nLocX, nLocY)
  local pNpc = self:GetFishBallNPC()
  if not pNpc and nStatusId == self.FishBallStatus.START then
    pNpc = self:AddFishBallNPC(nMapId, nLocX, nLocY)
  end
  if pNpc and self.FishBallSkill[nStatusId] then
    pNpc.CastSkill(self.FishBallSkill[nStatusId], 1, nLocX, nLocY)
  end
  if nStatusId == self.FishBallStatus.END or nStatusId == self.FishBallStatus.GET then
    self:DelFishBallNPC()
  end
end
