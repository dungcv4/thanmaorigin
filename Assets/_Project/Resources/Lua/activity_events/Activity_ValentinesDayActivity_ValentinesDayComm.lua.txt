Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ValentinesDay)

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Award/Awards/Activities/OperationAwards/ValentinesDayAchieve.tab"
  local tbField = {
    "nWareId",
    "szAchieveId"
  }
  self.ACTIVITY_CONFIG = {}
  self.tbAchievementConfig = Lib:LoadTabFileEx(szFilePath, "ds", "nWareId", tbField)
  self.ACTIVITY_CONFIG[1] = self.tbAchievementConfig
  Validator:Register(self.CheckValentinesDayCfg, self)
end

function tbActivity:CheckValentinesDayCfg()
  for nWareId, tbInfo in pairs(self.tbAchievementConfig) do
    if not Lib:IsInteger(nWareId) or not Shop.tbWares[nWareId] then
      LogErr(string.format("load ValentinesDayAchieve.tab failed, nWareId %d not found in RechargeProducts.tab", nWareId))
      return false
    end
  end
  return true
end
