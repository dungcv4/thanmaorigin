Require("CommonScript/Activity/XiongWangActivity/XiongWangActivity/XiongWangActivityCfg.lua")
ActivityMgr.XiongWangActivity = ActivityMgr.XiongWangActivity or {}
local XiongWangActivity = ActivityMgr.XiongWangActivity
XiongWangActivity.ActivityType = {
  JingWangJiuXiActivity = "JingWangJiuXiActivity"
}
local tbJumpFunc = {
  JingWangJiuXiActivity = function(nId)
    ActivityMgr.XiongWangActivity:OnActive(nId)
    if 0 == Ui.PlayerPrefs.GetInt("JingWangJiuXiFirst", 0) then
      Ui.PlayerPrefs.SetInt("JingWangJiuXiFirst", 1)
      RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.JingWangJiuxiFirstOpen)
    end
  end,
  TuanJieLiWu = function(nId)
    if nId then
      local pInstance = ActivityMgr:GetInstanceById(nId)
      if pInstance and pInstance:IsOpen() then
        Ui:OpenWindow("UITuanJieLiWu", nId)
        return
      end
    end
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
  end,
  [ActivityMgr.Def.ClassNameList.LaborHonor] = function(nId)
    ActivityMgr.XiongWangActivity:OnActive(nId)
  end,
  CelebrationGift = function()
    local pObj = ActivityMgr:GetInstanceByClassName(ActivityMgr.szCelebrationGift)
    if pObj and pObj:IsOpen() then
      pObj:TryOpenWindow()
    else
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    end
  end
}

function XiongWangActivity:GetActivityCfg(szType)
  local tbActivityList = self:GetActivityList()
  if not tbActivityList then
    Log(" XiongWangActivity:GetActivity() data nil ")
    return nil
  end
  for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
    local tbConfig = self:BuildInstanceConfig(nId, szType)
    return tbConfig
  end
  return nil
end

function XiongWangActivity:GetActivityList()
  local tbList = {}
  local tbConfig, szName, pInstance
  for _, szType in ipairs(ActivityMgr.XiongWangActivityCfg.ActivityList) do
    for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
      tbConfig, pInstance = self:BuildInstanceConfig(nId, szType)
      if tbConfig then
        szName = tbConfig.szName and i18n.Parse(tbConfig.szName) or ""
        table.insert(tbList, {
          szClassName = szType,
          nId = nId,
          szName = szName,
          szDesc = i18n.Parse(tbConfig.szDesc),
          nLevel = tbConfig.nLevel or tbConfig.nJoinLevel,
          tbAward = Award:GetAward(tbConfig.nShowAwardId or tbConfig.nAwardId or -1, tbConfig.nShowAwardKey or tbConfig.nAwardKey or -1) or {},
          pFunc = tbJumpFunc[szType],
          szShowTime = i18n.Parse(tbConfig.szShowTime or ""),
          nSortValue = tbConfig.nSortValue and tbConfig.nSortValue or ActivityMgr.XiongWangActivityCfg.SortList[nId] or 0,
          tbRedPointEventId = pInstance and pInstance.GetRedPointEventId and pInstance:GetRedPointEventId(),
          bDirectJump = Lib:IIF(tbConfig.nDirectJump and 1 == tbConfig.nDirectJump, true, false)
        })
      end
    end
  end
  table.sort(tbList, function(tbFirst, tbSecond)
    return tbFirst.nSortValue > tbSecond.nSortValue
  end)
  return tbList
end

function XiongWangActivity:BuildInstanceConfig(nId, szType)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if pInstance then
    return pInstance:GetConfig(), pInstance
  end
  local tmpInstance = ActivityMgr:Create(szType)
  if not tmpInstance then
    return
  end
  tmpInstance:OnInit(ActivityMgr:GetConfigById(nId))
  return tmpInstance:GetConfig()
end

function XiongWangActivity:OnActive(nId)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return
  end
  if pInstance:IsA(ActivityMgr.Def.ClassNameList.LaborHonor) then
    return self:OnLaborHonorActive(pInstance)
  elseif pInstance:IsA(ActivityMgr.Def.ClassNameList.JingWangJiuXiActivity) then
    return self:OnJingWangJiuXiActive(pInstance)
  end
end

function XiongWangActivity:OnJingWangJiuXiActive(pInstance)
  Ui:OpenWindow("UIJinWangJiuXi", pInstance:GetId())
end

function XiongWangActivity:OnLaborHonorActive(pInstance)
  local dwTongId = me.dwTongId
  if not dwTongId or dwTongId <= 0 then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\138\160\229\133\165\229\184\174\228\188\154"))
    return
  end
  Ui:OpenWindow("UILaoDongRongYao", pInstance:GetId())
end
