local tbWnd = Ui:CreateWindow("UIActivity_JiangHu2")
tbWnd.tbControls = {
  btnHelp = "node/imgBg/Title/btnHelp",
  btnClose = "node/imgBg/btnClose",
  txtSlogan = "node/imgBg/txtSlogan",
  txtRemainCoinNumber = "node/imgBg/content/ContYuanBao/txtRemainReturn/RemainCoin/txtRemainCoinNumber",
  txtNowCoinNumber = "node/imgBg/content/ContYuanBao/txtNowReturn/NowCoin/txtNowCoinNumber",
  btnGet = "node/imgBg/content/ContYuanBao/btnGet",
  txtGet = "node/imgBg/content/ContYuanBao/btnGet/Text",
  btnLv = "node/imgBg/content/btnTab/btnLv",
  btnYuanbao = "node/imgBg/content/btnTab/btnYuanBao",
  goLvPanel = "node/imgBg/content/ContLV",
  goYuanbaoPanel = "node/imgBg/content/ContYuanBao",
  templateLvList = "node/imgBg/content/ContLV/Viewport/Content/Template"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("FriendsReturnActivityHelp")
  end,
  btnGet = function(self)
    self:OnGet()
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ReturnerVNVipSignIn)
  end,
  btnLv = function(self)
    self.pPanel:Sprite_SetSprite(self.btnLv, "UI/atlas/welfare/Img_Btn01.png")
    self.pPanel:Sprite_SetSprite(self.btnYuanbao, "UI/atlas/welfare/Img_Btn02.png")
    self.pPanel:SetActive(self.goLvPanel, true)
    self.pPanel:SetActive(self.goYuanbaoPanel, false)
  end,
  btnYuanbao = function(self)
    self.pPanel:Sprite_SetSprite(self.btnLv, "UI/atlas/welfare/Img_Btn02.png")
    self.pPanel:Sprite_SetSprite(self.btnYuanbao, "UI/atlas/welfare/Img_Btn01.png")
    self.pPanel:SetActive(self.goLvPanel, false)
    self.pPanel:SetActive(self.goYuanbaoPanel, true)
  end
}

function tbWnd:OnCreate()
  self:UnRegistRedPoint()
  self.tbDailySignInList = Ui:NewScrollView(self, self.templateLvList, nil, false, true)
end

function tbWnd:OnOpen()
  self:RegistRedPoint()
  local nState = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipAwardCheck)
  local szServerName = FriendsReturn.szOldServerName
  local szPlayerName = FriendsReturn.szOldPlayerName
  self.pPanel:Label_SetText(self.txtSlogan, i18n.EasyGet("\231\148\177\228\186\142\229\164\167\228\190\160\229\156\168{0}\230\156\141\229\138\161\229\153\168\231\154\132\232\167\146\232\137\178{1}\229\183\178\231\187\143\232\191\158\231\187\17330\229\164\169\230\156\170\231\153\187\229\189\149\239\188\140\231\187\145\229\174\154\230\150\176\230\156\141\232\167\146\232\137\178\229\141\179\229\143\175\230\191\128\230\180\187\229\133\131\229\174\157\231\166\143\229\136\169\239\188\129", szServerName, szPlayerName))
  if 1 == nState then
    local nTotalYuanbaoCount = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipTotalYuanbaoCount)
    local nYuanbaoGetCount = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoGetCount)
    local nLeftYuanbaoCount = nTotalYuanbaoCount - nYuanbaoGetCount
    local nYuanbaoGetPerTime = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoGetPerTime)
    local nLastGetTime = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoCanGetTime)
    self.pPanel:Label_SetText(self.txtRemainCoinNumber, nLeftYuanbaoCount)
    self.pPanel:Label_SetText(self.txtNowCoinNumber, nYuanbaoGetPerTime)
    local nCurTime = GetTime()
    if not Lib:IsDiffDay(14400, nCurTime, nLastGetTime) then
      self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\228\187\138\230\151\165\229\183\178\233\162\134\229\143\150"))
      self.pPanel:Button_SetEnable(self.btnGet, false)
    elseif nLeftYuanbaoCount <= 0 then
      self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\229\183\178\231\187\147\230\157\159"))
      self.pPanel:Button_SetEnable(self.btnGet, false)
    else
      self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\233\162\134\229\143\150"))
      self.pPanel:Button_SetEnable(self.btnGet, true)
    end
  elseif 3 == nState then
    local nVipLevel = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipVIPLevel)
    local nTotalYuanbaoCount = FriendsReturn.tbVNVipAwardInfo[nVipLevel].nYuanBaoTotalNum
    local nYuanbaoGetPerTime = FriendsReturn.tbVNVipAwardInfo[nVipLevel].nYuanBaoGetPerDay
    self.pPanel:Label_SetText(self.txtRemainCoinNumber, nTotalYuanbaoCount)
    self.pPanel:Label_SetText(self.txtNowCoinNumber, nYuanbaoGetPerTime)
    self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\231\187\145\229\174\154\230\173\164\232\167\146\232\137\178"))
  end
  self.pPanel:SetActive(self.btnLv, 1 == nState)
  self.pPanel:SetActive(self.btnYuanbao, 1 == nState)
  self.pPanel:Sprite_SetSprite(self.btnLv, "UI/atlas/welfare/Img_Btn02.png")
  self.pPanel:Sprite_SetSprite(self.btnYuanbao, "UI/atlas/welfare/Img_Btn01.png")
  self.pPanel:SetActive(self.goYuanbaoPanel, true)
  self.pPanel:SetActive(self.goLvPanel, false)
  self:UpdateLVList()
end

function tbWnd:OnClose()
  local tbElementList = self.tbDailySignInList:GetAllElement()
  for _, tbElement in pairs(tbElementList) do
    RedPointMgr:DynamicUnRegist(tbElement.pPanel, "reward1/btnGet")
  end
end

function tbWnd:UpdateLVList()
  local nLevelupGetNum = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipLevelupGetState)
  local nLevelupExtraGetNum = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipLevelupExtraGetState)
  self.tbDailySignInList:Clear()
  for i = 1, Lib:CountTB(FriendsReturn.tbVNVipAwardInfoById) do
    self.tbDailySignInList:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtLv", i18n.EasyGet("{0}\231\186\167", FriendsReturn.tbVNVipAwardInfoById[i].nLevel))
      local tbAward = Award:GetAward(FriendsReturn.tbVNVipAwardInfoById[i].nAwardId, FriendsReturn.tbVNVipAwardInfoById[i].nAwardKey)
      local itemGrid = Ui:NewExistItemGrid(tbElement, "reward1/UIItemGrid")
      itemGrid:SetItemByTemplate(tbAward[1][2], tbAward[1][3])
      itemGrid:SetClickEnable(true)
      tbElement.pPanel:Button_BindEvent("reward1/btnGet", function()
        FriendsReturn:GetVNVipLevelupAwardReq(i, 1)
      end)
      tbElement.pPanel:SetActive("reward1/btnGet", me.nLevel >= FriendsReturn.tbVNVipAwardInfoById[i].nLevel and not Lib:GetBit(nLevelupGetNum, i))
      tbElement.pPanel:SetActive("reward1/img_Finish", Lib:GetBit(nLevelupGetNum, i))
      if me.nLevel >= FriendsReturn.tbVNVipAwardInfoById[i].nLevel and not Lib:GetBit(nLevelupGetNum, i) then
        RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "reward1/btnGet", {
          RedPointMgr.EVENT.ReturnerVNVIPNewServerLevelAward
        })
      end
      local tbExtraAward = Award:GetAward(FriendsReturn.tbVNVipAwardInfoById[i].nExtraAwardId, FriendsReturn.tbVNVipAwardInfoById[i].nExtraAwardKey)
      local itemGrid = Ui:NewExistItemGrid(tbElement, "reward2/UIItemGrid")
      itemGrid:SetItemByTemplate(tbExtraAward[1][2], tbExtraAward[1][3])
      itemGrid:SetClickEnable(true)
      tbElement.pPanel:Button_BindEvent("reward2/btnGet", function()
        FriendsReturn:GetVNVipLevelupAwardReq(i, 2)
      end)
      tbElement.pPanel:SetActive("reward2/btnGet", me.nLevel >= FriendsReturn.tbVNVipAwardInfoById[i].nLevel and not Lib:GetBit(nLevelupExtraGetNum, i))
      tbElement.pPanel:SetActive("reward2/img_Finish", Lib:GetBit(nLevelupExtraGetNum, i))
      tbElement.pPanel:Label_SetText("reward2/btnGet/txtCost", FriendsReturn.tbVNVipAwardInfoById[i].nExtraAwardPrice)
    end)
  end
end

function tbWnd:OnGet()
  local nState = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipAwardCheck)
  if 1 == nState then
    FriendsReturn:GetVNVipNewServerYuanBaoReq()
  elseif 3 == nState then
    local function fnCardPick()
      FriendsReturn:BecomeVNVipReq()
    end
    
    Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\228\184\186<color=#0A6F32>{0}</color>\230\156\141<color=#0A6F32>{1}</color>\232\167\146\232\137\178\231\187\145\229\174\154\229\134\141\233\151\175\230\177\159\230\185\150\230\150\176\230\156\141\231\137\185\230\157\131\239\188\159\239\188\136\231\161\174\232\174\164\229\144\142\230\151\160\230\179\149\230\146\164\233\148\128\239\188\140\230\151\160\230\179\149\230\155\180\230\141\162\232\167\146\232\137\178\239\188\137", Login.ServerName, me.szName), {
      {fnCardPick},
      {}
    }, {
      i18n.szOk,
      i18n.szCancel
    })
  end
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnGet, {
    RedPointMgr.EVENT.ReturnerVNVipSignIn
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, self.btnGet)
end
