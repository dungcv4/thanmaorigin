YuanBaoReturn.SaveGroup = GetGroupKey("YuanBaoReturn")
YuanBaoReturn.TotalNumKey = 1
YuanBaoReturn.LastGetDayKey = 2
YuanBaoReturn.TotalGetDayKey = 3
YuanBaoReturn.TotalGetNumKey = 4
YuanBaoReturn.RefreshOffSet = 14400
YuanBaoReturn.Protocol = {GetAward = 1}
YuanBaoReturn.Stage = {First = 10000, Second = 70000}
YuanBaoReturn.MaxGetDay = 7

function YuanBaoReturn:GetRemainNum(pPlayer)
  local nTotalNum = pPlayer.GetUserValue(self.SaveGroup, self.TotalNumKey)
  local nTotalGetNum = pPlayer.GetUserValue(self.SaveGroup, self.TotalGetNumKey)
  return nTotalNum - nTotalGetNum
end

function YuanBaoReturn:GenTodayYuanBaoNum(dwID, nTotalNum, nTotalGetNum, nTotalGetDayNum)
  if nTotalNum <= self.Stage.First then
    return nTotalNum
  end
  if nTotalNum > self.Stage.First and nTotalNum <= self.Stage.Second then
    local nMaxDay = math.ceil(nTotalNum / self.Stage.First)
    if nTotalGetDayNum + 1 == nMaxDay then
      return nTotalNum - nTotalGetDayNum * self.Stage.First
    end
    return self.Stage.First
  end
  local nNum = math.floor(nTotalNum / YuanBaoReturn.MaxGetDay)
  local nSpecialDay = nTotalNum % YuanBaoReturn.MaxGetDay
  if nTotalGetDayNum < nSpecialDay then
    nNum = nNum + 1
  end
  return nNum
end
