local tbWnd = Ui:CreateClass("UINewInfomation_GradeRankingActivity", "UIPanel")
local tbControls = {
  Template = "imgList/List/Template",
  ItemTemplate = "imgList/List/Template/AwardList/Viewport/Content/ItemTemplate",
  txtInfo2 = "txtInfo2"
}
tbWnd.tbControls = tbControls

function tbWnd:Init()
  self.tbActivity = ActivityMgr:GetGradeRankingActivityInstance()
  self.ACTIVITY_CONFIG = self.tbActivity.ACTIVITY_CONFIG
  self.AWARD_CONFIG = self.tbActivity.AWARD_CONFIG
  self.nAwardTypeCount = self.ACTIVITY_CONFIG[1].nAwardTypeCount
end

function tbWnd:OnCreate()
  if not self.tbActivity then
    self:Init()
  end
  self.tbActivityScrollView = Ui:NewScrollView(self, tbControls.Template, nil, true, false)
  self.tbActivityItemGridView = {}
  self.tbAwardTypeElements = {}
end

function tbWnd:OnOpen()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_GRADERANKING_DATA_SYNC, self.ReOpen, self)
  if self.tbActivity then
    ActivityMgr:CallServerScript(self.tbActivity.nId)
    self:ShowAwardList()
  end
end

function tbWnd:ShowAwardList()
  self.pPanel:Label_SetText(tbControls.txtInfo2, i18n.EasyGet("\229\137\141<color=#138446>{0}</color>\229\144\141\229\136\176\232\190\190<color=#138446>{1}</color>\231\186\167\231\154\132\228\190\160\229\163\171\229\176\134\232\142\183\229\190\151\228\184\176\229\142\154\229\165\150\229\138\177", self.ACTIVITY_CONFIG[1].nMaxPlayerCount, self.ACTIVITY_CONFIG[1].nLevelFlag))
  local tbRemainCount = self.tbActivity:_GetRemainCount()
  if #tbRemainCount ~= self.nAwardTypeCount then
    tbRemainCount = {
      1,
      9,
      90
    }
  end
  for nIndex = 1, self.nAwardTypeCount do
    self.tbActivityScrollView:SetElement(nIndex, function(tbElement)
      if nIndex <= 1 then
        tbElement.pPanel:Label_SetText("txtRanking", i18n.EasyGet("\231\172\172{0}\229\144\141", self.AWARD_CONFIG[nIndex].nRankLowerBound))
      else
        tbElement.pPanel:Label_SetText("txtRanking", i18n.EasyGet("\231\172\172{0}~{1}\229\144\141", self.AWARD_CONFIG[nIndex - 1].nRankLowerBound + 1, self.AWARD_CONFIG[nIndex].nRankLowerBound))
      end
      tbElement.pPanel:Label_SetText("Remain/txtRemainValue", tbRemainCount[nIndex])
      self.tbAwardTypeElements[nIndex] = tbElement
    end, false)
  end
  for nIndex = 1, self.nAwardTypeCount do
    self:SetItems(nIndex)
  end
end

function tbWnd:SetItems(nIndex)
  if not self.tbActivityItemGridView[nIndex] then
    self.tbActivityItemGridView[nIndex] = Ui:NewItemGridView(self.tbAwardTypeElements[nIndex], "AwardList/Viewport/Content/ItemTemplate")
  end
  local tbAwards = Award:GetAward(self.ACTIVITY_CONFIG[1].nAwardId, nIndex)
  for i = 1, #tbAwards do
    self.tbActivityItemGridView[nIndex]:SetItem(i, tbAwards[i], true)
  end
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_GRADERANKING_DATA_SYNC, self)
end

function tbWnd:ReOpen()
  if not self.pPanel then
    return
  end
  if self.pPanel then
    self:ShowAwardList()
  end
end

function tbWnd:OnDestroy()
  self.tbAwardTypeElements = {}
  self.tbActivityItemGridView = {}
end
