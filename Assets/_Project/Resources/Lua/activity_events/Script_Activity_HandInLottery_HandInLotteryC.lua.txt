local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.HandInLottery)
local Protocol = {
  [tbActivity.Protocol.SyncLotteryDataResp] = "OnSyncLotteryDataResp"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnClose()
  self.tbLotteryData = nil
end

function tbActivity:OnSyncLotteryDataResp(tbLotteryData)
  self.tbLotteryData = tbLotteryData
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_HANDINLOTTERY_DATA, self.tbLotteryData)
end

function tbActivity:GetLotteryData()
  return self.tbLotteryData
end

function tbActivity:GetRankList(nBatch)
  if self.tbLotteryData ~= nil and nil ~= self.tbLotteryData.tbAwardRecord then
    return self.tbLotteryData.tbAwardRecord[nBatch]
  end
  return nil
end
