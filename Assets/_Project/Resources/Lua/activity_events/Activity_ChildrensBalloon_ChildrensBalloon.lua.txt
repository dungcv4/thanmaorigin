Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ChildrensBalloon)
tbActivity.SaveID = {
  LastStartTimeKey = 1,
  EveryDayGetNum = 2,
  EveryDayFreeNum = 3,
  EveryDayInteractNum = 4
}
tbActivity.Protocol = {SyncData = 1, GetBalloon = 2}

function tbActivity:LoadConfig()
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile("Setting/Activity/ChildrensBalloon/ChildrensBalloonCfg.tab", {"nConfigId"})
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  return true
end

function tbActivity:GetEveryDayGetNum(pPlayer)
  if not self.SaveKey then
    self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
  end
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SaveID.EveryDayGetNum)
end

function tbActivity:GetEveryDayFreeNum(pPlayer)
  if not self.SaveKey then
    self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
  end
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SaveID.EveryDayFreeNum)
end

function tbActivity:GetEveryDayInteractNum(pPlayer)
  if not self.SaveKey then
    self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
  end
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SaveID.EveryDayInteractNum)
end

function tbActivity:CanJoin(nLevel)
  return nLevel >= self.tbConfig.nJoinLevel
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szTitle = i18n.Encode(self.tbConfig.szTitle)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szNotice = i18n.Encode(self.tbConfig.szNotice)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig = SetReadOnly(self.tbConfig, true)
  return true
end

function tbActivity:CheckConfigValid(tbConfig)
  return true
end

function tbActivity:AddNpc(nNpcId)
  if not self.tbNpcIdList then
    self.tbNpcIdList = {}
  end
  if self.tbNpcIdList[nNpcId] then
    return
  end
  self.tbNpcIdList[nNpcId] = true
end

function tbActivity:RemoveNpc(nNpcId)
  if not self.tbNpcIdList then
    return
  end
  if not self.tbNpcIdList[nNpcId] then
    return
  end
  local pNpc = KNpc.GetById(nNpcId)
  if pNpc and pNpc.nActivityId == self.nId then
    pNpc.Delete()
  end
  self.tbNpcIdList[nNpcId] = nil
end
