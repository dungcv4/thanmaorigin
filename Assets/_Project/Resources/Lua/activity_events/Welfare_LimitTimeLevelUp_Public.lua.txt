LimitTimeLevelUp = LimitTimeLevelUp or {}
LimitTimeLevelUp.GROUP = GetGroupKey("LimitTimeLevelUp")
LimitTimeLevelUp.AwardId = 1052

function LimitTimeLevelUp:LoadSetting()
  self.tbSetting = Lib:EasyLoadTabFile("Setting/Welfare/LimitTimeLevelUp.tab", {"nIndex"})
end

LimitTimeLevelUp:LoadSetting()

function LimitTimeLevelUp:GetSettngs()
  return self.tbSetting
end

function LimitTimeLevelUp:GetLevel(nIndex)
  if self.tbSetting[nIndex] then
    return self.tbSetting[nIndex].nLevel
  end
end

function LimitTimeLevelUp:GetUserValueKey(nIndex)
  if self.tbSetting[nIndex] then
    return self.tbSetting[nIndex].nUserValueKey
  end
end

function LimitTimeLevelUp:GetAward(nIndex)
  local nAwardKey, tbAward
  if self.tbSetting[nIndex] then
    nAwardKey = self.tbSetting[nIndex].nAwardKey
    tbAward = Award:GetAward(self.AwardId, nAwardKey)
  end
  return tbAward, nAwardKey
end

function LimitTimeLevelUp:CheckCanGet(pPlayer, nIndex)
  local nUserValue = pPlayer.GetUserValue(self.GROUP, self:GetUserValueKey(nIndex))
  if 1 == nUserValue then
    return false, true
  end
  if not pPlayer then
    return false, false
  end
  if pPlayer.nLevel < self:GetLevel(nIndex) then
    return false, false
  end
  return true, false
end
