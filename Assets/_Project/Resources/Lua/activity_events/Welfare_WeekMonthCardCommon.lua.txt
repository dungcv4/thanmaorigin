WeekMonthCard.SAVE_GROUP = GetGroupKey("WeekMonthCard")
WeekMonthCard.WeekGetLocalDay = 1
WeekMonthCard.WeekEndTime = 2
WeekMonthCard.MonthGetLocalDay = 3
WeekMonthCard.MonthEndTime = 4
WeekMonthCard.LimitTimeMonthEndTime = 5
WeekMonthCard.nLimitTimeBuffId = 122
WeekMonthCard.tbIndex2ProductId = {
  [1] = 17,
  [2] = 18
}
WeekMonthCard.tbProductId2Index = {
  [17] = 1,
  [18] = 2
}
WeekMonthCard.nOpenLevel = 11

function WeekMonthCard:GetWeekCardEndTime(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.WeekEndTime)
end

function WeekMonthCard:GetWeekCardLeftDay(pPlayer)
  local nLeftDay = Lib:GetLocalDay(pPlayer.GetUserValue(self.SAVE_GROUP, self.WeekEndTime) - Recharge.CARD_BUY_DAY_UPDATE_TIME) - Lib:GetLocalDay(GetTime() - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  return nLeftDay < 0 and 0 or nLeftDay
end

function WeekMonthCard:GetMonthCardEndTime(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.MonthEndTime)
end

function WeekMonthCard:CheckMonthCardState(pPlayer)
  local nMonthCardEndTime = self:GetMonthCardEndTime(pPlayer)
  if nMonthCardEndTime >= GetTime() then
    return true
  end
  local nLMMonthCardEndTime = self:GetLimitTimeMonthCardEndTime(pPlayer)
  if nLMMonthCardEndTime >= GetTime() then
    return true
  end
  return false
end

function WeekMonthCard:GetMonthCardLeftDay(pPlayer)
  local nLeftDay = Lib:GetLocalDay(pPlayer.GetUserValue(self.SAVE_GROUP, self.MonthEndTime) - Recharge.CARD_BUY_DAY_UPDATE_TIME) - Lib:GetLocalDay(GetTime() - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  return nLeftDay < 0 and 0 or nLeftDay
end

function WeekMonthCard:GetLimitTimeMonthCardEndTime(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.LimitTimeMonthEndTime)
end

function WeekMonthCard:GetWeedPerDayYuanBao()
  local nRet = self.Cfg.nWeedPerDayYuanBao
  local nRet2 = self.Cfg.nWeedAllYuanBao
  for _, tbV in ipairs(WeekMonthCard.Cfg.tbTimeframe2DayGet) do
    if 1 == GetTimeFrameState(tbV.szTimeFrame) then
      nRet = tbV.WeedDayGet
      nRet2 = tbV.WeedAllGet
    else
      break
    end
  end
  return nRet, nRet2
end

function WeekMonthCard:GetMonthPerDayYuanBao()
  local nRet = self.Cfg.nMonthPerDayYuanBao
  local nRet2 = self.Cfg.nMonthAllYuanBao
  for _, tbV in ipairs(WeekMonthCard.Cfg.tbTimeframe2DayGet) do
    if 1 == GetTimeFrameState(tbV.szTimeFrame) then
      nRet = tbV.MonthDayGet
      nRet2 = tbV.MonthAllGet
    else
      break
    end
  end
  return nRet, nRet2
end

WeekMonthCard.nMaxGiveYuanBao = 100000

function WeekMonthCard:CheckWeekMonthCardConfigValid()
  if WeekMonthCard:GetWeedPerDayYuanBao() > self.nMaxGiveYuanBao then
    LogErr(string.format("error CheckWeekMonthCardConfigValid nWeedPerDayYuanBao(%d)(%d)", WeekMonthCard:GetWeedPerDayYuanBao(), self.nMaxGiveYuanBao))
    return false
  end
  if self.Cfg.nWeedAllYuanBao > self.nMaxGiveYuanBao then
    LogErr(string.format("error CheckWeekMonthCardConfigValid nWeedAllYuanBao(%d)(%d)", self.Cfg.nWeedAllYuanBao, self.nMaxGiveYuanBao))
    return false
  end
  if WeekMonthCard:GetMonthPerDayYuanBao() > self.nMaxGiveYuanBao then
    LogErr(string.format("error CheckWeekMonthCardConfigValid nMonthPerDayYuanBao(%d)(%d)", WeekMonthCard:GetMonthPerDayYuanBao(), self.nMaxGiveYuanBao))
    return false
  end
  if self.Cfg.nMonthAllYuanBao > self.nMaxGiveYuanBao then
    LogErr(string.format("error CheckWeekMonthCardConfigValid nMonthAllYuanBao(%d)(%d)", self.Cfg.nMonthAllYuanBao, self.nMaxGiveYuanBao))
    return false
  end
  return true
end

Validator:Register(WeekMonthCard.CheckWeekMonthCardConfigValid, WeekMonthCard)
