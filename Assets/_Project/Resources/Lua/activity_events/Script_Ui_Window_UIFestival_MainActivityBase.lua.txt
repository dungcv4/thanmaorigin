local tbWnd = Ui:CreateWindow("UIFestival_MainActivityBase")

function tbWnd:OnCreate()
  self.tbMainActivityScrollView = Ui:NewScrollView(self, self.Template)
end

function tbWnd:OnOpen(nEntranceIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex)
  end
  for nIndex, nRedPointEventId in ipairs(self.tbFadeRedPointEvent) do
    RedPointMgr:FadeRedPoint(nRedPointEventId)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local szShowTime = ActivityMgr.EntranceSchedule:GetSmallTitle(nEntranceIndex)
  self.pPanel:Label_SetText(self.szTitle, ActivityMgr.EntranceSchedule:GetTitle(nEntranceIndex))
  self.pPanel:Label_SetText(self.szTime, ActivityMgr.EntranceSchedule:GetSmallTitle(nEntranceIndex))
  local nEndTime = Lib:GetConfigDate2Time(ActivityMgr.EntranceScheduleCfg[nEntranceIndex].szEndTime)
  if nEndTime > GetTime() then
    local nDiff = nEndTime - GetTime()
    local szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", szShowTime, Lib:TimeDesc2(nDiff))
    self.pPanel:Label_SetText(self.szTime, szTime)
  end
  self:UpdateActivityList()
  self.pPanel:SetActive(self.btnHelp, false)
end

function tbWnd:UpdateActivityList()
  self.tbMainActivityScrollView:Clear()
  for index, tbInfo in ipairs(self.tbActivityList) do
    self.tbMainActivityScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:Label_SetText(self.tbTemplateElement.txtActivityName, tbInfo.szName)
      tbElement.pPanel:Label_SetText(self.tbTemplateElement.txtTimes, tbInfo.szShowTime)
      tbElement.pPanel:Label_SetText(self.tbTemplateElement.txtInfo, tbInfo.szDesc)
      local pInst = ActivityMgr:GetInstanceById(tbInfo.nId)
      local bOpen = false
      local bEnd = false
      if pInst then
        local nStartTime = pInst:GetStartTime()
        local nEndTime = pInst:GetEndTime()
        if nStartTime < GetTime() then
          bOpen = true
        end
        if nEndTime < GetTime() then
          bEnd = true
        end
      end
      tbElement.pPanel:SetActive(self.tbTemplateElement.btnGo, bOpen and not bEnd)
      tbElement.pPanel:SetActive(self.tbTemplateElement.img_Finish, bEnd)
      tbElement.pPanel:SetActive(self.tbTemplateElement.txtUnOpened, not bOpen)
      Ui:Button_BindEvent(tbElement.pPanel, self.tbTemplateElement.btnGo, self.OnClickJoinBtn, self, tbInfo.pFunc, tbInfo.nId, table.unpack(Lib:SplitStrTry2Num(tbInfo.szJumpParam, "|")))
      local tbShowAwardScrollView = Ui:NewItemGridScrollView(tbElement, self.tbTemplateElement.ItemTemplate, 1, 1)
      tbShowAwardScrollView:UpdateGenericItem(tbInfo.tbAward, true)
      if tbInfo.tbRedPointEventId and 0 ~= #tbInfo.tbRedPointEventId then
        RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, self.tbTemplateElement.btnGo, tbInfo.tbRedPointEventId)
      end
    end)
  end
  self.tbMainActivityScrollView:PlayItemGridAnimation()
end

function tbWnd:OnClickJoinBtn(func, ...)
  func(...)
end

function tbWnd:OnClose()
  self.tbMainActivityScrollView:Clear()
end
