local tbWnd = Ui:CreateWindow("UIActivity_SuperStar")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnSignup = "node/imgBG/btnSignup",
  btnJoin = "node/imgBG/btnJoin",
  btnGameRule = "node/imgBG/btnGameRule",
  btnReward = "node/imgBG/btnReward",
  btnRule = "node/imgBG/btnRule ",
  btnGameReword = "node/imgBG/btnGameReword",
  btnGameFriend = "node/imgBG/btnGameFriend",
  liveCell = "node/imgBG/liveCell",
  btnLive = "node/imgBG/liveCell/btnLive",
  btnLiveReward = "node/imgBG/liveCell/btnLiveReward",
  txtRewardCountdown = "node/imgBG/liveCell/txtRewardCountdown",
  UIItemGrid = "node/imgBG/liveCell/liveReward/UIItemGrid",
  img_Have = "node/imgBG/liveCell/img_Have",
  txtTime = "node/imgBG/sloganCell/txtTime",
  txtRule = "node/imgBG/sloganCell/txtRule",
  btnProcess = "node/imgBG/processCell/btnProcess",
  txtProcessTitle = "node/imgBG/processCell/txtProcessTitle",
  txtProcess = "node/imgBG/processCell/txtProcess",
  txtProcessTime = "node/imgBG/processCell/txtProcessTime",
  btnStatus = "node/imgBG/processCell/btnStatus",
  RewardTemplate = "node/imgBG/rewardCell/RewardScrollView/Viewport/Content/RewardTemplate"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnRule = function(self)
  end,
  btnLive = function()
    local bRet, szErr = SuperstarWars:IsStageOpen(SuperstarWars.STAGE.FINAL)
    if not bRet then
      me.CenterMsg(szErr)
      return
    end
    if Sdk:GetOsType() == Sdk.eOSType_Windows then
      Ui:GotoWeb(SuperstarWars.Cfg.WEB_SITE_WINDOWS)
    else
      Ui:OpenWindow("UITransparentWebView", SuperstarWars.Cfg.WEB_SITE, 0, 0)
    end
  end,
  btnLiveReward = function()
    if SuperstarWars:IsStageOpen(SuperstarWars.STAGE.FINAL) then
      me.CenterMsg(i18n.EasyGet("\231\155\180\230\146\173\230\154\130\230\156\170\229\188\128\229\144\175"))
      return
    end
    SuperstarWars:SendProtocolGetUIAward()
  end,
  btnJoin = function()
    SuperstarWars:SendProtocolSwitchMap()
  end,
  btnGameRule = function()
    SuperstarWars:OpenRuleUI(1)
  end,
  btnReward = function()
    SuperstarWars:OpenRuleUI(2)
  end,
  btnGameFriend = function()
    Ui:OpenWindow("UIPartnerSettingWuLin", "SuperstarWars")
  end,
  btnGameReword = function()
    local szCrossServerRankWebsite = "https://kto.vnggames.net/bxh"
    Ui:OpenWindow("UIGeneralWebView", szCrossServerRankWebsite, 0, 0)
  end,
  btnProcess = function()
    SuperstarWars:OpenScheduleUI(true)
  end,
  btnStatus = function()
    SuperstarWars:OpenScheduleUI(false)
  end,
  btnSignup = function()
    SuperstarWars:SendProtocolSignUp()
  end
}

function tbWnd:OnCreate()
  self.pItemGrid = Ui:NewExistItemGrid(self, self.UIItemGrid)
  self.pPanel:SetActive(tbControls.RewardTemplate .. "/txtItemName", false)
  self.tbReAwardScrollView = Ui:NewItemGridScrollView(self, tbControls.RewardTemplate, 1, 1)
end

function tbWnd:OnOpen()
  self:UpdateUITimeAward()
  self:UpdateStage()
  self:UpdateReAward()
  self.pPanel:SetActive(tbControls.btnRule, false)
end

function tbWnd:OnClose()
  if self.nRemainTimer ~= nil then
    Timer:Close(self.nRemainTimer)
    self.nRemainTimer = nil
  end
end

function tbWnd:UpdateUITimeAward()
  if not OnOffController:Check(OnOffController.KEY.SuperstarWarsLive) then
    self.pPanel:SetActive(tbControls.liveCell, false)
    return
  end
  self.pPanel:SetActive(tbControls.liveCell, true)
  local tbAwards = Award:GetAward(SuperstarWars.UILiveAwardCfg.AWARD_ID, SuperstarWars.UILiveAwardCfg.AWARD_KEY, {})
  self.pItemGrid:SetGenericItem(tbAwards[1])
  self.pItemGrid:SetClickEnable(true)
  if SuperstarWars:IsStageOpen(SuperstarWars.STAGE.FINAL) then
    self.pPanel:SetActive(tbControls.img_Have, false)
    self.pPanel:SetActive(tbControls.btnLiveReward, false)
    self.pPanel:SetActive(tbControls.txtRewardCountdown, false)
    return
  end
  local nGotAward = me.GetUserValue(SuperstarWars.SAVE_GROUP, SuperstarWars.SAVE_UI_AWARD)
  local nOpenUITime = me.GetUserValue(SuperstarWars.SAVE_GROUP, SuperstarWars.SAVE_OPEN_UI_TIME)
  if 1 == nGotAward then
    self.pPanel:SetActive(tbControls.img_Have, true)
    self.pPanel:SetActive(tbControls.btnLiveReward, false)
    self.pPanel:SetActive(tbControls.txtRewardCountdown, false)
  else
    self.pPanel:SetActive(tbControls.img_Have, false)
    self.pPanel:SetActive(tbControls.txtRewardCountdown, true)
    if 0 == nOpenUITime then
      SuperstarWars:SendProtocolFirstOpenUI()
      self.pPanel:Label_SetText(tbControls.txtRewardCountdown, Lib:TimeDesc2(SuperstarWars.UILiveAwardCfg.AWARD_TIME))
    elseif GetTime() - nOpenUITime >= SuperstarWars.UILiveAwardCfg.AWARD_TIME then
      self.pPanel:SetActive(tbControls.btnLiveReward, true)
      self.pPanel:Label_SetText(tbControls.txtRewardCountdown, Lib:TimeDesc2(0))
    else
      self.pPanel:SetActive(tbControls.btnLiveReward, false)
      if self.nRemainTimer then
        Timer:Close(self.nRemainTimer)
      end
      self.pPanel:Label_SetText(tbControls.txtRewardCountdown, Lib:TimeDesc2(SuperstarWars.UILiveAwardCfg.AWARD_TIME - (GetTime() - nOpenUITime)))
      self.nRemainTimer = Timer:Register(Env.GAME_FPS, self.UpdateUITimeAward, self)
    end
  end
end

function tbWnd:UpdateStage()
  local nStartTime = Lib:GetConfigDate2Time(SuperstarWars.Cfg.START_DATE)
  local nEndTime = Lib:GetConfigDate2Time(SuperstarWars.Cfg.END_DATE)
  self.pPanel:Label_SetText(tbControls.txtTime, string.format("%s-%s", Lib:TimeDesc10(nStartTime), Lib:TimeDesc10(nEndTime)))
  self.pPanel:Label_SetText(tbControls.txtRule, i18n.Get(SuperstarWars.tbText.szMainUITips))
  if not SuperstarWars:IsSysOpen() then
    self.pPanel:Label_SetText(tbControls.txtProcessTitle, "")
    self.pPanel:Label_SetText(tbControls.txtProcess, i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    self.pPanel:Label_SetText(tbControls.txtProcessTime, "")
    return
  end
  local nNowTime = GetTime()
  local tbLastTimeSetting = SuperstarWars.tbTimeSettings[#SuperstarWars.tbTimeSettings]
  local nStageEndTime = Lib:GetConfigDate2Time(tbLastTimeSetting.nTimeEnd)
  if nNowTime > nStageEndTime then
    self.pPanel:Label_SetText(tbControls.txtProcessTitle, i18n.EasyGet("\229\183\178\231\187\147\230\157\159"))
    self.pPanel:Label_SetText(tbControls.txtProcess, i18n.Parse(tbLastTimeSetting.szDesc))
    self:UpdateProcessTime(nStartTime, nEndTime)
    return
  end
  local nNowTime = GetTime()
  for nStage, tbTimeSetting in ipairs(SuperstarWars.tbTimeSettings) do
    local nStartTime = Lib:GetConfigDate2Time(tbTimeSetting.nTimeStart)
    local nEndTime = Lib:GetConfigDate2Time(tbTimeSetting.nTimeEnd)
    if SuperstarWars:IsStageOpen(nStage) then
      self.pPanel:Label_SetText(tbControls.txtProcessTitle, i18n.EasyGet("\230\173\163\229\156\168\232\191\155\232\161\140"))
      self.pPanel:Label_SetText(tbControls.txtProcess, i18n.Parse(tbTimeSetting.szDesc))
      self:UpdateProcessTime(nStartTime, nEndTime)
      break
    elseif nNowTime < nStartTime then
      self.pPanel:Label_SetText(tbControls.txtProcessTitle, i18n.EasyGet("\228\184\139\228\184\170\232\181\155\231\168\139"))
      self.pPanel:Label_SetText(tbControls.txtProcess, i18n.Parse(tbTimeSetting.szDesc))
      self:UpdateProcessTime(nStartTime, nEndTime)
      break
    end
  end
end

function tbWnd:UpdateProcessTime(nStartTime, nEndTime)
  local szStartTime = os.date("%m/%d", nStartTime)
  local szEndTime = os.date("%m/%d", nEndTime)
  if nStartTime < nEndTime then
    self.pPanel:Label_SetText(tbControls.txtProcessTime, string.format("%s-%s", szStartTime, szEndTime))
  else
    self.pPanel:Label_SetText(tbControls.txtProcessTime, szStartTime)
  end
end

function tbWnd:UpdateReAward()
  if not SuperstarWars.UIShowAward then
    return
  end
  local tbAwards = Award:GetAward(SuperstarWars.UIShowAward.AWARD_ID, SuperstarWars.UIShowAward.AWARD_KEY)
  self.tbReAwardScrollView:UpdateGenericItem(tbAwards, true)
end
