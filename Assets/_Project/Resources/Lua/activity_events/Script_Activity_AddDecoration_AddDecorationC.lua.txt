Require("CommonScript/Activity/AddDecoration/AddDecoration.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.AddDecoration)

function tbActivity:GoToDynamicNpc(nNpcTemplateId, nMapTemplateId)
  if not nNpcTemplateId or not nMapTemplateId then
    LogErr("AddDecoration:GoToDynamicNpc some wrong happened", nNpcTemplateId, nMapTemplateId)
    return
  end
  local tbNpcPosInfo = self:GetNpcPosInfo()
  if tbNpcPosInfo and tbNpcPosInfo[nMapTemplateId] then
    for _, tbInfo in pairs(tbNpcPosInfo[nMapTemplateId]) do
      if tbInfo.NpcTemplateId == nNpcTemplateId then
        ActivityMgr:GotoPos(nNpcTemplateId, nMapTemplateId, tbInfo.PosX, tbInfo.PosY)
        break
      end
    end
  end
end
