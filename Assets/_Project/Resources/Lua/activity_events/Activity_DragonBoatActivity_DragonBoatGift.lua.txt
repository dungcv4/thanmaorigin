Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.DragonBoatGift)
tbActivity.Protocol = {
  RespPickResult = 1,
  OnPickResultResp = 2,
  RespNextLevel = 3,
  OnNextLevelResp = 4,
  OnRefreshResp = 5,
  RespRefresh = 6,
  RespClickStartBtn = 7,
  OnClickStartBtnResp = 8,
  RespClickCancelBtn = 9
}

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/DragonBoatActivity/DragonBoatGift/DragonBoatGiftCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/DragonBoatActivity/DragonBoatGift/GiftAward.tab"
  local tbGiftAward = Lib:EasyLoadTabFile(szFilepath)
  if not tbGiftAward then
    LogErr("%s not find", szFilepath)
    return false
  end
  for _, tbInfo in pairs(tbGiftAward) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("DragonBoatGift: Load GiftAward Config failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGiftAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGiftAward or {}
    if not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGiftAward[tbInfo.nLevel] then
      self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGiftAward[tbInfo.nLevel] = tbInfo
    else
      Lib:LogErrFormat("DragonBoatGift: Load GiftAward Config failed, duplicate nLevel ,nLevel=%d", tbInfo.nLevel)
      return false
    end
  end
  szFilepath = "Setting/Activity/DragonBoatActivity/DragonBoatGift/OpenRate.tab"
  local tbOpenRate = Lib:EasyLoadTabFile(szFilepath)
  if not tbOpenRate then
    LogErr("%s not find", szFilepath)
    return false
  end
  for _, tbInfo in pairs(tbOpenRate) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("DragonBoatGift: Load SimplePlayerRate Config failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbOpenRate = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbOpenRate or {}
    if not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbOpenRate[tbInfo.nOpenNum] then
      self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbOpenRate[tbInfo.nOpenNum] = {
        nSimpleRate = tbInfo.nSimpleRate,
        nWorkerRate = tbInfo.nWorkerRate
      }
    else
      LogErr("DragonBoatGift: Load OpenRate Config failed, duplicate nOpenNum, nOpenNum=%d", tbInfo.nOpenNum)
      return false
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.tbnGoldWareId = Lib:SplitStrTry2Num(self.tbConfig.tbnGoldWareId, "|")
  self.tbConfig.szSmallPanelText = i18n.Encode(self.tbConfig.szSmallPanelText)
  self.tbConfig.szBtnGoText = i18n.Encode(self.tbConfig.szBtnGoText)
  self.tbConfig.txtSecondTittle = i18n.Encode(self.tbConfig.txtSecondTittle or "")
  for _, tbInfo in ipairs(tbActCfg.GiftAward) do
    self.tbConfig.tbGiftAward = self.tbConfig.tbGiftAward or {}
    assert(not self.tbConfig.tbGiftAward[tbInfo.nLevel], string.format("DragonBoatGift:LoadConfigWithGroup duplicate level %d", tbInfo.nLevel))
    self.tbConfig.tbGiftAward[tbInfo.nLevel] = tbInfo
  end
  for _, tbInfo in ipairs(tbActCfg.OpenRate) do
    self.tbConfig.tbOpenRate = self.tbConfig.tbOpenRate or {}
    assert(not self.tbConfig.tbOpenRate[tbInfo.nOpenNum], string.format("DragonBoatGift:LoadConfigWithGroup duplicate nOpenNum %d", tbInfo.nOpenNum))
    self.tbConfig.tbOpenRate[tbInfo.nOpenNum] = tbInfo
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  local tbConfig = self:GetConfig()
  if Lib:IsEmptyStr(tbConfig.szSaveKey) then
    LogErr("[DragonBoatGift:CheckConfigValid] bad savekey! please check it! szSaveKey=", tbConfig.szSaveKey)
    return false
  end
  if Lib:IsEmptyStr(tbConfig.szScriptTableKey) then
    LogErr("[DragonBoatGift:CheckConfigValid] bad scripttable! please check it! szScriptTableKey=", tbConfig.szScriptTableKey)
    return false
  end
  if Lib:IsEmptyStr(tbConfig.szLogWay) then
    LogErr("[DragonBoatGift:CheckConfigValid] bad logway! please check it! szLogWay=", tbConfig.szLogWay)
    return false
  end
  return true
end

tbActivity.AwardType = {SimpleAward = 1, SpecialAward = 2}
tbActivity.CurLevelUV = 1
tbActivity.OpenNumUV = 2
