Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ActivityRemind)
tbActivity.nTimeoutClear = 2592000
tbActivity.szLogWay = "LogWay_ActivityRemind"

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/ActivityRemind/RemindCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/ActivityRemind/RemindInfo.tab"
  local tbRemindInfo = Lib:EasyLoadTabFile(szFilepath)
  for _, tbInfo in pairs(tbRemindInfo) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("ActivityRemind: LoadConfig RemindInfo nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRemindInfo = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRemindInfo or {}
    if self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRemindInfo[tbInfo.nRemindId] then
      LogErr(string.format("ActivityRemind: LoadConfig RemindInfo failed, duplicate nRemindId=%d", tbInfo.nRemindId))
      return false
    else
      local tbRemind = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRemindInfo
      tbRemind[tbInfo.nRemindId] = {}
      local nRemindTime = 0
      local nRemindTimeEnd = 0
      tbRemind[tbInfo.nRemindId] = tbInfo
      if tbInfo.szTimeDesc and not Lib:IsEmptyStr(tbInfo.szTimeDesc) then
        nRemindTime = Lib:ParseNumbericDateToTime(tbInfo.szTimeDesc)
      else
        LogErr(string.format("ActivityRemind: skip nRemindId = %d, szTimeDesc not exist", tbInfo.nRemindId))
      end
      tbRemind[tbInfo.nRemindId].nRemindTime = nRemindTime or 0
      if tbInfo.szTimeDescEnd and not Lib:IsEmptyStr(tbInfo.szTimeDescEnd) then
        nRemindTimeEnd = Lib:ParseNumbericDateToTime(tbInfo.szTimeDescEnd)
      end
      tbRemind[tbInfo.nRemindId].nRemindTimeEnd = nRemindTimeEnd or 0
    end
  end
  for _, tbCfg in pairs(self.ACTIVITY_CONFIG) do
    if not tbCfg.szLogWay then
      tbCfg.szLogWay = self.szLogWay
    end
  end
  self.bLoadConfig = true
  return true
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end
