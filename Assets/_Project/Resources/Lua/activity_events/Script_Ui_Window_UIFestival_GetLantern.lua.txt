local tbWnd = Ui:CreateWindow("UIFestival_GetLantern")
tbWnd.ElementPath = {
  btnGo = "ImgAll/btnGo",
  txtActivityName = "ImgAll/txtActivityName",
  btnWay = "ImgAll/btnWay",
  txtNum = "ImgAll/txtNum",
  txtInfo = "ImgAll/txtInfo"
}
tbWnd.tbType = {Person = 1, Rank = 2}
tbWnd.tbOnClick = {
  btnGift = function(self)
    Ui:OpenWindow("UIReward_Newyear", 190)
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.tbEntranceRankData = ActivityMgr.EntranceSchedule:GetEntranceRankData(nEntranceIndex)
  self.nIndex = nIndex
  self.szHelpKey = tbInfo.szHelpKey
  self.pPanel:Label_SetText(self.txtLV, i18n.EasyGet("\231\173\137\231\186\167\233\153\144\229\136\182{0}", tbInfo.nLevel))
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnGift, {
    RedPointMgr.EVENT.FestiveLanternAchieve
  })
  self:UpdateLeftActivity()
  self:UpdateRightActivity()
end

function tbWnd:UpdateListView(nType, tbAward, szTemplate)
  if not tbAward then
    return
  end
  if not self.tbListView[nType] then
    self.tbListView[nType] = Ui:NewScrollView(self, szTemplate)
  end
  self.tbListView[nType]:Clear()
  for i = 1, #tbAward do
    self.tbListView[nType]:SetElement(i, function(tbElement)
      self:UpdateListElement(tbElement, tbAward[i])
    end, true)
  end
end

function tbWnd:UpdateLeftActivity()
  local tbSetting = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].tbSetting
  if tbSetting then
    for index, szValue in ipairs(tbSetting.tbszInfo) do
      local tbPath = self.tbLeftElementPath[index]
      if tbPath then
        self.pPanel:Label_SetText(tbPath.txtSecondTittle, i18n.EasyGet("{0}", i18n.Get(szValue)))
      end
      local nShowAwardKey = tbSetting.tbShowAwardKey[index]
      local nShowItemId = tbSetting.tbShowItemId[index]
      if nShowAwardKey and nShowItemId then
        local tbAward = Award:GetAward(nShowItemId, nShowAwardKey)
        self:UpdateListView(index, tbAward, tbPath.Template)
      end
    end
  end
end

function tbWnd:UpdateRightActivity()
  local tbInfo = self.tbActivityList[self.nIndex]
  if not tbInfo then
    return
  end
  local nActId = self.tbActivityList[self.nIndex].nId
  local pInstance = ActivityMgr:GetInstanceById(nActId)
  local nEndTime = pInstance:GetEndTime()
  if nEndTime > GetTime() then
    local nDiff = nEndTime - GetTime()
    local szTime = i18n.EasyGet("<color=#1FFB24>{0}</color>{1}\229\144\142\231\187\147\230\157\159", tbInfo.szShowTime, Lib:TimeDesc2(nDiff))
    self.pPanel:Label_SetText(self.szTime, szTime)
  end
  self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\229\133\131\229\164\156\230\178\179\231\129\175"))
  self.tbScrollView:Clear()
  for index, tbInfo in ipairs(ActivityMgr.CommonEntryLinkData[nActId].tbSetp) do
    self.tbScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:SetActive(self.ElementPath.btnGo, false)
      tbElement.pPanel:SetActive(self.ElementPath.btnWay, false)
      local bIsGo = true
      if 1 == index then
        tbElement.pPanel:SetActive(self.ElementPath.btnWay, true)
        bIsGo = false
      else
        tbElement.pPanel:SetActive(self.ElementPath.btnGo, true)
      end
      tbElement.pPanel:Label_SetText(self.ElementPath.txtActivityName, i18n.Get(tbInfo.szInfo))
      tbElement.pPanel:Label_SetText(self.ElementPath.txtInfo, i18n.Get(tbInfo.szTime))
      tbElement.pPanel:Label_SetText(self.ElementPath.txtNum, i18n.EasyGet("{0}", index))
      if tbInfo.szCallBack then
        if bIsGo then
          Ui:Button_BindEvent(tbElement.pPanel, self.ElementPath.btnGo, self.OnClickGo, self, tbInfo.szCallBack, tbInfo.szParam, tbInfo.nRedId)
        else
          Ui:Button_BindEvent(tbElement.pPanel, self.ElementPath.btnWay, self.OnClickGo, self, tbInfo.szCallBack, tbInfo.szParam, tbInfo.nRedId)
        end
      end
      if tbInfo.nRedId and 0 ~= tbInfo.nRedId then
        RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, self.ElementPath.btnGo, {
          tbInfo.nRedId
        })
      end
    end)
  end
end

function tbWnd:OnClickGo(szCallBack, szParam, nRedId)
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam, "|")
  fn(self.tbActivityList[self.nIndex].nId, table.unpack(tbParam))
  if "GotoPos" == szCallBack or "GotoNPC" == szCallBack then
    Ui:CloseWindow("UITheLanternFestival")
  end
end

function tbWnd:OnClose()
  for key, ItemGridScrollView in pairs(self.tbItemGridScrollView or {}) do
    ItemGridScrollView:Clear()
  end
  self.tbScrollView:Clear()
  for _, tbValue in pairs(self.tbListView) do
    tbValue:Clear()
  end
end
