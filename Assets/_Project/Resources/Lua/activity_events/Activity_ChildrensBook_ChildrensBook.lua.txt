Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ChildrensBook)
tbActivity.SaveID = {LastStartTimeKey = 1}
tbActivity.Protocol = {SyncData = 1, GetChildrensBookAward = 2}
tbActivity.TaskLock = -1
tbActivity.Unfinished = 0
tbActivity.CanDraw = 1
tbActivity.HaveReceived = 2

function tbActivity:LoadConfig()
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile("Setting/Activity/ChildrensBook/ChildrensBook/ChildrensBookCfg.tab", {"nConfigId"})
  for _, tbInfo in pairs(self.ACTIVITY_CONFIG) do
    if 1 == tbInfo.nNeedAuctionModule then
      tbInfo.tbLevel = Lib:SplitStr2Num(tbInfo.szLevel, "|") or {}
    end
    tbInfo.nTimeOut = not Lib:IsEmptyStr(tbInfo.szTimeOut) and Lib:ParseDateTime(tbInfo.szTimeOut) or 0
  end
  local tbEvent = Lib:EasyLoadTabFile("Setting/Activity/ChildrensBook/ChildrensBook/ChildrensBookEvent.tab")
  for _, tbInfo in ipairs(tbEvent or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("ChildrensBook: LoadConfig EventAward nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEvent = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEvent or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEvent[tbInfo.nDay] = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEvent[tbInfo.nDay] or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEvent[tbInfo.nDay][tbInfo.szEventName] = tbInfo
  end
  local tbAward = Lib:EasyLoadTabFile("Setting/Activity/ChildrensBook/ChildrensBook/ChildrensBookAward.tab")
  for _, tbInfo in ipairs(tbAward or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("ChildrensBook: LoadConfig EventAward nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward[tbInfo.nDay] = tbInfo
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:CanJoin(nLevel)
  return nLevel >= self.tbConfig.nJoinLevel
end

function tbActivity:GetNowDays()
  return Lib:CalcDayOffset(14400, self.nStartTime, GetTime()) + 1
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szTitle = i18n.Encode(self.tbConfig.szTitle)
  self.tbConfig.szSecondTitle = i18n.Encode(self.tbConfig.szSecondTitle)
  self.tbConfig.szCheckAllTitle = i18n.Encode(self.tbConfig.szCheckAllTitle)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.nTimeOut = not Lib:IsEmptyStr(self.tbConfig.szTimeOut) and Lib:ParseDateTime(self.tbConfig.szTimeOut) or 0
  self.tbConfig.tbEvent = {}
  for _, tbEventInfo in pairs(tbActCfg.ChildrensBookEvent) do
    self.tbConfig.tbEvent[tbEventInfo.nDay] = self.tbConfig.tbEvent[tbEventInfo.nDay] or {}
    self.tbConfig.tbEvent[tbEventInfo.nDay][tbEventInfo.szEventName] = tbEventInfo
    self.tbConfig.tbEvent[tbEventInfo.nDay][tbEventInfo.szEventName].szTaskName = i18n.Encode(tbEventInfo.szTaskName)
  end
  self.tbConfig.tbAward = {}
  for _, tbEventInfo in pairs(tbActCfg.ChildrensBookAward) do
    self.tbConfig.tbAward[tbEventInfo.nDay] = tbEventInfo
    self.tbConfig.tbAward[tbEventInfo.nDay].szChildrensText = i18n.Encode(tbEventInfo.szChildrensText)
    self.tbConfig.tbAward[tbEventInfo.nDay].szTxtTitle = i18n.Encode(tbEventInfo.szTxtTitle)
    self.tbConfig.tbAward[tbEventInfo.nDay].szUnlock = i18n.Encode(tbEventInfo.szUnlock)
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
  return true
end

function tbActivity:CheckConfigValid(tbConfig)
  return true
end
