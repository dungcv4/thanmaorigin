function WeekMonthCard:OnChargeWeekCardReq()
  local tbSetting = Recharge:GetWeekCardSetting()
  if not tbSetting then
    me.CenterMsg("not had week card config!!!")
    return
  end
  Recharge:RequestCreateRechargeOrder(tbSetting.nProductId)
end

function WeekMonthCard:OnChargeMonthCardReq()
  local tbSetting = Recharge:GetMonthCardSetting()
  if not tbSetting then
    me.CenterMsg("not had month card config!!!")
    return
  end
  Recharge:RequestCreateRechargeOrder(tbSetting.nProductId)
end

function WeekMonthCard:GetWeekCardAward()
  RemoteServer.RequestWeekMonthCard("TakeWeekCardAward")
end

function WeekMonthCard:GetMonthCardAward()
  RemoteServer.RequestWeekMonthCard("TakeMonthCardAward")
end

function WeekMonthCard:GetMonthCardAwardState()
  local nNow = GetTime()
  local nToday = Lib:GetLocalDay(nNow - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  local nState = self.nNotBuy
  local nLeftAwardDay = self:GetMonthCardLeftDay(me)
  if nLeftAwardDay > 0 then
    if me.GetUserValue(self.SAVE_GROUP, self.MonthGetLocalDay) ~= nToday then
      nState = self.nNotGet
    else
      nState = self.nHadGet
    end
  end
  return nState, nLeftAwardDay
end

function WeekMonthCard:GetWeekCardAwardState()
  local nNow = GetTime()
  local nToday = Lib:GetLocalDay(nNow - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  local nState = self.nNotBuy
  local nLeftAwardDay = self:GetWeekCardLeftDay(me)
  if nLeftAwardDay > 0 then
    if me.GetUserValue(self.SAVE_GROUP, self.WeekGetLocalDay) ~= nToday then
      nState = self.nNotGet
    else
      nState = self.nHadGet
    end
  end
  return nState, nLeftAwardDay
end

function WeekMonthCard:OnTakeCardAwardScuess()
  EventNotify.OnNotify(EventNotify.emNOTIFY_WEEK_MONTH_CARD_GET_AWARD)
  self:UpdateReddotState()
  if Ui:WindowVisible("UIWelfare") then
    Ui:GetWindow("UIWelfare"):UpdateWeekMonthCardPanel()
  end
end

function WeekMonthCard:BuyWeekCardScucess()
  EventNotify.OnNotify(EventNotify.emNOTIFY_WEEK_MONTH_CARD_BUY)
  self:UpdateReddotState()
end

function WeekMonthCard:BuyMonthCardScucess()
  EventNotify.OnNotify(EventNotify.emNOTIFY_WEEK_MONTH_CARD_BUY)
  self:UpdateReddotState()
end

function WeekMonthCard:CheckIsOpen()
  if not me or me.nLevel < WeekMonthCard.nOpenLevel then
    return false
  end
  return true
end

function WeekMonthCard:_CheckEnableBuyDynReddot(szKey)
  self._tbDisableTemporaryReddot = self._tbDisableTemporaryReddot or {}
  if "WEEK_DAY_LIMIT" == szKey then
    return WeekMonthCard:GetWeekCardLeftDay(me) <= Recharge.WEEK_DAY_LIMIT and not self._tbDisableTemporaryReddot[szKey]
  end
  if "MON_DAY_LIMIT" == szKey then
    return WeekMonthCard:GetMonthCardLeftDay(me) <= Recharge.MON_DAY_LIMIT and not self._tbDisableTemporaryReddot[szKey]
  end
  return not self._tbDisableTemporaryReddot[szKey]
end

function WeekMonthCard:UpdateReddotState()
  if not self:CheckIsOpen() then
    return
  end
  local nWeekCardAwardState = self:GetWeekCardAwardState()
  local nMonthCardAwardState = self:GetMonthCardAwardState()
  self._tbDisableTemporaryReddot = self._tbDisableTemporaryReddot or {}
  local bNeedRedPoint = false
  if nWeekCardAwardState == self.nNotGet or nMonthCardAwardState == self.nNotGet then
    bNeedRedPoint = true
  else
    bNeedRedPoint = false
  end
  if not bNeedRedPoint and not self.bHaveClick and (nWeekCardAwardState == self.nNotBuy or nMonthCardAwardState == self.nNotBuy) then
    bNeedRedPoint = true
  end
  if bNeedRedPoint then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.WeekMonthCard)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.WeekMonthCard)
  end
end

function WeekMonthCard:OnLogin()
  self:UpdateReddotState()
end

function WeekMonthCard:RefreshTemporaryReddot(szType)
  self._tbDisableTemporaryReddot = self._tbDisableTemporaryReddot or {}
  self._tbDisableTemporaryReddot[szType] = true
  self:UpdateReddotState()
end

function WeekMonthCard:OnLogout()
  self._tbDisableTemporaryReddot = nil
end
