Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.HandInLottery)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/HandInLottery/HandInLotteryCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
    tbConfig.tbLevel = Lib:SplitStr2Num(tbConfig.szLevel, "|") or {}
    table.sort(tbConfig.tbLevel)
    tbConfig.tbForbidDuplicateRank = Lib:SplitStr2Num(tbConfig.szForbidDuplicateRank, "|") or {}
    tbConfig.tbForbidDuplicateRank = Lib:ListToSet(tbConfig.tbForbidDuplicateRank)
    assert(self.ACTIVITY_CONFIG[tbConfig.nConfigId].nLotteryNum >= Lib:CountTB(self.ACTIVITY_CONFIG[tbConfig.nConfigId].tbForbidDuplicateRank), string.format("HandInLottery: LoadConfig LotteryAward failed, nLotteryNum can't less than length of tbForbidDuplicateRank, nConfigId = %d", nConfigId))
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/HandInLottery/LotteryAward.tab"
  local tbLotteryAward = Lib:EasyLoadTabFile(szFilepath)
  if not tbLotteryAward then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  for _, tbInfo in ipairs(tbLotteryAward or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("HandInLottery: LoadConfig LotteryAward nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbLotteryAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbLotteryAward or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbLotteryAward[tbInfo.nRankStart] = tbInfo
  end
  for _, tbInfo in ipairs(tbLotteryAward or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("HandInLottery: LoadConfig tbRankingAward nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRankingAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRankingAward or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRankingAward, {
      nRankStart = tbInfo.nRankStart,
      nRankEnd = tbInfo.nRankEnd,
      nAwardKey = tbInfo.nAwardKey
    })
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

tbActivity.Protocol = {SyncLotteryDataResp = 1}
tbActivity.SaveId = {nLastJoinTime = 1, nFullFillMailIndex = 2}

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.tbForbidDuplicateRank = Lib:SplitStr2Num(self.tbConfig.szForbidDuplicateRank, "|") or {}
  self.tbConfig.tbForbidDuplicateRank = Lib:ListToSet(self.tbConfig.tbForbidDuplicateRank)
  self.tbConfig.tbLevel = Lib:SplitStr2Num(self.tbConfig.szLevel, "|") or {}
  table.sort(self.tbConfig.tbLevel)
  self.tbConfig.szFullFillMFrom = i18n.Encode(self.tbConfig.szFullFillMFrom)
  self.tbConfig.szFullFillMTitle = i18n.Encode(self.tbConfig.szFullFillMTitle)
  self.tbConfig.szFullFillMText = i18n.Encode(self.tbConfig.szFullFillMText)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.tbLotteryAward = {}
  self.tbConfig.tbRankingAward = {}
  for _, tbInfo in ipairs(tbActCfg.LotteryAward or {}) do
    tbInfo.szAwardRollTip = i18n.Encode(tbInfo.szAwardRollTip)
    self.tbConfig.tbLotteryAward[tbInfo.nRankStart] = tbInfo
    table.insert(self.tbConfig.tbRankingAward, {
      nRankStart = tbInfo.nRankStart,
      nRankEnd = tbInfo.nRankEnd,
      nAwardKey = tbInfo.nAwardKey,
      nShowAwardKey = tbInfo.nAwardKey
    })
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  return true
end
