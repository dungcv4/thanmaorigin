Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.CommonEntry)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile("Setting/Activity/Common/CommonEntry/CommonEntry.tab", {"nConfigId"}, nil, nil, true)
  self.bLoadConfig = true
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  if self.tbConfig.txtSecondTittle then
    self.tbConfig.txtSecondTittle = i18n.Encode(self.tbConfig.txtSecondTittle)
  end
  if self.tbConfig.szBtnGoText then
    self.tbConfig.szBtnGoText = i18n.Encode(self.tbConfig.szBtnGoText)
  end
  if self.tbConfig.szSmallPanelText then
    self.tbConfig.szSmallPanelText = i18n.Encode(self.tbConfig.szSmallPanelText)
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  return true
end

function tbActivity:GetRedPointEventList()
  if self.tbConfig.szRedPoint and not Lib:IsEmptyStr(self.tbConfig.szRedPoint) then
    local tbRedPoint = Lib:SplitStr(self.tbConfig.szRedPoint, "|")
    local tbRedPointEventList = {}
    for _, value in ipairs(tbRedPoint) do
      if RedPointMgr.EVENT[value] then
        table.insert(tbRedPointEventList, RedPointMgr.EVENT[value])
      end
    end
    return tbRedPointEventList
  else
    return {}
  end
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end

function tbActivity:GetLogWayFieldName()
  return ActivityMgr.NoLogWay
end
