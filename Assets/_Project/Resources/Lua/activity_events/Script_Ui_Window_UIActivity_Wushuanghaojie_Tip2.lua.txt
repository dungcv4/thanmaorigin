function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_WULINDAHUI_SYNC_GUESS,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    }
  }
end

  btnClose = function(self)

    self:AddItem()

function tbWnd:OnCreate()
  self.tbPageGroup = Ui:NewButtonGroup(self, self.tbPageBtnIndex, true, function(self, nIndex)
    self:ChangePanel(nIndex)
  end)
  self.tbScrollViewInfo = Ui:NewScrollView(self, tbControls.TemplateInfo)
  self.tbScrollViewTemplate = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  WuLinDaHui:RequestFinalMatch()
  self.tbPageGroup:Select(self.DEFAULT_TYPE)
  self.pPanel:SetActive(tbControls.UIMoneyPanel, false)
  self.pPanel:Label_SetText(tbControls.txtInfo_L06, i18n.EasyGet("\233\128\137\230\139\169\228\184\128\228\184\170\230\136\152\233\152\159\239\188\140\231\171\158\231\140\156\228\187\150\230\156\128\231\187\136\232\181\162\229\190\151\229\134\160\229\134\155"))
end

function tbWnd:OnClose()
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end

function tbWnd:UpdatePanel()
  self.tbPageGroup:Select(self.nType)
end

function tbWnd:ChangePanel(nType)
  self.nType = nType
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nNum1, nNum2 = 0, 0
  if nType == self.PANEL_TYPE.CHAMPION then
    self.pPanel:SetActive(tbControls.txtGongAndDrum01, true)
    self.pPanel:SetActive(tbControls.txtHorn01, false)
    self.pPanel:SetActive(tbControls.txtGongAndDrum02, true)
    self.pPanel:SetActive(tbControls.txtHorn02, false)
    self.pPanel:SetActive(tbControls.txtInfo_L06, true)
    self.pPanel:SetActive(tbControls.cont_txtInfo_R, true)
    self.pPanel:SetActive(tbControls.Cont_txtInfo_L, false)
    self.pPanel:SetActive(tbControls.ScrollViewCompetition, false)
    self.pPanel:SetActive(tbControls.ScrollViewChampion, true)
    self.pPanel:SetActive(tbControls.imgIcon01, false)
    self.pPanel:SetActive(tbControls.imgIcon02, false)
    self.pPanel:SetActive(tbControls.imgIcon03, true)
    self.pPanel:SetActive(tbControls.imgIcon04, true)
    nNum1 = 0
    if WuLinDaHui.tbMyChampionGuessData then
      for _, tbData in pairs(WuLinDaHui.tbMyChampionGuessData) do
        if not tbData.bSendAward then
          nNum1 = nNum1 + tbData.nGuessNum
        end
      end
    end
    nNum2 = me.GetItemCountByTypeInBag(WuLinDaHui.GuessCfg.nChampionGuessID)
    self:UpdateChampionScroll()
  elseif nType == self.PANEL_TYPE.COMPETITION then
    self.pPanel:SetActive(tbControls.txtGongAndDrum01, false)
    self.pPanel:SetActive(tbControls.txtHorn01, true)
    self.pPanel:SetActive(tbControls.txtGongAndDrum02, false)
    self.pPanel:SetActive(tbControls.txtHorn02, true)
    self.pPanel:SetActive(tbControls.txtInfo_L06, false)
    self.pPanel:SetActive(tbControls.cont_txtInfo_R, false)
    self.pPanel:SetActive(tbControls.Cont_txtInfo_L, true)
    self.pPanel:ContentSizeFitter_Refresh(tbControls.Cont_txtInfo_L .. "/GameObject")
    self.pPanel:Label_SetText(tbControls.Cont_txtInfo_L .. "/GameObject/txtInfo_L02", i18n.Get(WuLinDaHui.tbText.szMatchGuessTipsDesc))
    self.pPanel:Label_SetText(tbControls.Cont_txtInfo_L .. "/GameObject/txtInfo_L03", "")
    self.pPanel:Label_SetText(tbControls.Cont_txtInfo_L .. "/GameObject/txtInfo_L04", "")
    self.pPanel:Label_SetText(tbControls.Cont_txtInfo_L .. "/GameObject/txtInfo_L05", "")
    self.pPanel:SetActive(tbControls.ScrollViewCompetition, true)
    self.pPanel:SetActive(tbControls.ScrollViewChampion, false)
    self.pPanel:SetActive(tbControls.imgIcon01, true)
    self.pPanel:SetActive(tbControls.imgIcon02, true)
    self.pPanel:SetActive(tbControls.imgIcon03, false)
    self.pPanel:SetActive(tbControls.imgIcon04, false)
    nNum1 = 0
    if WuLinDaHui.tbMyMatchGuessData then
      for _, tbGuessData in pairs(WuLinDaHui.tbMyMatchGuessData) do
        for nTeamId, tbData in pairs(tbGuessData) do
          if not tbData.bSendAward then
            nNum1 = nNum1 + tbData.nGuessNum
          end
        end
      end
    end
    nNum2 = me.GetItemCountByTypeInBag(WuLinDaHui.GuessCfg.nMatchGuessID)
    self:UpdateMatchScroll()
  end
  self.pPanel:Label_SetText(tbControls.txtNum01, nNum1)
  self.pPanel:Label_SetText(tbControls.txtNum02, nNum2)
end

function tbWnd:AddItem()
  if self.nType == self.PANEL_TYPE.CHAMPION then
    Shop:OpenShopBuyWidDefault(WuLinDaHui.GuessCfg.nChampionGuessID)
  elseif self.nType == self.PANEL_TYPE.COMPETITION then
    Shop:OpenShopBuyWidDefault(WuLinDaHui.GuessCfg.nMatchGuessID)
  end
end

function tbWnd:UpdateChampionScroll()
  if not WuLinDaHui.tbFinalMatchList then
    return
  end
  local nResult = 0
  local nGuessedNum = 0
  if WuLinDaHui.tbMyChampionGuessData then
    for nIdx, tbData in pairs(WuLinDaHui.tbMyChampionGuessData) do
      if not tbData.bSendAward and WuLinDaHui.GuessSettingCfg[nIdx] then
        local nRightMultiple = WuLinDaHui.GuessSettingCfg[nIdx].RIGHT_MULTIPLE
        nResult = nResult + nRightMultiple * tbData.nGuessNum * WuLinDaHui.GuessCfg.nChampionYuanBao
        nGuessedNum = nGuessedNum + tbData.nGuessNum
      end
    end
  end
  self.pPanel:Label_SetText(tbControls.cont_txtInfo_R .. "/txtYuanBaoValue", nResult)
  if WuLinDaHui.tbFinalMatchList[WuLinDaHui.PHASE_CHAMPION] then
    local nTeamId = WuLinDaHui.tbFinalMatchList[WuLinDaHui.PHASE_CHAMPION][1].nTeamId
    if WuLinDaHui.tbFinalTeamData[nTeamId] then
      self.tbScrollViewTemplate:Reset()
      self.pPanel:SetActive(tbControls.Cont_Result, true)
      self.pPanel:SetActive(tbControls.txtGuessResult, false)
      self.pPanel:Label_SetText(tbControls.txtChampionpName, WuLinDaHui.tbFinalTeamData[nTeamId].szTeamName)
      self.pPanel:Label_SetText(tbControls.txtChampionpServer, GetServerName(WuLinDaHui.tbFinalTeamData[nTeamId].nServerId))
    end
    return
  end
  local tbNowGuessCfg = WuLinDaHui:GetNowGuessCfg(WuLinDaHui.tbFinalMatchList)
  if not tbNowGuessCfg then
    return
  end
  local bCanGuess = true
  if GetTime() >= Lib:ParseConfigDateTime(tbNowGuessCfg.END_TIME) then
    bCanGuess = false
  end
  local tbChampionGuessData
  if WuLinDaHui.tbMyChampionGuessData then
    for _, tbData in pairs(WuLinDaHui.tbMyChampionGuessData) do
      if not tbData.bSendAward then
        tbChampionGuessData = tbData
        break
      end
    end
  end
  self.tbScrollViewTemplate:Reset()
  local nIndex = 0
  for i, tbTeamData in pairs(WuLinDaHui.tbFinalMatchList[tbNowGuessCfg.PHASE]) do
    if tbTeamData then
      nIndex = nIndex + 1
      self.tbScrollViewTemplate:SetElement(nIndex, function(tbElementMatch)
        local pPanel = tbElementMatch.pPanel
        local nTeamId = tbTeamData.nTeamId
        if WuLinDaHui.tbFinalTeamData[nTeamId] then
          pPanel:Label_SetText("txtPlayerName01", tbTeamData.szTeamName)
          pPanel:Label_SetText("txtServer", GetServerName(WuLinDaHui.tbFinalTeamData[nTeamId].nServerId))
          pPanel:Label_SetText("txtFightingNum02", WuLinDaHui.tbFinalTeamData[nTeamId].nFightPower)
        end
        pPanel:Label_SetText("btnCheer_R/Text", i18n.EasyGet("\231\171\158\231\140\156"))
        pPanel:Label_SetText("btnFinish/Text", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
        if tbChampionGuessData then
          if tbChampionGuessData.nTeamId == nTeamId then
            pPanel:SetActive("btnCheer_R", true)
            pPanel:SetActive("btnFinish", false)
            pPanel:Button_SetEnable("btnCheer_R", bCanGuess)
            pPanel:SetActive("txtFinishNum", true)
            pPanel:Label_SetText("txtFinishNum", i18n.EasyGet("\229\183\178\231\171\158\231\140\156\239\188\154{0}", nGuessedNum))
            Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
              local nItemID = WuLinDaHui.GuessCfg.nChampionGuessID
              local szItemName = Item:GetName(nItemID, me)
              local nNum = me.GetItemCountByTypeInBag(nItemID)
              if nNum <= 0 then
                me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
                return
              end
              Ui:OpenWindow("UIActivity_Wushuanghaojie_Tip", WuLinDaHui.GUESS_TYPE.CHAMPION, nTeamId)
            end)
          else
            pPanel:SetActive("btnCheer_R", true)
            pPanel:SetActive("btnFinish", false)
            pPanel:SetActive("txtFinishNum", false)
            pPanel:Button_SetEnable("btnCheer_R", false)
          end
        else
          pPanel:SetActive("btnCheer_R", true)
          pPanel:SetActive("btnFinish", false)
          pPanel:Button_SetEnable("btnCheer_R", bCanGuess)
          pPanel:SetActive("txtFinishNum", false)
          Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
            local nItemID = WuLinDaHui.GuessCfg.nChampionGuessID
            local szItemName = Item:GetName(nItemID, me)
            local nNum = me.GetItemCountByTypeInBag(nItemID)
            if nNum <= 0 then
              me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
              return
            end
            Ui:OpenWindow("UIActivity_Wushuanghaojie_Tip", WuLinDaHui.GUESS_TYPE.CHAMPION, nTeamId)
          end)
        end
      end)
    end
  end
end

function tbWnd:UpdateMatchScroll()
  self.tbScrollViewInfo:Reset()
  self.pPanel:SetActive(tbControls.Cont_Result, false)
  local tbNowGuessCfg = WuLinDaHui:GetNowGuessCfg(WuLinDaHui.tbFinalMatchList)
  local bCanGuess = true
  local nGuessEndTime = Lib:ParseConfigDateTime(tbNowGuessCfg.END_TIME)
  if nGuessEndTime <= GetTime() then
    bCanGuess = false
  end
  local nIndexStage = 0
  for i = #WuLinDaHui.GuessSettingCfg, 1, -1 do
    local tbGuessCfg = WuLinDaHui.GuessSettingCfg[i]
    if tbGuessCfg and WuLinDaHui.tbFinalMatchList[tbGuessCfg.PHASE] then
      nIndexStage = nIndexStage + 1
      self.tbScrollViewInfo:SetElement(nIndexStage, function(tbElement)
        if not tbElement.tbScrollViewTong then
          tbElement.tbScrollViewTong = Ui:NewScrollView(tbElement, "Template")
        end
        if tbNowGuessCfg.GUESS_INDEX == i and bCanGuess then
          tbElement.pPanel:SetActive("imgBG/txtTimeTittle", true)
          tbElement.pPanel:SetActive("imgBG/txtTime", true)
          tbElement.pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc5(nGuessEndTime - GetTime()))
          self:UpdateLeftTime(tbElement.pPanel)
        else
          tbElement.pPanel:SetActive("imgBG/txtTimeTittle", false)
          tbElement.pPanel:SetActive("imgBG/txtTime", false)
        end
        tbElement.pPanel:Label_SetText("imgBG/txtActivityTittle", i18n.Get(tbGuessCfg.STAGE_NAME))
        tbElement.tbScrollViewTong:Reset()
        for j = 1, tbGuessCfg.PHASE / 2 do
          local nEnemy = tbGuessCfg.PHASE - j + 1
          tbElement.tbScrollViewTong:SetElement(j, function(tbElementMatch)
            local pPanel = tbElementMatch.pPanel
            pPanel:SetActive("btnCheer_L", true)
            pPanel:SetActive("btnCheer_R", true)
            local tbMatchTeam1 = WuLinDaHui.tbFinalMatchList[tbGuessCfg.PHASE][j]
            if tbMatchTeam1 then
              local nTeamId1 = tbMatchTeam1.nTeamId
              if nTeamId1 and WuLinDaHui.tbFinalTeamData[nTeamId1] then
                local tbTeam1 = WuLinDaHui.tbFinalTeamData[nTeamId1]
                local szTitle1 = ""
                if tbTeam1.nServerSubId then
                  szTitle1 = i18n.EasyGet("[{0}\230\156\141]", tbTeam1.nServerSubId)
                end
                pPanel:Label_SetText("txtPlayerName01", string.format("%s-%s", szTitle1, tbTeam1.szTeamName))
                pPanel:Label_SetText("txtFightingNum02", tbTeam1.nFightPower)
                pPanel:SetActive("Cont_Finish_L", false)
                if 0 == tbMatchTeam1.nResult and tbNowGuessCfg.PHASE == tbGuessCfg.PHASE and bCanGuess then
                  pPanel:Button_SetEnable("btnCheer_L", bCanGuess)
                  Ui:Button_BindEvent(pPanel, "btnCheer_L", function()
                    local nItemID = WuLinDaHui.GuessCfg.nMatchGuessID
                    local szItemName = Item:GetName(nItemID, me)
                    local nNum = me.GetItemCountByTypeInBag(nItemID)
                    if nNum <= 0 then
                      me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
                      return
                    end
                    Ui:OpenWindow("UIActivity_Wushuanghaojie_Tip", WuLinDaHui.GUESS_TYPE.COMPETITION, nTeamId1)
                  end)
                else
                  pPanel:SetActive("btnCheer_L", false)
                end
                if WuLinDaHui.tbMyMatchGuessData and WuLinDaHui.tbMyMatchGuessData[i] and WuLinDaHui.tbMyMatchGuessData[i][nTeamId1] then
                  pPanel:SetActive("Cont_Finish_L", true)
                  pPanel:SetActive("btnCheer_L", false)
                  pPanel:SetActive("Cont_Finish_R", false)
                  pPanel:SetActive("btnCheer_R", false)
                  pPanel:Label_SetText("Cont_Finish_L/txtPlayerName02 (4)", WuLinDaHui.tbMyMatchGuessData[i][nTeamId1].nGuessNum)
                end
              end
            else
              pPanel:Label_SetText("txtPlayerName01", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
              pPanel:Label_SetText("txtFightingNum02", 0)
              pPanel:SetActive("btnCheer_L", true)
              pPanel:Button_SetEnable("btnCheer_L", false)
              pPanel:SetActive("Cont_Finish_L", false)
            end
            local tbMatchTeam2 = WuLinDaHui.tbFinalMatchList[tbGuessCfg.PHASE][nEnemy]
            if tbMatchTeam2 then
              local nTeamId2 = tbMatchTeam2.nTeamId
              if nTeamId2 and WuLinDaHui.tbFinalTeamData[nTeamId2] then
                local tbTeam2 = WuLinDaHui.tbFinalTeamData[nTeamId2]
                local szTitle2 = ""
                if tbTeam2.nServerSubId then
                  szTitle2 = i18n.EasyGet("[{0}\230\156\141]", tbTeam2.nServerSubId)
                end
                pPanel:Label_SetText("txtPlayerName02", string.format("%s-%s", szTitle2, tbTeam2.szTeamName))
                pPanel:Label_SetText("txtFightingNum01", tbTeam2.nFightPower)
                pPanel:SetActive("Cont_Finish_R", false)
                if 0 == tbMatchTeam2.nResult and tbNowGuessCfg.PHASE == tbGuessCfg.PHASE and bCanGuess then
                  pPanel:Button_SetEnable("btnCheer_R", bCanGuess)
                  Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
                    local nItemID = WuLinDaHui.GuessCfg.nMatchGuessID
                    local szItemName = Item:GetName(nItemID, me)
                    local nNum = me.GetItemCountByTypeInBag(nItemID)
                    if nNum <= 0 then
                      me.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179", szItemName))
                      return
                    end
                    Ui:OpenWindow("UIActivity_Wushuanghaojie_Tip", WuLinDaHui.GUESS_TYPE.COMPETITION, nTeamId2)
                  end)
                else
                  pPanel:SetActive("btnCheer_R", false)
                end
                if WuLinDaHui.tbMyMatchGuessData and WuLinDaHui.tbMyMatchGuessData[i] and WuLinDaHui.tbMyMatchGuessData[i][nTeamId2] then
                  pPanel:SetActive("Cont_Finish_L", false)
                  pPanel:SetActive("btnCheer_L", false)
                  pPanel:SetActive("Cont_Finish_R", true)
                  pPanel:SetActive("btnCheer_R", false)
                  pPanel:Label_SetText("Cont_Finish_R/txtPlayerName02 (4)", WuLinDaHui.tbMyMatchGuessData[i][nTeamId2].nGuessNum)
                end
              end
            else
              pPanel:Label_SetText("txtPlayerName02", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
              pPanel:Label_SetText("txtFightingNum01", 0)
              pPanel:SetActive("btnCheer_R", true)
              pPanel:Button_SetEnable("btnCheer_R", false)
              pPanel:SetActive("Cont_Finish_R", false)
            end
            pPanel:Label_SetText("btnCheer_L/Text", i18n.EasyGet("\231\171\158\231\140\156"))
            pPanel:Label_SetText("btnCheer_R/Text", i18n.EasyGet("\231\171\158\231\140\156"))
            pPanel:Label_SetText("Cont_Finish_L/txtPlayerName02 (3)", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
            pPanel:Label_SetText("Cont_Finish_R/txtPlayerName02 (3)", i18n.EasyGet("\229\183\178\231\171\158\231\140\156"))
          end)
        end
      end)
    end
  end
end

function tbWnd:UpdateLeftTime(pPanel)
  local tbNowGuessCfg = WuLinDaHui:GetNowGuessCfg(WuLinDaHui.tbFinalMatchList)
  local nGuessEndTime = Lib:ParseConfigDateTime(tbNowGuessCfg.END_TIME)
  if not nGuessEndTime then
    return false
  end
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  local nTimeRemaining = nGuessEndTime - GetTime()
  if nTimeRemaining > 0 then
    pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc2(nGuessEndTime - GetTime()))
  else
    self:UpdateMatchScroll()
    return false
  end
  local nCurTime = tonumber(os.date("%Y%m%d%H%M", GetTime()))
  local nNextTime = 1
  if nTimeRemaining > 3600 then
    nNextTime = Lib:GetDate2Time(nCurTime + 1) - GetTime() + 1
  end
  if nTimeRemaining > 0 then
    self.nTimer = Timer:Register(nNextTime * Env.GAME_FPS, self.UpdateLeftTime, self, pPanel)
  end
  return false
end

function tbWnd:OnSyncItem(dwItemId, bNewItem, nAddCount)
  if dwItemId then
    local pItem = KItem.GetItemObj(dwItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId == WuLinDaHui.GuessCfg.nChampionGuessID or pItem.dwTemplateId == WuLinDaHui.GuessCfg.nMatchGuessID then
      self:UpdatePanel()
    end
  end
end
