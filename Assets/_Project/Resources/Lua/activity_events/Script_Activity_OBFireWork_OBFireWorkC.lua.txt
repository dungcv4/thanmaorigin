Require("CommonScript/Activity/OBFireWork/OBFireWork.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szOBFireWorkClassName)
local QuickUseItemWindowName = "UIQuickUseItemOB"

function tbActivity:OnOpen()
  if not self.bHasRegistered then
    Event:RegisterEvent(Event.emEVENT_OBFIREWORK_FIREWORK_QUICKUSING, self, "CheckFireWorkItemQuickUsing")
    self.bHasRegistered = true
  end
  if TongParty.nPartyStatus == TongParty.tbPartyStatus.Close then
    return
  end
  if Map:IsTongMap(me.nMapTemplateId) then
    self:CheckFireWorkItemQuickUsing(true)
  end
end

function tbActivity:OnModify(tbScheduleCfg)
  self.bLoadConfig = false
  if not self:CheckConfig(tbScheduleCfg.nConfigId) then
    return
  end
  self:InitMetaData(tbScheduleCfg)
  self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
end

function tbActivity:OnClose()
  if self.bHasRegistered then
    Event:UnRegisterEvent(Event.emEVENT_OBFIREWORK_FIREWORK_QUICKUSING, self)
    self.bHasRegistered = false
  end
  self:CheckFireWorkItemQuickUsing(false)
end

function tbActivity:CloseQuickUseItemWindow()
  if Ui:GetWindow(QuickUseItemWindowName) and Ui:GetWindow(QuickUseItemWindowName):GetItemId() == self.tbConfig.nFireWorkId then
    Ui:CloseWindow(QuickUseItemWindowName)
  end
end

function tbActivity:OpenQuickUseItemWindow(dwId)
  if not Ui:WindowVisible(QuickUseItemWindowName) then
    Ui:OpenWindow(QuickUseItemWindowName, dwId, i18n.EasyGet("\228\189\191\231\148\168"))
  else
    Ui:GetWindow(QuickUseItemWindowName):Update()
  end
end

function tbActivity:CheckFireWorkItemQuickUsing(bStatus)
  if not bStatus then
    self:CloseQuickUseItemWindow()
    return
  end
  local luaItem = me.FindItem(self.tbConfig.nFireWorkId)
  for _, pItem in ipairs(luaItem) do
    self:OpenQuickUseItemWindow(pItem.dwId)
    return
  end
  self:CloseQuickUseItemWindow()
end

function tbActivity:ServerRequest(szFuncName, ...)
  if not self[szFuncName] then
    return
  end
  self[szFuncName](self, ...)
end
