local tbWnd = Ui:CreateWindow("UIFestival_DailyBase")

function tbWnd:OnCreate()
  self.tbActivityScrollView = Ui:NewScrollView(self, self.ActivityTemplate)
  self.tbAwardScrollView = Ui:NewScrollView(self, self.AwardTemplate)
  self.tbRankAwardScrollView = Ui:NewScrollView(self, self.RankAwardTemplate)
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  self.nId = self.nId or self.tbActivityList[nIndex].nId
  self.tbSetting = self.tbSetting or ActivityMgr.CommonEntryLinkData[self.nId].tbSetting
  self.tbDailyActivityList = self.tbDailyActivityList or ActivityMgr.CommonEntryLinkData[self.nId].tbDailyActivityList
  local szShowTime = ActivityMgr.EntranceSchedule:GetSmallTitle(nEntranceIndex)
  self.szTitle = self.szTitle or self.tbActivityList[nIndex].szName
  self.pPanel:Label_SetText(self.txtPanelTitle, self.szTitle)
  local nEndTime = Lib:GetConfigDate2Time(ActivityMgr.EntranceScheduleCfg[nEntranceIndex].szEndTime)
  if nEndTime > GetTime() then
    local nDiff = nEndTime - GetTime()
    local szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", szShowTime, Lib:TimeDesc2(nDiff))
    self.pPanel:Label_SetText(self.txtTime, szTime)
  end
  self.pPanel:Label_SetText(self.txtLevel, i18n.Get(self.tbSetting.szLevelRestrict))
  self.pPanel:Label_SetText(self.txtPanelSubTitle, i18n.Get(self.tbSetting.szSubTitleInfo))
  self.pPanel:Label_SetText(self.txtHintInfo1, i18n.Get(self.tbSetting.szHintInfo1))
  self.pPanel:Label_SetText(self.txtHintInfo2, i18n.Get(self.tbSetting.szHintInfo2))
  self:UpdateActivityScrollView()
  self:UpdateAwardScrollView()
  self:UpdateRankAwardScrollView()
end

function tbWnd:UpdateActivityScrollView()
  self.tbActivityScrollView:Clear()
  for nIndex, tbInfo in ipairs(self.tbDailyActivityList) do
    self.tbActivityScrollView:SetElement(nIndex, function(tbElement)
      local tbActivityInfo = Calendar.tbCalendarSetting[tbInfo.nActivityId]
      tbElement.pPanel:Label_SetText(self.tbActivityTemplateElement.txtName, i18n.Parse(tbActivityInfo.szName))
      Ui:Button_BindEvent(tbElement.pPanel, self.tbActivityTemplateElement.btnGo, function(tbInfo)
        Ui:OpenWindow("UIActivity", Calendar:GetActTabType(tbInfo.nActivityId), tbInfo.nActivityId)
      end, tbInfo)
    end)
  end
end

function tbWnd:UpdateAwardScrollView()
  self.tbAwardScrollView:Clear()
  local tbItems = Award:GetAward(self.tbSetting.tbShowBoxInfo.nFixAwardId, self.tbSetting.tbShowBoxInfo.nFixAwardKey, {bNoRandom = true})
  for nIndex, tbItemInfo in ipairs(tbItems) do
    self.tbAwardScrollView:SetElement(nIndex, function(tbElement)
      tbElement.itemGrid = Ui:NewExistItemGrid(tbElement, self.tbAwardTemplateElement.itemGrid)
      tbElement.itemGrid:SetGenericItem(tbItemInfo)
      tbElement.itemGrid:SetClickEnable(true)
      tbElement.pPanel:Label_SetText(self.tbAwardTemplateElement.txtName, i18n.Parse(self:GetItemName(tbItemInfo)))
    end)
  end
end

function tbWnd:UpdateRankAwardScrollView()
  self.tbRankAwardScrollView:Clear()
  local tbItems = Award:GetAward(self.tbSetting.tbShowBoxInfo.nRandAwardId, self.tbSetting.tbShowBoxInfo.nRandAwardKey, {bNoRandom = true})
  for nIndex, tbItemInfo in ipairs(tbItems) do
    self.tbRankAwardScrollView:SetElement(nIndex, function(tbElement)
      tbElement.itemGrid = Ui:NewExistItemGrid(tbElement, self.tbAwardTemplateElement.itemGrid)
      tbElement.itemGrid:SetGenericItem(tbItemInfo)
      tbElement.itemGrid:SetClickEnable(true)
      tbElement.pPanel:Label_SetText(self.tbAwardTemplateElement.txtName, i18n.Parse(self:GetItemName(tbItemInfo)))
    end)
  end
end

function tbWnd:OnClose()
  self.tbActivityScrollView:Clear()
end

function tbWnd:GetItemName(tbItemInfo)
  local szItemType = tbItemInfo[1]
  local szName = ""
  if Player.AwardType[szItemType] == Player.award_type_item then
    szName = Item:GetItemBaseProp(tbItemInfo[2]).szName
  else
    szName = Currency:GetName(szItemType)
  end
  return szName
end
