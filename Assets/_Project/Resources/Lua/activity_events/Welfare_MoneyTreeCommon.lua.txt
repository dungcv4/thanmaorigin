Require("CommonScript/Welfare/MoneyTreeCfg.lua")
MoneyTree.GROUP_KEY = GetGroupKey("MoneyTree")
MoneyTree.TEN_COUNT = 10
MoneyTree.tbKeyId = {
  FreeTimesValue = 1,
  FreeTimesLastTime = 2,
  SecondConfirmValue = 3,
  SecondConfirmLastTime = 4
}
MoneyTree.C2SProtocol = {
  ShakeOnce = 1,
  ShakeTen = 2,
  SetTodaySecondConfirm = 3
}
MoneyTree.LOCKCMD = {
  [MoneyTree.C2SProtocol.ShakeOnce] = true,
  [MoneyTree.C2SProtocol.ShakeTen] = true
}
MoneyTree.RECORD_MAX_COUNT = 100

function MoneyTree:CheckCanShakeSeveralTimes(nTimes, bCanCostFreeTimes)
  local nNotFreeTimes = nTimes
  if bCanCostFreeTimes then
    nNotFreeTimes = nNotFreeTimes - DegreeCtrl:GetDegree(me, "MoneyTreeFreeTimes")
  end
  if nNotFreeTimes <= 0 then
    return true
  else
    return me.GetMoney("YuanBao") >= self.Cfg.nCostYuanBaoCountOnce * nNotFreeTimes
  end
end

function MoneyTree:CheckCanShakeOnce()
  return self:CheckCanShakeSeveralTimes(1, true)
end

function MoneyTree:CheckCanShakeTen()
  return self:CheckCanShakeSeveralTimes(self.TEN_COUNT, false)
end

function MoneyTree:HaveFreeTimes()
  return DegreeCtrl:GetDegree(me, "MoneyTreeFreeTimes") > 0
end

function MoneyTree:GetCoinCountFromRandomIndex(nRandomIndex)
  return self.Cfg.nGetCoinBaseCountOnce * self.tbProbabilityCfg[nRandomIndex].nMultiple * (1 + Vip:GetVipAddMoneyTreePercent(me.GetVipLevel()))
end

function MoneyTree:CheckIsOpen(pPlayer)
  return pPlayer.nLevel >= self.Cfg.nOpenLevel
end
