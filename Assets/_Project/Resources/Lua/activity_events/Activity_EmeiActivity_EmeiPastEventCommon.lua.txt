Require("CommonScript/Activity/ActivityMgrDef.lua")
Require("CommonScript/Activity/EmeiActivity/EmeiPastEvent/EmeiPastEventCfg.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.EmeiPastEvent)
local EmeiPastEventCfg = ActivityCfg.EmeiPastEventCfg
tbActivity.nSaveGroup = GetGroupKey("EmeiPastEvent")
tbActivity.tbGroupId = {HaveGetExtraAward = 1}
tbActivity.C2SProtocol = {GetExtraAward = 1}
tbActivity.S2CProtocol = {SyncAward = 1}

function tbActivity:LoadConfig()
  if not self.bLoadConfig then
    local szFilePath = "Setting/Activity/EmeiActivity/EmeiPastEvent/EmeiPastEvent.tab"
    local tbField = {
      "nConfigId",
      "szName",
      "szShowTime",
      "szDesc"
    }
    self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "dsss", "nConfigId", tbField, nil, nil, false)
    local tbSingleConfig = self.ACTIVITY_CONFIG[EmeiPastEventCfg.ActivityCfg.nConfigId]
    tbSingleConfig.nLevel = EmeiPastEventCfg.ActivityCfg.nLevel
    tbSingleConfig.nShowAwardId = EmeiPastEventCfg.ActivityCfg.nAwardId
    tbSingleConfig.nShowAwardKey = EmeiPastEventCfg.ActivityCfg.nAwardKey
    tbSingleConfig = SetReadOnly(tbSingleConfig)
    self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
    self.bLoadConfig = true
  end
end

function tbActivity:HaveGetExtraAwardTimesAndFinishTask(pPlayer)
  return not self:HaveGetAward(pPlayer) and self:HaveFinishEmeiTask(pPlayer)
end

function tbActivity:HaveFinishEmeiTask(pPlayer)
  return Task:IsTaskFinished(pPlayer, EmeiPastEventCfg.nEmeiSubTaskId)
end

function tbActivity:HaveGetAward(pPlayer)
  return 0 ~= pPlayer.GetUserValue(self.nSaveGroup, self.tbGroupId.HaveGetExtraAward)
end
