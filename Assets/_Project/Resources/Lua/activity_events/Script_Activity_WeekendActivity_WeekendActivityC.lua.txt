local tbActivity = ActivityMgr:GetClass(ActivityMgr.szWeekendActivityClassName)
tbActivity.tbWeekendCfg = {}

function tbActivity:OnOpen()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_CROSS_DAYS, self.CrossDay, self)
  self:TryLightRedPoint()
end

function tbActivity:TryLightRedPoint()
  if not RedPointMgr:CheckEvent(RedPointMgr.EVENT.WeekendActivity) and self:WeekendActivityIsOpen() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.WeekendActivity)
  end
end

function tbActivity:TryFadeRedPoint()
  if RedPointMgr:CheckEvent(RedPointMgr.EVENT.WeekendActivity) and self:WeekendActivityIsOpen() then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.WeekendActivity)
  end
end

function tbActivity:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CROSS_DAYS, self)
end

function tbActivity:CrossDay()
  if not self:WeekendActivityIsOpen() then
    self:TryFadeRedPoint()
  end
  self:TryLightRedPoint()
end
