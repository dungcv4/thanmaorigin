Require("CommonScript/GlobalTable.lua")
Require("CommonScript/Event/EventDefine.lua")

function Event:RegisterEvent(nEventId, tbCallBackTable, szFnName, param)
  assert(tbCallBackTable)
  if not tbCallBackTable[szFnName] then
    LogErr("Event:RegisterEvent Error, CallBack not exist", nEventId, szFnName, debug.traceback())
    return
  end
  self.tbEvent[nEventId] = self.tbEvent[nEventId] or {}
  if self.tbEvent[nEventId][tbCallBackTable] then
    LogErr("Event:RegisterEvent repeat: ", nEventId, debug.traceback())
  end
  self.tbEvent[nEventId][tbCallBackTable] = {
    tbCallBackTable,
    szFnName,
    param
  }
end

function Event:UnRegisterEvent(nEventId, tbCallBackTable)
  if not (nEventId and tbCallBackTable) or not self.tbEvent[nEventId] then
    LogErr("Event:UnRegisterEvent error: ", nEventId, debug.traceback())
    return
  end
  self.tbEvent[nEventId][tbCallBackTable] = nil
  if not next(self.tbEvent[nEventId]) then
    self.tbEvent[nEventId] = nil
  end
end

function Event:HasEvent(nEventId, tbCallBackTable)
  return self.tbEvent[nEventId] and self.tbEvent[nEventId][tbCallBackTable]
end

function Event:OnEvent(nEventId, ...)
  local tbEventCalls = self.tbEvent[nEventId]
  local fnFunc, tbTable, szFnName
  for _, tbEventCall in pairs(tbEventCalls or {}) do
    tbTable = tbEventCall[1]
    szFnName = tbEventCall[2]
    fnFunc = tbTable[szFnName]
    if tbEventCall[3] then
      Lib:CallBack({
        fnFunc,
        tbTable,
        tbEventCall[3],
        ...
      })
    else
      Lib:CallBack({
        fnFunc,
        tbTable,
        ...
      })
    end
  end
end

function Event:Init()
  self.tbEvent = self.tbEvent or {}
end

Event:Init()

end

end
