Require("CommonScript/Activity/ActivityMgrDef.lua")
Require("CommonScript/Activity/FirstAnniversaryActivity/FirstAnniversaryPaintingComm.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.FirstAnniversaryPainting)
local Protocol = {
  [tbActivity.Protocol.OpenCellBack] = "OnOpenCellBack",
  [tbActivity.Protocol.GetCellGroupRewardBack] = "OnGetCellGroupRewardBack",
  [tbActivity.Protocol.GetGrandPrizeBack] = "OnGetGrandPrizeBack",
  [tbActivity.Protocol.ReqPaintingLightRedPoint] = "OnCheckRedPoint"
}

function tbActivity:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_DEL_ITEM, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self)
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
end

function tbActivity:OnOpen()
  self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, tbActivity.OnAddItem, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_DEL_ITEM, tbActivity.OnDelItem, self)
  self:OnCheckRedPoint()
end

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("[LaborHonor] ServerRequest protocol is wrong", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:ReqSawPuzzle(bIsGold, nIndex)
  ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.ReqOnOpenCell, bIsGold, nIndex)
end

function tbActivity:ReqGetGroupReward(nGroupId)
  ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.ReqGetCellGroupReward, nGroupId)
end

function tbActivity:ReqGetGrandPrize()
  ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.ReqGetGrandPrize)
end

function tbActivity:OnOpenCellBack(nIndex)
  EventNotify.OnNotify(EventNotify.emNOTIFY_PAINTING_OPENCELL, nIndex)
end

function tbActivity:OnGetCellGroupRewardBack(nGroupId)
  EventNotify.OnNotify(EventNotify.emNOTIFY_PAINTING_GETGROUP, nGroupId)
end

function tbActivity:OnGetGrandPrizeBack()
  EventNotify.OnNotify(EventNotify.emNOTIFY_PAINTING_GETALLREWARD)
end

function tbActivity:GetRedPointEventList()
  if not self.tbConfig.szRedPoint or self.tbConfig.szRedPoint == "" then
    return nil
  end
  if not self.tbConfig.szItemRedPoint or "" == self.tbConfig.szItemRedPoint then
    return nil
  end
  return {
    RedPointMgr.EVENT[self.tbConfig.szRedPoint],
    RedPointMgr.EVENT[self.tbConfig.szItemRedPoint]
  }
end

function tbActivity:OnAddItem(nItemId)
  if nItemId then
    local pItem = KItem.GetItemObj(nItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId ~= self.tbConfig.nMaterialsId then
      return
    end
    self:OnCheckRedPoint()
  end
end

function tbActivity:OnDelItem(_, dwTemplateId)
  if dwTemplateId ~= self.tbConfig.nMaterialsId then
    return
  end
  self:OnCheckRedPoint()
end

function tbActivity:OnCheckRedPoint()
  if me.nLevel < self.tbConfig.nLevel then
    return
  end
  local nCount = me.GetItemCountByTypeInBag(self.tbConfig.nMaterialsId)
  if nCount > 0 then
    if not RedPointMgr:CheckEvent(RedPointMgr.EVENT[self.tbConfig.szItemRedPoint]) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT[self.tbConfig.szItemRedPoint])
    end
  elseif RedPointMgr:CheckEvent(RedPointMgr.EVENT[self.tbConfig.szItemRedPoint]) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[self.tbConfig.szItemRedPoint])
  end
  local nGrandPrize = self:GetGrandPrize(me)
  local nOpenCellNum = self:GetOpenCellNum(me)
  if 0 == nGrandPrize and nOpenCellNum >= self.tbConfig.nCellNum then
    if not RedPointMgr:CheckEvent(RedPointMgr.EVENT[self.tbConfig.szRedPoint]) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
    end
    return
  end
  local nGroupNum = #self.tbConfig.tbCellGroup
  local bShowRed = false
  for i = 1, nGroupNum do
    local bRet = self:IsGetCellGroupReward(me, i)
    local nGroupRewardData = self:GetCellGroupReward(me)
    local bIsGet = Lib:GetBit(nGroupRewardData, i)
    if bRet and not bIsGet then
      bShowRed = true
      break
    end
  end
  if bShowRed then
    if not RedPointMgr:CheckEvent(RedPointMgr.EVENT[self.tbConfig.szRedPoint]) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
    end
    return
  end
  if 0 == nGrandPrize and nCount > 0 then
    if not RedPointMgr:CheckEvent(RedPointMgr.EVENT[self.tbConfig.szRedPoint]) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
    end
  elseif RedPointMgr:CheckEvent(RedPointMgr.EVENT[self.tbConfig.szRedPoint]) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
  end
end

function tbActivity:DoCommonJump(nJumpId)
  Ui:CloseWindow("UIBag")
  Ui:CloseWindow("UIItemTips")
  Ui:OpenWindow("UIActivity_AnniversaryDraw", self.nId)
end
