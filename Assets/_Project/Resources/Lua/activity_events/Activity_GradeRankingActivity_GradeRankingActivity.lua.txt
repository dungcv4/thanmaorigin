Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Event/Event.lua")
ActivityMgr.szGradeRankingActivityClassName = "GradeRankingActivity"
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szGradeRankingActivityClassName)
tbActivity.tbTopList = {}
tbActivity.nConfigId = 1

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szGlobalFilePath = "Setting/Activity/GradeRankingActivity/GlobalSettings.tab"
  local szAwardFilePath = "Setting/Activity/GradeRankingActivity/GradeRankingActivity.tab"
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile(szGlobalFilePath, {"nConfigId"})
  self.AWARD_CONFIG = Lib:EasyLoadTabFile(szAwardFilePath, {"nAwardType"})
  for _, tbCfg in pairs(self.ACTIVITY_CONFIG) do
    if Lib:IsEmptyStr(tbCfg.szScriptDataKey) then
      tbCfg.szScriptDataKey = "GradeRankingActivityTopList"
    end
    if Lib:IsEmptyStr(tbCfg.szLogWay) then
      tbCfg.szLogWay = "LogWay_GradeRankingActivity"
    end
  end
  self.bLoadConfig = true
end

function tbActivity:OnModify(ActivityConfig)
  self.nId = ActivityConfig.nId
  self.nStartTime = ActivityConfig.nStartTime
  self.nEndTime = ActivityConfig.nEndTime
  self.tbGlobalConfig = self.ACTIVITY_CONFIG[ActivityConfig.nConfigId]
  self:InitMetaData(ActivityConfig)
end

function tbActivity:OnLogin(pPlayer)
  if not self:IsOpen() then
    return
  end
  self:SyncData(pPlayer)
  self:SynchronousDataToSingle()
end

function tbActivity:SyncData(pPlayer)
  pPlayer.CallClientScript("ActivityMgr:OnSyncActivityData", {
    {
      self:GetScheduleCfg()
    }
  })
end

function tbActivity:CheckConfig(nConfigId)
  self:LoadConfig()
  if not self.ACTIVITY_CONFIG[nConfigId] then
    LogErr(string.format("[GradeRankingActivityServer] CheckConfig fail. nConfigId =%d not found", nConfigId))
    return false
  end
  return true
end

function tbActivity:getAwardType(nRank)
  for i = 1, self.tbGlobalConfig.nAwardTypeCount do
    if nRank <= self.AWARD_CONFIG[i].nRankLowerBound then
      return i
    end
  end
  return -1
end

function tbActivity:IsServerOnly()
  return false
end

function ActivityMgr:GetGradeRankingActivityInstance()
  for _, pInstance in pairs(ActivityMgr.tbInstance) do
    if pInstance.szClassName == ActivityMgr.szGradeRankingActivityClassName and pInstance:IsOpen() then
      return pInstance
    end
  end
end

function tbActivity:IsOpenAndShow()
  local nfinishTime = self.nFinishTime
  if not nfinishTime then
    return true
  end
  return nfinishTime > GetTime()
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end

function tbActivity:GetScriptDataFieldName()
  return "szScriptDataKey"
end
