function CheapMonthGift:CheckIsOpen()
  return WeekMonthCard:CheckIsOpen() and DailyGift:CheckIsOpen()
end

function CheapMonthGift:OnChargeReq()
  local tbSetting = Recharge:GetCheapMonthGiftSetting()
  if not tbSetting then
    me.CenterMsg(i18n.EasyGet("\229\133\133\229\128\188\233\133\141\231\189\174\233\148\153\232\175\175"))
    return
  end
  Recharge:RequestCreateRechargeOrder(tbSetting.nProductId)
end

function CheapMonthGift:UpdateUI()
  self:UpdateReddotState()
  if Ui:WindowVisible("UIWelfare") == true then
    Ui:GetWindow("UIWelfare"):UpdateCheapMonthGiftPanel()
  end
end

function CheapMonthGift:UpdateReddotState()
  local bMonthGift = false
  if not self.bHaveClick and bMonthGift then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.CheapMonthGift)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.CheapMonthGift)
  end
end

function CheapMonthGift:OnLogin()
  self:UpdateReddotState()
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_CROSS_DAYS, CheapMonthGift.UpdateReddotState, CheapMonthGift)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_LEVEL, CheapMonthGift)
EventNotify:RegistNotify(EventNotify.emNOTIFY_CHANGE_PLAYER_LEVEL, CheapMonthGift.UpdateReddotState, CheapMonthGift)
