Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.LaborHonor)
tbActivity.SaveID = {
  LastStartTimeKey = 1,
  SubNumID = 2,
  EventIDList = {
    YiJunCompleted = 3,
    FuNiuMountain = 4,
    YOYOGameCompleted = 5,
    BattleCompleted = 6,
    BaiHuTangJoin = 7,
    JoinShiYueDao = 8,
    KinChallengeJoin = 9,
    WuLinGaoShouJoin = 10,
    PrimaryDailyGiftBuy = 11,
    MiddleDailyGiftBuy = 12,
    AdvancedDailyGiftBuy = 13
  }
}
tbActivity.Protocol = {
  Submit = 1,
  SubmitResp = 2,
  OnDialog = 3,
  SyncData = 4
}

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Activity/XiongWangActivity/LaborHonor/LaborHonor.tab"
  local tbField = {
    "nConfigId",
    "nJoinLevel",
    "szLogWay",
    "szDialogText",
    "nMaxSubmission",
    "nTemplateId",
    "nAwardId",
    "nAwardKey",
    "nGiftBoxBaseNum",
    "nGiftBoxAwardId",
    "nGiftBoxKey",
    "szSaveKey",
    "szLevel",
    "nNpcTemplateId",
    "nMapTemplateId",
    "szAucReadyMFrom",
    "szAucReadyMTitle",
    "szAucReadyMContent",
    "szSubmissionMFrom",
    "szSubmissionMTitle",
    "szSubmissionMContent",
    "szShowTime",
    "szName",
    "nSortValue",
    "szTimeOut",
    "szDesc",
    "nShowAwardId",
    "nShowAwardKey",
    "nAuctionAwardId"
  }
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "ddssdddddddssddssssssssdssddd", "nConfigId", tbField)
  for _, tbInfo in pairs(self.ACTIVITY_CONFIG) do
    tbInfo.tbLevel = Lib:SplitStr2Num(tbInfo.szLevel, "|") or {}
    tbInfo.nTimeOut = not Lib:IsEmptyStr(tbInfo.szTimeOut) and Lib:ParseDateTime(tbInfo.szTimeOut) or 0
  end
  szFilePath = "Setting/Activity/XiongWangActivity/LaborHonor/EventAward.tab"
  tbField = {
    "nConfigId",
    "szEventName",
    "nAwardId",
    "nAwardKey"
  }
  local tbEventAward = Lib:LoadTabFileEx(szFilePath, "dsdd", nil, tbField)
  for _, tbInfo in ipairs(tbEventAward or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("LaborHonor:LoadConfig EventAward nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEventAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEventAward or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbEventAward[tbInfo.szEventName] = tbInfo
  end
end

function tbActivity:CanJoin(nLevel)
  return nLevel >= self.tbConfig.nJoinLevel
end

function tbActivity:HasTong(pPlayer)
  local dwTongId = pPlayer.dwTongId
  return dwTongId and dwTongId > 0
end

function tbActivity:HasSubmissionItem(pPlayer)
  local tbItem = pPlayer.FindItem(self.tbConfig.nTemplateId)
  return tbItem and #tbItem > 0
end

function tbActivity:GenNeedSubmissionNum(nLevelIndex)
  if nLevelIndex <= 0 or nLevelIndex > #self.tbConfig.tbLevel then
    return 0
  end
  if not self.tbConfig.tbLevel[nLevelIndex - 1] then
    return self.tbConfig.tbLevel[nLevelIndex]
  end
  return self.tbConfig.tbLevel[nLevelIndex] - self.tbConfig.tbLevel[nLevelIndex - 1]
end
