Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.AddDecoration)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/Common/AddDecoration/AddDecorationCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/Common/AddDecoration/DecorationPosCfg.tab"
  local tbPosFileConfig = Lib:EasyLoadTabFile(szFilepath, nil)
  if not tbPosFileConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  for _, tbInfo in pairs(tbPosFileConfig) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("AddDecoration: Load tbPosFileConfig failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDecorationPos = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDecorationPos or {}
    if not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDecorationPos[tbInfo.nMapId] then
      self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDecorationPos[tbInfo.nMapId] = tbInfo.szPosFile
    else
      Lib:LogErrFormat("AddDecoration: Load tbPosFileConfig failed, duplicate nMapId = %d in same ConfigId, nConfigId = %d", tbInfo.nMapId, tbInfo.nConfigId)
      return false
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

function tbActivity:GetNpcPosInfo()
  local tbPosInfo = {}
  for nMapId, szPosFile in pairs(self.tbConfig.tbDecorationPos or {}) do
    local tbInfo = Lib:LoadTabFileEx(szPosFile, "ddd", nil, {
      "NpcTemplateId",
      "PosX",
      "PosY"
    })
    if not tbInfo then
      Lib:LogErrFormat("AddDecoration: GetNpcPosInfo failed, PosFile %s not exist, nMapId = %d", szPosFile, nMapId)
    else
      tbPosInfo[nMapId] = {}
      for _, tbNpcPos in pairs(tbInfo) do
        table.insert(tbPosInfo[nMapId], tbNpcPos)
      end
    end
  end
  return tbPosInfo
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = {}
  self.tbConfig.tbDecorationPos = {}
  for _, tbInfo in pairs(tbActCfg.DecorationPosCfg) do
    self.tbConfig.tbDecorationPos[tbInfo.nMapId] = tbInfo.szPosFile
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  return true
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end

function tbActivity:GetLogWayFieldName()
  return ActivityMgr.NoLogWay
end
