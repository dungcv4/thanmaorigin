Require("CommonScript/Activity/XiongWangActivity/JingWangJiuXiActivityCommon.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.TaoYuanJieYi)
tbActivity.LEADER_NPC_MAPID = {
  [1] = 31
}
tbActivity.LEADER_NPCID = 11042

function tbActivity:ServerRequest(szFuncName, ...)
  if not self[szFuncName] then
    return
  end
  self[szFuncName](self, ...)
end

function tbActivity:InitExtraData()
end

function tbActivity:S2CRefreshData()
  self:TryLightRedPoint()
end

function tbActivity:GetJieYiBoxAward()
  ActivityMgr:CallServerScriptV2(self.nId, tbActivity.Protocol.GetJieYiBoxAward)
end

function tbActivity:OnOpen()
  self:TryLightRedPoint()
end

function tbActivity:GetRedPointEventId()
  return {
    RedPointMgr.EVENT.TaoYuanJieYi,
    RedPointMgr.EVENT.TaoYuanJieYiFirstOpen
  }
end

function tbActivity:TryLightRedPoint()
  if self:IsJieYi(me) and not self:HasGetAwardBox(me) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.TaoYuanJieYi)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.TaoYuanJieYi)
  end
  if 0 == Ui.PlayerPrefs.GetInt("TaoYuanJieYiFirst", 0) and self:IsOpen() and self:PlayerLevelEnough(me) and not self:HasGetAwardBox(me) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.TaoYuanJieYiFirstOpen)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.TaoYuanJieYiFirstOpen)
  end
end

function tbActivity:GotoJieYiNPC()
  local nRandomNPCIndex = -1
  for nIndex, nMapID in ipairs(self.LEADER_NPC_MAPID) do
    if me.nMapId == nMapID then
      nRandomNPCIndex = nIndex
      break
    end
  end
  if -1 == nRandomNPCIndex then
    nRandomNPCIndex = math.random(1, #self.LEADER_NPC_MAPID)
  end
  local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(self.LEADER_NPCID, self.LEADER_NPC_MAPID[nRandomNPCIndex], false)
  
  local function fnCallback()
    local function __timer_Callback()
      local nNpcId = KNpc.GetNpcIdByTemplateId(self.LEADER_NPCID)
      
      if nNpcId then
        Operation:SimpleTap(nNpcId)
      end
    end
    
    Timer:Register(5, __timer_Callback)
  end
  
  AutoPath:GotoAndCallEx(self.LEADER_NPC_MAPID[nRandomNPCIndex], nPosX, nPosY, fnCallback, nWalkNearLength, nil, self.LEADER_NPCID)
end

function tbActivity:ClickGoOrGetAwardBtn()
  if not self:PlayerLevelEnough(me) then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179"))
    return
  end
  if 0 == Ui.PlayerPrefs.GetInt("TaoYuanJieYiFirst", 0) then
    Ui.PlayerPrefs.SetInt("TaoYuanJieYiFirst", 1)
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.TaoYuanJieYiFirstOpen)
  end
  if not self:IsJieYi(me) then
    self:GotoJieYiNPC()
    if Ui:WindowVisible("UIEmeiActivity") then
      Ui:CloseWindow("UIEmeiActivity")
    end
  elseif self:IsJieYi(me) and not self:HasGetAwardBox(me) then
    self:GetJieYiBoxAward()
  else
    me.CenterMsg(i18n.EasyGet("\230\130\168\229\183\178\231\187\143\233\162\134\229\143\150\232\191\135\229\174\157\231\174\177\228\186\134"))
  end
end
