FestivalActivity.nOffSet = 14400

function FestivalActivity:GotoOutGameWeb(szActivityID)
  if not szActivityID or not self.tbActivityList[szActivityID] then
    return
  end
  RemoteServer.RequestFestivalActivity(self.Protocol.OpenWeb, szActivityID)
end

function FestivalActivity:OpenWebC(szActivityID, szUrl)
  local szKey = self:GetRedPointKey(szActivityID)
  local nNow = GetTime()
  Ui.PlayerPrefs.SetInt(szKey, nNow)
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[self.tbActivityList[szActivityID].szRedPoint])
  Log("FestivalActivity:OpenWebC update", szKey, nNow)
  if WINDOWS then
    Ui:GotoWeb(szUrl)
    return
  end
  Ui:OpenWindow("UIGeneralWebView", szUrl, 0, 0)
end

function FestivalActivity:CheckLightingRedPointHelper()
  for szActivityID, tbInfo in pairs(self.tbActivityList) do
    if self:IsActivityOpening(szActivityID) and self:CanLightRedPoint(szActivityID, tbInfo) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT[tbInfo.szRedPoint])
      Log("FestivalActivity:CheckLightingRedPointHelper light", szActivityID)
    end
  end
end

function FestivalActivity:OnLogin()
  Log("FestivalActivity:OnLogin", me.dwID)
  self:CheckLightingRedPointHelper()
end

function FestivalActivity:GetRedPointKey(szActivityID)
  return table.concat({
    "FestivalActivity_",
    szActivityID,
    "_",
    me.dwID or 0
  })
end

function FestivalActivity:CanLightRedPoint(szActivityID, tbInfo)
  local szKey = self:GetRedPointKey(szActivityID)
  local nTime = Ui.PlayerPrefs.GetInt(szKey, 0)
  local nNow = GetTime()
  return not Lib:IsSameDay(self.nOffSet, nTime, nNow)
end

function FestivalActivity:OnPlayerFirstLoginToday()
  Log("FestivalActivity:OnPlayerFirstLoginToday", me.dwID)
  self:CheckLightingRedPointHelper()
end

function FestivalActivity:Init()
  if self.bHasInitialized then
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self)
  end
  EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self.OnPlayerFirstLoginToday, self)
  self.bHasInitialized = true
end

FestivalActivity:Init()
