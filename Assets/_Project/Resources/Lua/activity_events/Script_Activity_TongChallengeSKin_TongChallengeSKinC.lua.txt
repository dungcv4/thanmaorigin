local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.TongChallengeSkin)
local TongChallengeMapTid = 65

function tbActivity:OnOpen()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_TONGCHALLENGE_READY, self.LightRedPoint, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_TONGCHALLENGE_END, self.OnTongChallengeClose, self)
end

function tbActivity:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_TONGCHALLENGE_READY, self)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_TONGCHALLENGE_END, self)
end

function tbActivity:LightRedPoint()
  local nEventId = RedPointMgr.EVENT[self.tbConfig.szRedPointEvent]
  if not nEventId then
    return
  end
  if me.nMapTemplateId == TongChallengeMapTid then
    return
  end
  if not RedPointMgr:CheckEvent(nEventId) then
    RedPointMgr:LightRedPoint(nEventId)
  end
end

function tbActivity:FadeRedPoint()
  local nEventId = RedPointMgr.EVENT[self.tbConfig.szRedPointEvent]
  if not nEventId then
    return
  end
  if RedPointMgr:CheckEvent(nEventId) then
    RedPointMgr:FadeRedPoint(nEventId)
  end
end

function tbActivity:JoinTongChallenge()
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
  TongChallenge:OnJoinTongChallange()
  self:FadeRedPoint()
end

function tbActivity:OnTongChallengeClose()
  self:FadeRedPoint()
end
