local tbControls = {
  Img_Grid = "node/imgBG/Img_Grid",
  txtTitle = "node/imgBG/txtTitle",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  btnClose = "node/imgBG/btnClose",
  reward = "node/imgBG/ItemCell/reward",
  UIItemGrid = "node/imgBG/Template/UIItemGrid",
  txtCost = "node/imgBG/Template/txtCost",
  txtName = "node/imgBG/Template/txtName",
  btnAdd = "node/imgBG/Template/btnAdd",
  btnGo = "node/imgBG/btnGo",
  Cont_All = "node/imgBG/Cont_All",
  Hud_Effect = "node/imgBG/Cont_All/Hud_Effect",
  UIMoneyPanel = "node/imgBG/UIMoneyPanel",
  Txt_Info = "node/imgBG/Txt_Info",
  Img_FormalBg = "node/imgBG/Img_FormalBg",
  ImageCell = "node/imgBG/Img_Grid/Img_Cont%s/Image"
}
tbWnd.tbControls = tbControls

function tbWnd:UpdateListElement(tbElement, tbAward)
  local tbGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  if "Item" == tbAward[1] then
    tbGrid:SetItemByTemplate(tbAward[2], tbAward[3], nil, nil, {bShowTip = true})
  else
    tbGrid:SetDigitalItem(tbAward[1], tbAward[3])
  end
  tbGrid:SetClickEnable(true)
end
