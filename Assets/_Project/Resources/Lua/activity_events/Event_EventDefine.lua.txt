if EDITOR then
  local record = {}
  local mt = {}
  setmetatable(Event, {
    __index = mt,
    __newindex = function(t, k, v)
      if rawget(mt, k) ~= nil then
        error(string.format("Event.%s Redefine ", k))
      end
      if type(v) == "number" then
        if record[v] then
          error(string.format("Event.%s == Event.%s value is %d", k, record[v], v))
        else
          record[v] = k
        end
      end
      rawset(mt, k, v)
    end
  })
end
