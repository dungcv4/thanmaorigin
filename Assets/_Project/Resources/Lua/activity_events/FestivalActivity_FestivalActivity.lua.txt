FestivalActivity.Protocol = {OpenWeb = 1}

function FestivalActivity:LoadConfig()
  local szFilePath = "Setting/FestivalActivity/FestivalActivity.tab"
  self.tbActivityList = Lib:EasyLoadTabFile(szFilePath, {
    "szActivityID"
  }, nil, nil, true)
  szFilePath = "Setting/FestivalActivity/FestivalActivityAward.tab"
  self.tbAwardList = Lib:EasyLoadTabFile(szFilePath, {
    "szActivityID",
    "nAwardCallId"
  }, nil, nil, true)
end

FestivalActivity:LoadConfig()

function FestivalActivity:IsActivityOpening(szActivityID)
  if not szActivityID or not self.tbActivityList[szActivityID] then
    return false
  end
  local nNow = GetTime()
  if not Lib:IsEmptyStr(self.tbActivityList[szActivityID].szStartTime) then
    local nStartTime = Lib:ParseDateTime(self.tbActivityList[szActivityID].szStartTime)
    if nNow < nStartTime then
      return false
    end
  end
  if not Lib:IsEmptyStr(self.tbActivityList[szActivityID].szEndTime) then
    local nEndTime = Lib:ParseDateTime(self.tbActivityList[szActivityID].szEndTime)
    if nNow >= nEndTime then
      return false
    end
  end
  return true
end
