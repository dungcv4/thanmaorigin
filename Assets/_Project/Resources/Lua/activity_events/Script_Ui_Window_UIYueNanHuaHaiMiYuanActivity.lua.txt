tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnShop = function(self)
    local nRedPointEventId = self.pMiYuanActInstance:GetRedPointEventList()[1]
    if RedPointMgr:CheckEvent(nRedPointEventId) then
      RedPointMgr:FadeRedPoint(nRedPointEventId)
      LocalData:SetRoleBoolean(self.pMiYuanActInstance:GetRedPointStatusKey(), false)
    end
    Ui:OpenWindow("UINpcShop", self.nNpcShopId, nil, nil, nil, nil, Currency.TYPE.XunYuan)
  end,
  btnCostReward = function(self)
    Ui:OpenWindow("UIYueNanWomenDayReward", self.nAchieveActId)
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("LabaGiftHelp")
  end,
  btnXuniYuan1 = function(self)
    self.pMiYuanActInstance:ReqFindYuan(self.pMiYuanActInstance.FindTypeIndex.OnceFind)
  end,
  btnXuniYuan10 = function(self)
    self.pMiYuanActInstance:ReqFindYuan(self.pMiYuanActInstance.FindTypeIndex.BatchFind)
  end,
  btnClose = function(self)
    self:Close()
  end
}
