local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.GuMuMiQing)
local Protocol = {
  [tbActivity.Protocol.SyncActiveStatusResp] = "OnSyncActiveStatus"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("AchieveAwardC: ServerRequest failed, invalid Protocol = ", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnClose()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.GuMuMiQingTreeAppear)
end

function tbActivity:OnSyncActiveStatus(bIsActive)
  self.bIsActive = bIsActive
  if self.bIsActive then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.GuMuMiQingTreeAppear)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.GuMuMiQingTreeAppear)
  end
end

function tbActivity:IsInActiveTime()
  return self.bIsActive or false
end

function tbActivity:GetMapList()
  if self.tbConfig.tbTreeNumInfo ~= nil then
    local tbMapList = {}
    for nMapId, _ in pairs(self.tbConfig.tbTreeNumInfo) do
      table.insert(tbMapList, nMapId)
    end
    return tbMapList
  end
  return {}
end

function tbActivity:GotoRandomPos()
  if self:IsOpen() then
    ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.ReqGoToTree)
  else
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\176\154\230\156\170\229\188\128\229\144\175"))
  end
end
