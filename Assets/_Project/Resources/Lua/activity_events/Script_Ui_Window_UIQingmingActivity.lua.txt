local tbWnd = Ui:CreateClass("UIQingmingActivity", "UIFestivalBase")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  ItemTemplate = "ImgAll/ContList/Viewport/Content/ItemTemplate",
  txtTitle = "node/imgBG/Content/Main/Tex/txtTitle",
  txtTime = "node/imgBG/Content/Main/Tex/txtTime01",
  txtMain = "node/imgBG/Content/btnMain/Text"
}
tbWnd.tbControls = tbControls
tbWnd.tbSelectedColor = Ui.tbColor:GetUnityColorByDefineName("QingmingActivityOutline")
tbWnd.tbUnSelectedColor = Ui.tbColor:GetUnityColorByDefineName("ActivityDefaultOutline")

function tbWnd:OnCreate()
  self.nMainCommonEntryID = 286
  self.tbBtnPath = {}
  self.tbBtnTextPath = {}
  self.tbBtnEffectPath = {}
  self.tbBtnTextList = {}
  local szTemp = "node/imgBG/Content/btnList/Viewport/Content/State_UnOpened%d"
  for i = 1, self.nMaxButton do
    self.tbBtnPath[i] = string.format(szTemp .. "/btnTemplate", i)
    self.tbBtnTextPath[i] = string.format(szTemp .. "/Text", i)
    self.tbBtnEffectPath[i] = string.format(szTemp .. "/btnPitchOn", i)
  end
  self.tbBtnPath[self.nMainCommonEntryID] = "node/imgBG/Content/btnMain"
  self.tbBtnTextPath[self.nMainCommonEntryID] = "node/imgBG/Content/btnMain/Text"
  self.tbBtnEffectPath[self.nMainCommonEntryID] = "node/imgBG/Content/btnMain/BtnPitchOn"
  self.tbCommonEntryID2SubPanel = {
    [282] = "node/imgBG/Content/Kite",
    [283] = "node/imgBG/Content/VisitGrave",
    [284] = "node/imgBG/Content/Gift",
    [285] = "node/imgBG/Content/Ancestors",
    [286] = "node/imgBG/Content/Main"
  }
  self.tbCommonEntryID2UiName = {
    [282] = "UIQingmingKite",
    [283] = "UIQingmingVisitGrave",
    [284] = "UIQingmingGift",
    [285] = "UIQingmingAncestors",
    [286] = "UIQingmingActivityMain"
  }
  self.nButtonNum = 5
  for i = 1, self.nButtonNum - 1 do
    self.tbBtnTextList[i] = self.pPanel:GetObject(self.tbBtnTextPath[i])
  end
  self.tbBtnTextList[self.nButtonNum] = self.pPanel:GetObject(tbControls.txtMain)
end

function tbWnd:SetMainOutline()
  for i = 1, self.nButtonNum do
    local obj = self.tbBtnTextList[i]
    if obj then
      local comp = obj:GetComponent("Outline")
      if self.nCurLightIndex == i then
        comp.effectColor = self.tbSelectedColor
      else
        comp.effectColor = self.tbUnSelectedColor
      end
    end
  end
end
