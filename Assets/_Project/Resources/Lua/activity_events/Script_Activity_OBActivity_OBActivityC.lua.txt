ActivityMgr.OBActivity = ActivityMgr.OBActivity or {}
local obActivity = ActivityMgr.OBActivity
obActivity.ActivityType = {
  OBReward = "OBReward",
  OBFireWork = "OBFireWork",
  OBTreasureRabbit = "OBTreasureRabbit",
  CommonEntry = "CommonEntry"
}

function obActivity:GetOBRewardRewardPreview()
  for nId in pairs(ActivityMgr:GetIdListByClassName(self.ActivityType.OBReward)) do
    local pInstance = ActivityMgr:GetInstanceById(nId)
    if pInstance then
      return pInstance:GetPreviewReward()
    end
    pInstance = ActivityMgr:Create(self.ActivityType.OBReward)
    if pInstance then
      pInstance:OnInit(ActivityMgr:GetConfigById(nId))
      return pInstance:GetPreviewReward()
    end
  end
  return {}
end

function obActivity:BuildInstanceConfig(nId, szType)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if pInstance then
    return pInstance:GetConfig(), true
  end
  local tmpInstance = ActivityMgr:Create(szType)
  if not tmpInstance then
    return
  end
  tmpInstance:OnInit(ActivityMgr:GetConfigById(nId))
  return tmpInstance:GetConfig(), false
end

function obActivity:GetActivityList()
  local tbList = {}
  local tbConfig, szName
  for _, szType in ipairs(ActivityMgr.OBActivityCfg.ActivityList) do
    for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
      tbConfig = self:BuildInstanceConfig(nId, szType)
      if tbConfig then
        szName = tbConfig.szName and i18n.Parse(tbConfig.szName) or ""
        table.insert(tbList, {
          szClassName = szType,
          nId = nId,
          szName = szName,
          szDesc = i18n.Parse(tbConfig.szDesc),
          nLevel = tbConfig.nLevel,
          tbAward = Award:GetAward(tbConfig.nShowAwardId or tbConfig.nAwardId, tbConfig.nShowAwardKey or tbConfig.nAwardKey),
          pFunc = tbJumpFunc[szType],
          szShowTime = i18n.Parse(tbConfig.szShowTime or ""),
          nSortValue = ActivityMgr.OBActivityCfg.SortList[nId] or 0
        })
      end
    end
  end
  table.sort(tbList, function(tbFirst, tbSecond)
    return tbFirst.nSortValue > tbSecond.nSortValue
  end)
  return tbList
end
