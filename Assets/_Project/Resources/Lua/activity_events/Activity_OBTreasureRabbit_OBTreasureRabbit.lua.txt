Require("CommonScript/Activity/ActivityMgr.lua")
ActivityMgr.szOBTreasureRabbitClassName = "OBTreasureRabbit"
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szOBTreasureRabbitClassName)
tbActivity.LastStartTimeKey = 1
tbActivity.UsedBoxNumKey = 2
tbActivity.UsedBagNumKey = 3

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/OBTreasureRabbit/OBTreasureRabbit.tab"
  local tbField = {
    "nConfigId",
    "nRatio",
    "nSmallInterval",
    "nMaxSmallNum",
    "nMaxLifeTime",
    "nBigTemplateId",
    "nSmallTemplateId",
    "nMaxBoxUsedNum",
    "nMaxBagUsedNum",
    "szSaveKey",
    "nBoxAdvancedAwardId",
    "nBoxAdvancedAwardKey",
    "nBoxBaseAwardId",
    "nBoxBaseAwardKey",
    "nBagAwardId",
    "nBagAwardKey",
    "nBagBaseAwardId",
    "nBagBaseAwardKey",
    "szDesc",
    "szName",
    "nLevel",
    "nShowAwardId",
    "nShowAwardKey",
    "nPosInterval",
    "szShowTime"
  }
  local tbActivityConfig = Lib:LoadTabFileEx(szFilepath, "dddddddddsddddddddssdddds", "nConfigId", tbField)
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig or {}) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  self.bLoadConfig = true
end

function tbActivity:GetLastNumOfUsingBox(pPlayer)
  local nUsedBoxNum = pPlayer.GetUserValue(self.SaveKey, self.UsedBoxNumKey)
  return self.tbConfig.nMaxBoxUsedNum - nUsedBoxNum
end

function tbActivity:GetLastNumOfUsingBag(pPlayer)
  local nUsedBagNum = pPlayer.GetUserValue(self.SaveKey, self.UsedBagNumKey)
  return self.tbConfig.nMaxBagUsedNum - nUsedBagNum
end

function tbActivity:GetLogWayFieldName()
  return ActivityMgr.NoLogWay
end
