local tbWnd = Ui:CreateWindow("UIActivity_WuLinDaHuiGuessSelect")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  txtTitle = "node/imgBG/txtTitle",
  btnClose = "node/imgBG/btnClose",
  inputSearch = "node/imgBG/inputSearch",
  btnSearch = "node/imgBG/btnSearch",
  TemplateTeam = "node/imgBG/listScrollView/Viewport/Content/TemplateTeam",
  txtPage = "node/imgBG/txtPage",
  btnPrePage = "node/imgBG/btnPrePage",
  btnNextPage = "node/imgBG/btnNextPage",
  btnFirstPage = "node/imgBG/btnFirstPage",
  btnLastPage = "node/imgBG/btnLastPage",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  Title = "node/imgBG/listTitle"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnPrePage = function(self)
    self:OnClickBtnPrePage()
  end,
  btnNextPage = function(self)
    self:OnClickBtnNextPage()
  end,
  btnFirstPage = function(self)
    self:OnClickBtnFirstPage()
  end,
  btnLastPage = function(self)
    self:OnClickBtnLastPage()
  end,
  btnSearch = function(self)
    self:OnSearchClick()
  end,
  btnHelp = function(self)
  end
}
local MAX_COL_NUM = 4
local tbTitlePath = {
  [1] = "txtTitleTeam",
  [2] = "txtTitleServer",
  [3] = "txtTitleFightPower",
  [4] = "txtTitleBtn"
}

function tbWnd:OnCreate()
  self.tbScrollViewTeam = Ui:NewScrollView(self, tbControls.TemplateTeam)
end

function tbWnd:OnOpen(tbTitleList, tbAllList, tbCallBack, szTitle)
  self.tbTitleList = tbTitleList
  self.tbAllList = tbAllList
  self.tbCallBack = tbCallBack
  self.pPanel:Input_SetText(tbControls.inputSearch, "")
  self:OnSearchClick()
  self.pPanel:SetActive(tbControls.btnHelp, false)
  if szTitle then
    self.pPanel:Label_SetText(tbControls.txtTitle, szTitle)
  end
  Lib:CallBack({
    self._UpdateRankTitle,
    self
  })
end

function tbWnd:_UpdateRankTitle()
  for i = 1, MAX_COL_NUM do
    local szTitle = self.tbTitleList[i]
    local szPath = string.format("%s/%s", tbControls.Title, tbTitlePath[i])
    if false == szTitle or "string" == type(szTitle) and Lib:IsEmptyStr(szTitle) then
      self.pPanel:SetActive(szPath, false)
    else
      self.pPanel:Label_SetText(szPath, szTitle)
      self.pPanel:SetActive(szPath, true)
    end
  end
end

function tbWnd:OnSearchClick()
  local sInput = self.pPanel:Input_GetText(tbControls.inputSearch)
  if nil == sInput then
    return
  end
  self.tbAllTeamResult = self:SearchTeam(sInput)
  self.nMaxPage = math.ceil(#self.tbAllTeamResult / 4)
  self.nCurPage = 1
  self:UpdataTeam()
end

function tbWnd:OnClose()
end

function tbWnd:UpdataTeam()
  if not self.tbAllTeamResult then
    return
  end
  
  local function funSetTeamData(tbElement, nIndex)
    local pPanel = tbElement.pPanel
    local nTeamIndex = (self.nCurPage - 1) * 4 + nIndex
    if not self.tbAllTeamResult[nTeamIndex] or Lib:IsEmptyTB(self.tbAllTeamResult[nTeamIndex]) then
      pPanel:SetActive("txtTeam", false)
      pPanel:SetActive("txtServer", false)
      pPanel:SetActive("txtFightPower", false)
      pPanel:SetActive("btnChange", false)
      pPanel:SetActive("btnSelect", false)
    else
      local tbTeamData = self.tbAllTeamResult[nTeamIndex]
      pPanel:SetActive("txtTeam", true)
      pPanel:SetActive("txtServer", true)
      pPanel:SetActive("txtFightPower", true)
      pPanel:Label_SetText("txtTeam", tbTeamData[1].szName)
      pPanel:Label_SetText("txtServer", tbTeamData[2])
      pPanel:Label_SetText("txtFightPower", tbTeamData[3])
      local bSelected = tbTeamData[4]
      if bSelected then
        pPanel:SetActive("btnChange", false)
        pPanel:SetActive("btnSelect", false)
        Ui:Button_UnBindEvent(pPanel, "btnChange")
        Ui:Button_UnBindEvent(pPanel, "btnSelect")
      else
        pPanel:SetActive("btnChange", false)
        pPanel:SetActive("btnSelect", true)
        Ui:Button_UnBindEvent(pPanel, "btnChange")
        Ui:Button_BindEvent(pPanel, "btnSelect", function()
          if not Lib:IsEmptyTB(self.tbCallBack) then
            table.insert(self.tbCallBack, tbTeamData[1].nId)
            Lib:CallBack(self.tbCallBack)
          end
          self:Close()
        end)
      end
    end
  end
  
  self.tbScrollViewTeam:Reset()
  for i = 1, 4 do
    self.tbScrollViewTeam:SetElement(i, function(tbElement)
      funSetTeamData(tbElement, i)
    end)
  end
  self.pPanel:Label_SetText(tbControls.txtPage, string.format("%d/%d", self.nCurPage, self.nMaxPage))
end

function tbWnd:OnClickBtnPrePage()
  if self.nCurPage <= 1 then
    return
  end
  self.nCurPage = self.nCurPage - 1
  self:UpdataTeam()
end

function tbWnd:OnClickBtnNextPage()
  if self.nCurPage >= self.nMaxPage then
    return
  end
  self.nCurPage = self.nCurPage + 1
  self:UpdataTeam()
end

function tbWnd:OnClickBtnFirstPage()
  if 1 == self.nCurPage then
    return
  end
  self.nCurPage = 1
  self:UpdataTeam()
end

function tbWnd:OnClickBtnLastPage()
  if self.nCurPage == self.nMaxPage then
    return
  end
  self.nCurPage = self.nMaxPage
  self:UpdataTeam()
end

function tbWnd:SearchTeam(str)
  local tbSearchResult = {}
  if not Lib:IsEmptyStr(str) then
    for _, tbData in ipairs(self.tbAllList) do
      if string.find(tbData[1].szName, str) ~= nil then
        table.insert(tbSearchResult, tbData)
      end
    end
  else
    for _, tbData in ipairs(self.tbAllList) do
      table.insert(tbSearchResult, tbData)
    end
  end
  return tbSearchResult
end
