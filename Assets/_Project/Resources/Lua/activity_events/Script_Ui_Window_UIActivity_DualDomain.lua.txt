local tbWnd = Ui:CreateWindow("UIActivity_DualDomain")
local tbControls = {
  Slider = "Cont/Slider",
  Effect = "Cont/Effect"
}
tbWnd.tbControls = tbControls

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_BELONGUI_UPDATE_BELONG,
      self.OnUpdaBelong,
      self
    },
    {
      EventNotify.emNOTIFY_BELONGUI_UPDATE_TARGET,
      self.OnUpdateTarget,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.nBelong = 0
end

function tbWnd:OnOpen(nNpcId, nBelong)
  self.nCurShowNpc = nNpcId
  self:OnUpdaBelong(nNpcId, nBelong)
end

function tbWnd:OnUpdaBelong(nNpcId, nBelong)
  if nNpcId ~= self.nCurShowNpc then
    return
  end
  if self.nBelong == nBelong then
    return
  end
  self:SetBelong(nBelong)
  self.nBelong = nBelong
end

function tbWnd:OnUpdateTarget(nNpcId, nBelong)
  self.nCurShowNpc = nNpcId
  if self.nBelong == nBelong then
    return
  end
  self:SetBelong(nBelong)
end

function tbWnd:SetBelong(nBelong)
  self.pPanel:SetActive(tbControls.Effect, false)
  self.pPanel:SetActive(tbControls.Effect, true)
  self.pPanel:Slider_SetValue(tbControls.Slider, nBelong)
end
