Require("CommonScript/Activity/OBActivity/OBActivityCfg.lua")
ActivityMgr.OBActivity = ActivityMgr.OBActivity or {}
local obActivity = ActivityMgr.OBActivity

function obActivity:TryGainAward(nId)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\228\184\141\229\156\168\229\188\128\229\144\175\231\138\182\230\128\129"))
    return
  end
  if pInstance.szClassName ~= self.ActivityType.OBReward then
    return
  end
  ActivityMgr:CallServerScript(nId, pInstance.Protocol.GetReward)
end
