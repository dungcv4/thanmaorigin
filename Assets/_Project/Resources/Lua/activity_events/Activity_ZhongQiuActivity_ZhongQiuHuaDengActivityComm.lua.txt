Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ZhongQiuHuaDeng)
tbActivity.Protocol = {
  OnSyncQuestionId = 1,
  ReqAnswerQuestion = 2,
  ReqLightRedPoint = 3
}

local function SortFunc(tbNpcInfoDataA, tbNpcInfoDataB)
  return tbNpcInfoDataA.nNpcIndex < tbNpcInfoDataB.nNpcIndex
end

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Activity/ZhongQiuActivity/ZhongQiuHuaDengActivity/ZhongQiuHuaDengActivity.tab"
  local tbField = {
    "nConfigId",
    "szSaveKey",
    "szScriptTableKey",
    "nJoinLevel",
    "nQuestionTime",
    "nAwardId",
    "nCorrectAwardKey",
    "nFaultAwardKey",
    "nAllCompleteAwardKey",
    "nUseHuaDengAwardKey",
    "nHuaDengLaZhuItemId",
    "nMaxJoinCount",
    "nHuaDengLaZhuUp",
    "nMonthUp",
    "nMaxNotifyNpcTime"
  }
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "dssdddddddddddd", "nConfigId", tbField)
  local szFilePath = "Setting/Activity/ZhongQiuActivity/ZhongQiuHuaDengActivity/NpcPosList.tab"
  local tbField = {
    "nConfigId",
    "nMapTemplateId",
    "nPosX",
    "nPosY"
  }
  local tbNpcPosCfg = Lib:LoadTabFileEx(szFilePath, "dddd", nil, tbField)
  local tbMapTable = {}
  local tbMapHash = {}
  local tbPosTable = {}
  for _, tbInfo in ipairs(tbNpcPosCfg) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("ZhongQiuHuaDengActivity:LoadConfig NpcPosList nConfigId=%d not exist", tbInfo.nConfigId))
    if tbMapTable[tbInfo.nConfigId] == nil then
      tbMapTable[tbInfo.nConfigId] = {}
    end
    if tbPosTable[tbInfo.nConfigId] == nil then
      tbPosTable[tbInfo.nConfigId] = {}
    end
    if tbMapHash[tbInfo.nConfigId] == nil then
      tbMapHash[tbInfo.nConfigId] = {}
    end
    local bIsHave = false
    if tbMapHash[tbInfo.nConfigId][tbInfo.nMapTemplateId] ~= nil then
      bIsHave = true
    end
    if false == bIsHave then
      table.insert(tbMapTable[tbInfo.nConfigId], tbInfo.nMapTemplateId)
      tbMapHash[tbInfo.nConfigId][tbInfo.nMapTemplateId] = tbInfo.nMapTemplateId
    end
    if tbPosTable[tbInfo.nConfigId][tbInfo.nMapTemplateId] == nil then
      tbPosTable[tbInfo.nConfigId][tbInfo.nMapTemplateId] = {}
    end
    local tbPosData = {}
    tbPosData.nPosX = tbInfo.nPosX
    tbPosData.nPosY = tbInfo.nPosY
    table.insert(tbPosTable[tbInfo.nConfigId][tbInfo.nMapTemplateId], tbPosData)
  end
  for nConfigId, tbMapList in pairs(tbMapTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbMapList = tbMapList
    end
  end
  for nConfigId, tbMapPos in pairs(tbPosTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbMapPos = tbMapPos
    end
  end
  local szFilePath = "Setting/Activity/ZhongQiuActivity/ZhongQiuHuaDengActivity/NpcInfo.tab"
  local tbField = {
    "nConfigId",
    "nNpcIndex",
    "nNpcTemplateId",
    "szName"
  }
  local tbNpcInfoCfg = Lib:LoadTabFileEx(szFilePath, "ddds", nil, tbField)
  local tbNpcInfoTable = {}
  for i, tbNpcInfoCfgData in ipairs(tbNpcInfoCfg) do
    if tbNpcInfoTable[tbNpcInfoCfgData.nConfigId] == nil then
      tbNpcInfoTable[tbNpcInfoCfgData.nConfigId] = {}
    end
    local tbZhongQiuHuaDengNpcInfoData = {}
    tbZhongQiuHuaDengNpcInfoData.nNpcIndex = tbNpcInfoCfgData.nNpcIndex
    tbZhongQiuHuaDengNpcInfoData.nNpcTemplateId = tbNpcInfoCfgData.nNpcTemplateId
    tbZhongQiuHuaDengNpcInfoData.szName = tbNpcInfoCfgData.szName
    table.insert(tbNpcInfoTable[tbNpcInfoCfgData.nConfigId], tbZhongQiuHuaDengNpcInfoData)
  end
  for nConfigId, tbNpcInfoList in pairs(tbNpcInfoTable) do
    table.sort(tbNpcInfoList, SortFunc)
  end
  for nConfigId, tbNpcInfoList in pairs(tbNpcInfoTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbNpcInfoList = tbNpcInfoList
    end
  end
  local szFilePath = "Setting/Activity/ZhongQiuActivity/ZhongQiuHuaDengActivity/Question.tab"
  local tbField = {
    "nConfigId",
    "nId",
    "nAnswerId",
    "szQuestion",
    "szChoice1",
    "szChoice2",
    "szChoice3",
    "szChoice4"
  }
  local tbQuestionCfg = Lib:LoadTabFileEx(szFilePath, "dddsssss", nil, tbField)
  local tbQuestionTable = {}
  local tbQuestionIdTable = {}
  for i, tbQuestionCfgData in ipairs(tbQuestionCfg) do
    if tbQuestionTable[tbQuestionCfgData.nConfigId] == nil then
      tbQuestionTable[tbQuestionCfgData.nConfigId] = {}
    end
    if tbQuestionIdTable[tbQuestionCfgData.nConfigId] == nil then
      tbQuestionIdTable[tbQuestionCfgData.nConfigId] = {}
    end
    local tbZhongQiuHuaDengQuestionData = {}
    tbZhongQiuHuaDengQuestionData.nId = tbQuestionCfgData.nId
    tbZhongQiuHuaDengQuestionData.nAnswerId = tbQuestionCfgData.nAnswerId
    tbZhongQiuHuaDengQuestionData.szQuestion = tbQuestionCfgData.szQuestion
    tbZhongQiuHuaDengQuestionData.szChoice1 = tbQuestionCfgData.szChoice1
    tbZhongQiuHuaDengQuestionData.szChoice2 = tbQuestionCfgData.szChoice2
    tbZhongQiuHuaDengQuestionData.szChoice3 = tbQuestionCfgData.szChoice3
    tbZhongQiuHuaDengQuestionData.szChoice4 = tbQuestionCfgData.szChoice4
    tbQuestionTable[tbQuestionCfgData.nConfigId][tbQuestionCfgData.nId] = tbZhongQiuHuaDengQuestionData
    table.insert(tbQuestionIdTable[tbQuestionCfgData.nConfigId], tbQuestionCfgData.nId)
  end
  for nConfigId, tbQuestion in pairs(tbQuestionTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbQuestion = tbQuestion
    end
  end
  for nConfigId, tbQuestionIdList in pairs(tbQuestionIdTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbQuestionIdList = tbQuestionIdList
    end
  end
  local szFilePath = "Setting/Activity/ZhongQiuActivity/ZhongQiuHuaDengActivity/ContinuousScore.tab"
  local tbField = {
    "nConfigId",
    "nContinuousNum",
    "nScore"
  }
  local tbScoreCfg = Lib:LoadTabFileEx(szFilePath, "ddd", nil, tbField)
  local tbScoreTable = {}
  for i, tbScoreCfgData in ipairs(tbScoreCfg) do
    if tbScoreTable[tbScoreCfgData.nConfigId] == nil then
      tbScoreTable[tbScoreCfgData.nConfigId] = {}
    end
    tbScoreTable[tbScoreCfgData.nConfigId][tbScoreCfgData.nContinuousNum] = tbScoreCfgData.nScore
  end
  for nConfigId, tbScore in pairs(tbScoreTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbScore = tbScore
    end
  end
  local szFilePath = "Setting/Activity/ZhongQiuActivity/XuanLanHuaDeng/MapList.tab"
  local tbField = {"nConfigId", "szMapClass"}
  local tbHuaDengMapCfg = Lib:LoadTabFileEx(szFilePath, "ds", nil, tbField)
  for i, tbHuaDengMapCfgData in ipairs(tbHuaDengMapCfg) do
    if self.ACTIVITY_CONFIG[tbHuaDengMapCfgData.nConfigId] ~= nil then
      if nil == self.ACTIVITY_CONFIG[tbHuaDengMapCfgData.nConfigId].tbXuanLanHuaDengData then
        self.ACTIVITY_CONFIG[tbHuaDengMapCfgData.nConfigId].tbXuanLanHuaDengData = {}
      end
      if nil == self.ACTIVITY_CONFIG[tbHuaDengMapCfgData.nConfigId].tbXuanLanHuaDengData.tbMapClassList then
        self.ACTIVITY_CONFIG[tbHuaDengMapCfgData.nConfigId].tbXuanLanHuaDengData.tbMapClassList = {}
      end
      table.insert(self.ACTIVITY_CONFIG[tbHuaDengMapCfgData.nConfigId].tbXuanLanHuaDengData.tbMapClassList, tbHuaDengMapCfgData.szMapClass)
    end
  end
  local szFilePath = "Setting/Activity/ZhongQiuActivity/XuanLanHuaDeng/Content.tab"
  local tbField = {"nConfigId", "szText"}
  local tbHuaDengTextCfg = Lib:LoadTabFileEx(szFilePath, "ds", nil, tbField)
  for i, tbHuaDengTextCfgData in ipairs(tbHuaDengTextCfg) do
    if self.ACTIVITY_CONFIG[tbHuaDengTextCfgData.nConfigId] ~= nil then
      if nil == self.ACTIVITY_CONFIG[tbHuaDengTextCfgData.nConfigId].tbXuanLanHuaDengData then
        self.ACTIVITY_CONFIG[tbHuaDengTextCfgData.nConfigId].tbXuanLanHuaDengData = {}
      end
      if nil == self.ACTIVITY_CONFIG[tbHuaDengTextCfgData.nConfigId].tbXuanLanHuaDengData.tbTextList then
        self.ACTIVITY_CONFIG[tbHuaDengTextCfgData.nConfigId].tbXuanLanHuaDengData.tbTextList = {}
      end
      table.insert(self.ACTIVITY_CONFIG[tbHuaDengTextCfgData.nConfigId].tbXuanLanHuaDengData.tbTextList, tbHuaDengTextCfgData.szText)
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:GetQuestionTime()
  if self.ACTIVITY_CONFIG[self.nConfigId].nQuestionTime ~= nil then
    return self.ACTIVITY_CONFIG[self.nConfigId].nQuestionTime
  end
  return 0
end

function tbActivity:GetMapList()
  if self.ACTIVITY_CONFIG[self.nConfigId].tbMapList ~= nil then
    return self.ACTIVITY_CONFIG[self.nConfigId].tbMapList
  end
  return {}
end
