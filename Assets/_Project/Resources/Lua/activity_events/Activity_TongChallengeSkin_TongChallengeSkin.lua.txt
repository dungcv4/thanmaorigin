Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.TongChallengeSkin)

function tbActivity:LoadConfig()
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile("Setting/Activity/TongChallengeSkin/TongChallengeSkin.tab", {"nConfigId"}, nil, nil, false)
  local tbConfigs = Lib:EasyLoadTabFile("Setting/Activity/TongChallengeSkin/TongChallengeMsg.tab", {"nConfigId", "nStage"})
  for nConfigId, tbStage in pairs(tbConfigs) do
    self.ACTIVITY_CONFIG[nConfigId] = self.ACTIVITY_CONFIG[nConfigId] or {}
    self.ACTIVITY_CONFIG[nConfigId].tbMsg = self.ACTIVITY_CONFIG[nConfigId].tbMsg or {}
    for nStage, tbMsg in pairs(tbStage) do
      self.ACTIVITY_CONFIG[nConfigId].tbMsg[nStage] = tbMsg.szMsg
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:GetSkinName()
  return self.tbConfig.szName
end

function tbActivity:GetStageMsg(nStage)
  if not self.tbConfig.tbMsg then
    return
  end
  return self.tbConfig.tbMsg[nStage]
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.tbMsg = {}
  for _, tbInfo in ipairs(tbActCfg.TongChallengeMsg) do
    self.tbConfig.tbMsg[tbInfo.nStage] = i18n.Encode(tbInfo.szMsg)
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  return true
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end

function tbActivity:GetLogWayFieldName()
  return ActivityMgr.NoLogWay
end
