local tbWnd = Ui:CreateWindow("UIFestival_GetMoneyBase")

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  if not self.tbSubActivityScrollView then
    self.tbSubActivityScrollView = Ui:NewScrollView(self, self.szSubActivityTemplate)
  end
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
  EventNotify:RegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self.UpdateItem, self)
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.nIndex = nIndex
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  local tbExtraData = ActivityMgr.EntranceSchedule:GetExtraData(nEntranceIndex, nIndex) or {}
  self.tbExtraBtnList = ActivityMgr.EntranceSchedule:GetExtraBtnList(nEntranceIndex, nIndex) or {}
  self.tbSubActivityList = tbExtraData.SubActivity or {}
  self.tbActivityItem = tbExtraData.ActivityItem or {}
  self.szHelpKey = tbInfo.szHelpKey
  self.pPanel:Label_SetText(self.szTitle, tbInfo.szName)
  self.pInstance = ActivityMgr.EntranceSchedule:BuildInstance(tbInfo.nId, ActivityMgr.Def.ClassNameList.CommonEntry)
  local szTime = string.format("<color=#1FFB24>%s</color>", tbInfo.szShowTime)
  local nStartTime = self.pInstance:GetStartTime()
  local nEndTime = self.pInstance:GetEndTime()
  if nStartTime < GetTime() then
    local nDiff = nEndTime - GetTime()
    szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", tbInfo.szShowTime, Lib:TimeDesc2(nDiff))
  end
  if nEndTime < GetTime() then
    szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> \229\183\178\231\187\147\230\157\159", tbInfo.szShowTime)
  end
  self.pPanel:Label_SetText(self.szTime, szTime)
  self:UpdateListActivity()
  self:UpdateItem()
  self:DynamicRegisteRedPoint()
  self:UpdateExtraButton()
end

function tbWnd:_SetSubActivityElement(tbElement, tbSubActivityInfo)
  if self.tbSubActivityElementPath then
    tbElement.pPanel:Label_SetText(self.tbSubActivityElementPath.txtActivityName, i18n.Get(tbSubActivityInfo.szName) or "")
    tbElement.pPanel:Label_SetText(self.tbSubActivityElementPath.txtTimes, i18n.Get(tbSubActivityInfo.szShowTime or ""))
    tbElement.pPanel:Label_SetText(self.tbSubActivityElementPath.txtInfo, i18n.Get(tbSubActivityInfo.szDesc))
    local tbAward = Award:GetAward(tbSubActivityInfo.nShowAwardId, tbSubActivityInfo.nShowAwardKey, {bNotMerge = true})
    if not tbElement.tbItemGridScrollView then
      tbElement.tbItemGridScrollView = Ui:NewItemGridScrollView(tbElement, self.tbSubActivityElementPath.ItemTemplate, 1, 1)
    end
    tbElement.tbItemGridScrollView:Reset()
    tbElement.tbItemGridScrollView:UpdateGenericItem(tbAward, true)
    if self.tbSubActivityElementPath.btnView then
      if GetTime() > (Lib:ParseDateTime(tbSubActivityInfo.szEndTime) or 0) then
        tbElement.pPanel:Button_SetEnable(self.tbSubActivityElementPath.btnView, false)
        tbElement.pPanel:Button_SetText(self.tbSubActivityElementPath.btnView, i18n.EasyGet("\229\183\178\231\187\147\230\157\159"), "Text")
        if tbSubActivityInfo.szRedPoint then
          RedPointMgr:DynamicUnRegist(tbElement.pPanel, self.tbSubActivityElementPath.btnView)
        end
      else
        tbElement.pPanel:Button_SetEnable(self.tbSubActivityElementPath.btnView, true)
        tbElement.pPanel:Button_SetText(self.tbSubActivityElementPath.btnView, i18n.EasyGet("\229\137\141\229\190\128"), "Text")
        if tbSubActivityInfo.szRedPoint then
          RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, self.tbSubActivityElementPath.btnView, {
            RedPointMgr.EVENT[tbSubActivityInfo.szRedPoint]
          })
        end
        Ui:Button_BindEvent(tbElement.pPanel, self.tbSubActivityElementPath.btnView, self.OnClickJoinBtn, self, ActivityMgr.tbJumpFunc[tbSubActivityInfo.szCallBack], table.unpack(Lib:SplitStrTry2Num(tbSubActivityInfo.szParam, "|")))
      end
    end
  end
end

function tbWnd:UpdateListActivity()
  for nIndex, tbSubActivityInfo in ipairs(self.tbSubActivityList) do
    self.tbSubActivityScrollView:SetElement(nIndex, function(tbElement)
      self:_SetSubActivityElement(tbElement, tbSubActivityInfo)
    end)
  end
end

function tbWnd:UpdateExtraButton()
  if not self.tbExtraBtnView then
    self.tbExtraBtnView = Ui:NewScrollView(self, self.szExtraBtnTemplate)
  end
  self.tbExtraBtnView:Clear()
  for nIndex, tbExtraBtnInfo in ipairs(self.tbExtraBtnList) do
    self.tbExtraBtnView:SetElement(nIndex, function(tbElement)
      self:UpdateExtraBtnElement(tbElement, tbExtraBtnInfo)
    end)
  end
end

function tbWnd:UpdateExtraBtnElement(tbElement, tbBtnInfo)
  tbElement.pPanel:Label_SetText("txtBtn", i18n.Get(tbBtnInfo.szBtnName))
  tbElement.pPanel:Sprite_SetSprite("Image", tbBtnInfo.szImgInfo)
  local tbParam = Lib:SplitStrTry2Num(tbBtnInfo.szParam or "", "|")
  table.insert(tbParam, 1, self.pInstance.nId)
  tbElement.pPanel:Button_BindEvent("Main", self.OnClickJoinBtn, self, ActivityMgr.tbJumpFunc[tbBtnInfo.szCallBack], table.unpack(tbParam))
  if tbBtnInfo.szRedPoint and "" ~= tbBtnInfo.szRedPoint then
    RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "Main", ActivityMgr:GenRedPointEventIdList(Lib:SplitStr(tbBtnInfo.szRedPoint, "|")))
  end
end

function tbWnd:_SetActivityItemElement(tbElement, nItemId)
  local tbGrid = Ui:NewExistItemGrid(tbElement, self.tbActivityItemElementPath.UIItemGrid)
  tbGrid:SetItemByTemplate(nItemId, nil, nil, nil, {bShowTip = true})
  tbGrid:SetClickEnable(true)
  tbElement.pPanel:Label_SetText(self.tbActivityItemElementPath.txtItemCount, Item:GetName(nItemId, me))
  tbElement.pPanel:Label_SetText(self.tbActivityItemElementPath.txtItemCountValue, i18n.EasyGet("\230\139\165\230\156\137\239\188\154{0}", me.GetItemCountByTypeInBag(nItemId)))
end

function tbWnd:UpdateItem()
  if not Lib:IsEmptyStr(self.tbActivityItem.szItemId) then
    local tbItemIdList = Lib:SplitStr2Num(self.tbActivityItem.szItemId, "|")
    if not self.tbActivityItemScrollView then
      self.tbActivityItemScrollView = Ui:NewScrollView(self, self.szActivityItemTemplate)
    end
    for nIndex, nItemId in ipairs(tbItemIdList) do
      self.tbActivityItemScrollView:SetElement(nIndex, function(tbElement)
        self:_SetActivityItemElement(tbElement, nItemId)
      end)
    end
    if not Lib:IsEmptyStr(self.tbActivityItem.szPlusCallBack) and ActivityMgr.tbJumpFunc[self.tbActivityItem.szPlusCallBack] then
      local tbParam = Lib:SplitStrTry2Num(self.tbActivityItem.szPlusParam or "", "|")
      table.insert(tbParam, 1, self.pInstance.nId)
      Ui:Button_BindEvent(self.pPanel, self.szAddBtn, self.OnClickJoinBtn, self, ActivityMgr.tbJumpFunc[self.tbActivityItem.szPlusCallBack], table.unpack(tbParam))
    end
  end
end

function tbWnd:OnClickJoinBtn(func, ...)
  func(...)
end

function tbWnd:OnClickAddBtn(func, ...)
  func(...)
end

function tbWnd:UpdateOtherInfo()
end

function tbWnd:DynamicRegisteRedPoint()
end

function tbWnd:UnRegisteRedPoint()
end

function tbWnd:OnClose()
  self:UnRegisteRedPoint()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_SYNC_ITEM, self)
  if self.tbSubActivityScrollView then
    self.tbSubActivityScrollView:Clear()
    self.tbSubActivityScrollView = nil
  end
  if self.tbExtraBtnView then
    self.tbExtraBtnView:Clear()
    self.tbExtraBtnView = nil
  end
end
