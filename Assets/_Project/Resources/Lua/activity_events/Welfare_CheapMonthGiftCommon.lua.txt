CheapMonthGift.SAVE_GROUP = GetGroupKey("CheapMonthGift")
CheapMonthGift.BUY_CD_END_TIME = 1

function CheapMonthGift:GetBuyCdEndTime(pPlayer)
  return pPlayer.GetUserValue(self.SAVE_GROUP, self.BUY_CD_END_TIME)
end

function CheapMonthGift:CheckBuyState(pPlayer)
  local nCdEndTime = self:GetBuyCdEndTime(pPlayer)
  if nCdEndTime >= GetTime() then
    return true
  end
  return false
end

function CheapMonthGift:GetBuyCDEndLeftDay(pPlayer)
  local nLeftDay = Lib:GetLocalDay(pPlayer.GetUserValue(self.SAVE_GROUP, self.BUY_CD_END_TIME) - Recharge.CARD_BUY_DAY_UPDATE_TIME) - Lib:GetLocalDay(GetTime() - Recharge.CARD_BUY_DAY_UPDATE_TIME)
  return nLeftDay < 0 and 0 or nLeftDay
end
