function SevenDayWelfare:RefreshGetAwardData()
  RemoteServer.RefreshGetAwardData()
end

function SevenDayWelfare:OnS2CAwardGetData(tbWelfareGetData)
  SevenDayWelfare.tbWelfareGetData = tbWelfareGetData
  EventNotify.OnNotify(EventNotify.emNOTIFY_SEVENDAYWELFARE_SYNC)
  SevenDayWelfare:OnCheckRedPoint()
end

function SevenDayWelfare:OnS2CRoleIsBind(bIsBind)
  SevenDayWelfare.bIsBind = bIsBind
end

function SevenDayWelfare:IsBind()
  return SevenDayWelfare.bIsBind
end

function SevenDayWelfare:GetCanGetAwardNum()
  if SevenDayWelfare.tbWelfareGetData then
    return SevenDayWelfare.tbWelfareGetData.nCanGetAwardNum or 0
  else
    print("GetCanGetAwardNum nil ")
  end
end

function SevenDayWelfare:OnCheckRedPoint()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.SevenDayWelfare)
  if not self:CheckIsOpen(me) then
    return
  end
  local nLoginTime = SevenDayWelfare:GetCanGetAwardNum()
  for nDay = 1, SevenDayWelfare.DAY do
    local hasGetAward = 1 == SevenDayWelfare.tbWelfareGetData[nDay]
    local canGetAward = nDay <= nLoginTime
    if canGetAward and not hasGetAward then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.SevenDayWelfare)
      return
    end
  end
end

function SevenDayWelfare:BindRoleIsMe()
  if not SevenDayWelfare.tbWelfareGetData then
    return false
  end
  local dwBindId = SevenDayWelfare.tbWelfareGetData.dwBindId
  if not dwBindId then
    return true
  elseif dwBindId == me.dwID then
    return true
  else
    return false
  end
end

function SevenDayWelfare:CheckIsOpen(pPlayer)
  if not OnOffController:Check(OnOffController.KEY.SevenDayWelfare) then
    return false
  end
  if not SevenDayWelfare:BindRoleIsMe() then
    return false
  end
  if pPlayer.nLevel < SevenDayWelfare.nOPENLEVEL then
    return false, i18n.EasyGet("\229\141\129\230\151\165\231\166\143\229\136\169{0}\231\186\167\229\188\128\230\148\190\239\188\129", SevenDayWelfare.nOPENLEVEL)
  end
  for nDay = 1, SevenDayWelfare.DAY do
    local hasGetAward = 1 == SevenDayWelfare.tbWelfareGetData[nDay]
    if not hasGetAward then
      return true
    end
  end
  return false
end

function SevenDayWelfare:ActiveIsOpen()
  return false
end
