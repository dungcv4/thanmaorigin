function ActivityCheck:RegisterActivity(szActivityName, fnCustom, tbFnParam, bNeedKin, bNeedNoTeam, bNeedSafeArea, tbNeedCloseUi, bIngoreMap)
  self.tbActivity = self.tbActivity or {}
  if self.tbActivity[szActivityName] then
    Log("RegisterActivity Duplicate: " .. debug.traceback())
  end
  local tbReg = {}
  tbReg.szActivityName = szActivityName
  tbReg.bNeedNoTeam = bNeedNoTeam
  tbReg.bNeedKin = bNeedKin
  tbReg.bNeedSafeArea = bNeedSafeArea
  tbReg.fnCustom = fnCustom
  tbReg.tbFnParam = tbFnParam
  tbReg.tbNeedCloseUi = tbNeedCloseUi
  tbReg.bIngoreMap = bIngoreMap
  self.tbActivity[szActivityName] = tbReg
end

function ActivityCheck:CheckCanEnterActivityEx(pPlayer, fnCustom, fnEnter, bNeedKin, bNeedNoTeam, bNeedSafeArea, tbFnParam, tbNeedCloseUi, bIngoreMap)
  if not self:CheckIsInChuanGong(pPlayer) then
    return false, i18n.EasyGet("\228\188\160\229\138\159\231\138\182\230\128\129\228\184\141\232\131\189\229\143\130\229\138\160\230\173\164\230\180\187\229\138\168")
  end
  if JiangHuZhuJiu:IsInGame(pPlayer) then
    return false, i18n.EasyGet("\232\175\183\229\133\136\233\128\128\229\135\186\230\177\159\230\185\150\231\133\174\233\133\146\229\144\142\229\134\141\233\135\141\230\150\176\229\143\130\229\138\160")
  end
  if pPlayer.bInXiuWeiSit then
    return false, i18n.Get(XiuWei.tbText.szXiuWeiSitEnterActivityTips)
  end
  local bRet, szMsg = pPlayer.CanTranOpt()
  if not bRet then
    return bRet, szMsg
  end
  local tbReg = {}
  tbReg.bNeedNoTeam = bNeedNoTeam
  tbReg.bNeedKin = bNeedKin
  tbReg.bNeedSafeArea = bNeedSafeArea
  if tbReg.bNeedKin and not self:CheckIsInKin(pPlayer) then
    return false, i18n.EasyGet("\233\156\128\232\166\129\229\138\160\229\133\165\229\174\182\230\151\143\230\136\150\229\184\174\228\188\154\230\137\141\229\143\175\229\143\130\228\184\142\230\173\164\230\180\187\229\138\168")
  end
  if pPlayer.IsDead() then
    return false, i18n.EasyGet("\230\173\187\228\186\161\231\138\182\230\128\129\228\184\141\232\131\189\230\147\141\228\189\156")
  end
  if MODULE_GAMESERVER then
    if DungeonMgr:CheckIsWaitingMap(pPlayer) then
      return false, i18n.EasyGet("\229\183\178\229\156\168\229\137\175\230\156\172\233\152\159\229\136\151\228\184\173")
    end
    if pPlayer.InSwitchMapProcess() then
      return false, i18n.EasyGet("\230\173\163\229\156\168\228\188\160\233\128\129")
    end
    if tbReg.bNeedNoTeam and pPlayer.dwTeamID > 0 then
      return false, i18n.EasyGet("\231\187\132\233\152\159\231\138\182\230\128\129\228\184\141\232\131\189\229\143\130\229\138\160\230\173\164\230\180\187\229\138\168")
    end
    if tbReg.bNeedSafeArea and not self:CheckIsInSafeArea(pPlayer, bIngoreMap) then
      return false, i18n.EasyGet("\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186\228\184\141\232\131\189\229\143\130\229\138\160\230\173\164\230\180\187\229\138\168")
    end
    local _, bRet, szInfo
    if fnCustom then
      if tbFnParam then
        _, bRet, szInfo = Lib:CallBack({
          fnCustom,
          tbFnParam[1],
          pPlayer,
          table.unpack(tbFnParam, 2)
        })
      else
        _, bRet, szInfo = Lib:CallBack({fnCustom, pPlayer})
      end
    else
      bRet = true
    end
    return bRet, szInfo
  elseif MODULE_GAMECLIENT then
    if tbReg.bNeedNoTeam and self:CheckIsInTeam(pPlayer) then
      local function fnContinue()
        PlayerEvent:UnRegister(pPlayer, "OnQuiteTeam", ActivityCheck.nQuiteTeamRegisterID)
        
        ActivityCheck:CheckCanEnterActivityEx(pPlayer, fnCustom, fnEnter, bNeedKin, bNeedNoTeam, bNeedSafeArea, tbFnParam, tbNeedCloseUi, false)
      end
      
      pPlayer.MsgBox(i18n.EasyGet("\231\187\132\233\152\159\231\138\182\230\128\129\230\151\160\230\179\149\229\143\130\229\138\160\239\188\140\230\152\175\229\144\166\233\128\128\229\135\186\233\152\159\228\188\141\239\188\159"), {
        {
          i18n.EasyGet("\233\128\128\229\135\186\233\152\159\228\188\141"),
          ActivityCheck.QuiteTeam,
          fnContinue
        },
        {
          i18n.EasyGet("\229\143\150\230\182\136")
        }
      })
      return false, i18n.EasyGet("\231\187\132\233\152\159\231\138\182\230\128\129\228\184\141\232\131\189\229\143\130\229\138\160")
    end
    if tbReg.bNeedSafeArea and not self:CheckIsInSafeArea(pPlayer, bIngoreMap) then
      local function fnContinue()
        ActivityCheck:CheckCanEnterActivityEx(pPlayer, fnCustom, fnEnter, bNeedKin, bNeedNoTeam, bNeedSafeArea, tbFnParam, tbNeedCloseUi, bIngoreMap)
      end
      
      GeneralProcess:JoinGamePlayFormMessageBox(szMsg, tbNeedCloseUi, fnContinue, nil, nil)
      if GlobalHangUp:IsGlobalHangUpMap(pPlayer.nMapTemplateId) then
        return false, i18n.EasyGet("\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186\228\184\141\232\131\189\231\166\187\229\188\128")
      else
        return false, i18n.EasyGet("\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186\228\184\141\232\131\189\229\143\130\229\138\160\230\173\164\230\180\187\229\138\168")
      end
    end
    local _, bRet, szInfo
    if fnCustom then
      if tbFnParam then
        _, bRet, szInfo = Lib:CallBack({
          fnCustom,
          tbFnParam[1],
          pPlayer,
          table.unpack(tbFnParam, 2)
        })
      else
        _, bRet, szInfo = Lib:CallBack({fnCustom, pPlayer})
      end
    else
      bRet = true
    end
    if not bRet then
      return bRet, szInfo
    elseif fnEnter then
      self:_CloseUi(tbNeedCloseUi)
      Lib:CallBack({fnEnter})
      return true
    end
    self:_CloseUi(tbNeedCloseUi)
    return true
  end
end

function ActivityCheck:_CloseUi(tbNeedCloseUi)
  Ui:CloseWindow("UIActivity")
  for _, szUiName in pairs(tbNeedCloseUi or {}) do
    Ui:CloseWindow(szUiName)
  end
end

function ActivityCheck:CheckCanEnterActivity(szActivityName, pPlayer, fnEnter)
  Log("CheckCanEnterActivity", szActivityName)
  szActivityName = Lib:StrTrim(szActivityName)
  local tbReg = self.tbActivity[szActivityName]
  if not tbReg then
    LogErr("Could not find activity,", szActivityName)
    LogErr(debug.traceback())
    return
  end
  local fnCustom = ActivityCheck.tbCheckConfig[szActivityName]
  fnCustom = fnCustom or tbReg.fnCustom
  return self:CheckCanEnterActivityEx(pPlayer, fnCustom, fnEnter, tbReg.bNeedKin, tbReg.bNeedNoTeam, tbReg.bNeedSafeArea, tbReg.tbFnParam, tbReg.tbNeedCloseUi, false)
end

function ActivityCheck.QuiteTeam(fnContinue)
  Log("ActivityCheck.QuiteTeam")
  TeamMgr.Quite(true)
  ActivityCheck.nQuiteTeamRegisterID = PlayerEvent:Register(me, "OnQuiteTeam", fnContinue)
end

function ActivityCheck:CheckIsInChuanGong(pPlayer)
  if pPlayer.bInChuanGong then
    return false
  end
  return true
end

function ActivityCheck:CheckIsInSafeArea(pPlayer, bIngoreMap)
  if Map:IsTransForbid(pPlayer.nMapTemplateId) then
    return false
  end
  if not bIngoreMap and Map:IsFuben(pPlayer.nMapTemplateId) then
    return false
  end
  if (Map:IsFieldFightMap(pPlayer.nMapTemplateId) or Map:IsCityMap(pPlayer.nMapTemplateId) or Map:IsTongMap(pPlayer.nMapTemplateId)) and 0 ~= pPlayer.nFightMode then
    return false
  end
  return true
end

function ActivityCheck:CheckIsInKin(pPlayer)
  return pPlayer.dwKinId > 0
end

function ActivityCheck:CheckIsInTeam(pPlayer)
  Log("ActivityCheck:CheckIsInTeam")
  return TeamMgr:HasTeam()
end

function ActivityCheck:CheckTeaammate(me, nActivityId, tbExclude)
  tbExclude = tbExclude or {}
  local nTeamId = me.dwTeamID
  local tbTeam = TeamMgr:GetTeamById(nTeamId)
  local tbMember = tbTeam:GetMembers()
  for _, nPlayerID in pairs(tbMember) do
    if not tbExclude[nPlayerID] then
      local pMember = KPlayer.GetPlayerObjById(nPlayerID)
      local bRet, szInfo = self:CheckCanEnterActivity(nActivityId, pMember, tbExclude)
      if not bRet then
        return false, i18n.EasyGet("{0}\228\184\141\230\187\161\232\182\179\230\157\161\228\187\182: {1}", pMember.szName, szInfo)
      end
    end
  end
  return true
end
