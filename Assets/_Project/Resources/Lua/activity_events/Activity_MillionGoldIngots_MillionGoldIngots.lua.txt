Require("CommonScript/Activity/ActivityMgr.lua")
ActivityMgr.szMillionGoldIngotsClassName = "MillionGoldIngots"
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szMillionGoldIngotsClassName)
tbActivity.SaveKey = GetGroupKey(ActivityMgr.szMillionGoldIngotsClassName)
tbActivity.nLastStartTimeKey = 1
tbActivity.nMailTime = 2

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/MillionGoldIngots.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[nConfigId].szLogWay) then
      self.ACTIVITY_CONFIG[nConfigId].szLogWay = "LogWay_MillionGoldIngots"
    end
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[nConfigId].szSaveKey) then
      self.ACTIVITY_CONFIG[nConfigId].szSaveKey = ActivityMgr.szMillionGoldIngotsClassName
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
  return true
end

function tbActivity:CheckGoldIngots(pPlayer)
  if not self:IsOpen() then
    return false
  end
  local nGoldIngots = pPlayer.GetMoney("JinDing")
  if nGoldIngots <= 0 then
    return false
  end
  return true
end

function tbActivity:MillionGoldIngotsIsOpen()
  if not self:IsOpen() then
    return false
  end
  return true
end
