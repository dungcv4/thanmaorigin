Require("CommonScript/Activity/NewTimeFrameReminder/NewTimeFrameReminder.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.NewTimeFrameReminder)
tbActivity.tbCacheData = {}

function tbActivity:GetCurReminderInfo()
  for _, tbInfo in ipairs(self.tbConfig.tbReminder) do
    if (Lib:IsEmptyStr(tbInfo.szTimeFrameStart) or 1 == GetTimeFrameState(tbInfo.szTimeFrameStart)) and (Lib:IsEmptyStr(tbInfo.szTimeFrameEnd) or 0 == GetTimeFrameState(tbInfo.szTimeFrameEnd)) then
      return tbInfo
    end
  end
end

function tbActivity:LoadSyncData(tbData)
  self.tbCacheData.nIgnoredIndex = tbData.nIgnoredIndex
  self:CheckRedPoint()
end

function tbActivity:IsShowOpen()
  return self:GetCurReminderInfo() ~= nil
end

function tbActivity:GetPlayerPrefsKey()
  return table.concat({
    "NewTimeFrameReminder",
    self.nId,
    me.dwID or 0
  }, "_")
end

function tbActivity:CheckRedPoint(bFade)
  if not self.tbCacheData.nIgnoredIndex then
    return
  end
  local tbInfo = self:GetLastReminderInfo()
  if not tbInfo then
    return
  end
  local nCurReminderIndex = tbInfo.nIndex
  if nCurReminderIndex == self.tbCacheData.nIgnoredIndex then
    return
  end
  local szKey = self:GetPlayerPrefsKey()
  local nCurIndex = Ui.PlayerPrefs.GetInt(szKey, 0)
  if nCurIndex == nCurReminderIndex then
    return
  end
  if not bFade and self:IsShowOpen() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.NewTimeFrameReminder)
  else
    Ui.PlayerPrefs.SetInt(szKey, nCurReminderIndex)
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.NewTimeFrameReminder)
  end
end

function tbActivity:OnCrossDay()
  self:CheckRedPoint()
end
