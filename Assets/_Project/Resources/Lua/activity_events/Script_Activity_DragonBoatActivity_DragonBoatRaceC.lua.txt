Require("CommonScript/Activity/DragonBoatActivity/DragonBoatRaceRankCommon.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.DragonBoatRace)

function tbActivity:SignIn()
  local nTicketCount = me.GetItemCountInBags(DragonBoatRace.Cfg.nDragonBoatTicketId)
  local nBoatRaceJoinCount = DegreeCtrl:GetDegree(me, DragonBoatRace.szDegreeName)
  
  local function funcOK()
    if TeamMgr:HasTeam() then
      tbActivity:IfPlayerIsInTeam()
    elseif tbActivity:CheckTicketEnable() then
      RemoteServer.DragonBoatRaceRequest("SignUp", me.dwID)
    end
  end
  
  local function funcCancel()
  end
  
  local szText1 = i18n.EasyGet("\230\152\175\229\144\166\230\182\136\232\128\151<color=red>1</color>\230\172\161\229\143\130\228\184\142\230\172\161\230\149\176\230\138\165\229\144\141\229\143\130\229\138\160\230\175\148\232\181\155\239\188\159")
  local szText2 = i18n.EasyGet("\230\156\172\230\151\165\229\143\175\229\143\130\228\184\142\230\172\161\230\149\176 <color=red>{0}</color>", nBoatRaceJoinCount)
  Ui:OpenWindow("UIMessageBoxItem", DragonBoatRace.Cfg.nDragonBoatTicketId, nTicketCount, szText1, szText2, szText3, {
    {funcOK},
    {funcCancel}
  })
end

function tbActivity:CheckTicketEnable()
  local nTicketJoinCount = DegreeCtrl:GetDegree(me, DragonBoatRace.szDegreeName)
  local nTicketCount = me.GetItemCountInBags(DragonBoatRace.Cfg.nDragonBoatTicketId)
  if nTicketJoinCount <= 0 then
    if nTicketCount <= 0 then
      self:NotEnoughJoinCountAndNoTicket()
      return false
    elseif nTicketCount > 0 then
      self:NotEnoughJoinCountButHaveTicket()
      return false
    end
    return false
  end
  return true
end

  })
end

function tbActivity:NotEnoughJoinCountButHaveTicket()
  RemoteServer.DragonBoatRaceRequest("AddDragonBoatTicketTime", true)
end

function tbActivity:CheckRaceIsOpen()
  local tbOpenTimeInfo = ActivityMgr.DragonBoatActivityCfg.RaceTime
  local nCurTime = GetTime() + GetZoneTimeSecDiff()
  local nZeroTime = Lib:GetTodayZeroHour(GetTime())
  local nCurSecondTime = nCurTime - nZeroTime
  if nCurSecondTime >= tbOpenTimeInfo.nStartTimePerDay1 and nCurSecondTime < tbOpenTimeInfo.nEndTimePerDay1 then
    return true, 0
  end
  if nCurSecondTime >= tbOpenTimeInfo.nStartTimePerDay2 and nCurSecondTime < tbOpenTimeInfo.nEndTimePerDay2 then
    return true, 0
  end
  if nCurSecondTime >= tbOpenTimeInfo.nStartTimePerDay3 and nCurSecondTime < tbOpenTimeInfo.nEndTimePerDay3 then
    return true, 0
  end
  return false, 2
end
