Require("CommonScript/Activity/ActivityMgrDef.lua")
Require("CommonScript/Activity/AddMonster/AddMonsterComm.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.AddMonster)
local Protocol = {
  [tbActivity.Protocol.OnSyncNpcState] = "OnSyncNpcState",
  [tbActivity.Protocol.ReqLightRedPoint] = "OnLightRedPointReq"
}

function tbActivity:OnOpen()
  self.bIsNpcAppear = false
end

function tbActivity:OnClose()
end

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("[LaborHonor] ServerRequest protocol is wrong", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnSyncNpcState(bIsNpcAppear)
  if nil ~= bIsNpcAppear then
    self.bIsNpcAppear = bIsNpcAppear
  end
end

function tbActivity:IsShowOpen()
  return self.bIsNpcAppear
end

function tbActivity:OnLightRedPointReq(bNeedLight)
  if self.tbConfig.szRedPoint == nil or self.tbConfig.szRedPoint == "" then
    return
  end
  if bNeedLight then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
  end
end

function tbActivity:GetJumpMapList()
  local szTimeFrame = self:GetActivityMaxTimeFrame()
  local tbJumpMapList = {}
  for nMapTemplateId, tbPosData in pairs(self.tbConfig.tbMapPos) do
    local nCount = 1
    for k, v in ipairs(tbPosData) do
      local tbInfo = {}
      tbInfo.nMapId = nMapTemplateId
      tbInfo.nPosX = v.nPosX
      tbInfo.nPosY = v.nPosY
      tbInfo.nNpcTemplateId = self.tbConfig.tbNpcFrame[szTimeFrame].nTemplateId
      tbInfo.szMapName = i18n.Parse(self.tbConfig.tbMapNameList[nMapTemplateId][nCount])
      table.insert(tbJumpMapList, tbInfo)
      nCount = nCount + 1
    end
  end
  return tbJumpMapList
end

function tbActivity:DoCommonJump(szJumpType, tbParam)
  if tbParam and #tbParam > 0 and ActivityMgr:IsJumpTypeExist(szJumpType) then
    ActivityMgr:DoCommonJump(szJumpType, table.unpack(tbParam))
  end
end
