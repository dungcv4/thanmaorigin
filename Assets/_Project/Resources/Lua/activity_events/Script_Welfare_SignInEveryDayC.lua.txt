function SignInEveryDay:CheckIsOpen(pPlayer)
  return true
end

function SignInEveryDay:UpdateUI()
  self:UpdateReddotState()
  if Ui:WindowVisible("UIWelfare") == true then
    Ui:GetWindow("UIWelfare"):UpdateSignInEveryDayPanel()
  end
end

function SignInEveryDay:ReportLogToKTO(tbArgs, tbCleverTapArgs)
  Sdk:RecordAppsflyerEvent("af_daily_login_reward", tbArgs)
  Sdk:RecordFirebaseEvent("daily_login_reward", tbArgs)
  Sdk:RecordCleverTapEvent("daily_login_reward", tbCleverTapArgs)
end

function SignInEveryDay:ReportLogToKTOV2(tbArgs)
  Sdk:RecordAppsflyerEvent("grant_daily_reward", tbArgs)
  Sdk:RecordFirebaseEvent("grant_daily_reward", tbArgs)
  Sdk:RecordCleverTapEvent("grant_daily_reward", tbArgs)
end

function SignInEveryDay:ReqSignIn()
  RemoteServer.RequestSignInEveryDay("ApplySignIn")
end

function SignInEveryDay:ReqReSignIn()
  RemoteServer.RequestSignInEveryDay("ApplyReSignIn")
end

function SignInEveryDay:ReqGetStageAward(nIndex)
  RemoteServer.RequestSignInEveryDay("ApplyGetStageAward", nIndex)
end

function SignInEveryDay:ReqGetConsecutiveAward()
  RemoteServer.RequestSignInEveryDay("ApplyGetConsecutiveAward")
end

function SignInEveryDay:UpdateReddotState()
  local bRet1 = self:CheckSignUp(me)
  local bRet2 = self:CheckReSignIn(me)
  local bRet3 = false
  for nIndex, tbAward in ipairs(self.tbStageAward) do
    local bRetIdx = self:CheckStageAward(me, nIndex)
    bRet3 = bRet3 or bRetIdx
  end
  local bRet4 = self:CheckConsecutiveAward(me)
  if bRet1 or bRet2 or bRet3 or bRet4 then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.SignInEveryDay)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.SignInEveryDay)
  end
end

function SignInEveryDay:OnLogin()
  self:UpdateReddotState()
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_CROSS_DAYS, SignInEveryDay.UpdateReddotState, SignInEveryDay)
