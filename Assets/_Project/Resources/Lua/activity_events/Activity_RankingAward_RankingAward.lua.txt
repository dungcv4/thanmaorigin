Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.RankingAward)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/RankingAward/RankingAwardCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/RankingAward/RankingAward.tab"
  local tbRankingAward = Lib:EasyLoadTabFile(szFilepath)
  if not tbRankingAward then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  for _, tbInfo in pairs(tbRankingAward) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("RankingAward: LoadRankingAwardConfig failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRankingAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRankingAward or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbRankingAward, {
      nRankStart = tbInfo.nRankStart,
      nRankEnd = tbInfo.nRankEnd,
      nAwardKey = tbInfo.nAwardKey
    })
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szMailFrom = i18n.Encode(self.tbConfig.szMailFrom)
  self.tbConfig.szMailText = i18n.Encode(self.tbConfig.szMailText)
  self.tbConfig.szTitle = i18n.Encode(self.tbConfig.szTitle)
  self.tbConfig.szScoreTitle = i18n.Encode(self.tbConfig.szScoreTitle)
  self.tbConfig.szScoreName = i18n.Encode(self.tbConfig.szScoreName)
  self.tbConfig.szSubValueTitle = self.tbConfig.szSubValueTitle and i18n.Encode(self.tbConfig.szSubValueTitle)
  for _, tbInfo in ipairs(tbActCfg.RankingAward) do
    self.tbConfig.tbRankingAward = self.tbConfig.tbRankingAward or {}
    table.insert(self.tbConfig.tbRankingAward, tbInfo)
    if not tbInfo.nShowAwardKey or tbInfo.nShowAwardKey <= 0 then
      tbInfo.nShowAwardKey = tbInfo.nAwardKey
    end
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  local tbConfig = self:GetConfig()
  if Lib:IsEmptyStr(tbConfig.szLogWay) then
    LogErr("[RankingAward:CheckConfigValid] bad logway! please check it! szLogWay=", tbConfig.szLogWay, self:GetId())
    return false
  end
  for _, tbInfo in ipairs(self.tbConfig.tbRankingAward) do
    if tbInfo.nAwardKey and tbInfo.nAwardKey > 0 then
      if Lib:IsEmptyTB(Award:GetAward(self.tbConfig.nAwardId, tbInfo.nAwardKey)) then
        LogErr("[RankingAward:CheckConfigValid] bad award! please check it! awardid, awardkey=", self.tbConfig.nAwardId, tbInfo.nAwardKey)
        return false
      end
    elseif tbInfo.nShowAwardKey and tbInfo.nShowAwardKey > 0 then
      if Lib:IsEmptyTB(Award:GetAward(self.tbConfig.nAwardId, tbInfo.nShowAwardKey)) then
        LogErr("[RankingAward:CheckConfigValid] bad award! please check it! awardid, showAwardkey=", self.tbConfig.nAwardId, tbInfo.nShowAwardKey)
        return false
      end
    else
      LogErr("[RankingAward:CheckConfigValid] bad award! please check it! awardid, awardkey, showAwardkey=", self.tbConfig.nAwardId, tbInfo.nAwardKey, tbInfo.nShowAwardKey)
    end
  end
  return true
end

function tbActivity:GetSaveKeyFieldName()
  return ActivityMgr.NoSaveKey
end

function tbActivity:GetScriptDataFieldName()
  return "szScriptDataKey"
end

tbActivity.DataSegment = {AllRank = 1}
