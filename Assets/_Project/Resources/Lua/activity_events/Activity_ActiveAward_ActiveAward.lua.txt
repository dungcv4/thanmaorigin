Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ActiveAward)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/ActiveAward/ActiveAwardCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/ActiveAward/ActiveAward.tab"
  local tbActiveAwardConfig = Lib:EasyLoadTabFile(szFilepath)
  if not tbActiveAwardConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  for _, tbInfo in pairs(tbActiveAwardConfig) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("ActiveAward: LoadActiveAwardConfig failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActiveAwardConfig = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActiveAwardConfig or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActiveAwardConfig, tbInfo)
  end
  for _, tbConfig in pairs(self.ACTIVITY_CONFIG) do
    table.sort(tbConfig.tbActiveAwardConfig, function(tb1, tb2)
      return tb1.nActDegree < tb2.nActDegree
    end)
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.tbActiveAwardConfig = tbActCfg.AwardCfg
  table.sort(self.tbConfig.tbActiveAwardConfig, function(tb1, tb2)
    return tb1.nActDegree < tb2.nActDegree
  end)
  self.tbConfig.szMailFrom = i18n.Encode(self.tbConfig.szMailFrom)
  self.tbConfig.szMailTitle = i18n.Encode(self.tbConfig.szMailTitle)
  self.tbConfig.szMailTextNotFull = i18n.Encode(self.tbConfig.szMailTextNotFull)
  self.tbConfig.szMailFull = i18n.Encode(self.tbConfig.szMailFull)
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  return true
end

function tbActivity:GetLogWayFieldName()
  return "szLogName"
end

tbActivity.SaveID = {HistroryLevelS = 1, LastJoinTimeS = 2}
