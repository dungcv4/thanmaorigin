Require("CommonScript/Activity/ActivityMgrDef.lua")
Require("CommonScript/Activity/ZhongQiuActivity/ZhongQiuHuaDengActivityComm.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ZhongQiuHuaDeng)
local Protocol = {
  [tbActivity.Protocol.OnSyncQuestionId] = "OnSyncQuestionId",
  [tbActivity.Protocol.ReqLightRedPoint] = "OnLightRedPointReq"
}

function tbActivity:OnClose()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshChangE)
end

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("[LaborHonor] ServerRequest protocol is wrong", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnSyncQuestionId(nQuestionId, nNpcIndex)
  self.nNpcIndex = nNpcIndex
  Ui:OpenWindow("UIKinFlagAnswer", true, nQuestionId, self.nId)
end

function tbActivity:GetQuestionData(nQuestionId)
  if self.tbConfig.tbQuestion[nQuestionId] ~= nil then
    return self.tbConfig.tbQuestion[nQuestionId]
  end
  return nil
end

function tbActivity:ReqAnswerQuestion(nAnswerId)
  ActivityMgr:CallServerScriptV2(self.nId, tbActivity.Protocol.ReqAnswerQuestion, nAnswerId, self.nNpcIndex)
end

function tbActivity:OnLightRedPointReq(bNeedLight)
  if bNeedLight then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshChangE)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshChangE)
  end
end
