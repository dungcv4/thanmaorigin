Require("CommonScript/Activity/ActivityMgr.lua")
ActivityMgr.szOBFireWorkClassName = "OBFireWork"
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szOBFireWorkClassName)
tbActivity.LastStartTimeKey = 1
tbActivity.ActivePointFourKey = 2
tbActivity.ActivePointFiveKey = 3
tbActivity.FieldPracticeKey = 4
tbActivity.FireWorkUsingKey = 5
tbActivity.FireWorkAwardTop1Key = 6
tbActivity.FireWorkAwardTop2Key = 7
tbActivity.FireWorkAwardTop3Key = 8

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/OBFireWork/OBFireWork.tab"
  local tbField = {
    "nConfigId",
    "szSaveKey",
    "nTop",
    "nVip",
    "szMailTitle",
    "szMailMsg",
    "szMailFrom",
    "szLogWay",
    "szAwardDays",
    "szDesc",
    "szName",
    "nLevel",
    "nShowAwardId",
    "nShowAwardKey",
    "nFireWorkId",
    "szShowTime"
  }
  local tbActivityConfig = Lib:LoadTabFileEx(szFilepath, "dsddsssssssdddds", "nConfigId", tbField)
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig or {}) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
    local tbAwardDays = Lib:SplitStr2Num(tbConfig.szAwardDays)
    self.ACTIVITY_CONFIG[nConfigId].tbAwardDays = tbAwardDays
  end
  szFilepath = "Setting/Activity/OBFireWork/OBFireWorkActivePoint.tab"
  tbField = {
    "nConfigId",
    "nActivePointIndex",
    "nAwardId",
    "nAwardKey",
    "nMaxTime"
  }
  local tbActivePoint = Lib:LoadTabFileEx(szFilepath, "ddddd", nil, tbField)
  for _, tbInfo in ipairs(tbActivePoint or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("OBFireWork:LoadConfig OBFireWorkActivePoint nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActivePoint = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActivePoint or {}
    assert(not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActivePoint[tbInfo.nActivePointIndex], string.format("OBFireWork:LoadConfig OBFireWorkActivePoint nActivePointIndex=%d repetition", tbInfo.nActivePointIndex))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbActivePoint[tbInfo.nActivePointIndex] = tbInfo
  end
  szFilepath = "Setting/Activity/OBFireWork/OBFireWorkFieldPractice.tab"
  tbField = {
    "nConfigId",
    "nAwardId",
    "nAwardKey",
    "nMaxTime",
    "nRatio"
  }
  local tbFieldPractice = Lib:LoadTabFileEx(szFilepath, "ddddd", nil, tbField)
  for _, tbInfo in ipairs(tbFieldPractice or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("OBFireWork:LoadConfig OBFireWorkFieldPractice nConfigId=%d not exist", tbInfo.nConfigId))
    assert(not self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFieldPractice, string.format("OBFireWork:LoadConfig OBFireWorkFieldPractice repetition", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbFieldPractice = tbInfo
  end
  szFilepath = "Setting/Activity/OBFireWork/OBFireWorkAward.tab"
  tbField = {
    "nConfigId",
    "nTop",
    "nNum",
    "nAwardId",
    "nAwardKey",
    "nDay",
    "szNoticeMsg"
  }
  local tbAward = Lib:LoadTabFileEx(szFilepath, "dddddds", nil, tbField)
  local tbMulti = {}
  for _, tbInfo in ipairs(tbAward or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("OBFireWork:LoadConfig OBFireWorkAward nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward[tbInfo.nDay] = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward[tbInfo.nDay] or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAward[tbInfo.nDay], tbInfo)
  end
  for _, tbInfo in pairs(self.ACTIVITY_CONFIG) do
    for _, tbList in pairs(tbInfo.tbAward) do
      table.sort(tbList, function(tbFirst, tbSecond)
        return tbFirst.nTop < tbSecond.nTop
      end)
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end
