local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.YangDaoDaHui)
local Protocol = {
  [tbActivity.Protocol.OnSyncActivityState] = "OnSyncActivityState"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("AchieveAwardC: ServerRequest failed, invalid Protocol = ", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnClose()
  self.tbLodeBoss = nil
end

function tbActivity:GoToFightBoss()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.YangDaoDaHuiFightBoss)
  self:OnProceed()
end

function tbActivity:OnProceed()
  if not self.bInProgress then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return
  end
  if self.tbLodeBoss and not Lib:IsEmptyTB(self.tbLodeBoss) then
    local tbBossInfo
    local tbBossList = {}
    for nNpcId, tbInfo in pairs(self.tbLodeBoss) do
      if me.nMapId == tbInfo.nMapId then
        tbBossInfo = tbInfo
      end
      table.insert(tbBossList, tbInfo)
    end
    if not tbBossInfo then
      Lib:SmashTable(tbBossList)
      tbBossInfo = tbBossList[1]
    end
    if not tbBossInfo then
      return
    end
    TeamMgr:StopFollowMember()
    AFK:Stop()
    AutoPath:GotoAndCallEx(tbBossInfo.nMapId, tbBossInfo.nPosX, tbBossInfo.nPosY, function()
      if me.nFightMode == Npc.FIGHT_MODE.emFightMode_Fight then
        AFK:Start()
      end
    end)
  else
    local nGoToMapId
    local tbMapId = {}
    for nMapId, _ in pairs(self.tbConfig.tbMapBossPosInfo) do
      if nMapId == me.nMapId then
        me.CenterMsg(i18n.EasyGet("\229\183\178\229\156\168\230\180\187\229\138\168\229\156\176\229\155\190\228\184\173"))
        return
      end
      table.insert(tbMapId, nMapId)
    end
    if not nGoToMapId then
      Lib:SmashTable(tbMapId)
      nGoToMapId = tbMapId[1]
    end
    if not nGoToMapId then
      return
    end
    AutoPath:GotoAndCallEx(nGoToMapId)
  end
  if Ui:IsWindowLogicOpen("UIAccess_Tip1") then
    Ui:CloseWindow("UIAccess_Tip1")
  end
  if Ui:IsWindowLogicOpen("UIBaDaoActivity") then
    Ui:CloseWindow("UIBaDaoActivity")
  end
end

function tbActivity:OnSyncActivityState(bInProgress, tbLodeBoss)
  local bLightRedPoint = false
  if not self.bInProgress and bInProgress then
    bLightRedPoint = true
  end
  self.bInProgress = bInProgress
  self.tbLodeBoss = tbLodeBoss
  if bLightRedPoint then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.YangDaoDaHuiFightBoss)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.YangDaoDaHuiFightBoss)
  end
end
