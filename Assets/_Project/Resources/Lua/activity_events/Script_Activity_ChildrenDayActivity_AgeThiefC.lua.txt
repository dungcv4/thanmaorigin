Require("CommonScript/Activity/ChildrenActivity/PreTask/AgeThief.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.AgeThief)

function tbActivity:OnOpen()
  self.nSaveId = GetGroupKey(self.tbConfig.szSaveKey)
end

function tbActivity:OnClose()
  self.nSaveId = nil
end

function tbActivity:GetMapShowInfo()
  local tbList = {}
  local tbHash = {}
  for nMapId, tbInfo in pairs(self.tbConfig.tbAgeThiefConfig) do
    if not tbHash[nMapId] and tbInfo.szTimeFrame then
      local tbTimeFrame = Lib:SplitStr(tbInfo.szTimeFrame)
      for _, szTimeFrame in ipairs(tbTimeFrame) do
        if self:CheckIsInTimeFrame(szTimeFrame) then
          local tbMapDefaultPosInfo = Map.tbMapListEx[nMapId].tbDefaultPos
          if tbMapDefaultPosInfo then
            table.insert(tbList, {
              nMapTemplateId = nMapId,
              nPosX = tbMapDefaultPosInfo[1].PosX,
              nPosY = tbMapDefaultPosInfo[1].PosY,
              szName = Map:GetMapName(nMapId)
            })
            tbHash[nMapId] = true
          end
        end
      end
    end
  end
  return tbList
end
