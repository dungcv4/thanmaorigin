Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.CaiHuaDaoZei)
tbActivity.Protocol = {OnSyncNpcState = 1, ReqLightRedPoint = 2}

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Activity/WomenDayActivity/CaiHuaDaoZeiActivity.tab"
  local tbField = {
    "nConfigId",
    "szSaveKey",
    "szScriptTableKey",
    "nJoinLevel",
    "nNpcTemplateID",
    "nMaxNpcNumByMap",
    "szGradeNum",
    "nAwardId",
    "szGradeAwardKey",
    "nKillerAwardKey"
  }
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "dssdddsdsd", "nConfigId", tbField)
  local szFilePath = "Setting/Activity/WomenDayActivity/NpcPosList.tab"
  local tbField = {
    "nConfigId",
    "nMapTemplateId",
    "nPosX",
    "nPosY"
  }
  local tbNpcPosCfg = Lib:LoadTabFileEx(szFilePath, "dddd", nil, tbField)
  local tbPosTable = {}
  for _, tbInfo in ipairs(tbNpcPosCfg) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("CaiHuaDaoZeiActivity:LoadConfig NpcPosList nConfigId=%d not exist", tbInfo.nConfigId))
    if tbPosTable[tbInfo.nConfigId] == nil then
      tbPosTable[tbInfo.nConfigId] = {}
    end
    if tbPosTable[tbInfo.nConfigId][tbInfo.nMapTemplateId] == nil then
      tbPosTable[tbInfo.nConfigId][tbInfo.nMapTemplateId] = {}
    end
    local tbPosData = {}
    tbPosData.nPosX = tbInfo.nPosX
    tbPosData.nPosY = tbInfo.nPosY
    table.insert(tbPosTable[tbInfo.nConfigId][tbInfo.nMapTemplateId], tbPosData)
  end
  for nConfigId, tbMapPos in pairs(tbPosTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbMapPos = tbMapPos
    end
  end
  local szFilePath = "Setting/Activity/WomenDayActivity/NpcFrameList.tab"
  local tbField = {
    "nConfigId",
    "szTimeFrame",
    "nNpcTemplateID"
  }
  local tbNpcFrameCfg = Lib:LoadTabFileEx(szFilePath, "dsd", nil, tbField)
  local tbNpcFrameTable = {}
  for _, tbInfo in ipairs(tbNpcFrameCfg) do
    if tbNpcFrameTable[tbInfo.nConfigId] == nil then
      tbNpcFrameTable[tbInfo.nConfigId] = {}
    end
    tbNpcFrameTable[tbInfo.nConfigId][tbInfo.szTimeFrame] = tbInfo.nNpcTemplateID
  end
  for nConfigId, tbNpcFrame in pairs(tbNpcFrameTable) do
    if self.ACTIVITY_CONFIG[nConfigId] ~= nil then
      self.ACTIVITY_CONFIG[nConfigId].tbNpcFrame = tbNpcFrame
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:GetMapList()
  if self.ACTIVITY_CONFIG[self.nConfigId].tbMapPos ~= nil then
    local tbMap = {}
    local tbMapList = {}
    for nMapId, tbPosDataList in pairs(self.ACTIVITY_CONFIG[self.nConfigId].tbMapPos) do
      if nil == tbMap[nMapId] then
        tbMap[nMapId] = 1
        table.insert(tbMapList, nMapId)
      end
    end
    return tbMapList
  end
  return {}
end
