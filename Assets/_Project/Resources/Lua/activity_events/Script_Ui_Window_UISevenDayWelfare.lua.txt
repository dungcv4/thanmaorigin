function tbWnd:OnCreate()

end

function tbWnd:RefreshData()
  SevenDayWelfare:RefreshGetAwardData()
end

function tbWnd:UpdateView()
  SevenDayWelfare:OnCheckRedPoint()
  local nGetAwardNum = SevenDayWelfare:GetCanGetAwardNum()
  for nDay = 1, SevenDayWelfare.DAY do
    local hasGetAward = 1 == SevenDayWelfare.tbWelfareGetData[nDay]
    local btnGetPath = string.format("DayList/viewport/Contant/Day%d/btnGet", nDay)
    local imgComplete = string.format("DayList/viewport/Contant/Day%d/imgComplete", nDay)
    self.pPanel:SetActive(btnGetPath, not hasGetAward)
    self.pPanel:SetActive(imgComplete, hasGetAward)
    local canGetAward = nDay <= nGetAwardNum
    self.pPanel:Button_SetEnable(btnGetPath, canGetAward and SevenDayWelfare:BindRoleIsMe())
  end
end

function tbWnd:ShowAwardInfo()
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.ItemPath = self.ItemPath or {}
  for nDay = 1, SevenDayWelfare.DAY do
    self.tbEffect[nDay] = self.tbEffect[nDay] or {}
    self.ItemPath[nDay] = self.ItemPath[nDay] or {}
    self.ItemPath[nDay] = string.format(tbControls.UIAwardItem, nDay)
    local tbItemScrollView = Ui:NewScrollView(self, self.ItemPath[nDay])
    tbItemScrollView:Clear()
    local tbTodayWelfare = Award:GetAward(SevenDayWelfare.AWARDID, nDay)
    for i, tbAward in ipairs(tbTodayWelfare) do
      tbItemScrollView:SetElement(i, function(tbElement)
        self.tbEffect[nDay][i] = self.tbEffect[nDay][i] or {}
        local szType = tbAward[1]
        local tbItemGrid = tbElement.tbItemGrid
        if not tbItemGrid then
          tbItemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
          tbItemGrid:SetClickEnable(true)
        end
        if szType == Player.AwardType2Name[Player.award_type_item] then
          tbItemGrid:SetItemByTemplate(tbAward[2], tbAward[3])
        else
          tbItemGrid:SetDigitalItem(szType, tbAward[2])
        end
        local szEffectPath = "UI/Effect/UIItemGrid/Enhance_10.prefab"
        if not self.tbEffect[nDay][i].szkey then
          self.tbEffect[nDay][i].szkey = tbElement.pPanel:DynamicLoadGameObject("UIItemGrid/Enhance", szEffectPath, "effect")
          tbElement.pPanel:SetSortingOrder(self.tbEffect[nDay][i].szkey, 2)
        end
      end)
    end
  end
end
