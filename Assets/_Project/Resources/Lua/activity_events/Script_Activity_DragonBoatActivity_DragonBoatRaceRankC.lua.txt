local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.DragonBoatRace)

function tbActivity:C2SRequestPersonalRankData(nPageIndex)
  ActivityMgr:CallServerScript(self.nId, self.tbC2SProtocol.RequestPersonalRankData, nPageIndex)
end

function tbActivity:C2SRequestKinRankData(nPageIndex)
  ActivityMgr:CallServerScript(self.nId, self.tbC2SProtocol.RequestKinRankData, nPageIndex)
end

function tbActivity:ServerRequest(nProtocol, ...)
  if nProtocol == self.tbC2SProtocol.RequestPersonalRankData then
    self:OnS2CSyncPersonalRankData(...)
  elseif nProtocol == self.tbC2SProtocol.RequestKinRankData then
    self:OnS2CSyncKinRankData(...)
  end
end

function tbActivity:OnS2CSyncPersonalRankData(tbData)
  local tbWnd = Ui:GetWindow("UIRankingDuanWu")
  if nil ~= tbWnd then
    tbWnd:OnSyncPersonalRank(tbData)
  end
end

function tbActivity:OnS2CSyncKinRankData(tbData)
  local tbWnd = Ui:GetWindow("UIRankingDuanWu")
  if nil ~= tbWnd then
    tbWnd:OnSyncKinRank(tbData)
  end
end
