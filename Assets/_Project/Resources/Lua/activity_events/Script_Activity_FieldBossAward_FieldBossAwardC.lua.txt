Require("CommonScript/Activity/FieldBossAward/FieldBossAward.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.FieldBossAward)

function tbActivity:DoCommonJump()
  if not self.tbConfig.tbTips then
    return
  end
  local szMsg
  for _, tbInfo in ipairs(self.tbConfig.tbTips) do
    if Lib:IsEmptyStr(tbInfo.szTimeFrameStart) then
      if Lib:IsEmptyStr(tbInfo.szTimeFrameEnd) or 0 == GetTimeFrameState(tbInfo.szTimeFrameEnd) then
        szMsg = tbInfo.szMsg
        break
      end
    elseif 1 == GetTimeFrameState(tbInfo.szTimeFrameStart) and (Lib:IsEmptyStr(tbInfo.szTimeFrameEnd) or 0 == GetTimeFrameState(tbInfo.szTimeFrameEnd)) then
      szMsg = tbInfo.szMsg
      break
    end
  end
  if not szMsg then
    return
  end
  me.MsgBox(i18n.Parse(szMsg), {
    {
      i18n.szOk
    }
  })
end

function tbActivity:GetRedPointEventList()
  return {}
end
