Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.EmeiActivity)
tbActivity.nSaveGroup = GetGroupKey("EmeiActivitySendEndMail")
tbActivity.nSendEndMailKeyId = 1
local nDefaultConfigId = 1

function tbActivity:LoadConfig()
  if not self.bLoadConfig then
    self.ACTIVITY_CONFIG = {
      [nDefaultConfigId] = {}
    }
    self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
    self.bLoadConfig = true
  end
end
