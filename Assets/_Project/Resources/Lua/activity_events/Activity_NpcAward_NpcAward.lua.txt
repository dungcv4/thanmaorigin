Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.NpcAward)
tbActivity.SaveID = {
  LastStartTimeKey = 1,
  ReceivedVersionKey = 2,
  TodayNumKey = 3
}
tbActivity.Protocol = {
  SyncRedPointStatus = 1,
  OnSyncNpcPosition = 2,
  GoToNpcMap = 3,
  OnSyncNpcMap = 4
}

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Activity/ChildrenActivity/NpcAward/NpcAward.tab"
  local tbField = {
    "nConfigId",
    "szSaveKey",
    "nJoinLevel",
    "nNpcTemplateId",
    "nNpcNum",
    "nMaxNum",
    "szBeginNotice",
    "szEndNotice",
    "nAliveTime",
    "szDialogText",
    "szLogWay",
    "szName",
    "szDesc",
    "nShowAwardId",
    "nShowAwardKey",
    "szShowTime",
    "szNotifyType",
    "nNotifyTimeout",
    "szNotifyContent"
  }
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "dsddddssdssssddssds", "nConfigId", tbField)
  szFilePath = "Setting/Activity/ChildrenActivity/NpcAward/AwardList.tab"
  tbField = {
    "nConfigId",
    "nIndex",
    "nAwardId",
    "nAwardKey",
    "nNum"
  }
  local tbEventAward = Lib:LoadTabFileEx(szFilePath, "ddddd", nil, tbField)
  for _, tbInfo in ipairs(tbEventAward or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("NpcAward:LoadConfig AwardList nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAwardList = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAwardList or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAwardList, tbInfo)
  end
  for _, tbInfo in pairs(self.ACTIVITY_CONFIG) do
    table.sort(tbInfo.tbAwardList, function(tbFirst, tbSecond)
      return tbFirst.nIndex < tbSecond.nIndex
    end)
  end
  szFilePath = "Setting/Activity/ChildrenActivity/NpcAward/PosList.tab"
  tbField = {
    "nConfigId",
    "nMapTemplateId",
    "nPosX",
    "nPosY"
  }
  local tbPosList = Lib:LoadTabFileEx(szFilePath, "dddd", nil, tbField)
  for _, tbInfo in ipairs(tbPosList) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("NpcAward:LoadConfig PosList nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPosList = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPosList or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPosList, tbInfo)
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szBeginNotice = i18n.Encode(self.tbConfig.szBeginNotice)
  self.tbConfig.szEndNotice = i18n.Encode(self.tbConfig.szEndNotice)
  self.tbConfig.szDialogText = i18n.Encode(self.tbConfig.szDialogText)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.szNotifyContent = i18n.Encode(self.tbConfig.szNotifyContent)
  if self.tbConfig.szNumCellText then
    self.tbConfig.szNumCellText = i18n.Encode(self.tbConfig.szNumCellText)
  end
  if self.tbConfig.szSmallPanelText then
    self.tbConfig.szSmallPanelText = i18n.Encode(self.tbConfig.szSmallPanelText)
  end
  self.tbConfig.tbAwardList = tbActCfg.AwardList
  table.sort(self.tbConfig.tbAwardList, function(tbFirst, tbSecond)
    return tbFirst.nIndex < tbSecond.nIndex
  end)
  self.tbConfig.tbPosFileList = {}
  for _, tbInfo in ipairs(tbActCfg.PosList) do
    self.tbConfig.tbPosFileList[tbInfo.nMapTemplateId] = tbInfo.szPosFilePath
  end
  if tbActCfg.tbMapList then
    self.tbConfig.tbMapList = {}
    for nIndex, tbInfo in ipairs(tbActCfg.tbMapList) do
      self.tbConfig.tbMapList[nIndex] = tbInfo.nMapId
    end
  end
end

function tbActivity:CheckConfigValid(tbConfig)
  return true
end

function tbActivity:CanJoin(nLevel)
  return nLevel >= self.tbConfig.nJoinLevel
end

function tbActivity:GetTodayAwardTime(pPlayer)
  return pPlayer.GetUserValue(self.SaveKey, self.SaveID.TodayNumKey)
end

function tbActivity:GetTodayMaxAwardTime()
  return self.tbConfig.nMaxNum
end

function tbActivity:InitExtraData(tbScheduleCfg)
  self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
end
