Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.NewTimeFrameReminder)
tbActivity.SaveKey = GetGroupKey("NewTimeFrameReminder")
tbActivity.SaveID = {ReminderIndexID = 1}

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Activity/NewTimeFrameReminder/NewTimeFrameReminder.tab"
  local tbField = {
    "nConfigId",
    "szMFrom",
    "szMTitle",
    "nAwardId",
    "nAwardKey"
  }
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "dssdd", "nConfigId", tbField)
  szFilePath = "Setting/Activity/NewTimeFrameReminder/Reminder.tab"
  tbField = {
    "nConfigId",
    "nIndex",
    "szNewTimeFrame",
    "szTimeFrameStart",
    "szTimeFrameEnd",
    "szDesc",
    "szMContent",
    "szShowTitle"
  }
  local tbReminder = Lib:LoadTabFileEx(szFilePath, "ddssssss", nil, tbField)
  for _, tbInfo in ipairs(tbReminder or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("NewTimeFrameReminder:LoadConfig Reminder nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbReminder = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbReminder or {}
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[tbInfo.nConfigId].szSaveKey) then
      self.ACTIVITY_CONFIG[tbInfo.nConfigId].szSaveKey = "NewTimeFrameReminder"
    end
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[tbInfo.nConfigId].szLogWay) then
      self.ACTIVITY_CONFIG[tbInfo.nConfigId].szLogWay = "LogWay_NewTimeFrameReminder"
    end
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbReminder, tbInfo)
  end
  for _, tbInfo in pairs(self.ACTIVITY_CONFIG) do
    table.sort(tbInfo.tbReminder, function(tbFirst, tbSecond)
      return tbFirst.nIndex > tbSecond.nIndex
    end)
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:GetLastReminderInfo()
  for _, tbInfo in ipairs(self.tbConfig.tbReminder) do
    if 1 == GetTimeFrameState(tbInfo.szNewTimeFrame) then
      return tbInfo
    end
  end
  return nil
end
