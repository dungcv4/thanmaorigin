function tbWnd:OnCreate()
  self:InitCfg()
  self:SetTabBtn()
  self:SetClickEmptyToClose(tbControls.wndMonthPrerogative)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
  self:SetClickEmptyToClose(self.wndOnlineRewardPreview)
  self:SetClickEmptyToClose(tbControls.wndDailyAwards)
  self:SetClickEmptyToClose(tbControls.wndFortuneRecord)
  self.pAwardPreviewScorllView = Ui:NewScrollView(self, self.Template, function(_, nIndex, tbElement)
  end, false, false)
  self.pLeftTabScrollView = Ui:NewScrollView(self, self.btnTemplate, function(self, nIdx)
  end, false, false)
  self:InitAwardPreviewScrollView()
  self:InitLeftTabScrollView()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbDailyAwards = Ui:NewItemGridView(self, tbControls.DailyAwardsItemTemp)
  self.tbRecordScrollView = Ui:NewScrollView(self, tbControls.FortuneRecordTemp)
end

function tbWnd:OnOpen(szPanelType)
  local nSelectPanel = tbWnd.tbMapPanelType2Index[szPanelType]
  if not self.tbTabControl:OnOpen(nSelectPanel) then
    return 0
  end
  if nil == nSelectPanel then
    nSelectPanel = 1
  end
  self.pLeftTabScrollView:ScrollToIndex(nSelectPanel)
  self.pPanel:SetActive(tbControls.wndMonthPrerogative, false)
  self:SetPrivilegeDesc()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnOnHook, {
    RedPointMgr.EVENT.OfflineExp
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnSupplement, {
    RedPointMgr.EVENT.SupplementAward
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btn7daysSignInAward, {
    RedPointMgr.EVENT.SevenDayWelfare
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnMoneyTree, {
    RedPointMgr.EVENT.MoneyTreeFree
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnTimeLimit, LimitTimeLevelUp.tbIndex2RedPoint)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnOnlineReward, {
    RedPointMgr.EVENT.OnlineAward
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnMonthWeek, {
    RedPointMgr.EVENT.WeekMonthCard
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGrowthFund, {
    RedPointMgr.EVENT.GrowUpInvest
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnDailyGift, {
    RedPointMgr.EVENT.DailyGift
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnDailyAttendance, {
    RedPointMgr.EVENT.SignInEveryDay
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnYuanBaoReturn, {
    RedPointMgr.EVENT.YuanBaoReturn
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnCheapMonthGift, {
    RedPointMgr.EVENT.CheapMonthGift
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnReturnAward, {
    RedPointMgr.EVENT.ReturnerDailyFund,
    RedPointMgr.EVENT.ReturnerDailySignIn,
    RedPointMgr.EVENT.ReturnerFirstTime
  })
  self.tbMoneyPanel:OnOpen()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:InitAwardPreviewScrollView()
  self.pAwardPreviewScorllView:Clear()
  for nStageId, onlineAwardCfgData in pairs(OnlineAward.tbOnlineAwardCfg) do
    self.pAwardPreviewScorllView:SetElement(nStageId, function(tbElement)
      self:SetElementInfo(tbElement, onlineAwardCfgData)
    end)
  end
end

function tbWnd:InitLeftTabScrollView()
  self.pLeftTabScrollView:Clear()
  for nIdx, tbInfo in ipairs(tbWnd.tbPanelCfg) do
    self.pLeftTabScrollView:SetElement(tbInfo.nIdx, function(tbElement)
      tbElement.pPanel:Label_SetText("Text", i18n.Parse(tbInfo.szName))
      if tbInfo.szPanelType == "PANEL_TYPE_RETURN_AWARD" and FriendsReturn:CheckIsPlayerSpecialReturner(me) then
        tbElement.pPanel:Label_SetText("Text", i18n.EasyGet("\229\134\141\233\151\175\230\177\159\230\185\150"))
      end
    end)
  end
end

function tbWnd:SetElementInfo(tbElement, onlineAwardCfgData)
  local nStageId = onlineAwardCfgData.StageId
  local nTargetTimeSecond = onlineAwardCfgData.Time
  local nTargetTimeMinute = math.floor(nTargetTimeSecond / 60)
  tbElement.pPanel:Label_SetText("txtTitle", i18n.EasyGet("\233\162\134\229\143\150\230\156\172\230\151\165\229\156\168\231\186\191{0}\229\136\134\233\146\159\229\144\142\239\188\140\230\152\142\230\151\165\229\143\175\233\162\134\229\143\150", nTargetTimeMinute))
  local tbAwards = Award:GetAward(onlineAwardCfgData.AwardListId, onlineAwardCfgData.TotalAwardKey, {})
  local tbControls = {
    itemTemplate = "itemList/itemTemplate"
  }
  if tbElement.pAwardListScorllView == nil then
    tbElement.pAwardListScorllView = Ui:NewScrollView(tbElement, tbControls.itemTemplate, function(_, nIndex, tbElement)
    end, false, false)
  end
  
  local function awardFunc(tbAwardElement, awardData)
    local tbItemGrid = tbAwardElement.tbItemGrid
    tbItemGrid = tbItemGrid or Ui:NewExistItemGrid(tbAwardElement, "UIItemGrid")
    tbItemGrid:SetGenericItem(awardData)
    tbItemGrid:SetClickEnable(true)
  end
  
  tbElement.pAwardListScorllView:Clear()
  for i, awardData in pairs(tbAwards) do
    tbElement.pAwardListScorllView:SetElement(i, function(tbAwardElement)
      awardFunc(tbAwardElement, awardData)
    end)
  end
end

function tbWnd:UpdateDailyAwardsTip(nIndex)
  if not self.pPanel:IsActive(tbControls.wndDailyAwards) then
    self.pPanel:SetActive(tbControls.wndDailyAwards, true)
  end
  local tbPosition = self.pPanel:Object_GetPosition(tbControls.panelDailyAwards .. "/RightDayGroup/btnDay" .. nIndex)
  self.pPanel:Object_SetPosition(tbControls.wndDailyAwards, Ui.Vector3(tbPosition.x - 2.5, tbPosition.y + 3.5, tbPosition.z))
  local tbSetting = SignInEveryDay.tbStageAward[nIndex]
  local tbAwards = Lib:GetAwardFromString(tbSetting.szAward)
  for i = 1, #tbAwards do
    self.tbDailyAwards:SetItem(i, tbAwards[i], true)
  end
  self.tbDailyAwards:ResetFrom(#tbAwards + 1)
end

function tbWnd:ShowFortuneRecord()
  self.pPanel:SetActive(tbControls.wndFortuneRecord, true)
  local tbRecord = MoneyTree:GetClientRecord()
  self.tbRecordScrollView:Reset()
  for nIdx, szReocrd in ipairs(tbRecord) do
    self.tbRecordScrollView:SetElement(nIdx, function(tbElement)
      tbElement.pPanel:Label_SetText("", szReocrd)
    end)
  end
  self.tbRecordScrollView:GoBottom()
end
