local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ZhongQiuTuanYuan)
local Protocol = {
  [tbActivity.Protocol.ReqLightRedPoint] = "OnLightRedPointReq",
  [tbActivity.Protocol.ReqSendNotify] = "OnSynNotifyData"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    me.CenterMsg(i18n.EasyGet("\230\156\141\229\138\161\231\171\175\232\190\147\229\133\165\233\148\153\232\175\175"))
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnLightRedPointReq(bNeedLight)
  if bNeedLight then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshTable)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshTable)
  end
end

function tbActivity:OnClose()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshTable)
end

function tbActivity:GetRedPointEventId()
  return {
    RedPointMgr.EVENT.ZhongQiuRefreshTable
  }
end

function tbActivity:GetRedPointEventList()
  return {
    RedPointMgr.EVENT.ZhongQiuRefreshTable
  }
end

function tbActivity:DoCommonJump()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ZhongQiuRefreshTable)
  if not Tong:HasTong() then
    me.CenterMsg(i18n.EasyGet("\229\175\185\228\184\141\232\181\183\239\188\140\230\130\168\230\178\161\230\156\137\229\184\174\228\188\154"))
    return
  end
  if me.nMapTemplateId == Tong.Def.nMainMapTemplateId then
    me.CenterMsg(i18n.EasyGet("\230\130\168\229\183\178\231\187\143\229\156\168\229\184\174\228\188\154\229\156\176\229\155\190\228\184\173"))
    return
  end
  Tong:ReqEnterTongMap()
end

function tbActivity:OnSynNotifyData()
  Ui:SynNotifyMsg({
    szType = ActivityMgr.Def.ClassNameList.ZhongQiuTuanYuan,
    nTimeOut = GetTime() + 300,
    szTitle = i18n.Parse(self.tbConfig.szNotifyTitle),
    szContent = i18n.Parse(self.tbConfig.szNotifyContent),
    nId = self.nId
  })
end

function tbActivity:GoToCity(nMapId)
  local tbTableLocInfo = self.tbConfig.tbTableLocInfo
  for index, tbInfo in pairs(tbTableLocInfo) do
    if nMapId then
      if tbInfo.nMapTemplateId == nMapId then
        AutoPath:GotoAndCallEx(nMapId, tbInfo.PosX, tbInfo.PosY)
        break
      end
    else
      AutoPath:GotoAndCallEx(tbInfo.nMapTemplateId, tbInfo.PosX, tbInfo.PosY)
      break
    end
  end
end
