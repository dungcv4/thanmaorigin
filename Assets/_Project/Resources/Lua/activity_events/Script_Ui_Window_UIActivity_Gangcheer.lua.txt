local tbWnd = Ui:CreateWindow("UIActivity_Gangcheer")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_FIRSTTONG_SYNC_FINAL,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_FIRSTTONG_SYNC_RANK,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_FIRSTTONG_SYNC_GUESS,
      self.UpdatePanel,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    },
    {
      EventNotify.emNOTIFY_CHANGE_MONEY,
      self.OnChangeMoney,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnTipClose",
  UIMoneyPanel = "node/imgBG/imgBG/UIMoneyPanel",
  btnHelp = "node/imgBG/imgBG/txtTitle/btnHelp",
  btnChampion = "node/imgBG/node/FactionSect/btnChampion",
  btnCompetition = "node/imgBG/node/FactionSect/btnCompetition",
  ScrollView = "node/imgBG/node/Scroll View",
  ScrollView01 = "node/imgBG/node/Scroll View01",
  TemplateInfo = "node/imgBG/node/Scroll View/Viewport/Content/Info",
  TemplateInfo01 = "node/imgBG/node/Scroll View01/Viewport/Content/Info",
  txtInfo_Top = "node/imgBG/node/txtInfo_Top",
  imgIcon03 = "node/imgBG/node/txtInfo_Top/imgIcon03",
  txtGongAndDrum01 = "node/imgBG/node/txtInfo_Top/imgIcon03/txtGongAndDrum01",
  txtInfo_L = "node/imgBG/node/txtInfo_L"
}
tbWnd.tbControls = tbControls
tbWnd.PANEL_TYPE = {CHAMPION = 1, COMPETITION = 2}
tbWnd.tbPageBtnIndex = {
  [tbWnd.PANEL_TYPE.CHAMPION] = tbControls.btnCompetition,
  [tbWnd.PANEL_TYPE.COMPETITION] = tbControls.btnChampion
}
tbWnd.DEFAULT_TYPE = tbWnd.PANEL_TYPE.COMPETITION
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("TianXiaDiYiZhuWei")
  end
}

function tbWnd:OnCreate()
  self.tbPageGroup = Ui:NewButtonGroup(self, self.tbPageBtnIndex, true, function(self, nIndex)
    self:ChangePanel(nIndex)
  end)
  self.tbScrollView = Ui:NewScrollView(self, tbControls.TemplateInfo)
  self.tbScrollView01 = Ui:NewScrollView(self, tbControls.TemplateInfo01)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
end

function tbWnd:OnOpen()
  FirstTong:RequestTongRankData(false)
  FirstTong:RequestFinalStageMatch()
  FirstTong:RequestSyncGuessData()
  self.tbPageGroup:Select(self.DEFAULT_TYPE)
end

function tbWnd:OnClose()
end

function tbWnd:UpdatePanel()
  self.tbPageGroup:Select(self.nType)
end

function tbWnd:ChangePanel(nType)
  self.nType = nType
  self:OnChangeMoney()
  if self.nType == self.PANEL_TYPE.CHAMPION then
    self.pPanel:SetActive(tbControls.ScrollView, false)
    self.pPanel:SetActive(tbControls.ScrollView01, true)
    local nItemID = FirstTong.GuessCfg.nChampionGuessID
    local szItemName = Item:GetName(nItemID, me)
    local _, nIcon, _, _ = KItem.GetItemShowInfo(nItemID)
    local _, szIconPath = Item:GetIcon(nIcon)
    self.pPanel:Label_SetText(tbControls.txtInfo_L, i18n.EasyGet("\233\128\137\230\139\169\228\184\128\228\184\170\229\184\174\228\188\154\239\188\140\229\138\169\229\168\129\228\187\150\230\156\172\232\189\174\230\153\139\231\186\167"))
    self.pPanel:Sprite_SetSprite(tbControls.imgIcon03, szIconPath)
    self.pPanel:Label_SetText(tbControls.txtGongAndDrum01, i18n.EasyGet("\230\156\172\232\189\174\229\183\178\230\182\136\232\128\151{0}:", szItemName))
    local nNowPhase = FirstTong:GetNowFinalsSatge(FirstTong.tbFinalMatchData)
    local nGuessNum = 0
    if FirstTong.tbMyChampionGuessData and FirstTong.tbMyChampionGuessData[nNowPhase] then
      nGuessNum = FirstTong.tbMyChampionGuessData[nNowPhase].nGuessNum
    end
    self.pPanel:Label_SetText(tbControls.txtInfo_Top, nGuessNum)
    self:UpdateChampionScroll()
  elseif self.nType == self.PANEL_TYPE.COMPETITION then
    self.pPanel:SetActive(tbControls.ScrollView, true)
    self.pPanel:SetActive(tbControls.ScrollView01, false)
    local nItemID = FirstTong.GuessCfg.nMatchGuessID
    local szItemName = Item:GetName(nItemID, me)
    local _, nIcon, _, _ = KItem.GetItemShowInfo(nItemID)
    local _, szIconPath = Item:GetIcon(nIcon)
    self.pPanel:Label_SetText(tbControls.txtInfo_L, i18n.EasyGet("\229\138\169\229\168\129\229\175\185\233\152\181\231\154\132\229\133\182\228\184\173\228\184\128\228\184\170\229\184\174\228\188\154\232\142\183\229\190\151\232\131\156\229\136\169"))
    self.pPanel:Sprite_SetSprite(tbControls.imgIcon03, szIconPath)
    self.pPanel:Label_SetText(tbControls.txtGongAndDrum01, i18n.EasyGet("\230\156\172\232\189\174\229\183\178\230\182\136\232\128\151{0}:", szItemName))
    self.pPanel:Label_SetText(tbControls.txtInfo_Top, 999)
    local nNowPhase = FirstTong:GetNowFinalsSatge(FirstTong.tbFinalMatchData)
    local nGuessNum = 0
    if FirstTong.tbMyMatchGuessData and FirstTong.tbMyMatchGuessData[nNowPhase] then
      for _, tbGuess in pairs(FirstTong.tbMyMatchGuessData[nNowPhase]) do
        nGuessNum = nGuessNum + tbGuess.nGuessNum
      end
    end
    self.pPanel:Label_SetText(tbControls.txtInfo_Top, nGuessNum)
    self:UpdateMatchScroll()
  end
end

function tbWnd:OnChangeMoney()
  local nItemId = FirstTong.GuessCfg.nChampionGuessID
  if self.nType == self.PANEL_TYPE.COMPETITION then
    nItemId = FirstTong.GuessCfg.nMatchGuessID
  end
  local nItemCount = me.GetItemCountByTypeInBag(nItemId)
  local _, nIcon, _, _ = KItem.GetItemShowInfo(nItemId)
  local _, szPickItemIconPath = Item:GetIcon(nIcon)
  self.tbMoneyPanel:SetMoneyValue()
  self.tbMoneyPanel:SetCoinShow(false)
  self.tbMoneyPanel:SetDynamicShow(true)
  self.tbMoneyPanel:SetDynamicCurrencyInfo(nil, nItemCount, szPickItemIconPath)
  self.tbMoneyPanel:SetDynamicBtnPlusCallBack(self.OpenShop, self, nItemId)
end

function tbWnd:OpenShop(nItemId)
  Shop:OpenShopBuyWidDefault(nItemId)
end

function tbWnd:UpdateChampionScroll()
  self.tbScrollView01:Reset()
  if not FirstTong.tbFinalMatchData then
    return
  end
  local nIndex = 0
  for i = #FirstTong.GuessTimeCfg, 1, -1 do
    local tbCfg = FirstTong.GuessTimeCfg[i]
    if tbCfg and FirstTong.tbFinalMatchData[tbCfg.nPhase] then
      nIndex = nIndex + 1
      self.tbScrollView01:SetElement(nIndex, function(tbElement)
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("imgBG/txtActivityTittle", i18n.Get(tbCfg.szStageName))
        self:UpdateLeftTime(pPanel, Lib:GetConfigDate2Time(tbCfg.nEndTime))
        local bInGuessTime = false
        if GetTime() >= Lib:GetConfigDate2Time(tbCfg.nStartTime) and GetTime() < Lib:GetConfigDate2Time(tbCfg.nEndTime) then
          bInGuessTime = true
        end
        if not tbElement.tbScrollViewTong then
          tbElement.tbScrollViewTong = Ui:NewScrollView(tbElement, "TemplateBePromoted")
        end
        local tbGuessData = {}
        if FirstTong.tbMyChampionGuessData and FirstTong.tbMyChampionGuessData[tbCfg.nPhase] then
          for _, tbData in pairs(FirstTong.tbFinalMatchData[tbCfg.nPhase]) do
            if tbData.nTongId == FirstTong.tbMyChampionGuessData[tbCfg.nPhase].nTongId then
              tbGuessData = tbData
            end
          end
        end
        tbElement.tbScrollViewTong:Reset()
        tbElement.tbScrollViewTong:SetElement(1, function(tbElementMatch)
          local pPanel = tbElementMatch.pPanel
          pPanel:Label_SetText("txtName01", tbGuessData.szTongName or i18n.EasyGet("\230\154\130\230\156\170\233\128\137\230\139\169"))
          pPanel:Label_SetText("txtName02", i18n.EasyGet("\229\140\186\230\156\141\239\188\154{0}", tbGuessData.szServerName or i18n.EasyGet("\230\154\130\230\151\160")))
          pPanel:Label_SetText("txtFighting01", i18n.EasyGet("\230\142\146\229\144\141\239\188\154"))
          pPanel:Label_SetText("txtFightingNum01", self:GetTongRank(tbGuessData.nTongId))
          if bInGuessTime then
            pPanel:Button_SetEnable("BtnSelect", true)
            Ui:Button_BindEvent(pPanel, "BtnSelect", function()
              local tbTitleList = {
                [1] = i18n.EasyGet("\229\184\174\228\188\154"),
                [2] = i18n.EasyGet("\230\156\141\229\138\161\229\153\168"),
                [3] = i18n.EasyGet("\233\162\132\231\174\151\232\181\155\230\142\146\229\144\141"),
                [4] = i18n.EasyGet("\230\147\141\228\189\156")
              }
              local tbAllList = {}
              for _, tbData in pairs(FirstTong.tbFinalMatchData[tbCfg.nPhase]) do
                local bSelected = false
                table.insert(tbAllList, {
                  [1] = {
                    nId = tbData.nTongId,
                    szName = tbData.szTongName
                  },
                  [2] = tbData.szServerName,
                  [3] = self:GetTongRank(tbData.nTongId),
                  [4] = bSelected
                })
              end
              local tbCallBack = {
                FirstTong.ChangeChampionGuessTong,
                FirstTong
              }
              local szTitle = i18n.EasyGet("\229\138\169\229\168\129\229\184\174\228\188\154\233\128\137\230\139\169")
              Ui:OpenWindow("UIActivity_WuLinDaHuiGuessSelect", tbTitleList, tbAllList, tbCallBack, szTitle)
            end)
            pPanel:Button_SetEnable("btnCheer_R", true)
            Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
              if Lib:IsEmptyTB(tbGuessData) then
                me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\233\128\137\230\139\169\229\138\169\229\168\129\229\184\174\228\188\154"))
                return
              end
              local tbData = {
                {
                  nId = tbGuessData.nTongId,
                  szName = tbGuessData.szTongName
                },
                i18n.EasyGet("\233\162\132\233\128\137\232\181\155\230\142\146\229\144\141\239\188\154{0}", self:GetTongRank(tbGuessData.nTongId)),
                i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\231\165\168\230\149\176\239\188\154"),
                nil
              }
              local nItemID = FirstTong.GuessCfg.nChampionGuessID
              local tbCallBack = {
                FirstTong.ChangeChampionGuessNum,
                FirstTong
              }
              Ui:OpenWindow("UIActivity_Banghuizhengduosai_Tip", tbData, nItemID, nil, tbCallBack)
            end)
          else
            pPanel:Button_SetEnable("BtnSelect", false)
            pPanel:Button_SetEnable("btnCheer_R", false)
          end
        end)
      end)
    end
  end
end

function tbWnd:UpdateMatchScroll()
  self.tbScrollView:Reset()
  if not FirstTong.tbFinalMatchData then
    return
  end
  local nIndex = 0
  for i = #FirstTong.GuessTimeCfg, 1, -1 do
    local tbCfg = FirstTong.GuessTimeCfg[i]
    if tbCfg and FirstTong.tbFinalMatchData[tbCfg.nPhase] then
      nIndex = nIndex + 1
      self.tbScrollView:SetElement(nIndex, function(tbElement)
        local pPanel = tbElement.pPanel
        pPanel:Label_SetText("imgBG/txtActivityTittle", i18n.Get(tbCfg.szStageName))
        self:UpdateLeftTime(pPanel, Lib:GetConfigDate2Time(tbCfg.nEndTime))
        local bInGuessTime = false
        if GetTime() >= Lib:GetConfigDate2Time(tbCfg.nStartTime) and GetTime() < Lib:GetConfigDate2Time(tbCfg.nEndTime) then
          bInGuessTime = true
        end
        if not tbElement.tbScrollViewTong then
          tbElement.tbScrollViewTong = Ui:NewScrollView(tbElement, "TemplateFace")
        end
        local nElementIndex = 0
        tbElement.tbScrollViewTong:Reset()
        for nIndex = 1, tbCfg.nPhase / 2 do
          local tbMatchData = FirstTong.tbFinalMatchData[tbCfg.nPhase][nIndex]
          if tbMatchData then
            local nRightIndex = tbCfg.nPhase - nIndex + 1
            local tbLeftData = tbMatchData
            local tbRightData = FirstTong.tbFinalMatchData[tbCfg.nPhase][nRightIndex]
            nElementIndex = nElementIndex + 1
            tbElement.tbScrollViewTong:SetElement(nElementIndex, function(tbElement)
              local pPanel = tbElement.pPanel
              local tbGuessData = FirstTong.tbMyMatchGuessData and FirstTong.tbMyMatchGuessData[tbCfg.nPhase] and FirstTong.tbMyMatchGuessData[tbCfg.nPhase][nIndex]
              if tbLeftData then
                pPanel:Label_SetText("txtPlayerName01", tbLeftData.szTongName)
                pPanel:Label_SetText("txtFighting02", i18n.EasyGet("\230\142\146\229\144\141\239\188\154"))
                pPanel:Label_SetText("txtFightingNum02", self:GetTongRank(tbLeftData.nTongId))
              else
                pPanel:Label_SetText("txtPlayerName01", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
                pPanel:SetActive("btnCheer_L", false)
                pPanel:SetActive("txtFighting02", false)
                pPanel:SetActive("txtFightingNum02", false)
                pPanel:SetActive("Cont_Finish_L", false)
              end
              if tbRightData then
                pPanel:Label_SetText("txtPlayerName02", tbRightData.szTongName)
                pPanel:Label_SetText("txtFighting01", i18n.EasyGet("\230\142\146\229\144\141\239\188\154"))
                pPanel:Label_SetText("txtFightingNum01", self:GetTongRank(tbRightData.nTongId))
              else
                pPanel:Label_SetText("txtPlayerName02", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
                pPanel:SetActive("btnCheer_R", false)
                pPanel:SetActive("txtFighting01", false)
                pPanel:SetActive("txtFightingNum01", false)
                pPanel:SetActive("Cont_Finish_R", false)
              end
              if tbGuessData then
                if tbGuessData.nTongId == tbLeftData.nTongId then
                  pPanel:SetActive("btnCheer_L", false)
                  pPanel:SetActive("btnCheer_R", false)
                  pPanel:SetActive("Cont_Finish_L", true)
                  pPanel:Label_SetText("Cont_Finish_L/txtPlayerName02 (4)", tbGuessData.nGuessNum or 0)
                  pPanel:SetActive("Cont_Finish_R", false)
                elseif tbGuessData.nTongId == tbRightData.nTongId then
                  pPanel:SetActive("btnCheer_L", false)
                  pPanel:SetActive("btnCheer_R", false)
                  pPanel:SetActive("Cont_Finish_L", false)
                  pPanel:SetActive("Cont_Finish_R", true)
                  pPanel:Label_SetText("Cont_Finish_R/txtPlayerName02 (4)", tbGuessData.nGuessNum or 0)
                end
              else
                pPanel:SetActive("Cont_Finish_L", false)
                pPanel:SetActive("btnCheer_L", true)
                pPanel:SetActive("Cont_Finish_R", false)
                pPanel:SetActive("btnCheer_R", true)
                if tbLeftData and 0 == tbLeftData.nResult then
                  pPanel:Button_SetEnable("btnCheer_L", bInGuessTime)
                  Ui:Button_BindEvent(pPanel, "btnCheer_L", function()
                    local tbData = {
                      {
                        nId = tbLeftData.nTongId,
                        szName = tbLeftData.szTongName
                      },
                      i18n.EasyGet("\233\162\132\233\128\137\232\181\155\230\142\146\229\144\141\239\188\154{0}", self:GetTongRank(tbLeftData.nTongId)),
                      i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\231\165\168\230\149\176\239\188\154"),
                      nil
                    }
                    local nItemID = FirstTong.GuessCfg.nMatchGuessID
                    local tbCallBack = {
                      FirstTong.ChangeMatchGuess,
                      FirstTong,
                      tbLeftData.nTongId
                    }
                    Ui:OpenWindow("UIActivity_Banghuizhengduosai_Tip", tbData, nItemID, tbCfg.nGuessMax, tbCallBack)
                  end)
                else
                  pPanel:Button_SetEnable("btnCheer_L", false)
                end
                if tbRightData and 0 == tbRightData.nResult then
                  pPanel:Button_SetEnable("btnCheer_R", bInGuessTime)
                  Ui:Button_BindEvent(pPanel, "btnCheer_R", function()
                    local tbData = {
                      {
                        nId = tbRightData.nTongId,
                        szName = tbRightData.szTongName
                      },
                      i18n.EasyGet("\233\162\132\233\128\137\232\181\155\230\142\146\229\144\141\239\188\154{0}", self:GetTongRank(tbRightData.nTongId)),
                      i18n.EasyGet("\229\189\147\229\137\141\230\139\165\230\156\137\231\165\168\230\149\176\239\188\154"),
                      nil
                    }
                    local nItemID = FirstTong.GuessCfg.nMatchGuessID
                    local tbCallBack = {
                      FirstTong.ChangeMatchGuess,
                      FirstTong,
                      tbRightData.nTongId
                    }
                    Ui:OpenWindow("UIActivity_Banghuizhengduosai_Tip", tbData, nItemID, tbCfg.nGuessMax, tbCallBack)
                  end)
                else
                  pPanel:Button_SetEnable("btnCheer_R", false)
                end
              end
            end)
          end
        end
      end)
    end
  end
end

function tbWnd:UpdateLeftTime(pPanel, nGuessEndTime)
  if self.nTimer ~= nil then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  pPanel:Label_SetText("imgBG/txtTimeTittle", i18n.EasyGet("\230\156\172\232\189\174\231\187\147\230\157\159\229\128\146\232\174\161\230\151\182\239\188\154"))
  local nTimeRemaining = nGuessEndTime - GetTime()
  if nTimeRemaining > 0 then
    pPanel:Label_SetText("imgBG/txtTime", Lib:TimeDesc2(nGuessEndTime - GetTime()))
  else
    pPanel:Label_SetText("imgBG/txtTime", i18n.EasyGet("\229\183\178\231\187\147\230\157\159"))
    return false
  end
  local nNextTime = 1
  if nTimeRemaining > 3600 then
    local nCurTime = tonumber(os.date("%Y%m%d%H%M", GetTime()))
    nNextTime = Lib:GetDate2Time(nCurTime + 1) - GetTime() + 1
  end
  if nTimeRemaining > 0 then
    self.nTimer = Timer:Register(nNextTime * Env.GAME_FPS, self.UpdateLeftTime, self, pPanel, nGuessEndTime)
  end
  return false
end

function tbWnd:OnSyncItem(dwItemId, bNewItem, nAddCount)
  if dwItemId then
    local pItem = KItem.GetItemObj(dwItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId == FirstTong.GuessCfg.nChampionGuessID or pItem.dwTemplateId == FirstTong.GuessCfg.nMatchGuessID then
      self:UpdatePanel()
    end
  end
end

function tbWnd:GetTongRank(nTongId)
  if FirstTong.tbAllTongRank then
    for nRank, tbData in ipairs(FirstTong.tbAllTongRank) do
      if tbData.nTongId == nTongId then
        return nRank
      end
    end
  end
  return 0
end
