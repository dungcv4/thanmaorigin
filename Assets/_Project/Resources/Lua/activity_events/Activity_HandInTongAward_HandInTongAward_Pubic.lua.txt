local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.HandInTongAward)

function tbActivity:LoadConfig()
  local szFilepath = "Setting/Activity/HandInTongAward/HandInTongAwardCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr(string.format("%s not find", szFilepath))
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    tbConfig.tbLevel = Lib:SplitStr2Num(tbConfig.szLevel, "|") or {}
    table.sort(tbConfig.tbLevel)
    tbConfig.tbAwardKey = Lib:SplitStr2Num(tbConfig.szAwardKey, "|") or {}
    assert(#tbConfig.tbLevel == #tbConfig.tbAwardKey, string.format("HandInTongAward: LoadConfig failed, level nums ~= awardkey nums, nConfigId = %d", nConfigId))
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  Validator:Register(self.CheckAllConfig, self)
end

function tbActivity:GetLevel(nConfigId)
  return self.ACTIVITY_CONFIG[nConfigId].tbLevel
end

function tbActivity:GetAward(nConfigId, nLevel)
  local nAward = self.ACTIVITY_CONFIG[nConfigId].nAwardId
  local nAwardkey = self.ACTIVITY_CONFIG[nConfigId].tbAwardKey[nLevel]
  return Award:GetAward(nAward, nAwardkey)
end

function tbActivity:GetNumsByLevel(nLevel)
  return self.tbConfig.tbLevel[nLevel]
end

function tbActivity:CheckAllConfig()
  for nConfigId, tbConfigs in pairs(self.ACTIVITY_CONFIG) do
    if not (tbConfigs.szSaveKey and tbConfigs.nFormulaID and tbConfigs.szLogWay and tbConfigs.szCountEvent and tbConfigs.szAwardEvent and tbConfigs.nRecordDataTime and tbConfigs.nAwardTime and tbConfigs.tbLevel and tbConfigs.nAwardId and tbConfigs.tbAwardKey and tbConfigs.szMailFrom and tbConfigs.szMailTitle) or not tbConfigs.szMailText then
      LogErr("HandInTongAward Lost Config", nConfigId, tbConfigs.szSaveKey, tbConfigs.nFormulaID)
      return false
    end
    if not Env[tbConfigs.szLogWay] then
      LogErr("HandInTongAward szLogWay not Exsit", nConfigId)
      return false
    end
    if not Event[tbConfigs.szCountEvent] or not Event[tbConfigs.szAwardEvent] then
      LogErr("HandInTongAward Event not Exsit", nConfigId)
      return false
    end
    if #tbConfigs.tbLevel ~= #tbConfigs.tbAwardKey then
      LogErr("HandInTongAward Event not Exsit", nConfigId)
      return false
    end
    if tbConfigs.nAwardTime < tbConfigs.nRecordDataTime then
      LogErr("HandInTongAward Date Err", nConfigId)
      return false
    end
    for nIndex, nkey in pairs(tbConfigs.tbAwardKey) do
      local tbAward = Award:GetAward(tbConfigs.nAwardId, nkey)
      if not tbAward or 0 == #tbAward then
        LogErr("HandInTongAward Award not Exsit", nConfigId, tbConfigs.nAwardId, nkey)
        return false
      end
    end
  end
  return true
end
