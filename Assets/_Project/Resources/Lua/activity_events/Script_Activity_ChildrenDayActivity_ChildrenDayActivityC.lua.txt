Require("CommonScript/Activity/ChildrenActivity/ChildrenActivity/ChildrenActivityCfg.lua")
ActivityMgr.ChildrenActivity = ActivityMgr.ChildrenActivity or {}
local ChildrenActivity = ActivityMgr.ChildrenActivity
local tbJumpFunc = {
  [ActivityMgr.Def.ClassNameList.QingChunQuan] = function(nId)
    ActivityMgr.ChildrenActivity:OnActive(nId)
    if 0 == Ui.PlayerPrefs.GetInt("QingChunQuanFirst", 0) then
      Ui.PlayerPrefs.SetInt("QingChunQuanFirst", 1)
      RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.QingChunQuanFirstOpen)
    end
  end,
  NpcAward = function(nId)
    local tbWnd = Ui:GetWindow("UIChildrenDayActivity")
    if not tbWnd then
      return
    end
    local pInstance = ActivityMgr:GetInstanceById(nId)
    if not pInstance or not pInstance:IsOpen() then
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
      return
    end
    local tbMapInfoList = pInstance:GetMapShowInfo()
    tbWnd:ShowJumpToWnd(tbMapInfoList)
    pInstance:OnSyncRedPointStatus(false)
  end,
  [ActivityMgr.Def.ClassNameList.AgeThief] = function(nId)
    local tbWnd = Ui:GetWindow("UIChildrenDayActivity")
    if not tbWnd then
      return
    end
    local pInstance = ActivityMgr:GetInstanceById(nId)
    if not pInstance or not pInstance:IsOpen() then
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
      return
    end
    local tbMapInfoList = pInstance:GetMapShowInfo()
    tbWnd:ShowJumpToWnd(tbMapInfoList)
  end,
  [ActivityMgr.Def.ClassNameList.LimitAuction] = function(nId)
    local tbWnd = Ui:GetWindow("UIChildrenDayActivity")
    if not tbWnd then
      return
    end
    local pInstance = ActivityMgr:GetInstanceById(nId)
    if not pInstance or not pInstance:IsOpen() then
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
      return
    end
    pInstance:OpenLimitAuctionUI()
  end
}

function ChildrenActivity:GetActivityCfg(szType)
  local tbActivityList = self:GetActivityList()
  if not tbActivityList then
    Log(" ChildrenActivity:GetActivity() data nil ")
    return nil
  end
  for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
    local tbConfig = self:BuildInstanceConfig(nId, szType)
    return tbConfig
  end
  return nil
end

function ChildrenActivity:GetActivityList()
  local tbList = {}
  local tbConfig, szName, pInstance
  for _, szType in ipairs(ActivityMgr.ChildrenActivityCfg.ActivityList) do
    for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
      tbConfig, pInstance = self:BuildInstanceConfig(nId, szType)
      if tbConfig then
        szName = tbConfig.szName and i18n.Parse(tbConfig.szName) or ""
        table.insert(tbList, {
          szClassName = szType,
          nId = nId,
          szName = szName,
          szDesc = i18n.Parse(tbConfig.szDesc),
          nLevel = tbConfig.nLevel or tbConfig.nJoinLevel,
          tbAward = Award:GetAward(tbConfig.nShowAwardId or tbConfig.nAwardId or -1, tbConfig.nShowAwardKey or tbConfig.nAwardKey or -1) or {},
          pFunc = tbJumpFunc[szType],
          szShowTime = i18n.Parse(tbConfig.szShowTime or ""),
          nSortValue = ActivityMgr.ChildrenActivityCfg.SortList[nId] or 0,
          tbRedPointEventId = pInstance and pInstance.GetRedPointEventId and pInstance:GetRedPointEventId(),
          szTimeFrame = tbConfig.szTimeFrame or nil
        })
      end
    end
  end
  table.sort(tbList, function(tbFirst, tbSecond)
    return tbFirst.nSortValue > tbSecond.nSortValue
  end)
  return tbList
end

function ChildrenActivity:BuildInstanceConfig(nId, szType)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if pInstance then
    return pInstance:GetConfig(), pInstance
  end
  local tmpInstance = ActivityMgr:Create(szType)
  if not tmpInstance then
    return
  end
  tmpInstance:OnInit(ActivityMgr:GetConfigById(nId))
  return tmpInstance:GetConfig()
end

function ChildrenActivity:OnActive(nId)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return
  end
  if pInstance:IsA(ActivityMgr.Def.ClassNameList.QingChunQuan) then
    return self:OnLaborHonorActive(pInstance)
  end
end

function ChildrenActivity:OnLaborHonorActive(pInstance)
  Ui:OpenWindow("UIZhiQingChun", pInstance:GetId())
end
