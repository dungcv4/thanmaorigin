local tbWnd = Ui:CreateWindow("UIFestival_OtherWelfare")
local tbControls = {
  txtTitle = "Tex/txtTitle",
  Template = "Scroll View/Viewport/Content/Template",
  btnHelp = "Tex/txtTitle/btnHelp",
  btnLeft = "btnLeft",
  btnRight = "btnRight"
}
tbWnd.tbControls = tbControls
tbWnd.ElementPath = {
  txtNameNormal = "txtNameNormal",
  txtTime = "txtTime01",
  txtCont = "txtCont",
  btnGo = "btnGo",
  imgDiscountsBg = "imgDiscountsBg"
}
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  self.nIndex = nIndex
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.szHelpKey = tbInfo.szHelpKey
  self.pPanel:Label_SetText(tbControls.txtTitle, tbInfo.szName)
  self:UpdateViewList()
  self.pPanel:SetActive(tbControls.btnLeft, false)
  self.pPanel:SetActive(tbControls.btnRight, false)
end

function tbWnd:UpdateViewList()
  self.tbScrollView:Reset()
  local tbSmallActList = ActivityMgr.CommonEntryLinkData[self.tbActivityList[self.nIndex].nId].tbActivityList
  for index, nCommonEntryID in ipairs(tbSmallActList) do
    self.tbScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:SetActive(self.ElementPath.imgDiscountsBg, false)
      local tbInstConfig = ActivityMgr:GetConfigById(nCommonEntryID)
      local tbConfig = ActivityMgr.EntranceSchedule:BuildInstanceConfig(nCommonEntryID, ActivityMgr.Def.ClassNameList.CommonEntry)
      local szName = tbConfig.szName and i18n.Parse(tbConfig.szName) or ""
      tbElement.pPanel:Label_SetText(self.ElementPath.txtNameNormal, szName)
      local szTime = i18n.EasyGet("{0}\n\229\183\178\231\187\147\230\157\159", i18n.Parse(tbConfig.szShowTime))
      tbElement.pPanel:Label_SetText(self.ElementPath.txtTime, szTime)
      local bOpen = false
      local bEnd = false
      if GetTime() > tbInstConfig.nStartTime then
        bOpen = true
      end
      if GetTime() > tbInstConfig.nEndTime then
        bEnd = true
      end
      if not bEnd then
        local nDiff = tbInstConfig.nEndTime - GetTime()
        szTime = i18n.EasyGet("{0}\n{1}\229\144\142\231\187\147\230\157\159", i18n.Parse(tbConfig.szShowTime), Lib:TimeDesc2(nDiff))
        tbElement.pPanel:Label_SetText(self.ElementPath.txtTime, szTime)
      end
      if not bOpen then
        local nDiff = tbInstConfig.nStartTime - GetTime()
        szTime = i18n.EasyGet("{0}\n{1}\229\144\142\229\188\128\229\144\175", i18n.Parse(tbConfig.szShowTime), Lib:TimeDesc2(nDiff))
        tbElement.pPanel:Label_SetText(self.ElementPath.txtTime, szTime)
      end
      tbElement.pPanel:Label_SetText(self.ElementPath.txtCont, i18n.Parse(tbConfig.szDesc))
      if tbConfig.szJumpType and not Lib:IsEmptyStr(tbConfig.szJumpType) then
        Ui:Button_BindEvent(tbElement.pPanel, self.ElementPath.btnGo, self.OnClickGo, self, tbConfig.szJumpType, tbConfig.szJumpParam)
      else
        tbElement.pPanel:SetActive(self.ElementPath.btnGo, false)
      end
      tbElement.pPanel:SetActive(self.ElementPath.btnGo, not bEnd)
      tbElement.pPanel:Button_SetEnable(self.ElementPath.btnGo, bOpen)
    end)
  end
end

function tbWnd:OnClickGo(szCallBack, szParam)
  if not szCallBack or Lib:IsEmptyStr(szCallBack) then
    return
  end
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam, "|")
  fn(self.tbActivityList[self.nIndex].nId, table.unpack(tbParam))
end

function tbWnd:OnClose()
  self.tbScrollView:Reset()
end
