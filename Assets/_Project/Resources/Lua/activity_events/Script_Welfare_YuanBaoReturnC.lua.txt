YuanBaoReturn.nBuildingId = YuanBaoReturn.nBuildingId or -1
YuanBaoReturn.nYuanBaoNum = YuanBaoReturn.nYuanBaoNum or 0

function YuanBaoReturn:CheckIsOpen(pPlayer)
  return self.nBuildingId > -1
end

function YuanBaoReturn:OnSyncBuildingInfo(dwId, nYuanBaoNum)
  self.nBuildingId = dwId
  self.nYuanBaoNum = nYuanBaoNum
  self:CheckRedPoint()
end

function YuanBaoReturn:OnNewDay()
  EventNotify.OnNotify(EventNotify.emNOTIFY_YUANBAORETURN_REFRESH)
  self:CheckRedPoint()
end

function YuanBaoReturn:CheckRedPoint()
  local nRemainNum, nTodayNum = self:GetRemainAndTodayNum()
  if nRemainNum > 0 and not self:HasTodayGotten() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.YuanBaoReturn)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.YuanBaoReturn)
  end
end

function YuanBaoReturn:OnGetAward()
  EventNotify.OnNotify(EventNotify.emNOTIFY_YUANBAORETURN_REFRESH)
  self:CheckRedPoint()
  if self:GetRemainNum(me) <= 0 then
    me.CenterMsg(i18n.EasyGet("\230\130\168\229\183\178\233\162\134\229\143\150\230\137\128\230\156\137\232\191\148\232\191\152\231\154\132\229\133\131\229\174\157"))
  end
end

function YuanBaoReturn:GetRemainAndTodayNum()
  local nTotalNum = me.GetUserValue(self.SaveGroup, self.TotalNumKey)
  local nTotalGetNum = me.GetUserValue(self.SaveGroup, self.TotalGetNumKey)
  local nRemainNum = nTotalNum - nTotalGetNum
  local nTotalGetDayNum = me.GetUserValue(self.SaveGroup, self.TotalGetDayKey)
  local nTodayNum = self:GenTodayYuanBaoNum(me.dwID, nTotalNum, nTotalGetNum, nTotalGetDayNum)
  return nRemainNum, nTodayNum
end

function YuanBaoReturn:HasTodayGotten()
  local nNow = GetTime()
  local nToday = Lib:GetLocalDay(nNow - self.RefreshOffSet)
  return me.GetUserValue(self.SaveGroup, self.LastGetDayKey) == nToday
end

function YuanBaoReturn:TryGainAward()
  RemoteServer.RequestYuanBaoReturn(self.Protocol.GetAward)
end
