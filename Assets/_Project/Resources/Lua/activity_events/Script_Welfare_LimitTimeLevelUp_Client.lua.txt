LimitTimeLevelUp = LimitTimeLevelUp or {}
LimitTimeLevelUp.nOpenLevel = 18

function LimitTimeLevelUp:RPC_TryGetAward(nIndex)
  local bRet, szMsg = self:CheckCanGet(me, nIndex)
  if not bRet then
    me.CenterMsg(szMsg)
    return
  end
end

function LimitTimeLevelUp:OnSynResult(nIndex)
  EventNotify.OnNotify(EventNotify.emNOTIFY_LIMITTIME_LEVELUP, nIndex)
end

function LimitTimeLevelUp:CheckIsOpen()
  if not OnOffController:Check(OnOffController.KEY.LimitTimeLevelUp) then
    return false
  end
  if me.nLevel < LimitTimeLevelUp.nOpenLevel then
    return false
  end
  for index, value in ipairs(self.tbSetting) do
    local nUserValue = me.GetUserValue(self.GROUP, self:GetUserValueKey(index))
    if 1 ~= nUserValue then
      return true
    end
  end
  return false
end

function LimitTimeLevelUp.OnChangeOpen()
end

LimitTimeLevelUp.tbIndex2RedPoint = {
  [1] = RedPointMgr.EVENT.LimitTimeLevelUpGrade1,
  [2] = RedPointMgr.EVENT.LimitTimeLevelUpGrade2,
  [3] = RedPointMgr.EVENT.LimitTimeLevelUpGrade3,
  [4] = RedPointMgr.EVENT.LimitTimeLevelUpGrade4,
  [5] = RedPointMgr.EVENT.LimitTimeLevelUpGrade5
}

function LimitTimeLevelUp:CheckRedPoint()
  if not self.tbSetting then
    return
  end
  if not OnOffController:Check(OnOffController.KEY.LimitTimeLevelUp) then
    return
  end
  for index, value in ipairs(self.tbSetting) do
    if self:CheckCanGet(me, index) then
      RedPointMgr:LightRedPoint(self.tbIndex2RedPoint[index])
    elseif RedPointMgr:CheckEvent(self.tbIndex2RedPoint[index]) then
      RedPointMgr:FadeRedPoint(self.tbIndex2RedPoint[index])
    end
  end
end
