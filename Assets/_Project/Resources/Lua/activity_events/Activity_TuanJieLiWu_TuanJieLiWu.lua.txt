Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.TuanJieLiWu)
tbActivity.Protocol = {
  GetBuyInfo = 1,
  GetDiscountInfo = 2,
  OnGetBuyInfo = 3,
  OnGetDiscountInfo = 4
}

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/XiongWangActivity/TuanJieLiWu/ActivityConfig.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    tbConfig.nBuyEndTime = Lib:ParseNumbericDateToTime(tbConfig.nBuyEndTime)
    tbConfig.nRewardTime = Lib:ParseNumbericDateToTime(tbConfig.nRewardTime)
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[nConfigId].szSaveKey) then
      self.ACTIVITY_CONFIG[nConfigId].szSaveKey = ActivityMgr.Def.ClassNameList.TuanJieLiWu
    end
    if Lib:IsEmptyStr(self.ACTIVITY_CONFIG[nConfigId].szLogWay) then
      self.ACTIVITY_CONFIG[nConfigId].szLogWay = "LogWay_TuanJieLiWu"
    end
  end
  szFilepath = "Setting/Activity/XiongWangActivity/TuanJieLiWu/Discount.tab"
  local tbDiscountInfo = Lib:EasyLoadTabFile(szFilepath)
  for _, tbInfo in pairs(tbDiscountInfo) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("TuanJieLiWu:LoadConfig Discount nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDiscountInfo = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDiscountInfo or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbDiscountInfo[tbInfo.nGoodId] = tbInfo
  end
  szFilepath = "Setting/Activity/XiongWangActivity/TuanJieLiWu/Goods.tab"
  local tbGoodInfo = Lib:EasyLoadTabFile(szFilepath)
  for _, tbInfo in pairs(tbGoodInfo) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("TuanJieLiWu:LoadConfig Goods nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGoodInfo = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGoodInfo or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbGoodInfo[tbInfo.nGoodId] = tbInfo
  end
  self.bLoadConfig = true
  return true
end

function tbActivity:GetDiscountInfo()
  return self.tbConfig.tbDiscountInfo
end

function tbActivity:IsBuyEndTime()
  if self.tbConfig.nBuyEndTime then
    return self.tbConfig.nBuyEndTime <= GetTime()
  else
    LogErr("TuanJieLiWu: IsBuyEndTime failed, nBuyEndTime is nil")
  end
end

function tbActivity:IsRewardTime()
  if self.tbConfig.nRewardTime then
    return self.tbConfig.nRewardTime <= GetTime()
  else
    LogErr("TuanJieLiWu: IsRewardTime failed, nRewardTime is nil")
  end
end
