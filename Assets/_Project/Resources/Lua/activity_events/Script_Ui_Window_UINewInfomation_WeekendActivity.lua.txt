local tbWnd = Ui:CreateClass("UINewInfomation_WeekendActivity", "UIPanel")
tbWnd.szClassName = "UINewInfomation_WeekendActivity"
tbWnd.tbControls = {
  txtInfo2 = "txtInfo2",
  List = "imgList/List",
  Template = "imgList/List/Template"
}
tbWnd.tbOnClickBtnGo = {
  [1] = function()
    tbWnd:OnClickWeekMonthCard()
  end,
  [2] = function()
    tbWnd:OnClickTongChallenge()
  end,
  [3] = function()
    tbWnd:OnClickTongParty()
  end
}

function tbWnd:OnCreate()
  self.tbActivityScrollView = Ui:NewScrollView(self, self.Template, nil, false, false)
  self.tbWeekendActivityObj = ActivityMgr:GetInstanceByClassName(ActivityMgr.szWeekendActivityClassName)
  if not self.tbWeekendActivityObj then
    return
  end
  self.tbWeekendCfg = self.tbWeekendActivityObj:GetConfig()
  if not self.tbWeekendCfg then
    LogErr("Weekend Config is load error")
    return
  end
end

function tbWnd:OnOpen()
  self.tbWeekendActivityObj = ActivityMgr:GetInstanceByClassName(ActivityMgr.szWeekendActivityClassName)
  if not self.tbWeekendActivityObj then
    self:Close()
    return
  end
  self.tbWeekendCfg = self.tbWeekendActivityObj:GetConfig()
  if not self.tbWeekendCfg then
    LogErr("Weekend Config is load error")
    self:Close()
    return
  end
  self.pPanel:Label_SetText(self.tbControls.txtInfo2, self.tbWeekendActivityObj:GetShowTxtInfo())
  self.tbWeekendActivityObj:TryFadeRedPoint()
  local nIndex = 1
  for nActivetyId, tbInfo in pairs(self.tbWeekendCfg) do
    self.tbActivityScrollView:SetElement(nIndex, function(tbElement)
      tbElement.pPanel:Label_SetText("ListTitle/txtActivityName", i18n.EasyGet("\230\180\187\229\138\168{0}", nIndex))
      tbElement.pPanel:Label_SetText("ListTitle/txtSlogon", i18n.Parse(tbInfo.szActivityName))
      tbElement.pPanel:Label_SetText("txtActivityDetail", i18n.Parse(tbInfo.szActivityDesc))
      tbElement.pPanel:Button_BindEvent("btnGo", function(ClickBtnFunc)
        if not ClickBtnFunc then
          return
        end
        ClickBtnFunc()
      end, self.tbOnClickBtnGo[nIndex])
    end, false)
    nIndex = nIndex + 1
  end
end

function tbWnd:OnClose()
  self.tbActivityScrollView:Clear()
end

function tbWnd:OnClickWeekMonthCard()
  Ui:OpenWindow("UIWelfare", "PANEL_TYPE_WEEK_MONTH_CARD")
end

function tbWnd:OnClickTongParty()
  PlayerGuide.tbButtonFuncs.OpenUi("UIActivity|11")
end

function tbWnd:OnClickTongChallenge()
  PlayerGuide.tbButtonFuncs.OpenUi("UIActivity|10")
end
