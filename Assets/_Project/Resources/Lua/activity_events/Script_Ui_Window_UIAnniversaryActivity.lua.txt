  NumberCell = "node/imgBG/Content/tips/NumberCell",
  ServerTxtTitle = "node/imgBG/Content/tips/NumberCell/txtTitle",
  ServerTxtNumber = "node/imgBG/Content/tips/NumberCell/txtNumber",
  ServerScoreCell = "node/imgBG/Content/imgTitleBg/ServerScoreCell",
  txtTime = "node/imgBG/Content/timeCell/txtTime",
  txtCost1 = "node/imgBG/Content/costCell1/txtCost1",
  txtCost10 = "node/imgBG/Content/costCell10/txtCost10",

tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnShop = function(self)
    local nRedPointEventId = self.pMiYuanActInstance:GetRedPointEventList()[1]
    if RedPointMgr:CheckEvent(nRedPointEventId) then
      RedPointMgr:FadeRedPoint(nRedPointEventId)
    end
    Ui:OpenWindow("UINpcShop", self.nNpcShopId, nil, nil, nil, nil, Currency.TYPE.XunYuan)
  end,
  btnCostReward = function(self)
    Ui:OpenWindow("UIYueNanWomenDayReward", self.nAchieveActId)
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end,
  btnXuniYuan1 = function(self)
    self.pMiYuanActInstance:ReqFindYuan(self.pMiYuanActInstance.FindTypeIndex.OnceFind)
  end,
  btnXuniYuan10 = function(self)
    self.pMiYuanActInstance:ReqFindYuan(self.pMiYuanActInstance.FindTypeIndex.BatchFind)
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnBuyResult(bSuccess, nItemId)
  if self.nSyncTimer then
    Timer:Close(self.nSyncTimer)
    self.nSyncTimer = nil
  end
  if nItemId ~= self.pMiYuanActInstance.tbConfig.nFlowerTemplateId then
    return
  end
  local nHasFlowerCount = me.GetItemCountInBags(self.pMiYuanActInstance.tbConfig.nFlowerTemplateId)
  local szFlowerIconPath = self.pMiYuanActInstance.tbConfig.szFlowerIconPath
  self.tbMoneyPanel:SetCurrencyView(false)
  self.tbMoneyPanel:SetDynamicCurrencyInfo("Flower", nHasFlowerCount, szFlowerIconPath)
  self.tbMoneyPanel:SetDynamicBtnPlusCallBack(self.__OpenQuickBuyWindow, self)
  return false
end

function tbWnd:__OpenQuickBuyWindow()
  local tbYuanBaoWare = Shop:GetAWare(self.pMiYuanActInstance.tbConfig.nYuanBaoWareId)
  local nRemainCount = 0
  if tbYuanBaoWare then
    nRemainCount = Shop:GetShopRemainCount(tbYuanBaoWare, me)
    if nRemainCount > 0 then
      Ui:OpenWindow("UIShopBuy", tbYuanBaoWare, true, nRemainCount)
      return
    end
  end
  for nIndex, nGoldWareId in ipairs(self.pMiYuanActInstance.tbConfig.tbnGoldWareId) do
    local tbGoldWare = Shop:GetAWare(nGoldWareId)
    if tbGoldWare then
      nRemainCount = Shop:GetShopRemainCount(tbGoldWare, me)
      if nRemainCount > 0 then
        Ui:OpenWindow("UIShopBuy", tbGoldWare, true, nRemainCount)
        return
      end
    end
  end
  me.CenterMsg(i18n.EasyGet("\229\175\185\228\184\141\232\181\183\239\188\140\230\130\168\232\180\173\228\185\176{0}\231\154\132\230\149\176\233\135\143\229\183\178\232\190\190\229\136\176\228\184\138\233\153\144", Item:GetName(self.pMiYuanActInstance.tbConfig.nFlowerTemplateId, me)))
end

function tbWnd:OnRefreshUI(tbData)
  local nCurScore, nHasFindCount, _ = self.pMiYuanActInstance:GetPlayerScoreInfo()
  self.pPanel:Label_SetText(self.txtHaveCost, nHasFindCount)
  self.pPanel:Label_SetText(self.txtNumber, i18n.EasyGet("{0}\230\172\161", nHasFindCount))
  self.pPanel:Label_SetText(self.txtMyCurScoreTitle, i18n.EasyGet("\230\136\145\231\154\132\231\167\175\229\136\134"))
  self.pPanel:Label_SetText(self.txtMyCurScore, nCurScore)
  self.pPanel:SetActive(self.NumberCell, false)
  self.pPanel:Label_SetText(self.txtCost1, self.pMiYuanActInstance.FindType[self.pMiYuanActInstance.FindTypeIndex.OnceFind].nConsumeCount)
  self.pPanel:Label_SetText(self.txtCost10, self.pMiYuanActInstance.FindType[self.pMiYuanActInstance.FindTypeIndex.BatchFind].nConsumeCount)
  self.pPanel:Label_SetText(self.txtDiscounts, i18n.GetDiscountText(self.pMiYuanActInstance.FindType[self.pMiYuanActInstance.FindTypeIndex.BatchFind].nConsumeCount / self.pMiYuanActInstance.FindType[self.pMiYuanActInstance.FindTypeIndex.BatchFind].nFindCount * 10))
  self:UpdateTimeStr(Lib:IIF(self.pMiYuanActInstance.nEndTime - tbData.nServerTime > 0, self.pMiYuanActInstance.nEndTime - tbData.nServerTime, 0))
  self:OnBuyResult(true, self.pMiYuanActInstance.tbConfig.nFlowerTemplateId)
end
