local tbWnd = Ui:CreateClass("UIMayDaySignIn", "UIPanel")
local tbControls = {
  SignInTemplate = "List01/List/Viewport/Content/Template",
  btnHelp = "txtTitle/btnHelp",
  txtTime01 = "txtTime01",
  txtTitle = "txtTitle"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.tbSignInView = Ui:NewScrollView(self, tbControls.SignInTemplate)
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_UPDATE_DAILY_AWARD_WND, self)
  self.bRegist = false
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  if not self.bRegist then
    EventNotify:RegistNotify(EventNotify.emNOTIFY_UPDATE_DAILY_AWARD_WND, self.UpdateListView, self)
    self.bRegist = true
  end
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  self.nDailyAwardActId = tbInfo.nId
  self.pDailyAwardAct = ActivityMgr:GetInstanceById(self.nDailyAwardActId)
  if not self.pDailyAwardAct then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    self.pDailyAwardAct = ActivityMgr.EntranceSchedule:BuildInstance(self.nDailyAwardActId, ActivityMgr.Def.ClassNameList.DailyAward)
  end
  self.szHelpKey = tbInfo.szHelpKey
  if not self.pDailyAwardAct:IsOpen() then
    local nRestTime = self.pDailyAwardAct:GetStartTime() - GetTime()
    self.pPanel:Label_SetText(tbControls.txtTime01, i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\229\188\128\229\144\175", i18n.Parse(self.pDailyAwardAct.tbConfig.szShowTime), Lib:TimeDesc2(nRestTime)))
  else
    local nRestTime = self.pDailyAwardAct:GetEndTime() - GetTime()
    self.pPanel:Label_SetText(tbControls.txtTime01, i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", i18n.Parse(self.pDailyAwardAct.tbConfig.szShowTime), Lib:TimeDesc2(nRestTime)))
  end
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.Parse(self.pDailyAwardAct.tbConfig.szName))
  self:UpdateListView()
end

function tbWnd:UpdateListView()
  if not self.pDailyAwardAct then
    return
  end
  local tbAwardInfo = self.pDailyAwardAct:GetAwardInfo(true)
  self.tbSignInView:Clear()
  for nIndex, tbDailyAward in ipairs(tbAwardInfo or {}) do
    self.tbSignInView:SetElement(nIndex, function(tbElement)
      self:UpdateElement(tbElement, tbDailyAward)
    end, nil)
  end
end

function tbWnd:UpdateElement(tbElement, tbDailyAward)
  tbElement.pPanel:Label_SetText("ImgAll/txtRewardTitle", tbDailyAward.szTitle)
  if not tbElement.tbRewardView then
    tbElement.tbRewardView = Ui:NewItemGridScrollView(tbElement, "ImgAll/ContList/Viewport/Content/ItemTemplate", 1, 1)
  else
    tbElement.tbRewardView:Clean()
  end
  tbElement.tbRewardView:UpdateGenericItem(tbDailyAward.tbAwards, true)
  self:UpdateButtonStatus(tbElement, tbDailyAward)
end

function tbWnd:UpdateButtonStatus(tbElement, tbDailyAward)
  if tbDailyAward.nGetStatus == self.pDailyAwardAct.AwardStatus.HasGet then
    tbElement.pPanel:SetActive("ImgAll/btnGet", false)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnGet", false)
    tbElement.pPanel:SetActive("ImgAll/btnReceive", false)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnReceive", false)
    tbElement.pPanel:SetActive("ImgAll/img_Finish", true)
  elseif tbDailyAward.bCanGetAsSignAward then
    tbElement.pPanel:SetActive("ImgAll/btnReceive", false)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnReceive", false)
    tbElement.pPanel:SetActive("ImgAll/img_Finish", false)
    tbElement.pPanel:SetActive("ImgAll/btnGet", true)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnGet", true)
    RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "ImgAll/btnGet", {
      RedPointMgr.EVENT.DailyAwardSignIn
    })
    Ui:Button_BindEvent(tbElement.pPanel, "ImgAll/btnGet", function()
      self.pDailyAwardAct:ReqGetDailyAward(tbDailyAward.nIndex, self.pDailyAwardAct.GetType.SignIn)
      if RedPointMgr:CheckEvent(RedPointMgr.EVENT.DailyAwardSignIn) then
        RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.DailyAwardSignIn)
      end
    end)
  elseif tbDailyAward.bCanGetAsReSignAward then
    tbElement.pPanel:SetActive("ImgAll/btnGet", false)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnGet", false)
    tbElement.pPanel:SetActive("ImgAll/img_Finish", false)
    tbElement.pPanel:SetActive("ImgAll/btnReceive", true)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnReceive", true)
    tbElement.pPanel:Button_SetText("ImgAll/btnReceive", self.pDailyAwardAct.tbConfig.nRewardCost, "txt")
    RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "ImgAll/btnReceive", {
      RedPointMgr.EVENT.DailyAwardReSignIn
    })
    Ui:Button_BindEvent(tbElement.pPanel, "ImgAll/btnReceive", function()
      self.pDailyAwardAct:ReqGetDailyAward(tbDailyAward.nIndex, self.pDailyAwardAct.GetType.ReSign)
    end)
  else
    tbElement.pPanel:SetActive("ImgAll/btnGet", true)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnGet", false)
    tbElement.pPanel:SetActive("ImgAll/img_Finish", false)
    tbElement.pPanel:SetActive("ImgAll/btnReceive", false)
    tbElement.pPanel:Button_SetEnable("ImgAll/btnReceive", false)
  end
end
