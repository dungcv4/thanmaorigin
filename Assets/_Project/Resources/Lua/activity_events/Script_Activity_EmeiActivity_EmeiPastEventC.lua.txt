Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.EmeiPastEvent)
local EmeiPastEventCfg = ActivityCfg.EmeiPastEventCfg

function tbActivity:C2SGetExtraAward()
  ActivityMgr:CallServerScript(EmeiPastEventCfg.nActivityId, self.C2SProtocol.GetExtraAward)
end

function tbActivity:ServerRequest(nProtocol, ...)
  if nProtocol == self.S2CProtocol.SyncAward then
    self:OnS2CSyncGetAward()
  end
end

function tbActivity:OnS2CSyncGetAward()
  self:UpdateGetAwardRedPoint()
end

function tbActivity:OnOpen()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_FINISH_TASK, self.OnFinishTask, self)
  self:UpdateGetAwardRedPoint()
end

function tbActivity:OnClientClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_FINISH_TASK, self)
  self:UpdateGetAwardRedPoint()
end

function tbActivity:OnClickButton()
  if not self:HaveFinishEmeiTask(me) then
    if Task:IsTaskDoing(me, EmeiPastEventCfg.nEmeiSubTaskId) then
      Ui:OpenWindowWithCallback("UITask", function(_, tbWnd)
        tbWnd:SelectSubTaskInWindow(EmeiPastEventCfg.nEmeiSubTaskId)
        Ui:CloseWindow("UIEmeiActivity")
      end)
    else
      me.CenterMsg(i18n.Get(EmeiPastEventCfg.tbText.szNoAcceptTaskTips))
    end
  elseif not self:HaveGetAward(me) then
    self:C2SGetExtraAward()
  end
end

function tbActivity:OnFinishTask(nTaskId)
  if nTaskId == EmeiPastEventCfg.nEmeiSubTaskId then
    self:UpdateGetAwardRedPoint()
  end
end

function tbActivity:GetRedPointEventId()
  return {
    RedPointMgr.EVENT.EmeiPastEventGetAward
  }
end

function tbActivity:UpdateGetAwardRedPoint()
  if self:IsOpen() and self:HaveGetExtraAwardTimesAndFinishTask(me) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.EmeiPastEventGetAward)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.EmeiPastEventGetAward)
  end
end
