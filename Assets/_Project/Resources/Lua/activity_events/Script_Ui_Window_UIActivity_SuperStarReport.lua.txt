local tbWnd = Ui:CreateWindow("UIActivity_SuperStarReport")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle1 = "node/imgBG/txtTitle",
  NotInfo = "node/imgBG/Content/Ranking/NotInfo",
  txtTitle2 = "node/imgBG/Content/Ranking/txtTitle",
  txtTitleTeamName = "node/imgBG/Content/Ranking/titles/TitlePlayer/txtTitleTeamName",
  txtTitleScore = "node/imgBG/Content/Ranking/titles/TitlePlayer/txtTitleScore",
  ScrollView = "node/imgBG/Content/Ranking/Scroll View",
  Template = "node/imgBG/Content/Ranking/Scroll View/Viewport/Content/Template",
  Ranking = "node/imgBG/Content/Ranking",
  txtScoreValue = "node/imgBG/Content/Ranking/txtScoreValue",
  txtScore = "node/imgBG/Content/Ranking/txtScoreValue/txtScore",
  txtRank = "node/imgBG/Content/Ranking/txtRank",
  txtRankValue = "node/imgBG/Content/Ranking/txtRank/txtRankValue",
  txtTips = "node/imgBG/Content/Ranking/txtTips"
}
tbWnd.tbControls = tbControls
tbWnd.SPECIAL_RANKING = 10
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbRankView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(tbSetting, tbRankData, tbMyData)
  self.tbSetting = tbSetting or {}
  self.tbRankDataList = tbRankData or {}
  self.tbMyData = tbMyData or {}
  self:UpdateText()
  self:UpdateRank()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateText()
  self.pPanel:Label_SetText(tbControls.txtTitle1, self.tbSetting.szTitle1)
  self.pPanel:Label_SetText(tbControls.txtTitle2, self.tbSetting.szTitle2)
  self.pPanel:Label_SetText(tbControls.txtTitleScore, self.tbSetting.szTitleScore)
  self.pPanel:Label_SetText(tbControls.txtTitleTeamName, self.tbSetting.szTitleName)
  self.pPanel:Label_SetText(tbControls.txtRank, self.tbSetting.szTitleMyRank)
  self.pPanel:Label_SetText(tbControls.txtScore, self.tbSetting.szTitleMyScore)
  self.pPanel:Label_SetText(tbControls.txtTips, self.tbSetting.szTips)
end

function tbWnd:UpdateRank()
  if 0 == #self.tbRankDataList then
    self.pPanel:SetActive(tbControls.NotInfo, true)
    self.pPanel:SetActive(tbControls.ScrollView, false)
    return
  end
  self.pPanel:SetActive(tbControls.NotInfo, false)
  self.pPanel:SetActive(tbControls.ScrollView, true)
  self.tbRankView:Clear()
  for nRank, tbInfo in ipairs(self.tbRankDataList) do
    if nRank <= self.tbSetting.nRankMaxNum then
      self.tbRankView:SetElement(nRank, function(tbElement)
        self:_SetRankElement(tbElement, nRank, tbInfo)
      end)
    end
  end
  self.pPanel:Label_SetText(tbControls.txtRankValue, self.tbMyData.nRank or 0)
  self.pPanel:Label_SetText(tbControls.txtScoreValue, self.tbMyData.nScore or 0)
end

function tbWnd:_SetRankElement(tbElement, nRank, tbRankInfo)
  if nRank > 0 and nRank <= self.SPECIAL_RANKING then
    tbElement.pPanel:SetActive("UIRankNo", true)
    tbElement.pPanel:SetActive("txtRank", false)
    for i = 1, self.SPECIAL_RANKING do
      tbElement.pPanel:SetActive("UIRankNo/imgNO" .. i, i == nRank)
    end
  else
    tbElement.pPanel:SetActive("UIRankNo", false)
    tbElement.pPanel:SetActive("txtRank", true)
    tbElement.pPanel:Label_SetText("txtRank", nRank)
  end
  tbElement.pPanel:Label_SetText("txtServer", tbRankInfo.szServerName)
  tbElement.pPanel:Label_SetText("txtTeamName", tbRankInfo.szName)
  tbElement.pPanel:Label_SetText("txtScore", tbRankInfo.nScore)
end
