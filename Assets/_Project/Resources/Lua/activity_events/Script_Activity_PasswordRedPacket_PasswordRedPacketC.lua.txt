local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.PasswordRedPacket)
local Protocol = {
  [tbActivity.Protocol.OnLightRedPoint] = "CheckLightRedPoint_Client"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not Protocol or not Protocol[nProtocol] then
    LogErr("[PasswordRedPacket ] ServerRequest error", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:CheckLightRedPoint_Client(bOpen)
  if bOpen then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.PassWordRedPacket)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.PassWordRedPacket)
  end
end

function tbActivity:OnClose()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.PassWordRedPacket)
end
