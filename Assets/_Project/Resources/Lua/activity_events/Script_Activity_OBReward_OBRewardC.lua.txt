Require("CommonScript/Activity/OBReward/OBReward.lua")
local tbActivity = ActivityMgr:GetClass("OBReward")
tbActivity.tbCacheData = {}

function tbActivity:LoadSyncData(tbData)
  self.tbCacheData = tbData
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_OBACTIVITY_UPDATE, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_OBACTIVITY_UPDATE, self.CheckRedPoint, self)
  EventNotify.OnNotify(EventNotify.emNOTIFY_OBACTIVITY_UPDATE)
end

function tbActivity:GetPreviewReward()
  return Award:GetAward(self.tbConfig.nAwardId, self.tbConfig.nAwardKey)
end

function tbActivity:HasGotten()
  return self.tbCacheData.bHasGotten
end

function tbActivity:ServerRequest(tbData)
  self.tbCacheData.bHasGotten = tbData.bHasGotten
  EventNotify.OnNotify(EventNotify.emNOTIFY_OBACTIVITY_UPDATE)
end

function tbActivity:CheckRedPoint()
  if Lib:IsEmptyStr(self.tbConfig.szRedPoint) then
    return
  end
  if self:HasGotten() then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
  else
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT[self.tbConfig.szRedPoint])
  end
end

function tbActivity:OnClientClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_OBACTIVITY_UPDATE, self)
end
