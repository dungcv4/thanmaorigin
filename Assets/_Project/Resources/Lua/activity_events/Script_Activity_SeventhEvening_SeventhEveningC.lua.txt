Require("Setting/Npc/Ai/SeventhEvening/TongXinShuCfg.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.SeventhEvening)
local nTongXinItemId = Npc.TongXinShu.Cfg.nSeedWaterTemplateId

function tbActivity:OnOpen()
  self:CheckRedPoint()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, self.CheckRedPoint, self)
end

function tbActivity:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, self)
end

function tbActivity:CheckRedPoint()
  local nLastStartTime = LocalData:GetRoleNumber("SeventhEvening", 0)
  if ActivityMgr:CheckExpired(nLastStartTime, self.nStartTime) and not RedPointMgr:CheckEvent(RedPointMgr.EVENT.SeventhFirstOpen) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.SeventhFirstOpen)
  end
  local nCount = me.GetItemCountInBags(nTongXinItemId)
  if nCount > 0 and not RedPointMgr:CheckEvent(RedPointMgr.EVENT.SeventhPlantTree) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.SeventhPlantTree)
  end
end

function tbActivity:DoCommonJump()
  local tbUi = Ui:GetWindow("UIQiXiActivity")
  if tbUi then
    tbUi:DealJumpTo(self:GetMapInfoList())
  end
  if RedPointMgr:CheckEvent(RedPointMgr.EVENT.SeventhPlantTree) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.SeventhPlantTree)
  end
end

function tbActivity:GetMapInfoList()
  local tbList = {}
  for nMapId, tbInfo in pairs(self.tbConfig.tbMapLocInfo) do
    table.insert(tbList, {
      nMapTemplateId = nMapId,
      nPosX = tbInfo.nLocX,
      nPosY = tbInfo.nLocY,
      szName = Map:GetMapName(nMapId)
    })
  end
  return tbList
end
