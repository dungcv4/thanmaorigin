local tbWnd = Ui:CreateClass("UIGuMuActivity", "UICommonActWndBase")
local tbControls = {
  btnClose = "node/imgBg/btnClose",
  txtTitle = "node/imgBg/Title/txtTitle",
  txtSmallTitle = "node/imgBg/txtTime",
  wndActivityiTips = "content/wndActivityiTips",
  btnTemplate = "node/imgBg/btnList/Viewport/Content/btnTemplate",
  ActivityInfo = "content/wndActivityiTips/imgBg/ActivityInfo",
  btnJoin = "content/wndActivityiTips/imgBg/btnJoin",
  ItemTemplate = "content/wndActivityiTips/imgBg/ActivityInfo/AwardList/Viewport/Content/ItemTemplate",
  Award = "node/imgBg/Award",
  AwardList = "node/imgBg/Award/AwardList",
  btnGet = "node/imgBg/Award/btnGet",
  btnHelp = "node/imgBg/Title/btnHelp",
  btnWeb = "content/btnWeb",
  wndJumpTo = "content/wndJumpTo",
  MapTemplate = "content/wndJumpTo/imgBG/List/Viewport/Content/Template",
  timeWaring = "content/wndJumpTo/imgBG/List/timeWaring"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:CloseByDefault()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("GuMuImmortal")
  end,
  btnWeb = function(self)
    Ui:OpenWindow("UIGuMuActivityReward", 94)
  end
}

function tbWnd:OnCreate()
  local tbParent = Ui:GetClass(self.__Parent)
  tbParent.OnCreate(self)
  self.pMapListScorllView = Ui:NewScrollView(self, tbControls.MapTemplate, function(_, nIndex, tbElement)
    self:OnClickArrowBtn(tbElement.nMapId)
  end, false, false)
end

function tbWnd:OnOpen(...)
  local tbParent = Ui:GetClass(self.__Parent)
  tbParent.OnOpen(self, ...)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnWeb, {
    RedPointMgr.EVENT.GuMuExtraAward
  })
end

function tbWnd:OnClickArrowBtn(nMapId)
  ActivityMgr:GotoMap(nMapId)
  self:Close()
end

function tbWnd:OnClickEmptyClose()
  local tbParent = Ui:GetClass(self.__Parent)
  tbParent.OnClickEmptyClose(self)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
end

function tbWnd:OnHandleExtraParam(nLinkActivityId)
  self:ShowJumpToTips(nLinkActivityId)
end

function tbWnd:ShowJumpToTips(nActivityId)
  local bIsNpcAppear = false
  local pMiQingInstance = ActivityMgr:GetInstanceById(nActivityId)
  if nil ~= pMiQingInstance and pMiQingInstance.IsInActiveTime then
    bIsNpcAppear = pMiQingInstance:IsInActiveTime()
  end
  self.pPanel:SetActive(tbControls.wndJumpTo, true)
  self.pPanel:SetActive(tbControls.timeWaring, not bIsNpcAppear)
  self:InitMapListScorllView(nActivityId)
end

function tbWnd:InitMapListScorllView(nLinkActivityId)
  self.pMapListScorllView:Clear()
  local pMiQingInstance = ActivityMgr:GetInstanceById(nLinkActivityId)
  if nil == pMiQingInstance then
    return
  end
  local tbMapList = pMiQingInstance:GetMapList()
  local nIndex = 1
  for i, nMapId in pairs(tbMapList) do
    self.pMapListScorllView:SetElement(nIndex, function(tbElement)
      self:SetMapListScorllViewElementInfo(tbElement, nMapId)
    end)
    nIndex = nIndex + 1
  end
end

function tbWnd:SetMapListScorllViewElementInfo(tbElement, nMapId)
  tbElement.pPanel:Label_SetText("txtMapName", i18n.EasyGet("{0}", i18n.Parse(Map:GetMapSetting(nMapId).MapName)))
  tbElement.nMapId = nMapId
end

function tbWnd:DealActivityDetail(tbInfo)
  self.pPanel:SetActive(self.wndActivityiTips, true)
  self.pPanel:SetActive(self.wndActivityiTips .. "/imgBg", true)
  self.pPanel:SetActive(self.ActivityInfo .. "/txtActDegreeTip", true)
  self.pPanel:SetActive(self.ActivityInfo .. "/txtActDegree", true)
  self.pPanel:Label_SetText(self.ActivityInfo .. "/txtLevelLimit", tbInfo.nLevel)
  self.pPanel:Label_SetText(self.ActivityInfo .. "/txtLevelLimit", tbInfo.nLevel)
  self.pPanel:Label_SetText(self.wndActivityiTips .. "/imgBg/txtActivityName", tbInfo.szName)
  self.pPanel:Label_SetText(self.ActivityInfo .. "/ActivityDescScroll/Viewport/Content/txtActivityDesc", tbInfo.szDesc)
  self.pPanel:Label_SetText(self.ActivityInfo .. "/txtActDegreeTip", i18n.EasyGet("\230\180\187\229\138\168\230\151\182\233\151\180"))
  self.pPanel:Label_SetText(self.ActivityInfo .. "/txtActDegree", tbInfo.szShowTime)
  self.tbShowAwardScrollView:UpdateGenericItem(tbInfo.tbAward, true)
  self.tbShowAwardScrollView:ResetParent()
  if 100 == tbInfo.nId then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.GuMuMiQingTreeAppear)
  end
  if not tbInfo.pFunc then
    self.pPanel:SetActive(self.btnJoin, false)
  else
    self.pPanel:SetActive(self.btnJoin, true)
    Ui:Button_BindEvent(self.pPanel, self.btnJoin, self.OnClickJoinBtn, self, tbInfo.pFunc, tbInfo.nId, table.unpack(Lib:SplitStrTry2Num(tbInfo.szJumpParam, "|")))
    RedPointMgr:DynamicUnRegist(self.pPanel, self.btnJoin)
    if tbInfo.tbRedPointEventId then
      if 100 == tbInfo.nId then
        local tbTmp = Lib:GetSubTable(tbInfo.tbRedPointEventId, {
          RedPointMgr.EVENT.GuMuMiQingTreeAppear
        })
        RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnJoin, tbTmp)
      else
        RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnJoin, tbInfo.tbRedPointEventId)
      end
    end
  end
  self:SetGotoBtn(tbInfo.szClassName, tbInfo.szGotoText, tbInfo.bGotoEnable)
end
