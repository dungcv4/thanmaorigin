local tbWnd = Ui:CreateWindow("UIActivity_JiangHu")
tbWnd.tbControls = {
  btnHelp = "node/imgBg/Title/btnHelp",
  btnClose = "node/imgBg/btnClose",
  txtSlogan = "node/imgBg/txtSlogan",
  txtRemainCoinNumber = "node/imgBg/content/txtRemainReturn/RemainCoin/txtRemainCoinNumber",
  txtNowCoinNumber = "node/imgBg/content/txtNowReturn/NowCoin/txtNowCoinNumber",
  btnGet = "node/imgBg/content/btnGet",
  txtGet = "node/imgBg/content/btnGet/Text"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("FriendsReturnActivityHelp")
  end,
  btnGet = function(self)
    self:OnGet()
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.ReturnerVNVipSignIn)
  end
}

function tbWnd:OnCreate()
  self:UnRegistRedPoint()
end

function tbWnd:OnOpen()
  self:RegistRedPoint()
  local nState = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipAwardCheck)
  local szServerName = FriendsReturn.szOldServerName
  local szPlayerName = FriendsReturn.szOldPlayerName
  self.pPanel:Label_SetText(self.txtSlogan, i18n.EasyGet("\231\148\177\228\186\142\229\164\167\228\190\160\229\156\168{0}\230\156\141\229\138\161\229\153\168\231\154\132\232\167\146\232\137\178{1}\229\183\178\231\187\143\232\191\158\231\187\17330\229\164\169\230\156\170\231\153\187\229\189\149\239\188\140\231\187\145\229\174\154\230\150\176\230\156\141\232\167\146\232\137\178\229\141\179\229\143\175\230\191\128\230\180\187\229\133\131\229\174\157\231\166\143\229\136\169\239\188\129", szServerName, szPlayerName))
  if 1 == nState then
    local nTotalYuanbaoCount = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipTotalYuanbaoCount)
    local nYuanbaoGetCount = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoGetCount)
    local nLeftYuanbaoCount = nTotalYuanbaoCount - nYuanbaoGetCount
    local nYuanbaoGetPerTime = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoGetPerTime)
    local nLastGetTime = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoCanGetTime)
    self.pPanel:Label_SetText(self.txtRemainCoinNumber, nLeftYuanbaoCount)
    self.pPanel:Label_SetText(self.txtNowCoinNumber, nYuanbaoGetPerTime)
    local nCurTime = GetTime()
    if not Lib:IsDiffDay(14400, nCurTime, nLastGetTime) then
      self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\228\187\138\230\151\165\229\183\178\233\162\134\229\143\150"))
      self.pPanel:Button_SetEnable(self.btnGet, false)
    else
      self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\233\162\134\229\143\150"))
      self.pPanel:Button_SetEnable(self.btnGet, true)
    end
  elseif 3 == nState then
    local nVipLevel = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipVIPLevel)
    local nTotalYuanbaoCount = FriendsReturn.tbVNVipAwardInfo[nVipLevel].nYuanBaoTotalNum
    local nYuanbaoGetPerTime = FriendsReturn.tbVNVipAwardInfo[nVipLevel].nYuanBaoGetPerDay
    self.pPanel:Label_SetText(self.txtRemainCoinNumber, nTotalYuanbaoCount)
    self.pPanel:Label_SetText(self.txtNowCoinNumber, nYuanbaoGetPerTime)
    self.pPanel:Label_SetText(self.txtGet, i18n.EasyGet("\231\187\145\229\174\154\230\173\164\232\167\146\232\137\178"))
  end
end

function tbWnd:OnGet()
  local nState = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipAwardCheck)
  if 1 == nState then
    FriendsReturn:GetVNVipNewServerYuanBaoReq()
  elseif 3 == nState then
    local function fnCardPick()
      FriendsReturn:BecomeVNVipReq()
    end
    
    Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\228\184\186<color=#0A6F32>{0}</color>\230\156\141<color=#0A6F32>{1}</color>\232\167\146\232\137\178\231\187\145\229\174\154\229\134\141\233\151\175\230\177\159\230\185\150\230\150\176\230\156\141\231\137\185\230\157\131\239\188\159\239\188\136\231\161\174\232\174\164\229\144\142\230\151\160\230\179\149\230\146\164\233\148\128\239\188\140\230\151\160\230\179\149\230\155\180\230\141\162\232\167\146\232\137\178\239\188\137", Login.ServerName, me.szName), {
      {fnCardPick},
      {}
    }, {
      i18n.szOk,
      i18n.szCancel
    })
  end
end

function tbWnd:RegistRedPoint()
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnGet, {
    RedPointMgr.EVENT.ReturnerVNVipSignIn
  })
end

function tbWnd:UnRegistRedPoint()
  RedPointMgr:DynamicUnRegist(self.pPanel, self.btnGet)
end
