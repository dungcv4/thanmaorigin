function RedPacketMgr:LoadSetting()
  self.tbEventType = Lib:LoadTabFileEx("Setting/RedPacket/RedPacketSource.tab", "dssssd", "nTypeId", {
    "nTypeId",
    "szMark",
    "szRecordFormat",
    "szRelation",
    "szRepeat",
    "nSaveId"
  })
  local tbOrgSetting = Lib:LoadTabFileEx("Setting/RedPacket/RedPacket.tab", "ddsdsdsddds", "nEventId", {
    "nEventId",
    "nTypeId",
    "szTypeName",
    "nTarget",
    "szMoneyType",
    "nBaseNum",
    "szAddGoldList",
    "nCount",
    "nAverageMin",
    "nSpecialType",
    "szChatInfo"
  })
  local tbSettings = {}
  for nEventId, tb in pairs(tbOrgSetting) do
    local tbAddGoldList = Lib:SplitStr2Num(tb.szAddGoldList, "_")
    table.sort(tbAddGoldList)
    assert(tb.nBaseNum and tb.nBaseNum > 0)
    tbSettings[nEventId] = {
      nEventId = tb.nEventId,
      nTypeId = tb.nTypeId,
      szTypeName = tb.szTypeName,
      nTarget = tb.nTarget,
      szMoneyType = tb.szMoneyType,
      nBaseNum = tb.nBaseNum,
      nCount = tb.nCount,
      nAverageMin = tb.nAverageMin,
      tbAddAmountList = tbAddGoldList,
      nSpecialType = tb.nSpecialType,
      szChatInfo = tb.szChatInfo
    }
  end
  self.tbSettings = SetReadOnly(tbSettings)
  local tbTypeGroup = {}
  for nEventId, tb in pairs(tbSettings) do
    tbTypeGroup[tb.nTypeId] = tbTypeGroup[tb.nTypeId] or {}
    tbTypeGroup[tb.nTypeId][nEventId] = true
  end
  self.tbTypeGroup = SetReadOnly(tbTypeGroup)
end

RedPacketMgr:LoadSetting()

function RedPacketMgr:Test(nTestCount)
  for nEventId, tbInfo in ipairs(self.tbSettings) do
    self:CheckSetting(nTestCount, tbInfo.nBaseNum, tbInfo.nCount, tbInfo.nAverageMin, nEventId)
  end
  Log("Complete!")
end
