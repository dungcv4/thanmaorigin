Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.FieldBossAward)
tbActivity.SaveID = {LastStartTimeKey = 1, DailyAwardTimeKey = 2}

function tbActivity:LoadConfig()
  self.ACTIVITY_CONFIG = Lib:EasyLoadTabFile("Setting/Activity/Common/FieldBossAward/FieldBossAward.tab", {"nConfigId"})
  local tbMapInfo = Lib:EasyLoadTabFile("Setting/Activity/Common/FieldBossAward/MapInfo.tab")
  for _, tbInfo in ipairs(tbMapInfo) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("FieldBossAward:LoadConfig MapInfo nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbMap = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbMap or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbMap[tbInfo.nMapTemplateId] = tbInfo
  end
  local tbTips = Lib:EasyLoadTabFile("Setting/Activity/Common/FieldBossAward/Tips.tab")
  for _, tbInfo in ipairs(tbTips) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("FieldBossAward:LoadConfig Tips nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbTips = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbTips or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbTips, tbInfo)
  end
  local nFirstTime, nSecondTime
  for _, tbInfo in pairs(self.ACTIVITY_CONFIG) do
    for _, tbList in pairs(tbInfo.tbTips) do
      table.sort(tbList, function(tbFirst, tbSecond)
        nFirstTime = Lib:IsEmptyStr(tbFirst.szTimeFrameStart) and 0 or Lib:CalcTimeFrameOpenTime(tbFirst.szTimeFrameStart)
        nSecondTime = Lib:IsEmptyStr(tbSecond.szTimeFrameStart) and 0 or Lib:CalcTimeFrameOpenTime(tbSecond.szTimeFrameStart)
        return nFirstTime > nSecondTime
      end)
    end
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
end

function tbActivity:CanJoin(nLevel)
  return nLevel >= self.tbConfig.nJoinLevel
end
