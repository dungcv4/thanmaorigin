Require("CommonScript/Activity/OBTreasureRabbit/OBTreasureRabbit.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szOBTreasureRabbitClassName)

function tbActivity:OnModify(tbScheduleCfg)
  self.bLoadConfig = false
  if not self:CheckConfig(tbScheduleCfg.nConfigId) then
    return
  end
  self:InitMetaData(tbScheduleCfg)
  self.SaveKey = GetGroupKey(self.tbConfig.szSaveKey)
end

function ActivityMgr:GetOBTreasureRabbitLastNumOfUsingBox()
  local nNum, pInstance = 0
  for nId in pairs(self.tbClassName2Id[self.szOBTreasureRabbitClassName] or {}) do
    pInstance = self:GetInstanceById(nId)
    if pInstance and pInstance:IsOpen() then
      nNum = nNum + pInstance:GetLastNumOfUsingBox(me)
    end
  end
  return nNum
end

function ActivityMgr:GetOBTreasureRabbitLastNumOfUsingBag()
  local nNum, pInstance = 0
  for nId in pairs(self.tbClassName2Id[self.szOBTreasureRabbitClassName] or {}) do
    pInstance = self:GetInstanceById(nId)
    if pInstance and pInstance:IsOpen() then
      nNum = nNum + pInstance:GetLastNumOfUsingBag(me)
    end
  end
  return nNum
end

function ActivityMgr:GetOBTreasureRabbitInstance()
  local pInstance
  for nId in pairs(self.tbClassName2Id[self.szOBTreasureRabbitClassName] or {}) do
    pInstance = self:GetInstanceById(nId)
    if pInstance and pInstance:IsOpen() then
      return pInstance
    end
  end
  return nil
end
