SevenDayWelfare.DAY = 10
SevenDayWelfare.AWARDNUM = 2
SevenDayWelfare.nOPENLEVEL = 10
SevenDayWelfare.AWARDID = 1055
SevenDayWelfare.nSaveGroupId = GetGroupKey("SevenDayWelfare")

function SevenDayWelfare:CheckCanGetAward(pPlayer, nDay)
  if nDay > Server:GetServerOpenDay() then
    pPlayer.CenterMsg(i18n.EasyGet("\229\165\150\229\138\177\230\156\170\229\188\128\230\148\190!"))
    return false
  end
  local SevenDayBindData = ScriptData:GetValue("SevenDayAccountBind", 0, {})
  if not SevenDayBindData then
    Log("SevenDayWelfareGetRequest SevenDayBindData nil !!")
    return false
  end
  local BindRoleId = SevenDayBindData[pPlayer.szAccount]
  if not BindRoleId then
    SevenDayBindData[pPlayer.szAccount] = pPlayer.dwID
    pPlayer.CenterMsg(i18n.EasyGet("\231\187\145\229\174\154\230\136\144\229\138\159!"))
    ScriptData:SaveValue("SevenDayAccountBind", 0, SevenDayBindData)
  elseif BindRoleId ~= pPlayer.dwID then
    pPlayer.CenterMsg(i18n.EasyGet("\232\180\166\229\143\183\231\154\132\229\133\182\228\187\150\232\167\146\232\137\178\229\183\178\231\187\143\231\187\145\229\174\154\232\191\135\239\188\140\230\151\160\230\179\149\233\162\134\229\143\150\239\188\129"))
    return false
  end
  return true
end

function SevenDayWelfare:AccountHasBind(pPlayer)
  local SevenDayBindData = ScriptData:GetValue("SevenDayAccountBind", 0, {})
  if not SevenDayBindData then
    Log("SevenDayWelfareGetRequest SevenDayBindData nil !!")
    return false
  end
  local BindRoleId = SevenDayBindData[pPlayer.szAccount]
  if BindRoleId then
    return true
  else
    return false
  end
end

function SevenDayWelfare:GetBindRoleId(pPlayer)
  if SevenDayWelfare:AccountHasBind(pPlayer) then
    local SevenDayBindData = ScriptData:GetValue("SevenDayAccountBind", 0, {})
    if not SevenDayBindData then
      Log("SevenDayWelfareGetRequest SevenDayBindData nil !!")
      return
    end
    return SevenDayBindData[pPlayer.szAccount]
  end
  return
end

function SevenDayWelfare:ClearBindRoleId(pPlayer)
  if SevenDayWelfare:AccountHasBind(pPlayer) then
    local SevenDayBindData = ScriptData:GetValue("SevenDayAccountBind", 0, {})
    if not SevenDayBindData then
      Log("SevenDayWelfareGetRequest SevenDayBindData nil !!")
      return
    end
    SevenDayBindData[pPlayer.szAccount] = nil
  end
end

function SevenDayWelfare:ClearGetAwardData(pPlayer)
  for nDay = 1, SevenDayWelfare.DAY do
    pPlayer.SetUserValue(SevenDayWelfare.nSaveGroupId, nDay, 0)
  end
  SevenDayWelfare:ClearBindRoleId(pPlayer)
  SevenDayWelfare:RefreshGetAwardData(pPlayer)
end

function SevenDayWelfare:ClearGetAwardDataById(dwPlayerId)
  local pPlayer = KPlayer.GetPlayerObjById(dwPlayerId)
  SevenDayWelfare:ClearGetAwardData(pPlayer)
end
