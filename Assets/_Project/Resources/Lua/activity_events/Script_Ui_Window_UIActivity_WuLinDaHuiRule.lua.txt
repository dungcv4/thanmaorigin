local tbWnd = Ui:CreateWindow("UIActivity_WuLinDaHuiRule")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/txtTitle",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  UITree = "node/imgBG/Content/Left/TypeList/UITree",
  Awards = "node/imgBG/Content/Awards",
  txtAwardTitle = "node/imgBG/Content/Awards/txtAwardTitle",
  txtSlogan = "node/imgBG/Content/Awards/txtSlogan",
  rewardRound = "node/imgBG/Content/Awards/rewardRound",
  winItem = "node/imgBG/Content/Awards/rewardRound/win/UIItemGrid",
  defeatItem = "node/imgBG/Content/Awards/rewardRound/defeat/UIItemGrid",
  AwardsListPlayer = "node/imgBG/Content/Awards/AwardsListPlayer",
  Template = "node/imgBG/Content/Awards/AwardsListPlayer/Viewport/Content/Template",
  NotInfo = "node/imgBG/Content/Awards/NotInfo",
  txtMyRank = "node/imgBG/Content/Awards/txtMyRank",
  txtRewardTips = "node/imgBG/Content/Awards/txtRewardTips",
  Rule = "node/imgBG/Content/Rule",
  txtRuleTitle = "node/imgBG/Content/Rule/txtRuleTitle",
  txtTime = "node/imgBG/Content/Rule/timeCell/txtTime",
  txtTimeCountDown = "node/imgBG/Content/Rule/timeCell/txtTimeCountDown",
  txtContent = "node/imgBG/Content/Rule/contentScrollView/Viewport/Content/txtContent",
  btnWay = "node/imgBG/Content/Rule/btnWay"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
  end
}
tbWnd.TREE_BIG_INDEX = {RULE = 1, AWARD = 2}

function tbWnd:GetTreeData()
  local tbUITreeBigData = {}
  local tbUITreeSmallData = {}
  for nIndex, tbData in ipairs(self.tbSetting.tbBigData) do
    table.insert(tbUITreeBigData, {
      nType = nIndex,
      szText = i18n.Get(tbData.szText),
      tbInfo = {nOnClickIndex = nIndex}
    })
  end
  for nIndex, tbData in ipairs(self.tbSetting.tbSmallData) do
    table.insert(tbUITreeSmallData, {
      nType = tbData.nBigType,
      szText = i18n.Get(tbData.szText),
      tbInfo = {
        nOnClickIndex = nIndex,
        nBigIndex = tbData.nBigType
      }
    })
  end
  return tbUITreeBigData, tbUITreeSmallData
end

function tbWnd:OnCreate()
  self.uiTree = Ui:NewTree(self, tbControls.UITree, self.onClickTreeBigBtn, self.onClickTreeSmallBtn)
  self.pWinItemGrid = Ui:NewExistItemGrid(self, tbControls.winItem)
  self.pDefeatItemGrid = Ui:NewExistItemGrid(self, tbControls.defeatItem)
  self.ScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen(tbSetting, nBigIndex)
  self.tbSetting = tbSetting
  local tbBigData, tbSmallData = self:GetTreeData()
  self.uiTree:SetData(tbBigData, tbSmallData)
  self.uiTree:SelectNode(nBigIndex, 1, true)
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.pPanel:Label_SetText(tbControls.txtTitle, tbSetting.szTitle)
end

function tbWnd:OnClose()
end

function tbWnd:onClickTreeBigBtn(tbInfo)
  self.uiTree:AutoSelectSmall(tbInfo.nOnClickIndex)
end

function tbWnd:onClickTreeSmallBtn(tbInfo)
  local nOnClickIndex = tbInfo.nOnClickIndex
  if nil ~= nOnClickIndex and self.nPanelIndex ~= nOnClickIndex then
    self.nPanelIndex = nOnClickIndex
    if tbInfo.nBigIndex == self.TREE_BIG_INDEX.AWARD then
      self:UpdateAwardPanel()
    elseif tbInfo.nBigIndex == self.TREE_BIG_INDEX.RULE then
      self:UpdateRulePanel()
    end
  end
end

function tbWnd:UpdateAwardPanel()
  self.pPanel:SetActive(tbControls.Awards, true)
  self.pPanel:SetActive(tbControls.Rule, false)
  self.pPanel:SetActive(tbControls.AwardsListPlayer, false)
  self.pPanel:SetActive(tbControls.txtMyRank, false)
  self.pPanel:SetActive(tbControls.NotInfo, true)
  local tbSetting = self.tbSetting.tbSmallData[self.nPanelIndex]
  if tbSetting then
    self.pPanel:Label_SetText(tbControls.txtRuleTitle, i18n.Get(tbSetting.szText))
    self.pPanel:Label_SetText(tbControls.txtAwardTitle, i18n.Get(tbSetting.szText))
    if tbSetting.tbWinAwardData and tbSetting.tbDefeatAwardData then
      self.pPanel:SetActive(tbControls.rewardRound, true)
      local tbWinAwards = Award:GetAward(tbSetting.tbWinAwardData.nAwardId, tbSetting.tbWinAwardData.nAwardKey, {})
      local tbDefeatAwards = Award:GetAward(tbSetting.tbDefeatAwardData.nAwardId, tbSetting.tbDefeatAwardData.nAwardKey, {})
      self.pWinItemGrid:SetGenericItem(tbWinAwards[1])
      self.pWinItemGrid:SetClickEnable(true)
      self.pDefeatItemGrid:SetGenericItem(tbDefeatAwards[1])
      self.pDefeatItemGrid:SetClickEnable(true)
    else
      self.pPanel:SetActive(tbControls.rewardRound, false)
    end
    if tbSetting.tbRankAwards then
      self.pPanel:SetActive(tbControls.AwardsListPlayer, true)
      self.pPanel:SetActive(tbControls.NotInfo, false)
      self.ScrollView:Reset()
      for i, tbAward in ipairs(tbSetting.tbRankAwards) do
        self.ScrollView:SetElement(i, function(tbElement)
          tbElement.pPanel:Label_SetText("txtRank", i18n.Get(tbAward.szDesc))
          local tbAwards = Award:GetAward(tbAward.nAwardId, tbAward.nAwardKey)
          local tbItemList = tbElement.tbItemListGroup or Ui:NewItemGridView(tbElement, "Awards/ItemTemplate")
          tbItemList:Clear()
          tbElement.tbItemListGroup = tbItemList
          for j = 1, #tbAwards do
            tbItemList:SetItem(j, tbAwards[j], true)
          end
          tbItemList:ResetFrom(#tbAwards + 1)
        end)
      end
    end
    if tbSetting.szRewardTips then
      self.pPanel:Label_SetText(tbControls.txtRewardTips, i18n.Get(tbSetting.szRewardTips))
    end
    if tbSetting.nRank then
      self.pPanel:SetActive(tbControls.txtMyRank, true)
      self.pPanel:Label_SetText(tbControls.txtMyRank, i18n.EasyGet("\230\136\145\231\154\132\230\142\146\229\144\141\239\188\154{0}", tbSetting.nRank))
    end
  end
end

function tbWnd:UpdateRulePanel()
  self.pPanel:SetActive(tbControls.Awards, false)
  self.pPanel:SetActive(tbControls.Rule, true)
  local tbSetting = self.tbSetting.tbSmallData[self.nPanelIndex]
  if tbSetting then
    self.pPanel:Label_SetText(tbControls.txtRuleTitle, i18n.Get(tbSetting.szText))
    local nStartTime = Lib:GetConfigDate2Time(tbSetting.nStartTime)
    local nEndTime = Lib:GetConfigDate2Time(tbSetting.nEndTime)
    self.pPanel:Label_SetText(tbControls.txtTime, string.format("%s-%s", Lib:TimeDesc13(nStartTime), Lib:TimeDesc13(nEndTime)))
    if nEndTime > GetTime() then
      self.pPanel:Label_SetText(tbControls.txtTimeCountDown, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154{0}", Lib:TimeDesc2(nEndTime - GetTime())))
    else
      self.pPanel:Label_SetText(tbControls.txtTimeCountDown, i18n.EasyGet("\229\183\178\231\187\147\230\157\159"))
    end
    self.pPanel:Label_SetText(tbControls.txtContent, i18n.Get(tbSetting.szDesc))
    if tbSetting.tbCallBack then
      self.pPanel:SetActive(tbControls.btnWay, true)
      Ui:Button_BindEvent(self.pPanel, tbControls.btnWay, function()
        Lib:CallBack(tbSetting.tbCallBack)
      end)
    else
      self.pPanel:SetActive(tbControls.btnWay, false)
    end
  end
end
