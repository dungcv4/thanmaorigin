local tbWnd = Ui:CreateClass("UIHandInActivity", "UIPanel")
local tbControls = {
  RightStepTemplate = "Right/List/Viewport/Content/Template",
  btnHelp = "Right/txtTitle/btnHelp",
  txtTitle = "Right/txtTitle",
  SingleHandInAwardTemplate = "Left/List01/List/Viewport/Content/Template",
  MultiHandInAwardTemplate = "Left/List02/List/Viewport/Content/Template",
  txtTime01 = "Right/txtTime01",
  txtLevel = "Right/txtLV",
  btnGift = "Right/btnGift",
  txtSecondTittleUp = "Left/List01/txtSecondTittle",
  txtSecondTittleDown = "Left/List02/txtSecondTittle"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnGift = function(self)
    self:OnClickGift()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.tbRightStepListView = Ui:NewScrollView(self, tbControls.RightStepTemplate)
  self.tbSingleHandInAwardView = Ui:NewItemGridView(self, tbControls.SingleHandInAwardTemplate)
  self.tbMultiHandInAwardView = Ui:NewItemGridView(self, tbControls.MultiHandInAwardTemplate)
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.nIndex = nIndex
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  self.szHelpKey = tbInfo.szHelpKey
  self.pInstance = ActivityMgr:GetInstanceById(tbInfo.nId)
  local tbExtraData = ActivityMgr.EntranceSchedule:GetExtraData(self.nEntranceIndex, tbInfo.nIndex)
  self.bAchieveYuanBaoAward = tbExtraData.bAchieveYuanBaoAward or false
  self.nAchieveActId = tbExtraData.nAchieveActId
  self.pPanel:Label_SetText(tbControls.txtSecondTittleUp, i18n.Get(tbExtraData.txtSecondTittleUp))
  self.pPanel:Label_SetText(tbControls.txtSecondTittleDown, i18n.Get(tbExtraData.txtSecondTittleDown))
  self.pPanel:Label_SetText(tbControls.txtTitle, tbInfo.szName)
  if self.nAchieveActId then
    local pAchieveInstance = ActivityMgr:GetInstanceById(self.nAchieveActId)
    if pAchieveInstance then
      RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGift, pAchieveInstance:GetRedPointEventId())
    end
  end
  self:UpdateListView(tbInfo)
  self:UpdateAwardPreview(tbInfo)
  self:UpdateSecondAwardPreview(tbExtraData)
  self:UpdateRightInfo(tbInfo)
end

function tbWnd:UpdateRightInfo(tbInfo)
  if not self.pInstance then
    return
  end
  local nRestTime = self.pInstance.nEndTime - GetTime()
  self.pPanel:Label_SetText(tbControls.txtTime01, i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", tbInfo.szShowTime, Lib:TimeDesc2(nRestTime)))
  self.pPanel:Label_SetText(tbControls.txtLevel, i18n.EasyGet("\231\173\137\231\186\167\233\153\144\229\136\182\239\188\154{0}", tbInfo.nLevel))
end

function tbWnd:UpdateListView(tbInfo)
  local tbStepCfg = ActivityMgr.EntranceSchedule:GetStepList(self.nEntranceIndex, tbInfo.nIndex)
  self.tbRightStepListView:Clear()
  for nStep, tbStepInfo in ipairs(tbStepCfg) do
    self.tbRightStepListView:SetElement(nStep, function(tbElement, nIndex)
      self:UpdateListElement(tbElement, nStep, tbStepInfo)
    end)
  end
end

function tbWnd:UpdateListElement(tbElement, nIndex, tbStepInfo)
  tbElement.pPanel:Label_SetText("ImgAll/txtNum", nIndex)
  tbElement.pPanel:Label_SetText("ImgAll/txtInfo", i18n.Get(tbStepInfo.szInfo))
  tbElement.pPanel:Button_SetText("ImgAll/btnGo", i18n.Get(tbStepInfo.szBtnName), "Text")
  tbElement.pPanel:Button_BindEvent("ImgAll/btnGo", self.OnStepGoBtnClick, self, tbStepInfo.szCallBack, tbStepInfo.szParam)
  if not Lib:IsEmptyStr(tbStepInfo.szRedPoint) then
    local tbRedPointEvent = Lib:SplitStr(tbStepInfo.szRedPoint, "|")
    if tbRedPointEvent and #tbRedPointEvent > 0 then
      RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "ImgAll/btnGo", ActivityMgr:GenRedPointEventIdList(tbRedPointEvent))
    end
  end
end

function tbWnd:OnStepGoBtnClick(szCallBack, szParam)
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam or "", "|")
  if fn and tbParam and #tbParam > 0 then
    fn(self.tbActivityList[self.nIndex].nId, table.unpack(tbParam))
    if "GotoPos" == szCallBack or "GotoNPC" == szCallBack then
      Ui:CloseWindow("UIMayDayActivity")
    end
  end
end

function tbWnd:UpdateAwardPreview(tbInfo)
  if not self.pInstance then
    return
  end
  for nIndex, tbAward in ipairs(tbInfo.tbAward) do
    self.tbSingleHandInAwardView:SetItem(nIndex, tbAward, true)
    local szItemKey = self.tbSingleHandInAwardView:GetItemKey(nIndex)
    self.pPanel:Label_SetText(szItemKey .. "/txtActivityName", i18n.Parse(self:GetAwardName(tbAward, false)))
  end
end

function tbWnd:UpdateSecondAwardPreview(tbExtraData)
  local tbAwardList = Award:GetAward(tbExtraData.nComposeAwardId, tbExtraData.nComposeAwardKey)
  for nIndex, tbAward in ipairs(tbAwardList) do
    self.tbMultiHandInAwardView:SetItem(nIndex, tbAward, true)
    local szItemKey = self.tbMultiHandInAwardView:GetItemKey(nIndex)
    self.pPanel:Label_SetText(szItemKey .. "/txtActivityName", i18n.Parse(self:GetAwardName(tbAward, false)))
  end
end

function tbWnd:GetAwardName(tbAward, bNeedCount)
  local nAwardType = Player.AwardType[tbAward[1]]
  local szName = ""
  if 3 == Player.AwardTypeCountIndex[nAwardType] then
    szName = Item:GetName(tbAward[2], me)
  elseif nAwardType == Player.award_type_money then
    szName = Currency:GetName(tbAward[1])
  end
  if Lib:IsEmptyStr(szName) then
    return ""
  end
  if bNeedCount then
    szName = string.format("%sx%s", szName, Player:GetAwardValue(tbAward))
  end
  return szName
end

function tbWnd:OnClickGift()
  if 1 == self.bAchieveYuanBaoAward then
    Ui:OpenWindow("UIReward_Common", self.nAchieveActId)
    return
  end
  Ui:OpenWindow("UIReward_Newyear", self.nAchieveActId)
end
