Require("CommonScript/Activity/ActivityMgr.lua")
Require("CommonScript/Event/Event.lua")
ActivityMgr.szGradeRankingActivityClassName = "GradeRankingActivity"
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szGradeRankingActivityClassName)
tbActivity.tbTopList = {}
tbActivity.nConfigId = 1
tbActivity.tbRemainCount = {}
tbActivity.nFinishTime = nil
local tbS2CPotocolId = {
  SetTopList = 1,
  SetRemainCount = 2,
  SetFinishTime = 3
}
local Protocol = {
  [tbS2CPotocolId.SetTopList] = "SetTopList",
  [tbS2CPotocolId.SetRemainCount] = "SetRemainCount",
  [tbS2CPotocolId.SetFinishTime] = "SetFinishTime"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    me.CenterMsg(i18n.EasyGet("\229\174\162\230\136\183\231\171\175\232\190\147\229\133\165\233\148\153\232\175\175"))
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

tbActivity.tbS2CPotocolId = tbS2CPotocolId

function tbActivity:OnOpen()
  ActivityMgr:CallServerScript(self.nId)
end

function tbActivity:getAwardType(nRank)
  for i = 1, self.tbGlobalConfig.nAwardTypeCount do
    if nRank <= self.AWARD_CONFIG[i].nRankLowerBound then
      return i
    end
  end
  return -1
end

function tbActivity:_GetRemainCount()
  return self.tbRemainCount
end

function tbActivity:IsOpenAndShow()
  local nfinishTime = self.nFinishTime
  if not nfinishTime then
    return true
  end
  return nfinishTime > GetTime()
end

function tbActivity:IsServerOnly()
  return false
end

function tbActivity:SetTopList(tbTopList)
  self.tbTopList = tbTopList
  EventNotify.OnNotify(EventNotify.emNOTIFY_GRADERANKING_DATA_SYNC)
end

function tbActivity:SetRemainCount(tbRemainCount)
  self.tbRemainCount = tbRemainCount
  EventNotify.OnNotify(EventNotify.emNOTIFY_GRADERANKING_DATA_SYNC)
end

function tbActivity:SetFinishTime(nFinishTime)
  self.nFinishTime = nFinishTime
  EventNotify.OnNotify(EventNotify.emNOTIFY_GRADERANKING_DATA_SYNC)
end
