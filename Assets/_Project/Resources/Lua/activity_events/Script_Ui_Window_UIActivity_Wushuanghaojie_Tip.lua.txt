local tbWnd = Ui:CreateWindow("UIActivity_Wushuanghaojie_Tip")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/txtTitle",
  SliderNum = "node/imgBG/Content/SliderNum",
  txtTongName = "node/imgBG/Content/txtSetNumTip",
  txtFightPower = "node/imgBG/Content/txtSetNumValue",
  txtNowNum = "node/imgBG/Content/txtNowNum",
  txtMin = "node/imgBG/Content/txtMin",
  txtMax = "node/imgBG/Content/txtMax",
  btnAdd = "node/imgBG/Content/btnAdd",
  btnOK = "node/imgBG/Content/btnOK",
  txtInfo = "node/imgBG/Content/txtInfo",
  txtNum = "node/imgBG/Content/txtNum",
  Image = "node/imgBG/Content/Image",
  Img_Luogu = "node/imgBG/Content/Img_Luogu"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnAdd = function(self)
    self:AddItem()
  end,
  btnOK = function(self)
    if self.nSelectCount > 0 then
      self:ClickButtonOK()
    end
    self:Close()
  end
}
tbWnd.tbOnSliderClick = {
  SliderNum = function(self, nValue)
    self.nSelectCount = nValue
    self.pPanel:Label_SetText(tbControls.txtNum, string.format("%d/%d", self.nSelectCount, self.nMax))
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(nType, nTeamId)
  self.nType = nType
  self.nTeamId = nTeamId
  self.nSelectCount = 1
  local tbTeamData = WuLinDaHui.tbFinalTeamData[nTeamId]
  self.pPanel:Label_SetText(tbControls.txtTongName, tbTeamData.szTeamName)
  self.pPanel:Label_SetText(tbControls.txtFightPower, tbTeamData.nFightPower)
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\230\151\160\229\143\140\232\177\170\230\157\176\229\138\169\229\168\129"))
  self:Update()
end

function tbWnd:Update()
  local tbNowGuessCfg = WuLinDaHui:GetNowGuessCfg(WuLinDaHui.tbFinalMatchList)
  if not tbNowGuessCfg then
    return
  end
  if self.nType == WuLinDaHui.GUESS_TYPE.CHAMPION then
    self.nHaveNum = me.GetItemCountByTypeInBag(WuLinDaHui.GuessCfg.nChampionGuessID)
    self.nMax = self.nHaveNum
    self.pPanel:SetActive(tbControls.txtInfo, false)
    self.pPanel:SetActive(tbControls.Image, false)
    self.pPanel:SetActive(tbControls.Img_Luogu, true)
  elseif self.nType == WuLinDaHui.GUESS_TYPE.COMPETITION then
    self.nHaveNum = me.GetItemCountByTypeInBag(WuLinDaHui.GuessCfg.nMatchGuessID)
    self.nMax = math.min(self.nHaveNum, tbNowGuessCfg.MATCH_GUESS_MAX)
    self.pPanel:SetActive(tbControls.txtInfo, true)
    self.pPanel:SetActive(tbControls.Image, true)
    self.pPanel:SetActive(tbControls.Img_Luogu, false)
    self.pPanel:Label_SetText(tbControls.txtInfo, i18n.EasyGet("\230\175\148\232\181\155\231\171\158\231\140\156\230\156\128\229\164\154\229\143\175\230\138\149{0}\231\165\168", tbNowGuessCfg.MATCH_GUESS_MAX))
  end
  self.pPanel:Label_SetText(tbControls.txtNowNum, self.nHaveNum)
  self.pPanel:Label_SetText(tbControls.txtMin, 0)
  self.pPanel:Label_SetText(tbControls.txtMax, self.nMax)
  self.pPanel:Slider_SetRange(tbControls.SliderNum, 0, self.nMax)
  self.pPanel:Slider_SetValue(tbControls.SliderNum, self.nSelectCount)
  self.pPanel:Label_SetText(tbControls.txtNum, string.format("%d/%d", self.nSelectCount, self.nMax))
end

function tbWnd:AddItem()
  if self.nType == WuLinDaHui.GUESS_TYPE.CHAMPION then
    Shop:OpenShopBuyWidDefault(WuLinDaHui.GuessCfg.nChampionGuessID)
  elseif self.nType == WuLinDaHui.GUESS_TYPE.COMPETITION then
    Shop:OpenShopBuyWidDefault(WuLinDaHui.GuessCfg.nMatchGuessID)
  end
end

function tbWnd:ClickButtonOK()
  if self.nSelectCount <= 0 then
    me.CenterMsg(i18n.EasyGet("\231\171\158\231\140\156\230\149\176\233\135\143\228\184\141\232\131\189\228\184\1860"))
    return
  end
  if self.nType == WuLinDaHui.GUESS_TYPE.CHAMPION then
    WuLinDaHui:RequestGuessChampion(self.nTeamId, self.nSelectCount)
  elseif self.nType == WuLinDaHui.GUESS_TYPE.COMPETITION then
    WuLinDaHui:RequestGuessMatch(self.nTeamId, self.nSelectCount)
  end
end

function tbWnd:OnSyncItem(dwItemId, bNewItem, nAddCount)
  if dwItemId then
    local pItem = KItem.GetItemObj(dwItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId == WuLinDaHui.GuessCfg.nChampionGuessID or pItem.dwTemplateId == WuLinDaHui.GuessCfg.nMatchGuessID then
      self:Update()
    end
  end
end
