local tbActivity = ActivityMgr:GetClass(ActivityMgr.szNewServerNoviceGiftName)

function tbActivity:OnOpen()
  self:UpdateRedPoint()
end

function tbActivity:UpdateView()
  if Ui:WindowVisible("UINewInformation") == true then
    Ui:GetPanelByName("UINewInformation_NewServerNoviceGift"):UpdateView()
  end
end

function tbActivity:UpdateRedPoint()
  if self:IsOpen() and self:CheckHaveAwards(me) then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.NewServerNoviceGift)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.NewServerNoviceGift)
  end
end

function tbActivity:OnClose()
end
