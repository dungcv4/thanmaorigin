Require("CommonScript/Activity/NpcAward/NpcAward.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.YuTuNpcAward)
local Protocol = {
  [tbActivity.Protocol.OnSyncNpcPosition] = "OnSyncNpcPosition",
  [tbActivity.Protocol.OnSyncNpcMap] = "OnSyncNpcMap",
  [tbActivity.Protocol.OnSyncNpcMapList] = "OnSyncNpcMapList",
  [tbActivity.Protocol.OnGoToGatherMap] = "OnGoToGatherMap"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("[LaborHonor] ServerRequest protocol is wrong", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:GetMapShowInfo()
  local tbList = {}
  local tbHash = {}
  for _, tbInfo in ipairs(self.tbConfig.tbPosList) do
    if not tbHash[tbInfo.nMapTemplateId] then
      table.insert(tbList, {
        nMapTemplateId = tbInfo.nMapTemplateId,
        nPosX = tbInfo.nPosX,
        nPosY = tbInfo.nPosY,
        szName = Map:GetMapName(tbInfo.nMapTemplateId)
      })
      tbHash[tbInfo.nMapTemplateId] = true
    end
  end
  return tbList
end

function tbActivity:GetRedPointEventId()
  return {
    RedPointMgr.EVENT.ChildrenDoll
  }
end

function tbActivity:GetRedPointEventList()
  return self:GetRedPointEventId()
end

function tbActivity:OnSyncNpcPosition(nNpcTemplateId, nMapId, nPosX, nPosY)
  local function func()
    local nNpcId = KNpc.GetNpcIdByTemplateId(nNpcTemplateId)
    
    if nNpcId then
      Operation:SimpleTap(nNpcId)
    end
  end
  
  ActivityMgr:GotoDynamicNpc(nMapId, nPosX, nPosY, func, GetNpcDialogDistance(nNpcTemplateId))
end

function tbActivity:GoToNpcMap()
  ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.GoToNpcMap)
end

function tbActivity:OnGoToGatherMap(nMapTId, nPosX, nPosY)
  ActivityMgr:GotoDynamicNpc(nMapTId, nPosX, nPosY, function()
  end)
end

function tbActivity:OnSyncNpcMap(nNpcTemplateId, nMapId)
  ActivityMgr:GotoMap(nMapId)
end

function tbActivity:OnSyncNpcMapList(tbNpcMapList)
  if tbNpcMapList and next(tbNpcMapList) then
    self.tbNpcMapList = {}
    for nMapId, _ in pairs(tbNpcMapList) do
      table.insert(self.tbNpcMapList, nMapId)
    end
  else
    self.tbNpcMapList = nil
  end
end

function tbActivity:DoCommonJump()
  if not self.tbNpcMapList or not next(self.tbNpcMapList) then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\231\142\137\229\133\148\232\191\152\230\156\170\229\135\186\231\142\176\239\188\140\232\175\183\229\164\167\228\190\160\231\173\137\229\190\133\231\142\137\229\133\148\229\135\186\231\142\176\229\144\142\230\143\144\228\186\164"))
    return false
  end
  local nMapId = self.tbNpcMapList[MathRandom(#self.tbNpcMapList)]
  ActivityMgr:GotoMap(nMapId)
  return true
end

function tbActivity:GetJoinNum()
  return me.GetUserValue(self.SaveKey, self.SaveID.TodayNumKey)
end

function tbActivity:IsShowOpen()
  if not self.tbNpcMapList or not next(self.tbNpcMapList) then
    return false
  end
  return true
end
