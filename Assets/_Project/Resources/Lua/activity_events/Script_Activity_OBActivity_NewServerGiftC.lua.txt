Require("CommonScript/Activity/NewServerGift/NewServerGift.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.szNewServerGiftClassName)
tbActivity.tbCurGiftStatus = {}
tbActivity.nTimerId = 0

function tbActivity:OnOpen()
  self.bIsFirstLogin = true
  self.bisLight = false
  self:CreateTimer()
  for _, nConfigId in pairs(self.tbProductId2ConfigId) do
    local nOpenStatus = self:OpenStatusByConfigId(nConfigId)
    self.tbCurGiftStatus[nConfigId] = nOpenStatus
  end
  self:CheckShowRedPoint()
end

function tbActivity:CreateRechargeOrderReq(nProductId)
  if not me then
    return
  end
  if self:CheckIsBuy(me, self.tbProductId2ConfigId[nProductId]) then
    me.CenterMsg(i18n.EasyGet("\229\183\178\232\180\173\228\185\176\230\173\164\230\161\163\231\164\188\229\140\133,\228\184\141\229\143\175\233\135\141\229\164\141\232\180\173\228\185\176"))
    return
  end
  Recharge:RequestCreateRechargeOrder(nProductId)
end

function tbActivity:CheckShowRedPoint()
  local bHasNewGift = false
  local bHasCanBuy = false
  for nProductId, nConfigId in pairs(self.tbProductId2ConfigId) do
    local nOpenStatus = self:OpenStatusByConfigId(nConfigId)
    if nOpenStatus == self.GiftOpenStatus.Opening and self.tbCurGiftStatus[nConfigId] == self.GiftOpenStatus.NoOpen then
      bHasNewGift = true
      self.tbCurGiftStatus[nConfigId] = nOpenStatus
    end
    if nOpenStatus == self.GiftOpenStatus.Opening and not self:CheckIsBuy(me, self.tbProductId2ConfigId[nProductId]) then
      bHasCanBuy = true
    end
  end
  if self.bIsFirstLogin and bHasCanBuy then
    self:LightRedPoint()
    self.bIsFirstLogin = false
    return
  end
  if bHasCanBuy and bHasNewGift then
    self:LightRedPoint()
  end
end

function tbActivity:OnOpenUnRegisterRedPoint()
  self:FadeRedPoint()
end

function tbActivity:CreateTimer()
  if 0 ~= self.nTimerId then
    Timer:Close(self.nTimerId)
    self.nTimerId = 0
  end
  self.nTimerId = Timer:Register(Env.GAME_FPS * 10, function()
    self:CheckShowRedPoint()
    local tbActivityObj = ActivityMgr:GetInstanceByClassName(ActivityMgr.szNewServerNoviceGiftName)
    if tbActivityObj and not tbActivityObj:IsOpen() then
      return false
    end
    return true
  end)
end

function tbActivity:CloseTimer()
  Timer:Close(self.nTimerId)
  self.nTimerId = 0
end

function tbActivity:LightRedPoint()
  if self.bisLight then
    return
  end
  RedPointMgr:LightRedPoint(RedPointMgr.EVENT.NewServerGift)
  self.bisLight = true
end

function tbActivity:FadeRedPoint()
  if not self.bisLight then
    return
  end
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.NewServerGift)
  self.bisLight = false
end
