local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.MonthlySpecialsActivity)
local Protocol = {
  [tbActivity.Protocol.OnGetGiftPackRewardResp] = "OnGetGiftPackRewardResp"
}

function tbActivity:ServerRequest(nProtocol, ...)
  if not nProtocol or not Protocol[nProtocol] then
    LogErr("MonthlySpecialsActivityC: ServerRequest failed, invalid Protocol = ", nProtocol)
    return
  end
  self[Protocol[nProtocol]](self, ...)
end

function tbActivity:OnOpen()
  if not self.bRegisterEvent then
    self.bRegisterEvent = true
    EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self.OnPlayerFirstLoginToday, self)
  end
  self:TryLightRedPoint()
end

function tbActivity:OnLogin()
  self:TryLightRedPoint()
end

function tbActivity:OnClose()
  if self.bRegisterEvent then
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY, self)
  end
end

function tbActivity:OnGetGiftPackRewardResp(bResult)
  if bResult then
    self:TryLightRedPoint()
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_MONTHLYSPECIALSACTIVITY_GETAWARD, bResult)
end

function tbActivity:OnGetGiftPackRewardReq()
  if not self:IsOpen(me) then
    return false
  end
  local nAwardIndex = self:GetCurRewardProgress(me)
  if nAwardIndex + 1 > #self.tbAwardList then
    return false
  end
  local tbAwardCfg = self.tbAwardList[nAwardIndex + 1]
  if tbAwardCfg.nCondition == self.tbConditions.Free then
    ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.GetGiftPackRewardReq)
  elseif tbAwardCfg.nCondition == self.tbConditions.Recharge then
    Recharge:RequestCreateRechargeOrder(tbAwardCfg.nProductId)
  elseif tbAwardCfg.nCondition == self.tbConditions.LimitedTime then
    if self:CheckExpiredCurTime(tbAwardCfg.nLimitedTime) then
      local nHaveMoney = me.GetMoney(tbAwardCfg.szMoneyType)
      if nHaveMoney < tbAwardCfg.nPrice then
        me.CenterMsg(i18n.EasyGet("\232\180\167\229\184\129\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\232\180\173\228\185\176\231\164\188\229\140\133"))
        GainCurrency:ReadyForJump(tbAwardCfg.szMoneyType)
        return false
      end
    end
    ActivityMgr:CallServerScriptV2(self.nId, self.Protocol.GetGiftPackRewardReq)
  end
  return true
end

function tbActivity:OnPlayerFirstLoginToday()
  if self:HaveBuyRewardAvailable() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MonthlySpecialsActivity)
  end
end

function tbActivity:TryLightRedPoint()
  if self:HaveFreeRewardAvailable() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MonthlySpecialsActivity)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MonthlySpecialsActivity)
  end
end

function tbActivity:TryFadeRedPoint()
  if not self:HaveFreeRewardAvailable() then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MonthlySpecialsActivity)
  end
end

function tbActivity:HaveFreeRewardAvailable()
  local nAwardIndex = self:GetCurRewardProgress(me)
  if nAwardIndex + 1 > #self.tbAwardList then
    return false
  end
  local tbAward = self.tbAwardList[nAwardIndex + 1]
  if not tbAward then
    return false
  end
  if tbAward.nCondition == self.tbConditions.Free then
    return true
  elseif tbAward.nCondition == self.tbConditions.LimitedTime then
    return not self:CheckExpiredCurTime(tbAward.nLimitedTime)
  end
end

function tbActivity:HaveBuyRewardAvailable()
  local nAwardIndex = self:GetCurRewardProgress(me)
  if nAwardIndex + 1 > #self.tbAwardList then
    return false
  end
  local tbAward = self.tbAwardList[nAwardIndex + 1]
  if not tbAward then
    return false
  end
  if tbAward.nCondition == self.tbConditions.Recharge then
    return true
  elseif tbAward.nCondition == self.tbConditions.LimitedTime then
    return self:CheckExpiredCurTime(tbAward.nLimitedTime)
  end
end

function tbActivity:GetActivityCfg()
  return self.tbConfig
end

function tbActivity:GetActivityAwardList()
  return self.tbAwardList
end

function ActivityMgr:MonthlySpecialsActivity_CheckCanShowHud()
  local tbAct = self:GetInstanceByClassName(tbActivity.szClassName)
  if not tbAct then
    return false
  end
  local tbConfig = tbAct:GetActivityCfg()
  if not tbConfig then
    return false
  end
  return tbAct:IsOpen(me)
end
