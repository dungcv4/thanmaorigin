function PlayerEvent:OnDailyTimer()
  EventNotify.OnNotify(EventNotify.emNOTIFY_CROSS_DAYS)
  PlayerEvent:CheckIfPlayerIsFirstLoginToday()
  self.nDailyTimer = Timer:Register(Env.GAME_FPS * 24 * 3600, self.OnDailyTimer, self)
end

function PlayerEvent:RegDailyTimer()
  if self.nDailyTimer then
    Timer:Close(self.nDailyTimer)
    self.nDailyTimer = nil
  end
  local nSeconds = 14400 - Lib:GetLocalDayTime()
  if nSeconds <= 0 then
    nSeconds = nSeconds + 86400
  end
  Log(string.format("PlayerEvent:RegDailyTimer next time=%f hour", nSeconds / 3600))
  self.nDailyTimer = Timer:Register(Env.GAME_FPS * (nSeconds + 5), self.OnDailyTimer, self)
end

function PlayerEvent:CheckIfPlayerIsFirstLoginToday()
  local nLastTime = LocalData:GetRoleNumber("ClientFirstLoginToday_Last_Login_Time", 0)
  if Lib:IsDiffDay(14400, nLastTime, GetTime()) then
    self.bFirstLoginToday = true
    LocalData:SetRoleNumber("ClientFirstLoginToday_Last_Login_Time", GetTime())
    EventNotify.OnNotify(EventNotify.emNOTIFY_PLAYER_FIRST_LOGIN_TODAY)
  end
end

function PlayerEvent:UnRegDailyTimer()
  if self.nDailyTimer then
    Timer:Close(self.nDailyTimer)
    self.nDailyTimer = nil
  end
  Log("PlayerEvent:UnRegDailyTimer")
end

function PlayerEvent:StartAutoPath(nDesX, nDesY, nPathLen, nPathSize)
  local pNpc = me.GetNpc()
  if not pNpc then
    return
  end
  if 0 == IsAlone() then
    local bRet = AutoRunSpeed:CheckCanMapRunSpeed(me, nPathLen)
    if bRet then
      RemoteServer.StartMapAutoRunSpeed(nDesX, nDesY, nPathLen)
      me.bStartAutoRunSpeed = true
    end
  end
  if nPathLen >= Lib:Map2Logic(200) then
    me.bStartAutoPath = true
    if Ui:WindowVisible("UIHud") then
      Ui:GetWindow("UIHud"):SetAutoPathFlagVisible(true)
    end
  end
end

function PlayerEvent:StopAutoPath(nStopType)
  local pNpc = me.GetNpc()
  if not pNpc then
    return
  end
  me.bStartAutoPath = false
  local tbWndHud = Ui:GetWindow("UIHud")
  if tbWndHud then
    tbWndHud:SetAutoPathFlagVisible(false)
  end
  local tbWndMiniMap = Ui:GetWindow("UIMiniMap")
  if tbWndMiniMap then
    tbWndMiniMap:SetAutoPathFlagVisible(false)
  end
end
