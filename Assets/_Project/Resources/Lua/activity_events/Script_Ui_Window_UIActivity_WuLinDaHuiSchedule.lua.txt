local tbWnd = Ui:CreateWindow("UIActivity_WuLinDaHuiSchedule")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnTipClose = "node/imgBG/btnTipClose",
  btnHelp = "node/imgBG/imgTitle/btnHelp",
  processCell = "node/imgBG/content/processCell",
  ScheduleCell = "node/imgBG/content/ScheduleCell",
  btnScheduleCell1 = "node/imgBG/content/ScheduleCell1/btnTeam",
  btnScheduleCell2 = "node/imgBG/content/ScheduleCell2/btnReport",
  btnScheduleCell3 = "node/imgBG/content/ScheduleCell3/btnReport",
  btnScheduleCell4 = "node/imgBG/content/ScheduleCell4/btnReport",
  btnScheduleCell5 = "node/imgBG/content/ScheduleCell5/btnReport",
  imDayBg = "node/imgBG/content/ScheduleCell2/imDayBg",
  txtGuessTime = "node/imgBG/content/guessCell/txtGuessTime",
  btnGuess = "node/imgBG/content/guessCell/btnGuess"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnTipClose = function(self)
    self:Close()
  end,
  btnGuess = function(self)
    if not Lib:IsEmptyTB(self.tbBtnCallBack) then
      Lib:CallBack(self.tbBtnCallBack)
    end
  end,
  btnScheduleCell1 = function(self)
    if not Lib:IsEmptyTB(self.tbStateStting[1].tbCallBack) then
      Lib:CallBack(self.tbStateStting[1].tbCallBack)
    end
  end,
  btnScheduleCell2 = function(self)
    if not Lib:IsEmptyTB(self.tbStateStting[2].tbCallBack) then
      Lib:CallBack(self.tbStateStting[2].tbCallBack)
    end
  end,
  btnScheduleCell3 = function(self)
    if not Lib:IsEmptyTB(self.tbStateStting[3].tbCallBack) then
      Lib:CallBack(self.tbStateStting[3].tbCallBack)
    end
  end,
  btnScheduleCell4 = function(self)
    if not Lib:IsEmptyTB(self.tbStateStting[4].tbCallBack) then
      Lib:CallBack(self.tbStateStting[4].tbCallBack)
    end
  end,
  btnScheduleCell5 = function(self)
    if not Lib:IsEmptyTB(self.tbStateStting[5].tbCallBack) then
      Lib:CallBack(self.tbStateStting[5].tbCallBack)
    end
  end
}
tbWnd.STATE = {
  NOT_STARTED = 1,
  IN_PROGRESS = 2,
  ENDED = 3
}
tbWnd.tbNotStartColor = Ui.tbColor:GetUnityColorByDefineName("wulindahui6")
tbWnd.tbStartedColor = Ui.tbColor:GetUnityColorByDefineName("wulindahui5")

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(tbStateStting, nBtnStartTime, tbBtnCallBack)
  self.tbStateStting = tbStateStting
  self.nBtnStartTime = nBtnStartTime
  self.tbBtnCallBack = tbBtnCallBack
  self:UpdateSchedule()
  self:UpdateGuessCelll()
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.pPanel:SetActive(tbControls.imDayBg, false)
end

function tbWnd:OnClose()
  if self.nRemainTimer ~= nil then
    Timer:Close(self.nRemainTimer)
    self.nRemainTimer = nil
  end
end

function tbWnd:UpdateSchedule()
  local nNowTime = GetTime()
  for i, tbTime in ipairs(self.tbStateStting) do
    if nNowTime < Lib:GetConfigDate2Time(tbTime.nStartTime) then
      self:UpdateScheduleState(i, self.STATE.NOT_STARTED, tbTime)
    elseif nNowTime <= Lib:GetConfigDate2Time(tbTime.nEndTime) then
      self:UpdateScheduleState(i, self.STATE.IN_PROGRESS, tbTime)
    else
      self:UpdateScheduleState(i, self.STATE.ENDED, tbTime)
    end
  end
end

function tbWnd:UpdateScheduleState(i, nState, tbTime)
  if nState == self.STATE.NOT_STARTED then
    self.pPanel:SetActive(tbControls.ScheduleCell .. i .. "/imgSpot", false)
    self.pPanel:SetActive(tbControls.ScheduleCell .. i .. "/ingOngoingBg", false)
    self.pPanel:Sprite_SetSprite(tbControls.ScheduleCell .. i .. "/imgBg", "UI/atlas/activity_WuLinDaHui/img_ScheduleBg2.png")
    self.pPanel:Button_SetEnable(tbControls["btnScheduleCell" .. i], false)
    self.pPanel:Sprite_SetColorByName(tbControls.ScheduleCell .. i .. "/imgTimeBg", "wulindahui6")
    self.pPanel:Label_SetColorByName(tbControls.ScheduleCell .. i .. "/txtScheduleTitle", "wulindahui4")
    self.pPanel:Label_SetColorByName(tbControls.ScheduleCell .. i .. "/imgTimeBg/txtTime", "wulindahui4")
    local obj = self.pPanel:GetObject(tbControls.ScheduleCell .. i .. "/txtScheduleTitle")
    if obj then
      local comp = obj:GetComponent("Outline")
      comp.effectColor = self.tbNotStartColor
    end
    if i > 1 then
      self.pPanel:SetActive(tbControls.processCell .. "/line" .. i - 1 .. "/line1HL", false)
    end
  elseif nState == self.STATE.IN_PROGRESS then
    self.pPanel:SetActive(tbControls.ScheduleCell .. i .. "/imgSpot", true)
    self.pPanel:SetActive(tbControls.ScheduleCell .. i .. "/ingOngoingBg", true)
    self.pPanel:Sprite_SetSprite(tbControls.ScheduleCell .. i .. "/imgBg", "UI/atlas/activity_WuLinDaHui/img_ScheduleBg.png")
    self.pPanel:Button_SetEnable(tbControls["btnScheduleCell" .. i], true)
    self.pPanel:Sprite_SetColorByName(tbControls.ScheduleCell .. i .. "/imgTimeBg", "wulindahui5")
    self.pPanel:Label_SetColorByName(tbControls.ScheduleCell .. i .. "/txtScheduleTitle", "wulindahui3")
    self.pPanel:Label_SetColorByName(tbControls.ScheduleCell .. i .. "/imgTimeBg/txtTime", "wulindahui3")
    local obj = self.pPanel:GetObject(tbControls.ScheduleCell .. i .. "/txtScheduleTitle")
    if obj then
      local comp = obj:GetComponent("Outline")
      comp.effectColor = self.tbStartedColor
    end
    if i > 1 then
      self.pPanel:SetActive(tbControls.processCell .. "/line" .. i - 1 .. "/line1HL", true)
    end
  elseif nState == self.STATE.ENDED then
    self.pPanel:SetActive(tbControls.ScheduleCell .. i .. "/imgSpot", true)
    self.pPanel:SetActive(tbControls.ScheduleCell .. i .. "/ingOngoingBg", false)
    self.pPanel:Sprite_SetSprite(tbControls.ScheduleCell .. i .. "/imgBg", "UI/atlas/activity_WuLinDaHui/img_ScheduleBg.png")
    self.pPanel:Button_SetEnable(tbControls["btnScheduleCell" .. i], true)
    self.pPanel:Sprite_SetColorByName(tbControls.ScheduleCell .. i .. "/imgTimeBg", "wulindahui5")
    self.pPanel:Label_SetColorByName(tbControls.ScheduleCell .. i .. "/txtScheduleTitle", "wulindahui3")
    self.pPanel:Label_SetColorByName(tbControls.ScheduleCell .. i .. "/imgTimeBg/txtTime", "wulindahui3")
    local obj = self.pPanel:GetObject(tbControls.ScheduleCell .. i .. "/txtScheduleTitle")
    if obj then
      local comp = obj:GetComponent("Outline")
      comp.effectColor = self.tbStartedColor
    end
    if i > 1 then
      self.pPanel:SetActive(tbControls.processCell .. "/line" .. i - 1 .. "/line1HL", true)
    end
  end
  local nStartTime = Lib:GetConfigDate2Time(tbTime.nStartTime)
  local nEndTime = Lib:GetConfigDate2Time(tbTime.nEndTime)
  if Lib:IsDiffDay(0, nStartTime, nEndTime) then
    self.pPanel:Label_SetText(tbControls.ScheduleCell .. i .. "/imgTimeBg/txtTime", string.format("%s-%s", Lib:TimeDesc13(nStartTime), Lib:TimeDesc13(nEndTime)))
  else
    self.pPanel:Label_SetText(tbControls.ScheduleCell .. i .. "/imgTimeBg/txtTime", Lib:TimeDesc13(nStartTime))
  end
  self.pPanel:Label_SetText(tbControls.ScheduleCell .. i .. "/txtScheduleTitle", tbTime.szStateName)
  self.pPanel:Button_SetText(tbControls["btnScheduleCell" .. i], tbTime.szBtnName, "Text")
end

function tbWnd:UpdateGuessCelll()
  if not self.nBtnStartTime or not self.tbBtnCallBack then
    self.pPanel:SetActive(tbControls.txtGuessTime, false)
    self.pPanel:SetActive(tbControls.btnGuess, false)
    return
  end
  local nGuessStartTime = Lib:GetConfigDate2Time(self.nBtnStartTime)
  local nRemainTime = nGuessStartTime - GetTime()
  if nRemainTime <= 0 then
    self.pPanel:SetActive(tbControls.txtGuessTime, false)
    self.pPanel:Button_SetEnable(tbControls.btnGuess, true)
    return
  end
  if self.nRemainTimer then
    Timer:Close(self.nRemainTimer)
  end
  self.pPanel:SetActive(tbControls.txtGuessTime, true)
  self.pPanel:Label_SetText(tbControls.txtGuessTime, i18n.EasyGet("{0}\229\144\142\229\188\128\229\144\175\229\138\169\229\168\129", Lib:TimeDesc2(nRemainTime)))
  self.nRemainTimer = Timer:Register(Env.GAME_FPS, self.UpdateGuessCelll, self)
  self.pPanel:Button_SetEnable(tbControls.btnGuess, false)
end
