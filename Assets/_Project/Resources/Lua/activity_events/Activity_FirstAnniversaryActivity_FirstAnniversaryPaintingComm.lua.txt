Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.FirstAnniversaryPainting)
tbActivity.Protocol = {
  ReqOnOpenCell = 1,
  ReqGetCellGroupReward = 2,
  ReqGetGrandPrize = 3,
  OpenCellBack = 4,
  GetCellGroupRewardBack = 5,
  GetGrandPrizeBack = 6,
  ReqPaintingLightRedPoint = 7
}
tbActivity.SAVE_OPEN_CELL = 1
tbActivity.SAVE_CELLGROUP_REWARD = 2
tbActivity.SAVE_GRAND_PRIZE = 3
tbActivity.SAVE_ACTIVITE_STARTTIME = 4

function tbActivity:GetOpenCellData(pPlayer)
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SAVE_OPEN_CELL)
end

function tbActivity:GetCellGroupReward(pPlayer)
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SAVE_CELLGROUP_REWARD)
end

function tbActivity:GetGrandPrize(pPlayer)
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SAVE_GRAND_PRIZE)
end

function tbActivity:GetStartTime(pPlayer)
  return pPlayer.GetUserValue(self.SaveKey, tbActivity.SAVE_ACTIVITE_STARTTIME)
end

function tbActivity:GetOpenCellNum(pPlayer)
  local nOpenCellData = self:GetOpenCellData(pPlayer)
  local cnt = 0
  for i = 1, self.tbConfig.nCellNum do
    if Lib:GetBit(nOpenCellData, i) then
      cnt = cnt + 1
    end
  end
  return cnt
end

function tbActivity:IsGetCellGroupReward(pPlayer, nGroupId)
  local nOpenCellData = self:GetOpenCellData(pPlayer)
  local tbCfg = self.tbConfig.tbCellGroup[nGroupId]
  if not tbCfg then
    return false
  end
  for _, nValue in ipairs(tbCfg) do
    if not Lib:GetBit(nOpenCellData, nValue) then
      return false
    end
  end
  return true
end

function tbActivity:CheckConfigValid()
  return true
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.tbCellGroup = tbActCfg.tbCellGroup
  self.tbConfig.tbCellGroupReward = tbActCfg.tbCellGroupReward
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end
