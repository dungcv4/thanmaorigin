Require("CommonScript/Activity/DragonBoatActivity/DragonBoatActivity/DragonBoatActivityCfg.lua")
ActivityMgr.DragonBoatActivity = ActivityMgr.DragonBoatActivity or {}
local DragonBoatActivity = ActivityMgr.DragonBoatActivity
local tbJumpFunc = {
  [ActivityMgr.Def.ClassNameList.DragonBoatRace] = function(nId)
    local tbWnd = Ui:GetWindow("UIDuanWuActivity")
    if not tbWnd then
      return
    end
    local pInstance = ActivityMgr:GetInstanceById(nId)
    if not pInstance or not pInstance:IsOpen() then
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
      return
    end
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.DragonBoatRace)
    local bCanEnter, nCode = pInstance:CheckRaceIsOpen()
    if bCanEnter then
      pInstance:SignIn()
    elseif 1 == nCode then
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\128\187\229\143\130\228\184\142\230\172\161\230\149\176\229\183\178\232\190\19026\230\172\161\229\143\130\228\184\142\228\184\138\233\153\144"))
    else
      me.CenterMsg(i18n.Get(DragonBoatRace.Cfg.Msg.NotInOpenTime))
    end
  end,
  [ActivityMgr.Def.ClassNameList.DragonBoatGift] = function(nId)
    local tbWnd = Ui:GetWindow("UIDuanWuActivity")
    if not tbWnd then
      return
    end
    local pInstance = ActivityMgr:GetInstanceById(nId)
    if not pInstance or not pInstance:IsOpen() then
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
      return
    end
    Ui:OpenWindow("UIDuanWuGift")
  end
}

function DragonBoatActivity:GetActivityCfg(szType)
  local tbActivityList = self:GetActivityList()
  if not tbActivityList then
    Log("DragonBoatActivity:GetActivity() data nil")
    return nil
  end
  for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
    local tbConfig = self:BuildInstanceConfig(nId, szType)
    return tbConfig
  end
  return nil
end

function DragonBoatActivity:GetActivityList()
  local tbList = {}
  local tbConfig, szName, pInstance
  for _, szType in ipairs(ActivityMgr.DragonBoatActivityCfg.ActivityList) do
    for nId in pairs(ActivityMgr:GetIdListByClassName(szType)) do
      tbConfig, pInstance = self:BuildInstanceConfig(nId, szType)
      if tbConfig then
        szName = tbConfig.szName and i18n.Parse(tbConfig.szName) or ""
        table.insert(tbList, {
          szClassName = szType,
          nId = nId,
          szName = szName,
          szDesc = i18n.Parse(tbConfig.szDesc),
          nLevel = tbConfig.nLevel or tbConfig.nJoinLevel,
          tbAward = Award:GetAward(tbConfig.nShowAwardId or tbConfig.nAwardId or -1, tbConfig.nShowAwardKey or tbConfig.nAwardKey or -1) or {},
          pFunc = tbJumpFunc[szType],
          szShowTime = i18n.Parse(tbConfig.szShowTime or ""),
          nSortValue = ActivityMgr.DragonBoatActivityCfg.SortList[nId] or 0,
          tbRedPointEventId = pInstance and pInstance.GetRedPointEventId and pInstance:GetRedPointEventId(),
          szTimeFrame = tbConfig.szTimeFrame or nil
        })
      end
    end
  end
  table.sort(tbList, function(tbFirst, tbSecond)
    return tbFirst.nSortValue > tbSecond.nSortValue
  end)
  return tbList
end

function DragonBoatActivity:BuildInstanceConfig(nId, szType)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if pInstance then
    return pInstance:GetConfig(), pInstance
  end
  local tmpInstance = ActivityMgr:Create(szType)
  if not tmpInstance then
    return
  end
  tmpInstance:OnInit(ActivityMgr:GetConfigById(nId))
  return tmpInstance:GetConfig()
end

function DragonBoatActivity:OnActive(nId)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return
  end
end
