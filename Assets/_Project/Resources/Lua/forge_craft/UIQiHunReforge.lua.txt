local tbWnd = Ui:CreateClass("UIQiHunReforge", "UIPanel")
local tbControls = {
  PanelNormal = "PanelNormal",
  txtCurName = "PanelNormal/txtCurName",
  btnRefine = "PanelNormal/btnRefine",
  btnRate = "PanelNormal/btnRate",
  PanelResult = "PanelResult",
  txtResultName = "PanelResult/txtResultName",
  btnGetBack = "PanelResult/btnGetBack",
  btnContinue = "PanelResult/btnContinue",
  txtRefineTimes = "PanelResult/txtRefineTimes",
  txtCongratulation = "PanelResult/txtCongratulation",
  Success = "ResultTips/Success",
  Fail = "ResultTips/Fail",
  Fail_Effect = "Effect/Fail_Effect",
  Progress_Effect = "Effect/Progress_Effect",
  Succeed_Effect = "Effect/Succeed_Effect",
  itemNew = "itemNew",
  txt1 = "Effect/Succeed_Effect/txt1",
  btnMask1 = "PanelResult/imgMask1"
}
local QiHun = Item.QiHun
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnRefine = function(self)
    self:DoReforge()
  end,
  btnRate = function(self)
    self.tbOwner:ShowRate()
  end,
  btnGetBack = function(self)
    local function fnOk()
      QiHun:ReforgeGetBackReq()
    end
    
    Ui:OpenWindow("UIMessageBox", i18n.EasyGet("\230\152\175\229\144\166\231\161\174\232\174\164\229\143\150\229\135\186\229\153\168\233\173\130"), {
      {fnOk},
      {}
    })
    self.pPanel:Button_SetEnable(tbControls.btnRefine, true)
  end,
  btnContinue = function(self)
    self:DoReforge()
  end,
  btnMask1 = function(self)
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\143\150\229\135\186\229\153\168\233\173\130"))
  end
}

function tbWnd:DoReforge()
  local nLevel
  if self.nCurReforgeLevel > 0 then
    nLevel = self.nCurReforgeLevel
  else
    nLevel = self.tbOwner:GetSelectLevel()
  end
  if not nLevel then
    me.CenterMsg(i18n.EasyGet("\230\156\170\233\128\137\230\139\169\229\153\168\233\173\130"))
    return
  end
  if QiHun:GetCurReforgeTimes(me) >= QiHun.nReforgeMaxCount then
    me.CenterMsg(i18n.EasyGet("\229\141\149\232\189\174\229\183\178\233\135\141\233\147\184\229\133\171\230\172\161\239\188\140\232\175\183\229\133\136\229\143\150\229\155\158\229\153\168\233\173\130"))
    return
  end
  local bRet, szErr = QiHun:CanReforge(me, nLevel)
  if not bRet then
    return me.CenterMsg(szErr)
  end
  if self.bIsClicking then
    return
  end
  self.pPanel:Button_SetEnable(tbControls.btnRefine, false)
  self.bIsClicking = true
  self.pPanel:SetActive(tbControls.Progress_Effect, true)
  Timer:Register(40, function()
    self.pPanel:SetActive(tbControls.Progress_Effect, false)
    QiHun:ReforgeReq(nLevel)
    self.bIsClicking = false
  end)
end

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self.pPanel:Label_SetText(tbControls.txt1, i18n.EasyGet("\233\135\141\233\147\184\230\136\144\229\138\159"))
  self.pPanel:SetActive(tbControls.Progress_Effect, false)
  self.pPanel:Button_SetEnable(tbControls.btnRefine, true)
  self.bIsClicking = false
  self.nCurReforgeLevel = QiHun:GetCurReforgeLevel(me)
  if 0 == self.nCurReforgeLevel then
    self:UpdateNormalView()
  else
    self:UpdateResultView(nil, self.nCurReforgeLevel, true)
  end
end

function tbWnd:OnClose()
  QiHun:ReforgeGetBackReq()
  self:RemoveQiHunEffect()
end

function tbWnd:OnSelectQiHun()
  self:UpdateNormalView()
end

function tbWnd:UpdateNormalView()
  self.pPanel:SetActive(tbControls.PanelNormal, true)
  self.pPanel:SetActive(tbControls.PanelResult, false)
  if not self.tbOwner.nSelectIndex or not self.tbOwner.tbQiHunItem[self.tbOwner.nSelectIndex] then
    self.pPanel:Label_SetText(tbControls.txtCurName, "")
  else
    local tbData = self.tbOwner.tbQiHunItem[self.tbOwner.nSelectIndex]
    local nSelectLevel = self.tbOwner:GetSelectLevel()
    self:ShowQiHunEffect(nSelectLevel)
    local szName = Item:GetName(tbData[1], me)
    local nQuality = Item:GetQuality(tbData[1])
    local _, szColor = Item:GetQualityColor(nQuality)
    self.pPanel:Label_SetText(tbControls.txtCurName, i18n.EasyGet("<color=#{0}>{1}</color>", szColor, szName))
  end
end

function tbWnd:UpdateResultView(nOldLevel, nNewLevel, bIsOpen)
  self.pPanel:SetActive(tbControls.PanelNormal, false)
  self.pPanel:SetActive(tbControls.PanelResult, true)
  if nNewLevel then
    local nTemplateId = QiHun.tbLevelInfo[nNewLevel][1]
    self:ShowQiHunEffect(nNewLevel)
    local szName = Item:GetName(nTemplateId, me)
    local nQuality = Item:GetQuality(nTemplateId)
    local _, szColor = Item:GetQualityColor(nQuality)
    self.pPanel:Label_SetText(tbControls.txtResultName, i18n.EasyGet("<color=#{0}>{1}</color>", szColor, szName))
    if nOldLevel and nOldLevel == nNewLevel then
      self.pPanel:Label_SetText(tbControls.txtCongratulation, i18n.EasyGet("\230\129\173\229\150\156\229\164\167\228\190\160\239\188\140\229\153\168\233\173\130\228\191\157\231\186\167\230\136\144\229\138\159\229\185\182\233\162\157\229\164\150\232\142\183\229\190\151{0}\231\186\167\229\153\168\233\173\130 ", nNewLevel - QiHun.nQiHunReForgeOffsetLevel))
    else
      self.pPanel:Label_SetText(tbControls.txtCongratulation, i18n.EasyGet("\230\129\173\229\150\156\229\164\167\228\190\160\239\188\140\229\176\134\229\153\168\233\173\130\231\173\137\231\186\167\230\143\144\229\141\135\229\136\176{0}\231\186\167", nNewLevel))
    end
    self.pPanel:Label_SetText(tbControls.txtRefineTimes, i18n.EasyGet("\229\189\147\229\137\141\229\183\178\229\174\140\230\136\144{0}/{1}\230\172\161\233\135\141\233\147\184\239\188\140\230\152\175\229\144\166\231\187\167\231\187\173\239\188\159", QiHun:GetCurReforgeTimes(me), QiHun.nReforgeMaxCount))
    self.nCurReforgeLevel = nNewLevel
    if not bIsOpen then
      self.pPanel:SetActive(tbControls.Succeed_Effect, true)
      Timer:Register(40, function()
        self.pPanel:SetActive(tbControls.Succeed_Effect, false)
      end)
    end
  else
    self.pPanel:Button_SetEnable(tbControls.btnRefine, true)
    self.nCurReforgeLevel = QiHun:GetCurReforgeLevel(me)
    self.pPanel:Label_SetText(tbControls.txtResultName, "")
    self.pPanel:Label_SetText(tbControls.txtCongratulation, "")
    if not bIsOpen then
      self.pPanel:SetActive(tbControls.Fail_Effect, true)
      Timer:Register(18, function()
        self.pPanel:SetActive(tbControls.Fail_Effect, false)
      end)
    end
    self:ReforgeGetBack()
  end
end

function tbWnd:ReforgeGetBack()
  self.nCurReforgeLevel = QiHun:GetCurReforgeLevel(me)
  self:UpdateNormalView()
end

function tbWnd:ShowQiHunEffect(nLevel)
  if not nLevel then
    self:RemoveQiHunEffect()
    return
  end
  local szPath = "UI/Effect/UIQiHun_Effect/QiHun_"
  if nLevel < 10 then
    szPath = szPath .. "0" .. nLevel .. ".prefab"
  else
    szPath = szPath .. nLevel .. ".prefab"
  end
  if szPath then
    if not self.tbTargetEffect then
      self.tbTargetEffect = {}
      self.tbTargetEffect.szPath = szPath
      self.tbTargetEffect.szKey = self.pPanel:DynamicLoadGameObject(tbControls.itemNew, szPath, "effect")
      self.pPanel:SetSortingOrder(self.tbTargetEffect.szKey, 1)
    elseif szPath ~= self.tbTargetEffect.szPath then
      self.pPanel:DeleteObject(self.tbTargetEffect.szKey)
      self.tbTargetEffect.szKey = self.pPanel:DynamicLoadGameObject(tbControls.itemNew, szPath, "effect")
      self.pPanel:SetSortingOrder(self.tbTargetEffect.szKey, 1)
      self.tbTargetEffect.szPath = szPath
    end
    self.pPanel:SetActive(self.tbTargetEffect.szKey, true)
  end
end

function tbWnd:RemoveQiHunEffect()
  if self.tbTargetEffect then
    self.pPanel:DeleteObject(self.tbTargetEffect.szKey)
    self.tbTargetEffect = nil
  end
end
