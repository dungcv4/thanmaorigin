local tbWnd = Ui:CreateWindow("UISpecialForget")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnApplyForget = "node/imgBG/Content/btnConfirm",
  btnPlusItme = "node/imgBG/Content/ItemNum/btnPlus",
  txtForgetConditionTips = "node/imgBG/Content/txtTip1",
  txtForgetTimeLimit = "node/imgBG/Content/txtTip2",
  txtItemCount = "node/imgBG/Content/ItemNum/txtNumValue",
  UIItemGrid = "node/imgBG/Content/UIItemGrid"
}
tbWnd.tbControls = tbControls
local tbTextDef = {
  ForgetTimeLimit = function()
    return i18n.EasyGet("\233\129\151\229\191\152\228\184\147\231\178\190\231\154\132\233\151\180\233\154\148\230\151\182\233\151\180\230\156\170\229\136\176\239\188\140\232\191\152\232\175\183\232\128\144\229\191\131\231\173\137\229\190\133")
  end,
  ForgetConditionTips = function(...)
    return i18n.EasyGet("\233\129\151\229\191\152\228\184\147\231\178\190\233\156\128\232\166\129\230\182\136\232\128\151\228\184\128\233\162\151{0}\239\188\140\228\184\148\228\189\191\231\148\168\233\151\180\233\154\148\228\184\186{1}\229\164\169\227\128\130", ...)
  end,
  ForgetTimeLimitTips = function(nSec)
    if nSec <= 0 then
      return ""
    end
    if nSec <= 3600 then
      return i18n.EasyGet("\228\184\139\230\172\161\228\189\191\231\148\168\233\151\180\233\154\148\239\188\154{0}\229\136\134\233\146\159", math.floor(nSec / 60))
    end
    return i18n.EasyGet("\228\184\139\230\172\161\228\189\191\231\148\168\233\151\180\233\154\148\239\188\154{0}\229\164\169{1}\230\151\182", math.floor(nSec / 86400), math.ceil(nSec % 86400 / 3600))
  end
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnApplyForget = function(self)
    local tbPlayerData = LifeSkill:GetClientData()
    if (tbPlayerData.nForgetCD or 0) > GetTime() then
      me.CenterMsg(tbTextDef.ForgetTimeLimit())
    else
      RemoteServer.OnLifeSkillRequest(LifeSkill.Protocol.C2SForgetSpecialty)
    end
  end,
  btnPlusItme = function(self)
    local nWareId = Shop:GetWareIDByItemID(LifeSkill.tbForgetCost.nItemId)
    local tbWare = Shop:GetAWare(nWareId)
    local nRemainCount = Shop:GetShopRemainCount(tbWare, me)
    Ui:OpenWindow("UIShopBuy", tbWare, true, nRemainCount)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.pItemGrid = Ui:NewExistItemGrid(self, self.UIItemGrid)
end

function tbWnd:OnOpen()
  EventNotify:RegistNotify(EventNotify.emNOTIFY_LIFESKILL_MODIFY_SPECIALTY, self.OnModifySpecialty, self)
  self:UpdateView()
end

function tbWnd:OnClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_LIFESKILL_MODIFY_SPECIALTY, self)
  if Ui:WindowVisible("UILifeSkill") then
    Ui:GetWindow("UILifeSkill"):UpdateView()
  end
end

function tbWnd:UpdateView()
  local nItemId, nCostCount = LifeSkill.tbForgetCost.nItemId, LifeSkill.tbForgetCost.nCount
  local szName = i18n.Parse(KItem.GetItemShowInfo(nItemId))
  local tbPlayerData = LifeSkill:GetClientData()
  self.pPanel:Label_SetText(tbControls.txtForgetConditionTips, tbTextDef.ForgetConditionTips(szName, LifeSkill.nForgetCD / 60 / 60 / 24))
  self.pPanel:Label_SetText(tbControls.txtForgetTimeLimit, tbTextDef.ForgetTimeLimitTips((tbPlayerData.nForgetCD or 0) - GetTime()))
  self.pItemGrid:SetItemByTemplate(nItemId)
  self.pItemGrid:SetClickEnable(true)
  local nCount = me.GetItemCountInBags(nItemId)
  self.pPanel:Label_SetText(tbControls.txtItemCount, i18n.EasyGet("{0}\228\184\170", nCount))
end

function tbWnd:OnModifySpecialty()
  self:Close()
end
