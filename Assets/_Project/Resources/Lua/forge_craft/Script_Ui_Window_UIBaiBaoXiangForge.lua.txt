local tbWnd = Ui:CreateWindow("UIBaiBaoXiangForge")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnForge = "node/imgBG/Content/btnForge",
  txtMin = "node/imgBG/Content/txtMin",
  txtMax = "node/imgBG/Content/txtMax",
  txtTips = "node/imgBG/Content/txtTips",
  txtSetNumValue = "node/imgBG/Content/SetNum/txtSetNumValue",
  SliderNum = "node/imgBG/Content/SliderNum",
  UIItemGrid = "node/imgBG/Content/UIItemGrid"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
    Ui:OpenWindow("UIBaiBaoXiang")
  end,
  btnForge = function(self)
    if self.nValue > 0 then
      TreasureChest:ForgeGoldShell(self.nValue)
    end
    self:Close()
  end
}
tbWnd.tbOnSliderClick = {
  SliderNum = function(self, nValue)
    if not self.bHaveCard then
      return
    end
    self.nValue = nValue
    self.pPanel:Button_SetEnable(tbControls.btnForge, nValue > 0)
    local nJingHuo = me.GetMoney("JingHuo")
    local nNeed = nValue * TreasureChest.FORGE_USE
    self.pPanel:Label_SetText(tbControls.txtSetNumValue, string.format("%d/%d", nNeed, nJingHuo))
    self.tbItemGrid:SetLabelSuffixText(self.nValue)
  end
}

function tbWnd:OnCreate()
  self.tbItemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
end

function tbWnd:OnOpen()
  local nJingHuo = me.GetMoney("JingHuo")
  local nMaxNum = math.floor(nJingHuo / TreasureChest.FORGE_USE)
  nMaxNum = math.min(nMaxNum, TreasureChest.ONCE_FORGE_MAX)
  self.nValue = 0
  self.pPanel:Label_SetText(tbControls.txtSetNumValue, string.format("%d/%d", 0, nJingHuo))
  self.tbItemGrid:SetItemByTemplate(TreasureChest.nShellTemplateId)
  self.tbItemGrid:SetLabelSuffixText(self.nValue)
  self.pPanel:Slider_SetRange(tbControls.SliderNum, 0, nMaxNum)
  self.pPanel:Slider_SetValue(tbControls.SliderNum, 0)
  self.pPanel:Label_SetText(tbControls.txtMin, 0)
  self.pPanel:Label_SetText(tbControls.txtMax, nMaxNum)
  self.bHaveCard = WeekMonthCard:CheckMonthCardState(me)
  self.pPanel:SetActive(tbControls.txtTips, not self.bHaveCard)
  self.pPanel:Button_SetEnable(tbControls.btnForge, self.bHaveCard)
end
