  else
    return self:GetTarget_Item(tbInfo)
  end
end

function tbLogic_KillNpc:GetTarget_Npc(tbInfo)
  local tbGain = {}
  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.tbGain then
    tbGain = tbData.tbGain
  end
  local szTarget = i18n.EasyGet("\229\135\187\232\180\165")
  for nIndex, tbTarget in ipairs(tbInfo.tbNpcTargetList) do
    local nNpcTemplateId = tbTarget.nNpcTemplateId
    local nCount = tbTarget.nCount
    local nGain = tbGain[nIndex] or 0
    local szNpcName = Npc:GetNameByTemplateId(nNpcTemplateId)
    local szMsg = ""
    if nCount > 1 then
      szMsg = string.format("<color=#ffb431>%s</color><color=#c5b505>\239\188\136%s/%s\239\188\137</color>", szNpcName, nGain, nCount)
    else
      szMsg = string.format("<color=#ffb431>%s</color>", szNpcName)
    end
    if nIndex ~= #tbInfo.tbNpcTargetList then
      szMsg = szMsg .. "\227\128\129"
    end
    szTarget = szTarget .. szMsg
  end
  return szTarget
end

function tbLogic_KillNpc:GetTarget_Item(tbInfo)

  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.tbGain then
    tbGain = tbData.tbGain
  end
  local szNpcTarget = ""
  for nIndex, tbTarget in ipairs(tbInfo.tbNpcTargetList) do
    local nNpcTemplateId = tbTarget.nNpcTemplateId
    local szNpcName = Npc:GetNameByTemplateId(nNpcTemplateId)
    local szMsg = string.format("<color=#ffb431>%s</color>", szNpcName)
    if nIndex ~= #tbInfo.tbNpcTargetList then
      szMsg = szMsg .. "\227\128\129"
    end
    szNpcTarget = szNpcTarget .. szMsg
  end
  local szItemTarget = ""
  for nIndex, tbTarget in ipairs(tbInfo.tbItemTargetList) do
    local nItemTemplateId = tbTarget.nItemTemplateId
    local tbProp = Item:GetItemBaseProp(nItemTemplateId)

      szMsg = string.format("<color=%s>%s</color>", szColor, szItemName)

      szMsg = szMsg .. "\227\128\129"

    szItemTarget = szItemTarget .. szMsg

  local szTarget = i18n.EasyGet("\229\135\187\232\180\165{0}\239\188\140\232\142\183\229\143\150{1}", szNpcTarget, szItemTarget)

function tbLogic_KillNpc:DoAction_Client_Npc(pPlayer, tbInfo)
  local tbGain = {}
  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.tbGain then
    tbGain = tbData.tbGain
  end
  for nIndex, tbTarget in ipairs(tbInfo.tbNpcTargetList) do
    local nMapTemplateId = tbTarget.nMapTemplateId
    local nNpcTemplateId = tbTarget.nNpcTemplateId
    local nCount = tbTarget.nCount
    local nGain = tbGain[nIndex] or 0

      return

  end
end

function tbLogic_KillNpc:DoAction_Client_Item(pPlayer, tbInfo)
  local tbGain = {}
  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.tbGain then
    tbGain = tbData.tbGain
  end
  local bFinishAll = true
  for nIndex, tbTarget in ipairs(tbInfo.tbItemTargetList) do
    local nCount = tbTarget.nCount
    local nGain = tbGain[nIndex] or 0
    if nCount > nGain then
      bFinishAll = false
      break
    end
  end
  if bFinishAll then
    return
  end
  local nRandIndex = MathRandom(1, #tbInfo.tbNpcTargetList)
  local tbNpcTarget = tbInfo.tbNpcTargetList[nRandIndex]
  local nMapTemplateId = tbNpcTarget.nMapTemplateId
  local nNpcTemplateId = tbNpcTarget.nNpcTemplateId
  Task:GotoNpc(tbInfo.nTaskId, nNpcTemplateId, nMapTemplateId, true)
end

Task.tbLogic_Gather = Task.tbLogic_Gather or Lib:NewClass(Task.tbLogic_Default)
local tbLogic_Gather = Task.tbLogic_Gather

function tbLogic_Gather:GetTarget(tbInfo)
  local tbGain = {}
  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.tbGain then
    tbGain = tbData.tbGain
  end
  local szMsg = ""
  for i, tbGather in ipairs(tbInfo.tbGatherList) do
    local nGain = tbGain[i] or 0
    local tbNpc = tbGather.tbNpc
    local tbGet = tbGather.tbGet
    if tbNpc.nCount then
      local nCount = tbNpc.nCount
      assert(nCount >= 1)
      local nNpcTemplateId = tbNpc.nNpcTemplateId
      local szNpcName = Npc:GetNameByTemplateId(nNpcTemplateId)
      local szColor = nGain >= nCount and "green" or "#ffb431"
      if nCount > 1 then
        szMsg = szMsg .. string.format("<color=%s>%s</color>(%s/%s)", szColor, szNpcName, nGain, nCount)
      else
        szMsg = szMsg .. string.format("<color=%s>%s</color>", szColor, szNpcName)
      end
    elseif tbGet.nCount then
      local nCount = tbGet.nCount
      assert(nCount >= 1)
      local nItemTemplateId = tbGet.nItemTemplateId
      local tbProp = Item:GetItemBaseProp(nItemTemplateId)
      local szItemName = tbProp.szName
      local szColor = nGain >= nCount and "green" or "#60A5FF"
      if nCount > 1 then
        szMsg = szMsg .. string.format("<color=%s>%s</color>(%s/%s)", szColor, szItemName, nGain, nCount)
      else
        szMsg = szMsg .. string.format("<color=%s>%s</color>", szColor, szItemName)
      end
    else
      assert(false, tbInfo.nTaskId)
    end
    if i ~= #tbInfo.tbGatherList then
      szMsg = szMsg .. "\227\128\129"
    end
  end
  local szTarget = i18n.EasyGet("\233\135\135\233\155\134{0}", szMsg)
  return szTarget
end

function tbLogic_Gather:DoAction_Client(pPlayer, tbInfo)
  local tbGain = {}
  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.tbGain then
    tbGain = tbData.tbGain
  end
  for i, tbGather in ipairs(tbInfo.tbGatherList) do
    local nGain = tbGain[i] or 0
    local nCount = 0
    local tbNpc = tbGather.tbNpc
    local tbGet = tbGather.tbGet
    if tbNpc.nCount then
      nCount = tbNpc.nCount
      assert(nCount >= 1, tbInfo.nTaskId)
    elseif tbGet.nCount then
      nCount = tbGet.nCount
      assert(nCount >= 1, tbInfo.nTaskId)
    else
      assert(false, tbInfo.nTaskId)
    end
    if nGain < nCount then
      local nMapTemplateId = tbNpc.nMapTemplateId
      local nNpcTemplateId = tbNpc.nNpcTemplateId
      Task:GotoNpc(tbInfo.nTaskId, nNpcTemplateId, nMapTemplateId, false)
      return
    end
  end
end

Task.tbLogic_SwitchAward = Task.tbLogic_SwitchAward or Lib:NewClass(Task.tbLogic_Default)

end

function tbLogic_SwitchAward:GetTarget(tbInfo)
  local szAward = ""
  local tbSendAwardList = tbInfo.tbSendAwardList
  for i, tbAward in ipairs(tbSendAwardList) do
    local szType = tbAward[1]
    local szAwardName = ""
    local nNeed = 0
    local nCurNum = 0
    local nType = Player.AwardType[szType]
    local bIsMoneyType = Currency:Valid(szType)
    if nType == Player.award_type_item or nType == Player.award_type_item_Unsafe then
      local nItemTemplateId = tbAward[2]
      nNeed = tbAward[3]
      local tbProp = Item:GetItemBaseProp(nItemTemplateId)
      szAwardName = tbProp.szName
      nCurNum = me.GetItemCountByTypeInBag(nItemTemplateId)
    end
    if bIsMoneyType then
      nNeed = tbAward[2]
      szAwardName = Shop:GetMoneyName(szType)
      nCurNum = me.GetMoney(szType)
    end
    if "" ~= szAwardName and nNeed > 0 then
      local szColor = nNeed <= nCurNum and "green" or "#60A5FF"
      szAward = szAward .. string.format("<color=#60A5FF>%s</color><color=%s>(%s/%s)</color>", szAwardName, szColor, nCurNum, nNeed)
      if i ~= #tbSendAwardList then
        szAward = szAward .. "\227\128\129"
      end
    end
  end
  if "" == szAward then
    return ""
  end
  local szNpcName = Npc:GetNameByTemplateId(tbInfo.nNpcTemplateId)
  local szTarget = i18n.EasyGet("\230\138\138{0}\228\186\164\231\187\153<color=#ffb431>{1}</color>", szAward, szNpcName)
  return szTarget
end

  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  if tbData.nConvoyNpcId ~= nil then
    local pNpc = KNpc.GetById(tbData.nConvoyNpcId)
    if pNpc then
      self:StopConvoy(tbInfo, pNpc)
    end
    return
  end
  local _, nX, nY = me.GetWorldPos()
  if me.nMapTemplateId ~= tbInfo.nMapTemplateId then
    return
  end
  local pNpc = KNpc.Add(tbInfo.nTargetTemplateId, me.nLevel, -1, 1, nX, nY, 0, 0)
  if not pNpc then
    return
  end
  pNpc.SetName(i18n.EasyGet("{0}", pNpc.szName))
  pNpc.SetTitle(i18n.EasyGet("{0}\230\138\164\233\128\129\231\154\132", me.szName))
  local nNpcId = pNpc.nId
  tbData.nConvoyNpcId = nNpcId
  pNpc.AI_SetWaitNpc(me.GetNpc().nId)
  pNpc.AI_ClearMovePathPoint()
  for i = 1, #tbInfo.tbNpcPath do
    pNpc.AI_AddMovePos(tbInfo.tbNpcPath[i][1], tbInfo.tbNpcPath[i][2])
  end
  local tbDest = tbInfo.tbNpcPath[#tbInfo.tbNpcPath]
  pNpc.AI_StartPath(0)
  Npc:AddTimer(pNpc, Env.GAME_FPS, function()
    local nMapId, nX, nY = me.GetWorldPos()
    local nNpcMapId, nNpcX, nNpcY = pNpc.GetWorldPos()
    if nMapId ~= nNpcMapId then
      self:StopConvoy(tbInfo, pNpc)
      return
    end
    local nDistsSquare = Lib:GetDistsSquare(nNpcX, nNpcY, nX, nY)
    if Lib:GetDistsSquare(tbDest[1], tbDest[2], nNpcX, nNpcY) <= 122500 then
      RemoteServer.TaskCmd("EndConvoy", tbInfo.nTaskId)
      self:StopConvoy(tbInfo, pNpc)
      return
    end
    return true, true
  end)
end

function tbLogic_Convoy:StopConvoy(tbInfo, pNpc)
  local tbData = Task:GetSingleTaskData(tbInfo.nTaskId)
  tbData.nConvoyNpcId = nil
  pNpc.Delete()
end
