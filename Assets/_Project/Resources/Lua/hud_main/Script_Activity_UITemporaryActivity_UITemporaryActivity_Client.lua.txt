local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.UITemporaryActivity)

function tbActivity:ServerRequest(szFuncName, ...)
  if not self[szFuncName] then
    return
  end
  self[szFuncName](self, ...)
end

function tbActivity:S2CRefreshData()
  self:TryLightRedPoint()
end

function tbActivity:TryLightRedPoint()
  if 0 == Ui.PlayerPrefs.GetInt(self:GetFirstOpenSz(), 0) and self:IsOpen() then
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.UITemporaryActivityFirstOpen)
  else
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.UITemporaryActivityFirstOpen)
  end
end

function tbActivity:GetRedPointEventId()
  local tbStepCfg = self:GetStepsCfg()
  local RedPointEvents = {}
  table.insert(RedPointEvents, RedPointMgr.EVENT.UITemporaryActivityFirstOpen)
  for _, tbStepInfo in pairs(tbStepCfg) do
    if 0 ~= tbStepInfo.nLinkActivityID then
      local pInstance = ActivityMgr:GetInstanceById(tbStepInfo.nLinkActivityID)
      if pInstance and pInstance.GetRedPointEventId then
        local tbActivityRedPoint = pInstance:GetRedPointEventId()
        if tbActivityRedPoint then
          Lib:MergeTable(RedPointEvents, tbActivityRedPoint)
        end
      end
    end
  end
  return RedPointEvents
end

function tbActivity:DoCommonJump(nJumpId)
  local tbInfo = self.tbConfig.tbJumpInfo[nJumpId]
  return ActivityMgr:DoCommonJump(tbInfo.szJumpType, self.nId, tbInfo.szWndName, tbInfo.nStep)
end
