  tbTemp.nTaskId = tbInfo.nTaskId
  tbTemp.nStep = tbInfo.nStep
  tbTemp.nBranch = tbInfo.nBranch

  tbTemp.nNpcTemplateId = tonumber(tbInfo.param2)

  for _, tbAward in ipairs(tbTemp.tbGetItemList) do
    for k, value in pairs(tbAward) do
      if type(value) == "string" and "bAwardMsgView" == value then
        tbAward[value] = 1
        tbAward[k] = nil
      end
    end
  end
  tbTemp.tbDialog = {}
  local tbDialogInfo = {}
  for i = 1, 29 do
    local szKey = string.format("param%s", i + 3)
    local szDialog = tbInfo[szKey]
    if szDialog and "" ~= szDialog then
      tbTemp.tbDialog[i] = Task:ParseDialogMsg(szDialog)
      tbTemp.tbDialog[i].nMapTemplateId = tbTemp.nMapTemplateId
      tbTemp.tbDialog[i].nNpcTemplateId = tbTemp.nNpcTemplateId
    end
  end
  if 0 == tbTemp.nMapTemplateId or 0 == tbTemp.nNpcTemplateId then
    assert(tbInfo.szTarget and "" ~= tbInfo.szTarget, string.format("taskid = %s, mapid == 0 or npcid == 0, szTarget cannot be empty", tbInfo.nTaskId))
  end

  Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep] = Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep] or {}

local tbLogic_KillNpc = Task.tbLogic_KillNpc

function tbLogic_KillNpc:ParseCfg(tbInfo)
  local tbTemp = {}
  tbTemp.nTaskId = tbInfo.nTaskId
  tbTemp.nStep = tbInfo.nStep
  tbTemp.nBranch = tbInfo.nBranch
  tbTemp.szAction = tbInfo.szAction
  tbTemp.nNpcKindNum = tonumber(tbInfo.param1)
  tbTemp.nItemKindNum = tonumber(tbInfo.param2)
  assert(tbTemp.nNpcKindNum > 0)
  assert(tbTemp.nItemKindNum >= 0)
  local nIndex = 2
  tbTemp.tbNpcTargetList = {}
  for i = 1, tbTemp.nNpcKindNum do
    nIndex = nIndex + 1
    local szKey = string.format("param%s", nIndex)
    local szParam = tbInfo[szKey]
    if szParam and "" ~= szParam then
      local tbTarget = {}
      local tbParamInfo = Lib:SplitStr(szParam, "|")
      tbTarget.nMapTemplateId = tonumber(tbParamInfo[1])
      tbTarget.nNpcTemplateId = tonumber(tbParamInfo[2])
      if tbTemp.nItemKindNum > 0 then
        assert(2 == #tbParamInfo)
      else
        assert(3 == #tbParamInfo)
        tbTarget.nCount = tonumber(tbParamInfo[3])
      end
      table.insert(tbTemp.tbNpcTargetList, tbTarget)
    end
  end
  tbTemp.tbItemTargetList = {}
  for i = 1, tbTemp.nItemKindNum do
    nIndex = nIndex + 1
    local szKey = string.format("param%s", nIndex)
    local szParam = tbInfo[szKey]
    if szParam and "" ~= szParam then
      local tbTarget = {}
      local tbParamInfo = Lib:SplitStr(szParam, "|")
      assert(3 == #tbParamInfo)
      tbTarget.nItemTemplateId = tonumber(tbParamInfo[1])
      tbTarget.nRate = tonumber(tbParamInfo[2])
      tbTarget.nCount = tonumber(tbParamInfo[3])
      table.insert(tbTemp.tbItemTargetList, tbTarget)
    end
  end
  Task.CFG_DETAIL[tbTemp.nTaskId] = Task.CFG_DETAIL[tbTemp.nTaskId] or {}
  Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep] = Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep] or {}

Task.tbLogic_Gather = Task.tbLogic_Gather or Lib:NewClass(Task.tbLogic_Default)
local tbLogic_Gather = Task.tbLogic_Gather

function tbLogic_Gather:ParseCfg(tbInfo)
  local tbTemp = {}
  tbTemp.nTaskId = tbInfo.nTaskId
  tbTemp.nStep = tbInfo.nStep
  tbTemp.nBranch = tbInfo.nBranch
  tbTemp.szAction = tbInfo.szAction
  tbTemp.tbGatherList = {}
  local nGatherKindNum = tonumber(tbInfo.param1)
  assert(nGatherKindNum > 0, tbInfo.nTaskId)
  local nIndex = 1
  for i = 1, nGatherKindNum do
    nIndex = nIndex + 1
    local szKeyGahter = string.format("param%s", nIndex)
    nIndex = nIndex + 1
    local szKeyCost = string.format("param%s", nIndex)
    nIndex = nIndex + 1
    local szKeyGet = string.format("param%s", nIndex)
    local szGatherParam = tbInfo[szKeyGahter]
    local szCostParam = tbInfo[szKeyCost]
    local szGetParam = tbInfo[szKeyGet]
    local tbNpc = {}
    local tbGatherParam = Lib:SplitStr(szGatherParam, "|")
    tbNpc.nMapTemplateId = tonumber(tbGatherParam[1])
    tbNpc.nNpcTemplateId = tonumber(tbGatherParam[2])
    tbNpc.bDelete = tonumber(tbGatherParam[3])
    if tbGatherParam[4] then
      tbNpc.nCount = tonumber(tbGatherParam[4])
    end
    if tbGatherParam[5] then
      tbNpc.nMsgType = tonumber(tbGatherParam[5])
    end
    local tbCost = {}
    if szCostParam and "" ~= szCostParam then
      tbCost.nItemTemplateId = tonumber(szCostParam)
    end
    local tbGet = {}
    if szGetParam and "" ~= szGetParam then
      local tbGetParam = Lib:SplitStr(szGetParam, "|")
      tbGet.nItemTemplateId = tonumber(tbGetParam[1])
      tbGet.nCount = tonumber(tbGetParam[2])
    end
    if tbGet.nCount then
      assert(tbGet.nCount > 0 and tbNpc.nCount == nil, tbInfo.nTaskId)
    end
    if tbNpc.nCount then
      assert(tbNpc.nCount > 0 and tbGet.nCount == nil, tbInfo.nTaskId)
    end
    local tbGather = {}
    tbGather.tbNpc = tbNpc
    tbGather.tbCost = tbCost
    tbGather.tbGet = tbGet
    table.insert(tbTemp.tbGatherList, tbGather)
  end
  Task.CFG_DETAIL[tbTemp.nTaskId] = Task.CFG_DETAIL[tbTemp.nTaskId] or {}
  Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep] = Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep] or {}
  Task.CFG_DETAIL[tbTemp.nTaskId][tbTemp.nStep][tbTemp.nBranch] = tbTemp
end

  tbTemp.nBranch = tbInfo.nBranch

  tbTemp.nNpcTemplateId = tonumber(tbInfo.param2)

  if szSendAwardList and "" ~= szSendAwardList then
    local tbAwardList = Task:ParseAwardString(szSendAwardList)

      table.insert(tbTemp.tbSendAwardList, tbAward)
    end
  end
  local szGetAwardList = tbInfo.param4
  if szGetAwardList and "" ~= szGetAwardList then
    local tbAwardList = Task:ParseAwardString(szGetAwardList)
    for _, tbAward in ipairs(tbAwardList) do
      local szType = tbAward[1]
      assert(Player.AwardType[szType], string.format("award type error, szType = %s, taskid = %s, step = %s, branch = %s", szType, tbInfo.nTaskId, tbInfo.nStep, tbInfo.nBranch))
      if "item" == szType or "Item" == szType then
      end
      table.insert(tbTemp.tbGetAwardList, tbAward)
    end
  end
  local nIndex = 4
  for i = 1, 28 do
    nIndex = nIndex + 1
    local szKey = string.format("param%s", nIndex)
    local szDialog = tbInfo[szKey]
    if not szDialog or "" == szDialog then
      break
    end
