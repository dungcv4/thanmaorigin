local tbWnd = Ui:CreateWindow("UIGangCompetitionReport")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  NotInfo = "node/imgBG/Content/Ranking/NotInfo",
  txtTitleScore = "node/imgBG/Content/Ranking/titles/TitlePlayer/txtTitleScore",
  btnLV = "node/imgBG/Content/Ranking/btnTab/btnLV",
  btnRecharge = "node/imgBG/Content/Ranking/btnTab/btnRecharge",
  ScrollView = "node/imgBG/Content/Ranking/Scroll View",
  Template = "node/imgBG/Content/Ranking/Scroll View/Viewport/Content/Template",
  txtScoreValue = "node/imgBG/Content/Ranking/txtScoreValue",
  txtRankValue = "node/imgBG/Content/Ranking/txtRank/txtRankValue"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
  end
}

function tbWnd:OnCreate()
  self.tbRankView = Ui:NewScrollView(self, tbControls.Template)
  self.tbButtonGroup = Ui:NewButtonGroup(self, {
    [1] = tbControls.btnLV,
    [2] = tbControls.btnRecharge
  }, true, function(self, nIndex)
    self:ChangeRankSelec(nIndex)
  end)
end

function tbWnd:OnOpen(tbRankData1, tbRankData2, tbMyTongRank, nMyTongPoint)
  self.tbRankDataList = {}
  self.tbRankDataList[1] = tbRankData1 or {}
  self.tbRankDataList[2] = tbRankData2 or {}
  self.tbMyTongRank = tbMyTongRank or {}
  self.nMyTongPoint = nMyTongPoint or 0
  self:UpdateText()
  self.tbButtonGroup:Select(1)
  self.pPanel:SetActive(tbControls.btnHelp, false)
end

function tbWnd:OnClose()
end

function tbWnd:UpdateText()
  self.pPanel:Button_SetText(tbControls.btnLV, i18n.EasyGet("\230\136\152\229\140\186\230\142\146\229\144\141"), "Text")
  self.pPanel:Button_SetText(tbControls.btnRecharge, i18n.EasyGet("\229\133\168\229\140\186\230\142\146\229\144\141"), "Text")
  self.pPanel:Label_SetText(tbControls.txtTitleScore, i18n.EasyGet("\229\184\174\228\188\154\231\167\175\229\136\134"))
end

function tbWnd:ChangeRankSelec(nIndex)
  if 0 == #self.tbRankDataList[nIndex] then
    self.pPanel:SetActive(tbControls.NotInfo, true)
    self.pPanel:SetActive(tbControls.ScrollView, false)
    return
  end
  self.pPanel:SetActive(tbControls.NotInfo, false)
  self.pPanel:SetActive(tbControls.ScrollView, true)
  self.tbRankView:Clear()
  for nRank, tbInfo in ipairs(self.tbRankDataList[nIndex]) do
    if nRank <= FirstTong.tbRankMaxNum[nIndex] then
      self.tbRankView:SetElement(nRank, function(tbElement)
        self:_SetRankElement(tbElement, nRank, tbInfo)
      end)
    end
  end
  self.pPanel:Label_SetText(tbControls.txtRankValue, self.tbMyTongRank[nIndex] or 0)
  self.pPanel:Label_SetText(tbControls.txtScoreValue, self.nMyTongPoint)
end

function tbWnd:_SetRankElement(tbElement, nRank, tbRankInfo)
  tbElement.pPanel:Label_SetText("txtRank", nRank)
  tbElement.pPanel:Label_SetText("txtServer", tbRankInfo.szServerName)
  tbElement.pPanel:Label_SetText("txtTeamName", tbRankInfo.szTongName)
  tbElement.pPanel:Label_SetText("txtScore", tbRankInfo.nJiJiePoint)
  tbElement.pPanel:Label_SetText("txtState", tbRankInfo.bZoneFirst and i18n.EasyGet("\230\152\175") or i18n.EasyGet("\229\144\166"))
end
