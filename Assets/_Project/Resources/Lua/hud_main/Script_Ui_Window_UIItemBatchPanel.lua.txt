function tbWnd:OnCreate(nItemId, nTemplateId)
  self.itemGrid = Ui:NewExistItemGrid(self, "imgBG/imgBG2/UIItemGrid")
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, "imgBG/imgBG2/UIAddAndSubGroup")
  self.AddAndSubGroup:BindEvent(function(nValue)
    return self:OnValueUpdate(nValue)
  end)
  self.AddAndSubGroup:SetNumPadPos(-274, 0)
  self.AddAndSubGroup:SetLongPressLoop(true, true, true)
end

function tbWnd:OnOpen(nItemId, nTemplateId, szType, nMaxCanUse, tbFnParam)
  if nItemId then
    self.pItem = KItem.GetItemObj(nItemId)
    if not self.pItem then
      Ui:CloseWindow(self.ClassName)
      return
    end
  end
  if not nTemplateId then
    return 0
  end
  self.nItemId = nItemId
  self.nTemplateId = nTemplateId
  self.szType = szType
  self.nMaxCanUse = nMaxCanUse
  self.tbFnParam = tbFnParam or {}
  self.nCurValue = 0
  self.pPanel:SetActive(self.txtTip2, false)
  self.tbInfo = Item:GetItemBaseProp(nTemplateId)
  assert(self.tbInfo)
  local szClassName = self.tbInfo.szClass
  local tbFuncSet = tbItemClassSetting[szClassName] or tbItemClassSetting[self.szType]
  assert(tbFuncSet)
  tbFuncSet(self, self.nItemId, self.nTemplateId)
  self:ShowPanel()
end

function tbWnd:ShowPanel()
  if self.nItemId then
    local pItem = KItem.GetItemObj(self.nItemId)
    if not pItem then
      self:Close()
      return
    end
    local nUseLimit = Item:GetItemUseManyLimit(pItem.szClass)
    if not nUseLimit then
      self:Close()
      return
    end
    if self.nMaxCanUse then
      self.nMaxCount = math.min(self.nMaxCanUse, nUseLimit)
      self.nMaxCount = math.min(pItem.nCount, self.nMaxCount)
    elseif nUseLimit < 0 then
      self.nMaxCount = pItem.nCount
    else
      self.nMaxCount = math.min(pItem.nCount, nUseLimit)
    end
  else
    if not self.nMaxCanUse then
      Ui:CloseWindow(self.ClassName)
      return
    end
    self.nMaxCount = self.nMaxCanUse
  end
  self.nCurValue = self.nMaxCount
  self.AddAndSubGroup:SetNumRange(1, self.nMaxCount)
  self.AddAndSubGroup:SetInputValue(self.nCurValue)
  if self.fnChangeTip then
    self.fnChangeTip(self)
  else
    self.pPanel:Label_SetText(self.txtTip, i18n.EasyGet("\229\183\178\232\190\190\229\136\176\230\156\128\229\164\167\228\189\191\231\148\168\233\135\143"))
    self.pPanel:SetActive(self.txtTip, true)
  end
end

function tbWnd:SetItem(nItemId, nTemplateId)
  if not nTemplateId and not nItemId then
    return
  end
  local pItem
  if nItemId then
    pItem = KItem.GetItemObj(nItemId)
    if not pItem then
      return
    end
    nTemplateId = pItem.dwTemplateId
  end
  self.itemGrid:SetItem(self.nItemId, self.nTemplateId, {bShowTip = false})
  local szName = KItem.GetItemShowInfo(self.nTemplateId, me.nFaction, me.nFactionSect, me.nSex)
  if szName then
    if EDITOR then
      szName = string.format("%s (ID:%d)", szName, nTemplateId)
    end
    self.pPanel:Label_SetText(self.txtName, i18n.Parse(szName))
  end
end

function tbWnd:OnValueUpdate(nValue)
  if nValue < 1 then
    nValue = 1
  end
  if nValue > self.nMaxCount then
    nValue = self.nMaxCount
  end
  self.nCurValue = nValue
  self.AddAndSubGroup:SetInputValue(self.nCurValue)
  if self.fnChangeTip then
    self.fnChangeTip(self)
  else
    self.pPanel:SetActive(self.txtTip, self.nCurValue == self.nMaxCount)
  end
  return self.nCurValue
end

function tbWnd:UseManyItems()
  if self.fnUseManyItems then
    self.fnUseManyItems(self)
  else
    Item:ClientUseManyItem(self.nItemId, self.nCurValue)
  end
end
