local tbWnd = Ui:CreateWindow("UICuilingCompose")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_EQUIP_LINGZHU_COMPOSE_SUCCESS,
      self.OnComposeRes,
      self
    }
  }
end

local tbControls = {
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  btnClose = "node/imgBG/btnClose",
  UIItemGridOutPutTemplate = "node/imgBG/Content/Cuiling/panelItemList/imgList/List/Template0%d/UIItemGrid",
  txtProbabilityOutPutTemplate = "node/imgBG/Content/Cuiling/panelItemList/imgList/List/Template0%d/txtName",
  UIItemGridGet = "node/imgBG/Content/Cuiling/panelCompose/PossibleGet/Product/imgPossibleProduct/UIItemGrid",
  txtProductGet = "node/imgBG/Content/Cuiling/panelCompose/PossibleGet/Product/txtProduct",
  UIItemGridConsumeTemplate = "node/imgBG/Content/Cuiling/panelCompose/Item0%d/UIItemGrid",
  txtConsumeNameTemplate = "node/imgBG/Content/Cuiling/panelCompose/Item0%d/txtPossibility",
  txtPossibility = "node/imgBG/Content/Cuiling/panelCompose/Item01/txtPossibility",
  btnCompose = "node/imgBG/Content/Cuiling/panelCompose/btnCompose",
  UIAddAndSubGroup = "node/imgBG/Content/Cuiling/panelCompose/UIAddAndSubGroup",
  txtCostCoin = "node/imgBG/Content/Cuiling/panelCompose/UIAddAndSubGroup/Cost/CostMoney/Text"
}
tbWnd.tbControls = tbControls
tbControls.tbShowOutPutItemGridPath = {}
tbControls.tbShowOutPutItemProbabilityPath = {}
for i = 1, 3 do
  tbControls.tbShowOutPutItemGridPath[i] = string.format(tbControls.UIItemGridOutPutTemplate, i)
  tbControls.tbShowOutPutItemProbabilityPath[i] = string.format(tbControls.txtProbabilityOutPutTemplate, i)
end
tbControls.tbConsumeItemGridPath = {}
tbControls.tbConsumeNamePath = {}
for i = 1, 3 do
  tbControls.tbConsumeItemGridPath[i] = string.format(tbControls.UIItemGridConsumeTemplate, i)
  tbControls.tbConsumeNamePath[i] = string.format(tbControls.txtConsumeNameTemplate, i)
end
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked("CuilingCompose")
  end,
  btnClose = function(self)
    self:Close()
  end,
  btnCompose = function(self)
    if self.nCount <= 0 then
      return
    end
    EquipCuiLing:ECLLingZhuComposReq(self.nCount)
  end
}

function tbWnd:OnCreate()
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroup)
  self.AddAndSubGroup:BindEvent(function(nValue)
    self.nCount = self:UpdateNumberInput(nValue, true)
    return self.nCount
  end)
  self.AddAndSubGroup:SetEnable(true)
  self.tbShowItemGrid = {}
  for i = 1, 3 do
    self.tbShowItemGrid[i] = Ui:NewExistItemGrid(self, tbControls.tbShowOutPutItemGridPath[i])
  end
  self.tbConsumeItemGrid = {}
  for i = 1, 3 do
    self.tbConsumeItemGrid[i] = Ui:NewExistItemGrid(self, tbControls.tbConsumeItemGridPath[i])
  end
  self.tbItemGridGet = Ui:NewExistItemGrid(self, tbControls.UIItemGridGet)
end

function tbWnd:OnDestroy()
end

function tbWnd:OnOpen()
  self.nCount = 0
  self.pPanel:SetActive(tbControls.UIItemGridGet, false)
  self:InitView()
  self:UpdateView()
  if self:UpdateNumberInput(1) > 0 then
    self.AddAndSubGroup:InputValueChanged(1)
  else
    self.AddAndSubGroup:InputValueChanged(0)
  end
end

function tbWnd:OnClose()
end

function tbWnd:InitView()
  local tbAddLingZhuId = EquipCuiLing:GetCurTimeFrameComposeLZId()
  for i = 1, 3 do
    self.tbShowItemGrid[i]:SetItemByTemplate(tbAddLingZhuId[i])
    self.tbShowItemGrid[i]:SetClickEnable(true)
    self.pPanel:Label_SetText(tbControls.tbShowOutPutItemProbabilityPath[i], i18n.EasyGet("\230\166\130\231\142\135\239\188\154{0}%", EquipCuiLing.tbLZAttrCountWeight[i] / 10000))
  end
  for i = 1, 3 do
    local nItemTemplated = EquipCuiLing.Cfg["nLZSuiPianId" .. i]
    self.tbConsumeItemGrid[i]:SetItemByTemplate(nItemTemplated)
    self.tbConsumeItemGrid[i]:SetClickEnable(true)
    local szItemName = Item:GetName(nItemTemplated, me)
    self.pPanel:Label_SetText(tbControls.tbConsumeNamePath[i], szItemName)
  end
end

function tbWnd:UpdateView()
  for i = 1, 3 do
    local nItemTemplated = EquipCuiLing.Cfg["nLZSuiPianId" .. i]
    local nHadSuiPian = me.GetItemCountByTypeInBag(nItemTemplated)
    self.tbConsumeItemGrid[i]:SetLabelSuffixText(nHadSuiPian)
  end
end

function tbWnd:UpdateNumberInput(nInput, bTips)
  local nMaxNum = 20
  for i = 1, 3 do
    local nItemTemplated = EquipCuiLing.Cfg["nLZSuiPianId" .. i]
    local nHadSuiPian = me.GetItemCountByTypeInBag(nItemTemplated)
    nMaxNum = math.min(nMaxNum, nHadSuiPian)
  end
  if nInput > nMaxNum then
    nInput = nMaxNum
    if bTips then
      me.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\229\134\141\229\162\158\229\138\160\228\186\134"))
    end
    self:UpdateCoinCost(nInput)
    return nInput
  end
  if nInput <= 0 and nMaxNum > 0 then
    nInput = 1
    if bTips then
      me.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\229\134\141\229\135\143\229\176\145\228\186\134"))
    end
    self:UpdateCoinCost(nInput)
    return nInput
  end
  self:UpdateCoinCost(nInput)
  return nInput
end

function tbWnd:OnComposeRes()
  self.AddAndSubGroup:InputValueChanged(1)
  self:UpdateView()
end

function tbWnd:UpdateCoinCost(nInput)
  local nCostCoin = EquipCuiLing.Cfg.nComposeCostCoin * nInput
  self.pPanel:Label_SetText(tbControls.txtCostCoin, nCostCoin)
  if nCostCoin <= me.GetMoney("Coin") then
    self.pPanel:Label_SetColorByName(tbControls.txtCostCoin, "Enough")
  else
    self.pPanel:Label_SetColorByName(tbControls.txtCostCoin, "NotEnough")
  end
end
