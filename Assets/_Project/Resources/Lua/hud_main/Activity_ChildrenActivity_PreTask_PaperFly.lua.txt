Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.PaperFly)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/ChildrenActivity/PreTask/PaperFlyCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/ChildrenActivity/PreTask/PaperFly.tab"
  local tbPaperFlyConfig = Lib:EasyLoadTabFile(szFilepath)
  if not tbPaperFlyConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  for _, tbInfo in pairs(tbPaperFlyConfig) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("PreTask: LoadPaperFlyConfig failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPaperFlyConfig = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPaperFlyConfig or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbPaperFlyConfig, {
      nActDegree = tbInfo.nActDegree,
      nNum = tbInfo.nNum,
      nAwardKey = tbInfo.nAwardKey
    })
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end
