local tbItemClass = Item:GetClass("JumpNpcShopCurrency")
tbItemClass.PARAM_NPCSHOPID = 1

function tbItemClass:OnClientUse(it)
  local nNpcShopId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPCSHOPID)
  local tbWareList = Shop:GetANpcShopWares(nNpcShopId)
  local tbWare = Shop:GetAWare(tbWareList[1].nWareId)
  local szCurrencyType = tbWare and tbWare.szCurrencyType or ""
  Ui:OpenWindow("UINpcShop", nNpcShopId, nil, nil, nil, nil, szCurrencyType, true)
  return 0
end
