local tbWnd = Ui:CreateWindow("UIStoneTipsCompare")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UPDATE_COMPARE_PANEL,
      self.OnCompare
    }
  }
end

local tbControls = {
  imgBG = "imgBG",
  UIStoneTips = "imgBG/UIStoneTips",
  Cont_Compare = "Cont_Compare",
  MyTips = "Cont_Compare/UIStoneTips",
  NewTips = "Cont_Compare/UIStoneTips_New"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbStoneTips = Ui:CreatePanel("UIStoneTips", tbControls.UIStoneTips, self, self)
  self.tbMyTips = Ui:CreatePanel("UIStoneTips", tbControls.MyTips, self, self)
  self.tbNewTips = Ui:CreatePanel("UIStoneTips", tbControls.NewTips, self, self)
  self:SetClickEmptyToClose()
end

function tbWnd:OnOpen(nItemId, nTemplateId, tbInfo)
  self.nItemId = nItemId
  self.nTemplateId = nTemplateId
  self.tbInfo = tbInfo
  self.tbStoneTips:OnOpen(nItemId, nTemplateId, tbInfo)
  self.pPanel:SetActive(tbControls.imgBG, true)
  self.pPanel:SetActive(tbControls.Cont_Compare, false)
end

function tbWnd:OnCompare(nNewTemplateId)
  local tbCompareTipsInfo = {
    nFaction = me.nFaction,
    nFactionSect = me.nFactionSect,
    nBodySex = me.GetNpc().GetBodySex(),
    szItemOpt = self.tbInfo.szItemOpt,
    tbItemExtParams = self.tbInfo.tbItemExtParams
  }
  self.pPanel:SetActive(tbControls.imgBG, false)
  self.pPanel:SetActive(tbControls.Cont_Compare, true)
  self.tbMyTips:OnOpen(self.nItemId, self.nTemplateId, self.tbInfo)
  self.tbNewTips:OnOpen(nil, nNewTemplateId, tbCompareTipsInfo)
end
