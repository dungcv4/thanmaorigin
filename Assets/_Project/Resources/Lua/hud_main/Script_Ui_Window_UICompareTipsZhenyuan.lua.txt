local tbWnd = Ui:CreateWindow("UICompareTipsZhenyuan")
local tbControls = {
  EquipDetail = "imgBG/EquipDetail",
  panelRoot = "imgBG"
}
tbWnd.tbControls = tbControls

function tbWnd:OnCreate()
  self.tbEquipDetail = Ui:CreatePanel("UIZhenyuanEquipDetail", tbControls.EquipDetail, self)
end

function tbWnd:OnOpen(tbInfo)
  self.pAsyncData = tbInfo.pAsyncData
  self.nItemId = tbInfo.nItemId
  self.nTemplateId = tbInfo.nTemplateId
  self.nFaction = tbInfo.nFaction
  self.nFactionSect = tbInfo.nFactionSect
  self.nSex = tbInfo.nSex
  self.nSeries = tbInfo.nSeries
  self.pPanel:Object_SetLocalPosition(tbControls.panelRoot, tbInfo.pos)
  if 0 == self.nItemId then
    self.nItemId = nil
  end
  if self.nItemId then
    local pItem = KItem.GetItemObj(self.nItemId)
    self.pItem = pItem
    self.nTemplateId = pItem.dwTemplateId
  else
    self.pItem = nil
  end
  local tbEquipDetailInfo = {
    szOpt = tbInfo.szItemOpt or "ZhenYuanEquip",
    pItem = self.pItem,
    nItemId = self.nItemId,
    nTemplateId = self.nTemplateId,
    nFaction = self.nFaction,
    nFactionSect = self.nFactionSect,
    nSex = self.nSex,
    pAsyncData = tbInfo.pAsyncData,
    bIsCompare = self.ClassName == "UICompareTipsZhenyuan",
    tbItemExtParams = tbInfo.tbItemExtParams,
    nSeries = self.nSeries
  }
  self.tbEquipDetail:Update(tbEquipDetailInfo)
end

function tbWnd:CloseByDefault()
  self:Close()
  Ui:CloseWindow("UIZhenyuanTips")
end

function tbWnd:OnFixedUpdate()
end
