local tbItem = Item:GetClass("ImperialEdict")

function tbItem:OnClientUse(it)
  if DegreeCtrl:GetDegree(me, "Battle") > 0 then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\164\169\232\191\152\230\156\137\229\137\169\228\189\153\230\172\161\230\149\176\230\156\170\229\174\140\230\136\144\239\188\140\230\151\160\230\179\149\228\189\191\231\148\168\229\174\139\233\135\145\232\175\143\228\185\166"))
    return 0
  end
  if DegreeCtrl:GetDegree(me, "BattleImperialEdict") > 0 then
    return 1
  else
    me.CenterMsg(i18n.Get(Battle.Cfg.tbText.szNoImperialEdictUseTimes))
    return 0
  end
end

function tbItem:OnUse(it)
  if DegreeCtrl:ReduceDegree(me, "BattleImperialEdict", 1) then
    DegreeCtrl:AddDegree(me, "Battle", 1)
    me.CallClientScript("me.BuyTimesSuccess", i18n.Get(Battle.Cfg.tbText.szImperialEdictUseSuccessTips))
    return 1
  else
    return 0
  end
end
