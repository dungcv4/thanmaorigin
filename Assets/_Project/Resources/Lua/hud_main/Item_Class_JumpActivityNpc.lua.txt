local tbItem = Item:GetClass("JumpActivityNpc")
tbItem.PARAM_MAP_TEMPLATE_ID = 1
tbItem.PARAM_NPC_TEMPLATE_ID = 2

function tbItem:OnClientUse(it)
  local nMapTemplateId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_MAP_TEMPLATE_ID)
  local nNpcTemplateId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPC_TEMPLATE_ID)
  if Tong.Def.nMainMapTemplateId == nMapTemplateId and not Tong:HasTong() then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\138\160\229\133\165\229\184\174\228\188\154"))
    return 0
  end
  if Ui:GetWindow("UIItemTips") then
    Ui:CloseWindow("UIItemTips")
  end
  if Ui:GetWindow("UIBag") then
    Ui:CloseWindow("UIBag")
  end
  ActivityMgr:GotoNpc(nNpcTemplateId, nMapTemplateId)
  return 0
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "JumpActivityNpc")
end
