Task.tbLogicFieldShareNpc = Task.tbLogicFieldShareNpc or {}
local tbLogicFieldShareNpc = Task.tbLogicFieldShareNpc
tbLogicFieldShareNpc.tbNpcList = tbLogicFieldShareNpc.tbNpcList or {}
tbLogicFieldShareNpc.nShareRange = 5000

function tbLogicFieldShareNpc:OnKillNpc(nNpcId, pKillerNpc)
  if not MODULE_GAMESERVER then
    return
  end
  local tbNpcInfo = self.tbNpcList[nNpcId]
  if not tbNpcInfo then
    return
  end
  local pDeathNpc = KNpc.GetById(nNpcId)
  for nPlayerNpcId, _ in pairs(tbNpcInfo) do
    local pNpc = KNpc.GetById(nPlayerNpcId)
    if pNpc then
      local pPlayer = pNpc.GetPlayer()
      if pPlayer and (not pKillerNpc or pNpc.nId ~= pKillerNpc.nId) then
        Lib:CallBack({
          Task.OnKillNpc,
          Task,
          pDeathNpc,
          pNpc
        })
        Lib:CallBack({
          YiJunTask.OnKillNpc,
          YiJunTask,
          pDeathNpc,
          pNpc
        })
        Lib:CallBack({
          GuanFuTongJi.OnKillNpc,
          GuanFuTongJi,
          pDeathNpc,
          pNpc
        })
      end
    end
  end
  local tbPlayerList, nCount = KNpc.GetAroundPlayerList(pDeathNpc.nId, self.nShareRange)
  if nCount > 1 then
    for _, pPlayer in ipairs(tbPlayerList) do
      if pPlayer and not tbNpcInfo[pPlayer.GetNpc().nId] and (not pKillerNpc or pPlayer.GetNpc().nId ~= pKillerNpc.nId) then
        Lib:CallBack({
          Task.OnKillNpc,
          Task,
          pDeathNpc,
          pPlayer.GetNpc()
        })
        Lib:CallBack({
          YiJunTask.OnKillNpc,
          YiJunTask,
          pDeathNpc,
          pPlayer.GetNpc()
        })
        Lib:CallBack({
          GuanFuTongJi.OnKillNpc,
          GuanFuTongJi,
          pDeathNpc,
          pPlayer.GetNpc()
        })
      end
    end
  end
  self.tbNpcList[nNpcId] = nil
end

function tbLogicFieldShareNpc:OnBeAttackByPlayerOnce(nPlayerNpcId, nNpcId)
  if not MODULE_GAMESERVER then
    return
  end
  local pNpc = KNpc.GetById(nNpcId)
  if not pNpc then
    return
  end
  local tbNpcCfg = Map:GetMapNpcInfoByNpcTemplate(pNpc.nMapTemplateId, pNpc.nTemplateId)
  if not (tbNpcCfg and not (#tbNpcCfg <= 0) and tbNpcCfg[1].nShareType) or tbNpcCfg[1].nShareType <= 0 then
    return
  end
  self.tbNpcList[nNpcId] = self.tbNpcList[nNpcId] or {}
  self.tbNpcList[nNpcId][nPlayerNpcId] = 1
end
