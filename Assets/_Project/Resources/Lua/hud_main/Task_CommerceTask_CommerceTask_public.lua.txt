Require("CommonScript/Task/CommerceTask/CommerceTaskCfg.lua")
CommerceTask.tbUserValueKey = {
  nCurTaskId = 1,
  nFinishCount = 2,
  nLastTaskId = 3,
  nCanGetRoundAward = 6,
  nReplaceCount = 7,
  nReplaceRemainder = 9
}
CommerceTask.KEY = "CommerceTask"
CommerceTask.nTaskGroup = GetGroupKey(CommerceTask.KEY)

function CommerceTask:GetCurTaskId()
  return me.GetUserValue(self.nTaskGroup, self.tbUserValueKey.nCurTaskId)
end

function CommerceTask:GetFinishCount()
  return me.GetUserValue(self.nTaskGroup, self.tbUserValueKey.nFinishCount)
end

function CommerceTask:GetLastTaskId()
  return me.GetUserValue(self.nTaskGroup, self.tbUserValueKey.nLastTaskId)
end

function CommerceTask:GetReplaceRemainder()
  return me.GetUserValue(self.nTaskGroup, self.tbUserValueKey.nReplaceRemainder)
end

function CommerceTask:CanGetWholeAward()
  return 0 ~= me.GetUserValue(self.nTaskGroup, self.tbUserValueKey.nCanGetRoundAward)
end

function CommerceTask:IsFinishRound()
  return self:GetFinishCount() >= self.ROUND_TASKNUM
end

function CommerceTask:IsDoingTask()
  return self:GetCurTaskId() > 0
end

CommerceTask.Protocol = {
  SubmitItem = 1,
  SubmitTask = 2,
  GetWholeAward = 3,
  OnRelpaceTask = 4,
  BuyRelpaceTaskCount = 5
}
CommerceTask.Count2KinSalaryKey = {
  [10] = 18,
  [20] = 18,
  [30] = 18,
  [40] = 19
}

function CommerceTask:__LoadFile_TaskFile()
  self.CFG_TASK = {}
  local szFile = "Setting/Task/CommerceTask.tab"
  self.CFG_TASK = Lib:LoadTabFileEx(szFile, "dssddddddddsdd", "nTaskId", {
    "nTaskId",
    "szFilterType",
    "FilterParam",
    "nDifficulty",
    "nWeight",
    "nAwardKey",
    "nItemTemplateId",
    "nIconId",
    "nItemType",
    "nItemLevel",
    "nCount",
    "szItemName",
    "nWay1Id",
    "nWay2Id"
  })
  for _, tbInfo in pairs(self.CFG_TASK) do
    self:ParseCfg(tbInfo)
  end
  self.CFG_TASK = SetReadOnly(self.CFG_TASK)
end

function CommerceTask:ParseCfg(tbInfo)
  local tbFilter = {}
  local szFilterType = tbInfo.szFilterType
  assert("level" == szFilterType, string.format("CommerceTask.tab, nIndex = %s, szFilterType Wrong!", tbInfo.nId))
  local FilterParam = tbInfo.FilterParam
  if "level" == szFilterType then
    local tbParam = Lib:SplitStr(FilterParam, "|")
    assert(2 == Lib:CountTB(tbParam), string.format("CommerceTask.tab, nId = %s, FilterParam = %s, Wrong!", tbInfo.nId, FilterParam))
    tbFilter = {
      szFilterType,
      tonumber(tbParam[1]),
      tonumber(tbParam[2])
    }
  end
  tbInfo.tbFilter = tbFilter
  tbInfo.tbWay = {}
  if tbInfo.nWay1Id > 0 then
    table.insert(tbInfo.tbWay, tbInfo.nWay1Id)
  end
  if tbInfo.nWay2Id > 0 then
    table.insert(tbInfo.tbWay, tbInfo.nWay2Id)
  end
  return tbInfo
end

function CommerceTask:LoadFile()
  self:__LoadFile_TaskFile()
end

function CommerceTask:GetTaskCfg(nTaskId)
  return self.CFG_TASK[nTaskId]
end

CommerceTask:LoadFile()
