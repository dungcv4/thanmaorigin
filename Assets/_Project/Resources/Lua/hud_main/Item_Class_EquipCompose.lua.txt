local tbEquipCompose = Item:GetClass("EquipCompose")

function tbEquipCompose:OnUse(pItem)
  local EquipPieceId = pItem.dwTemplateId
  local tbCfgEquipPiece = self.tbEquipPieceInfo[EquipPieceId]
  if not tbCfgEquipPiece then
    me.CenterMsg(i18n.EasyGet("{0}\230\151\160\233\133\141\231\189\174", EquipPieceId))
    return
  end
  local ComposeId = tbCfgEquipPiece.ComposeId
  if not ComposeId then
    me.CenterMsg(i18n.EasyGet("\232\175\165\233\129\147\229\133\183\230\151\160\230\179\149\229\144\136\230\136\144"))
    return
  end
  local nHadNum = me.GetItemCountByTypeInBag(EquipPieceId)
  local nNeedNum = tbCfgEquipPiece.NeedNum
  if nHadNum < nNeedNum then
    me.CenterMsg(i18n.EasyGet("\231\162\142\231\137\135\230\149\176\233\135\143\228\184\141\232\182\179{0}\228\184\170,\230\151\160\230\179\149\229\144\136\230\136\144", nNeedNum))
    return
  end
  if nNeedNum <= pItem.nCount then
    if nNeedNum > me.ConsumeItem(pItem, nNeedNum, Env.LogWay_EquipCompose) then
      me.CenterMsg(i18n.EasyGet("\231\137\169\229\147\129\230\182\136\232\128\151\229\188\130\229\184\184"))
      return
    end
  else
    local curNum = pItem.nCount
    if me.ConsumeItem(pItem, pItem.nCount, Env.LogWay_EquipCompose) < pItem.nCount then
      me.CenterMsg(i18n.EasyGet("\231\137\169\229\147\129\230\182\136\232\128\151\229\188\130\229\184\184"))
      return
    end
    if me.ConsumeItemInBag(EquipPieceId, nNeedNum - curNum, Env.LogWay_EquipCompose) < nNeedNum - curNum then
      me.CenterMsg(i18n.EasyGet("\231\137\169\229\147\129\230\182\136\232\128\151\229\188\130\229\184\184"))
      return
    end
  end
  local tbItemExtParams = {
    nTradeLimitType = Item.eTRADELIMITTYPE.NONE
  }
  local tbAward = {
    {
      "Item",
      ComposeId,
      1,
      tbItemExtParams
    }
  }
  if 0 == me.GetFreeBagCount() then
    me.CenterMsg(i18n.EasyGet("\232\131\140\229\140\133\229\183\178\230\187\161\239\188\140\232\175\183\228\187\142\233\130\174\228\187\182\233\162\134\229\143\150"))
  end
  me.SendAward(tbAward, false, true, Env.LogWay_EquipCompose)
end

function tbEquipCompose:LoadEquipPieceSetting()
  local tbParam = {
    "TemplateId",
    "ComposeId",
    "Name",
    "CostCoin",
    "NeedNum"
  }
  local tbCfg = Lib:LoadTabFileEx("Setting/Item/Other/EquipCompose.tab", "ddsdd", nil, {
    table.unpack(tbParam)
  })
  self.tbEquipPieceInfo = {}
  for i, v in ipairs(tbCfg) do
    self.tbEquipPieceInfo[v.TemplateId] = {
      ComposeId = v.ComposeId,
      Name = v.Name,
      CostCoin = v.CostCoin,
      NeedNum = v.NeedNum
    }
  end
  Validator:Register(self.CheckEquipComposeCfg, self)
end

function tbEquipCompose:CheckEquipComposeCfg()
  for TemplateId, tbInfo in pairs(self.tbEquipPieceInfo) do
    if not Item:IsExist(TemplateId) then
      LogErr(string.format("load EquipCompose.tab failed, TemplateId %d not exist", TemplateId))
      return false
    end
    if not Item:IsExist(tbInfo.ComposeId) then
      LogErr(string.format("load EquipCompose.tab failed, ComposeId %d not exist", tbInfo.ComposeId))
      return false
    end
  end
  return true
end

tbEquipCompose:LoadEquipPieceSetting()
