local tbItem = Item:GetClass("TaskItem_progress_compose")
tbItem.EXT_PARAM_PROGRESSTIME = 1
tbItem.EXT_PARAM_COSTNUM = 2
tbItem.EXT_PARAM_COMPOSEITEM = 3

function tbItem:OnUse(pItem)
  local nProgressTime = KItem.GetItemExtParam(pItem.dwTemplateId, self.EXT_PARAM_PROGRESSTIME)
  if nProgressTime > 0 then
    Progress:BeginProgress(me, " ", nProgressTime, self.OnEndProgress, {
      self,
      me.dwID,
      pItem.dwId
    })
  else
    self:OnEndProgress(me.dwID, pItem.dwId)
  end
  return 0
end

function tbItem:OnEndProgress(nPlayerId, nItemId)
  local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
  if not pPlayer then
    return
  end
  local pItem = KItem.GetItemObj(nItemId)
  if not pItem then
    return
  end
  local nLogWay = Env.LogWay_Task
  local nTaskId = Task:GetItemTaskId(pPlayer, pItem.dwTemplateId)
  if nTaskId and nTaskId > 0 then
    nLogWay = Task:GetTaskLogWay(nTaskId)
  end
  local nCostNum = KItem.GetItemExtParam(pItem.dwTemplateId, self.EXT_PARAM_COSTNUM)
  local nComposeItemTemplateId = KItem.GetItemExtParam(pItem.dwTemplateId, self.EXT_PARAM_COMPOSEITEM)
  assert(nCostNum > 0, string.format("taskitem compose, cost num wrong, costnum = %s", nCostNum or 0))
  assert(nComposeItemTemplateId > 0, string.format("taskitem compose, compose num wrong, composenum = %s", nComposeItemTemplateId or 0))
  local nCurNum = pPlayer.GetItemCountByTypeInBag(pItem.dwTemplateId)
  if nCostNum > nCurNum then
    pPlayer.CenterMsg(i18n.EasyGet("{0}\230\149\176\233\135\143\228\184\141\232\182\179<color=red>{1}</color>\239\188\129", pItem.szName, nCostNum))
    return
  end
  assert(1 == pPlayer.ConsumeItemInBag(pItem.dwTemplateId, nCostNum, nLogWay))
  Task:GiveItem(pPlayer, nComposeItemTemplateId, 1, nLogWay)
  Task:OnUseItem(pPlayer, pItem.dwTemplateId)
end
