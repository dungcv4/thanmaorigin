local tbWnd = Ui:CreateWindow("UIPKCompetitionSectSelect")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_DINGFENGPK_SYNC_BAN,
      self.UpdateBan,
      self
    },
    {
      EventNotify.emNOTIFY_DINGFENGPK_SYNC_PICK,
      self.UpdatePick,
      self
    }
  }
end

local tbControls = {
  txtTitle = "imgBG/Mail/txtTitle",
  txtTime = "imgBG/Mail/txtTime",
  txtRule = "imgBG/Mail/txtRule",
  TemplateTime = "imgBG/Mail/contentCell/Template"
}
tbWnd.tbControls = tbControls
tbWnd.ElementPath = {
  imgPlayName = "imgPlayName",
  txtPlayerName = "imgPlayName/txtPlayerName",
  imgSelected = "imgSelected",
  txtSect = "txtSect",
  imgFactionIcon = "imgFactionIcon",
  imgBan = "imgBan",
  imgSelect = "imgSelect"
}
tbWnd.tbOnClick = {}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.TemplateTime)
end

function tbWnd:OnOpen()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateBan(nRestTime, tbEnemyFactionList, tbBanFactionList)
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\232\175\183\228\189\160\233\128\137\230\139\169\229\175\185\230\137\139\231\154\132\228\184\128\228\184\170\232\183\175\231\186\191\229\176\129\231\166\129"))
  self.pPanel:Label_SetText(tbControls.txtRule, i18n.Get(DingFengPK.tbText.szBanRule))
  self:UpdateTime(nRestTime)
  self.tbScrollView:Clear()
  for index, tbInfo in ipairs(tbEnemyFactionList) do
    self.tbScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:Label_SetText(self.ElementPath.txtSect, Faction:GetSectName(tbInfo.nFaction, tbInfo.nFactionSect))
      tbElement.pPanel:Sprite_SetSprite(self.ElementPath.imgFactionIcon, Faction:GetIcon(tbInfo.nFaction))
      tbElement.pPanel:SetActive(self.ElementPath.imgPlayName, false)
      tbElement.pPanel:SetActive(self.ElementPath.imgSelected, false)
      tbElement.pPanel:SetActive(self.ElementPath.imgBan, false)
      tbElement.pPanel:SetActive(self.ElementPath.imgSelect, false)
      local bCanBan = true
      for nBanPlayerId, tbBanFactionInfo in pairs(tbBanFactionList) do
        if tbBanFactionInfo.nFaction == tbInfo.nFaction and tbBanFactionInfo.nFactionSect == tbInfo.nFactionSect then
          bCanBan = false
          tbElement.pPanel:SetActive(self.ElementPath.imgPlayName, true)
          tbElement.pPanel:SetActive(self.ElementPath.imgBan, true)
          tbElement.pPanel:Label_SetText(self.ElementPath.txtPlayerName, tbBanFactionInfo.szPlayerName)
          break
        end
      end
      if bCanBan then
        Ui:Button_BindEvent(tbElement.pPanel, "", DingFengPK.BanFactionSect, DingFengPK, tbInfo.nFaction, tbInfo.nFactionSect)
      end
    end)
  end
end

function tbWnd:UpdatePick(nRestTime, tbFactionList, tbEnemyBanFactionList, tbPickFactionList)
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\232\175\183\228\189\160\233\128\137\230\139\169\228\189\160\229\156\168\230\175\148\232\181\155\228\184\173\231\154\132\232\183\175\231\186\191"))
  self.pPanel:Label_SetText(tbControls.txtRule, i18n.Get(DingFengPK.tbText.szPickRule))
  self:UpdateTime(nRestTime)
  self.tbScrollView:Clear()
  for index, tbInfo in ipairs(tbFactionList) do
    self.tbScrollView:SetElement(index, function(tbElement)
      tbElement.pPanel:Label_SetText(self.ElementPath.txtSect, Faction:GetSectName(tbInfo.nFaction, tbInfo.nFactionSect))
      tbElement.pPanel:Sprite_SetSprite(self.ElementPath.imgFactionIcon, Faction:GetIcon(tbInfo.nFaction))
      tbElement.pPanel:SetActive(self.ElementPath.imgPlayName, false)
      tbElement.pPanel:SetActive(self.ElementPath.imgSelected, false)
      tbElement.pPanel:SetActive(self.ElementPath.imgBan, false)
      tbElement.pPanel:SetActive(self.ElementPath.imgSelect, false)
      local bCanPick = true
      for nBanPlayerId, tbBanFactionInfo in pairs(tbEnemyBanFactionList) do
        if tbBanFactionInfo.nFaction == tbInfo.nFaction and tbBanFactionInfo.nFactionSect == tbInfo.nFactionSect then
          bCanPick = false
          tbElement.pPanel:SetActive(self.ElementPath.imgPlayName, false)
          tbElement.pPanel:SetActive(self.ElementPath.imgBan, true)
          Ui:Button_BindEvent(tbElement.pPanel, "", function()
            me.CenterMsg(i18n.EasyGet("\230\173\164\232\183\175\231\186\191\229\183\178\232\162\171\231\166\129\231\148\168"))
          end)
          break
        end
      end
      if bCanPick then
        for nPickPlayerId, tbPickFactionInfo in pairs(tbPickFactionList) do
          if tbPickFactionInfo.nFaction == tbInfo.nFaction and tbPickFactionInfo.nFactionSect == tbInfo.nFactionSect then
            bCanPick = false
            tbElement.pPanel:SetActive(self.ElementPath.imgPlayName, true)
            tbElement.pPanel:SetActive(self.ElementPath.imgSelected, true)
            tbElement.pPanel:Label_SetText(self.ElementPath.txtPlayerName, tbPickFactionInfo.szPlayerName)
            Ui:Button_BindEvent(tbElement.pPanel, "", function()
              if nPickPlayerId == me.dwID then
                me.CenterMsg(i18n.EasyGet("\228\189\160\229\183\178\233\128\137\231\148\168\230\173\164\232\183\175\231\186\191"))
              else
                me.CenterMsg(i18n.EasyGet("\228\189\160\231\154\132\233\152\159\229\143\139\229\183\178\233\128\137\231\148\168\230\173\164\232\183\175\231\186\191"))
              end
            end)
            break
          end
        end
      end
      if bCanPick and not Faction:IsOpenSex(tbInfo.nFaction, me.nSex) then
        bCanPick = false
        Ui:Button_BindEvent(tbElement.pPanel, "", function()
          me.CenterMsg(i18n.EasyGet("\228\189\160\231\154\132\230\128\167\229\136\171\228\184\141\231\172\166\229\144\136\230\173\164\233\151\168\230\180\190"))
        end)
      end
      if bCanPick then
        Ui:Button_BindEvent(tbElement.pPanel, "", DingFengPK.PickFactionSect, DingFengPK, tbInfo.nFaction, tbInfo.nFactionSect)
      end
    end)
  end
end

function tbWnd:UpdateTime(nRemainSec)
  if self.nRemainTimer then
    Timer:Close(self.nRemainTimer)
  end
  self.nRemainSec = nRemainSec
  
  local function pSetTimeFunc()
    self.nRemainSec = self.nRemainSec - 1
    if self.nRemainSec <= 0 then
      self:Close()
      return false
    else
      self.pPanel:Label_SetText(self.txtTime, Lib:TimeDesc2(self.nRemainSec))
      return true
    end
  end
  
  self.nRemainTimer = Timer:Register(Env.GAME_FPS, pSetTimeFunc, self)
end
