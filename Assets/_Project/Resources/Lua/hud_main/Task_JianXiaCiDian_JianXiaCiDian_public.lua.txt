JianXiaCiDian.nTaskGroup = GetGroupKey("AnswerQuestion")
JianXiaCiDian.szActivePointKey = "JianXiaCiDian"
JianXiaCiDian.nTaskCurQuesIdx = 1
JianXiaCiDian.nTaskAnswQuesNum = 2
JianXiaCiDian.nTaskCorrectNum = 3
JianXiaCiDian.nTaskDate = 4
JianXiaCiDian.nTotalQuestionNum = 10
JianXiaCiDian.nTimeOutSec = 600
JianXiaCiDian.nLevelLimit = 10
JianXiaCiDian.nTaskNpcTemplate = 9
JianXiaCiDian.nTaskNpcMapTemplate = 1
JianXiaCiDian.nAwardId = 1002
JianXiaCiDian.nCorrectKey = 1
JianXiaCiDian.nWrongKey = 2
JianXiaCiDian.nFinishKey = 3
JianXiaCiDian.szCloseTimeFrame = "OpenLevel69"

function JianXiaCiDian:LoadSetting()
  self.tbQuestions = Lib:LoadTabFileEx("Setting/AnswerQuestions/JianXiaCiDian.tab", "ddddsssss", "Id", {
    "Id",
    "AnswerId",
    "Group",
    "LevelMin",
    "Question",
    "Choice_1",
    "Choice_2",
    "Choice_3",
    "Choice_4"
  })
  self.tbRandomTable = {}
  for i = 1, self.nTotalQuestionNum do
    self.tbRandomTable[i] = {}
  end
  for nIndex, tbQuestion in ipairs(self.tbQuestions) do
    table.insert(self.tbRandomTable[tbQuestion.Group], {
      nIndex = nIndex,
      nLevelMin = tbQuestion.LevelMin or 1
    })
  end
end

function JianXiaCiDian:GetCurrentQustionIdx()
  return me.GetUserValue(self.nTaskGroup, self.nTaskCurQuesIdx)
end

function JianXiaCiDian:GetAnswerQustionNum()
  return me.GetUserValue(self.nTaskGroup, self.nTaskAnswQuesNum)
end

function JianXiaCiDian:GetCorrectNum()
  return me.GetUserValue(self.nTaskGroup, self.nTaskCorrectNum)
end

function JianXiaCiDian:GetDate()
  return me.GetUserValue(self.nTaskGroup, self.nTaskDate)
end

function JianXiaCiDian:CheckIsClose()
  if 1 == GetTimeFrameState(JianXiaCiDian.szCloseTimeFrame) then
    return true
  end
  return false
end

JianXiaCiDian:LoadSetting()
