local tbPanel = Ui:CreateClass("UITradeItemPanel", "UIPanel")
tbPanel.tbControls = {
  btnTrade = "btnTrade",
  btnCancel = "btnCancel",
  txtTradeInfo = "txtTradeInfo"
}
tbPanel.tbOnClick = {
  btnTrade = function(self)
    if Ui:WindowVisible("UICompareTips") then
      Ui:CloseWindow("UICompareTips")
    end
    if Ui:WindowVisible("UIItemTips") then
      Ui:CloseWindow("UIItemTips")
    end
    if Ui:WindowVisible("UIEquipTips") then
      Ui:CloseWindow("UIEquipTips")
    end
    if Ui:WindowVisible("UIZhenyuanTips") then
      Ui:CloseWindow("UIZhenyuanTips")
    end
    if Ui:WindowVisible("UIStoneTipsCompare") then
      Ui:CloseWindow("UIStoneTipsCompare")
    end
    if Ui:WindowVisible("UICompareTipsZhenyuan") then
      Ui:CloseWindow("UICompareTipsZhenyuan")
    end
    if Ui:WindowVisible("UIMeridiansTips") then
      Ui:CloseWindow("UIMeridiansTips")
    end
    if Ui:WindowVisible("UIPartnerEquipTips") then
      Ui:CloseWindow("UIPartnerEquipTips")
    end
    if Ui:WindowVisible("UICompareTipsPartnerEquip") then
      Ui:CloseWindow("UICompareTipsPartnerEquip")
    end
    if Ui:WindowVisible("UIDivineWeaponEquipTips") then
      Ui:CloseWindow("UIDivineWeaponEquipTips")
    end
    if Ui:WindowVisible("UICompareTipsDivineWeaponEquip") then
      Ui:CloseWindow("UICompareTipsDivineWeaponEquip")
    end
    if self.nOnTradeCount and 0 ~= self.nOnTradeCount then
      self.nAmount = self.nAmount + self.nOnTradeCount
    end
    if self.nItemId then
      Trade:MakeItemDecision(Trade.Operation.ADD, self.nItemId, self.nAmount)
    elseif self.nTemplateId then
      MiningAgency:ReqAddItem(self.nTemplateId, self.nAmount)
    end
  end,
  btnCancel = function(self)
    if Ui:WindowVisible("UICompareTips") then
      Ui:CloseWindow("UICompareTips")
    end
    if Ui:WindowVisible("UIItemTips") then
      Ui:CloseWindow("UIItemTips")
    end
    if Ui:WindowVisible("UIEquipTips") then
      Ui:CloseWindow("UIEquipTips")
    end
    if Ui:WindowVisible("UIZhenyuanTips") then
      Ui:CloseWindow("UIZhenyuanTips")
    end
    if Ui:WindowVisible("UIStoneTipsCompare") then
      Ui:CloseWindow("UIStoneTipsCompare")
    end
    if Ui:WindowVisible("UICompareTipsZhenyuan") then
      Ui:CloseWindow("UICompareTipsZhenyuan")
    end
    if Ui:WindowVisible("UIMeridiansTips") then
      Ui:CloseWindow("UIMeridiansTips")
    end
    if Ui:WindowVisible("UIPartnerEquipTips") then
      Ui:CloseWindow("UIPartnerEquipTips")
    end
    if Ui:WindowVisible("UICompareTipsPartnerEquip") then
      Ui:CloseWindow("UICompareTipsPartnerEquip")
    end
    if Ui:WindowVisible("UIDivineWeaponEquipTips") then
      Ui:CloseWindow("UIDivineWeaponEquipTips")
    end
    if Ui:WindowVisible("UICompareTipsDivineWeaponEquip") then
      Ui:CloseWindow("UICompareTipsDivineWeaponEquip")
    end
    if self.nItemId then
      Trade:MakeItemDecision(Trade.Operation.REMOVE, self.nItemId, self.nAmount)
    elseif self.nTemplateId then
      MiningAgency:ReqDeleteItem(self.nTemplateId, self.nAmount)
    end
  end
}

function tbPanel:UpdateNumberInput(nInput)
  local ret = nInput
  if nInput > self.nMaxCount then
    me.CenterMsg(i18n.EasyGet("\231\137\169\229\147\129\230\149\176\233\135\143\229\176\177\232\191\153\228\185\136\229\164\154\228\186\134"))
    ret = self.nMaxCount
    return ret
  end
  if nInput < 1 then
    me.CenterMsg(i18n.EasyGet("\231\137\169\229\147\129\230\149\176\233\135\143\228\184\141\232\131\189\229\134\141\229\176\145\228\186\134"))
    ret = 1
    return ret
  end
  self:SetCurCount(ret)
  return ret
end

function tbPanel:OnCreate()
  self.AmountAddAndSubGroup = Ui:NewExistAddAndSubGroup(self, "txtAmount/UIAddAndSubGroup")
  self.AmountAddAndSubGroup:SetNumPadPos(-200, -20)
end

function tbPanel:OnOpen(nItemIdOrTemplateId, nOnTradeCount, nMaxCount, bOnTrade)
  local pItem = KItem.GetItemObj(nItemIdOrTemplateId)
  if pItem then
    self.nItemId = nItemIdOrTemplateId
    self.nTemplateId = nil
  else
    self.nItemId = nil
    self.nTemplateId = nItemIdOrTemplateId
  end
  self.nMaxCount = nMaxCount or pItem.nCount
  self.nOnTradeCount = nOnTradeCount
  self.AmountAddAndSubGroup:SetNumRange(1, self.nMaxCount)
  if bOnTrade then
    self:SetCurCount(self.nMaxCount)
    self.AmountAddAndSubGroup:SetNumPadNotOpen()
  else
    self:SetCurCount(1)
    self.AmountAddAndSubGroup:SetNumPadOpen()
    self.AmountAddAndSubGroup:BindEvent(function(nValue)
      return self:UpdateNumberInput(nValue)
    end)
  end
  self.pPanel:SetActive("txtAmount/UIAddAndSubGroup/btnReduce", not bOnTrade)
  self.pPanel:SetActive("txtAmount/UIAddAndSubGroup/btnPlus", not bOnTrade)
  self.pPanel:SetActive("txtAmount/UIAddAndSubGroup/btnMax", not bOnTrade)
  self.pPanel:SetActive(self.btnTrade, not bOnTrade)
  self.pPanel:SetActive(self.btnCancel, bOnTrade)
  if self.nItemId then
    local bCanSell = true
    local szForeverInLimitMsg = ""
    local szNotInLimitMsg = ""
    if not pItem.dwTradeLimitType or pItem.dwTradeLimitType == Item.eTRADELIMITTYPE.NONE then
      szForeverInLimitMsg = i18n.EasyGet("<color=#b60000>\228\184\141\229\143\175\228\186\164\230\152\147</color>")
      bCanSell = false
    elseif pItem.dwTradeLimitType == Item.eTRADELIMITTYPE.NORMAL then
      szForeverInLimitMsg = i18n.EasyGet("\229\143\175\228\186\164\230\152\147")
    elseif pItem.dwTradeLimitType == Item.eTRADELIMITTYPE.TRADE_TIMES then
      if pItem.dwTradeLimitValue <= 0 then
        szForeverInLimitMsg = i18n.EasyGet("\228\186\164\230\152\147\230\172\161\230\149\176\239\188\154<color=#b60000>0</color>")
        bCanSell = false
      else
        szForeverInLimitMsg = i18n.EasyGet("\228\186\164\230\152\147\230\172\161\230\149\176\239\188\154<color=#28ff28>{0}</color>", pItem.dwTradeLimitValue)
      end
    elseif pItem.dwTradeLimitType == Item.eTRADELIMITTYPE.REFRESH_PROTECTED_TIME then
      if pItem.dwTradeLimitValue <= GetTime() then
        szNotInLimitMsg = i18n.EasyGet("\228\186\164\230\152\147\228\191\157\230\138\164\230\151\182\233\151\180\239\188\154\230\151\160")
      end
      local nTime = Lib:TimeDesc2(pItem.dwTradeLimitValue - GetTime())
      szNotInLimitMsg = i18n.EasyGet("\228\186\164\230\152\147\228\191\157\230\138\164\230\151\182\233\151\180\239\188\154<color=#b60000>{0}</color>", nTime)
      bCanSell = false
    end
    self.pPanel:SetActive("btnTrade/txtCanTrade", bCanSell)
    self.pPanel:SetActive("btnTrade/txtCanNotTrade", not bCanSell)
    self.pPanel:Button_SetEnable(self.btnTrade, bCanSell)
    self.pPanel:Label_SetText(self.txtTradeInfo, string.format("%s%s", szForeverInLimitMsg, szNotInLimitMsg))
    if bCanSell then
      self.pPanel:Label_SetColorByID(self.txtTradeInfo, 500)
    end
  elseif self.nTemplateId then
    local bCanMining, nErrCode = MiningAgency:CheckItemCanMining(self.nTemplateId)
    self.pPanel:SetActive("btnTrade/txtCanTrade", bCanMining)
    self.pPanel:SetActive("btnTrade/txtCanNotTrade", not bCanMining)
    self.pPanel:Button_SetEnable(self.btnTrade, bCanMining)
    self.pPanel:Label_SetText(self.txtTradeInfo, string.format("%s%s", "", ""))
    if bCanMining then
      self.pPanel:Label_SetColorByID(self.txtTradeInfo, 500)
    end
  end
end

function tbPanel:SetCurCount(count)
  self.AmountAddAndSubGroup:SetInputValue(count)
  self.nAmount = count
end
