function tbWnd:FinishEventHandle_ButtonEvent(szPath)
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_BUTTON_EVENT, self)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_BUTTON_EVENT, function(self, szPathInner)
    if szPath == szPathInner then
      self:FinishGuideStep()
    end
  end, self)
  table.insert(self.funcRelease, function()
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_BUTTON_EVENT, self)
  end)
end

function tbWnd:FinishEventHandle_ToggleEvent(szPath, bIsOn)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_TOGGLE_EVENT, function(self, szPathInner, bInnerIsOn)
    if szPath == szPathInner and bIsOn == bInnerIsOn then
      self:FinishGuideStep()
    end
  end, self)
  table.insert(self.funcRelease, function()
    EventNotify:UnRegistNotify(EventNotify.emNOTIFY_TOGGLE_EVENT, self)
  end)
end

    local scrollRect = tbInnerWnd[szScrollRect]
    local fViewportHeight = scrollRect.viewport.rect.height
    local fContentHeight = scrollRect.content.rect.height
    if fViewportHeight >= fContentHeight then
      self:FinishGuideStep()
    else
      local showUITransform = tbInnerWnd.pPanel:GetObject(szShowUIPath).transform
      local fShowUILocalYValue = showUITransform.rect.yMin + showUITransform.localPosition.y
      local fShowUIVerticalNormalizedPosition = (fContentHeight + fShowUILocalYValue) / (fContentHeight - fViewportHeight)
      
      local function fnListenMoveDown(vec2)
        if vec2.y <= fShowUIVerticalNormalizedPosition then
          self:FinishGuideStep()
        end
      end
      
      scrollRect.onValueChanged:AddListener(fnListenMoveDown)
      table.insert(self.funcRelease, function()
        scrollRect.onValueChanged:RemoveAllListeners()
      end)
    end

function tbWnd:CheckJumpCondition(szJumpCondition)
  local EventList = Lib:SplitStr(szJumpCondition, "|")
  for _, Event in pairs(EventList) do
    local Params = Lib:SplitStr(Event, ",")
    local Args = {}
    for idx = 2, #Params do
      table.insert(Args, Params[idx])
    end
    if tbWnd.JumpCondition[Params[1]] and tbWnd.JumpCondition[Params[1]](self, table.unpack(Args)) == true then
      return true
    end
  end
  return false
end

function tbWnd:ProcessDynamicMask(szDynamicMask)
  local Params = Lib:SplitStr(szDynamicMask, ",")
  local Args = {}
  for idx = 2, #Params do
    table.insert(Args, Params[idx])
  end
  if tbWnd.DynamicMaskElemHandle[Params[1]] then
    tbWnd.DynamicMaskElemHandle[Params[1]](self, table.unpack(Args))
  else
    self:DynamicTraceByPath(Params[1], Params[2])
  end
end

function tbWnd:DynamicTraceByPath(szWindowName, szPath)
  local wnd = Ui:GetWindow(szWindowName)
  if wnd and Ui:IsWindowLogicOpen(szWindowName) then
    local go = wnd.pPanel:GetObject(szPath)
    if go then
      self.DynamicMask:SetTarget(go)
      self.RelativeSortingGroup:SetSortingOrder(wnd.pPanel:GetMaxSortingOrderInChildren() + 1)
      self.pPanel:SetActive(self.DynamicHighLight, true)
      self.pPanel:SetActive(self.Region, true)
      local nStep = Guide:GetNowStep()
      EventNotify:RegistNotify(EventNotify.emNOTIFY_WND_CLOSED, function(self, szUI)
        if szWindowName == szUI then
          Timer:Register(1, function()
            if not Ui:IsWindowLogicOpen(szUI) and nStep == Guide:GetNowStep() then
              self:CloseGuideStep()
            end
          end)
        end
      end, self)
      table.insert(self.funcRelease, function()
        EventNotify:UnRegistNotify(EventNotify.emNOTIFY_WND_CLOSED, self)
      end)
      return
    end
  end
  self:CloseGuideStep()
end

function tbWnd:OnOpen(config)
  self.nOverrideSortingOrder = tonumber(config.nOverrideSortingOrder)
  self:Init()
  self.GuideID = tonumber(config.ID)
  self.GuideStep = tonumber(config.Step)
  if config.JumpCondition and config.JumpCondition ~= "" and self:CheckJumpCondition(config.JumpCondition) == true and config.JumpTo and "" ~= config.JumpTo then
    if config.JumpTo == "end" then
      Guide:FinishWholeGuide(self.GuideID)
    else
      local nToward = tonumber(config.JumpTo)
      if nil ~= nToward and nToward > 1 and nToward <= #Guide.Config[self.GuideID] then
        self:JumpCurrentGuide2step(nToward)
      end
    end
    return
  end
  if not config.FinishEvent or "" == config.FinishEvent then
    LogErr(string.format("There is no valid FinishEvent in Guild[%d, %d]", self.GuideID, self.GuideStep))
    self:Close()
    return
  end
  self:ProcessFinishEvent(config.FinishEvent)
  if config.FailEvent and "" ~= config.FailEvent then
    self:ProcessFailEvent(config.FailEvent)
  end
  if config.Foreground and "" ~= config.Foreground then
    if self.ForegroundHandle[config.Foreground] then
      self.ForegroundHandle[config.Foreground](self)
    else
      self.WorkingForeground = tbControls.ForegroundGroup .. "/" .. config.Foreground
      self.pPanel:SetActive(self.WorkingForeground, true)
    end
  end
  self.pPanel:SetActive(tbControls.Background, false)
  if config.DynamicHighLight and "" ~= config.DynamicHighLight then
    local objDynamicHighLight = self.pPanel:GetObject(tbControls.DynamicHighLight)
    if objDynamicHighLight then
      self.DynamicMask = objDynamicHighLight:GetComponent(Ui.DynamicMask)
      if config.HighLightOverrideSize and "" ~= config.HighLightOverrideSize then
        local tbSzCount = Lib:SplitStr(config.HighLightOverrideSize, ",")
        self.DynamicMask:SetGuideOverrideSize(Ui.Vector2(tonumber(tbSzCount[1]), tonumber(tbSzCount[2])))
      else
        self.DynamicMask:SetGuideSizeNotOverride()
      end
      if config.HighLightOffset and "" ~= config.HighLightOffset then
        local tbSzCount = Lib:SplitStr(config.HighLightOffset, ",")
        self.DynamicMask:SetGuideOffset(Ui.Vector2(tonumber(tbSzCount[1]), tonumber(tbSzCount[2])))
      else
        self.DynamicMask:SetGuideNoOffset()
      end
      self:ProcessDynamicMask(config.DynamicHighLight)
      if config.Comment and "" ~= config.Comment and "" ~= config.CommentType then
        self.pPanel:Label_SetText(self[config.CommentType] .. "/imgTxtBg/Text", i18n.Parse(config.Comment))
        self.pPanel:SetActive(self[config.CommentType], true)
      end
    end
  end
  if config.CommonDialog and "" ~= config.CommonDialog then
    self.pPanel:Label_SetText(tbControls.CommonDialogText, i18n.Parse(config.CommonDialog))
    self.pPanel:SetActive(tbControls.CommonDialog, true)
  end
  if config.ShowJumpBtn and config.ShowJumpBtn == "1" then
    self.pPanel:SetActive(tbControls.BtnSkip, true)
  end
  if config.HightLightHandIcon and "" ~= config.HightLightHandIcon then
    self.pPanel:SetActive(tbControls.HightLightHand .. "/long", config.HightLightHandIcon == "long")
    self.pPanel:SetActive(tbControls.HightLightHand .. "/short", config.HightLightHandIcon == "short")
    if config.HandOffset and "" ~= config.HandOffset then
      local offset = Lib:SplitStr(config.HandOffset, ",")
      self.DynamicMask:SetHandOffset(tonumber(offset[1]), tonumber(offset[2]))
    else
      self.DynamicMask:SetHandOffset(0, 0)
    end
    self.pPanel:SetActive(tbControls.HightLightHand, true)
  end
end

function tbWnd:OnClose()
  self:ReleaseUI()
end

function tbWnd:ReleaseUI()
  if self.nTimer_Finish then
    Timer:Close(self.nTimer_Finish)
    self.nTimer_Finish = nil
  end
  if self.nTimer_Fail then
    Timer:Close(self.nTimer_Fail)
    self.nTimer_Fail = nil
  end
  if self.WorkingHighLight then
    self.pPanel:SetActive(self.WorkingHighLight, false)
  end
  if self.WorkingForeground then
    self.pPanel:SetActive(self.WorkingForeground, false)
  end
  if self.WorkingBtnGroup then
    self.pPanel:SetActive(self.WorkingBtnGroup, false)
  end
  self.WorkingHighLight = nil
  self.WorkingForeground = nil
  self.WorkingBtnGroup = nil
  for _, funcRelease in ipairs(tbWnd.funcRelease) do
    funcRelease()
  end
  self.funcRelease = {}
end

function tbWnd:CloseGuideStep()
  self:Close()
  Guide:UnLockGuide()
  Guide:UpdateTrigger(self.GuideID)
end

function tbWnd:FinishGuideStep()
  self:Close()
  Guide:FinishGuideStep(self.GuideID, self.GuideStep)
end

function tbWnd:JumpCurrentGuide2step(nStep)
  self:Close()
  Guide:JumpToStep(self.GuideID, nStep)
end
