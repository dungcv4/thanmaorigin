local tbWnd = Ui:CreateClass("UIPartnerJumpTo", "UIPanel")
local tbControls = {
  Template = "imgBG/List/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls

function tbWnd:UpdateView(tbGetWayCfg)
  if not tbGetWayCfg then
    LogErr("UIPartnerJumpTo::UpdateView Error tbGetWayCfg is nil")
    return
  end
  if not self.tbScrollView then
    self.tbScrollView = Ui:NewScrollView(self, tbControls.Template, self.OnClickJumpTo)
  end
  self.tbGetWayCfg = tbGetWayCfg
  self.tbScrollView:Clear()
  for nIndex, tbInfo in ipairs(self.tbGetWayCfg) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      self:SetElement(tbElement, tbInfo)
    end)
  end
end

function tbWnd:SetElement(tbElement, tbWayInfo)
  tbElement.pPanel:Label_SetText("txtName", i18n.Parse(tbWayInfo.szTitle))
  tbElement.pPanel:Label_SetText("txtDesc", i18n.Parse(tbWayInfo.szDesc))
  tbElement.pPanel:SetActive("imgArrow", not Lib:IsEmptyStr(tbWayInfo.szJumpType))
end

function tbWnd:OnClickJumpTo(nIndex)
  if self.tbGetWayCfg[nIndex] and self.tbGetWayCfg[nIndex].szJumpType == "OpenWindow" then
    Ui:OpenWindow(table.unpack(Lib:SplitStrTry2Num(self.tbGetWayCfg[nIndex].szJumpParam, "|")))
    self.pPanel:SetActive("Main", false)
  end
end
