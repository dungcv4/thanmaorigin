local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.UITemporaryActivity)

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  self.ACTIVITY_CONFIG = {}
  local szFilepath = "Setting/Activity/Common/UITemporaryActivityCfg.tab"
  local tbField = {
    "nConfigID",
    "szName",
    "nLevel",
    "szShowTime",
    "szUIName",
    "szDesc",
    "nShowAwardId",
    "nShowAwardKey",
    "szImgBg",
    "szFirstOpen"
  }
  local tbLoadConfigs = Lib:LoadTabFileEx(szFilepath, "dsdsssddss", nil, tbField)
  for _, tbConfig in pairs(tbLoadConfigs or {}) do
    self.ACTIVITY_CONFIG[tbConfig.nConfigID] = self.ACTIVITY_CONFIG[tbConfig.nConfigID] or {}
    self.ACTIVITY_CONFIG[tbConfig.nConfigID] = tbConfig
  end
  szFilepath = "Setting/Activity/Common/UITemporaryStep.tab"
  tbField = {
    "nConfigID",
    "nStepID",
    "szName",
    "szDesc",
    "szImg",
    "szBtnName",
    "szCallBack",
    "szParam1",
    "szParam2",
    "szParam3",
    "nLinkActivityID"
  }
  tbLoadConfigs = Lib:LoadTabFileEx(szFilepath, "ddssssssssd", nil, tbField)
  for _, tbConfig in pairs(tbLoadConfigs or {}) do
    self.ACTIVITY_CONFIG[tbConfig.nConfigID].tbStepInfo = self.ACTIVITY_CONFIG[tbConfig.nConfigID].tbStepInfo or {}
    self.ACTIVITY_CONFIG[tbConfig.nConfigID].tbStepInfo[tbConfig.nStepID] = tbConfig
  end
  local tbJump = Lib:EasyLoadTabFile("Setting/Activity/Common/Jump.tab")
  for _, tbInfo in ipairs(tbJump) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("UITemporaryActivity:LoadConfig Jump nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpInfo = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpInfo or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpInfo[tbInfo.nJumpId] = tbInfo
  end
  local tbJumpMapLoc = Lib:EasyLoadTabFile("Setting/Activity/Common/JumpMapLoc.tab")
  for _, tbInfo in ipairs(tbJumpMapLoc) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("UITemporaryActivity:LoadConfig tbJumpMapLoc nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpMapLoc = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpMapLoc or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpMapLoc[tbInfo.nStepId] = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpMapLoc[tbInfo.nStepId] or {}
    table.insert(self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbJumpMapLoc[tbInfo.nStepId], tbInfo)
  end
  self.ACTIVITY_CONFIG = SetReadOnly(self.ACTIVITY_CONFIG)
  self.bLoadConfig = true
end

function tbActivity:GetUITemporarayCfg()
  return self.ACTIVITY_CONFIG[self.nConfigId]
end

function tbActivity:GetFirstOpenSz()
  return self.ACTIVITY_CONFIG[self.nConfigId].szFirstOpen
end

function tbActivity:GetStepsCfg()
  return self.ACTIVITY_CONFIG[self.nConfigId].tbStepInfo
end

tbActivity.tbMapDataGenFunc = {
  MidAutumnMap = function(tbJumpMapLoc)
    local pAchieveInstance = ActivityMgr:GetInstanceById(77)
    if pAchieveInstance and pAchieveInstance:IsOpen() then
      local tbAchieveAward = pAchieveInstance:GetAchieveData()
      local tbNotCollectIdList = pAchieveInstance:GetNotCollectIdList(4, me, tbAchieveAward)
      for _, tbInfo in ipairs(tbJumpMapLoc) do
        tbInfo.szExtraText = i18n.EasyGet("\229\183\178\232\181\160\233\128\129")
        for _, nCollectId in ipairs(tbNotCollectIdList) do
          if tbInfo.nMapTemplateId == nCollectId then
            tbInfo.szExtraText = nil
          end
        end
      end
    end
  end
}
