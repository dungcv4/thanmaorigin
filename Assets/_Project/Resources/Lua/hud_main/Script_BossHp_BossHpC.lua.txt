BossHp.defaultHpNum = 10
BossHp.defaultNoDamageCloseTime = 15
BossHp.bloodColor = {
  {
    nR = 243,
    nG = 231,
    nB = 17,
    nA = 255
  },
  {
    nR = 195,
    nG = 71,
    nB = 71,
    nA = 255
  }
}

function BossHp:UpdateBossHp(nNpcID, szBossName, nCurHP, nMaxHP)
  if self.nNpcId ~= nNpcID then
    return
  end
  if self.pNpc == nil then
    self.pNpc = KNpc.GetById(self.nNpcId)
  end
  local tbBossHpData = {}
  tbBossHpData.nBossId = nNpcID
  tbBossHpData.szName = self.pNpc.szName
  if 0 ~= self.pNpc.i64MaxLife then
    tbBossHpData.nHpNum = math.ceil(self.pNpc.i64CurLife / self.pNpc.i64MaxLife * BossHp.defaultHpNum)
    tbBossHpData.nHpRate = self.pNpc.i64CurLife / self.pNpc.i64MaxLife
  else
    tbBossHpData.nHpNum = 0
    tbBossHpData.nHpRate = 0
  end
  tbBossHpData.nSeries = self.pNpc.nSeries
  if Ui:WindowVisible("UIBloodBarBossTop") == false then
    Ui:OpenWindow("UIBloodBarBossTop", tbBossHpData)
  else
    Ui:GetWindow("UIBloodBarBossTop"):RefreshView(tbBossHpData)
  end
end

function BossHp:CloseBossHpView(nNpcID)
  self.nNpcId = nil
  if nil ~= self.pNpc then
    if Ui:WindowVisible("UIBloodBarBossTop") == true then
      Ui:GetWindow("UIBloodBarBossTop"):CloseView()
    end
    EventNotify.OnNotify(EventNotify.emNOTIFY_BOSS_HP_CLOSE)
  else
    if self.nHossHpCloseTimer then
      Timer:Close(self.nHossHpCloseTimer)
    end
    self.nHossHpCloseTimer = Timer:Register(Env.GAME_FPS * 2, function()
      if Ui:WindowVisible("UIBloodBarBossTop") == true then
        Ui:GetWindow("UIBloodBarBossTop"):CloseView()
      end
      EventNotify.OnNotify(EventNotify.emNOTIFY_BOSS_HP_CLOSE)
      return false
    end)
  end
end

function BossHp:OpenBossHpView(nNpcID)
  self.nNpcId = nNpcID
  self:UpdateBossHp(nNpcID, nil, nil, nil)
  EventNotify.OnNotify(EventNotify.emNOTIFY_BOSS_HP_OPEN)
end

function BossHp:OnBossDeath(nNpcID, nCurHP, nMaxHP)
  if nNpcID == self.nNpcId and nCurHP <= 0 then
    BossHp:CloseBossHpView()
  end
end

function BossHp:ClearData()
  self.nNpcId = nil
  self.pNpc = nil
  Timer:Close(self.nHossHpCloseTimer)
  self.nHossHpCloseTimer = nil
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_BOSS_ATTACKED, BossHp)
EventNotify:RegistNotify(EventNotify.emNOTIFY_BOSS_ATTACKED, BossHp.UpdateBossHp, BossHp)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_BOSS_HP, BossHp)
EventNotify:RegistNotify(EventNotify.emNOTIFY_BOSS_HP, BossHp.UpdateBossHp, BossHp)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_BOSS_INSIDE, BossHp)
EventNotify:RegistNotify(EventNotify.emNOTIFY_BOSS_INSIDE, BossHp.OpenBossHpView, BossHp)
EventNotify:UnRegistNotify(EventNotify.emNOTIFY_BOSS_OUTSIDE, BossHp)
EventNotify:RegistNotify(EventNotify.emNOTIFY_BOSS_OUTSIDE, BossHp.CloseBossHpView, BossHp)
