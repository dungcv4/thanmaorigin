local tbNpc = Npc:GetClass("IdentityComp")

function tbNpc:OnInit(pNpc)
  if MODULE_GAMECLIENT then
    Operation:DisableWalking()
  end
end

function tbNpc:OnExit(pNpc)
  if MODULE_GAMECLIENT then
    Operation:EnableWalking()
  end
end

function tbNpc:BanOperation(pNpc)
  if MODULE_GAMECLIENT then
    Operation:DisableWalking()
  end
end

function tbNpc:CanOperation(pNpc)
  if MODULE_GAMECLIENT then
    Operation:EnableWalking()
  end
end
