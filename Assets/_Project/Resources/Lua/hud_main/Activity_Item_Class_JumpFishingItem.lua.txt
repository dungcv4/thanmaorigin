Require("CommonScript/Item/ItemDefine.lua")
local tbJumpFishingItem = Item:GetClass("JumpFishingItem")
local ENTRANCE_PARAM = 1

function tbJumpFishingItem:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    local function func(tbWnd)
      if not tbWnd then
        return
      end
      local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.SummerFishing)
      if pInstance and pInstance:IsOpen() then
        tbWnd:OnClickActivity(pInstance.nId)
      else
        me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
      end
    end
    
    local szWndName = "UISummerActivity"
    if Ui:WindowVisible(szWndName) then
      return func(Ui:GetWindow(szWndName))
    end
    local nEntranceIndex = KItem.GetItemExtParam(nTemplateId, ENTRANCE_PARAM)
    Ui:OpenWindowWithCallback(szWndName, function(_, tbWnd)
      func(tbWnd)
    end, nil, nEntranceIndex)
  end
  
  return tbSetting
end

function tbJumpFishingItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "JumpFishingItem")
end
