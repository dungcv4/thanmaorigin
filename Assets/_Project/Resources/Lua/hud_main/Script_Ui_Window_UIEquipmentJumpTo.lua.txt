function tbWnd:UpdateView(nType)
  if not self.tbScrollView then
    self.tbScrollView = Ui:NewScrollView(self, tbControls.Template, self.OnClickJumpTo)
  end
  self.tbScrollView:Clear()
  self.nTargetListCfg = nil
  if 1 == nType then
    self.nTargetListCfg = self.tbEquipQiHunGetWayCfg
  elseif 2 == nType then
    self.nTargetListCfg = self.tbEquipBaoShiGetWayCfg
  elseif 3 == nType then
    self.nTargetListCfg = self.tbEquipLingZhuGetWayCfg
  elseif 4 == nType then
    self.nTargetListCfg = self.tbEquipLZWenYangGetWayCfg
  end
  for nIndex, tbInfo in ipairs(self.nTargetListCfg) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      self:SetElement(tbElement, tbInfo)
    end)
  end
end

function tbWnd:SetElement(tbElement, tbWayInfo)
  tbElement.pPanel:Label_SetText("txtName", tbWayInfo.szTitle())
  tbElement.pPanel:Label_SetText("txtDesc", tbWayInfo.szDesc())
  if tbWayInfo.bHideArrow then
    tbElement.pPanel:SetActive("imgArrow", false)
  end
end

function tbWnd:OnClickJumpTo(nIndex)
  if self.nTargetListCfg[nIndex] then
    self.nTargetListCfg[nIndex].func(self)
  end
end
