local tbWnd = Ui:CreateWindow("UITemporaryActivityComposeSet")
tbWnd.nMaxInputCount = 50
tbWnd.nMinInputCount = 1
local tbControls = {
  txtMax = "node/imgBG/Content/txtMax",
  btnOK = "node/imgBG/Content/btnOK",
  SliderNum = "node/imgBG/Content/SliderNum",
  txtMin = "node/imgBG/Content/txtMin",
  btnClose = "node/imgBG/btnClose",
  txtSetNumTip = "node/imgBG/Content/SetNum/txtSetNumTip",
  txtInputNumValue = "node/imgBG/Content/SetNum/txtSetNumValue",
  txtTitle = "node/imgBG/txtTitle"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    local nSliderValue = self.pPanel:Slider_GetValue(tbControls.SliderNum)
    if self.nValue > 0 and nSliderValue > 0 then
      table.insert(self.tbCallback, nSliderValue)
      Lib:CallBack(self.tbCallback)
    end
    self:Close()
  end
}
tbWnd.tbOnSliderClick = {
  SliderNum = function(self, nValue)
    self.nValue = nValue
    self.pPanel:Label_SetText(self.txtInputNumValue, nValue)
  end
}

function tbWnd:OnOpen(tbProduct, nMaxNum, nTargetNum, tbCallback)
  if 0 == nMaxNum then
    me.CenterMsg(i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179\239\188\140\232\175\183\229\137\141\229\142\187\232\142\183\229\143\150"))
    self:Close()
    return
  end
  self:SetShowInfo(tbProduct)
  self.nTargetNum = nTargetNum or self.nMinInputCount
  self.nValue = nMaxNum or self.nMaxInputCount
  self.tbCallback = tbCallback
  self.pPanel:Slider_SetRange(tbControls.SliderNum, 1, nMaxNum)
  self.pPanel:Slider_SetValue(tbControls.SliderNum, nMaxNum)
  self.pPanel:Label_SetText(tbControls.txtMin, 1)
  self.pPanel:Label_SetText(tbControls.txtMax, nMaxNum)
end

function tbWnd:SetShowInfo(tbProduct)
  local szProductName, _ = Award:GetAwardNameAndIcon(me, table.unpack(tbProduct))
  self.pPanel:Label_SetText(tbControls.txtSetNumTip, i18n.EasyGet("\233\128\137\230\139\169\229\136\182\228\189\156<color=#c56d05>{0}</color>\231\154\132\230\149\176\233\135\143", i18n.Parse(szProductName)))
  self.pPanel:Label_SetText(tbControls.txtTitle, i18n.Parse(szProductName))
end
