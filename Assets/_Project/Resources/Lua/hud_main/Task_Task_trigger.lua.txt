Require("CommonScript/lib.lua")
Task.tbTrigger_Level = Task.tbTrigger_Level or {}
local tbTrigger_Level = Task.tbTrigger_Level

function tbTrigger_Level:ParseTrigger(tbInfo)
  local tbCond = {}
  tbCond.szType = tbInfo[1]
  tbCond.nValue = tonumber(tbInfo[2])
  return tbCond
end

Task.tbTrigger_FightPower = Task.tbTrigger_FightPower or {}
local tbTrigger_FightPower = Task.tbTrigger_FightPower

function tbTrigger_FightPower:ParseTrigger(tbInfo)
  local tbCond = {}
  tbCond.szType = tbInfo[1]
  tbCond.nValue = tonumber(tbInfo[2])
  return tbCond
end

Task.tbTrigger_Task = Task.tbTrigger_Task or {}
local tbTrigger_Task = Task.tbTrigger_Task

function tbTrigger_Task:ParseTrigger(tbInfo)
  local tbCond = {}
  tbCond.szType = tbInfo[1]
  tbCond.nValue = tonumber(tbInfo[2])
  return tbCond
end

if MODULE_GAMESERVER then
  function tbTrigger_Level:CheckTrigger(pPlayer, tbTriggerInfo)
    return pPlayer.nLevel >= tbTriggerInfo.nValue
  end
  
  function tbTrigger_FightPower:CheckTrigger(pPlayer, tbTriggerInfo)
    return pPlayer.GetFightPower() >= tbTriggerInfo.nValue
  end
  
  function tbTrigger_Task:CheckTrigger(pPlayer, tbTriggerInfo)
    local nNeedFinishTaskId = tbTriggerInfo.nValue
    local tbTaskData = Task:GetTaskData(pPlayer)
    for _, nFinishedTaskId in pairs(tbTaskData.tbFinishedTaskIdList) do
      if nNeedFinishTaskId == nFinishedTaskId then
        return true
      end
    end
    return false
  end
end
Task.tbAllTriggerLogic = {
  level = Task.tbTrigger_Level,
  fightpower = Task.tbTrigger_FightPower,
  task = Task.tbTrigger_Task
}

function Task:GetTriggerLogic(szAction)
  return self.tbAllTriggerLogic[szAction]
end

function Task:ParseTrigger(szTrigger)
  if not szTrigger or "" == szTrigger then
    return {}
  end
  local tbCond = Lib:SplitStr(szTrigger, "|")
  local tbCLogic = self:GetTriggerLogic(tbCond[1])
  assert(tbCLogic, tbCond[1])
  return tbCLogic:ParseTrigger(tbCond)
end
