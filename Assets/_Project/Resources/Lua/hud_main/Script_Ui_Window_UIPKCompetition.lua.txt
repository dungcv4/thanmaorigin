local tbWnd = Ui:CreateWindow("UIPKCompetition")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UI_MESSAGEBOX_OPEN,
      self.OnMsgBoxOpen,
      self
    },
    {
      EventNotify.emNOTIFY_UI_MESSAGEBOX_CLOSE,
      self.OnMsgBoxClose,
      self
    },
    {
      EventNotify.emNOTIFY_DINGFENGPK_SYNC_STATE,
      self.UpdateRight,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/Cont_Video/btnClose",
  UIMoneyPanel = "node/imgBG/Cont_Video/UIMoneyPanel",
  ViewWeb = "node/imgBG/Cont_Video/Center/RawImage",
  txtActivityName = "node/imgBG/Cont_Video/Center/txtActivityName",
  btnGo = "node/imgBG/Cont_Video/Right/btnGo",
  btnRule = "node/imgBG/Cont_Video/Right/btnRule",
  btnBattle = "node/imgBG/Cont_Video/Right/btnBattle",
  btnRank = "node/imgBG/Cont_Video/Right/btnRank",
  btnTeam = "node/imgBG/Cont_Video/Right/btnTeam",
  btnGuess = "node/imgBG/Cont_Video/Right/btnguess",
  txtStartTime = "node/imgBG/Cont_Video/Right/txtStartTime",
  yingcaiCell = "node/imgBG/Cont_Video/Right/yingcaiCell",
  btnSubmit = "node/imgBG/Cont_Video/Right/yingcaiCell/btnSubmit",
  txtInfoHave = "node/imgBG/Cont_Video/Right/yingcaiCell/txtInfoHave",
  btnRewardCheck = "node/imgBG/Cont_Video/Right/yingcaiCell/btnRewardCheck",
  matchCell = "node/imgBG/Cont_Video/Right/matchCell",
  btnCheck = "node/imgBG/Cont_Video/Right/matchCell/btnCheck",
  ScrollView_Gift = "node/imgBG/Cont_Video/Right/matchCell/Scroll View_Gift",
  GiftList = "node/imgBG/Cont_Video/Right/matchCell/Scroll View_Gift/Viewport/Content/GiftList",
  txtInfo01 = "node/imgBG/Cont_Video/Right/matchCell/txtInfo01",
  txtInfo02 = "node/imgBG/Cont_Video/Right/matchCell/txtInfo02",
  txtNumberYingCai = "node/imgBG/Cont_Video/Right/matchCell/txtNumberYingCai",
  TemplateTime = "node/imgBG/Cont_Video/Down/Scroll View/Viewport/Content/Award",
  ZBimgBg = "node/imgBG/Cont_Video/Center/imgBg"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnGo = function(self)
    self:ClickButtonGo()
  end,
  btnRule = function(self)
    Ui:OpenWindow("UIGeneralWebView", DingFengPK.Website.szRuleWeb, 0, 0)
  end,
  btnBattle = function(self)
    if GetTime() < Lib:ParseDateTime(tostring(DingFengPK.TimeCfg.MID_END)) then
      Ui:OpenWindow("UIGeneralWebView", DingFengPK.Website.szJiJieWeb, 0, 0)
    elseif Lib:GetLocalDay(GetTime()) < Lib:GetLocalDay(Lib:GetDate2Time(tostring(DingFengPK.TimeCfg.FINAL_2))) then
      Ui:OpenWindow("UIGeneralWebView", DingFengPK.Website.szFinalWeb, 0, 0)
    else
      Ui:OpenWindow("UIPKCompetitionReport2")
      self:Close()
    end
  end,
  btnRank = function(self)
    if GetTime() < Lib:ParseDateTime(tostring(DingFengPK.TimeCfg.MID_START)) then
      Ui:OpenWindow("UIPKCompetitionReport1", DingFengPK.OPEN_TYPE.PRE)
    else
      Ui:OpenWindow("UIPKCompetitionReport1", DingFengPK.OPEN_TYPE.MID)
    end
    self:Close()
  end,
  btnTeam = function(self)
    Ui:OpenWindow("UIPKCompetitionTeam")
    self:Close()
  end,
  btnGuess = function(self)
    self:Close()
    Ui:OpenWindow("UIPKCompetitionGuess")
  end,
  btnSubmit = function(self)
    ActivityMgr:GotoNpc(DingFengPK.JieChuYingCaiCfg.NPC_ID, DingFengPK.JieChuYingCaiCfg.NPC_MAP_ID)
    self:Close()
  end,
  btnRewardCheck = function(self)
    Ui:OpenWindow("UIReward_Newyear", DingFengPK.YingCaiAward)
    self:Close()
  end,
  btnCheck = function(self)
    Ui:OpenWindow("UIGeneralWebView", DingFengPK.Website.szAwardWeb, 0, 0)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbScrollViewTime = Ui:NewScrollView(self, tbControls.TemplateTime)
  self.tbAwardListView = Ui:NewScrollView(self, tbControls.GiftList)
end

function tbWnd:OnOpen()
  DingFengPK:RequestGameState()
  self.pPanel:SetActive(tbControls.UIMoneyPanel, false)
  self.pPanel:SetActive(tbControls.ZBimgBg, false)
  self.pPanel:SetActive(tbControls.txtActivityName, false)
  self.pPanel:Button_SetText(tbWnd.btnSubmit, i18n.EasyGet("\228\184\138\228\186\164\232\139\177\230\137\141\228\187\164"), "Text")
  self:UpdateCenter()
  self:UpdateRight()
  self:UpdateDown()
end

function tbWnd:OnClose()
  ClientUtil:SetMusicVolume(self.nMusicVolume, false)
  ClientUtil:SetSoundVolume(self.nSoundVolume, false)
end

function tbWnd:UpdateCenter()
  self.pPanel:ClearCamera()
  self.pPanel:ShowUrl(tbControls.ViewWeb, DingFengPK.Website.szLiveWeb, 0, 0)
  self.nMusicVolume = ClientUtil:SetMusicVolume(0, false)
  self.nSoundVolume = ClientUtil:SetSoundVolume(0, false)
end

function tbWnd:UpdateRight()
  if GetTime() < Lib:ParseDateTime(tostring(DingFengPK.TimeCfg.MID_END)) then
    self.pPanel:Button_SetText(tbControls.btnBattle, i18n.EasyGet("\232\139\177\230\137\141\230\166\156"), "Text")
  else
    self.pPanel:Button_SetText(tbControls.btnBattle, i18n.EasyGet("\229\134\179\232\181\155\229\175\185\233\152\181"), "Text")
  end
  if GetTime() >= Lib:ParseDateTime(tostring(DingFengPK.TimeCfg.MID_START)) and DingFengPK:GetBattleTeamId(me.dwID) <= 0 then
    self.pPanel:Button_SetEnable(tbControls.btnTeam, false)
  else
    self.pPanel:Button_SetEnable(tbControls.btnTeam, true)
  end
  if DingFengPK:GetBattleTeamId(me.dwID) > 0 then
    self.pPanel:Button_SetText(tbControls.btnTeam, i18n.EasyGet("\230\136\145\231\154\132\230\136\152\233\152\159"), "Text")
  else
    self.pPanel:Button_SetText(tbControls.btnTeam, i18n.EasyGet("\229\136\155\229\187\186\230\136\152\233\152\159"), "Text")
  end
  local nStartTime = Lib:ParseDateTime(tostring(DingFengPK.tbGuessSetting[1].nStartTime))
  if nStartTime > GetTime() then
    local szStartTime = i18n.EasyGet("{0}\229\188\128\229\144\175", Lib:TimeDesc13(nStartTime))
    self.pPanel:Label_SetText(tbControls.txtStartTime, szStartTime)
    self.pPanel:Button_SetEnable(tbControls.btnGuess, false)
  else
    self.pPanel:Label_SetText(tbControls.txtStartTime, "")
    self.pPanel:Button_SetEnable(tbControls.btnGuess, true)
  end
  if DingFengPK:CheckInZoneRankime() then
    self.pPanel:SetActive(tbControls.yingcaiCell, true)
    self.pPanel:SetActive(tbControls.matchCell, false)
    local nNum = me.GetItemCountByTypeInBag(DingFengPK.JieChuYingCaiCfg.TOKEN_ID)
    self.pPanel:Label_SetText(tbControls.txtInfoHave, i18n.EasyGet("\229\183\178\230\139\165\230\156\137:{0}", nNum))
  else
    self.pPanel:SetActive(tbControls.yingcaiCell, false)
    self.pPanel:SetActive(tbControls.matchCell, true)
    self.pPanel:SetActive(tbControls.ScrollView_Gift, true)
    self.pPanel:SetActive(tbControls.txtNumberYingCai, false)
    self.pPanel:Button_SetText(tbControls.btnCheck, i18n.EasyGet("\230\159\165\231\156\139\230\175\148\232\181\155\229\165\150\229\138\177"), "Text")
    self:UpdateAward()
    local nGetOpenType = DingFengPK:GetOpenType()
    if nGetOpenType then
      if nGetOpenType == DingFengPK.OPEN_TYPE.PRE then
        self.pPanel:Label_SetText(tbControls.txtInfo02, i18n.EasyGet("\230\130\168\229\143\175\229\143\130\229\138\160{0}", i18n.Get(DingFengPK.TimeStageCfg[2].szTitle)))
      elseif nGetOpenType == DingFengPK.OPEN_TYPE.MID then
        self.pPanel:Label_SetText(tbControls.txtInfo02, i18n.EasyGet("\230\130\168\229\143\175\229\143\130\229\138\160{0}", i18n.Get(DingFengPK.TimeStageCfg[3].szTitle)))
      else
        local tbSetting = DingFengPK.FINAL_SETTING[nGetOpenType]
        if tbSetting and tbSetting[1] and tbSetting[1].nPhase then
          self.pPanel:Label_SetText(tbControls.txtInfo02, i18n.EasyGet("\230\130\168\231\154\132\230\175\148\232\181\155\230\151\182\233\151\180:\n{0}", i18n.Get(DingFengPK.ShowMatchTimeCfg[nGetOpenType - DingFengPK.OPEN_TYPE.MID])))
        end
      end
      if DingFengPK.bCanJoin then
        self.pPanel:SetActive(tbControls.txtInfo01, false)
        self.pPanel:SetActive(tbControls.txtInfo02, true)
      else
        self.pPanel:SetActive(tbControls.txtInfo01, true)
        self.pPanel:SetActive(tbControls.txtInfo02, false)
      end
    else
      self.pPanel:SetActive(tbControls.txtInfo01, false)
      self.pPanel:SetActive(tbControls.txtInfo02, false)
    end
  end
  self.pPanel:Button_SetText(tbControls.btnGo, i18n.EasyGet("\229\143\130\228\184\142\230\175\148\232\181\155"), "Text")
  if DingFengPK.nGameState == DingFengPK.GAME_STATE.OPEN then
    self.pPanel:Button_SetEnable(tbControls.btnGo, true)
  else
    self.pPanel:Button_SetEnable(tbControls.btnGo, false)
  end
end

function tbWnd:UpdateAward()
  self.tbAwardListView:Reset()
  for i = 1, 1 do
    self.tbAwardListView:SetElement(i, function(tbElement)
      if not tbElement.tbAwardView then
        tbElement.tbAwardView = Ui:NewItemGridScrollView(tbElement, "Content/ItemTemplate", 1, 1)
      end
      local tbAwards = Award:GetAward(DingFengPK.UIShowAward.nAwardId, DingFengPK.UIShowAward.nAwardKey)
      tbElement.tbAwardView:UpdateGenericItem(tbAwards, false)
    end)
  end
end

function tbWnd:UpdateDown()
  self.tbScrollViewTime:Reset()
  for i, tbData in ipairs(DingFengPK.TimeStageCfg) do
    self.tbScrollViewTime:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtTitle", i18n.Get(tbData.szTitle))
      local nStartTime = Lib:ParseDateTime(tostring(tbData.szTimeStart))
      local nEndTime = Lib:ParseDateTime(tostring(tbData.szTimeEnd))
      local szStartTime = os.date("%Y/%m/%d %H:%M", nStartTime)
      local szEndTime = os.date("%Y/%m/%d %H:%M", nEndTime)
      pPanel:Label_SetText("txtTime01", szStartTime)
      pPanel:Label_SetText("txtTime03", szEndTime)
      if not tbElement.tbCPStateBg then
        tbElement.tbCPStateBg = Ui:CreatePanel("UICP_StateBg", "UICP_StateBg", tbElement)
      end
      local nTime = GetTime()
      tbElement.tbCPStateBg:HideAll()
      if nStartTime > nTime then
        tbElement.tbCPStateBg:ShowNotStarted()
      elseif nEndTime < nTime then
        tbElement.tbCPStateBg:ShowEnd()
      else
        tbElement.tbCPStateBg:ShowInProgress()
      end
    end)
  end
end

function tbWnd:ClickButtonGo()
  DingFengPK:RequestAttend()
  self:Close()
end

function tbWnd:OnMsgBoxOpen()
  self:Close()
end

function tbWnd:OnMsgBoxClose()
  self:Close()
end
