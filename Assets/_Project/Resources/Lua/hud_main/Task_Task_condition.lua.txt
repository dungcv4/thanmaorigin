Task.tbCond_TimeFrame = Task.tbCond_TimeFrame or {}
local tbCond_TimeFrame = Task.tbCond_TimeFrame

function tbCond_TimeFrame:ParseCondition(tbInfo)
  local tbCond = {}
  tbCond.szType = tbInfo[1]
  local szOpenTimeFrame = tbInfo[2]
  if Calendar.tbMapTimeFrame[szOpenTimeFrame] == nil then
    LogErr("Task Condition, timeframe:", szOpenTimeFrame, "is not exsit")
    return
  end
  tbCond.szOpenTimeFrame = szOpenTimeFrame
  return tbCond
end

if MODULE_GAMESERVER then
  function tbCond_TimeFrame:CheckCondition(pPlayer, tbCond)
    return 0 ~= GetTimeFrameState(tbCond.szOpenTimeFrame)
  end
else
  function tbCond_TimeFrame:GetTarget(tbCond)
    return i18n.EasyGet("\230\173\164\228\187\187\229\138\161\229\176\154\230\156\170\229\136\176\229\188\128\230\148\190\230\151\182\233\151\180\232\189\180")

Task.tbAllConditionLogic = {
  level = Task.tbCond_Level,
  fightPower = Task.tbCond_FightPower,
  days = Task.tbCond_Days,
  timeFrame = Task.tbCond_TimeFrame,
  lastMainTask = Task.tbCond_LastMainTask
}

function Task:GetConditonLogic(szAction)
  return self.tbAllConditionLogic[szAction]
end

function Task:ParseCondition(szCondition)
  if not szCondition or "" == szCondition then
    return {}
  end
  local tbCond = Lib:SplitStr(szCondition, "|")
  local tbCLogic = self:GetConditonLogic(tbCond[1])
  return tbCLogic:ParseCondition(tbCond)
end
