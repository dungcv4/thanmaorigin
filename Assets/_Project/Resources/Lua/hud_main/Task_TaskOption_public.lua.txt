Task.tbOption_GoToPos = Task.tbOption_GoToPos or Lib:NewClass(Task.tbOption_Default)
local tbOption_GoToPos = Task.tbOption_GoToPos

function tbOption_GoToPos:ParseCfg(tbInfo)
  local nTaskId = tbInfo.nTaskId
  local nStep = tbInfo.nStep
  local nBranch = tbInfo.nBranch
  Task.CFG_DETAIL[nTaskId][nStep][nBranch].tbOption = {}
  local tbTemp = Task.CFG_DETAIL[nTaskId][nStep][nBranch].tbOption
  tbTemp.szOptTriggerTime = tbInfo.szOptTriggerTime
  tbTemp.szOptType = tbInfo.szOptType
  local szOptParam = tbInfo.OptParam1
  local tbOptParam = Lib:SplitStr(szOptParam, "|")
  tbTemp.nMapTemplateId = tonumber(tbOptParam[1])
  tbTemp.nX = tonumber(tbOptParam[2])
  tbTemp.nY = tonumber(tbOptParam[3])
end

Task.tbOption_Dialog = Task.tbOption_Dialog or Lib:NewClass(Task.tbOption_Default)
local tbOption_Dialog = Task.tbOption_Dialog

function tbOption_Dialog:ParseCfg(tbInfo)
  local nTaskId = tbInfo.nTaskId
  local nStep = tbInfo.nStep
  local nBranch = tbInfo.nBranch
  Task.CFG_DETAIL[nTaskId][nStep][nBranch].tbOption = {}
  local tbTemp = Task.CFG_DETAIL[nTaskId][nStep][nBranch].tbOption
  tbTemp.szOptTriggerTime = tbInfo.szOptTriggerTime
  tbTemp.szOptType = tbInfo.szOptType
  tbTemp.tbDialog = {}
  for i = 1, 10 do
    local szKey = string.format("OptParam%s", i)
    local szParam = tbInfo[szKey]
    if szParam and "" ~= szParam then
      tbTemp.tbDialog[i] = Task:ParseDialogMsg(szParam)
      assert(tbTemp.tbDialog[i].szIdentity ~= "him", szParam)
    end
  end
end
