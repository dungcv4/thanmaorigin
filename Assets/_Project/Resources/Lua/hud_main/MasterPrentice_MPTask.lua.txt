    "ID",
    "Title",
    "AchievementSubKindID",
    "Count",
    "PrenticeExp",
    "MasterMoney"

function MasterPrentice.MPTask:IsFinishGoal(pPlayer, nGoalId, tbMPData)
  local tbGoal = self.tbGoalSetting[nGoalId]
  if not tbGoal then
    return false
  end
  if not tbGoal.AchievementSubKindID or 0 == tbGoal.AchievementSubKindID then
    return false
  end
  local nCount = tbMPData.tbGoalData[nGoalId] or 0
  if nCount < tbGoal.Count then
    return false
  end
  return true
end

function MasterPrentice.MPTask:GetGoalData(tbTask, tbMPData)
  local nFinish, nCount = 0, 0
  local nGoalId = tbTask.ID
  tbMPData.tbGoalData = tbMPData.tbGoalData or {}
  local nCount = tbMPData.tbGoalData[nGoalId] or 0
  if 1 == self:GetGoalState(nGoalId, tbMPData.tbGoalFlagsBefore) then
    nFinish = TaskStatus.Expired
  elseif 1 == self:GetGoalState(nGoalId, tbMPData.tbGoalFlags) then
    nFinish = TaskStatus.Reported
  elseif self:IsFinishGoal(me, nGoalId, tbMPData) then
    nFinish = TaskStatus.Finished
  else
    nFinish = TaskStatus.Unfinish
  end
  return nFinish, nCount
end

local function _GetBitPos(nGoalId)
  return math.ceil(nGoalId / 30), math.mod(nGoalId, 30) + 1
end

function MasterPrentice.MPTask:GetGoalState(nGoalId, tbFlags)
  local nValueId, nBitIndex = _GetBitPos(nGoalId)
  tbFlags[nValueId] = tbFlags[nValueId] or 0
  return KLib.GetBit(tbFlags[nValueId], nBitIndex)
end

function MasterPrentice.MPTask:SetGoalState(nGoalId, tbFlags, bValue)
  local nValueId, nBitIndex = _GetBitPos(nGoalId)
  bValue = 0 ~= bValue and 1 or 0
  tbFlags[nValueId] = KLib.SetBit(tbFlags[nValueId], nBitIndex, bValue)
end

function MasterPrentice.MPTask:ReportGoal(masterId, prenticeId, tbMPData, bGiveAward, bForce)
  if 1 == tbMPData.bDismiss then
    return false, i18n.EasyGet("\232\167\163\233\153\164\229\184\136\229\190\146\229\133\179\231\179\187\231\138\182\230\128\129\228\184\173\230\151\160\230\179\149\230\177\135\230\138\165\228\187\187\229\138\161")
  end
  if tbMPData.status ~= MasterPrentice.Def.STATUS_APPRENTICE and 1 == bGiveAward then
    return false, i18n.EasyGet("\230\156\170\229\135\186\229\184\136\229\188\159\229\173\144\230\137\141\232\131\189\230\177\135\230\138\165\228\187\187\229\138\161")
  end
  local pPrentice = KPlayer.GetPlayerObjById(prenticeId)
  local nGoalCount, nPrenticeExp, nMasterMoney = 0, 0, 0
  for i = 1, #self.tbGoalSetting do
    for _continue = 1, 1 do
      local tbGoal = self.tbGoalSetting[i]
      local nGoalId = tbGoal.ID
      if not (1 ~= self:GetGoalState(nGoalId, tbMPData.tbGoalFlagsBefore) and 1 ~= self:GetGoalState(nGoalId, tbMPData.tbGoalFlags) and self:IsFinishGoal(pPrentice, nGoalId, tbMPData)) then
        break
      end
      nGoalCount = nGoalCount + 1
      self:SetGoalState(nGoalId, tbMPData.tbGoalFlags, 1)
      nPrenticeExp = nPrenticeExp + tbGoal.PrenticeExp
      nMasterMoney = nMasterMoney + tbGoal.MasterMoney
    end
  end
  if 0 == nGoalCount then
    return 1 == bForce and true or false, i18n.EasyGet("\230\178\161\230\156\137\229\174\140\230\136\144\228\187\187\228\189\149\230\150\176\231\154\132\229\135\186\229\184\136\228\187\187\229\138\161")
  end
  if 1 == bGiveAward then
    tbMPData.nFinishedGoalCount = (tbMPData.nFinishedGoalCount or 0) + nGoalCount
    MasterPrentice:AsyncCall(masterId, "MasterPrentice.MPTask:GiveGoalMasterAward", masterId, prenticeId, nGoalCount, nMasterMoney)
    MasterPrentice:AsyncCall(prenticeId, "MasterPrentice.MPTask:GiveGoalPrenticeAward", prenticeId, nGoalCount, nPrenticeExp)
  end
  MasterPrentice:NotifyGoalReported(pPrentice, masterId)
  local tbMasterInfo = KPlayer.GetRoleStayInfo(masterId)
  if tbMasterInfo then
    Lib:TrackEvent(prenticeId, Env.track_shitu_huibao, tbMasterInfo.szAccount, tbMasterInfo.dwID, tbMasterInfo.szName, nGoalCount)
  end
  Event:OnEvent(Event.emEVENT_MASTERPRENTICE_TASK, prenticeId, nGoalCount)
  return true, "", nGoalCount
end

  pMaster.CenterMsg(szMsg)
  MasterPrentice:NotifyMasterAward(pMaster, szMsg, i18n.EasyGet("\229\184\136\229\190\146\231\155\174\230\160\135"))
end

function MasterPrentice.MPTask:GiveGoalPrenticeAward(prenticeId, nGoalCount, nPrenticeExp)

  MasterPrentice:NotifyMasterAward(pPrentice, szMsg, i18n.EasyGet("\229\184\136\229\190\146\231\155\174\230\160\135"))
end

function MasterPrentice.MPTask:GetTaskData(tbTask, tbMPData)
  local nFinish, nCount = 0, 0
  local nTaskId = tbTask.ID
  if not tbMPData.tbTaskList then
    return TaskStatus.Unfinish, 0
  end
  local nCount = tbMPData.tbTaskList[nTaskId] or 0
  if self:IsFinishTask(me, nTaskId, tbMPData) then
    nFinish = TaskStatus.Finished
  else
    nFinish = TaskStatus.Unfinish
  end
  return nFinish, nCount
end

function MasterPrentice.MPTask:GetFinishedTaskCount(tbMPData)
  local nFinishedTaskCount, nPrenticeExp = 0, 0
  local pPrentice = me
  for nTaskId, nValue in pairs(tbMPData.tbTaskList) do
    if self:IsFinishTask(pPrentice, nTaskId, tbMPData) then
      nFinishedTaskCount = nFinishedTaskCount + 1
    end
  end
  nPrenticeExp = nFinishedTaskCount * MasterPrentice:GetPrenticeExpByLevel(pPrentice.nLevel)
  return nFinishedTaskCount, nPrenticeExp
end

function MasterPrentice.MPTask:IsFinishTask(pPlayer, nTaskId, tbMPData)
  local tbGoal = self.tbTaskSetting[nTaskId]
  if not tbGoal then
    return false
  end
  if not tbGoal.AchievementSubKindID or 0 == tbGoal.AchievementSubKindID then
    return false
  end
  local nCount = tbMPData.tbTaskList[nTaskId] or 0
  if nCount < tbGoal.Count then
    return false
  end
  return true
end

    return false

function MasterPrentice.MPTask:OnAchiChange(nPlayerId, nSubKindID, nCurCount, nNewCount)
  local pPlayer = KPlayer.GetPlayerObjById(nPlayerId)
  if not pPlayer then
    return
  end
  local masterCtrl = MasterPrentice:GetMPInfo(pPlayer)
  if not masterCtrl.tbMyMaster then
    return
  end
  local tbTask = self.tbTaskBySubKind[nSubKindID]
  local bCompleted
  if tbTask then
    for i, masterId in pairs(masterCtrl.tbMyMaster) do
      bCompleted = true
      local tbMPData = MasterPrentice:GetMPData(masterId, nPlayerId)
      if tbMPData and tbMPData.status == MasterPrentice.Def.STATUS_AFTER_APPENTICE and tbMPData.tbTaskList and tbMPData.tbTaskList[tbTask.ID] then
        local tbTaskList = tbMPData.tbTaskList
        tbTaskList[tbTask.ID] = tbTaskList[tbTask.ID] + nNewCount - nCurCount
        MasterPrentice:SaveMPData(masterId, nPlayerId, tbMPData)
        for nTaskId, nCount in pairs(tbTaskList) do
          local tbTaskSetting = self.tbTaskSetting[nTaskId]
          if not tbTaskSetting or nCount < tbTaskSetting.Count then
            bCompleted = false
            break
          end
        end
        if bCompleted then
          MasterPrentice:NotifyTaskCompleted(pPlayer, masterId, tbTask.ID)
        end
      end
    end
  end
  local tbGoals = self.tbGoalBySubKind[nSubKindID]
  if tbGoals then
    for i, masterId in pairs(masterCtrl.tbMyMaster) do
      local tbMPData = MasterPrentice:GetMPData(masterId, nPlayerId)
      if tbMPData and tbMPData.status == MasterPrentice.Def.STATUS_APPRENTICE then
        tbMPData.tbGoalData = tbMPData.tbGoalData or {}
        for _, aGoal in pairs(tbGoals) do
          tbMPData.tbGoalData[aGoal.ID] = nNewCount
          MasterPrentice:SaveMPData(masterId, nPlayerId, tbMPData)
          local nFinish = self:GetGoalData(aGoal, tbMPData)
          if nFinish == MasterPrentice.Def.TaskStatus.Finished then
            MasterPrentice:NotifyGoalCompleted(pPlayer, masterId, aGoal.ID)
          end
        end
      end
    end
  end
  MasterPrentice:SyncPlayerMPData(pPlayer)
  for i, masterId in pairs(masterCtrl.tbMyMaster) do
    local pMaster = KPlayer.GetPlayerObjById(masterId)
    if pMaster then
      MasterPrentice:SyncPlayerMPData(pMaster)
    end
  end
end

function MasterPrentice.MPTask:UpdateGoalData(pPlayer, tbMPData, nMasterId)
  local nSubKindID
  for _, tbGoal in pairs(self.tbGoalSetting) do
    nSubKindID = tbGoal.AchievementSubKindID
    if 0 ~= nSubKindID then
      local nCount = Achievement:GetSubKindCount(pPlayer, nSubKindID)
      tbMPData.tbGoalData[tbGoal.ID] = nCount
    end
  end
end

function MasterPrentice.MPTask:CheckAchievementIsValid()
  local nSubKindID
  for _, tbGoal in pairs(self.tbGoalSetting) do
    nSubKindID = tbGoal.AchievementSubKindID
    if not nSubKindID or nSubKindID <= 0 then
      LogErr("MasterPrentice.MPTask:CheckAchievementIsValid wrong, subKindID is not exist", tbGoal.AchievementSubKindID)
    end
  end
  for _, tbTask in pairs(self.tbTaskSetting) do
    nSubKindID = tbTask.AchievementSubKindID
    if not nSubKindID or nSubKindID <= 0 then
      LogErr("MasterPrentice.MPTask:CheckAchievementIsValid tbTaskSetting wrong, subKindID is not exist", tbTask.AchievementSubKindID)
    end
  end
end
