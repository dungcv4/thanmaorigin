Require("Script/ShenBingPu/ShenBingPuC.lua")
local tbWnd = Ui:CreateClass("UIDivineCompose", "UIPanel")
local ShenBingPuC = Item.ShenBingPu
local tbControls = {
  MiddleTopItem = "Middle/TopItem",
  MiddleLeftItem = "Middle/LeftItem",
  MiddleRightItem = "Middle/RightItem",
  MiddleSuccessItem = "Middle/ComposeInfo/Item01",
  MiddleFailureItem = "Middle/ComposeInfo/ItemFailure",
  MiddleTxtProbability = "Middle/ComposeInfo/TxtProbability01",
  MiddleTxtSuccessTittle = "Middle/ComposeInfo/TxtTittle",
  MiddleTxtFailureTittle = "Middle/ComposeInfo/TxtFailureTittle",
  RightbtnTabMain = "Right/Conternt/Tab/btnFaction1",
  RightbtnTabVice = "Right/Conternt/Tab/btnFaction2",
  RightIngredientTemplate = "Right/Conternt/Scroll View/Viewport/Content/Template",
  RightDropdownQuality = "Right/panelItemList/DropdownQuality",
  RightToggleAuto = "Right/ToggleAuto",
  RightbtnCompose = "Right/Conternt/btnCompose",
  RightDropDown = "Right/panelItemList/DropdownQuality/Template",
  RightDropDownItem = "Right/panelItemList/DropdownQuality/Template/Viewport/Content/Item",
  btnObtainDivineWeapon = "Btn_ObtainDivineWeapon",
  Fail_Effect = "Fail_Effect",
  Success_Effect = "Success_Effect"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnObtainDivineWeapon = function(self)
    self:OnGetMoreWeaponClicked()
  end,
  RightbtnCompose = function(self)
    if not self.nMainWeaponId then
      me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\230\148\190\229\133\165\228\184\187\230\157\144\230\150\153"))
      return
    end
    if not self.nViceWeaponId then
      local bAuto = self.pPanel:Toggle_GetChecked(tbControls.RightToggleAuto)
      if not bAuto then
        me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\230\148\190\229\133\165\232\190\133\230\157\144\230\150\153"))
      else
        self:CheckAutoCompose()
      end
      return
    end
    ShenBingPuC:DoComposeReq(self.nMainWeaponId, self.nViceWeaponId)
  end
}
tbWnd.RIGHT_BTN_GROUP = {
  [1] = tbControls.RightbtnTabMain,
  [2] = tbControls.RightbtnTabVice
}

function tbWnd:OnCreate()
  self.tbMiddleTopGrid = Ui:NewExistItemGrid(self, tbControls.MiddleTopItem)
  self.tbMiddleTopGrid:Plus_SetActive(false)
  self.tbMiddleLeftGrid = Ui:NewExistItemGrid(self, tbControls.MiddleLeftItem)
  self.tbMiddleLeftGrid:Plus_SetActive(false)
  self.tbMiddleRightGrid = Ui:NewExistItemGrid(self, tbControls.MiddleRightItem)
  self.tbMiddleRightGrid:Plus_SetActive(false)
  self.tbMiddleSuccessGrid = Ui:NewExistItemGrid(self, tbControls.MiddleSuccessItem)
  self.tbMiddleSuccessGrid:Plus_SetActive(false)
  self.tbMiddleFailureGrid = Ui:NewExistItemGrid(self, tbControls.MiddleFailureItem)
  self.tbMiddleFailureGrid:Plus_SetActive(false)
  self.tbRightIngredientScrollView = Ui:NewScrollView(self, tbControls.RightIngredientTemplate, self.OnRightIngredientClick, true)
  self.pPanel:Dropdown_BindEvent(tbControls.RightDropdownQuality, function(value)
    self:OnDropDownChange(value)
  end)
  self.tbIngredientBtnGroup = Ui:NewButtonGroup(self, tbWnd.RIGHT_BTN_GROUP, true, self.OnRightTabBtnClick)
end

function tbWnd:OnPanelOpen(nSelectIndex)
  self.nMainWeaponId = nil
  self.nViceWeaponId = nil
  self.nTabIndex = 1
  self.nDropDownValue = 0
  self.pPanel:Toggle_SetChecked(tbControls.RightToggleAuto, false)
  self:UpdateMiddleAndRight(nSelectIndex)
end

function tbWnd:OnSelectWeaponChange(nSelectIndex)
  self:UpdateMiddleAndRight(nSelectIndex)
end

function tbWnd:OnRightTabBtnClick(nBtnIndex)
  self.nTabIndex = nBtnIndex
  local nSlot = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + self.nDropDownValue
  self:UpdateRigthIngredientView(nSlot)
  self.tbMiddleLeftGrid.pPanel:SetActive("Img_Select", 1 == self.nTabIndex)
  self.tbMiddleRightGrid.pPanel:SetActive("Img_Select", 2 == self.nTabIndex)
end

function tbWnd:UpdateMiddleAndRight(nSelectIndex)
  local nSlot = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + nSelectIndex - 1
  self:UpdateMiddleInfo(nSlot)
  self:UpdateRightInfo(nSlot)
end

function tbWnd:UpdateMiddleGrid(tbMiddleGrid, bPlusActive)
  tbMiddleGrid:Clear()
  tbMiddleGrid:BaseColor_SetSpriteByQuality(1)
  tbMiddleGrid:BaseColor_SetActive(true)
  tbMiddleGrid:Plus_SetActive(bPlusActive or false)
  if tbMiddleGrid.pPanel:GetObject("Img_Select", false) then
    tbMiddleGrid.pPanel:SetActive("Img_Select", false)
  end
end

function tbWnd:UpdateMiddleInfo(nSlot)
  self:UpdateMiddleGrid(self.tbMiddleTopGrid)
  self:UpdateMiddleGrid(self.tbMiddleLeftGrid, true)
  self:UpdateMiddleGrid(self.tbMiddleRightGrid, true)
  self:UpdateMiddleGrid(self.tbMiddleSuccessGrid)
  self:UpdateMiddleGrid(self.tbMiddleFailureGrid)
  self.tbMiddleTopGrid:BaseColor_SetActive(true)
  self.tbMiddleTopGrid:Plus_SetActive(false)
  self.tbMiddleTopGrid.pPanel:SetActive("Img_Select", false)
  self.tbMiddleLeftGrid:SetClickEnable(true, self.OnRightTabBtnClick, self, 1)
  self.tbMiddleRightGrid:SetClickEnable(true, self.OnRightTabBtnClick, self, 2)
  self.pPanel:SetActive(tbControls.MiddleTxtProbability, false)
end

function tbWnd:UpdateRightInfo(nSlot)
  self.tbRightIngredientScrollView:ResetFrom(1)
  self:UpdateDropDownOpt(nSlot)
  self.tbIngredientBtnGroup:Select(self.nTabIndex)
end

function tbWnd:UpdateDropDownOpt(nSlot)
  if not self.tbOptList then
    self.tbOptList = {}
    for i = Item.ShenBingPu.WEAPON_1, Item.ShenBingPu.WEAPON_3 do
      local szWeaponTypeName = ShenBingPuC:GetFactionWeaponName(me.nFaction, me.nFactionSect, i)
      table.insert(self.tbOptList, szWeaponTypeName)
    end
    self.pPanel:Dropdown_ResetOptions(tbControls.RightDropdownQuality, self.tbOptList)
  end
  local nNewValue = nSlot and nSlot - Item.ShenBingPu.WEAPON_1 or 0
  self.pPanel:Dropdown_SetValue(tbControls.RightDropdownQuality, nNewValue)
end

function tbWnd:OnDropDownChange(nValue)
  if nValue + 1 ~= self.tbOwner.tbLeftBtnGroup.nSelectIndex then
    self.tbOwner:OnLeftBtnClick(nValue + 1)
  end
  self.nDropDownValue = nValue
  self.nMainWeaponId = nil
  self.nViceWeaponId = nil
  local nSlot = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + self.nDropDownValue
  self:UpdateMiddleInfo(nSlot)
  self:UpdateRigthIngredientView(nSlot)
  self:OnRightTabBtnClick(self.nTabIndex)
end

function tbWnd:UpdateRigthIngredientView(nSlot)
  self.tbRightIngredientScrollView:ResetFrom(1)
  local tbCandidateWeapon = ShenBingPuC:GetAllCanComposeShenBingPuInBagBySlot(me, nSlot)
  for nIndex, pWeapon in ipairs(tbCandidateWeapon) do
    self.tbRightIngredientScrollView:SetElement(nIndex, function(tbElement, _)
      self:__SetRightIngredientElement(tbElement, pWeapon)
    end)
  end
end

function tbWnd:__SetRightIngredientElement(tbElement, pWeapon)
  if not tbElement.tbItemGrid then
    tbElement.tbItemGrid = Ui:NewExistItemGrid(tbElement, "Item")
  end
  tbElement.tbItemGrid:SetItem(pWeapon.dwId, pWeapon.dwTemplateId, {bShowTip = true})
  tbElement.nWeaponId = pWeapon.dwId
  tbElement.nWeaponTemplateId = pWeapon.dwTemplateId
  tbElement.tbItemGrid:SetClickEnable(true)
  tbElement.tbItemGrid:Plus_SetActive(false)
  local szName, _, _, _ = KItem.GetItemShowInfo(pWeapon.dwTemplateId, me.nFaction, me.nFactionSect, me.nSex)
  tbElement.pPanel:Label_SetText("txtName", i18n.Parse(szName))
  if pWeapon.szClass == "ShenBingPuSeed" then
    tbElement.pPanel:Label_SetText("txtNum", i18n.EasyGet("\230\136\152\229\138\155\239\188\154{0}", ShenBingPuC:GetWeaponFightPowerForSeed(me, pWeapon)))
  elseif pWeapon.szClass == "ShenBingPuItem" then
    tbElement.pPanel:Label_SetText("txtNum", i18n.EasyGet("\230\136\152\229\138\155\239\188\154{0}", ShenBingPuC:GetWeaponFightPower(me, pWeapon)))
  end
  if self.nMainWeaponId == pWeapon.dwId then
    tbElement.pPanel:SetActive("Main", 1 == self.nTabIndex)
    tbElement.pPanel:SetActive("ImgSelcet", 1 == self.nTabIndex)
  elseif self.nViceWeaponId == pWeapon.dwId then
    tbElement.pPanel:SetActive("Main", 2 == self.nTabIndex)
    tbElement.pPanel:SetActive("ImgSelcet", 2 == self.nTabIndex)
  else
    tbElement.pPanel:SetActive("ImgSelcet", false)
    tbElement.pPanel:SetActive("Main", true)
  end
end

function tbWnd:OnRightIngredientClick(nIndex, tbElement)
  self:UpdateMiddleGrid(self.tbMiddleTopGrid, false)
  if 1 == self.nTabIndex then
    if self.nMainWeaponId == tbElement.nWeaponId then
      self.nMainWeaponId = nil
      self:UpdateMiddleGrid(self.tbMiddleLeftGrid, true)
      self.tbMiddleLeftGrid:SetClickEnable(false)
      local trans = tbElement.pPanel:GetComponent(Ui.BGroupTrans)
      trans:OnUnSelected()
    else
      self.nMainWeaponId = tbElement.nWeaponId
      self.tbMiddleLeftGrid:SetItem(self.nMainWeaponId, nil, {bShowTip = true})
      self.tbMiddleLeftGrid:Plus_SetActive(false)
      self.tbMiddleLeftGrid:SetClickEnable(true)
    end
  elseif 2 == self.nTabIndex then
    if self.nViceWeaponId == tbElement.nWeaponId then
      self.nViceWeaponId = nil
      self:UpdateMiddleGrid(self.tbMiddleRightGrid, true)
      self.tbMiddleRightGrid:SetClickEnable(false)
      local trans = tbElement.pPanel:GetComponent(Ui.BGroupTrans)
      trans:OnUnSelected()
    else
      self.nViceWeaponId = tbElement.nWeaponId
      self.tbMiddleRightGrid:SetItem(self.nViceWeaponId, nil, {bShowTip = true})
      self.tbMiddleRightGrid:Plus_SetActive(false)
      self.tbMiddleRightGrid:SetClickEnable(true)
    end
  end
  if self.nMainWeaponId and self.nViceWeaponId then
    local nSuccessRate, nComposeTemplateId, nRefundCoinNum = ShenBingPuC:GetComposeSuccessResultStrict(self.nMainWeaponId, self.nViceWeaponId)
    self.pPanel:Label_SetText(tbControls.MiddleTxtProbability, string.format("%d%%", nSuccessRate))
    self.pPanel:SetActive(tbControls.MiddleTxtProbability, true)
    if nSuccessRate > 0 and nComposeTemplateId > 0 then
      self.tbMiddleSuccessGrid:SetItemByTemplate(nComposeTemplateId)
      self.tbMiddleFailureGrid:SetDigitalItem(Currency.TYPE.Coin, nRefundCoinNum)
    else
      _, nComposeTemplateId, nRefundCoinNum = ShenBingPuC:GetComposeSuccessResultRelex(self.nMainWeaponId)
      self.pPanel:Label_SetText(tbControls.MiddleTxtProbability, "0%")
      self.pPanel:SetActive(tbControls.MiddleTxtProbability, true)
      self.tbMiddleSuccessGrid:SetItemByTemplate(nComposeTemplateId)
      self.tbMiddleFailureGrid:SetDigitalItem(Currency.TYPE.Coin, nRefundCoinNum)
    end
    self.pPanel:SetActive(tbControls.MiddleFailureItem, true)
    self.pPanel:SetActive(tbControls.MiddleSuccessItem, true)
  elseif not (not self.nMainWeaponId or self.nViceWeaponId) or not self.nMainWeaponId and self.nViceWeaponId then
    local _, nComposeTemplateId, nRefundCoinNum = ShenBingPuC:GetComposeSuccessResultRelex(self.nMainWeaponId or self.nViceWeaponId)
    self.pPanel:Label_SetText(tbControls.MiddleTxtProbability, "0%")
    self.pPanel:SetActive(tbControls.MiddleTxtProbability, true)
    self.tbMiddleSuccessGrid:SetItemByTemplate(nComposeTemplateId)
    self.tbMiddleFailureGrid:SetDigitalItem(Currency.TYPE.Coin, nRefundCoinNum)
    self.pPanel:SetActive(tbControls.MiddleFailureItem, true)
    self.pPanel:SetActive(tbControls.MiddleSuccessItem, true)
  else
    self.pPanel:SetActive(tbControls.MiddleTxtProbability, false)
    self.pPanel:SetActive(tbControls.MiddleTxtSuccessTittle, false)
    self.pPanel:SetActive(tbControls.MiddleTxtFailureTittle, false)
    self.pPanel:SetActive(tbControls.MiddleFailureItem, false)
    self.pPanel:SetActive(tbControls.MiddleSuccessItem, false)
  end
  tbElement.pPanel:Button_SetEnable("Main", true)
end

function tbWnd:OnComposeSucc(nNewWeaponId)
  self.nMainWeaponId = nil
  self.nViceWeaponId = nil
  self:UpdateMiddleAndRight(self.nDropDownValue + 1)
  self.tbMiddleTopGrid:SetItem(nNewWeaponId, nil, {bShowTip = true})
  self.tbMiddleTopGrid:SetClickEnable(true)
  self.pPanel:Toggle_SetChecked(tbControls.RightToggleAuto, false)
  if self.nAutoTimer then
    Timer:Close(self.nAutoTimer)
  end
  self:ShowSuccessEffect()
end

function tbWnd:OnComposeFail(nMainItemId)
  local nSlot = Item.ExtEquip.CFG.SLOT.SHENBINGPU_WEAPON1 + self.nDropDownValue
  self.nMainWeaponId = nMainItemId
  self.nViceWeaponId = nil
  self:UpdateMiddleInfo(self.nDropDownValue + 1)
  self:UpdateRigthIngredientView(nSlot)
  self.tbMiddleLeftGrid:SetItem(self.nMainWeaponId, nil, {bShowTip = true})
  self.tbMiddleLeftGrid:SetClickEnable(true)
  self.tbMiddleLeftGrid:Plus_SetActive(false)
  if self.nAutoTimer then
    Timer:Close(self.nAutoTimer)
  end
  self.nAutoTimer = Timer:Register(Env.GAME_FPS * 2, self.CheckAutoCompose, self)
  self:ShowFailEffect()
end

function tbWnd:ShowSuccessEffect()
  if self.nSuccessEffectTimer then
    Timer:Close(self.nSuccessEffectTimer)
    self.pPanel:SetActive(tbControls.Success_Effect, false)
  end
  self.pPanel:Label_SetText(tbControls.Success_Effect .. "/txtSuccess", i18n.EasyGet("\229\144\136\230\136\144\230\136\144\229\138\159"))
  self.pPanel:SetActive(tbControls.Success_Effect, true)
  self.nSuccessEffectTimer = Timer:Register(Env.GAME_FPS, function()
    self.pPanel:SetActive(tbControls.Success_Effect, false)
    self.nSuccessEffectTimer = nil
  end)
end

function tbWnd:ShowFailEffect()
  if self.nFailEffectTimer then
    Timer:Close(self.nFailEffectTimer)
    self.pPanel:SetActive(tbControls.Fail_Effect, false)
  end
  self.pPanel:Label_SetText(tbControls.Fail_Effect .. "/txtSuccess", i18n.EasyGet("\229\144\136\230\136\144\229\164\177\232\180\165"))
  self.pPanel:SetActive(tbControls.Fail_Effect, true)
  self.nFailEffectTimer = Timer:Register(Env.GAME_FPS, function()
    self.pPanel:SetActive(tbControls.Fail_Effect, false)
    self.nFailEffectTimer = nil
  end)
end

function tbWnd:CheckAutoCompose()
  local bAutoCompose = self.pPanel:Toggle_GetChecked(tbControls.RightToggleAuto)
  if not bAutoCompose then
    self.nAutoTimer = nil
    return false
  end
  if not self.nMainWeaponId then
    self.nAutoTimer = nil
    return false
  end
  local pWeapon = me.GetItemInBag(self.nMainWeaponId)
  if not pWeapon then
    self.nAutoTimer = nil
    return false
  end
  if 2 ~= self.nTabIndex then
    self.tbIngredientBtnGroup:Select(2)
  end
  local tbAllElement = self.tbRightIngredientScrollView:GetAllElement()
  for nIndex, tbElement in ipairs(tbAllElement) do
    if tbElement.pPanel:IsActive("Main") then
      local tbEquipComposeCfg = ShenBingPuC.tbEquipComposeCfg[pWeapon.dwTemplateId][tbElement.nWeaponTemplateId]
      if tbEquipComposeCfg then
        self.tbRightIngredientScrollView:SelectElement(nIndex)
        ShenBingPuC:DoComposeReq(self.nMainWeaponId, self.nViceWeaponId)
        break
      end
    end
  end
  self.nAutoTimer = nil
  return false
end

function tbWnd:OnGetMoreWeaponClicked()
  self.tbOwner:ShowGetMorePanel()
end

function tbWnd:OnClose()
  self.nMainWeaponId = nil
  self.nViceWeaponId = nil
  self.nTabIndex = 1
  self.nDropDownValue = 0
  self.tbOptList = nil
  self:FadeAllSlotRedPoint()
  if self.nAutoTimer then
    Timer:Close(self.nAutoTimer)
  end
  if self.nFailEffectTimer then
    Timer:Close(self.nFailEffectTimer)
    self.nFailEffectTimer = nil
  end
  if self.nSuccessEffectTimer then
    Timer:Close(self.nSuccessEffectTimer)
    self.nSuccessEffectTimer = nil
  end
end

function tbWnd:RefreshSlotRedPoint()
end

function tbWnd:FadeAllSlotRedPoint()
end

function tbWnd:OnSyncItem(dwItemId, dwTemplateId)
  self.nMainWeaponId = nil
  self.nViceWeaponId = nil
  self:UpdateMiddleAndRight(self.nDropDownValue + 1)
end

function tbWnd:OnDelItem(dwTemplateId)
  self.nMainWeaponId = nil
  self.nViceWeaponId = nil
  self:UpdateMiddleAndRight(self.nDropDownValue + 1)
end
