Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.AgeThief)
tbActivity.nSaveId = nil
tbActivity.AgeThiefOpenCount = 1
tbActivity.nLastStartTimeKey = 2

function tbActivity:LoadConfig()
  if self.bLoadConfig then
    return
  end
  local szFilepath = "Setting/Activity/ChildrenActivity/PreTask/AgeThiefCfg.tab"
  local tbActivityConfig = Lib:EasyLoadTabFile(szFilepath, {"nConfigId"})
  if not tbActivityConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  self.ACTIVITY_CONFIG = {}
  for nConfigId, tbConfig in pairs(tbActivityConfig) do
    self.ACTIVITY_CONFIG[nConfigId] = tbConfig
  end
  szFilepath = "Setting/Activity/ChildrenActivity/PreTask/AgeThief.tab"
  local tbAgeThiefConfig = Lib:EasyLoadTabFile(szFilepath)
  if not tbAgeThiefConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  for _, tbInfo in pairs(tbAgeThiefConfig) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("AgeThief:LoadAgeThiefConfig failed, nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAgeThiefConfig = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAgeThiefConfig or {}
    if self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAgeThiefConfig[tbInfo.nMapId] then
      LogErr("AgeThief: LoadAgeThiefConfig Error, duplicate nMapId=%d", tbInfo.nMapId)
    else
      self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAgeThiefConfig[tbInfo.nMapId] = {
        szTimeFrame = tbInfo.szTimeFrame
      }
    end
  end
  self.bLoadConfig = true
end

function tbActivity:LoadConfigWithGroup(tbConfig)
  local szGroupName = tbConfig.szGroupName
  local szGroupSubType = tbConfig.szGroupSubType
  local tbActCfg = ActivityMgr:GetActivityCfgWithGroup(szGroupName, szGroupSubType)
  self.tbConfig = tbActCfg.Cfg
  self.tbConfig.szDesc = i18n.Encode(self.tbConfig.szDesc)
  self.tbConfig.szName = i18n.Encode(self.tbConfig.szName)
  self.tbConfig.szShowTime = i18n.Encode(self.tbConfig.szShowTime)
  self.tbConfig.tbAgeThiefConfig = {}
  for _, tbInfo in ipairs(tbActCfg.AgeThief) do
    self.tbConfig.tbAgeThiefConfig[tbInfo.nMapId] = tbInfo
  end
  self.tbConfig = SetReadOnly(self.tbConfig, true)
end

function tbActivity:CheckConfigValid()
  local tbConfig = self:GetConfig()
  if Lib:IsEmptyStr(tbConfig.szSaveKey) then
    LogErr("[AgeThief:CheckConfigValid] bad savekey! please check it! szSaveKey=", tbConfig.szSaveKey, self:GetId())
    return false
  end
  return true
end

function tbActivity:CheckIsInTimeFrame(szTimeFrame)
  if szTimeFrame then
    local szTime = Lib:GetMaxTimeFrame(self.TimeFrame)
    return szTimeFrame == szTime
  else
    return false
  end
end

function tbActivity:GetRemainAwardCount(pPlayer)
  if not pPlayer then
    return 0
  end
  local nCount = pPlayer.GetUserValue(self.nSaveId, self.AgeThiefOpenCount)
  return nCount
end

function tbActivity:SetRemainAwardCount(pPlayer, nCount)
  if not pPlayer or not nCount then
    return
  end
  pPlayer.SetUserValue(self.nSaveId, self.AgeThiefOpenCount, nCount)
end

function tbActivity:GetLogWayFieldName()
  return ActivityMgr.NoLogWay
end
