local tbWnd = Ui:CreateWindow("UIResourceCompetition")
local nMaxHome = 8
tbWnd.tbOnClick = {}
local tbControls = {
  txtTime = "node/imgBG/txtTime"
}
for i = 1, nMaxHome do
  local szBtnName = "Btn0" .. i
  tbControls[szBtnName] = "node/imgBG/Content/Point0" .. i
  tbWnd.tbOnClick[szBtnName] = function(self)
    RemoteServer.RequestZResourceBattle("OnC2ZSelectHome", i)
  end
end
tbWnd.tbControls = tbControls

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_WND_OPENED,
      self.OnWindowOpen,
      self
    },
    {
      EventNotify.emNOTIFY_WND_CLOSED,
      self.OnWindowClose,
      self
    }
  }
end

function tbWnd:OnOpen()
  self:IniBtn()
end

function tbWnd:IniBtn()
  for i = 1, 8 do
    self.pPanel:SetActive(tbControls["Btn0" .. i] .. "/NotChosen", true)
    self.pPanel:Label_SetText(tbControls["Btn0" .. i] .. "/NotChosen/txtNum", i)
    self.pPanel:SetActive(tbControls["Btn0" .. i] .. "/Selected", false)
    self.pPanel:Label_SetText(tbControls["Btn0" .. i] .. "/Selected/txtNum", i)
    self.pPanel:SetActive(tbControls["Btn0" .. i] .. "/Main", false)
    self.pPanel:Label_SetText(tbControls["Btn0" .. i] .. "/Main/txtNum", i)
  end
end

function tbWnd:Update(tbHome2LegionId, nResTime, nMyHome)
  self:UpdateBtn(tbHome2LegionId, nMyHome)
  self:UpdateTime(nResTime)
end

function tbWnd:UpdateBtn(tbHome2LegionId, nMyHome)
  self:IniBtn()
  for nHomeIndex in pairs(tbHome2LegionId) do
    self.pPanel:SetActive(tbControls["Btn0" .. nHomeIndex] .. "/NotChosen", false)
    self.pPanel:SetActive(tbControls["Btn0" .. nHomeIndex] .. "/Selected", true)
    self.pPanel:SetActive(tbControls["Btn0" .. nHomeIndex] .. "/Main", nMyHome == nHomeIndex)
  end
end

function tbWnd:UpdateTime(nResTime)
  self.nEndTime = GetTime() + nResTime
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
  self.nTimer = Timer:Register(Env.GAME_FPS, self.OnTimer, self)
end

function tbWnd:OnTimer()
  local nLeftTime = self.nEndTime - GetTime()
  if nLeftTime > 0 then
    self.pPanel:Label_SetText(tbControls.txtTime, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154{0}\231\167\146", nLeftTime))
    return true
  end
  self.pPanel:Label_SetText(tbControls.txtTime, i18n.EasyGet("\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154{0}\231\167\146", 0))
  return false
end

function tbWnd:OnWindowOpen(szName)
  if "UIChat" ~= szName then
    return
  end
  local Animator = self.pPanel:GetAnimator("Main")
  self.pPanel:Animator_SetEnable("Main", true)
  if Animator then
    Animator:Play("UIResourceCompetition_InTo", 0, 0)
  end
end

function tbWnd:OnWindowClose(szName)
  if "UIChat" ~= szName then
    return
  end
  local Animator = self.pPanel:GetAnimator("Main")
  self.pPanel:Animator_SetEnable("Main", true)
  if Animator then
    Animator:Play("UIResourceCompetition_InTo 0", 0, 0)
  end
end

function tbWnd:OnClose()
  if self.nTimer then
    Timer:Close(self.nTimer)
    self.nTimer = nil
  end
end
