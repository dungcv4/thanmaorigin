local tbItem = Item:GetClass("QuickJumpNPC")
tbItem.PARAM_MAP_TEMPLATE_ID = 1
tbItem.PARAM_NPC_TEMPLATE_ID = 2
tbItem.PARAM_NPC_POSX = 3
tbItem.PARAM_NPC_POSY = 4
tbItem.PARAM_ACTIVITY_ID = 5

function tbItem:OnClientUse(it)
  local nMapTemplateId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_MAP_TEMPLATE_ID)
  local nNpcTemplateId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPC_TEMPLATE_ID)
  local nPosX = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPC_POSX)
  local nPosY = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_NPC_POSY)
  local nActivityId = KItem.GetItemExtParam(it.dwTemplateId, self.PARAM_ACTIVITY_ID)
  if Tong.Def.nMainMapTemplateId == nMapTemplateId and not Tong:HasTong() then
    me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\229\138\160\229\133\165\229\184\174\228\188\154"))
    return 0
  end
  if Ui:GetWindow("UIItemTips") then
    Ui:CloseWindow("UIItemTips")
  end
  if Ui:GetWindow("UIBag") then
    Ui:CloseWindow("UIBag")
  end
  if nPosX and nPosY and 0 ~= nPosX and 0 ~= nPosY then
    ActivityMgr:GotoPos(nNpcTemplateId, nMapTemplateId, nPosX, nPosY)
  elseif nActivityId and 0 ~= nActivityId then
    ActivityMgr.tbJumpFunc[ActivityMgr.tbJumpType.GoToDynamicNpc](nActivityId, nNpcTemplateId, nMapTemplateId)
  else
    ActivityMgr:GotoNpc(nNpcTemplateId, nMapTemplateId)
  end
  return 0
end

function tbItem:OpenQuickUseItem(nItemId, pPlayer)
  local tbWnd = Ui:GetWindow("UIQuickItemLaoDong")
  if tbWnd and tbWnd:Visible() then
    Log("UIQuickItemLaoDong is opened")
    return
  end
  Ui:OpenQuickUseItem(nItemId, i18n.EasyGet("\229\137\141\229\190\128\228\184\138\228\186\164"), nil, function(tbUIQuickUseItem)
    if nItemId > 0 then
      Item:ClientUseItem(nItemId)
    end
    EventNotify.OnNotify(EventNotify.emNOTIFY_QUICKJUMPNPC_CLICK)
    tbUIQuickUseItem:Close()
  end)
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "QuickJumpNPC")
end
