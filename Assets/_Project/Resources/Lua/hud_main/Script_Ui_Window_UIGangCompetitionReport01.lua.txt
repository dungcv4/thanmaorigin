local tbWnd = Ui:CreateWindow("UIGangCompetitionReport01")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_FIRSTTONG_SYNC_FINAL,
      self.UpdatePanel,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.tbButtonGroup = Ui:NewButtonGroup(self, {
    [1] = tbControls.btnTeam1,
    [2] = tbControls.btnTeam2,
    [3] = tbControls.btnTeam3,
    [4] = tbControls.btnTeam4
  }, true, function(self, nPanelIndex)
    self.nPanelIndex = nPanelIndex
    self:UpdatePanel()
  end)
end

function tbWnd:OnOpen(nType)
  self.nType = nType
  self:SetShowHide()
  self:UpdateGuess()
  self.tbButtonGroup:Select(1)
  FirstTong:RequestFinalStageMatch()
end

function tbWnd:OnClose()
end

function tbWnd:SetShowHide()
  if self.nType == self.TYPE.TYPE64_16 then
    self.pPanel:SetActive(tbControls.Champion, false)
    self.pPanel:SetActive(tbControls.Team .. 29, false)
    self.pPanel:SetActive(tbControls.Team .. 30, false)
    self.pPanel:SetActive(tbControls.imgGroup .. 15, false)
    self.pPanel:SetActive(tbControls.imgGroup .. 16, false)
    self.pPanel:SetActive(tbControls.btnTab, true)
    self.pPanel:SetActive(tbControls.Schedule, false)
  elseif self.nType == self.TYPE.TYPE16_1 then
    self.pPanel:SetActive(tbControls.Champion, true)
    self.pPanel:SetActive(tbControls.Team .. 29, true)
    self.pPanel:SetActive(tbControls.Team .. 30, true)
    self.pPanel:SetActive(tbControls.imgGroup .. 15, true)
    self.pPanel:SetActive(tbControls.imgGroup .. 16, false)
    self.pPanel:SetActive(tbControls.imgGroup .. 16, true)
    self.pPanel:SetActive(tbControls.btnTab, false)
    self.pPanel:SetActive(tbControls.Schedule, true)
  end
end

function tbWnd:UpdatePanel()
  if not FirstTong.tbFinalMatchData then
    return
  end
  if self.nType == self.TYPE.TYPE64_16 then
    self:UpdatePanelType1()
  elseif self.nType == self.TYPE.TYPE16_1 then
    self:UpdatePanelType2()
  end
end

function tbWnd:UpdatePanelType1()
  for i = 1, 28 do
    self.pPanel:Label_SetText(tbControls.Team .. i .. "/txtName", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
    self.pPanel:SetActive(tbControls.Team .. i .. "/imgWin", false)
  end
  self.pPanel:Label_SetText(tbControls.Champion .. "/txtName", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
  for nPhase, tbPhaseData in pairs(FirstTong.tbFinalMatchData) do
    for nIndex, tbData in pairs(tbPhaseData) do
      local nPanelIndex = nIndex % 4
      if 0 == nPanelIndex then
        nPanelIndex = 4
      end
      local nTempIdx = math.floor((nIndex - 1) / 4) + 1
      if 0 == nTempIdx % 2 then
        nPanelIndex = 5 - nPanelIndex
      end
      if self.tbPhase2Idx_1[nPhase] and self.tbPhase2Idx[self.tbPhase2Idx_1[nPhase]] and self.tbPhase2Idx[self.tbPhase2Idx_1[nPhase]][nTempIdx] then
        local nIdx = self.tbPhase2Idx[self.tbPhase2Idx_1[nPhase]][nTempIdx]
        if self.nPanelIndex == nPanelIndex then
          self.pPanel:Label_SetText(tbControls.Team .. nIdx .. "/txtName", tbData.szTongName or i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
          self.pPanel:SetActive(tbControls.Team .. nIdx .. "/imgWin", tbData.nResult == FirstTong.Win)
          self.pPanel:SetActive(tbControls.Team .. nIdx .. "/imgSelect", false)
        end
      end
    end
  end
  local tbReportDesc = {
    FirstTong.tbText.szReportDesc1,
    FirstTong.tbText.szReportDesc2,
    FirstTong.tbText.szReportDesc3,
    FirstTong.tbText.szReportDesc4
  }
  self.pPanel:Label_SetText(tbControls.txtTIPS, i18n.Get(tbReportDesc[self.nPanelIndex]) or "")
end

function tbWnd:UpdatePanelType2()
  for i = 1, 30 do
    self.pPanel:Label_SetText(tbControls.Team .. i .. "/txtName", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
    self.pPanel:SetActive(tbControls.Team .. i .. "/imgWin", false)
  end
  self.pPanel:Label_SetText(tbControls.Champion .. "/txtName", i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
  for nPhase, tbPhaseData in pairs(FirstTong.tbFinalMatchData) do
    if self.tbPhase2Idx_2[nPhase] and self.tbPhase2Idx[self.tbPhase2Idx_2[nPhase]] then
      for nIdx, nIndex in ipairs(self.tbPhase2Idx[self.tbPhase2Idx_2[nPhase]]) do
        if tbPhaseData[nIdx] then
          if 1 == nPhase then
            self.pPanel:Label_SetText(tbControls.Champion .. "/txtName", tbPhaseData[nIdx].szTongName or i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
          else
            self.pPanel:Label_SetText(tbControls.Team .. nIndex .. "/txtName", tbPhaseData[nIdx].szTongName or i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"))
            self.pPanel:SetActive(tbControls.Team .. nIndex .. "/imgWin", tbPhaseData[nIdx].nResult == FirstTong.Win)
            self.pPanel:SetActive(tbControls.Team .. nIndex .. "/imgSelect", false)
          end
        end
      end
    end
  end
  self.pPanel:Label_SetText(tbControls.txtTIPS, "")
  self:UpdateSchedule()
end

function tbWnd:UpdateSchedule()
  local function getShowTime(nDate)
    local tbTime = os.date("*t", Lib:GetConfigDate2Time(nDate))
    
    return string.format("%d.%d %02d:%02d", tbTime.month, tbTime.day, tbTime.hour, tbTime.min)
  end
  
  for i = 1, 7 do
    self.pPanel:Label_SetText(tbControls.txtTime .. i, getShowTime(FirstTong.FinalUiTimeCfg[i]))
  end
end

function tbWnd:UpdateGuess()
  local nStartTime = Lib:GetConfigDate2Time(FirstTong.GuessTimeCfg[1].nStartTime)
  if nStartTime > GetTime() then
    local szStartTime = i18n.EasyGet("{0}\229\188\128\229\144\175", Lib:TimeDesc13(nStartTime))
    self.pPanel:SetActive(tbControls.txtCountdownValue, true)
    self.pPanel:SetActive(tbControls.txtCountdown, true)
    self.pPanel:Label_SetText(tbControls.txtCountdownValue, szStartTime)
    self.pPanel:Button_SetEnable(tbControls.btnGuess, false)
  else
    self.pPanel:SetActive(tbControls.txtCountdownValue, false)
    self.pPanel:SetActive(tbControls.txtCountdown, false)
    self.pPanel:Button_SetEnable(tbControls.btnGuess, true)
  end
end
