Require("CommonScript/Task/CommerceTask/CommerceTask_public.lua")
CommerceTask.tbWayHandle = {
  ui = function(tbInfo)
    local szUi = tbInfo.Param1
    return function()
      Ui:OpenWindow(szUi)
    end
  end,
  lifeskill = function(tbInfo)
    local nLifeSkillId = tonumber(tbInfo.Param1)
    return function()
      local function func(tbWnd)
        if not tbWnd then
          return
        end
        tbWnd:OpenLifeSkillIndex(nLifeSkillId)
      end
      
      local szWndName = "UILifeSkill"
      if Ui:WindowVisible(szWndName) then
        return func(Ui:GetWindow(szWndName))
      end
      Ui:OpenWindowWithCallback(szWndName, function(_, tbWnd)
        func(tbWnd)
      end)
    end
  end,
  gotonpc = function(tbInfo)
    local nMapTemplateId = tonumber(tbInfo.Param1)
    local nNpcTemplateId = tonumber(tbInfo.Param2)
    return function()
      CommerceTask:GoToNpc(nNpcTemplateId, nMapTemplateId)
    end
  end,
  gotomap = function(tbInfo)
    local nMapTemplateId = tonumber(tbInfo.Param1)
    return function()
      if nMapTemplateId == me.nMapTemplateId then
        me.CenterMsg(i18n.Get(CommerceTask.Msg.AlreadyInMap))
      else
        TeamMgr:StopFollowMember()
        AFK:Stop()
        Map:SendSwitchMapRequest(nMapTemplateId)
      end
    end
  end,
  OpenMarket = function(tbInfo)
    return function(nItemTemplateId)
      MarketStallBase:OpenWindowByItemTempId(tonumber(tbInfo.Param1), nItemTemplateId)
    end
  end
}

function CommerceTask:__LoadFile_WayFile()
  self.CFG_WAY = {}
  local szFile = "Setting/Task/CommerceTaskWay.tab"
  self.CFG_WAY = Lib:LoadTabFileEx(szFile, "sdssss", "nWayId", {
    "szWayName",
    "nWayId",
    "szWayDesc",
    "szWayType",
    "Param1",
    "Param2",
    "Param3"
  })
  for _, tbInfo in ipairs(self.CFG_WAY) do
    local handle = CommerceTask.tbWayHandle[tbInfo.szWayType]
    if handle then
      tbInfo.fnOnTrack = handle(tbInfo)
    end
  end
  self.CFG_WAY = SetReadOnly(self.CFG_WAY)
end

CommerceTask:__LoadFile_WayFile()
