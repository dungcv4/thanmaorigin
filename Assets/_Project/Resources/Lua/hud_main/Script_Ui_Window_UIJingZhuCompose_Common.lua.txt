local tbWnd = Ui:CreateWindow("UIJingZhuCompose_Common")
local tbControls = {
  txtTitle = "imgBg/txtTitle",
  goItemGrid = "imgBg/panelCompose/UIItemGrid",
  txtMax = "imgBg/panelCompose/txtMax",
  btnOK = "imgBg/btnOK",
  SliderNum = "imgBg/panelCompose/SliderNum",
  txtMin = "imgBg/panelCompose/txtMin",
  btnClose = "imgBg/btnClose"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    if self.nValue > 0 and type(self.tbCallBack) == "table" then
      table.insert(self.tbCallBack, self.nTemplateId)
      table.insert(self.tbCallBack, self.nValue)
      Lib:CallBack(self.tbCallBack)
    end
    self:Close()
  end
}
tbWnd.tbOnSliderClick = {
  SliderNum = function(self, nValue)
    self.nValue = nValue
    self.tbItemGrid:SetLabelSuffixText(self.nValue * self.nTargetNum)
  end
}

function tbWnd:OnCreate()
  self.tbItemGrid = Ui:NewExistItemGrid(self, tbControls.goItemGrid)
end

function tbWnd:OnOpen(nTemplateId, nMaxNum, nTargetId, nTargetNum, szTitle, tbBtnCallBack)
  self.nTargetNum = nTargetNum
  self.nValue = 1
  self.nTemplateId = nTemplateId
  self.tbCallBack = tbBtnCallBack
  self.tbItemGrid:SetItemByTemplate(nTargetId)
  self.tbItemGrid:SetLabelSuffixText(self.nValue * self.nTargetNum)
  self.pPanel:Label_SetText(tbControls.txtTitle, szTitle)
  self.pPanel:Slider_SetRange(tbControls.SliderNum, 0, nMaxNum)
  self.pPanel:Slider_SetValue(tbControls.SliderNum, nMaxNum)
  self.pPanel:Label_SetText(tbControls.txtMin, 0)
  self.pPanel:Label_SetText(tbControls.txtMax, nMaxNum)
end
