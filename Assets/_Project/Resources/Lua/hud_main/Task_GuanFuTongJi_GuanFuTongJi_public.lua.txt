Require("CommonScript/lib.lua")
GuanFuTongJi = GuanFuTongJi or {}
GuanFuTongJi.szActivePointKey = "GuanFuTongJi"
GuanFuTongJi.CFG_TASK = {}
GuanFuTongJi.CFG_GROUP = {}
GuanFuTongJi.CFG_AWARD = {}
GuanFuTongJi.tbMapIDList = {17}
GuanFuTongJi.nTaskNpcTemplate = 7001
GuanFuTongJi.nMaxCount = 6
GuanFuTongJi.nLevelLimit = 20
GuanFuTongJi.REFRESHTIME = 14400
GuanFuTongJi.nAwardPanelHide = 3
GuanFuTongJi.nStartFindWay = 2
GuanFuTongJi.nMinJoinmember = 1
GuanFuTongJi.nAwardId = 108
GuanFuTongJi.nKinSalaryCount = 5
GuanFuTongJi.nCompleteRoundCount = 6

function GuanFuTongJi:IsFinishBeforeTask(pPlayer)
  if MODULE_GAMESERVER then
    local tbTaskData = self:GetTaskData(pPlayer)
    tbTaskData.bFinishStartTask = tbTaskData.bFinishStartTask or false
    return tbTaskData.bFinishStartTask
  else
    self.tbTaskData.bFinishStartTask = self.tbTaskData.bFinishStartTask or false
    return self.tbTaskData.bFinishStartTask
  end
end

function GuanFuTongJi:GetPlayerFinishCount(pPlayer)
  if not pPlayer then
    return
  end
  if MODULE_GAMESERVER then
    local tbTaskData = self:GetTaskData(pPlayer)
    return tbTaskData and tbTaskData.nFinishCount
  else
    self.tbTaskData.nFinishCount = self.tbTaskData.nFinishCount or false
    return self.tbTaskData.nFinishCount
  end
end

function GuanFuTongJi:GetTaskCfg(nTaskId)
  if not GuanFuTongJi.CFG_TASK[nTaskId] then
    return
  end
  return self.CFG_TASK[nTaskId]
end

function GuanFuTongJi:GetMemberIDList(pPlayer)
  local tbMemberList = {}
  if TeamMgr:HasTeam(pPlayer) then
    local tbteamData = TeamMgr:GetTeamById(pPlayer.dwTeamID)
    local tbMemberIDList = tbteamData:GetMembers()
    for _, nMemberID in pairs(tbMemberIDList) do
      table.insert(tbMemberList, nMemberID)
    end
  else
    tbMemberList = {
      pPlayer.dwID
    }
  end
  return tbMemberList
end

function GuanFuTongJi:RemoveFromList(value, tb)
  local key
  for k, v in pairs(tb) do
    if v == value then
      key = k
    end
  end
  if key then
    table.remove(tb, key)
    return true
  end
  return false
end

function GuanFuTongJi:CheckCanShow()
  if me.nLevel < self.nLevelLimit then
    return false
  end
  return true
end

function GuanFuTongJi:LoadFile()
  self:__LoadFile_Award()
  self:__LoadFile_Task()
end

function GuanFuTongJi:__LoadFile_Task()
  self.CFG_TASK = Lib:LoadTabFileEx("Setting/Task/GuanFuTongJi.tab", "ddddd", "nTaskId", {
    "nTaskId",
    "nLevel",
    "nMapTemplateId",
    "nNpcTemplateId",
    "nWeight"
  })
  for k, v in pairs(self.CFG_TASK) do
    self.CFG_GROUP[v.nLevel] = self.CFG_GROUP[v.nLevel] or {}
    table.insert(self.CFG_GROUP[v.nLevel], v.nTaskId)
  end
end

function GuanFuTongJi:__LoadFile_Award()
  local tbGuanFuTongJiAwardSetting = Lib:LoadTabFileEx("Setting/Task/GuanFuTongJi_AwardKey.tab", "ddddddd", nil, {
    "nLevel",
    "nAwardKey1",
    "nAwardKey2",
    "nAwardKey3",
    "nAwardKey4",
    "nAwardKey5",
    "nAwardKey6"
  })
  for k, v in pairs(tbGuanFuTongJiAwardSetting) do
    GuanFuTongJi.CFG_AWARD[v.nLevel] = {
      v.nAwardKey1,
      v.nAwardKey2,
      v.nAwardKey3,
      v.nAwardKey4,
      v.nAwardKey5,
      v.nAwardKey6
    }
  end
end

GuanFuTongJi:LoadFile()
