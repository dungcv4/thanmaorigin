local tbItem = Item:GetClass("ComposeItem")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local nActivityId = KItem.GetItemExtParam(nItemTemplateId, 1)
  local pInstance = ActivityMgr:GetInstanceById(nActivityId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return
  end
  
  local function fnCompose()
    Ui:OpenWindow("UICompose_Tip", nActivityId)
  end
  
  return {
    szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnFirst = fnCompose
  }
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "ComposeItem")
end
