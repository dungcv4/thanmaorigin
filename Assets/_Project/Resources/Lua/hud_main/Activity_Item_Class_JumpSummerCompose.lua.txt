Require("CommonScript/Item/ItemDefine.lua")
local tbJumpSummerCompose = Item:GetClass("JumpSummerCompose")
local COMPOSE_ACTID_PARAM = 1

function tbJumpSummerCompose:GetUseSetting(nTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  local nComposeActId = KItem.GetItemExtParam(nTemplateId, COMPOSE_ACTID_PARAM)
  local tbSetting = {}
  tbSetting.bHideShareButton = false
  tbSetting.szFirstName = i18n.EasyGet("\228\189\191\231\148\168")
  
  function tbSetting.fnFirst()
    local pInstance = ActivityMgr:GetInstanceById(nComposeActId)
    if pInstance and pInstance:IsOpen() then
      Ui:OpenWindow("UICompose_Tip", nComposeActId)
    else
      me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    end
  end
  
  return tbSetting
end

function tbJumpSummerCompose:OnTimeOut(pItem, pPlayer)
  local nItemTid = pItem.dwTemplateId
  local nCount = pItem.nCount
  local szName = pItem.szName
  local szCurrencyType, nNum = Item.tbItemSell:GetSellCurrencyValue(nItemTid)
  if not szCurrencyType then
    return
  end
  if nCount ~= pPlayer.ConsumeItem(pItem, nCount, Env.LogWay_TimeOut) then
    Log("tbJumpSummerCompose:OnTimeOut", pPlayer.szName, "pid:", pPlayer.dwID, "item:", nItemTid, "count:", nCount, "errConsumeItem")
    return
  end
  local nTotalNum = nNum * nCount
  local tbAward = {
    {szCurrencyType, nTotalNum}
  }
  pPlayer.SendAward(tbAward, true, true, Env.LogWay_TimeOut, nil, nil)
  local szMsg = i18n.EasyGet("\230\130\168\230\156\137{0}\228\184\170{1}\229\183\178\232\191\135\230\156\159\239\188\140\232\191\148\232\191\152\230\130\168{2}{3}", nCount, szName, nTotalNum, Currency:GetName(szCurrencyType))
  pPlayer.Msg(szMsg)
  Log("tbJumpSummerCompose:OnTimeOut", pPlayer.szName or "", "pid:" .. (pPlayer.dwID or 0), "item:" .. nItemTid, "count:" .. nCount)
end
