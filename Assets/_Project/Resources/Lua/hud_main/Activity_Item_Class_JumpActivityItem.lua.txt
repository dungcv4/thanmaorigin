local tbItem = Item:GetClass("JumpActivityItem")
tbItem.ACTIVITY_ID = 1
tbItem.JUMP_ID = 2

function tbItem:OnClientUse(it)
  local nId = KItem.GetItemExtParam(it.dwTemplateId, self.ACTIVITY_ID)
  local nJumpId = KItem.GetItemExtParam(it.dwTemplateId, self.JUMP_ID)
  local pInstance = ActivityMgr:GetInstanceById(nId)
  if not pInstance then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return 0
  end
  local bSuccess = pInstance:DoCommonJump(nJumpId)
  if not bSuccess then
    return 0
  end
  if Ui:GetWindow("UIItemTips") then
    Ui:CloseWindow("UIItemTips")
  end
  if Ui:GetWindow("UIBag") then
    Ui:CloseWindow("UIBag")
  end
  return 0
end

function tbItem:OnTimeOut(pItem, pPlayer)
  Item:OnTimeOutHelper(pItem, pPlayer, "JumpActivityItem")
end

function tbItem:OpenQuickUseItem(nItemId, pPlayer)
  local tbWnd = Ui:GetWindow("UIQuickItemLaoDong")
  if tbWnd and tbWnd:Visible() then
    Log("UIQuickItemLaoDong is opened")
    return
  end
  Ui:OpenQuickUseItem(nItemId, i18n.EasyGet("\229\137\141\229\190\128\228\184\138\228\186\164"), nil, function(tbUIQuickUseItem)
    if nItemId > 0 then
      Ui:CloseWindow("UITemporaryActivityCompose")
      Item:ClientUseItem(nItemId)
    end
    tbUIQuickUseItem:Close()
  end)
end
