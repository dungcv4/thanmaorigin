function tbWnd:OnCreate()
  self.AddAndSubGroup = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroup)
  self.AddAndSubGroup:BindEvent(function(nValue)
    self.nbuyCount = self:UpdateNumberInput(nValue)
    return self.nbuyCount
  end)
  self.AddAndSubGroup:SetNumPadPos(-148, 12)
  self.AddAndSubGroup:SetEnable(true)
  self.tbTree = Ui:NewTree(self, tbControls.UITree, self.onClickTreeBigBtn)
end

function tbWnd:OnOpen(nComposeActivityId, nIndex, szSubmitPrimaryRedPoint, szSubmitAdvancedRedPoint)
  szSubmitPrimaryRedPoint = szSubmitPrimaryRedPoint or "QingmingSubmitPrimary"
  szSubmitAdvancedRedPoint = szSubmitAdvancedRedPoint or "QingmingSubmitAdvanced"
  self.pInstance = ActivityMgr:GetInstanceById(nComposeActivityId)
  if not self.pInstance or not self.pInstance:IsOpen() then
    return 0
  end
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.nComposeActivityId = nComposeActivityId
  self.nComposeCount = 0
  self:UpdateTreeData()
  self.tbShopMaterialsList = tbShopMaterialsList[nComposeActivityId]
  if nIndex then
    self.tbTree:OnClickBigButton(nIndex, self.tbBigData[nIndex].tbInfo)
  end
  self.pRedPointFunc = tbRedPointFunc[nComposeActivityId]
  if self.pRedPointFunc then
    if nIndex then
      self.pRedPointFunc[nIndex](self)
    else
      self.pRedPointFunc[self.nComposeType or 1](self)
    end
  else
    local tbRedPointEventId = self.pInstance:GetRedPointEventList()
    self.pRedPointFunc = {
      [1] = function(tbWnd)
        RedPointMgr:DynamicUnRegist(tbWnd.pPanel, tbWnd.btnCompose)
        RedPointMgr:DynamicRegisteRedPoint(tbWnd.pPanel, tbWnd.btnCompose, {
          tbRedPointEventId[1]
        })
        RedPointMgr:DynamicUnRegist(tbWnd.pPanel, tbWnd.btnReceive)
        RedPointMgr:DynamicRegisteRedPoint(tbWnd.pPanel, tbWnd.btnReceive, {
          tbRedPointEventId[3]
        })
        RedPointMgr:DynamicUnRegist(tbWnd.pPanel, tbWnd.btnGo)
        RedPointMgr:DynamicRegisteRedPoint(tbWnd.pPanel, tbWnd.btnGo, {
          RedPointMgr.EVENT[szSubmitAdvancedRedPoint]
        })
      end,
      [2] = function(tbWnd)
        RedPointMgr:DynamicUnRegist(tbWnd.pPanel, tbWnd.btnCompose)
        RedPointMgr:DynamicRegisteRedPoint(tbWnd.pPanel, tbWnd.btnCompose, {
          tbRedPointEventId[2]
        })
        RedPointMgr:DynamicUnRegist(tbWnd.pPanel, tbWnd.btnGo)
        RedPointMgr:DynamicRegisteRedPoint(tbWnd.pPanel, tbWnd.btnGo, {
          RedPointMgr.EVENT[szSubmitPrimaryRedPoint]
        })
      end
    }
  end
end

function tbWnd:UpdateTreeData()
  self.tbBigData = {}
  local tbFormulaCfg = self.pInstance:GetAllFormula()
  for nComposeType, tbInfo in pairs(tbFormulaCfg) do
    table.insert(self.tbBigData, {
      nType = nComposeType,
      szText = i18n.Parse(tbInfo.szTitle),
      tbInfo = {nComposeType = nComposeType}
    })
  end
  self.tbTree:SetData(self.tbBigData, {})
end

function tbWnd:onClickTreeBigBtn(tbInfo)
  self.nComposeType = tbInfo.nComposeType
  self:Refresh()
end

function tbWnd:Refresh()
  local tbComposeCfg = self.pInstance:GetFormula(self.nComposeType)
  local tbProduct = tbComposeCfg.tbProduct
  local tbMaterials = tbComposeCfg.tbMaterials
  if self.tbShopMaterialsList then
    self.tbShopMaterials = self.tbShopMaterialsList[self.nComposeType]
  end
  self.tbExtraAward = self.pInstance:GetAllExAwardCfg()
  if self.tbExtraAward and self.tbExtraAward[self.nComposeType] then
    self.pPanel:SetActive(tbControls.btnReceive, true)
    self.pPanel:Label_SetText(tbControls.btnReceive .. "/txtBtn", i18n.EasyGet("\233\162\157\229\164\150\229\165\150\229\138\177"))
    Ui:Button_BindEvent(self.pPanel, tbControls.btnReceive, function(self)
      Ui:OpenWindow("UIReward_Newyear", self.pInstance.nId, self.nComposeType)
    end, self)
  else
    self.pPanel:SetActive(tbControls.btnReceive, false)
  end
  self.pPanel:SetActive(tbControls.Img_StateAdd, false)
  local nMaxComposeCount = self.pInstance:GetLeftFormulaMaxComposeCount(me, self.nComposeType)
  self.pPanel:Label_SetText(tbControls.txt_Num, i18n.EasyGet("\229\137\169\228\189\153\229\144\136\230\136\144\233\135\143\239\188\154{0}", nMaxComposeCount))
  local nMaxCanComposeCount = self.pInstance:GetPlayerCanMakeNum(me, self.nComposeType)
  local nMinComposeCount = 0
  if nMaxCanComposeCount > 0 then
    nMinComposeCount = 1
  end
  self.AddAndSubGroup:SetNumRange(nMinComposeCount, nMaxCanComposeCount)
  self:UpdateNumberInput(nMinComposeCount)
  self.AddAndSubGroup:SetInputValue(nMaxCanComposeCount)
  local szItemName = Item:GetName(tbProduct[2], me)
  self.pPanel:Label_SetText(tbControls.txt_Name, i18n.EasyGet("{0}", szItemName))
  self.pItemGridMain = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  self.pItemGridMain:SetItemByTemplate(tbProduct[2], tbProduct[3])
  self.pItemGridMain:SetClickEnable(true)
  local nMaterialsNum = #tbMaterials
  local szMaterialsPath
  for i = 2, 6 do
    local szContItemPath = string.format(tbControls.Cont_Item, i)
    if nMaterialsNum == i then
      self.pPanel:SetActive(szContItemPath, true)
      szMaterialsPath = szContItemPath
    else
      self.pPanel:SetActive(szContItemPath, false)
    end
  end
  local szContItemPath = string.format(tbControls.Cont_Item, nMaterialsNum)
  for index, tbItemInfo in ipairs(tbMaterials) do
    local nItemNum = me.GetItemCountByTypeInBag(tbItemInfo[2])
    local szImg_StateAddPath = string.format(szContItemPath .. "/UIItemGrid%d/Img_StateAdd", index)
    if self.tbShopMaterials then
      if self.tbShopMaterials[index] and nItemNum <= 0 then
        self.pPanel:SetActive(szImg_StateAddPath, true)
        Ui:Button_BindEvent(self.pPanel, szImg_StateAddPath, self.GetMaterials, self, tbItemInfo[2])
      else
        self.pPanel:SetActive(szImg_StateAddPath, false)
      end
    elseif nItemNum <= 0 and index == nMaterialsNum then
      local tbWareList = Shop:GetAvaliableRemainWareList(Shop.ShopType2ViewIndex.LimitBuy, tbItemInfo[2])
      if tbWareList[1] then
        self.pPanel:SetActive(szImg_StateAddPath, true)
        Ui:Button_BindEvent(self.pPanel, szImg_StateAddPath, self.GetMaterials, self, tbItemInfo[2], tbWareList[1].nWareId)
      else
        self.pPanel:SetActive(szImg_StateAddPath, false)
        Ui:Button_UnBindEvent(self.pPanel, szImg_StateAddPath)
      end
    else
      self.pPanel:SetActive(szImg_StateAddPath, false)
    end
  end
  if not szMaterialsPath then
    return
  end
  self.tbpItemGridList = {}
  for nIndex, tbValue in ipairs(tbMaterials) do
    local pItemGrid = Ui:NewExistItemGrid(self, szMaterialsPath .. "/UIItemGrid" .. nIndex)
    local nHaveNum = 0
    local nNeedNum = 0
    if Currency.TYPE[tbValue[1]] == nil then
      pItemGrid:SetItemByTemplate(tbValue[2], tbValue[3])
      nHaveNum = me.GetItemCountByTypeInBag(tbValue[2])
      nNeedNum = tbValue[3]
    else
      pItemGrid:SetDigitalItem(tbValue[1], tbValue[2])
      nHaveNum = me.GetMoney(Currency.TYPE[tbValue[1]])
      nNeedNum = tbValue[2]
    end
    local szShowText = string.format("%d", nHaveNum)
    pItemGrid:LabelSuffix_SetText(szShowText)
    if nHaveNum < nNeedNum then
      pItemGrid:LabelSuffix_SetColor("Red")
    else
      pItemGrid:LabelSuffix_SetColor("ItemGirdGreen")
    end
    self.tbpItemGridList[nIndex] = pItemGrid
    pItemGrid:SetClickEnable(true)
  end
  local tbFormulaCfg = self.pInstance:GetAllFormula()
  if tbFormulaCfg[self.nComposeType] then
    if not Lib:IsEmptyStr(tbFormulaCfg[self.nComposeType].szCallBack1) then
      self.pPanel:SetActive(tbControls.btnGo, true)
      Ui:Button_BindEvent(self.pPanel, tbControls.btnGo, self.OnClickGo, self, tbFormulaCfg[self.nComposeType].szCallBack1, tbFormulaCfg[self.nComposeType].szParam1)
    else
      self.pPanel:SetActive(tbControls.btnGo, false)
    end
    if not Lib:IsEmptyStr(tbFormulaCfg[self.nComposeType].szCallBack2) then
      self.pPanel:SetActive(tbControls.btnGet, true)
      Ui:Button_BindEvent(self.pPanel, tbControls.btnGet, self.OnClickGo, self, tbFormulaCfg[self.nComposeType].szCallBack2, tbFormulaCfg[self.nComposeType].szParam2)
    else
      self.pPanel:SetActive(tbControls.btnGet, false)
    end
  end
  if self.pRedPointFunc then
    self.pRedPointFunc[self.nComposeType](self)
  end
end

function tbWnd:GetMaterials(nItemTemplateId, nWareId)
  Shop:OpenShopBuyWidDefault(nItemTemplateId, nil, nWareId)
end

function tbWnd:OnClickGo(szCallBack, szParam)
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam, "|")
  fn(self.nComposeActivityId, table.unpack(tbParam))
  Ui:CloseWindow("UICompose_Tip")
end

function tbWnd:OnChangeMoney(szType)
  self:Refresh()
end

function tbWnd:OnAddItem(nItemId)
  if nItemId then
    local pItem = KItem.GetItemObj(nItemId)
    if not pItem then
      return false
    end
    local tbComposeCfg = self.pInstance:GetFormula(self.nComposeType)
    local tbMaterials = tbComposeCfg.tbMaterials or {}
    if Lib:CountTB(tbMaterials) <= 0 then
      return
    end
    local bIsRefresh = false
    for nIndex, tbValue in ipairs(tbMaterials) do
      if tbValue[2] == pItem.dwTemplateId then
        bIsRefresh = true
        break
      end
    end
    if bIsRefresh then
      self:Refresh()
    end
  end
end

function tbWnd:OnDelItem(_, dwTemplateId)
  local tbComposeCfg = self.pInstance:GetFormula(self.nComposeType)
  local tbMaterials = tbComposeCfg.tbMaterials or {}
  if Lib:CountTB(tbMaterials) <= 0 then
    return
  end
  local bIsRefresh = false
  for nIndex, tbValue in ipairs(tbMaterials) do
    if tbValue[2] == dwTemplateId then
      bIsRefresh = true
      break
    end
  end
  if bIsRefresh then
    self:Refresh()
  end
end

function tbWnd:UpdateNumberInput(nNum)
  local nMaxComposeCount = self.pInstance:GetPlayerCanMakeNum(me, self.nComposeType)
  if nNum > nMaxComposeCount then
    nNum = nMaxComposeCount
    me.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\229\134\141\229\162\158\229\138\160\228\186\134"))
    return nNum
  end
  if nNum <= 0 and nMaxComposeCount > 0 then
    nNum = 1
    me.CenterMsg(i18n.EasyGet("\228\184\141\232\131\189\229\134\141\229\135\143\229\176\145\228\186\134"))
    return nNum
  end
  return nNum
end

function tbWnd:Make()
  if self.pInstance:GetLeftFormulaMaxComposeCount(me, self.nComposeType) <= 0 then
    me.CenterMsg(i18n.EasyGet("\229\183\178\232\190\190\229\136\176\229\144\136\230\136\144\233\135\143\228\184\138\233\153\144\239\188\140\230\151\160\230\179\149\229\136\182\228\189\156"))
    return
  elseif self.AddAndSubGroup.nValue <= 0 then
    me.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\228\184\141\232\182\179\239\188\140\230\151\160\230\179\149\229\136\182\228\189\156"))
    return
  end
  self.pInstance:MakeByFormula(self.nComposeType, self.AddAndSubGroup.nValue)
end

function tbWnd:OnClose()
  self.tbShopMaterialsList = nil
end
