local tbWnd = Ui:CreateWindow("UIPKCompetitionGuess_Tip1")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnOK = "node/imgBG/Content/btnOK",
  txtTeamName = "node/imgBG/Content/txtTeamName",
  txtFighting = "node/imgBG/Content/txtFighting",
  txtInfo = "node/imgBG/Content/infoCell/txtInfo",
  imgIconVotable1 = "node/imgBG/Content/infoCell/ votableNumber01/imgIcon1 (1)",
  txtNumVotable1 = "node/imgBG/Content/infoCell/ votableNumber01/txtNum",
  imgIconVotable2 = "node/imgBG/Content/infoCell/ votableNumber01 (1)/imgIcon1 (1)",
  txtNumVotable2 = "node/imgBG/Content/infoCell/ votableNumber01 (1)/txtNum",
  imgIcon01 = "node/imgBG/Content/resources01/imgIcon01",
  txtNowNum01 = "node/imgBG/Content/resources01/txtNowNum01",
  btnAdd01 = "node/imgBG/Content/resources01/btnAdd01",
  imgIcon02 = "node/imgBG/Content/resources02/imgIcon02",
  txtNowNum02 = "node/imgBG/Content/resources02/txtNowNum02",
  btnAdd02 = "node/imgBG/Content/resources02/btnAdd02",
  UIAddAndSubGroup1 = "node/imgBG/Content/resources01/UIAddAndSubGroup01",
  UIAddAndSubGroup2 = "node/imgBG/Content/resources02/UIAddAndSubGroup02"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnAdd01 = function(self)
    Shop:OpenShopBuyWidDefault(TongBattle.GuessCfg.nLaBaItemID)
  end,
  btnAdd02 = function(self)
    Shop:OpenShopBuyWidDefault(TongBattle.GuessCfg.nLuoGuItemID)
  end,
  btnOK = function(self)
    self:ClickButtonOK()
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.AddAndSubGroup1 = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroup1)
  self.AddAndSubGroup1:BindEvent(function(nValue)
    self:SetNum(TongBattle.GuessCfg.nLaBaItemID, nValue)
  end)
  self.AddAndSubGroup2 = Ui:NewExistAddAndSubGroup(self, tbControls.UIAddAndSubGroup2)
  self.AddAndSubGroup2:BindEvent(function(nValue)
    self:SetNum(TongBattle.GuessCfg.nLuoGuItemID, nValue)
  end)
end

function tbWnd:OnOpen(nStage, nGuessMatch, nGuessTongID)
  self.tbGuessLimitCfg = nil
  for nStageIndex, tbCfg in pairs(TongBattle.GuessCfg.Limit) do
    if nStageIndex == nStage then
      self.tbGuessLimitCfg = tbCfg
      break
    end
  end
  if not self.tbGuessLimitCfg then
    me.CenterMsg(i18n.EasyGet("\231\171\158\231\140\156\229\156\186\230\172\161\228\184\141\229\173\152\229\156\168"))
    self:Close()
    return
  end
  self.nStage = nStage
  self.nGuessMatch = nGuessMatch
  self.nGuessTongID = nGuessTongID
  self.tbNums = {
    [TongBattle.GuessCfg.nLaBaItemID] = 0,
    [TongBattle.GuessCfg.nLuoGuItemID] = 0
  }
  local tbTongData = TongBattle.tbTongData[nGuessTongID]
  if tbTongData then
    self.pPanel:Label_SetText(tbControls.txtTeamName, tbTongData.szTongName)
    self.pPanel:Label_SetText(tbControls.txtFighting, tbTongData.nFightPower)
  end
  self.AddAndSubGroup1:SetInputValue(0)
  self.AddAndSubGroup2:SetInputValue(0)
  self:UpdateIcon()
  self:UpdateItemLimit()
  self:UpdateItemCount()
end

function tbWnd:Update()
end

function tbWnd:ClickButtonOK()
  TongBattle:RequestGuessMatch(self.nGuessMatch, self.nGuessTongID, self.tbNums)
end

function tbWnd:OnSyncItem(nItemId, bNewItem, nAddCount)
  if nItemId then
    local pItem = KItem.GetItemObj(nItemId)
    if not pItem then
      return false
    end
    if pItem.dwTemplateId == TongBattle.GuessCfg.nLaBaItemID or pItem.dwTemplateId == TongBattle.GuessCfg.nLuoGuItemID then
      self:UpdateItemCount()
    end
  end
end

function tbWnd:UpdateItemCount()
  local nNum1 = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLaBaItemID)
  self.pPanel:Label_SetText(tbControls.txtNowNum01, nNum1)
  local nNum2 = me.GetItemCountByTypeInBag(TongBattle.GuessCfg.nLuoGuItemID)
  self.pPanel:Label_SetText(tbControls.txtNowNum02, nNum2)
  local tbGuessData = TongBattle.tbMyMatchGuessData and TongBattle.tbMyMatchGuessData[self.nGuessMatch]
  local tbGuessedNum = {}
  if tbGuessData then
    tbGuessedNum[TongBattle.GuessCfg.nLaBaItemID] = tbGuessData.tbMyMatchGuessData[TongBattle.GuessCfg.nLaBaItemID].nGuessNum
    tbGuessedNum[TongBattle.GuessCfg.nLuoGuItemID] = tbGuessData.tbMyMatchGuessData[TongBattle.GuessCfg.nLuoGuItemID].nGuessNum
  end
  local nLimit1 = self.tbGuessLimitCfg.nLaBaMaxNum - (tbGuessedNum[TongBattle.GuessCfg.nLaBaItemID] or 0)
  local nMaxNum1 = math.min(nLimit1, nNum1)
  self.AddAndSubGroup1:SetNumRange(0, nMaxNum1)
  local nLimit2 = self.tbGuessLimitCfg.nLuoGuMaxNum - (tbGuessedNum[TongBattle.GuessCfg.nLuoGuItemID] or 0)
  local nMaxNum2 = math.min(nLimit2, nNum2)
  self.AddAndSubGroup2:SetNumRange(0, nMaxNum2)
end

function tbWnd:UpdateItemLimit()
  self.pPanel:Label_SetText(tbControls.txtInfo, i18n.EasyGet("\230\173\164\233\152\182\230\174\181\230\175\143\229\156\186\229\143\175\230\138\149\231\165\168"))
  self.pPanel:Label_SetText(tbControls.txtNumVotable1, self.tbGuessLimitCfg.nLaBaMaxNum)
  self.pPanel:Label_SetText(tbControls.txtNumVotable2, self.tbGuessLimitCfg.nLuoGuMaxNum)
end

function tbWnd:UpdateIcon()
  local nIcon, szIconSprite
  _, nIcon, _, _ = KItem.GetItemShowInfo(TongBattle.GuessCfg.nLaBaItemID)
  _, szIconSprite = Item:GetIcon(nIcon)
  self.pPanel:Sprite_SetSprite(tbControls.imgIconVotable1, szIconSprite)
  self.pPanel:Sprite_SetSprite(tbControls.imgIcon01, szIconSprite)
  _, nIcon, _, _ = KItem.GetItemShowInfo(TongBattle.GuessCfg.nLuoGuItemID)
  _, szIconSprite = Item:GetIcon(nIcon)
  self.pPanel:Sprite_SetSprite(tbControls.imgIconVotable2, szIconSprite)
  self.pPanel:Sprite_SetSprite(tbControls.imgIcon02, szIconSprite)
end

function tbWnd:SetNum(nItemId, nNum)
  if self.tbNums[nItemId] == nNum then
    return
  end
  self.tbNums[nItemId] = nNum
end
