local tbWnd = Ui:CreateWindow("UIJingZhuCompose")
tbWnd.nMaxInputCount = 99
tbWnd.nMinInputCount = 1
tbWnd.tbControls = {
  goItemGrid = "imgBg/panelCompose/UIItemGrid",
  txtMax = "imgBg/panelCompose/txtMax",
  btnOK = "imgBg/btnOK",
  SliderNum = "imgBg/panelCompose/SliderNum",
  txtMin = "imgBg/panelCompose/txtMin",
  btnClose = "imgBg/btnClose"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    if self.nValue > 0 then
      RemoteServer.RequestEquipJingZhu("JingZhuSynthesisReq", self.nTemplateId, self.nValue)
    end
    self:Close()
  end
}
tbWnd.tbOnSliderClick = {
  SliderNum = function(self, nValue)
    self.nValue = nValue
    self.tbItemGrid:SetLabelSuffixText(self.nValue * self.nTargetNum)
  end
}

function tbWnd:NotifyEvents()
  return {}
end

function tbWnd:OnCreate()
  self.tbItemGrid = Ui:NewExistItemGrid(self, self.goItemGrid)
end

function tbWnd:OnOpen(nTemplateId, nMaxNum, nTargetId, nTargetNum)
  self.nTargetNum = nTargetNum
  self.nValue = 1
  self.nTemplateId = nTemplateId
  self.tbItemGrid:SetItemByTemplate(nTargetId)
  self.tbItemGrid:SetLabelSuffixText(self.nValue * self.nTargetNum)
  self.pPanel:Slider_SetRange(self.SliderNum, 0, nMaxNum)
  self.pPanel:Slider_SetValue(self.SliderNum, nMaxNum)
  self.pPanel:Label_SetText(self.txtMin, 0)
  self.pPanel:Label_SetText(self.txtMax, nMaxNum)
end
