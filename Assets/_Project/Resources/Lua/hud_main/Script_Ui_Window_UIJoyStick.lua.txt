local tbWnd = Ui:CreateWindow("UIJoyStick")
local tbControls = {
  imgHand = "panel/Joystick/imgHand",
  KeyboardMark = "panel/Joystick/KeyboardMark",
  A = "panel/Joystick/KeyboardMark/A",
  W = "panel/Joystick/KeyboardMark/W",
  D = "panel/Joystick/KeyboardMark/D",
  S = "panel/Joystick/KeyboardMark/S",
  panel = "panel"
}
tbWnd.tbControls = tbControls

function tbWnd:OnOpen()
  self:SetHandActive(false)
  self:UpdateShowTips()
  self:RefreshHotkeyDisplay()
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_UISetting_ACTIVE_KEYBOARD_TIPS,
      self.UpdateShowTips,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.uiJoyStick = self.pPanel:GetComponent(Ui.UIJoyStick)
end

function tbWnd:SetHandActive(bActive)
  self.pPanel:SetActive(tbControls.imgHand, bActive)
end

function tbWnd:OnClose()
end

function tbWnd:UpdateShowTips()
  if not self.uiJoyStick then
    self.uiJoyStick = self.pPanel:GetComponent(Ui.UIJoyStick)
  end
  local nActive = Ui:GetKeyBoardTipActiveSetting()
  if 0 == nActive then
    self.pPanel:SetActive(tbControls.KeyboardMark, false)
    if Sdk:GetOsType() == Sdk.eOSType_Windows and self.uiJoyStick then
      self.pPanel:SetActive(tbControls.panel, false)
      self.uiJoyStick.bEnable = false
    end
  else
    self.pPanel:SetActive(tbControls.KeyboardMark, true)
    if Sdk:GetOsType() == Sdk.eOSType_Windows and self.uiJoyStick then
      self.pPanel:SetActive(tbControls.panel, true)
      self.uiJoyStick.bEnable = true
    end
  end
end

function tbWnd:RefreshHotkeyDisplay()
  if not CustomHotkey:CheckPlatform() then
    return
  end
  local moveForwardKeycode = CustomHotkey:GetKeycodeGameDisplayName("HOTKEY_MOVE_FORWARD")
  self.pPanel:Label_SetText(tbControls.W, moveForwardKeycode)
  local moveBackKeycode = CustomHotkey:GetKeycodeGameDisplayName("HOTKEY_MOVE_BACK")
  self.pPanel:Label_SetText(tbControls.S, moveBackKeycode)
  local moveLeftKeycode = CustomHotkey:GetKeycodeGameDisplayName("HOTKEY_MOVE_LEFT")
  self.pPanel:Label_SetText(tbControls.A, moveLeftKeycode)
  local moveLeftKeycode = CustomHotkey:GetKeycodeGameDisplayName("HOTKEY_MOVE_RIGHT")
  self.pPanel:Label_SetText(tbControls.D, moveLeftKeycode)
end
