  return {
    {

local tbPanelIndex = {
  nNone = 0,
  nAllRank = 1,
  nRankAward = 2,
  nCustomRank = 3
}
local nRankMaxNum = 6

function tbWnd:OnCreate()
  self.uiTree = Ui:NewTree(self, tbControls.TreeRoot, self.OnClickBigButton, self.OnClickSamllButton)
  self.tbAllRankScrollView = Ui:NewScrollView(self, tbControls.RankTemplatePlayer)
  self.tbRankAwardScrollView = Ui:NewScrollView(self, tbControls.AwardTemplatePlayer)
  self.tbCustomRankScrollView = Ui:NewScrollView(self, tbControls.CustomRankTemplatePlayer)
end

function tbWnd:OnOpen(tbEntranceRankData, nButtonIndex, tbPermanantRankData)
  self.tbPermanantRankData = tbPermanantRankData
  self.tbUITreeBigDataList, self.tbUITreeSmallDataList = ActivityRank:GetTreeData(tbEntranceRankData.tbRankLayerIdList)
  self.curType = nil
  self.nPanelIndex = nil
  self.uiTree:SetData(self.tbUITreeBigDataList, self.tbUITreeSmallDataList)
  if nil ~= tbEntranceRankData.szTitle and tbEntranceRankData.szTitle ~= "" then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.Get(tbEntranceRankData.szTitle))
  end
  self.uiTree:SelectNode(nButtonIndex, 1, true)
end

function tbWnd:CheckActivityValid(nActivityIndex)
  if not Lib:IsInteger(nActivityIndex) then
    return false
  end
  local pActInstance = ActivityMgr:GetInstanceById(nActivityIndex)
  if not pActInstance or not pActInstance:IsOpen() then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\230\156\170\229\188\128\229\144\175"))
    return false
  end
  return true
end

function tbWnd:OnClickBigButton(tbInfo)
  if self.curType ~= tbInfo.nType then
    self.uiTree:AutoSelectSmall(tbInfo.nType)
    self.curType = tbInfo.nType
  end
  local nActivityIndex = tbInfo.nActivityIndex
  local nOnClickIndex = tbInfo.nOnClickIndex
  local bNotSameButton = self.nPanelIndex ~= nOnClickIndex or self.curType ~= tbInfo.nType
  if nil ~= nOnClickIndex and bNotSameButton and 0 ~= nOnClickIndex then
    self:UpdatePanel(tbInfo, nActivityIndex)
  end
end

function tbWnd:OnClickSamllButton(tbInfo)
  local nActivityIndex = tbInfo.nActivityIndex
  local nOnClickIndex = tbInfo.nOnClickIndex
  local bNotSameButton = self.nPanelIndex ~= nOnClickIndex or self.curType ~= tbInfo.nType or self.nSamllButtonId ~= tbInfo.nSecendLayerId
  if nil ~= nOnClickIndex and bNotSameButton and 0 ~= nOnClickIndex then
    self.nSamllButtonId = tbInfo.nSecendLayerId
    self:UpdatePanel(tbInfo, nActivityIndex)
  end
end

function tbWnd:UpdatePanel(tbInfo, nActivityIndex)
  local nPanelIndex = tbInfo.nOnClickIndex
  self.nPanelIndex = nPanelIndex
  local bShowRankPanel = nPanelIndex == tbPanelIndex.nAllRank
  local bShowAwardPanel = nPanelIndex == tbPanelIndex.nRankAward
  local bShowCustomRankPanel = nPanelIndex == tbPanelIndex.nCustomRank
  self.pPanel:SetActive(tbControls.Ranking, bShowRankPanel)
  self.pPanel:SetActive(tbControls.Awards, bShowAwardPanel)
  self.pPanel:SetActive(tbControls.RankingChristmas, bShowCustomRankPanel)
  self:InitScrollView()
  if not self.tbPermanantRankData and not self:CheckActivityValid(nActivityIndex) then
    return
  end
  if self.tbPermanantRankData then
    self.szRankType = self.tbPermanantRankData.szRankType
  else
    self.pActInstance = ActivityMgr:GetInstanceById(nActivityIndex)
    self.szRankType = self.pActInstance.tbConfig.szRankType
  end
  if nPanelIndex == tbPanelIndex.nAllRank then
    self:ShowAllRankPanel()
  elseif nPanelIndex == tbPanelIndex.nRankAward then
    self:ShowRankAwardPanel()
  elseif nPanelIndex == tbPanelIndex.nCustomRank then
    self:ShowCustomRankPanel(table.unpack(Lib:SplitStrTry2Num(tbInfo.szExtraParam, "|")))
  end
end

function tbWnd:InitScrollView()
  self.pPanel:SetActive(tbControls.Empty, false)
  self.tbRankAwardScrollView:Clear()
  self.tbAllRankScrollView:Clear()
  self.tbCustomRankScrollView:Clear()
end

function tbWnd:ShowRankAwardPanel()
  self.pPanel:SetActive(tbControls.AwardsListPlayer, true)
  self.pPanel:SetActive(tbControls.AwardsListKin, false)
  local tbRankingAward
  if self.tbPermanantRankData then
    tbRankingAward = self.tbPermanantRankData.tbRankingAward
  else
    if not self.pActInstance then
      return
    end
    tbRankingAward = self.pActInstance.tbConfig.tbRankingAward
  end
  for nIndex, tbAwardCfg in ipairs(tbRankingAward) do
    self.tbRankAwardScrollView:SetElement(nIndex, function(tbElement)
      if tbAwardCfg.nRankStart == tbAwardCfg.nRankEnd then
        tbElement.pPanel:Label_SetText("txtRank", i18n.EasyGet("\231\172\172{0}\229\144\141", tbAwardCfg.nRankEnd))
      else
        tbElement.pPanel:Label_SetText("txtRank", i18n.EasyGet("\231\172\172{0}-{1}\229\144\141", tbAwardCfg.nRankStart, tbAwardCfg.nRankEnd))
      end
      if tbElement.tbLevelAwardShowScrollView == nil then
        tbElement.tbLevelAwardShowScrollView = Ui:NewItemGridScrollView(tbElement, "Awards/ItemTemplate", 1, 1)
      end
      if self.tbPermanantRankData then
        local tbAward = Award:GetAward(self.tbPermanantRankData.nAwardId, tbAwardCfg.nShowAwardKey)
        tbElement.tbLevelAwardShowScrollView:UpdateGenericItem(tbAward, true)
      else
        local tbAward = Award:GetAward(self.pActInstance.tbConfig.nAwardId, tbAwardCfg.nShowAwardKey)
        tbElement.tbLevelAwardShowScrollView:UpdateGenericItem(tbAward, true)
      end
    end)
  end
end

function tbWnd:ShowAllRankPanel()
  if self.tbPermanantRankData then
    self.pPanel:Label_SetText(tbControls.txtKinName, Lib:IIF(Lib:IsEmptyStr(self.tbPermanantRankData.szSubValueTitle), i18n.EasyGet("\229\174\182\230\151\143"), i18n.Parse(self.tbPermanantRankData.szSubValueTitle)))
    self.pPanel:Label_SetText(tbControls.txtScoreTitle, i18n.Parse(self.tbPermanantRankData.szScoreTitle))
  else
    self.pPanel:Label_SetText(tbControls.txtKinName, Lib:IIF(Lib:IsEmptyStr(self.pActInstance.tbConfig.szSubValueTitle), i18n.EasyGet("\229\174\182\230\151\143"), i18n.Parse(self.pActInstance.tbConfig.szSubValueTitle)))
    self.pPanel:Label_SetText(tbControls.txtScoreTitle, i18n.Parse(self.pActInstance.tbConfig.szScoreTitle))
  end
  self:AllRankSwitchPage(1)
end

function tbWnd:AllRankSwitchPage(nPageIndex)
  self.nPageIndex = nPageIndex
  self.tbAllRankScrollView:Clear()
  self:SetAllBottomRankAndScoreText()
  self:SetAllNowPageAndSumPageText()
  self:RequestAllRankDataByPage(nPageIndex)
end

function tbWnd:SetAllNowPageAndSumPageText()
  self:SetNowPageAndSumPage(self.nPageIndex, self.nAllSumPage or 1)
end

function tbWnd:SetAllBottomRankAndScoreText()
  if self.nMyPersonalRank ~= nil and nil ~= self.nMyCardNum then
    if self.tbPermanantRankData then
      self:SetMyRankAndScoreShow(i18n.EasyGet("\230\136\145\231\154\132\230\142\146\229\144\141\239\188\154"), self.nMyPersonalRank, i18n.Parse(self.tbPermanantRankData.szScoreTitle), self.nMyCardNum)
    else
      self:SetMyRankAndScoreShow(i18n.EasyGet("\230\136\145\231\154\132\230\142\146\229\144\141\239\188\154"), self.nMyPersonalRank, i18n.Parse(self.pActInstance.tbConfig.szScoreName), self.nMyCardNum)
    end
  else
    self:SetNotOnRank()
  end
end

function tbWnd:RequestAllRankDataByPage(nPageIndex)
  RankBoard:RequestOpenRankBoard(self.szRankType, nPageIndex, true)
end

function tbWnd:OnSyncAllRank(szKey, nPage)
  if szKey ~= self.szRankType then
    return
  end
  local tbData = RankBoard.tbAllData[self.szRankType][self.nPageIndex]
  local tbMyData = RankBoard.tbMyRankInfo[self.szRankType]
  self.nMyPersonalRank = tbMyData.nPosition
  self.nMyValue = tbMyData.nValue or 0
  self.nMyCardNum = math.floor(self.nMyValue / (RankBoard.ExtraCfg[self.szRankType] or 1))
  if self.tbPermanantRankData and self.tbPermanantRankData.fnProcessHighValue then
    self.nMyCardNum = self.tbPermanantRankData.fnProcessHighValue(self.nMyCardNum)
  end
  self.nAllSumPage = math.ceil(tbMyData.nLength / RankBoard.PAGE_NUM)
  if self.nAllSumPage < 1 then
    self.nAllSumPage = 1
  end
  if self.nPanelIndex == tbPanelIndex.nAllRank then
    self:SetAllBottomRankAndScoreText()
    self:SetAllNowPageAndSumPageText()
    local bRankDataNotEmpty = #tbData > 0
    self.pPanel:SetActive(tbControls.RankContent, bRankDataNotEmpty)
    self.pPanel:SetActive(tbControls.RankNotInfo, not bRankDataNotEmpty)
    if bRankDataNotEmpty then
      local nPlayerTotalNum = 0
      for nIndex, tbPlayerData in ipairs(tbData) do
        self.tbAllRankScrollView:SetElement(nIndex, function(tbElement)
          local pPanel = tbElement.pPanel
          local nRank = (self.nPageIndex - 1) * RankBoard.PAGE_NUM + nIndex
          self:RankElement_SetRank(tbElement, "UIRankNo", "txtRank", nRank)
          pPanel:Label_SetText("txtPlayerName", tbPlayerData.szName)
          Ui:SetVipLevelEffect(pPanel, tbPlayerData.nVipLevel, "txtPlayerName")
          self:RankElement_SetHonor(tbElement, "txtPlayerName/HonorTitle", tbPlayerData.nHonorStarLevel)
          pPanel:Label_SetText("txtKinName", Lib:IIF(tbPlayerData.bReplaceKinName, tbPlayerData.nLowValue, tbPlayerData.szKinName))
          nPlayerTotalNum = math.floor(tbPlayerData.nHighValue / (RankBoard.ExtraCfg[self.szRankType] or 1))
          if self.tbPermanantRankData and self.tbPermanantRankData.fnProcessHighValue then
            nPlayerTotalNum = self.tbPermanantRankData.fnProcessHighValue(nPlayerTotalNum)
          end
          pPanel:Label_SetText("txtScore", nPlayerTotalNum)
          self:RankElement_SetBackGroundColor(tbElement, me.dwID == tbPlayerData.nPlayerId)
        end)
      end
    end
  end
end

function tbWnd:SetMyRankAndScoreShow(szRankName, nRank, szScoreName, nScore)
  self:SetBottomActive(true)
  self.pPanel:Label_SetText(tbControls.txtRank, szRankName)
  self.pPanel:Label_SetText(tbControls.txtRankValue, nRank)
  self.pPanel:Label_SetText(tbControls.txtScore, szScoreName)
  self.pPanel:Label_SetText(tbControls.txtScoreValue, nScore)
end

function tbWnd:SetNotOnRank()
  self:SetBottomActive(false)
end

function tbWnd:SetBottomActive(bIsOnRank)
  self.pPanel:SetActive(tbControls.txtRank, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtRankValue, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtScore, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtScoreValue, bIsOnRank)
  self.pPanel:SetActive(tbControls.txtMyInfoLabelNoData, not bIsOnRank)
end

function tbWnd:SetNowPageAndSumPage(nNowPage, nSumPage)
  self.pPanel:Label_SetText(tbControls.txtPage, string.format("%d/%d", nNowPage, nSumPage))
end

function tbWnd:RankElement_SetRank(tbElement, szTop3RankNoKey, szTxtRankKey, nRank)
  local pPanel = tbElement.pPanel
  local bTop3 = nRank <= 3
  pPanel:SetActive(szTxtRankKey, not bTop3)
  pPanel:SetActive(szTop3RankNoKey, bTop3)
  if bTop3 then
    pPanel:SetActive(szTop3RankNoKey .. "/imgNO1", 1 == nRank)
    pPanel:SetActive(szTop3RankNoKey .. "/imgNO2", 2 == nRank)
    pPanel:SetActive(szTop3RankNoKey .. "/imgNO3", 3 == nRank)
  else
    pPanel:Label_SetText(szTxtRankKey, nRank)
  end
end

function tbWnd:RankElement_SetHonor(tbElement, szHonorKey, nHonorStarLevel)
  local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(nHonorStarLevel, false)
  local obj = tbElement.pPanel:GetObject(szHonorKey)
  local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
  pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
end

function tbWnd:RankElement_SetBackGroundColor(tbElement, bIsSelf)
  if bIsSelf then
    tbElement.pPanel:Sprite_SetColor("Main", 255, 255, 255, 255)
  else
    tbElement.pPanel:Sprite_SetColor("Main", 0, 0, 0, 20)
  end
end

function tbWnd:GetNowShowRankSumPage()
  if self.nPanelIndex == tbPanelIndex.nAllRank then
    return self.nAllSumPage
  else
    return nil
  end
end

function tbWnd:OnClickBtnPrePage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 and nil ~= self.nPageIndex and self.nPageIndex - 1 >= 1 then
    self:OnClickSwitchRankPage(self.nPageIndex - 1)
  end
end

function tbWnd:OnClickBtnNextPage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 and nil ~= self.nPageIndex and nSumPage >= self.nPageIndex + 1 then
    self:OnClickSwitchRankPage(self.nPageIndex + 1)
  end
end

function tbWnd:OnClickBtnFirstPage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 then
    self:OnClickSwitchRankPage(1)
  end
end

function tbWnd:OnClickBtnLastPage()
  local nSumPage = self:GetNowShowRankSumPage()
  if nil ~= nSumPage and nSumPage > 0 then
    self:OnClickSwitchRankPage(nSumPage)
  end
end

function tbWnd:OnClickSwitchRankPage(nPageIndex)
  if self.nPageIndex ~= nPageIndex and self.nPanelIndex == tbPanelIndex.nAllRank then
    self:AllRankSwitchPage(nPageIndex)
  end
end

function tbWnd:OnEnterMap()
  self:Close()
end

function tbWnd:ShowCustomRankPanel(...)
  local tbCustomRankList = self.pActInstance:GetRankList(...)
  local bIsEmpty = nil == tbCustomRankList or 0 == #tbCustomRankList
  self.pPanel:SetActive(tbControls.Empty, bIsEmpty)
  if bIsEmpty then
    return
  end
  for nIndex, tbPlayerData in ipairs(tbCustomRankList) do
    self.tbCustomRankScrollView:SetElement(nIndex, function(tbElement)
      self:SetCustomRankElementInfo(tbElement, tbPlayerData)
    end)
  end
end

function tbWnd:SetCustomRankElementInfo(tbElement, tbPlayerData)
  local pPanel = tbElement.pPanel
  local nRank = tbPlayerData.nRank
  self:RankElement_SetRank(tbElement, "UIRankNo", "txtRank", nRank)
  pPanel:Label_SetText("txtPlayerName", tbPlayerData.szName)
  Ui:SetVipLevelEffect(pPanel, tbPlayerData.nVipLevel, "txtPlayerName")
  self:RankElement_SetHonor(tbElement, "txtPlayerName/HonorTitle", tbPlayerData.nHonorStarLevel)
  pPanel:Label_SetText("txtKinName", tbPlayerData.szKinName)
  self:RankElement_SetBackGroundColor(tbElement, me.dwID == tbPlayerData.nPlayerId)
end
