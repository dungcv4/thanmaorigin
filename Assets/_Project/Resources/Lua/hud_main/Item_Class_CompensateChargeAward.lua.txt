local tbItem = Item:GetClass("CompensateChargeAward")

function tbItem:OnUse(it)
  local nProductId = KItem.GetItemExtParam(it.dwTemplateId, 1)
  local nExtParam2 = KItem.GetItemExtParam(it.dwTemplateId, 2)
  local bWeb = nExtParam2 and nExtParam2 > 0 or false
  local tbProductSetting = Recharge:GetProductSettingById(nProductId)
  if not tbProductSetting then
    me.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\229\143\130\230\149\176\233\148\153\232\175\175, \230\151\160\230\179\149\228\189\191\231\148\168"))
    LogErr(string.format("CompensateChargeAward \233\129\147\229\133\183\229\143\130\230\149\176\233\148\153\232\175\175 nProductId = %d", nProductId))
    return
  end
  if not IsPayOpen() then
    me.CenterMsg(i18n.EasyGet("\229\133\133\229\128\188\230\156\170\229\188\128\230\148\190"), true)
    Log("CompensateChargeAward tbProductCreateOrderCallBackFun not open", me.dwID, me.szName, it.dwTemplateId, nProductId)
    return
  end
  local tbProductInfo = Recharge:GetProductSettingById(nProductId)
  if not Recharge.tbProductCreateOrderCallBackFun[tbProductInfo.szProductClass] then
    me.CenterMsg(i18n.EasyGet("\229\136\155\229\187\186\232\174\162\229\141\149\232\191\135\231\168\139\230\156\170\231\159\165\233\148\153\232\175\175\239\188\140\232\175\183\232\129\148\231\179\187\229\174\162\230\156\141"))
    Log("CompensateChargeAward not Recharge.tbProductCreateOrderCallBackFun[tbProductInfo.szProductClass]!!", tbProductInfo.szChannelProductId, tbProductInfo.szProductClass, me.dwID, me.szName, it.dwTemplateId, nProductId)
    return
  end
  local bRet = Recharge.tbProductCreateOrderCallBackFun[tbProductInfo.szProductClass](Recharge, me, tbProductInfo)
  if not bRet then
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\229\164\177\232\180\165\239\188\140\232\175\183\232\129\148\231\179\187\229\174\162\230\156\141"), true)
    Log("CompensateChargeAward tbProductCreateOrderCallBackFun fail", me.dwID, me.szName, it.dwTemplateId, nProductId)
    return
  end
  if me.GetVipLevel() < tbProductInfo.nVipLevelLimit then
    local szMsg = i18n.EasyGet("VIP\231\173\137\231\186\167\228\184\141\232\182\179{0}", tbProductInfo.nVipLevelLimit)
    me.CenterMsg(szMsg, true)
    Log("CompensateChargeAward tbProductCreateOrderCallBackFun", szMsg, me.dwID, me.szName, it.dwTemplateId, nProductId)
    return
  end
  Log("CompensateChargeAward OnUse", me.dwID, me.szName, me.nLevel, it.dwTemplateId, nProductId)
  local bRet = 1
  xpcall(function()
    bRet = Recharge:DeliverProductByProductId(nProductId, bWeb) and 1 or nil
  end, Lib.ShowStack)
  me.CallClientScript("Ui:CloseWindow", "UIItemTips")
  if not bRet then
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\229\164\177\232\180\165\239\188\140\232\175\183\232\129\148\231\179\187\229\174\162\230\156\141"), true)
    Log("CompensateChargeAward use fail", me.dwID, me.szName, it.dwTemplateId, nProductId)
  end
  return bRet
end
