local tbIntroducer = Npc:GetClass("DungeonFactionIntroducer")

function tbIntroducer:OnDialog()
  if YanWuChang.bInDungeon then
    if YanWuChang.emStep == YanWuChang.tbEachStepEnum.DialogBeforeFight then
      YanWuChang:ShowDialogs(YanWuChang.tbDialog.BeforeFight, YanWuChang.StepLimitNextStep, YanWuChang, YanWuChang.tbEachStepEnum.DialogBeforeFight)
    elseif YanWuChang.emStep == YanWuChang.tbEachStepEnum.DialogBeforeLeave then
      YanWuChang:ShowDialogs(YanWuChang.tbDialog.BeforeLeave, RemoteServer.YanWuChangRequest, YanWuChang.tbC2SProtocol.Leave)
    end
  end
end
