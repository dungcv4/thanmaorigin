local tbWnd = Ui:CreateWindow("UIKeYiMenSelectCamp")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MAP_LEAVE,
      self.OnLeaveMap
    }
  }
end

tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  Toggle1 = "node/imgBG/Content/Camp1/Toggle",
  Toggle2 = "node/imgBG/Content/Camp2/Toggle",
  btnOK = "node/imgBG/Content/btnOK"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnOK = function(self)
    KeYiMen.C:RequestSetTongJoinCamp(self.nNewJoinCamp)
    self:Close()
  end
}
tbWnd.tbOnToggleClick = {
  Toggle1 = function(self, isOn)
    self:OnToggleClick(KeYiMen.Def.nCampMengGu, isOn)
  end,
  Toggle2 = function(self, isOn)
    self:OnToggleClick(KeYiMen.Def.nCampXiXia, isOn)
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self.nCurJoinCamp = KeYiMen.C:GetTongJoinCamp()
  self.nNewJoinCamp = self.nCurJoinCamp
  self:UpdateToggle()
end

function tbWnd:OnLeaveMap()
  Ui:CloseWindow("UIActivity")
end

function tbWnd:UpdateToggle()
  self.bDoNothing = true
  self.pPanel:Toggle_SetChecked(self.Toggle1, self.nNewJoinCamp == KeYiMen.Def.nCampMengGu)
  self.pPanel:Toggle_SetChecked(self.Toggle2, self.nNewJoinCamp == KeYiMen.Def.nCampXiXia)
  self.bDoNothing = false
  self.pPanel:SetActive(self.btnOK, self.nNewJoinCamp ~= self.nCurJoinCamp)
end

function tbWnd:OnToggleClick(nCamp, bIsCheck)
  if self.bDoNothing then
    return
  end
  if nCamp ~= self.nNewJoinCamp then
    self.nNewJoinCamp = nCamp
  end
  self:UpdateToggle()
end
