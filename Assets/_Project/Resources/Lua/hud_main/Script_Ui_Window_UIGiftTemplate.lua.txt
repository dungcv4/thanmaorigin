local tbWnd = Ui:CreateClass("UIGiftTemplate", "UIPanel")
local tbControls = {
  Template = "List01/RewardScrollView/Viewport/Content/Template",
  szTitle = "txtTitle",
  szTime = "txtCell/txtTime01",
  szSmallList = "List01",
  btnHelp = "txtTitle/btnHelp",
  btnGo = "B/btn/btnGo",
  szGoText = "B/btn/btnGo/Text",
  btnIconTemplate = "B/btn/btnIconTemplate",
  txtInfo = "B/txtInfo",
  NumCell = "B/btn/NumCell",
  btnGet = "B/btn/btnGet",
  szGetText = "B/btn/btnGet/Text",
  txtLV = "txtCell/txtLV",
  txtSecondTittle = "List01/txtSecondTittle",
  btnPlus = "B/btn/NumCell/btnPlus"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}

function tbWnd:OnCreate()
  self.btnTemplateListView = Ui:NewScrollView(self, tbControls.btnIconTemplate)
end

function tbWnd:OnOpen(nEntranceIndex, nIndex)
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  if self.nEntranceIndex then
    self:UpdateInfo(self.nEntranceIndex, nIndex)
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_ACTIVITY_PANEL_ONOPEN, self.nId)
end

function tbWnd:UpdateInfo(nEntranceIndex, nIndex)
  self.tbActivityList = ActivityMgr.EntranceSchedule:GetActivityList(nEntranceIndex)
  local tbInfo = self.tbActivityList[nIndex]
  if not tbInfo then
    return
  end
  self.nEntranceIndex = nEntranceIndex or self.nEntranceIndex
  self.nId = tbInfo.nId
  self.pInstance = ActivityMgr:GetInstanceById(tbInfo.nId)
  if not self.pInstance then
    self.pInstance = ActivityMgr.EntranceSchedule:BuildInstance(tbInfo.nId, tbInfo.szClassName)
  end
  self.nIndex = nIndex
  self.szHelpKey = tbInfo.szHelpKey
  self.pPanel:Label_SetText(self.szTitle, tbInfo.szName)
  local szTime = string.format("<color=#1FFB24>%s</color>", tbInfo.szShowTime)
  local nStartTime = self.pInstance:GetStartTime()
  local nEndTime = self.pInstance:GetEndTime()
  if nStartTime < GetTime() then
    local nDiff = nEndTime - GetTime()
    szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> {1}\229\144\142\231\187\147\230\157\159", tbInfo.szShowTime, Lib:TimeDesc2(nDiff))
  end
  self.bTimeOut = nEndTime < GetTime()
  if self.bTimeOut then
    szTime = i18n.EasyGet("<color=#1FFB24>{0}</color> \229\183\178\231\187\147\230\157\159", tbInfo.szShowTime)
  end
  self.pPanel:Label_SetText(tbControls.szTime, szTime)
  self.pPanel:Label_SetText(tbControls.txtLV, i18n.EasyGet("\231\173\137\231\186\167\233\153\144\229\136\182: {0}", tbInfo.nLevel))
  self.pPanel:Label_SetText(tbControls.txtSecondTittle, i18n.Parse(self.pInstance.tbConfig.txtSecondTittle))
  self.pPanel:Label_SetText(tbControls.szGoText, i18n.Parse(self.pInstance.tbConfig.szBtnGoText))
  self:UpdateListView(tbInfo.tbAward)
  self:UpdateBtnTemplateListView(tbInfo)
  self:BindFindPath()
  self.szSmallPanelText = self.pInstance.tbConfig.szSmallPanelText
  if self.szSmallPanelText then
    self.pPanel:Label_SetText(tbControls.txtInfo, i18n.Parse(self.szSmallPanelText))
    self.pPanel:SetActive(tbControls.txtInfo, true)
  else
    self.pPanel:SetActive(tbControls.txtInfo, false)
  end
end

function tbWnd:UpdateListView(tbAward)
  if not tbAward then
    return
  end
  local szTemplate = self.Template
  if self.szTemplate ~= szTemplate then
    if self.tbListView then
      self.tbListView:Clear()
    end
    self.tbListView = Ui:NewScrollView(self, szTemplate)
    self.szTemplate = szTemplate
  end
  self.pPanel:SetActive(szTemplate, bUseSmall)
  self.tbListView:Clear()
  for i = 1, #tbAward do
    self.tbListView:SetElement(i, function(tbElement)
      self:UpdateListElement(tbElement, tbAward[i])
    end, true)
  end
end

function tbWnd:UpdateListElement(tbElement, tbAward)
  local tbGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  if "Item" == tbAward[1] then
    tbGrid:SetItemByTemplate(tbAward[2], tbAward[3], nil, nil, {bShowTip = true})
    local szName = Item:GetName(tbAward[2], me)
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  else
    tbGrid:SetDigitalItem(tbAward[1], tbAward[3])
    local szName = Currency:GetName(tbAward[1])
    tbElement.pPanel:Label_SetText("txtActivityName", szName)
  end
  tbGrid:SetClickEnable(true)
end

function tbWnd:BindFindPath()
  if self.pInstance.tbConfig.szNumCellText then
    self.pPanel:SetActive(tbControls.NumCell, true)
    self.pPanel:Label_SetText(tbControls.NumCell .. "/txtNum", i18n.Parse(i18n.Get(self.pInstance.tbConfig.szNumCellText, self.pInstance.tbConfig.nMaxNum - self.pInstance:GetJoinNum())))
  else
    self.pPanel:SetActive(tbControls.NumCell, false)
  end
  if self.pInstance.tbConfig.szNumCellJumpType then
    self.pPanel:SetActive(tbControls.btnPlus, true)
    Ui:Button_BindEvent(self.pPanel, tbControls.btnPlus, self.OnGoBtnClick, self, self.pInstance.tbConfig.szNumCellJumpType, self.pInstance.tbConfig.szNumCellJumpParam)
  else
    self.pPanel:SetActive(tbControls.btnPlus, false)
  end
  if not self.pInstance.tbConfig.szBtnGetText then
    self.pPanel:SetActive(tbControls.btnGet, false)
  else
    self.pPanel:SetActive(tbControls.btnGet, true)
    self.pPanel:Label_SetText(tbControls.szGetText, i18n.Parse(self.pInstance.tbConfig.szBtnGetText))
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGet, ActivityMgr:GenRedPointEventIdList(Lib:SplitStr(self.pInstance.tbConfig.szGetBtnRedPoint or "", "|")))
  end
  if self.pInstance.tbConfig.nIsYanHua and 1 == self.pInstance.tbConfig.nIsYanHua then
    Ui:Button_BindEvent(self.pPanel, tbControls.btnGo, self.GoTo, self)
    return
  end
  if not self.pInstance.GetJumpMapList or not self.pInstance:GetJumpMapList() then
    Ui:Button_BindEvent(self.pPanel, tbControls.btnGo, self.OnGoBtnClick, self, self.pInstance.tbConfig.szJumpType, self.pInstance.tbConfig.szJumpParam)
  else
    Ui:Button_BindEvent(self.pPanel, tbControls.btnGo, self.OnRecommendBtnClicked, self)
  end
  if self.pInstance.GetRedPointEventList then
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGo, ActivityMgr:GenRedPointEventIdList(Lib:SplitStr(self.pInstance.tbConfig.szGoBtnRedPoint or "", "|")))
  end
end

function tbWnd:OnGoBtnClick(szCallBack, szParam)
  local fn = ActivityMgr.tbJumpFunc[szCallBack]
  local tbParam = Lib:SplitStrTry2Num(szParam or "", "|")
  if fn and tbParam and #tbParam > 0 then
    fn(self.tbActivityList[self.nIndex].nId, table.unpack(tbParam))
    if "GotoPos" == szCallBack or "GotoNPC" == szCallBack then
      Ui:CloseWindow("UIGiftTemplate")
    end
  end
end

function tbWnd:GoTo()
  if self.bTimeOut then
    me.CenterMsg(i18n.EasyGet("\230\180\187\229\138\168\229\183\178\231\187\147\230\157\159"))
    return
  end
  self.pInstance:DoCommonJump()
end

function tbWnd:UpdateBtnTemplateListView(tbInfo)
  local tbExtraBtnList = ActivityMgr.EntranceSchedule:GetExtraBtnList(self.nEntranceIndex, tbInfo.nIndex)
  self.btnTemplateListView:Clear()
  for nIndex, tbBtnInfo in ipairs(tbExtraBtnList) do
    self.btnTemplateListView:SetElement(nIndex, function(tbElement, nIndex)
      self:UpdateBtnListElement(tbElement, nIndex, tbBtnInfo)
    end)
  end
end

function tbWnd:UpdateBtnListElement(tbElement, nIndex, tbBtnInfo)
  tbElement.pPanel:Label_SetText("Text", i18n.Get(tbBtnInfo.szBtnName))
  tbElement.pPanel:Sprite_SetSprite("imgIcon", tbBtnInfo.szImgInfo)
  tbElement.pPanel:Button_BindEvent("Main", self.OnClickGift, self, tbBtnInfo.szCallBack, tbBtnInfo.szParam)
  if tbBtnInfo.szRedPoint and tbBtnInfo.szRedPoint ~= "" then
    RedPointMgr:DynamicRegisteRedPoint(tbElement.pPanel, "Main", ActivityMgr:GenRedPointEventIdList(Lib:SplitStr(tbBtnInfo.szRedPoint, "|")))
  end
end

function tbWnd:OnClickGift(szCallBack, szParam)
  if "TypeRank" == szParam then
    if not self.tbEntranceRankData then
      self.tbEntranceRankData = ActivityMgr.EntranceSchedule:GetEntranceRankData(self.nEntranceIndex)
    end
    Ui:OpenWindow("UITemporaryActivityRank", self.tbEntranceRankData, 1)
  else
    self:OnGoBtnClick(szCallBack, szParam)
  end
end

function tbWnd:OnGotoRecommendSucc()
  self.pPanel:SetActive(self.wndJumpTo, false)
  self:Close()
end

function tbWnd:OnRecommendBtnClicked()
  if Ui:WindowVisible("UIActivityTemplate") == true then
    Ui:GetWindow("UIActivityTemplate"):OnOpenWndJumpTo(self.nId)
  end
end

function tbWnd:OnClose()
  if self.tbListView then
    self.tbListView:Clear()
  end
  if self.btnTemplateListView then
    self.btnTemplateListView:Clear()
  end
  self.szSmallPanelText = nil
end
