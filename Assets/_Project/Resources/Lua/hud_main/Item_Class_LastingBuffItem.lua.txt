local tbLastingBuffItem = Item:GetClass("LastingBuffItem")
local tbExtParam = {
  BUFFID = 1,
  BUFFIDLV = 2,
  BUFFTIME = 3
}

function tbLastingBuffItem:OnUse(pItem)
  local nTemplateId = pItem.dwTemplateId
  local nHave = me.GetItemCountByTypeInBag(nTemplateId)
  if nHave < 1 then
    me.CenterMsg(i18n.EasyGet("\233\129\147\229\133\183\228\184\141\232\182\179"))
    return 0
  end
  local nBuffId = KItem.GetItemExtParam(nTemplateId, tbExtParam.BUFFID)
  local nBuffLv = KItem.GetItemExtParam(nTemplateId, tbExtParam.BUFFIDLV)
  local nBuffTime = KItem.GetItemExtParam(nTemplateId, tbExtParam.BUFFTIME)
  local nCurTime = GetTime()
  local nEndTime = nCurTime + nBuffTime
  local pNpc = me.GetNpc()
  if pNpc then
    local tbSkillState = pNpc.GetSkillState(nBuffId)
    if tbSkillState and tbSkillState.nTrueTime and nCurTime < tbSkillState.nTrueTime then
      nEndTime = tbSkillState.nTrueTime + nBuffTime
    end
  else
    me.CenterMsg(i18n.EasyGet("\228\189\191\231\148\168\229\164\177\232\180\165"))
    return 0
  end
  me.AddSkillState(nBuffId, nBuffLv, FightSkill.STATE_TIME_TYPE.state_time_truetime, nEndTime)
  me.CenterMsg(i18n.EasyGet("{0}\228\189\191\231\148\168\230\136\144\229\138\159!", i18n.Parse(pItem.szName)))
  return 1
end
