local tbWnd = Ui:CreateWindow("UIStoneCompose")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItemEvent,
      self
    },
    {
      EventNotify.emNOTIFY_WND_CLOSED,
      self.OnWndClose,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  DropdownLevel = "node/imgBG/Content/StoneCompose/panelStoneList/DropdownLevel",
  btnTemplate = "node/imgBG/btnList/ScrollView/Viewport/Content/btnTemplate",
  TargetSV = "node/imgBG/Content/StoneCompose/panelStoneList/imgList",
  TargetTemplate = "node/imgBG/Content/StoneCompose/panelStoneList/imgList/List/Template",
  YuanShi = "node/imgBG/Content/StoneCompose/panelCompose/YuanShi",
  YuanShiimgFrame = "node/imgBG/Content/StoneCompose/panelCompose/YuanShi/Stone/imgFrame",
  YuanShibtnEuipAdd = "node/imgBG/Content/StoneCompose/panelCompose/YuanShi/Stone/btnEuipAdd",
  YuanShiimgSelected = "node/imgBG/Content/StoneCompose/panelCompose/YuanShi/Stone/imgSelected",
  YuanShiUIItemGrid = "node/imgBG/Content/StoneCompose/panelCompose/YuanShi/Stone/UIItemGrid",
  YuanShiEffect = "node/imgBG/Content/StoneCompose/panelCompose/YuanShi/Stone/YuanShiEffect",
  Stone = "node/imgBG/Content/StoneCompose/panelCompose/StoneList/Stone",
  BagTemplate = "node/imgBG/Content/StoneCompose/panelStoneBag/imgList/List/Template",
  btnCompose = "node/imgBG/Content/StoneCompose/panelCompose/btnCompose",
  txtComposeCost = "node/imgBG/Content/StoneCompose/panelCompose/txtComposeCost",
  MakeEffect = "node/imgBG/Content/StoneCompose/panelCompose/MakeEffect"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnCompose = function(self)
    local nTargetId = self.tbTargetStone[self.nEquipPos][self.nTargetIndex]
    local bRet, szErr = Item.Stone:CanStoneForge(me, nTargetId, self.tbSelectConsumeId)
    if not bRet then
      return me.CenterMsg(szErr)
    end
    for i = 1, 5 do
      local Animator = self.pPanel:GetAnimator(tbControls.Stone .. i .. "/LinEffect")
      Animator:Play("LineEffect_UP")
    end
    self.pPanel:Animator_SetEnable(tbControls.YuanShi, true)
    Timer:Register(Env.GAME_FPS * 1.5, function()
      RemoteServer.OnEquipStoneRequest("Forge", nTargetId, self.tbSelectConsumeId)
    end)
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("StoneForgeHelp")
  end
}

function tbWnd:OnSyncItemEvent()
  self:UpdateDetailInfo()
end

function tbWnd:OnForgeStone(nTemplateId)
  if not self.nStoneIndex then
    for i = 1, Item.Stone.nForgeNeed do
      if not self.tbSelectConsumeId[i] then
        self.pPanel:SetActive(tbControls.Stone .. i .. "/imgSelected", true)
        self.nStoneIndex = i
        break
      end
    end
  end
  self.pPanel:SetActive(tbControls.Stone .. self.nStoneIndex .. "/LinEffect", true)
  self.tbSelectConsumeId[self.nStoneIndex] = nTemplateId
  self.tbItemGrid[self.nStoneIndex]:SetItemByTemplate(nTemplateId)
  self.tbItemGrid[self.nStoneIndex]:SetActive(true)
  local nTargetId = self.tbTargetStone[self.nEquipPos][self.nTargetIndex]
  local bRet, szErr = Item.Stone:CanStoneForge(me, nTargetId, self.tbSelectConsumeId)
  if bRet then
    self.pPanel:SetActive(tbControls.MakeEffect, true)
  end
  for i = 1, Item.Stone.nForgeNeed do
    if not self.tbSelectConsumeId[i] then
      self.pPanel:SetActive(tbControls.Stone .. self.nStoneIndex .. "/imgSelected", false)
      self.nStoneIndex = i
      self.pPanel:SetActive(tbControls.Stone .. i .. "/imgSelected", true)
      break
    end
  end
  self:UpdateBagSSV()
end

function tbWnd:OnCreate()
  self.pPanel:SetActive(tbControls.MakeEffect, false)
  self.tbBtnScrollView = Ui:NewScrollView(self, tbControls.btnTemplate, self.OnBtnClick)
  self.tbTargetSScrollView = Ui:NewScrollView(self, tbControls.TargetTemplate, self.OnTargetSClick)
  self.tbBagSScrollView = Ui:NewScrollView(self, tbControls.BagTemplate, self.OnBagSClick)
  self.tbTargetStone = {}
  self.tbAllStone = {}
  self.yuanShiItemGrid = Ui:NewExistItemGrid(self, tbControls.YuanShiUIItemGrid)
  self.yuanShiItemGrid:SetClickEnable(true)
  self.itemGridConsume1 = Ui:NewExistItemGrid(self, tbControls.Stone .. "4/UIItemGrid")
  self.itemGridConsume2 = Ui:NewExistItemGrid(self, tbControls.Stone .. "5/UIItemGrid")
  self.itemGridConsume1:SetClickEnable(true)
  self.itemGridConsume2:SetClickEnable(true)
  self.tbItemGrid = {}
  
  local function funcSelect(index, status)
    self.pPanel:SetActive(tbControls.Stone .. index .. "/imgSelected", status)
  end
  
  for i = 1, Item.Stone.nForgeNeed do
    self.tbItemGrid[i] = Ui:NewExistItemGrid(self, tbControls.Stone .. i .. "/UIItemGrid")
    self.tbItemGrid[i]:SetClickEnable(true, function()
      self.tbSelectConsumeId[i] = nil
      self.tbItemGrid[i]:SetEmpty()
      self.tbItemGrid[i]:SetActive(false)
      self.pPanel:SetActive(tbControls.MakeEffect, false)
      self.pPanel:SetActive(tbControls.Stone .. i .. "/LinEffect", false)
      if self.nStoneIndex then
        funcSelect(self.nStoneIndex, false)
      end
      self.nStoneIndex = i
      funcSelect(self.nStoneIndex, true)
      self:UpdateBagSSV()
    end)
  end
  for i = 1, Item.Stone.nForgeNeed do
    Ui:Button_BindEvent(self.pPanel, tbControls.Stone .. i .. "/btnEuipAdd", function()
      if self.nStoneIndex then
        funcSelect(self.nStoneIndex, false)
      end
      self.nStoneIndex = i
      funcSelect(self.nStoneIndex, true)
    end)
  end
  self.tbLevelOptions = {}
  for _, v in ipairs(Item.Stone.tbFrameTime2ForgeLevel) do
    if 1 == GetTimeFrameState(v.szTimeFrame) then
      table.insert(self.tbLevelOptions, v.nForgeLevel)
    else
      break
    end
  end
  self:InitLevelDropDown()
  self.pPanel:Dropdown_BindEvent(tbControls.DropdownLevel, function(index)
    self.nDropLevelIndex = index
    self.nTargetIndex = 1
    self:UpdateTargetSSV()
    self:UpdateView()
  end)
end

function tbWnd:OnClose()
  self.pPanel:SetActive(tbControls.MakeEffect, false)
end

function tbWnd:OnOpen(nEquipPos, nStoneId)
  nEquipPos = nEquipPos or 0
  self:InitBtnSV()
  self.tbBtnScrollView:OnOpenDoHightLight(nEquipPos + 1)
  self.nStoneIndex = nil
  self.nEquipPos = nEquipPos or 0
  self.nTargetIndex = 1
  self.tbSelectConsumeId = {}
  self.nDropLevelIndex = 0
  self:UpdateTargetSSV(nStoneId)
  self.tbTargetSScrollView:ScrollToIndex(self.nTargetIndex)
  self:UpdateDetailInfo()
  self.pPanel:Dropdown_SetValue(tbControls.DropdownLevel, self.nDropLevelIndex)
end

function tbWnd:InitLevelDropDown()
  local tbLevelOptions = {
    i18n.EasyGet("\229\133\168\233\131\168")
  }
  for i, v in ipairs(self.tbLevelOptions) do
    table.insert(tbLevelOptions, i18n.EasyGet("{0}\231\186\167", v))
  end
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownLevel, tbLevelOptions)
end

function tbWnd:UpdateView(szParam)
  self.nStoneIndex = nil
  self.tbSelectConsumeId = {}
  self:UpdateDetailInfo()
  if szParam and "ForgeRsp" == szParam then
    self.pPanel:SetActive(tbControls.MakeEffect, false)
  end
end

function tbWnd:OnBtnClick(nIndex)
  self.nEquipPos = nIndex - 1
  self.nTargetIndex = 1
  self:UpdateTargetSSV()
  self:UpdateView()
  self.pPanel:SetActive(tbControls.MakeEffect, false)
end

function tbWnd:InitBtnSV()
  local tbName = {}
  for i = 1, Item.EQUIP_AMULET do
    tbName[i] = i18n.Get(Item:GetNotExtEquipName(i))
  end
  self.tbBtnScrollView:Reset()
  for i, v in pairs(tbName) do
    self.tbBtnScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("Text", tbName[i])
    end)
  end
end

function tbWnd:OnTargetSClick(nIndex)
  local tbOldElement = self.tbTargetSScrollView:GetElement(self.nTargetIndex)
  if tbOldElement then
    tbOldElement.pPanel:SetActive("imgSelected", false)
  end
  self.nTargetIndex = nIndex
  local tbCurElement = self.tbTargetSScrollView:GetElement(self.nTargetIndex)
  tbCurElement.pPanel:SetActive("imgSelected", true)
  self.nStoneIndex = nil
  self.tbSelectConsumeId = {}
  self:UpdateDetailInfo()
  self.pPanel:SetActive(tbControls.MakeEffect, false)
end

function tbWnd:UpdateTargetSSV(nStoneId)
  self:UpdateTargetStoneList(nStoneId)
  self.tbTargetSScrollView:Clear()
  for i, v in pairs(self.tbTargetStone[self.nEquipPos]) do
    self.tbTargetSScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      pPanel:SetActive("imgSelected", i == self.nTargetIndex)
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(v)
      local szName = KItem.GetItemShowInfo(v, me.nFaction, me.nFactionSect, me.nSex)
      pPanel:Label_SetText("txtStoneName", i18n.Parse(szName))
      self:SetStoneAttr(v, pPanel)
    end)
  end
end

function tbWnd:OnBagSClick(nIndex, tbElement)
  if tbElement.nCount <= 0 then
    return
  end
  self:OnForgeStone(tbElement.nTemplateId)
end

function tbWnd:UpdateBagSSV()
  local tbBag = {}
  local nTargetId = self.tbTargetStone[self.nEquipPos][self.nTargetIndex]
  local tbCfg = Item.Stone.tbYuanShiInfo[nTargetId]
  for i, v in ipairs(tbCfg[6]) do
    local nCount, tbItem = me.GetItemCountByTypeInBag(v)
    for j = 1, Item.Stone.nForgeNeed do
      if self.tbSelectConsumeId[j] == v then
        nCount = nCount - 1
      end
    end
    if tbCfg[3] == v then
      nCount = nCount - 1
    end
    if tbCfg[4] == v then
      nCount = nCount - 1
    end
    if nCount > 0 then
      table.insert(tbBag, {v, nCount})
    else
      table.insert(tbBag, {v, 0})
    end
  end
  self.tbBagSScrollView:Reset()
  for i, v in pairs(tbBag) do
    self.tbBagSScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
      itemGrid:SetItemByTemplate(v[1], tostring(v[2]))
      itemGrid:SetClickEnable(true)
      itemGrid.szItemOpt = "StoneForge"
      local szName = KItem.GetItemShowInfo(v[1], me.nFaction, me.nFactionSect, me.nSex)
      pPanel:Label_SetText("txtStoneName", i18n.Parse(szName))
      self:SetStoneAttr(v[1], pPanel, true)
      tbElement.nTemplateId = v[1]
      tbElement.nCount = v[2]
    end)
  end
end

function tbWnd:UpdateTargetStoneList(nStoneId)
  if not self.tbAllStone[self.nEquipPos] then
    local nOpenLevel = Item.Stone:GetForgeTimeframeOpenLevel()
    self.tbAllStone[self.nEquipPos] = {}
    local tbCfgStone = Item.Stone.tbEquipPos2Stone[self.nEquipPos]
    for i, v in pairs(Item.Stone.tbYuanShiInfo) do
      local nTemplateId = v[1]
      local nLevel = Item:GetLevel(nTemplateId)
      if tbCfgStone[nTemplateId] and nOpenLevel >= nLevel then
        table.insert(self.tbAllStone[self.nEquipPos], nTemplateId)
      end
    end
  end
  if 0 == self.nDropLevelIndex then
    self.tbTargetStone[self.nEquipPos] = self.tbAllStone[self.nEquipPos]
  else
    self.tbTargetStone[self.nEquipPos] = {}
    for i, nTemplateId in ipairs(self.tbAllStone[self.nEquipPos]) do
      local nLevel = Item:GetLevel(nTemplateId)
      if nLevel == self.tbLevelOptions[self.nDropLevelIndex] then
        table.insert(self.tbTargetStone[self.nEquipPos], nTemplateId)
      end
    end
  end
  if nStoneId then
    for i, v in ipairs(self.tbTargetStone[self.nEquipPos]) do
      if nStoneId == v then
        self.nTargetIndex = i
      end
    end
  else
    self.nTargetIndex = 1
  end
end

function tbWnd:UpdateDetailInfo()
  local nTargetId = self.tbTargetStone[self.nEquipPos][self.nTargetIndex]
  local tbCfg = Item.Stone.tbYuanShiInfo[nTargetId]
  if not tbCfg then
    return
  end
  self.pPanel:Label_SetText(tbControls.txtComposeCost, tbCfg[2])
  local nYuanShiId = tbCfg[5]
  local nHadNum = me.GetItemCountByTypeInBag(nYuanShiId)
  self.yuanShiItemGrid:SetItemByTemplate(nYuanShiId, nHadNum .. "/1")
  if nHadNum < 1 then
    self.yuanShiItemGrid:LabelSuffix_SetColor("Red")
    self.pPanel:SetActive(tbControls.YuanShiEffect, false)
  else
    self.yuanShiItemGrid:LabelSuffix_SetColor("Green")
    self.pPanel:SetActive(tbControls.YuanShiEffect, true)
  end
  self.yuanShiItemGrid:SetActive(true)
  self.yuanShiItemGrid:BaseColor_SetActive(false)
  nHadNum = me.GetItemCountByTypeInBag(tbCfg[3])
  self.itemGridConsume1:SetItemByTemplate(tbCfg[3], nHadNum .. "/1")
  if nHadNum < 1 then
    self.itemGridConsume1:LabelSuffix_SetColor("Red")
    self.pPanel:SetActive(tbControls.Stone .. "4/LinEffect", false)
  else
    self.itemGridConsume1:LabelSuffix_SetColor("Green")
    self.pPanel:SetActive(tbControls.Stone .. "4/LinEffect", true)
  end
  self.itemGridConsume1:SetActive(true)
  self.itemGridConsume1:BaseColor_SetActive(false)
  nHadNum = me.GetItemCountByTypeInBag(tbCfg[4])
  self.itemGridConsume2:SetItemByTemplate(tbCfg[4], nHadNum .. "/1")
  if nHadNum < 1 then
    self.itemGridConsume2:LabelSuffix_SetColor("Red")
    self.pPanel:SetActive(tbControls.Stone .. "5/LinEffect", false)
  else
    self.itemGridConsume2:LabelSuffix_SetColor("Green")
    self.pPanel:SetActive(tbControls.Stone .. "5/LinEffect", true)
  end
  self.itemGridConsume2:SetActive(true)
  self.itemGridConsume2:BaseColor_SetActive(false)
  for i = 1, Item.Stone.nForgeNeed do
    if self.tbSelectConsumeId[i] then
      self.tbItemGrid[i]:SetActive(true)
      self.tbItemGrid[i]:SetItemByTemplate(self.tbSelectConsumeId[i])
      self.pPanel:SetActive(tbControls.Stone .. i .. "/LinEffect", true)
    else
      self.tbItemGrid[i]:SetActive(false)
      self.pPanel:SetActive(tbControls.Stone .. i .. "/LinEffect", false)
    end
    self.tbItemGrid[i]:BaseColor_SetActive(false)
    if self.nStoneIndex and self.nStoneIndex == i then
      self.pPanel:SetActive(tbControls.Stone .. i .. "/imgSelected", true)
    else
      self.pPanel:SetActive(tbControls.Stone .. i .. "/imgSelected", false)
    end
  end
  self:UpdateBagSSV()
end

function tbWnd:SetStoneAttr(nStoneId, pPanel, bOnlyOne)
  local tbStoneAttrCfg = Item.Stone:GetStoneAttri(nStoneId)
  if tbStoneAttrCfg[1] then
    local szDesc, szValueCur = FightSkill:GetMagicDescSplit(tbStoneAttrCfg[1][1], {
      tbStoneAttrCfg[1][2],
      tbStoneAttrCfg[1][3],
      tbStoneAttrCfg[1][4]
    })
    pPanel:Label_SetText("txtAttr1", szDesc)
    pPanel:Label_SetText("txtAttr1/txtAttr1Value", szValueCur)
  else
    pPanel:Label_SetText("txtAttr1", "")
    pPanel:Label_SetText("txtAttr1/txtAttr1Value", "")
  end
  if bOnlyOne then
    return
  end
  if tbStoneAttrCfg[2] then
    local szDesc, szValueCur = FightSkill:GetMagicDescSplit(tbStoneAttrCfg[2][1], {
      tbStoneAttrCfg[2][2],
      tbStoneAttrCfg[2][3],
      tbStoneAttrCfg[2][4]
    })
    pPanel:Label_SetText("txtAttr2", szDesc)
    pPanel:Label_SetText("txtAttr2/txtAttr2Value", szValueCur)
  else
    pPanel:Label_SetText("txtAttr2", "")
    pPanel:Label_SetText("txtAttr2/txtAttr2Value", "")
  end
end

function tbWnd:OnWndClose(szUiName)
  if "UIGetAwardShow" == szUiName then
    local Animator = self.pPanel:GetAnimator(tbControls.YuanShi)
    Animator:Play("YuanSh_Open", 0, 0)
    Animator:Update(0)
    Animator.enabled = false
  end
end
