Guide.STATE_NONT = 1
Guide.STATE_RUNNING = 2
Guide.STATE_WAITTING = 3
Guide.SAVE_GROUP = GetGroupKey("Guide")

function Guide:ReadProcessData(pPlayer, nGuideID)
  return pPlayer.GetUserValue(Guide.SAVE_GROUP, nGuideID)
end

function Guide:HaveFinishGuide(pPlayer, nGuideId)
  return self:ReadProcessData(pPlayer, nGuideId) >= #self.Config[nGuideId]
end
