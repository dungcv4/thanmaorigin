local tbWnd = Ui:CreateWindow("UIPKCompetitionReport1")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_DINGFENGPK_SYNC_RANK,
      self.OnSynRankData,
      self
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  txtTitle = "node/imgBG/txtTitle",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  NotInfo = "node/imgBG/Content/Ranking/NotInfo",
  txtRankTitle = "node/imgBG/Content/Ranking/txtTitle",
  txtCondition = "node/imgBG/Content/Ranking/txtCondition",
  txtWarTime = "node/imgBG/Content/Ranking/txtWarTime",
  Template = "node/imgBG/Content/Ranking/rankContent/Template",
  txtScoreValue = "node/imgBG/Content/Ranking/txtScoreValue",
  txtRank = "node/imgBG/Content/Ranking/txtRank",
  txtRankValue = "node/imgBG/Content/Ranking/txtRank/txtRankValue",
  txtTips = "node/imgBG/Content/Ranking/txtTips",
  txtPage = "node/imgBG/Content/Ranking/txtPage",
  btnPrePage = "node/imgBG/Content/Ranking/btnPrePage",
  btnNextPage = "node/imgBG/Content/Ranking/btnNextPage",
  btnFirstPage = "node/imgBG/Content/Ranking/btnFirstPage",
  btnLastPage = "node/imgBG/Content/Ranking/btnLastPage"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnPrePage = function(self)
    self:PrePage()
  end,
  btnNextPage = function(self)
    self:NextPage()
  end,
  btnLastPage = function(self)
    self:LastPage()
  end,
  btnFirstPage = function(self)
    self:FirstPage()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(tbWnd, tbControls.Template)
end

function tbWnd:OnClose()
end

function tbWnd:OnOpen(nOpenType)
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.nOpenType = nOpenType
  if self.nOpenType == DingFengPK.OPEN_TYPE.PRE then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\233\162\132\233\128\137\232\181\155\230\136\152\230\138\165"))
    self.pPanel:Label_SetText(tbControls.txtRankTitle, i18n.EasyGet("\233\162\132\233\128\137\232\181\155\230\136\152\230\138\165"))
  elseif self.nOpenType == DingFengPK.OPEN_TYPE.MID then
    self.pPanel:Label_SetText(tbControls.txtTitle, i18n.EasyGet("\230\183\152\230\177\176\232\181\155\230\136\152\230\138\165"))
    self.pPanel:Label_SetText(tbControls.txtRankTitle, i18n.EasyGet("\230\183\152\230\177\176\232\181\155\230\136\152\230\138\165"))
  end
  self.nCurPage = 1
  self:DoUpdateRank()
  self:UpdateTip()
end

function tbWnd:UpdateTip()
  self.pPanel:Label_SetText(tbControls.txtCondition, i18n.EasyGet(""))
  self.pPanel:Label_SetText(tbControls.txtWarTime, i18n.EasyGet(""))
end

function tbWnd:SetMyData(tbMyRankInfo)
  self.pPanel:SetActive(tbControls.txtRank, true)
  self.pPanel:SetActive(tbControls.txtScoreValue, true)
  self.pPanel:Label_SetText(tbControls.txtRank, i18n.EasyGet("\230\136\152\233\152\159\230\142\146\229\144\141\239\188\154"))
  self.pPanel:Label_SetText(tbControls.txtRankValue, tbMyRankInfo[DingFengPK.RANK] or "-")
  self.pPanel:Label_SetText(tbControls.txtScoreValue, tbMyRankInfo[DingFengPK.SCORE] or "-")
end

function tbWnd:SetElement(tbElement, tbRecord)
  local pPanel = tbElement.pPanel
  if tbRecord.nRank <= 3 then
    pPanel:SetActive("txtRank", false)
    pPanel:SetActive("UIRankNo", true)
    pPanel:SetActive("UIRankNo/imgNO1", 1 == tbRecord.nRank)
    pPanel:SetActive("UIRankNo/imgNO2", 2 == tbRecord.nRank)
    pPanel:SetActive("UIRankNo/imgNO3", 3 == tbRecord.nRank)
  else
    pPanel:SetActive("UIRankNo", false)
    pPanel:SetActive("txtRank", true)
    pPanel:Label_SetText("txtRank", tbRecord.nRank)
  end
  if tbRecord.nTeamId ~= DingFengPK:GetBattleTeamId(me.dwID) then
    pPanel:Sprite_SetColor("Main", 0, 0, 0, 20)
  else
    pPanel:Sprite_SetColor("Main", 255, 255, 255, 255)
  end
  pPanel:Label_SetText("txtTeamName", tbRecord.szTeamName)
  pPanel:Label_SetText("txtTime", Lib:TimeDesc(tbRecord.nBattleTime or 0))
  pPanel:Label_SetText("txtNumber", tbRecord.nWinCount or 0)
  pPanel:Label_SetText("txtScore", tbRecord.nScore or 0)
  pPanel:Label_SetText("txtServer", GetServerName(tbRecord.nServerId))
  Ui:Button_BindEvent(pPanel, "", function()
    Ui:OpenWindow("UIPKCompetitionTeam", tbRecord.nTeamId)
  end)
end

function tbWnd:DoUpdateRank()
  local nPage = self.nCurPage
  local tbData = DingFengPK:CheckUpdateData(self.nOpenType, nPage)
  self:UpdateMyRankInfo()
  if tbData then
    self:SetRankData(tbData)
  else
    self.tbScrollView:Clear()
    self.pPanel:SetActive(tbControls.NotInfo, true)
  end
end

function tbWnd:SetRankData(tbData)
  self.tbScrollView:Clear()
  if not tbData or Lib:IsEmptyTB(tbData) then
    self.pPanel:SetActive(tbControls.NotInfo, true)
    return
  end
  for nIndex, tbRecord in ipairs(tbData) do
    self.tbScrollView:SetElement(nIndex, function(tbElement)
      tbWnd:SetElement(tbElement, tbRecord)
    end)
  end
  self.pPanel:SetActive(tbControls.NotInfo, false)
end

function tbWnd:UpdateMyRankInfo()
  local tbMyInfo = DingFengPK.tbMyTeamInfo
  if not (tbMyInfo and tbMyInfo[DingFengPK.RANK]) or 0 == tbMyInfo[DingFengPK.RANK] or tbMyInfo[DingFengPK.RANK] > DingFengPK.PreCfg.RANK_COUNT then
    self:SetMyRankInfo_NOData()
  else
    self:SetMyRankInfo(tbMyInfo)
  end
  local nMaxNum = DingFengPK.tbMaxLength[self.nOpenType] or 0
  self.nMaxPage = math.max(math.ceil(nMaxNum / DingFengPK.PAGE_NUM), 1)
  self.pPanel:Label_SetText(tbControls.txtPage, string.format("%d/%d", self.nCurPage, self.nMaxPage))
end

function tbWnd:SetMyRankInfo_NOData()
  self.pPanel:SetActive(tbControls.txtRank, false)
  self.pPanel:SetActive(tbControls.txtScoreValue, false)
end

function tbWnd:SetMyRankInfo(tbData)
  self:SetMyData(tbData)
end

function tbWnd:OnSynRankData()
  self:DoUpdateRank()
end

function tbWnd:FirstPage()
  if 1 == self.nCurPage then
    return
  end
  self.nCurPage = 1
  self:DoUpdateRank()
end

function tbWnd:LastPage()
  if self.nCurPage == self.nMaxPage then
    return
  end
  self.nCurPage = self.nMaxPage
  self:DoUpdateRank()
end

function tbWnd:PrePage()
  if self.nCurPage <= 1 then
    return
  end
  self.nCurPage = self.nCurPage - 1
  self:DoUpdateRank()
end

function tbWnd:NextPage()
  if self.nCurPage >= self.nMaxPage then
    return
  end
  self.nCurPage = self.nCurPage + 1
  self:DoUpdateRank()
end
