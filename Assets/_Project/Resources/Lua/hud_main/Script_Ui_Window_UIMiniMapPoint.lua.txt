local tbWnd = Ui:CreateClass("UIMiniMapPoint", "UIPanel")
tbWnd.tbControls = {
  Npc = "Npc",
  Tuwen = "Tuwen",
  Wenzi = "Wenzi",
  Img = "Img"
}
tbWnd.tbUseResPath = {Tuwen = 1, Img = 1}
tbWnd.tbUseNpcModel = {Npc = 1}

function tbWnd:OnCreate()
  self.tbUsedObjs = {}
  Map:ChangeMapPosToUI()
end

function tbWnd:ShowMapPoints(nMapTId, nOffsetX, nOffsetY)
  self:ResetMapPoints()
  local tbMapPoints = Map:GetMapPointSetting(nMapTId)
  local nCount = 1
  for _, tbPoint in pairs(tbMapPoints) do
    if tbPoint.bIsPosMultiple then
      for _, tbPos in pairs(tbPoint.tbPos) do
        self:GetPoint(nCount, tbPoint, i18n.Parse(tbPos.szName), tbPos.x + nOffsetX, tbPos.y + nOffsetY)
        nCount = nCount + 1
      end
    else
      self:GetPoint(nCount, tbPoint, i18n.Parse(tbPoint.szName), tbPoint.tbPos.x + nOffsetX, tbPoint.tbPos.y + nOffsetY)
      nCount = nCount + 1
    end
  end
end

function tbWnd:GetPoint(nIndex, tbPoint, szName, posX, posY)
  local tbNextPoint = self:GetNextAvailableObj(nIndex, tbPoint.szType)
  local tbPointTransform = self.pPanel:GetRectTransform(tbNextPoint.szKey)
  tbPointTransform.anchoredPosition = Ui.Vector2(posX, posY)
  local szPointName = szName
  if tbPoint.szTextColor and not Lib:IsEmptyStr(tbPoint.szTextColor) then
    szPointName = string.format("<color=#%s>%s</color>", tbPoint.szTextColor, szName)
  end
  self.pPanel:Label_SetText(tbNextPoint.szKey .. "/txtName", szPointName)
  if tbPoint.szResName then
    if self.tbUseResPath[tbPoint.szType] then
      self.pPanel:Sprite_SetSprite(tbNextPoint.szKey .. "/imgRes", tbPoint.szResName)
    elseif self.tbUseNpcModel[tbPoint.szType] then
      if not tbNextPoint.pNpcModel then
        local obj = self.pPanel:GetObject(tbNextPoint.szKey .. "/model")
        tbNextPoint.pNpcModel = obj:GetComponent(Ui.UINpcModel)
      end
      tbNextPoint.pNpcModel:ShowNpc(tonumber(tbPoint.szResName))
    end
  end
end

function tbWnd:GetNextAvailableObj(nIndex, szType)
  local tbPoint
  local szKey = szType .. nIndex
  if not self.tbUsedObjs[szKey] then
    tbPoint = {
      szObj = self.pPanel:CloneObject(szType, szKey),
      szKey = szKey
    }
    self.tbUsedObjs[szKey] = tbPoint
  else
    tbPoint = self.tbUsedObjs[szKey]
  end
  self.pPanel:SetActive(szKey, true)
  return tbPoint
end

function tbWnd:ResetMapPoints()
  for _, v in pairs(self.tbUsedObjs) do
    self.pPanel:SetActive(v.szKey, false)
    if v.pNpcModel then
      v.pNpcModel:Release()
    end
  end
end

function tbWnd:OnClose()
  self:ResetMapPoints()
end
