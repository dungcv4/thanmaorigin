SocialJieYi.Def = {
  nGroupKey = GetGroupKey("SocialJieYi"),
  FirstOpenMail = 1,
  dwSocialJieYiID = 2
}

function SocialJieYi:Log(...)
  if 1 ~= self.Def.Debug then
    return
  end
  local arg = {
    ...
  }
  local tbString = {}
  tbString[#tbString + 1] = string.format("[%s] [%d]", me and me.szName or "unknown", me and me.dwID or 0)
  for _, value in ipairs(arg) do
    local szType = type(value)
    if "table" == szType then
      tbString[#tbString + 1] = Lib:TinyTBToJson(value)
    elseif "number" == szType or "string" == szType or "boolean" == szType then
      tbString[#tbString + 1] = tostring(value)
    else
      tbString[#tbString + 1] = szType
    end
  end
  Log("[SocialJieYi]", table.concat(tbString, ", "))
end

function SocialJieYi:IsOpen()
  if 1 ~= GetTimeFrameState(self.Cfg.OPEN_TIME_FRAME) then
    return false
  end
  return true
end
