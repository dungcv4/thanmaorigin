function tbUi:OnCreate()
  self.tbSelectedTask = {}
  self.tbSelectedTask2TaskId = {}
  self.tbIdList = {}
  self.nGraduated = 0
  self.tbTaskScrollView = Ui:NewLoopScrollView(self, template.taskItemTemplate)
  self.tbStuScrollView = Ui:NewScrollView(self, template.StuTemplate, function(self, nIndex, tbElement)
    if self.nTeaStuIndex == nIndex + 2 then
      return
    end
    self.nTeaStuIndex = nIndex + 2
    self.bSelectTeaStu = true
    self.nSelectId = self.tbIdList[self.nTeaStuIndex]
    self:UpdateView()
  end, nil, true)
  self.PutongStuAwardsItem = Ui:NewExistItemGrid(self, "EmptyObj/TeacherRulesObj/PutongStuAwardsItem/UIItemGrid")
  self.JiechuStuAwardsItem = Ui:NewExistItemGrid(self, "EmptyObj/TeacherRulesObj/JiechuStuAwardsItem/UIItemGrid")
  self.PutongTeaAwardsItem = Ui:NewExistItemGrid(self, "EmptyObj/StudentRulesObj/PutongTeaAwardsItem/UIItemGrid")
  self.JiechuTeaAwardsItem = Ui:NewExistItemGrid(self, "EmptyObj/StudentRulesObj/JiechuTeaAwardsItem/UIItemGrid")
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnApplyList, {
    RedPointMgr.EVENT.MP_TeacherApply
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, self.btnReport, {
    RedPointMgr.EVENT.MP_btnTaskReport,
    RedPointMgr.EVENT.MP_btnGoalReport
  })
end

function tbUi:OnOpen(nSelectId)
  self.nSelectId = nSelectId
  self.nGraduated = 0
  self.nTeaStuIndex = 0
  self.tbIdList = {}
  self.bCompleted = false
  self:UpdateView()
end

function tbUi:UpdateView()
  local tbTeaStuData = self:UpdateLeftList()
  self:UpdateGuideInfo(not tbTeaStuData)
  if tbTeaStuData then
    self.bSelectTeaStu = false
    if self.nSelectId then
      for key, value in pairs(tbTeaStuData) do
        if self.nSelectId and self.nSelectId == value.dwId then
          self.nTeaStuIndex = key
          self.bSelectTeaStu = true
          break
        end
      end
    end
    if not self.bSelectTeaStu then
      for key, value in pairs(tbTeaStuData) do
        self.nTeaStuIndex = key
        break
      end
    end
    self.tbLastTeacherData = tbTeaStuData[self.nTeaStuIndex]
    self:UpdateTeacherInfo(self.tbLastTeacherData)
  end
  for i = 1, 2 do
    if self.nTeaStuIndex == i then
      self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/PlayerInfor/imgSelect", i), true)
    else
      self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/PlayerInfor/imgSelect", i), false)
    end
  end
  for i = 1, self.nGraduated + 2 do
    local nIdx = i + 2
    if self.nTeaStuIndex == nIdx then
      self.pPanel:SetActive(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d/PlayerInfor/imgSelect", i), true)
    else
      self.pPanel:SetActive(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d/PlayerInfor/imgSelect", i), false)
    end
    self.nGraduated = 0
  end
end

function tbUi:UpdateLeftList()
  local masterCtrl = MasterPrentice:GetMPInfo()
  local tbTeaStuData = {}
  local bUpdate = false
  local bEmpty = true
  if masterCtrl.tbMyMaster then
    for i = 1, 2 do
      if masterCtrl.tbMyMaster[i] then
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/btnTeaAdd%d", i, i), false)
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/PlayerInfor", i), true)
        local dwId = masterCtrl.tbMyMaster[i]
        local tbTempData = {dwId = dwId, bMaster = 1}
        tbTeaStuData[i] = tbTempData
        bUpdate = true
        bEmpty = false
        self.tbIdList[i] = dwId
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/PlayerInfor/imgRedPoint", i), false)
        self:updatePlayerInfo(string.format("TeaAndStuList/Player%d", i), tbTempData)
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/btnChat", i), true)
        Ui:Button_BindEvent(self.pPanel, string.format("TeaAndStuList/Player%d/btnChat", i), self.OnChatSomeone, self, dwId)
      else
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/btnTeaAdd%d", i, i), true)
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/PlayerInfor", i), false)
        self.pPanel:SetActive(string.format("TeaAndStuList/Player%d/btnChat", i), false)
      end
    end
  else
    self.pPanel:SetActive("TeaAndStuList/Player1/btnTeaAdd1", true)
    self.pPanel:SetActive("TeaAndStuList/Player2/btnTeaAdd2", true)
    self.pPanel:SetActive("TeaAndStuList/Player1/PlayerInfor", false)
    self.pPanel:SetActive("TeaAndStuList/Player2/PlayerInfor", false)
    self.pPanel:SetActive("TeaAndStuList/Player1/btnChat", false)
    self.pPanel:SetActive("TeaAndStuList/Player2/btnChat", false)
  end
  self.tbStuScrollView:Reset()
  if masterCtrl.tbMyPrentices then
    if masterCtrl.tbMyPrentices[1] then
      for i = 1, #masterCtrl.tbMyPrentices[1] do
        local dwId = masterCtrl.tbMyPrentices[1][i]
        local tbTempData = {dwId = dwId, bMaster = 0}
        tbTeaStuData[2 + i] = tbTempData
        bUpdate = true
        bEmpty = false
        self.tbStuScrollView:SetElement(i, function(tbItem)
          self:updatePlayerInfo(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d", i), tbTempData, true)
          self.tbIdList[i + 2] = dwId
        end)
      end
      if 1 == #masterCtrl.tbMyPrentices[1] then
        self.tbStuScrollView:SetElement(2, function(tbItem)
          self:updatePlayerInfo(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d", 2), nil, true, true)
        end)
      elseif 0 == #masterCtrl.tbMyPrentices[1] then
        for i = 1, 2 do
          self.tbStuScrollView:SetElement(i, function(tbItem)
            self:updatePlayerInfo(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d", i), nil, true, true)
          end)
        end
      end
    end
    self.pPanel:Label_SetText(self.txtStudentNum, string.format("%d/%d", #masterCtrl.tbMyPrentices[1], MasterPrentice.Def.MAX_PRENTICE_COUNT))
    if masterCtrl.tbMyPrentices[2] then
      self.nGraduated = #masterCtrl.tbMyPrentices[2]
      for i = 1, #masterCtrl.tbMyPrentices[2] do
        local dwId = masterCtrl.tbMyPrentices[2][i]
        local tbTempData = {
          dwId = dwId,
          bMaster = 0,
          bGraduated = 1
        }
        local nElement = 2 + i
        tbTeaStuData[2 + nElement] = tbTempData
        bUpdate = true
        bEmpty = false
        self.tbStuScrollView:SetElement(nElement, function(tbItem)
          self:updatePlayerInfo(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d", nElement), tbTempData, true)
          self.tbIdList[nElement + 2] = dwId
        end)
      end
    end
  else
    for i = 1, 2 do
      self.tbStuScrollView:SetElement(i, function(tbItem)
        self:updatePlayerInfo(string.format("TeaAndStuList/StuScrollView/Viewport/paneBG/Element%d", i), nil, true, true)
      end)
    end
    self.pPanel:Label_SetText(self.txtStudentNum, string.format("0/%d", MasterPrentice.Def.MAX_PRENTICE_COUNT))
  end
  if bUpdate then
    self.tbTeaStuData = tbTeaStuData
  end
  if bEmpty then
    self.tbTeaStuData = nil
    self.nTeaStuIndex = 0
  end
  return self.tbTeaStuData
end

function tbUi:updatePlayerInfo(szPlayer, tbItem, bPrentice, bEmpty)
  self.pPanel:SetActive(szPlayer .. "/btnChat", false)
  if bPrentice and true == bEmpty then
    self.pPanel:SetActive(szPlayer .. "/btnStuAdd", true)
    Ui:Button_BindEvent(self.pPanel, szPlayer .. "/btnStuAdd", function()
      self:TryFindPlayer(1)
    end)
    self.pPanel:SetActive(szPlayer .. "/PlayerInfor", false)
    return
  else
    if bPrentice then
      self.pPanel:SetActive(szPlayer .. "/btnStuAdd", false)
      self.pPanel:SetActive(szPlayer .. "/PlayerInfor", true)
    end
    local tbInfo = FriendShip:GetFriendDataInfo(tbItem.dwId)
    local tbPlayerInfo = MasterPrentice._CacheTable[tbItem.dwId]
    local tbPortraitObj = Ui:NewExistPortraitObjPanel(self, szPlayer .. "/PlayerInfor/PortraitObj")
    if tbInfo then
      local tbInfo2 = {
        szName = tbInfo.szName,
        nPortrait = tbInfo.nPortrait,
        nPortraitFrame = tbInfo.nPortraitFrame
      }
      tbPortraitObj:SetRoleIcon(tbInfo2.nPortrait)
      tbPortraitObj:SetPortraitFrame(tbInfo2.nPortraitFrame)
      tbPortraitObj:SetFactionIcon(tbInfo.nFaction)
      tbPortraitObj:SetLevel(tbInfo.nLevel)
      self.pPanel:Label_SetText(szPlayer .. "/PlayerInfor/PlayerName/txtPlayerName", tbInfo2.szName)
      Ui:Button_BindEvent(self.pPanel, szPlayer .. "/PlayerInfor/PortraitObj", self.onClickShowFriendDetails, self, tbInfo)
      local szHonorName, nFontColor = HonorLevel:GetHonorNameByStarLevel(tbInfo.nHonorStarLevel, false)
      local szPath = szPlayer .. "/PlayerInfor/PlayerName/HonorTitle"
      local obj = self.pPanel:GetObject(szPath)
      local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
      pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), nFontColor)
    elseif tbPlayerInfo then
      local tbInfo2 = {
        szName = tbPlayerInfo.szName,
        nPortrait = tbPlayerInfo.nPortrait
      }
      tbPortraitObj:SetRoleIcon(tbInfo2.nPortrait)
      tbPortraitObj:SetFactionIcon(tbPlayerInfo.nFaction)
      tbPortraitObj:SetLevel(tbPlayerInfo.nLevel)
      self.pPanel:Label_SetText(szPlayer .. "/PlayerInfor/PlayerName/txtPlayerName", tbInfo2.szName)
      Ui:Button_BindEvent(self.pPanel, szPlayer .. "/PlayerInfor/PortraitObj", self.onClickShowFriendDetails, self, tbPlayerInfo)
      local szHonorName, nFontColor = HonorLevel:GetHonorNameByStarLevel(tbPlayerInfo.nHonorStarLevel, false)
      local szPath = szPlayer .. "/PlayerInfor/PlayerName/HonorTitle"
      local obj = self.pPanel:GetObject(szPath)
      local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
      pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), nFontColor)
    else
      self.pPanel:SetActive(szPlayer .. "/PlayerInfor/PlayerName/HonorTitle", false)
    end
    local bOnline = tbInfo and tbInfo.nState == Player.emPLAYER_STATE_NORMAL
    local szOnlineState
    if tbPlayerInfo then
      szOnlineState = self:GetOnlineState(bOnline, tbPlayerInfo.nLastOnlineTime)
    else
      szOnlineState = self:GetOnlineState(bOnline, GetTime())
    end
    if szOnlineState then
      self.pPanel:SetActive(szPlayer .. "/PlayerInfor/txtOutline", true)
      self.pPanel:SetActive(szPlayer .. "/PlayerInfor/txtOnline", false)
      self.pPanel:Label_SetText(szPlayer .. "/PlayerInfor/txtOutline", szOnlineState)
      self.pPanel:SetActive(szPlayer .. "/btnChat", true)
      Ui:Button_BindEvent(self.pPanel, szPlayer .. "/btnChat", self.OnChatSomeone, self, tbItem.dwId)
    else
      self.pPanel:SetActive(szPlayer .. "/PlayerInfor/txtOutline", false)
      self.pPanel:SetActive(szPlayer .. "/PlayerInfor/txtOnline", true)
      self.pPanel:SetActive(szPlayer .. "/btnChat", true)
      Ui:Button_BindEvent(self.pPanel, szPlayer .. "/btnChat", self.OnChatSomeone, self, tbItem.dwId)
    end
    self.pPanel:SetActive(szPlayer .. "/PlayerInfor/imgMarkCom", tbItem.bGraduated)
    tbPortraitObj:SetOutlineObjActive(tbInfo and tbInfo.nState ~= Player.emPLAYER_STATE_NORMAL)
    tbPortraitObj:SetOutlineText(Lib:IIF(tbInfo and tbInfo.nState == Player.emPLAYER_STATE_ZONE, i18n.EasyGet("\232\183\168\230\156\141"), i18n.EasyGet("\231\166\187\231\186\191")))
    self.pPanel:SetActive(szPlayer .. "/PlayerInfor", true)
    local tbMPData = MasterPrentice:GetMPData(tbItem.dwId, me.dwID)
    if tbMPData and tbInfo then
      if tbMPData.status == MasterPrentice.Def.STATUS_AFTER_APPENTICE then
        if MasterPrentice.tbTaskComlated and MasterPrentice.tbTaskComlated[tbItem.dwId] and 2 == tbInfo.nState then
          self.pPanel:SetActive(szPlayer .. "/PlayerInfor/imgRedPoint", true)
        end
      elseif tbMPData.status == MasterPrentice.Def.STATUS_APPRENTICE and MasterPrentice.tbGoalComlated and MasterPrentice.tbGoalComlated[tbItem.dwId] and 2 == tbInfo.nState then
        self.pPanel:SetActive(szPlayer .. "/PlayerInfor/imgRedPoint", true)
      end
    end
  end
end

local function _GetReport(tbLastTeacherData, tbMPData)
  if tbMPData.status == MasterPrentice.Def.STATUS_APPRENTICE then
    if 1 == tbLastTeacherData.bMaster then
      return true, fnReportGoal, i18n.EasyGet("\230\177\135\230\138\165\228\187\187\229\138\161")
    else
      return false, nil, ""
    end
  elseif tbMPData.status == MasterPrentice.Def.STATUS_AFTER_APPENTICE then
    local taskCount, bReported = 0, tbMPData.bReportedTask
    for i, v in pairs(tbMPData.tbTaskList or {}) do
      taskCount = taskCount + 1
    end
    if 0 == tbLastTeacherData.bMaster then
      if 0 == taskCount then
        return true, fnSetTask, i18n.EasyGet("\229\143\145\229\184\131\228\187\187\229\138\161"), true
      end
    elseif 0 == taskCount then
      return true, fnAskTask, i18n.EasyGet("\231\148\179\232\175\183\228\187\187\229\138\161")
    elseif 1 ~= bReported then
      return true, fnReportTask, i18n.EasyGet("\228\184\138\228\186\164\228\187\187\229\138\161")
    end
  end
  return false
end

function tbUi:UpdateTaskInfo(tbLastTeacherData, tbMPData)
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MP_btnTaskReport)
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MP_btnGoalReport)
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.MP_TaskReport)
  self.tbSelectedTask = {}
  self.tbSelectedTask2TaskId = {}
  local taskType, tbTasks = 0, {}
  local bShowButton, szButton
  if tbMPData.status == MasterPrentice.Def.STATUS_APPRENTICE then
    tbTasks = MasterPrentice.MPTask.tbGoalSetting
    taskType = 1
    local tbInfo = FriendShip:GetFriendDataInfo(tbLastTeacherData.dwId)
    if MasterPrentice.tbGoalComlated and MasterPrentice.tbGoalComlated[tbLastTeacherData.dwId] and tbInfo and 2 == tbInfo.nState then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MP_btnGoalReport)
    end
  elseif tbMPData.status == MasterPrentice.Def.STATUS_AFTER_APPENTICE then
    tbTasks = MasterPrentice.MPTask.tbTaskSetting
    taskType = 2
    local tbInfo = FriendShip:GetFriendDataInfo(tbLastTeacherData.dwId)
    if MasterPrentice.tbTaskComlated and MasterPrentice.tbTaskComlated[tbLastTeacherData.dwId] and tbInfo and 2 == tbInfo.nState then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.MP_btnTaskReport)
    end
  else
    return
  end
  bShowButton, _, szButton = _GetReport(tbLastTeacherData, tbMPData)
  self.pPanel:Label_SetText("RightPlayerInfo/TaskListObj/btnReport/Text", szButton)
  self.pPanel:SetActive("RightPlayerInfo/TaskListObj/btnReport", bShowButton)
  self.pPanel:SetActive("RightPlayerInfo/TaskListObj/ListTitleObj/txtStuAwards", 1 == tbLastTeacherData.bMaster)
  self.pPanel:SetActive("RightPlayerInfo/TaskListObj/ListTitleObj/txtTeaAwards", 0 == tbLastTeacherData.bMaster)
  self.tbTaskScrollView:Clear()
  if tbMPData.tbTaskList and Lib:CountTB(tbMPData.tbTaskList) > 0 then
    self:updateWeekTaskInfo(tbMPData, tbTasks, taskType, tbLastTeacherData)
  else
    self.bCompleted = false
    local tbNewTasks = {}
    for nId, tbInfo in pairs(tbTasks) do
      local nStartTimeFrame = 1
      local nEndTimeFrame = 0
      if not Lib:IsEmptyStr(tbInfo.StartTimeFrame) then
        nStartTimeFrame = GetTimeFrameState(tbInfo.StartTimeFrame)
      end
      if not Lib:IsEmptyStr(tbInfo.EndTimeFrame) then
        nEndTimeFrame = GetTimeFrameState(tbInfo.EndTimeFrame)
      end
      if 1 == nStartTimeFrame and 0 == nEndTimeFrame then
        table.insert(tbNewTasks, tbInfo)
        table.insert(self.tbSelectedTask2TaskId, tbInfo.ID)
      end
    end
    self:updateGoalInfo(tbMPData, tbNewTasks, taskType, tbLastTeacherData)
  end
end

function tbUi:updateGoalInfo(tbMPData, tbTasks, taskType, tbLastTeacherData)
  local bApprentice, bCanChuanGong
  if tbMPData.status == MasterPrentice.Def.STATUS_APPRENTICE then
    local nLastChuangong = tbMPData.nLastChuangongDate or 0
    bCanChuanGong = nLastChuangong >= Lib:GetLocalDay(GetTime() - 14400) and 0 or 1
    bApprentice = true
  end
  
  local function setItemFunc(tbLoopScroll, nIndex, this)
    local tbElement = tbLoopScroll:GetElement(nIndex)
    if bApprentice then
      if 1 == nIndex then
        tbElement.pPanel:Label_SetText("txtTask", i18n.EasyGet("\229\184\136\229\190\146\228\188\160\229\138\159"))
        tbElement.pPanel:SetActive("txtProgress", true)
        tbElement.pPanel:Label_SetText("txtProgress", string.format("%d/1", 1 == bCanChuanGong and 0 or 1))
        tbElement.pPanel:SetActive("btnChuangong", true)
        tbElement.pPanel:SetActive("TaskCheckBox", false)
        tbElement.pPanel:SetActive("txtAwards", false)
        tbElement.pPanel:SetActive("txtCompleted", false)
        tbElement.pPanel:SetActive("txtSubmission", false)
        tbElement.pPanel:SetActive("txtCompletedAgo", false)
        local dwRoleId, bMaster = self.tbLastTeacherData.dwId, self.tbLastTeacherData.bMaster
        local Relation = MasterPrentice:GetRelation(me, dwRoleId)
        if 1 == bCanChuanGong then
          tbElement.pPanel:Button_SetEnable("btnChuangong", true)
          Ui:Button_BindEvent(tbElement.pPanel, "btnChuangong", self.OnClickChuanGong, self)
          if Relation == MasterPrentice.Def.Relation_Master then
            tbElement.pPanel:Sprite_SetSprite("btnChuangong", tbChuanGongIcon.szRecieveIconPath)
          elseif Relation == MasterPrentice.Def.Relation_Prentice then
            tbElement.pPanel:Sprite_SetSprite("btnChuangong", tbChuanGongIcon.szSendIconPath)
          end
        else
          Ui:Button_UnBindEvent(tbElement.pPanel, "btnChuangong")
          if Relation == MasterPrentice.Def.Relation_Master then
            tbElement.pPanel:Sprite_SetSprite("btnChuangong", tbChuanGongIcon.szDisableRecieveIconPath)
          elseif Relation == MasterPrentice.Def.Relation_Prentice then
            tbElement.pPanel:Sprite_SetSprite("btnChuangong", tbChuanGongIcon.szDisableSendIconPath)
          end
        end
        return
      end
      nIndex = nIndex - 1
    end
    local tbInfo = tbTasks[nIndex]
    if 1 == tbLastTeacherData.bGraduated then
      tbElement.pPanel:SetActive("TaskCheckBox", true)
      Ui:Toggle_BindEvent(tbElement.pPanel, "TaskCheckBox", function(self, isOn)
        if true == isOn then
          self.tbSelectedTask[nIndex] = 1
        else
          self.tbSelectedTask[nIndex] = nil
        end
        local nNowTaskNum = 0
        for k, v in pairs(self.tbSelectedTask) do
          if v then
            nNowTaskNum = nNowTaskNum + 1
          end
        end
        if nNowTaskNum > MasterPrentice.Cfg.WEEK_TASK_COUNT then
          me.CenterMsg(i18n.EasyGet("\230\175\143\229\145\168\230\156\128\229\164\154\229\143\170\232\131\189\229\184\131\231\189\174{0}\228\184\170\228\187\187\229\138\161\239\188\129", MasterPrentice.Cfg.WEEK_TASK_COUNT))
          tbElement.pPanel:Toggle_SetChecked("TaskCheckBox", false)
          return
        end
        if nNowTaskNum > 0 then
          if nNowTaskNum > MasterPrentice.Cfg.WEEK_TASK_COUNT then
            nNowTaskNum = MasterPrentice.Cfg.WEEK_TASK_COUNT
          end
          self.pPanel:SetActive("RightPlayerInfo/txtDismissTime", true)
          self.pPanel:Label_SetText("RightPlayerInfo/txtDismissTime", i18n.EasyGet("\229\183\178\233\128\137\230\139\169\239\188\136{0}/{1}\239\188\137", nNowTaskNum, MasterPrentice.Cfg.WEEK_TASK_COUNT))
        else
          self.pPanel:SetActive("RightPlayerInfo/txtDismissTime", false)
        end
      end, this)
      if not self.tbSelectedTask[nIndex] then
        tbElement.pPanel:Toggle_SetChecked("TaskCheckBox", false)
      end
      Ui:Button_BindEvent(tbElement.pPanel, "", this.OnTaskItemClick, this, nIndex, tbElement)
    else
      tbElement.pPanel:SetActive("TaskCheckBox", false)
    end
    this:SetTaskListData(tbElement, tbInfo, taskType, tbMPData, this, tbLastTeacherData.bMaster, this.tbSelectedTask[nIndex])
  end
  
  self.tbTaskScrollView:Clear()
  local nCount = Lib:CountTB(tbTasks) + (bApprentice and 1 or 0)
  self.tbTaskScrollView:SetElement(nCount, setItemFunc)
  self.tbTaskScrollView:GoTop()
end

function tbUi:SetTaskListData(tbItem, tbTask, taskType, tbMPData, pParent, bMaster, bSelectable)

  if 1 == taskType then
    nFinish, nCount = MasterPrentice.MPTask:GetGoalData(tbTask, tbMPData)
  else
    nFinish, nCount = MasterPrentice.MPTask:GetTaskData(tbTask, tbMPData)
  end
  local panel = tbItem.pPanel
  self.pParent = pParent
  panel:Label_SetText("txtTask", i18n.Parse(tbTask.Title))
  panel:SetActive("btnChuangong", false)
  panel:SetActive("txtProgress", false)
  panel:SetActive("txtCompleted", false)
  panel:SetActive("txtSubmission", false)
  panel:SetActive("txtCompletedAgo", false)
  panel:SetActive("txtAwards", true)
  if TaskStatus.Expired == nFinish then
    panel:SetActive("txtCompletedAgo", true)
  elseif TaskStatus.Reported == nFinish then
    panel:SetActive("txtSubmission", true)
  elseif TaskStatus.Finished == nFinish then
    panel:SetActive("txtCompleted", true)
    self.bCompleted = true
  else
    panel:SetActive("txtProgress", true)
    panel:Label_SetText("txtProgress", nCount .. "/" .. tbTask.Count)
  end
  local nAward
  if 1 ~= bMaster then
    if 1 == taskType then
      nAward = tbTask.MasterMoney
    else
      nAward = MasterPrentice.Cfg.MASTER_MONEY_OF_TASK
    end
  elseif 1 == taskType then
    nAward = tbTask.PrenticeExp * me.GetBaseAwardExp()
  else
    nAward = MasterPrentice:GetPrenticeExpByLevel(me.nLevel)
  end
  panel:Label_SetText("txtAwards", nAward)
  if bSelectable then
    panel:Toggle_SetChecked("TaskCheckBox", true)
  else
    panel:Toggle_SetChecked("TaskCheckBox", false)
  end
end

function tbUi:GetOnlineState(bOnline, nLastOnlineTime)
  if bOnline then
    return nil
  end
  local nToday = Lib:GetLocalDay()
  local nLoginDay = Lib:GetLocalDay(nLastOnlineTime)
  if nToday == nLoginDay then
    return i18n.EasyGet("\228\187\138\229\164\169")
  elseif nToday == nLoginDay + 1 then
    return i18n.EasyGet("\230\152\168\229\164\169")
  elseif nToday > nLoginDay + 1 and nToday < nLoginDay + 7 then
    return i18n.EasyGet("2\229\164\169\229\137\141")
  else
    return i18n.EasyGet("7\229\164\169\229\137\141")
  end
end

function tbUi:TryFindPlayer(bMaster)
  local bOpen, szMsg = MasterPrentice:IsOpen()
  if not bOpen then
    me.CenterMsg(szMsg)
    return
  end
  local tbLimitLevel = MasterPrentice.Cfg.MASTER_LEVEL
  local nLimitLevel = 0
  for szOpenTime, nLevel in pairs(tbLimitLevel) do
    if 1 == GetTimeFrameState(szOpenTime) then
      nLimitLevel = math.max(nLimitLevel, nLevel)
    end
  end
  if 1 == bMaster and nLimitLevel > me.nLevel then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\229\164\159{0}\231\186\167,\228\184\141\232\131\189\230\148\182\229\190\146", nLimitLevel))
    return
  elseif 1 ~= bMaster and me.nLevel < MasterPrentice.Cfg.PRENTICE_LEVEL then
    me.CenterMsg(i18n.EasyGet("\231\173\137\231\186\167\228\184\141\229\164\159{0}\231\186\167,\228\184\141\232\131\189\230\139\156\229\184\136", MasterPrentice.Cfg.PRENTICE_LEVEL))
    return
  end
  local prenticeCount = 0
  if prenticeCount >= 2 then
    me.CenterMsg(i18n.EasyGet("\230\156\170\229\135\186\229\184\136\229\188\159\229\173\144\230\149\176\229\183\178\231\187\143\232\190\190\229\136\176\228\184\138\233\153\144"))
    return
  end
  Ui:OpenWindow("UITeacherFind", bMaster)
end

function tbUi:GetMPData(tbTeacherData)
  local tbMPData
  if 1 == tbTeacherData.bMaster then
    tbMPData = MasterPrentice:GetMPData(tbTeacherData.dwId, me.dwID)
  else
    tbMPData = MasterPrentice:GetMPData(me.dwID, tbTeacherData.dwId)
  end
  if not tbMPData then
    tbMPData = {}
    MasterPrentice:CheckMPInfo()
  end
  return tbMPData
end

    end

end
