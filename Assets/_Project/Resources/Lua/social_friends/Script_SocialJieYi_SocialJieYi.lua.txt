function SocialJieYi:OnSyncData(dwJieYiID, tbJieYiData)
  self._cacheData = self._cacheData or {}
  
  if not dwJieYiID or not tbJieYiData then
    self._cacheData[me.dwID] = nil
    EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_JIEYI_DATA)
    return
  end
  self._cacheData[me.dwID] = {dwJieYiID = dwJieYiID, tbJieYiData = tbJieYiData}
  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_JIEYI_DATA)
  SocialJieYi:Log("OnSyncData", self._cacheData[me.dwID])
end

function SocialJieYi:GetJieYiID()
  if not self._cacheData or not self._cacheData[me.dwID] then
    return nil
  end
  return self._cacheData[me.dwID].dwJieYiID
end

function SocialJieYi:GetJieYiData()
  if not self._cacheData or not self._cacheData[me.dwID] then
    return nil
  end
  return self._cacheData[me.dwID].tbJieYiData
end

function SocialJieYi:IsMyJieYiBrother(dwRoleId2)
  local tbJieYiData = self:GetJieYiData()
  if not tbJieYiData then
    return false
  end
  if not tbJieYiData.tbBrothers[dwRoleId2] then
    return false
  end
  return true
end

function SocialJieYi:ApplyKickAgree(dwRoleId, bAgree)

  if not tbJieYiData then
    return
  end
  local tbBrotherData = tbJieYiData.tbBrothers[dwRoleId]
  if not tbBrotherData then
    return
  end
  local nSustainTime = GetTime() - tbBrotherData.nJoinTime
  if nSustainTime < SocialJieYi.Cfg.JIEYI_QUIT_COOLING_TIME then
    return
  end
  local tbInfo = FriendShip:GetFriendDataInfo(dwRoleId)
  if not tbInfo then
    return
  end
  local tbKickInfo = tbBrotherData.tbKickInfo
  if not tbKickInfo or 0 ~= tbKickInfo.tbResult[me.dwID] then
    return
  end
  RemoteServer.OnSocialJieYiRequest("OnKickAgree", dwRoleId, bAgree)
end

function SocialJieYi:GotoNpcAndTap()
  TeamMgr:StopFollowMember()
  AFK:Stop()
  local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(SocialJieYi.Cfg.JIEYI_NPC_TEMPLATE_ID, SocialJieYi.Cfg.JIEYI_NPC_MAP_ID, true)
  
  local function fnCallback()
    local function __timer_Callback()
      local nNpcId = KNpc.GetNpcIdByTemplateId(SocialJieYi.Cfg.JIEYI_NPC_TEMPLATE_ID)
      
      if nNpcId then
        Operation:SimpleTap(nNpcId)
      end
    end
    
    Timer:Register(5, __timer_Callback)
  end
  
  AutoPath:GotoAndCallEx(SocialJieYi.Cfg.JIEYI_NPC_MAP_ID, nPosX, nPosY, fnCallback, nWalkNearLength, nil, SocialJieYi.Cfg.JIEYI_NPC_TEMPLATE_ID)
  me.CenterMsg(i18n.EasyGet("\230\173\163\229\156\168\229\137\141\229\190\128\231\187\147\228\185\137\229\143\184\228\187\170<color=#ffff0e>{0}</color>", KNpc.GetNameByTemplateId(SocialJieYi.Cfg.JIEYI_NPC_TEMPLATE_ID)))
end

function SocialJieYi:GetBuffSelfMaxLevel()
  local tbJieYiData = self:GetJieYiData()
  if not tbJieYiData then
    return
  end
  local nImityValue = 0
  for dwRoleId, _ in pairs(tbJieYiData.tbBrothers) do
    if me.dwID ~= dwRoleId then
      local nImityLevel = FriendShip:GetFriendImityLevel(me.dwID, dwRoleId) or 0
      nImityLevel = math.min(nImityLevel, SocialJieYi.Cfg.FRIEND_IMITITY_LEVEL_MAX)
      SocialJieYi:Log("nImityLevel", dwRoleId, nImityLevel)
      nImityValue = nImityValue + nImityLevel
    end
  end
  nImityValue = math.floor(nImityValue / 10)
  nImityValue = math.min(nImityValue, SocialJieYi.Cfg.JIEYI_BUFF_SKILL_MAX_LEVEL)
  nImityValue = math.max(nImityValue, 1)
  return nImityValue
end
