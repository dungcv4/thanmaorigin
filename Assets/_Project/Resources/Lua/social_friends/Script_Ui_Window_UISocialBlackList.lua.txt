local tbWnd = Ui:CreateWindow("UISocialBlackList")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNoTIFY_SYNC_FRIEND_DATA,
      self.GenerateList
    }
  }
end

tbWnd.tbControls = {
  btnClearAll = "node/imgBG/Content/btnClearAll",
  BlackTemplate = "node/imgBG/Content/imgBlackList/BlackList/BlackTemplate",
  btnClose = "node/imgBG/btnClose",
  NotInfo = "node/imgBG/Content/NotInfo"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnClearAll = function(self)
    self:RemoveAll()
  end
}

function tbWnd:OnCreate()
  self.tbBlackListScrollView = Ui:NewScrollView(self, self.BlackTemplate)
end

function tbWnd:OnOpen()
  FriendShip:RequestUpdateBlackInfo()
  self:GenerateList()
end

function tbWnd:GenerateList()
  self.tbBlackListScrollView:Clear()
  self.tbBlackList = FriendShip:GetBlackList()
  local nBlackNum = #self.tbBlackList
  self.pPanel:SetActive(self.NotInfo, 0 == nBlackNum)
  self.pPanel:SetActive(self.btnClearAll, nBlackNum > 0)
  for i = 1, nBlackNum do
    self.tbBlackListScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
      tbPortraitObjPanel:SetRoleIcon(self.tbBlackList[i].nPortrait)
      tbPortraitObjPanel:SetPortraitFrame(self.tbBlackList[i].nPortraitFrame)
      tbPortraitObjPanel:SetFactionIcon(self.tbBlackList[i].nFaction)
      tbPortraitObjPanel:SetLevel(self.tbBlackList[i].nLevel)
      pPanel:Label_SetText("txtName", self.tbBlackList[i].szName)
      pPanel:Label_SetText("txtPlayerTitle", "")
      Ui:Button_BindEvent(pPanel, "btnRemove", self.Remove, self, i, self.tbBlackList[i].dwID)
    end)
  end
end

function tbWnd:Remove(nIndex, dwPlayerID)
  local function fnDelBlack()
    FriendShip:OnDelBlack(dwPlayerID)
    
    self:GenerateList()
  end
  
  local szTip = i18n.EasyGet("\231\161\174\229\174\154\229\176\134<color=#ff0000>{0}</color>\228\187\142\233\187\145\229\144\141\229\141\149\231\167\187\233\153\164\229\144\151\239\188\159", self.tbBlackList[nIndex].szName)
  me.MsgBox(szTip, {
    {
      i18n.szOk,
      function()
        fnDelBlack()
      end
    },
    {
      i18n.szCancel
    }
  })
end

function tbWnd:RemoveAll()
  local function fnClearAll()
    FriendShip:DelAllBlack()
    
    self:GenerateList()
  end
  
  local szTip = i18n.EasyGet("\230\152\175\229\144\166\230\184\133\231\169\186\233\187\145\229\144\141\229\141\149")
  me.MsgBox(szTip, {
    {
      i18n.szOk,
      function()
        fnClearAll()
      end
    },
    {
      i18n.szCancel
    }
  })
end
