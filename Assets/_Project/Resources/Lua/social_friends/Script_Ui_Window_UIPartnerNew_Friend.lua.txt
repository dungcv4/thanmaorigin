function tbWnd:OnCreateFriend()
  self.tbFriendScrollView = Ui:NewScrollView(self, tbControls.goFriendListTemplate, function(self, nIndex, tbElement)
    local nFetterId = self.tbFetterIndex2Id[nIndex]
    self.nCurFetterId = nFetterId
    self:SetFriendMoreInfo(nFetterId, tbElement)
  end, false, true)
  self.tbAttribListScrollView = Ui:NewScrollView(self, tbControls.goAttribListTemplate, nil, false, true)
  self.nCurFetterId = 0
end

function tbWnd:OnCreateGallery()
  self.tbPartnerInfoList = {}
  for i = 1, #tbQualitySetting do
    self.tbPartnerInfoList[i] = Ui:NewScrollView(self, string.format(tbControls.PartnerGalleryTemplatePath, tbQualitySetting[i]), function(tbOwner, nIndex, element)
      Ui:OpenWindow("UIPartnerGalleryDetail", self.tbPartnerQualityInfoList[i][nIndex])
    end, false, true)
  end
end

function tbWnd:OnCreateDecompose()
  self.nMaxElements = 0
  self.tbDecomposePartnerView = Ui:NewScrollView(self, tbControls.partnerDecomposeList, function(tbOwner, nIndex, element)
    local nHaveCount = 0
    if self.tbSelectedPartnerList[self.tbPartnerList[nIndex].nTemplateId] ~= nil then
      nHaveCount = self.tbSelectedPartnerList[self.tbPartnerList[nIndex].nTemplateId]
    end
    Ui:OpenWindow("UIPartnerDecomposeSetting", self.tbPartnerList[nIndex].nTemplateId, self.tbPartnerList[nIndex].nCount, nHaveCount)
    element.pPanel:Button_SetEnable("", true)
  end, false, false)
  self.pPanel:Dropdown_BindEvent(self.dropDownQuality, function(value)
    self.nCurSelectQuality = value
    self:HideParterInfo()
    if self:CheckNowIsAllToggle() then
      self.pPanel:Toggle_SetChecked(self.togToggleAll, true)
    else
      self.pPanel:Toggle_SetChecked(self.togToggleAll, false)
    end
  end)
end

end

function tbWnd:SetActive(bActive, nPanel)

function tbWnd:OnOpenFriend()
  self.nRedPointIndexPos = -1
  self.nSelecttbElementIndex = 0
  self.tbFetterId2Index = {}
  self.tbFetterIndex2Id = {}
  self.tbSelecttbElement = nil
  self.tbAllMainPartner = Partner:GetAllPartnersDetailInfoForFetters()
  self.tbFriendScrollView:Clear()
  local nListIndex = 0
  for nFetterId = 1, #Partner.tbPartnerFetterSetting do
    if Partner:CheckFetterCanShow(nFetterId) then
      nListIndex = nListIndex + 1
      self.tbFetterId2Index[nFetterId] = nListIndex
      self.tbFetterIndex2Id[nListIndex] = nFetterId
      if 0 == self.nSelecttbElementIndex then
        self.nSelecttbElementIndex = nListIndex
      end
      self.tbFriendScrollView:SetElement(nListIndex, function(tbElement)
        self:SetFriendInfo(tbElement, nFetterId)
      end)
    end
  end
  if self.nRedPointIndexPos > 0 then
    self.nSelecttbElementIndex = self.nRedPointIndexPos
  end
  self.tbFriendScrollView:ScrollToIndex(self.nSelecttbElementIndex, true)
end

function tbWnd:SetFriendMoreInfo(nIndex, tbElement)
  self.nSelecttbElementIndex = self.tbFetterId2Index[nIndex]
  if self.tbSelecttbElement then
    self.tbSelecttbElement.pPanel:SetActive("imgSelected", false)
  end
  self.tbSelecttbElement = tbElement
  tbElement.pPanel:SetActive("imgSelected", true)
  local tbPartnerList = {}
  for i = 1, 4 do
    local nPartnerId = 0
    if 1 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nIndex].nPartnerId1
    elseif 2 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nIndex].nPartnerId2
    elseif 3 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nIndex].nPartnerId3
    elseif 4 == i then
      nPartnerId = Partner.tbPartnerFetterSetting[nIndex].nPartnerId4
    end
    if nPartnerId <= 0 then
      self.pPanel:SetActive(string.format(tbControls.goPartner, i), false)
    else
      self.pPanel:SetActive(string.format(tbControls.goPartner, i), true)
      local szPartnerName = Partner.tbAllPartnerTemplateInfo[nPartnerId].szName
      self.pPanel:Label_SetText(string.format(tbControls.txtPartnerName, i), i18n.Parse(szPartnerName))
    end
    for j = 1, 3 do
      self.pPanel:SetActive(string.format(tbControls.goPartner .. tbImgQualityBgEffectList[j], i), false)
    end
    if nPartnerId > 0 then
      local pCurPartner = self:GetFetterPartner(nPartnerId)
      if pCurPartner then
        table.insert(tbPartnerList, pCurPartner)
        self.pPanel:SetActive(string.format(tbControls.imgPartnerNameBg, i), true)
        local nQualityLevel = Partner.tbAllPartnerInfo[pCurPartner.nTemplateId].nQualityLevel
        self.pPanel:Sprite_SetSprite(string.format(tbControls.imgPartnerNameBg, i), Partner.tbPartnerNameBg[nQualityLevel])
        if nQualityLevel <= 3 then
          self.pPanel:SetActive(string.format(tbControls.goPartner .. tbImgQualityBgEffectList[nQualityLevel], i), true)
        end
      else
        self.pPanel:SetActive(string.format(tbControls.imgPartnerNameBg, i), false)
      end
    end
  end
  local nFriendLevel = 0
  nFriendLevel = self:CalcFriendLevel(tbPartnerList, nIndex)
  nFriendLevel = Partner:GetPartnerFetterLevel(me, self.nCurFetterId)
  self.pPanel:Label_SetText(tbControls.txtInfoTitle, i18n.EasyGet("\228\190\160\231\188\152{0}\231\186\167", nFriendLevel))
  if 0 == nFriendLevel then
    self.pPanel:Label_SetText(tbControls.txtLevelUp, i18n.EasyGet("\230\191\128\230\180\187"))
  else
    tbElement.pPanel:SetActive("Activate", true)
    self.pPanel:Label_SetText(tbControls.txtLevelUp, i18n.EasyGet("\229\141\135\231\186\167"))
  end
  local bFetterFull, bFetterMaxLevel = Partner:CheckPartnerFetterFull(me, self.nCurFetterId)
  if not bFetterFull then
    self.pPanel:SetActive(tbControls.btnLevelUp, false)
  elseif bFetterMaxLevel then
    self.pPanel:SetActive(tbControls.btnLevelUp, false)
  else
    self.pPanel:SetActive(tbControls.btnLevelUp, true)
  end
  self:CheckRedPoint(nIndex)
end

function tbWnd:SetFriendInfo(tbElement, nIndex)
  tbElement.pPanel:SetActive("Effect_Activate", false)
  local tbOneFriendList = Partner.tbPartnerFetterSetting[nIndex]
  local nPartnerNum = 0
  local nPartnerWeHave = 0
  local tbPartnerList = {}
  for i = 1, 4 do
    local nPartnerNpcTemplateId = 0
    nPartnerNpcTemplateId = Partner.tbPartnerFetterSetting[tbOneFriendList.nId][string.format("nPartnerId%d", i)]
    local pCurPartner = self:GetFetterPartner(nPartnerNpcTemplateId)
    if pCurPartner then
      table.insert(tbPartnerList, pCurPartner)
    end
    if nPartnerNpcTemplateId > 0 then
      nPartnerNum = nPartnerNum + 1
      if self.tbAllMainPartner[nPartnerNpcTemplateId] then
        nPartnerWeHave = nPartnerWeHave + 1
      end
    end
  end
  self.nCurFetterId = nIndex
  local nFriendLevel = self:CalcFriendLevel(tbPartnerList, nIndex)
  if nFriendLevel > 0 then
    tbElement.pPanel:SetActive("Activate", true)
    tbElement.pPanel:Label_SetText("txtDecomposeLevel", i18n.EasyGet("{0}\231\186\167", nFriendLevel))
  else
    tbElement.pPanel:Label_SetText("txtDecomposeLevel", i18n.EasyGet("\230\148\182\233\155\134\232\191\155\229\186\166 {0}/{1}", nPartnerWeHave, nPartnerNum))
  end
  tbElement.pPanel:Label_SetText("txtDecomposeName", i18n.Parse(tbOneFriendList.szName))
  local nPartner1Id = Partner.tbPartnerFetterSetting[nIndex].nPartnerId1
  local nQualityLevel = Partner.tbAllPartnerTemplateInfo[nPartner1Id].nQualityLevel
  tbElement.pPanel:Sprite_SetSprite("imgQualityBg", tbImgListQualityBgList[nQualityLevel])
  self:CheckRedPoint(nIndex)
end

    return
  end
  if bFetterMaxLevel then
    me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\229\144\140\228\188\180\230\136\152\229\138\155\229\128\188\228\184\141\232\182\179\228\187\165\229\141\135\231\186\167\232\175\165\228\190\160\231\188\152\229\177\158\230\128\167"))

function tbWnd:GetFetterPartner(nNpcPartnerId)
  local pCurPartner
  for nCount, pPartner in pairs(self.tbAllMainPartner) do
    local nNpcTemplateId = Partner.tbAllPartnerInfo[pPartner.nTemplateId].nNpcTemplateId
    if nNpcTemplateId == nNpcPartnerId then
      pCurPartner = pPartner
      break
    end
  end
  return pCurPartner
end

function tbWnd:CalcFriendLevel(tbPartnerList, nFriendCount)
  local nFriendLevel = 0
  self.tbAttribListScrollView:Clear()
  local nFriendCountEnoughFightPower = 0
  local nFriendPartnerCount = 0
  if 0 ~= Partner.tbPartnerFetterSetting[nFriendCount].nPartnerId5 then
    nFriendPartnerCount = 5
  elseif 0 ~= Partner.tbPartnerFetterSetting[nFriendCount].nPartnerId4 then
    nFriendPartnerCount = 4
  elseif 0 ~= Partner.tbPartnerFetterSetting[nFriendCount].nPartnerId3 then
    nFriendPartnerCount = 3
  elseif 0 ~= Partner.tbPartnerFetterSetting[nFriendCount].nPartnerId2 then
    nFriendPartnerCount = 2
  else
    nFriendPartnerCount = 1
  end
  local nRequireFightPower = Partner.tbPartnerFetterSetting[nFriendCount].nRequireFightPower
  nFriendLevel = Partner:GetPartnerFetterLevel(me, self.nCurFetterId)
  for _, pPartner in pairs(tbPartnerList) do
    local nFightPowerNeed = nRequireFightPower * nFriendLevel
    if nFightPowerNeed <= pPartner.nFightPower then
      nFriendCountEnoughFightPower = nFriendCountEnoughFightPower + 1
    end
  end
  local nImproveType1 = Partner.tbPartnerFetterSetting[nFriendCount].nImproveType1
  local nImproveType2 = Partner.tbPartnerFetterSetting[nFriendCount].nImproveType2
  local nImproveType3 = Partner.tbPartnerFetterSetting[nFriendCount].nImproveType3
  local nImproveValue1 = Lib:SplitStr2Num(Partner.tbPartnerFetterSetting[nFriendCount].nImproveValue1, "|")
  local nImproveValue2 = Lib:SplitStr2Num(Partner.tbPartnerFetterSetting[nFriendCount].nImproveValue2, "|")
  local nImproveValue3 = Lib:SplitStr2Num(Partner.tbPartnerFetterSetting[nFriendCount].nImproveValue3, "|")
  local bShowUpdateInfo = true
  if nImproveType1 and "" ~= nImproveType1 then
    local a, b = FightSkill:GetMagicDescSplit(nImproveType1, nImproveValue1)
    self.tbAttribListScrollView:SetElement(1, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttrib", a)
      tbElement.pPanel:Label_SetText("txtAttribNum", "+" .. tonumber(b) * nFriendLevel)
      if bShowUpdateInfo then
        tbElement.pPanel:SetActive("imgArrows", true)
        tbElement.pPanel:SetActive("txtNum", true)
        tbElement.pPanel:Label_SetText("txtNum", b)
      else
        tbElement.pPanel:SetActive("imgArrows", false)
        tbElement.pPanel:SetActive("txtNum", false)
      end
    end)
  end
  if nImproveType2 and "" ~= nImproveType2 then
    local a, b = FightSkill:GetMagicDescSplit(nImproveType2, nImproveValue2)
    self.tbAttribListScrollView:SetElement(2, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttrib", a)
      tbElement.pPanel:Label_SetText("txtAttribNum", "+" .. tonumber(b) * nFriendLevel)
      if bShowUpdateInfo then
        tbElement.pPanel:SetActive("imgArrows", true)
        tbElement.pPanel:SetActive("txtNum", true)
        tbElement.pPanel:Label_SetText("txtNum", b)
      else
        tbElement.pPanel:SetActive("imgArrows", false)
        tbElement.pPanel:SetActive("txtNum", false)
      end
    end)
  end
  if nImproveType3 and "" ~= nImproveType3 then
    local a, b = FightSkill:GetMagicDescSplit(nImproveType3, nImproveValue3)
    self.tbAttribListScrollView:SetElement(3, function(tbElement)
      tbElement.pPanel:Label_SetText("txtAttrib", a)
      tbElement.pPanel:Label_SetText("txtAttribNum", "+" .. tonumber(b) * nFriendLevel)
      if bShowUpdateInfo then
        tbElement.pPanel:SetActive("imgArrows", true)
        tbElement.pPanel:SetActive("txtNum", true)
        tbElement.pPanel:Label_SetText("txtNum", b)
      else
        tbElement.pPanel:SetActive("imgArrows", false)
        tbElement.pPanel:SetActive("txtNum", false)
      end
    end)
  end
  self.pPanel:SetActive(tbControls.txtTips, true)
  if nFriendPartnerCount == #tbPartnerList and nFriendLevel >= 1 then
    self.pPanel:Label_SetText(tbControls.txtTips, i18n.EasyGet("\229\144\132\229\144\140\228\188\180\230\136\152\229\138\155\232\190\190\229\136\176{0}\229\143\175\229\141\135\231\186\167 ({1}/{2})", nRequireFightPower * nFriendLevel, nFriendCountEnoughFightPower, nFriendPartnerCount))
  else
    self.pPanel:Label_SetText(tbControls.txtTips, i18n.EasyGet("\233\155\134\233\189\144\230\137\128\230\156\137\229\144\140\228\188\180\229\143\175\230\191\128\230\180\187 ({0}/{1})", tostring(#tbPartnerList), nFriendPartnerCount))
  end
  return nFriendLevel
end

function tbWnd:GetPartnerQualityInfoList()
  local tbPartnerQualityInfoList = {
    [1] = {},
    [2] = {},
    [3] = {},
    [4] = {},
    [5] = {}
  }
  for nIndex, tbPartner in pairs(Partner.tbAllPartnerInfo) do
    local nQuality = tbPartner.nQualityLevel
    local nTemplateId = tbPartner.nId
    if nil ~= nTemplateId then
      local szTimeFrame = Partner.tbAllPartnerInfo[nTemplateId].szTimeFrame
      if nil ~= nQuality and ("" == szTimeFrame or 0 ~= GetTimeFrameState(szTimeFrame)) then
        table.insert(tbPartnerQualityInfoList[nQuality], nIndex)
      end
    end
  end
  return tbPartnerQualityInfoList
end

function tbWnd:GetOwnPartnerList()
  local tbOwnPartnerList = {}
  local tbAllPartner = me.GetAllPartner()
  for nPartnerId in pairs(tbAllPartner) do
    local pPartner = me.GetPartnerObj(nPartnerId)
    if Partner.tbAllPartnerInfo[pPartner.nTemplateId].szTimeFrame == "" or 0 ~= GetTimeFrameState(Partner.tbAllPartnerInfo[pPartner.nTemplateId].szTimeFrame) then
      tbOwnPartnerList[pPartner.nTemplateId] = true
    end
  end
  return tbOwnPartnerList
end

  for _, tbPartnerInfo in pairs(self.tbPartnerList) do
    local nQualityLevel = Partner.tbAllPartnerInfo[tbPartnerInfo.nTemplateId].nQualityLevel

      end

function tbWnd:DecomposeCurQualityOnly(tbSelectedPartnerList)

    return tbSelectedPartnerList
  end
  for nPartnerTemplateId, v in pairs(tbSelectedPartnerList) do
    if Partner.tbAllPartnerInfo[nPartnerTemplateId].nQualityLevel == self.nCurSelectQuality then
      tbDecompose[nPartnerTemplateId] = v
    end
  end
  return tbDecompose
end

function tbWnd:CheckNowIsAllToggle()
  local nNowQuality = self.nCurSelectQuality
  local tbAllPartner = self.tbPartnerList
  local tbAllSelectPartner = self.tbSelectedPartnerList
  if nNowQuality > 0 then
    local nPartnerCount = 0
    local nSelectCount = 0
    for nIndex, tbPartnerInfo in pairs(tbAllPartner) do
      local nPartnerQuality = Partner.tbAllPartnerInfo[tbPartnerInfo.nTemplateId].nQualityLevel
      if nPartnerQuality == nNowQuality then
        nPartnerCount = nPartnerCount + tbPartnerInfo.nCount
      end
    end
    for nTemplateId, nHaveCount in pairs(tbAllSelectPartner) do
      local nPartnerQuality = Partner.tbAllPartnerInfo[nTemplateId].nQualityLevel
      if nPartnerQuality == nNowQuality then
        nSelectCount = nSelectCount + nHaveCount
      end
    end
    if nPartnerCount == nSelectCount then
      return true
    else
      return false
    end
  else
    local nPartnerCount = 0
    local nSelectCount = 0
    for nIndex, tbPartnerInfo in pairs(tbAllPartner) do
      nPartnerCount = nPartnerCount + tbPartnerInfo.nCount
    end
    for nTemplateId, nHaveCount in pairs(tbAllSelectPartner) do
      nSelectCount = nSelectCount + nHaveCount
    end
    if nPartnerCount == nSelectCount then
      return true
    else
      return false
    end
  end
end
