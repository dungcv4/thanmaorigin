function FriendsReturn:GetDayCountByOfflineTime(pPlayer)
  if not pPlayer then
    return 0
  end
  local nLastOfflineTime = OfflineExp:LastOfflineTime(pPlayer)
  local nDayOffset = Lib:CalcDayOffset(14400, nLastOfflineTime)
  return nDayOffset
end

function FriendsReturn:CalcPlayerReturnerAwardTime(tbAwardData, tbVipData, nDayLeft, nVipLevel)
  if not tbAwardData or Lib:CountTB(tbAwardData) <= 0 then
    return 0
  end
  if not tbVipData or Lib:CountTB(tbVipData) <= 0 then
    return 0
  end
  local nTotalCount = math.min(tbAwardData[1], math.floor(nDayLeft / tbAwardData[2]) * tbAwardData[3])
  local nVipOffset = tbVipData[nVipLevel + 1] or tbVipData[#tbVipData]
  nTotalCount = math.floor(nTotalCount * nVipOffset)
  return nTotalCount
end

function FriendsReturn:CheckIfPlayerCanBeReturner(pPlayer)
  if not pPlayer then
    return false
  end
  local nPlayerLevel = pPlayer.nLevel
  local nCurTime = GetTime()
  local nLastOfflineTime = OfflineExp:LastOfflineTime(pPlayer)
  local nActiveCheck = pPlayer.GetUserValue(self.Def.nGroup, self.Def.KEY.nReturnerActiveCheck)
  if nPlayerLevel < self.Cfg.nNeedLevel then
    return false
  end
  if nLastOfflineTime <= 0 then
    return false
  end
  if nCurTime - nLastOfflineTime < self.Cfg.nOfflineTime then
    return false
  end
  if nCurTime - nActiveCheck < self.Cfg.nOfflineTime then
    return false
  end
  return true
end

function FriendsReturn:CheckIsPlayerReturner(pPlayer)
  if not pPlayer then
    return false
  end
  local nReturnerState = pPlayer.GetUserValue(self.Def.nGroup, self.Def.KEY.nReturnerState)
  if 0 == nReturnerState then
    return false, self.Def.tbReturnerState.notReturner, 0
  end
  local nNowTime = GetTime()
  local nReturnerStartTime = pPlayer.GetUserValue(self.Def.nGroup, self.Def.KEY.nReturnerStartTime)
  local nLeftTime = self.Cfg.nReturnerContinueTIme + nReturnerStartTime - nNowTime
  if nLeftTime <= 0 then
    if MODULE_GAMESERVER then
      self:ClearReturnerUserValue(pPlayer)
      self:ClearReturnerBuff(pPlayer)
    end
    return false, self.Def.tbReturnerState.isReturnerButOutOfTime, 0
  end
  return true, self.Def.tbReturnerState.isReturner, nLeftTime
end

function FriendsReturn:CheckIfPlayerCanRequestVNVipReturnAward(pPlayer)

function FriendsReturn:GetMaxLevel()
  local nCreateTime = GetServerCreateTime()
  local nCurOpenServerDay = Lib:GetLocalDay(nCreateTime)
  local nToday = Lib:GetLocalDay(GetTime() - 14400)
  local nCurMaxLevel = 0
  local nCurMaxZongShiLevel = 0
  for i, v in ipairs(Calendar.tbOpenLevelTimeFrame) do
    local nOpenDay = v.OpenDay + nCurOpenServerDay - 1
    if nToday < nOpenDay then
      break
    end
    nCurMaxLevel = v.NewMaxLevel
    nCurMaxZongShiLevel = v.NewMaxZongShiLevel
  end
  return nCurMaxLevel + nCurMaxZongShiLevel
end
