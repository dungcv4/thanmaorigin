function tbWnd:OnCreate()
  self.tbSwitchPanelButtonList = self.tbSwitchPanelButtonList or {
    tbControls.btnFriend,
    tbControls.btnTeacher,
    tbControls.btnJieYi,
    tbControls.btnEnemy,
    tbControls.btnSwordsmen
  }
  self.tbSocialKindGroup = Ui:NewButtonGroup(self, self.tbSwitchPanelButtonList, true, function(self, nIndex)
    self:UpdatePanel(nIndex)
  end)
  self.tbFriendScrollView = Ui:NewLoopScrollView(self, tbControls.FriendTemplate)
  self.tbEnemyScrollView = Ui:NewScrollView(self, tbControls.EnemyTemplate)
  self.pTeacher = Ui:CreatePanel("UISocial_TeacherPanel", tbControls.panelTeacher, self)
  self.tbJieYiList = Ui:NewLoopScrollView(self, tbControls.JieYiTemplate)
  self.tbJieYiAttrList = Ui:NewLoopScrollView(self, tbControls.AttrInfo_AttrTemplate)
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnTeacher, {
    RedPointMgr.EVENT.MP_TeacherApply,
    RedPointMgr.EVENT.MP_TaskReport,
    RedPointMgr.EVENT.MP_GoalReport
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnFriend, {
    RedPointMgr.EVENT.FriendApply
  })
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnFriendApply, {
    RedPointMgr.EVENT.FriendApply
  })
  self:SetListTitleTong()
  local obj = self.pPanel:GetObject(tbControls.MyNpcModel)
  self.MyModel = obj:GetComponent(Ui.UINpcModel)
  obj = self.pPanel:GetObject(tbControls.LoverModel)
  self.LoverModel = obj:GetComponent(Ui.UINpcModel)
  self.tbMarrySkillMagic = Ui:NewScrollView(self, tbControls.MarrySkillTemplate)
end

function tbWnd:OnOpen(nButtonIdx, nSelectId)
  self.nTeacherPanelUpdateCount = 0
  self.tbSocialKindGroup:Select(nButtonIdx or 1)
  self.nSelectId = nSelectId
end

function tbWnd:Refresh()
  if not self:Visible() then
    return
  end
  if self.nIndex and 2 ~= self.nIndex then
    tbWnd:UpdatePanel(self.nIndex)
  end
end

function tbWnd:OnClose()
  if self.nCdTimer then
    Timer:Close(self.nCdTimer)
    self.nCdTimer = nil
  end
  self.MyModel:Release()
  self.LoverModel:Release()
end

function tbWnd:UpdatePanel(nIndex)
  self.nIndex = nIndex
  for i = 1, #self.tbPanelIndex2Controls do
    if i == nIndex then
      self.pPanel:SetActive(self.tbPanelIndex2Controls[i], true)
    else
      self.pPanel:SetActive(self.tbPanelIndex2Controls[i], false)
    end
  end
  if 1 == nIndex then
    FriendShip:RequestUpdateFriendInfo()
    self:UpdateFriendPanel()
    self.szCurPanel = "panelFriend"
  elseif 2 == nIndex then
    Ui:GetPanelByName("UISocial_TeacherPanel"):OnOpen(self.nSelectId)
    self.szCurPanel = "panelTeacher"
  elseif 3 == nIndex then
    SocialJieYi:ReqJieYiData()
    self:UpdateJieYi()
    self.szCurPanel = "panelJieYi"
  elseif 4 == nIndex then
    FriendShip:RequestUpdateEnemyInfo()
    self:UpdateEnemyPanel()
    self.szCurPanel = "panelEnemy"
  elseif 5 == nIndex then
    self:UpdateMarry()
    self.szCurPanel = "panelSwordsmen"
  end
  self.pPanel:Label_SetText(tbControls.txtTitle, self.tbPanelIndex2Title[nIndex]())
  local btnHelpObj = self.pPanel:GetObject(tbControls.btnHelp)
  Ui:AddUIAnimScaleDown(btnHelpObj, 0.1, 0.7)
end

function tbWnd:UpdateFriendPanel()
  local tbFriendList = FriendShip:GetAllFriendData()
  
  local function funSort(tbA, tbB)
    if tbA.dwID == Marry.tbMarryData.dwLoverID then
      return true
    elseif tbB.dwID == Marry.tbMarryData.dwLoverID then
      return false
    end
    if tbA.nState == Player.emPLAYER_STATE_NORMAL and tbB.nState ~= Player.emPLAYER_STATE_NORMAL then
      return true
    end
    if tbA.nState ~= Player.emPLAYER_STATE_NORMAL and tbB.nState == Player.emPLAYER_STATE_NORMAL then
      return false
    end
    return tbA.nImity > tbB.nImity
  end
  
  table.sort(tbFriendList, funSort)
  local nFriendNum = #tbFriendList
  self.pPanel:SetActive(self.panelNoFriend, 0 == nFriendNum)
  self.pPanel:Label_SetText(self.txtFriendCount, string.format("%d/%d", nFriendNum, FriendShip:GetMaxFriendNum(me.nLevel, me.GetVipLevel())))

function tbWnd:UpdateMarrySkill()
  local nImitityLevel, nSkillLevel = Marry:CalImitityLevel(Marry.tbMarryData.nImitity)
  if nSkillLevel > 0 then
    local tbMagic = FightSkill:GetSkillMagicDescAndValue(Marry.Cfg.nBuffSkillID, nSkillLevel)
    local tbNextMagic = FightSkill:GetSkillMagicDescAndValue(Marry.Cfg.nBuffSkillID, nSkillLevel + 1)
    self.tbMarrySkillMagic:Reset()
    for i, tbInfo in ipairs(tbMagic) do
      self.tbMarrySkillMagic:SetElement(i, function(tbElement)
        tbElement.pPanel:Label_SetText("txtLife", tbInfo.szFinalName)
        tbElement.pPanel:Label_SetText("txtLifeNow", tbInfo.szFinalValue)
        tbElement.pPanel:Label_SetText("txtLifeReborn", tbNextMagic[i].szFinalValue)
      end)
    end
  end
  local tbStateEffect = FightSkill:GetStateEffectBySkill(Marry.Cfg.nBuffSkillID, nSkillLevel)
  if tbStateEffect then
    self.pPanel:Sprite_SetSprite(tbControls.IconBuff, tbStateEffect.Icon)
    self.pPanel:Sprite_SetSprite(tbControls.IconBuff1, tbStateEffect.Icon)
    self.pPanel:Sprite_SetSprite(tbControls.IconBuff2, tbStateEffect.Icon)
    self.pPanel:Label_SetText(tbControls.txtBuffName, i18n.Parse(tbStateEffect.StateName))
    self.pPanel:Label_SetText(tbControls.txtBuffName1, i18n.Parse(tbStateEffect.StateName))
    self.pPanel:Label_SetText(tbControls.txtBuffName2, i18n.Parse(tbStateEffect.StateName))
  end
end

function tbWnd:ShowLoverModel(nLevel)
  local tbFeature = {}
  local pAsyncRole = KPlayer.GetAsyncData(Marry.tbMarryData.dwLoverID)
  if not pAsyncRole then
    return
  end
  for nEquipPos, nIndex in pairs(Item.EQUIPPOS_TO_NPC_RES_PART) do
    local nResId
    local nFashionId = pAsyncRole.GetFashion(nIndex + 1)
    if nFashionId > 0 then
      local tbFashionSetting = Fashion:GetFashionSettingNotPlayer(nFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
      if nil ~= tbFashionSetting then
        nResId = tbFashionSetting.nResId
      end
    end
    if nil == nResId then
      local tbEquip, tbExtEquip, tbPartnerInfo = ViewRole:GetCurResInfo(Marry.tbMarryData.dwLoverID)
      if tbEquip then
        local nItemId = tbEquip[nEquipPos]
        if nItemId and nItemId > 0 then
          local pEquip = KItem.GetItemObj(nItemId)
          local _, _, _, _, nInnerResId = KItem.GetItemShowInfo(pEquip.dwTemplateId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
          nResId = nInnerResId
        end
      end
    end
    if nil ~= nResId then
      tbFeature[nIndex] = nResId
    else
      tbFeature[nIndex] = 0
    end
  end
  if tbFeature[Npc.NpcResPartsDef.npc_part_body] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_body] = 1
  end
  if tbFeature[Npc.NpcResPartsDef.npc_part_head] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_head] = 1
  end
  if nLevel then
    local nFashionId = Marry.Cfg.WeddingClothing[nLevel].nFashionId1
    local tbFashionSetting = Fashion:GetFashionSettingNotPlayer(nFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
    if tbFashionSetting then
      tbFeature[tbFashionSetting.nPart] = tbFashionSetting.nResId
    end
    nFashionId = Marry.Cfg.WeddingClothing[nLevel].nFashionId2
    tbFashionSetting = Fashion:GetFashionSettingNotPlayer(nFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
    if tbFashionSetting then
      tbFeature[tbFashionSetting.nPart] = tbFashionSetting.nResId
    end
  end
  local nLoverSex = 1 == me.nSex and 0 or 1
  self.LoverModel:ShowModel(nLoverSex, 1, tbFeature[Npc.NpcResPartsDef.npc_part_body], tbFeature[Npc.NpcResPartsDef.npc_part_weapon], tbFeature[Npc.NpcResPartsDef.npc_part_mantle], 0, tbFeature[Npc.NpcResPartsDef.npc_part_head], tbFeature[Npc.NpcResPartsDef.npc_part_wind])
end

function tbWnd:ResetSearchEnemyInfo()
  local nTimestamp = GetTime()
  local nLeftTime = FriendShip.Cfg.nGetEnemyPositionCd - (nTimestamp - FriendShip:GetUseSearchCdTime())
  local nDate = Lib:GetTimeNum(GetTime())
  local nFreeCount = 0
  local nLeftSearchTimes = FriendShip:GetLeftUseTimes()
  if nDate ~= FriendShip:GetLastSearchDate() then
    nFreeCount = 1
  end
  if 0 == nLeftSearchTimes then
    self.pPanel:Label_SetText(self.txtSearchCount, nFreeCount)
  else
    self.pPanel:Label_SetText(self.txtSearchCount, string.format("%d+%d", nFreeCount, nLeftSearchTimes))
  end
  if self.nCdTimer then
    Timer:Close(self.nCdTimer)
    self.nCdTimer = nil
  end
  local bIsSearchInCd = nLeftTime > 0
  self.pPanel:Button_SetEnable(self.btnClearSearchCd, bIsSearchInCd)
  if bIsSearchInCd then
    self.pPanel:Label_SetText(self.txtCDTimes, Lib:TimeDesc3(nLeftTime))
    self.nCdTimer = Timer:Register(Env.GAME_FPS, function()
      nLeftTime = nLeftTime - 1
      local bInnerIsSearchInCd = nLeftTime > 0
      self.pPanel:Button_SetEnable(self.btnClearSearchCd, bInnerIsSearchInCd)
      if bInnerIsSearchInCd then
        self.pPanel:Label_SetText(self.txtCDTimes, Lib:TimeDesc3(nLeftTime))
      else
        self.pPanel:Label_SetText(self.txtCDTimes, "0:00")
        self.nCdTimer = nil
      end
      return bInnerIsSearchInCd
    end)
  else
    self.pPanel:Label_SetText(self.txtCDTimes, "0:00")
  end
end

function tbWnd:SetSearchCdShow(nLeftTime)
  if nLeftTime <= 0 then
    self.pPanel:Label_SetText(self.txtCDTimes, "0:00")
    self.nCdTimer = nil
    return false
  else
    self.pPanel:Label_SetText(self.txtCDTimes, string.format("%d:%d", nLeftTime / 60, nLeftTime % 60))
  end
  return true
end
