local tbWnd = Ui:CreateWindow("UISocialAddFriend")
tbWnd.tbControls = {
  btnAdd = "node/imgBG/Content/btnAdd",
  inputPlayer = "node/imgBG/Content/inputPlayer",
  btnClose = "node/imgBG/btnClose"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnAdd = function(self)
    self:AddFriend()
    self:Close()
  end
}

function tbWnd:OnOpen()
  self.pPanel:Input_SetText(self.inputPlayer, "")
end

function tbWnd:AddFriend()
  local szInput = self.pPanel:Input_GetText(self.inputPlayer)
  local dwPlayerID = tonumber(szInput)
  if string.lower(szInput) == string.lower(me.szName) or szInput == me.dwID then
    me.CenterMsg(i18n.EasyGet("\228\184\141\229\133\129\232\174\184\230\144\156\231\180\162\232\135\170\229\183\177"))
    return
  end
  local tbAllFriend = FriendShip:GetAllFriendData()
  local bIsFriend = false
  if dwPlayerID then
    local tbRelationInfo = FriendShip:GetFriendDataInfo(dwPlayerID)
    if tbRelationInfo and FriendShip:IsFriend(me.dwID, dwPlayerID) then
      bIsFriend = true
    end
  else
    for i, v in ipairs(tbAllFriend) do
      if string.lower(v.szName) == string.lower(szInput) then
        bIsFriend = true
        break
      end
    end
  end
  if bIsFriend then
    me.CenterMsg(i18n.EasyGet("\228\187\150\229\183\178\231\187\143\230\152\175\228\189\160\231\154\132\229\165\189\229\143\139\228\186\134"))
    return
  end
  if dwPlayerID then
    FriendShip:RequestAddFriend(dwPlayerID)
  else
    FriendShip:RequestAddFriendByName(szInput)
  end
  self:Close()
end
