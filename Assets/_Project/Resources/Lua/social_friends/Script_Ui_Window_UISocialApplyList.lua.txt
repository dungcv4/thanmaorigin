local tbWnd = Ui:CreateWindow("UISocialApplyList")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnClearAllApply = "node/imgBG/Content/btnClearAllApply",
  ApplyTemplate = "node/imgBG/Content/imgApplyList/ApplyList/ApplyTemplate",
  NotInfo = "node/imgBG/Content/NotInfo"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnClearAllApply = function(self)
    self:ClearAll()
  end
}

function tbWnd:OnCreate()
  self.tbApplyListScrollView = Ui:NewScrollView(self, self.ApplyTemplate)
end

function tbWnd:OnOpen()
  self.tbApplyerList = FriendShip:GetAllFriendRequestData()
  self:GenerateApplyList()
end

function tbWnd:GenerateApplyList()
  self.tbApplyListScrollView:Clear()
  local nApplyerNum = #self.tbApplyerList
  self.pPanel:SetActive(self.NotInfo, 0 == nApplyerNum)
  self.pPanel:SetActive(self.btnClearAllApply, nApplyerNum > 0)
  for i = 1, nApplyerNum do
    self.tbApplyListScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "PortraitObj")
      tbPortraitObjPanel:SetRoleIcon(self.tbApplyerList[i].nPortrait)
      tbPortraitObjPanel:SetPortraitFrame(self.tbApplyerList[i].nPortraitFrame)
      tbPortraitObjPanel:SetFactionIcon(self.tbApplyerList[i].nFaction)
      tbPortraitObjPanel:SetLevel(self.tbApplyerList[i].nLevel)
      pPanel:Label_SetText("txtName", self.tbApplyerList[i].szName)
      local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(self.tbApplyerList[i].nHonorStarLevel, false)
      local obj = pPanel:GetObject("HonorTitle")
      local pHonorTitle = obj:GetComponent(Ui.UIHonorTitle)
      pHonorTitle:SetHonorTitle(i18n.Parse(szHonorName), szHonorColor)
      Ui:Button_BindEvent(pPanel, "btnAgree", self.AgreeApply, self, i, self.tbApplyerList[i].dwID)
      Ui:Button_BindEvent(pPanel, "btnIgnore", self.IgnoreApply, self, i, self.tbApplyerList[i].dwID)
    end)
  end
end

function tbWnd:AgreeApply(nIndex, dwPlayerId)
  local tbRequest = self.tbApplyerList[nIndex]
  FriendShip:AcceptFriendRequest(dwPlayerId, tbRequest)
  table.remove(self.tbApplyerList, nIndex)
  self:GenerateApplyList()
end

function tbWnd:IgnoreApply(nIndex, dwPlayerId)
  FriendShip:RefuseAddFriend(dwPlayerId)
  table.remove(self.tbApplyerList, nIndex)
  self:GenerateApplyList()
end

function tbWnd:ClearAll()
  FriendShip:RefuseAllRequet()
  self:Close()
end
