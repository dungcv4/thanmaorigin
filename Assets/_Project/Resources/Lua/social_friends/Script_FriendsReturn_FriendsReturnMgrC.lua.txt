function FriendsReturn:GetVNVipLevelupAwardReq(nId, nType)

function FriendsReturn:OnLogin()
  local nNowTime = GetTime()
  local nActivityStartTime = Lib:ParseDateTime(self.Cfg.SpecialActivity.nNewSerActivityStartTime)
  local nActivityEndTime = Lib:ParseDateTime(self.Cfg.SpecialActivity.nNewSerActivityEndTime)
  local nNewSerActivityGetEndTime = nActivityEndTime + 4320000
  local nTotalYuanbaoCount = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipTotalYuanbaoCount)
  local nYuanbaoGetCount = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoGetCount)
  local nLeftYuanbaoCount = nTotalYuanbaoCount - nYuanbaoGetCount
  if nNowTime >= nActivityStartTime and nNowTime <= nNewSerActivityGetEndTime then
    local nActCheck = me.GetUserValue(self.Def.nGroup, self.Def.KEY.nReturnerVNVipAwardCheck)
    if 1 == nActCheck or 3 == nActCheck then
      RemoteServer.OnFriendsReturnRequest("TryGetVNVipNameInfo")
      local nLastGetTime = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipYuanbaoCanGetTime)
      local nCurTime = GetTime()
      if Lib:IsDiffDay(14400, nCurTime, nLastGetTime) and nLeftYuanbaoCount > 0 then
        RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ReturnerVNVipSignIn)
      end
    end
  end
  RemoteServer.OnFriendsReturnRequest("UpdateFundCanGetState")
  if not FriendsReturn:CheckIsPlayerReturner(me) then
    return
  end
  RemoteServer.OnFriendsReturnRequest("UpdateDailyAwardCanGetState")
end

function FriendsReturn:OnLevelUp()
  local nLevelupGetNum = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerVNVipLevelupGetState)
  for i = 1, Lib:CountTB(FriendsReturn.tbVNVipAwardInfoById) do
    local bCanGet = self:CheckIfPlayerCanRequestVNVipReturnLevelupAward(me, i)
    if bCanGet and not Lib:GetBit(nLevelupGetNum, i) then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ReturnerVNVIPNewServerLevelAward)
      break
    end
  end
end

function FriendsReturn:OnSyncActivePoint()
  if not FriendsReturn:CheckIsPlayerReturner(me) then
    return
  end
  RemoteServer.OnFriendsReturnRequest("UpdateDailyAwardCanGetState")
end

function FriendsReturn:CheckDailySignInStateAndRedPoint()
  local nNowSignInDay = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerDailySignInCheck)
  local nNowSignCanGetDay = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerDailySignInCanGetCheck)
  local bIsSpecialReturner = FriendsReturn:CheckIsPlayerSpecialReturner(me)
  local tbSignInAwardInfo = FriendsReturn.tbDailySignInInfo
  if bIsSpecialReturner then
    tbSignInAwardInfo = FriendsReturn.tbSpecialDailySignInInfo
  end
  if nNowSignInDay >= Lib:CountTB(tbSignInAwardInfo) then
    return
  end
  if nNowSignInDay >= nNowSignCanGetDay then
    return
  end
  RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ReturnerDailySignIn)
  self:UpdateReturnerUIAck()
end

function FriendsReturn:CheckDailyFundStateAndRedPoint()
  local nBuyFund = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerDailyFundBuy)
  local nNowFundDay = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerDailyFundCheck)
  local nNowFundCanGetDay = me.GetUserValue(FriendsReturn.Def.nGroup, FriendsReturn.Def.KEY.nReturnerDailyFundCanGetCheck)
  if 0 == nBuyFund then
    return
  end
  if nNowFundDay >= Lib:CountTB(FriendsReturn.tbDailyFundInfo) then
    return
  end
  if nNowFundDay >= nNowFundCanGetDay then
    return
  end
  RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ReturnerDailyFund)
  self:UpdateReturnerUIAck()
end

function FriendsReturn:CheckCanShowHint()
  local bCanBeReturner = self:CheckIfPlayerCanBeReturner(me)
  local bIsReturner = self:CheckIsPlayerReturner(me)
  if bCanBeReturner or bIsReturner then
    return true
  else
    return false
  end
end

function FriendsReturn:CheckCanShowNewServerHint()
  if me.nLevel < self.Cfg.SpecialActivity.nNewSerTriggerLevel then
    return false
  end
  local nInfo = me.GetUserValue(self.Def.nGroup, self.Def.KEY.nReturnerVNVipAwardCheck)
  if 1 == nInfo then
    local nNowTime = GetTime()
    local nActivityStartTime = Lib:ParseDateTime(self.Cfg.SpecialActivity.nNewSerActivityStartTime)
    local nActivityEndTime = Lib:ParseDateTime(self.Cfg.SpecialActivity.nNewSerActivityEndTime)
    local nNewSerActivityGetEndTime = nActivityEndTime + 4320000
    if nNowTime < nActivityStartTime or nNowTime > nNewSerActivityGetEndTime then
      return false
    end
    return true
  end
  if 3 == nInfo then
    local nNowTime = GetTime()
    local nActivityStartTime = Lib:ParseDateTime(self.Cfg.SpecialActivity.nNewSerActivityStartTime)
    local nActivityEndTime = Lib:ParseDateTime(self.Cfg.SpecialActivity.nNewSerActivityEndTime)
    if nNowTime < nActivityStartTime or nNowTime > nActivityEndTime then
      return false
    end
    return true
  end
  return false
end

function FriendsReturn:GotoPos()
  local tbInfo = XiSuiDao.LeaderPos[me.nFaction]
  local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(tbInfo.nNpcID, tbInfo.nTID, true)
  
  local function fnCallback()
    local nNpcId = KNpc.GetNpcIdByTemplateId(tbInfo.nNpcID)
    if nNpcId then
      Operation:SimpleTap(nNpcId)
    end
  end
  
  Ui:CloseWindow("UIBag")
  Ui:CloseWindow("UIWelfare")
  AutoPath:GotoAndCallEx(tbInfo.nTID, nPosX, nPosY, fnCallback, nWalkNearLength, nil, tbInfo.nNpcID)
  return
end
