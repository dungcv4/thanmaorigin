Require("CommonScript/Shop/Shop.lua")

function Shop:LoadReturnerShop()
  self.tbReturnerShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/ReturnerShop.tab", "dd", nil, {"nWareId", "nSort"}) or {}
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.Returner then
      print("Load FriendsReturnShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.Returner, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbReturnerShopWares, v)
    end
  end
end

Shop:LoadReturnerShop()

function Shop:GetReturnerWareAvaliableList()
  local tbRtn = {}
  for _, v in pairs(self.tbReturnerShopWares) do
    local tbWare = self:GetAWare(v.nWareId)
    if tbWare and self:CheckWareAvaliable(tbWare, me) then
      table.insert(tbRtn, v)
    end
  end
  if not Lib:IsEmptyTB(tbRtn) then
    table.sort(tbRtn, function(a, b)
      return a.nSort < b.nSort
    end)
  end
  return tbRtn
end

function FriendsReturn:LoadReturnerVipAwardSetting()
  self.tbOldServerVipAwardInfo = {}
  self.tbNewServerVipAwardInfo = {}
  local tbTabInfo = Lib:EasyLoadTabFile("Setting/FriendsReturn/FriendsReturnVipAward.tab", {"nId"})
  for _, tbInfo in pairs(tbTabInfo) do
    if 0 == tbInfo.nNewServer then
      table.insert(self.tbOldServerVipAwardInfo, tbInfo)
    elseif 1 == tbInfo.nNewServer then
      table.insert(self.tbNewServerVipAwardInfo, tbInfo)
    end
  end
  self.tbOldServerVipAwardInfo = SetReadOnly(self.tbOldServerVipAwardInfo)
  self.tbNewServerVipAwardInfo = SetReadOnly(self.tbNewServerVipAwardInfo)
end
