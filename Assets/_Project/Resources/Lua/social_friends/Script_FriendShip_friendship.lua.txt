function FriendShip:OnSynFriendShipData(tbData)
  local dwID = tbData.dwID
  local nFaction = tbData.nFaction
  local nLevel = tbData.nLevel
  local szName = tbData.szName
  local nState = tbData.nState
  local nPortrait = tbData.nPortrait
  local dwKinId = tbData.dwKinId
  local dwTongId = tbData.dwTongId
  local szKinOrTongName = tbData.szKinOrTongName
  local nHonorStarLevel = tbData.nHonorStarLevel
  local nSex = tbData.nSex
  local emType = tbData.emType
  local emImity = tbData.emImity
  local emEnemy_Left = tbData.emEnemy_Left
  local emEnemy_Right = tbData.emEnemy_Right
  local emBlackOrRequest = tbData.emBlackOrRequest
  local emTemp_Refuse = tbData.emTemp_Refuse
  local emSendPhysicalPower = tbData.emSendPhysicalPower
  local emGetPhysicalPower = tbData.emGetPhysicalPower
  local nVipLevel = tbData.nVipLevel
  local szAccount = tbData.szAccount
  local dwTeamId = tbData.dwTeamId
  local nPortraitFrame = tbData.nPortraitFrame or 0
  local bModify = self.tbAllData[dwID] and self.tbAllData[dwID][emFriendData_Type] == emType and 0 ~= emType and true or false
  self.tbAllData[dwID] = {
    dwRoleId = dwID,
    dwID = dwID,
    dwTeamId = dwTeamId,
    nFaction = nFaction,
    nLevel = nLevel,
    szName = szName,
    szAccount = szAccount,
    nState = nState,
    nPortrait = nPortrait,
    nPortraitFrame = nPortraitFrame,
    dwKinId = dwKinId,
    dwTongId = dwTongId,
    szKinOrTongName = szKinOrTongName,
    nHonorStarLevel = nHonorStarLevel,
    nSex = nSex,
    nImity = emImity,
    nVipLevel = nVipLevel,
    nHate = dwID > me.dwID and emEnemy_Left or emEnemy_Right
  }
  local tbData = self.tbAllData[dwID]
  tbData[emFriendData_Type] = emType
  tbData[emFriendData_Imity] = emImity
  tbData[emFriendData_Enemy_Left] = emEnemy_Left
  tbData[emFriendData_Enemy_Right] = emEnemy_Right
  tbData[emFriendData_BlackOrRequest] = emBlackOrRequest
  tbData[emFriendData_SendPhysicalPower] = emSendPhysicalPower
  tbData[emFriendData_GetPhysicalPower] = emGetPhysicalPower
  if 0 ~= emImity then
    local nImityLevel, nMaxImity = FriendShip:GetImityLevel(emImity)
    tbData.nImityLevel = nImityLevel
    tbData.nMaxImity = nMaxImity
    if self.tbMyBlackList[dwID] then
      FriendShip:DelBlack(dwID)
    end
  end
  if FriendShip:IsMeRequested(me.dwID, dwID, self.tbAllData[dwID]) and self:IsHeInMyBlack(dwID) then
    RemoteServer.RefuseAddFriend(dwID)
  end
  self.nVersion = self.nVersion + 1
  self:UpdateFriendInfoListWithRoleInfoCache({dwID})
  EventNotify.OnNotify(EventNotify.emNoTIFY_SYNC_FRIEND_DATA, bModify and dwID or nil)
end

function FriendShip:TransBlackListAndRemarkList()
  local bHasMyBlack = LocalData:HasRoleKey("FriendShipBlackList")
  local bHasMyRemark = LocalData:HasRoleKey("FriendShipRemarkList")
  if bHasMyBlack and bHasMyRemark then
    return
  end
  local szFileContent, nLen = Lib:ReadFile_W(g_szUserPath .. "User/UserInfo.lua", "r")
  if 0 == nLen then
    szFileContent = "return {};"
  end
  local tbTmpEnv = {}
  szFileContent = "local fnLoad = function () " .. szFileContent .. " end tbUserInfo = fnLoad();"
  local fnFile = load(szFileContent, nil, "bt", tbTmpEnv)
  if fnFile then
    fnFile()
  end
  local tbUserInfo = tbTmpEnv.tbUserInfo or {}
  tbUserInfo[me.dwID] = tbUserInfo[me.dwID] or {}
  tbUserInfo[me.dwID].BlackList = tbUserInfo[me.dwID].BlackList or {}
  tbUserInfo[me.dwID].RemarkList = tbUserInfo[me.dwID].RemarkList or {}
  local nRet = cjson.encode_sparse_array()
  if not nRet then
    cjson.encode_sparse_array(true)
  end
  if not bHasMyBlack then
    LocalData:SetRoleTable("FriendShipBlackList", tbUserInfo[me.dwID].BlackList)
  end
  if not bHasMyRemark then
    LocalData:SetRoleTable("FriendShipRemarkList", tbUserInfo[me.dwID].RemarkList)
  end
  if not nRet then
    cjson.encode_sparse_array(false)
  end
  tbUserInfo[me.dwID].BlackList = nil
  tbUserInfo[me.dwID].RemarkList = nil
  if Client.tbUserInfo and Client.tbUserInfo[me.dwID] then
    Client.tbUserInfo[me.dwID].BlackList = nil
    Client.tbUserInfo[me.dwID].RemarkList = nil
  end
  local szValue = Lib:Val2Str(tbUserInfo)
  local szTable = "return" .. szValue
  local bResult = Lib:WriteFile_W(g_szUserPath .. "User/UserInfo.lua", "w+", szTable)
  if not bResult then
    return
  end
end

function FriendShip:CheckImitiyAchieve(nImityLevel, nOldImityLevel)
  Log("FriendShip:CheckImitiyAchieve", nImityLevel, nOldImityLevel)
  if self.nCheckImityAchieveVersion == self.nVersion then
    return
  end
  self.nCheckImityAchieveVersion = self.nVersion
  for i = 1, #self.tbImityAchivementLevel do
    local nCheckLevel = self.tbImityAchivementLevel[i]
    if nOldImityLevel < nCheckLevel and nImityLevel >= nCheckLevel then
      local tbAllFriendData = self:GetAllFriendData()
      local tbVaildFriendIds = {}
      for _, tbFriend in ipairs(tbAllFriendData) do
        if nCheckLevel <= tbFriend.nImityLevel then
          table.insert(tbVaildFriendIds, tbFriend.dwID)
        end
      end
      if FriendShip.nAchievementNeedFriendNum <= #tbVaildFriendIds then
        RemoteServer.TryCheckImityAchievement(nCheckLevel, tbVaildFriendIds)
      end
    end
  end
end

function FriendShip:haveFriendInImityLevel(nCurLevel, nNum)
  local tbAllFriendData = self:GetAllEnemyData()
  local nValidNum = 0
  for _, tbFriend in ipairs(tbAllFriendData) do
    if nCurLevel <= tbFriend.nImityLevel then
      nValidNum = nValidNum + 1
    end
  end
  if nNum <= nValidNum then
    return true
  else
    return false
  end
end

function FriendShip:AddBlack(dwRoleId2, tbRoleInfo)
  if dwRoleId2 == me.dwID then
    return
  end
  self.tbAllData[dwRoleId2] = nil
  self.tbMyBlackList[dwRoleId2] = tbRoleInfo
  self:SaveBlackList()
  self.nVersion = self.nVersion + 1
  ChatMgr.PrivateChatUnReadCache[dwRoleId2] = nil
  ChatMgr.PrivateChatReadCache[dwRoleId2] = nil
  ChatMgr:RemoveRecentPrivateTarget(dwRoleId2)
  EventNotify.OnNotify(EventNotify.emNoTIFY_NEW_PRIVATE_MSG)
  EventNotify.OnNotify(EventNotify.emNoTIFY_SYNC_FRIEND_DATA)
end

function FriendShip:RequestUpdateBlackInfo()
  if self.bUpdateBlackInfo then
    return
  end
  local tbIds = {}
  for dwId, _ in pairs(self.tbMyBlackList) do
    table.insert(tbIds, dwId)
  end
  if not next(tbIds) then
    return
  end
  self.bUpdateBlackInfo = true
  RemoteServer.RequestFriendShip("UpdateBlackInfo", tbIds)
end

function FriendShip:OnUpdateBlackInfo(tbInfos)
  local bUpdate
  for dwId, tbInfo in pairs(tbInfos) do
    if self.tbMyBlackList[dwId] then
      self.tbMyBlackList[dwId] = tbInfo
      bUpdate = bUpdate or true
    end
  end
  if bUpdate then
    self:SaveBlackList()
    EventNotify.OnNotify(EventNotify.emNoTIFY_SYNC_FRIEND_DATA)
  end
end

function FriendShip:UpdateFriendInfoListWithRoleInfoCache(tbIds)
  local bAllHit, nSession, tbRoleInfoList = Player.RoleInfoCache:GetCache(tbIds)
  if not bAllHit then
    self.nCacheUpdateSession = nSession
  end
  self:UpdateFriendRoleInfoWithCacheList(tbRoleInfoList)
  self:RequestExtraInfo(tbIds)
end

function FriendShip:RequestExtraInfo(tbIds)
  self.tbExtraIdList = self.tbExtraIdList or {}
  for _, nId in ipairs(tbIds) do
    if not self.tbExtraIdList[nId] then
      self.tbExtraIdList[nId] = true
    end
  end
  if self.nQuestExtraInfoTimer then
    return
  end
  local nInterval = 3
  self.nQuestExtraInfoTimer = Timer:Register(nInterval * Env.GAME_FPS, self.RequestExtraInfoHelper, self)
end

function FriendShip:RequestExtraInfoHelper()
  if self.nQuestExtraInfoTimer then
    Timer:Close(self.nQuestExtraInfoTimer)
    self.nQuestExtraInfoTimer = nil
  end
  self.nVersion = self.nVersion + 1
  local tbIds = {}
  for nId in pairs(self.tbExtraIdList or {}) do
    table.insert(tbIds, nId)
    if #tbIds >= self.MAX_SYNC_NUM then
      RemoteServer.RequestFriendShip("UpdateExtraInfo", tbIds)
      tbIds = {}
    end
  end
  self.tbExtraIdList = {}
  RemoteServer.RequestFriendShip("UpdateExtraInfo", tbIds)
end

function FriendShip:OnUpdateFriendInfo(tbInfos)
  if not tbInfos then
    return
  end
  for i = 1, #tbInfos do
    local tbInfo = tbInfos[i]
    local tbData = self.tbAllData[tbInfo.dwID]
    if not tbData then
      return
    end
    tbData.dwId = tbInfo.dwID
    tbData.szName = tbInfo.szName
    tbData.nLevel = tbInfo.nLevel
    tbData.nHonorStarLevel = tbInfo.nHonorStarLevel

    tbData.nPortrait = tbInfo.nPortrait

function FriendShip:GetAllFriendData(bNoSort)
  if self.tbAllFriendData and self.nAllFriendDataVersion == self.nVersion then
    return self.tbAllFriendData
  end
  if bNoSort and self.tbAllFriendData then
    for i, v in ipairs(self.tbAllFriendData) do
      self.tbAllFriendData[i] = self.tbAllData[v.dwID]
    end
    return self.tbAllFriendData
  end
  local tbFriends = {}
  for k, v in pairs(self.tbAllData) do
    if v[emFriendData_Type] == emFriend_Type_Friend then
      table.insert(tbFriends, v)
    end
  end
  self.tbAllFriendData = tbFriends
  self.nAllFriendDataVersion = self.nVersion
  return self.tbAllFriendData
end

function FriendShip:GetAllFriendRequestData()
  if self.tbAllRequestData and self.nAllRequestDataVersion == self.nVersion then
    return self.tbAllRequestData
  end
  local tbFriends = {}
  for k, v in pairs(self.tbAllData) do
    if k > me.dwID and v[emFriendData_BlackOrRequest] == emFriend_Type_Request_Right or k < me.dwID and v[emFriendData_BlackOrRequest] == emFriend_Type_Request_Left then
      table.insert(tbFriends, v)
    end
  end
  self.tbAllRequestData = tbFriends
  self.nAllRequestDataVersion = self.nVersion
  return self.tbAllRequestData
end

function FriendShip:GetAllEnemyData()
  if self.tbAllEnemyData and self.nAllEnemyDataVersion == self.nVersion then
    return self.tbAllEnemyData
  end
  local tbFriends = {}
  for k, v in pairs(self.tbAllData) do
    if k > me.dwID and v[emFriendData_Enemy_Left] and v[emFriendData_Enemy_Left] > 0 or k < me.dwID and v[emFriendData_Enemy_Right] and v[emFriendData_Enemy_Right] > 0 then
      table.insert(tbFriends, v)
    end
  end
  self.tbAllEnemyData = tbFriends
  self.nAllEnemyDataVersion = self.nVersion
  return self.tbAllEnemyData
end

function FriendShip:AcceptFriendRequest(dwRoleId2, tbFriendData)

function FriendShip:RefuseAllRequet()
  RemoteServer.RefuseAllRequet()
  local tbNeedDel = {}
  for k, v in pairs(self.tbAllData) do
    if k > me.dwID and v[emFriendData_BlackOrRequest] == emFriend_Type_Request_Right or k < me.dwID and v[emFriendData_BlackOrRequest] == emFriend_Type_Request_Left then
      tbNeedDel[k] = 1
    end
  end
  for k, _ in pairs(tbNeedDel) do
    self.tbAllData[k] = nil
  end
  self.tbAllRequestData = {}
  EventNotify.OnNotify(EventNotify.emNoTIFY_SYNC_FRIEND_DATA)
end

function FriendShip:RefuseAddFriend(dwID)
  RemoteServer.RefuseAddFriend(dwID)
  if not self.tbAllData then
    return
  end
  if self.tbAllData[dwID] and (dwID > me.dwID and self.tbAllData[dwID][emFriendData_BlackOrRequest] == emFriend_Type_Request_Right or dwID < me.dwID and self.tbAllData[dwID][emFriendData_BlackOrRequest] == emFriend_Type_Request_Left) then
    self.tbAllData[dwID] = nil
  end
  self.tbAllRequestData[dwID] = nil
  if 0 == Lib:CountTB(self.tbAllRequestData) then
    EventNotify.OnNotify(EventNotify.emNoTIFY_SYNC_FRIEND_DATA)
  end
end

function FriendShip:UpdateIntimacyBuff()
  local pNpc = me.GetNpc()
  if not TeamMgr:HasTeam() or GlobalHangUp:IsGlobalHangUpMap(me.nMapTemplateId) then
    if pNpc and pNpc.GetSkillState(FriendShip.nIntimacyBuffId) then
      pNpc.RemoveSkillState(FriendShip.nIntimacyBuffId)
    end
    return
  end
  local tbMembers = TeamMgr:GetMembers()
  self.tbIntimacyBuffTip = {}
  for _, tbMember in ipairs(tbMembers) do
    local tbFriendInfo = FriendShip:GetFriendDataInfo(tbMember.nPlayerID)
    if tbFriendInfo and tbFriendInfo.nPlayerID ~= "" then
      local nBuff = FriendShip:GetFriendImityExpP(tbFriendInfo.nImityLevel)
      if nBuff > 0 then
        local szIntimacyBuffTip = i18n.EasyGet("\228\184\142 <color={0}>{1}</color> \233\135\142\229\164\150\230\137\147\230\128\170\231\187\143\233\170\140\229\133\177\228\186\171 <color={2}>+{3}%</color>\n", FriendShip.tbIntimacyBuffColor[1], tbMember.szName, FriendShip.tbIntimacyBuffColor[2], nBuff)
        table.insert(self.tbIntimacyBuffTip, szIntimacyBuffTip)
      end
    end
  end
  if Lib:CountTB(self.tbIntimacyBuffTip) > 0 and pNpc then
    pNpc.AddSkillState(FriendShip.nIntimacyBuffId, 1, 0, -1)
  else
    pNpc.RemoveSkillState(FriendShip.nIntimacyBuffId)
  end
end
