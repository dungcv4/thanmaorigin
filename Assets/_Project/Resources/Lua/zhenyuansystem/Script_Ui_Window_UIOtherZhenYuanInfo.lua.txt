function tbWnd:OnCreate()
  self.tbEquipPos2ItemGrid = {}
  local obj = self.pPanel:GetObject(tbControls.imgModel)
  self.NpcModel = obj:GetComponent(Ui.UINpcModel)
end

function tbWnd:OnOpen()
  self.pAsyncRole = self.tbOwner.pAsyncRole
  self.tbExtEquips = self.tbOwner.tbExtEquips
  self:UpdataInfo()
  for nGroupId, _ in pairs(Item.ZhenYuan.tbHuTiAttrCfg) do
    self:RefreshSeriesAttr(nGroupId)
  end
  self:ShowEffect()
  self:ShowModel()
  self:ShowHuTiStar()
end

function tbWnd:OnClose()
  self.NpcModel:Release()
end

function tbWnd:ShowModel()
  local tbFeature = {}
  local pAsyncRole = self.pAsyncRole
  for nEquipPos, nIndex in pairs(Item.EQUIPPOS_TO_NPC_RES_PART) do
    local nResId
    local nFashionId = pAsyncRole.GetFashion(nIndex + 1)
    if nFashionId > 0 then
      local tbFashionSetting = Fashion:GetFashionSettingNotPlayer(nFashionId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
      if nil ~= tbFashionSetting then
        nResId = tbFashionSetting.nResId
      end
    end
    if nil == nResId then
      local nItemId = self.tbOwner.tbEquips[nEquipPos]
      if nItemId and nItemId > 0 then
        local pEquip = KItem.GetItemObj(nItemId)
        local _, _, _, _, nInnerResId = KItem.GetItemShowInfo(pEquip.dwTemplateId, pAsyncRole.GetFaction(), pAsyncRole.GetFactionSect(), pAsyncRole.GetSex())
        nResId = nInnerResId
      end
    end
    if nil ~= nResId then
      tbFeature[nIndex] = nResId
    else
      tbFeature[nIndex] = 0
    end
  end
  if tbFeature[Npc.NpcResPartsDef.npc_part_body] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_body] = 1
  end
  if tbFeature[Npc.NpcResPartsDef.npc_part_head] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_head] = 1
  end
  local nWeaponEffectResId = 0
  if tbFeature[Npc.NpcResPartsDef.npc_part_weapon] <= 0 then
    tbFeature[Npc.NpcResPartsDef.npc_part_weapon] = 1
  elseif tbFeature[Npc.NpcResPartsDef.npc_part_weapon] > 0 then
    local nWeaponId = self.tbOwner.tbEquips[Item.EQUIPPOS_WEAPON] or 0
    if nWeaponId > 0 then
      local pWeapon = KItem.GetItemObj(nWeaponId)
      local tbPlayer = {
        nFaction = self.pAsyncRole.GetFaction(),
        nFactionSect = self.pAsyncRole.GetFactionSect(),
        nSex = self.pAsyncRole.GetSex()
      }
      nWeaponEffectResId = Item.EquipEnhance:GetEffectResId(tbPlayer, self.pAsyncRole.GetEnhance(Item.EQUIPPOS_WEAPON + 1))
    end
  end
  self.NpcModel:ShowModel(self.pAsyncRole.GetSex(), 1, tbFeature[Npc.NpcResPartsDef.npc_part_body], tbFeature[Npc.NpcResPartsDef.npc_part_weapon], tbFeature[Npc.NpcResPartsDef.npc_part_mantle], tbFeature[Npc.NpcResPartsDef.npc_part_horse], tbFeature[Npc.NpcResPartsDef.npc_part_head], tbFeature[Npc.NpcResPartsDef.npc_part_wind], nWeaponEffectResId)
  self.NpcModel:SetAction(Npc.NpcAction.SitDown)
end

function tbWnd:ShowHuTiStar()
  local nCurStar = self.pAsyncRole.GetZhenYuanHutiStar()
  local nStar2 = 0
  if nCurStar > 9 then
    nStar2 = nCurStar - 9
  end
  for i = 1, 9 do
    if i <= nCurStar then
      self.pPanel:SetActive(tbControls.HutiStar .. "/HutiStar0" .. i, true)
      self.pPanel:Sprite_SetSprite(tbControls.HutiStar .. "/HutiStar0" .. i, Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar1)
    else
      self.pPanel:SetActive(tbControls.HutiStar .. "/HutiStar0" .. i, false)
    end
    if i <= nStar2 then
      self.pPanel:Sprite_SetSprite(tbControls.HutiStar .. "/HutiStar0" .. i, Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar2)
    end
  end
end

end

function tbWnd:CalcFightPower()
  local nTotalPower = 0
  for nEquipPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local nItemId = self.tbExtEquips[nEquipPos]
    local pEquip
    if nItemId and nItemId > 0 then
      pEquip = KItem.GetItemObj(nItemId)
    end
    if pEquip then
      nTotalPower = nTotalPower + Item.ZhenYuan:GetEquipPower(pEquip)
      local nLevel = self.pAsyncRole.GetZhenYuanHoleLevel(nEquipPos - Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL + 1)
      local tbHoleCfg = Item.ZhenYuan:GetHoleEnhanceCfg(nEquipPos, nLevel)
      if tbHoleCfg then
        nTotalPower = nTotalPower + tbHoleCfg.FightPower
      end
    end
  end
  for i = 1, 5 do
    if self:CheckHuTiGroupActive(i) then
      local nLevel = self.pAsyncRole.GetZhenYuanHutiLevel(i)
      local tbHuTiCfg = Item.ZhenYuan:GetHTGroupCfg(i, nLevel + 1)
      if tbHuTiCfg then
        nTotalPower = nTotalPower + tbHuTiCfg.FightPower
      end
    end
  end
  return nTotalPower
end

function tbWnd:UpdataInfo()
  for nPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local nItemId = self.tbExtEquips[nPos]
    local pEquip
    if nItemId and nItemId > 0 then
      pEquip = KItem.GetItemObj(nItemId)
    end
    if pEquip then
      self.pPanel:SetActive(tbZhenYuanPosKey[nPos].ItemGrid, true)
      local pItemGrid = self.tbEquipPos2ItemGrid[nPos]
      if not pItemGrid then
        pItemGrid = Ui:NewExistItemGrid(self, tbZhenYuanPosKey[nPos].ItemGrid .. "/UIItemGrid")
        self.tbEquipPos2ItemGrid[nPos] = pItemGrid
      end
      pItemGrid:SetItem(pEquip.dwId, pEquip.dwTemplateId, nil, self.pAsyncRole.GetFaction(), self.pAsyncRole.GetFactionSect(), self.pAsyncRole.GetSex(), self.pAsyncRole, nil, self.tbExtEquips)
      pItemGrid:SetClickEnable(true, self.OnClickItem, self, nPos, pItemGrid)
      pItemGrid.szItemOpt = "ViewEquip"
      self.pPanel:SetActive(tbZhenYuanWordActive[nPos][1], false)
      self.pPanel:SetActive(tbZhenYuanWordActive[nPos][2], false)
    else
      self.pPanel:SetActive(tbZhenYuanPosKey[nPos].ItemGrid, false)
      self.pPanel:SetActive(tbZhenYuanWordActive[nPos][1], true)
      self.pPanel:SetActive(tbZhenYuanWordActive[nPos][2], true)
    end
  end
  local nActive = 0
  for nGroupId = 1, 5 do
    local bRet = self:CheckHuTiGroupActive(nGroupId)
    if bRet then
      nActive = nActive + 1
    end
  end
  self.pPanel:SetActive(tbControls.Activate_Effect, false)
  for nIndex, tbValue in ipairs(HuTiActiveStat) do
    if nActive >= tbValue[1] and nActive <= tbValue[2] then
      self.pPanel:SetActive(tbControls.txtActivateName, true)
      self.pPanel:SetActive(tbControls.txtActivateValue, true)
      self.pPanel:Label_SetText(tbControls.txtActivateName, tbValue[3]())
      local nStar = self.pAsyncRole.GetZhenYuanHutiStar()
      if nStar >= #Item.ZhenYuan.nHTSkillLevel then
        nStar = nStar - 1
      end
      local szNowLevelDesc = FightSkill:GetSkillMagicDesc(Item.ZhenYuan.nHTSkillId, Item.ZhenYuan.nHTSkillLevel[nStar + 1])
      local szStr = string.gsub(szNowLevelDesc, "\n", "")
      self.pPanel:Label_SetText(tbControls.txtActivateValue, i18n.Parse(szStr))
      if 3 == nIndex then
        self.pPanel:Label_SetColorByName(tbControls.txtActivateValue, "NewGreen")
      else
        self.pPanel:Label_SetColorByName(tbControls.txtActivateValue, "Gray")
      end
      if nIndex > 1 then
        self.pPanel:SetActive(tbControls.Activate_Effect, true)
      end
    end
  end
  self.pPanel:Label_SetText(tbControls.txtFightPower, i18n.EasyGet("\230\136\152\229\138\155\229\128\188\239\188\154{0}", self:CalcFightPower()))
  return true
end

function tbWnd:OnClickItem(nIndex, tbItemGrid)
  tbItemGrid.pAsyncRole = self.pAsyncRole
  Item:ShowItemDetail(tbItemGrid)
end

function tbWnd:CheckHuTiGroupActive(nGroupId)
  local tbHTCfg = Item.ZhenYuan:GetHTGroupCfg(nGroupId, 1)
  if not tbHTCfg then
    return false
  end
  for _, nSeries in ipairs(tbHTCfg.tbnSeries) do
    local nItemId = self.tbExtEquips[nSeries]
    if not nItemId then
      return false
    end
    local pEquip = KItem.GetItemObj(nItemId)
    if not pEquip then
      return false
    end
  end
  return true
end

function tbWnd:RefreshSeriesAttr(nGroupId)
  local nLevel = self.pAsyncRole.GetZhenYuanHutiLevel(nGroupId)
  local bIsAct = self:CheckHuTiGroupActive(nGroupId)
  self.pPanel:SetActive(tbSeriesActiveBaseMap[nGroupId], not bIsAct)
  local tbValue = Item.ZhenYuan.tbHuTiAttrCfg[nGroupId]
  local tbHTCfg = tbValue[nLevel + 1]
  if tbHTCfg then
    self.pPanel:Label_SetText(tbSeriesNameMap[nGroupId], i18n.Parse(tbHTCfg.szName))
    for i = 1, tbHTCfg.nAttrNum do
      local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbHTCfg["szAttribType" .. i], tbHTCfg["tbnAttribValue" .. i])
      self.pPanel:SetActive(tbSeriesAttrList[nGroupId], true)
      self.pPanel:Label_SetText(tbSeriesAttrType[nGroupId], i18n.Parse(szAttrType))
      self.pPanel:Label_SetColorByName(tbSeriesAttrType[nGroupId], bIsAct and "NewOrange" or "Gray")
      self.pPanel:Label_SetText(tbSeriesAttrValue[nGroupId], i18n.Parse(szAttrValue))
      self.pPanel:Label_SetColorByName(tbSeriesAttrValue[nGroupId], bIsAct and "NewOrange" or "Gray")
      self.pPanel:Label_SetText(tbSeriesLevelValue[nGroupId], i18n.EasyGet("\231\173\137\231\186\167{0}", nLevel))
    end
  end
end

function tbWnd:OnDestroy()
  self.tbEquipPos2ItemGrid = nil
end

function tbWnd:OnSelectZhenYuan(nIndex, tbElement)
  if self.nSelectIndex then
    local tbOldElement = self.tbQiHunScrollView:GetElement(self.nSelectIndex)
    tbOldElement.pPanel:SetActive("imgSelected", false)
  end
  self.nSelectIndex = nIndex
  tbElement.pPanel:SetActive("imgSelected", true)
  self.tbTabControl:UpdateSubPanel(self.nPanelType, "OnSelectQiHun")
end

function tbWnd:ResetGTActiveStat()
  for nIndex, _ in ipairs(HuTiActiveStat) do
    self.pPanel:SetActive(tbControls.Huti .. "/ZhenYuan0" .. nIndex, false)
  end
end

function tbWnd:ShowEffect()

    local bIsAct = self:CheckHuTiGroupActive(i)
    self.pPanel:SetActive(tbSeriesEffectList[i], bIsAct)
  end
end
