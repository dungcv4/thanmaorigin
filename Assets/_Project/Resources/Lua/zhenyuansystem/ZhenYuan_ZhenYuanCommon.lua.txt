function ZhenYuan:InitCfg()
  if self.bLoadConfig then
    return false
  end
  local szFilepath = "Setting/ZhenYuan/ZYCompoundCfg.tab"
  local tbZhenYuanConfig = Lib:EasyLoadTabFile(szFilepath, {"nId"})
  if not tbZhenYuanConfig then
    LogErr("%s not find", szFilepath)
    return false
  end
  szFilepath = "Setting/ZhenYuan/ZhenYuanItemCfg.tab"
  local tbZhenYuanItemCfg = Lib:EasyLoadTabFile(szFilepath, {
    "nTemplateId"
  })
  if not tbZhenYuanItemCfg then
    LogErr("%s not find", szFilepath)
    return false
  end
  szFilepath = "Setting/ZhenYuan/ZhenYuanExtAttributeCfg.tab"
  local tbZhenYuanExtAttriCfg = Lib:EasyLoadTabFile(szFilepath)
  if not tbZhenYuanExtAttriCfg then
    LogErr("%s not find", szFilepath)
    return false
  end
  szFilepath = "Setting/ZhenYuan/ZYHoleEnhanceCfg.tab"
  local tbTmpZhenYuanHoleCfg = Lib:EasyLoadTabFile(szFilepath, {"nId"})
  if not tbTmpZhenYuanHoleCfg then
    LogErr("%s not find", szFilepath)
    return false
  end
  szFilepath = "Setting/ZhenYuan/ZYHTAttrCfg.tab"
  local tbZYHTAttrCfg = Lib:EasyLoadTabFile(szFilepath)
  if not tbZYHTAttrCfg then
    LogErr("%s not find", szFilepath)
    return false
  end
  szFilepath = "Setting/ZhenYuan/ZhenYuanCostCfg.tab"
  local tbZhenYuanCostCfg = Lib:EasyLoadTabFile(szFilepath, {"nId"})
  if not tbZhenYuanCostCfg then
    LogErr("%s not find", szFilepath)
    return false
  end
  if MODULE_GAMESERVER then
    szFilepath = "ServerSetting/Studio/StudioZhenYuan.tab"
    local tbZhenYuanStudioCfg = Lib:EasyLoadTabFile(szFilepath)
    if not tbZhenYuanStudioCfg then
      LogErr("%s not find", szFilepath)
      return false
    end
    for _, tbl in ipairs(tbZhenYuanStudioCfg) do
      if not ZhenYuan.tbZhenYuanStudioCfg[tbl.nPartnerId] then
        ZhenYuan.tbZhenYuanStudioCfg[tbl.nPartnerId] = {}
      end
      if not ZhenYuan.tbZhenYuanStudioCfg[tbl.nPartnerId][tbl.nOrder] then
        ZhenYuan.tbZhenYuanStudioCfg[tbl.nPartnerId][tbl.nOrder] = {}
      end
      table.insert(ZhenYuan.tbZhenYuanStudioCfg[tbl.nPartnerId][tbl.nOrder], tbl)
    end
  end
  for _, tbl in ipairs(tbZhenYuanConfig) do
    if not ZhenYuan.tbZhenYuanCfg[tbl.nPartnerId] then
      ZhenYuan.tbZhenYuanCfg[tbl.nPartnerId] = {}
    end
    table.insert(ZhenYuan.tbZhenYuanCfg[tbl.nPartnerId], tbl)
  end
  for _, tbl in ipairs(tbTmpZhenYuanHoleCfg) do
    if not ZhenYuan.tbZhenYuanAllHoleCfg[tbl.nHole] then
      ZhenYuan.tbZhenYuanAllHoleCfg[tbl.nHole] = {}
    end
    table.insert(ZhenYuan.tbZhenYuanAllHoleCfg[tbl.nHole], tbl)
  end
  for _, tbl in ipairs(tbZYHTAttrCfg) do
    if not ZhenYuan.tbHuTiAttrCfg[tbl.nGroupId] then
      ZhenYuan.tbHuTiAttrCfg[tbl.nGroupId] = {}
    end
    table.insert(ZhenYuan.tbHuTiAttrCfg[tbl.nGroupId], tbl)
  end
  for _, tbl in ipairs(tbZhenYuanExtAttriCfg) do
    if not ZhenYuan.tbZhenYuanExtAttriCfg[tbl.nGroupId] then
      ZhenYuan.tbZhenYuanExtAttriCfg[tbl.nGroupId] = {}
    end
    if not ZhenYuan.tbZhenYuanExtAttriCfg[tbl.nGroupId][tbl.nOrder] then
      ZhenYuan.tbZhenYuanExtAttriCfg[tbl.nGroupId][tbl.nOrder] = {}
    end
    table.insert(ZhenYuan.tbZhenYuanExtAttriCfg[tbl.nGroupId][tbl.nOrder], tbl)
  end
  for _, tbValue in ipairs(tbZhenYuanCostCfg) do
    ZhenYuan.tbZhenYuanCostSeriesMap[tbValue.nSeries] = tbValue
  end
  ZhenYuan.tbZhenYuanItemCfg = tbZhenYuanItemCfg

function ZhenYuan:GetGalleryCfg()
  if not self.tbGalleryCfg then
    self.tbGalleryCfg = {}
    for nTemplateId, tbV in pairs(self.tbZhenYuanItemCfg) do
      local nSeries = tbV.nSeries - self.nSeriesOffSet
      if not self.tbGalleryCfg[nSeries] then
        self.tbGalleryCfg[nSeries] = {}
      end
      if not self.tbGalleryCfg[nSeries][tbV.nOrder] then
        self.tbGalleryCfg[nSeries][tbV.nOrder] = {}
      end
      table.insert(self.tbGalleryCfg[nSeries][tbV.nOrder], tbV)
    end
  end
  for _, tbValue in ipairs(self.tbGalleryCfg) do
    for nOrder = 1, self:GetMaxOrder() do
      if tbValue[nOrder] then
        table.sort(tbValue[nOrder], function(tbA, tbB)
          local nQualityA = Item:GetQuality(tbA.nTemplateId)
          local nQualityB = Item:GetQuality(tbB.nTemplateId)
          if nQualityA ~= nQualityB then
            return nQualityA > nQualityB
          end
          return tbA.nFightPower > tbB.nFightPower
        end)
      end
    end
  end
  return self.tbGalleryCfg
end

function ZhenYuan:IsSystemOpen(pPlayer)
  pPlayer = pPlayer or me
  if pPlayer.nLevel < Item.ZhenYuan.nPlayerOpenLevel then
    return false, i18n.EasyGet("\231\142\169\229\174\182\231\173\137\231\186\167\228\184\141\232\182\179")
  end
  if 1 ~= GetTimeFrameState(Item.ZhenYuan.szOpenTimeframe) then
    return false, i18n.EasyGet("\231\179\187\231\187\159\229\176\154\230\156\170\229\188\128\229\144\175")
  end
  return true
end

function ZhenYuan:GetMaxOrder()
  local nMaxLevel = 0
  for k, v in ipairs(Item.ZhenYuan.tbMaxLevel) do
    if 1 == GetTimeFrameState(v.szTimeFrame) then
      nMaxLevel = v.nLevel
    end
  end
  return nMaxLevel
end

function ZhenYuan:GetPlayerMaxOrder(pPlayer)
  local nSystemMaxnOrder = 0
  local nPlayernOrder = 0
  for k, v in ipairs(Item.ZhenYuan.tbMaxLevel) do
    if 1 == GetTimeFrameState(v.szTimeFrame) then
      nSystemMaxnOrder = v.nLevel
    end
    if pPlayer.nTotalLevel >= v.RequireLevel then
      nPlayernOrder = v.nLevel
    end
  end
  return math.min(nPlayernOrder, nSystemMaxnOrder)
end

function ZhenYuan:GetMaxStar()
  local nSystemMaxnStar = 0
  for k, v in ipairs(Item.ZhenYuan.tbMaxLevel) do
    if 1 == GetTimeFrameState(v.szTimeFrame) then
      nSystemMaxnStar = v.MaxStar
    end
  end
  return nSystemMaxnStar
end

function ZhenYuan:CanCompound(pPlayer, nPartnerId, nOrder, nCompound)
  if not (pPlayer and Lib:IsInteger(nPartnerId) and Lib:IsInteger(nOrder)) or not Lib:IsInteger(nCompound) then
    return false
  end
  if nCompound <= 0 then
    return false
  end
  local bRet, szErr = self:IsSystemOpen(pPlayer)
  if not bRet then
    return false, szErr
  end
  if nOrder > self:GetPlayerMaxOrder(pPlayer) then
    return false, i18n.EasyGet("\229\135\157\232\129\154\231\154\132\231\156\159\229\133\131\232\182\133\232\191\135\229\189\147\229\137\141\230\156\128\229\164\167\231\169\191\230\136\180\231\173\137\233\152\182")
  end
  local pPartner = pPlayer.GetPartnerObj(nPartnerId)
  if not pPartner then
    return false, i18n.EasyGet("\230\130\168\230\156\170\230\191\128\230\180\187\232\175\165\229\144\140\228\188\180")
  end
  local nLeftBag = pPlayer.GetFreeBagCount()
  if nLeftBag <= 0 then
    return false, i18n.EasyGet("\232\131\140\229\140\133\231\169\186\233\151\180\228\184\141\232\182\179")
  end
  local tbCfg = self:GetCompoundCfg(pPartner.nTemplateId, nOrder)
  if not (tbCfg and Lib:IsInteger(tbCfg.nConsumeNum)) or not Lib:IsInteger(tbCfg.nYuanQiNum) then
    return false
  end
  if tbCfg.nConsumeNum * nCompound > pPlayer.GetItemCountByTypeInBag(tbCfg.nMaterialsId) then
    return false, i18n.EasyGet("\230\157\144\230\150\153\228\184\141\232\182\179")
  end
  if pPlayer.GetMoney(Currency.TYPE.ZhenQi) < tbCfg.nYuanQiNum * nCompound then
    return false, i18n.EasyGet("\229\133\131\230\176\148\228\184\141\232\182\179")
  end
  return true
end

function ZhenYuan:CanHunDunEnhance(pPlayer, nPos, nCurrencyType)
  local bRet, szErr = self:IsSystemOpen(pPlayer)
  if not bRet then
    return false, szErr
  end
  if nPos < Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL or nPos > Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH then
    return false, i18n.EasyGet("\229\143\130\230\149\176\233\148\153\232\175\175")
  end
  if pPlayer.IsMaskPlayerAttr() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\228\184\141\229\143\175\229\188\186\229\140\150")
  end
  local pEquip = Item.ExtEquip:GetEquipObj(pPlayer, nPos)
  if not pEquip or pEquip.dwId <= 0 then
    return false, i18n.EasyGet("\229\176\154\230\156\170\231\169\191\230\136\180\232\163\133\229\164\135")
  end
  local nIntensifyLv = self:GetHoleEnhanLevel(pPlayer, nPos)
  local tbEnhanceCfg = self:GetHoleEnhanceCfg(nPos, nIntensifyLv + 1)
  if not tbEnhanceCfg then
    return false, i18n.EasyGet("\229\183\178\232\190\190\229\188\186\229\140\150\228\184\138\233\153\144")
  end
  local nMaxStar = self:GetMaxStar()
  if nIntensifyLv >= nMaxStar * 10 then
    return false, i18n.EasyGet("\229\183\178\232\190\190\229\136\176\230\151\182\233\151\180\232\189\180\229\143\175\229\141\135\231\186\167\228\184\138\233\153\144")
  end
  local tbCostCfg = self.tbZhenYuanCostCfg[nCurrencyType]
  if not tbCostCfg then
    return false
  end
  if pPlayer.GetMoney(tbCostCfg.szCurrencyType) < tbEnhanceCfg.nMaterialsNum then
    return false, i18n.EasyGet("\229\133\131\231\159\179\228\184\141\232\182\179")
  end
  if pPlayer.GetMoney(Currency.TYPE.Coin) < tbEnhanceCfg.nSilver then
    return false, i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179")
  end
  return true
end

function ZhenYuan:CanHTEnhance(pPlayer, nGroupId, nCurrencyType)
  local bRet, szErr = self:IsSystemOpen(pPlayer)
  if not bRet then
    return false, szErr
  end
  if pPlayer.IsMaskPlayerAttr() then
    return false, i18n.EasyGet("\229\189\147\229\137\141\231\138\182\230\128\129\228\184\141\229\143\175\229\188\186\229\140\150")
  end
  bRet = self:IsHuTiActiveByGroupId(pPlayer, nGroupId)
  if not bRet then
    return false, i18n.EasyGet("\232\175\183\229\133\136\232\163\133\229\164\135\231\156\159\229\133\131")
  end
  local nOldLv = self:GetHuTiEnhanceLevel(pPlayer, nGroupId)
  local tbHTCfg = self:GetHTGroupCfg(nGroupId, nOldLv + 2)
  if not tbHTCfg then
    return false
  end
  local nCurStar = self:GetHuTiStar(pPlayer)
  if nCurStar >= self:GetMaxStar() then
    return false, i18n.EasyGet("\229\183\178\232\190\190\229\136\176\230\151\182\233\151\180\232\189\180\229\143\175\229\141\135\231\186\167\228\184\138\233\153\144")
  end
  if nCurStar < tbHTCfg.nNeedPreStar then
    return false, i18n.EasyGet("\232\175\183\229\133\136\230\143\144\229\141\135\229\133\182\228\187\150\228\186\148\232\161\140\231\155\184\231\148\159\231\130\188\229\140\150\231\173\137\231\186\167\239\188\140\228\189\191\230\138\164\228\189\147\231\156\159\229\133\131\229\141\135\230\152\159\229\144\142\230\150\185\229\143\175\231\187\167\231\187\173\231\130\188\229\140\150")
  end
  local tbCostCfg = self.tbZhenYuanCostCfg[nCurrencyType]
  if not tbCostCfg then
    return false
  end
  if pPlayer.GetMoney(tbCostCfg.szCurrencyType) < tbHTCfg.nMaterialsNum then
    return false, i18n.EasyGet("\229\133\131\231\159\179\228\184\141\232\182\179")
  end
  if pPlayer.GetMoney(Currency.TYPE.Coin) < tbHTCfg.nSilver then
    return false, i18n.EasyGet("\233\147\182\228\184\164\228\184\141\232\182\179")
  end
  return true
end

function ZhenYuan:CanUpStar(pPlayer)
  local bRet, szErr = self:IsSystemOpen(pPlayer)
  if not bRet then
    return false
  end
  if pPlayer.IsMaskPlayerAttr() then
    return false
  end
  local nCurStar = self:GetHuTiStar(pPlayer)
  if not Item.ZhenYuan.tbHTStar[nCurStar + 1] then
    return false
  end
  local nNeedLevel = Item.ZhenYuan.tbHTStar[nCurStar + 1]
  bRet = true
  for nGroupId, _ in pairs(self.tbHuTiAttrCfg) do
    local nLv = self:GetHuTiEnhanceLevel(pPlayer, nGroupId)
    if nNeedLevel > nLv then
      bRet = false
      break
    end
  end
  if not bRet then
    return false
  end
  return true
end

function ZhenYuan:IsHuTiActiveByGroupId(pPlayer, nGroupId)
  local tbHTCfg = self:GetHTGroupCfg(nGroupId, 1)
  if not tbHTCfg then
    return false
  end
  for _, nSeries in ipairs(tbHTCfg.tbnSeries) do
    local pEquip = Item.ExtEquip:GetEquipObj(pPlayer, nSeries)

function ZhenYuan:GetHTAttr(pPlayer, nPos)
  local tbActivatSeries = {}
  for nEquipPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local pEquip = Item.ExtEquip:GetEquipObj(pPlayer, nEquipPos)
    if pEquip and pEquip.dwId > 0 then
      tbActivatSeries[nEquipPos] = 1
    end
  end
  local tbActivatAttr = {}
  for nGId, tbValue in ipairs(self.tbHuTiAttrCfg) do
    local bRet = true
    local bIsNeed = false
    for _, nValue in ipairs(tbValue[1].tbnSeries) do
      if nValue ~= nPos and not tbActivatSeries[nValue] then
        bRet = false
        break
      end
      if not nPos or nValue == nPos then
        bIsNeed = true
      end
    end
    if bRet and bIsNeed then
      local nHtLv = self:GetHuTiEnhanceLevel(pPlayer, nGId)
      table.insert(tbActivatAttr, {nGroupId = nGId, lv = nHtLv})
    end
  end
  return tbActivatAttr
end

end

function tbItem:GetRandomAttrInfo(pItem)
  local tbRet = {}
  for i = 1, 10 do
    local nAttrGroupId = pItem.GetIntValue((i - 1) * Item.ZhenYuan.ATTR_OFFSET + ZhenYuan.ATTR_POWER)
    local nAttrIndex = pItem.GetIntValue((i - 1) * Item.ZhenYuan.ATTR_OFFSET + ZhenYuan.ATTR_POWER + 1)
    if nAttrGroupId > 0 and nAttrIndex > 0 then
      local tbV = {}
      for j = 1, 3 do
        tbV[j] = pItem.GetIntValue((i - 1) * Item.ZhenYuan.ATTR_OFFSET + j + ZhenYuan.ATTR_POWER + 1)
      end
      tbRet[i] = {
        nAttrGroupId,
        nAttrIndex,
        tbV
      }
    end
  end
  return tbRet
end

function ZhenYuan:GetEquipPower(pItem, pPlayer, nPos)
  if not self:IsZhenYuanEquip(pItem) then
    return 0
  end
  local nEquipPower = pItem.GetIntValue(0)
  local nHolePower = 0
  if pPlayer and nPos then
    local nLevel = self:GetHoleEnhanLevel(pPlayer, nPos)
    local tbHoleCfg = self:GetHoleEnhanceCfg(nPos, nLevel)
    if tbHoleCfg then
      nHolePower = tbHoleCfg.FightPower
    end
  end
  return math.floor(nEquipPower + nHolePower)
end

  local tbHoleCfg = self:GetHoleEnhanceCfg(nPos, nLevel)

function ZhenYuan:ApplyAttrib(pPlayer, pEquip)
  self:OnUseOrUnuse(pPlayer.GetNpc(), pEquip, true, pPlayer.dwID)
  local nSlot = Item:GetExtEquipSlot(pEquip)
  local nLevel = self:GetHoleEnhanLevel(pPlayer, nSlot)
  self:ApplyHDZYAttrib(pPlayer.GetNpc(), nSlot, nLevel, pPlayer.dwID)
  self:AddSeriesXiangShengAttr(pPlayer, pEquip)
end

function ZhenYuan:AddSeriesXiangShengAttr(pPlayer, pEquip)
  if Item.bAttrApplyLock then
    return
  end
  local nSlot = Item:GetExtEquipSlot(pEquip)
  local tbAttrList = self:GetHTAttr(pPlayer, nSlot)
  if #tbAttrList > 0 and not self:IsHuTiActive(pPlayer) then
    self:SetHuTiActive(pPlayer, 1)
  end
  for _, tbValue in ipairs(tbAttrList) do
    self:ApplyHTAttrib(pPlayer.GetNpc(), tbValue.nGroupId, tbValue.lv, pPlayer.dwID)
  end
  if self:CheckHTSkillActve(pPlayer) then
    local nCurStar = self:GetHuTiStar(pPlayer)
    local nSkilllv = 1
    if nCurStar + 1 > #Item.ZhenYuan.nHTSkillLevel then
      nSkilllv = Item.ZhenYuan.nHTSkillLevel[#Item.ZhenYuan.nHTSkillLevel]
    else
      nSkilllv = Item.ZhenYuan.nHTSkillLevel[nCurStar + 1]
    end
    pPlayer.GetNpc().AddSkill(Item.ZhenYuan.nHTSkillId, nSkilllv)
  end
end

function ZhenYuan:ApplyAllItemAttrib(pPlayer)
  if not Item.bAttrApplyLock then
    for nGroupId = 1, 5 do
      local bRet = self:IsHuTiActiveByGroupId(pPlayer, nGroupId)
      if bRet then
        local nHtLv = self:GetHuTiEnhanceLevel(pPlayer, nGroupId)
        self:ApplyHTAttrib(pPlayer.GetNpc(), nGroupId, nHtLv, pPlayer.dwID)
      end
    end
  end
  if self:CheckHTSkillActve(pPlayer) then
    local nCurStar = self:GetHuTiStar(pPlayer)
    local nSkilllv = 1
    if nCurStar + 1 > #Item.ZhenYuan.nHTSkillLevel then
      nSkilllv = Item.ZhenYuan.nHTSkillLevel[#Item.ZhenYuan.nHTSkillLevel]
    else
      nSkilllv = Item.ZhenYuan.nHTSkillLevel[nCurStar + 1]
    end
    pPlayer.GetNpc().AddSkill(Item.ZhenYuan.nHTSkillId, nSkilllv)
  end
end

function ZhenYuan:RemoveAttrib(pPlayer, pEquip)

  local nSlot = Item:GetExtEquipSlot(pEquip)
  self:RemoveSeriesXiangShengAttr(pPlayer, pEquip)
  local nLevel = self:GetHoleEnhanLevel(pPlayer, nSlot)
  self:RemoveHDZYAttrib(pPlayer, nSlot, nLevel)
end

function ZhenYuan:RemoveSeriesXiangShengAttr(pPlayer, pEquip)
  if Item.bAttrRemoveLock then
    return
  end
  local nSlot = Item:GetExtEquipSlot(pEquip)
  local tbAttrList = self:GetHTAttr(pPlayer, nSlot)
  for _, tbValue in ipairs(tbAttrList) do
    self:RemoveHTAttrib(pPlayer, tbValue.nGroupId, tbValue.lv)
  end
  if self:CheckHTSkillActve(pPlayer, nSlot) then
    pPlayer.GetNpc().RemoveSkill(Item.ZhenYuan.nHTSkillId)
  end
end

function ZhenYuan:RemoveAllItemAttrib(pPlayer)
  if not Item.bAttrRemoveLock then
    for nGroupId = 1, 5 do
      local bRet = self:IsHuTiActiveByGroupId(pPlayer, nGroupId)
      if bRet then
        local nHtLv = self:GetHuTiEnhanceLevel(pPlayer, nGroupId)
        self:RemoveHTAttrib(pPlayer, nGroupId, nHtLv)
      end
    end
  end
  if self:CheckHTSkillActve(pPlayer) then
    local nCurStar = self:GetHuTiStar(pPlayer)
    pPlayer.GetNpc().RemoveSkill(Item.ZhenYuan.nHTSkillId, Item.ZhenYuan.nHTSkillLevel[nCurStar + 1])
  end
end

function ZhenYuan:GetMaxAndMinAttr(pEquip)
  local tbRet = {}
  local tbZYItemCfg = ZhenYuan.tbZhenYuanItemCfg[pEquip.dwTemplateId]
  if not tbZYItemCfg then
    return tbRet
  end
  local tbRandomAttr = tbItem:GetRandomAttrInfo(pEquip)
  for _, tbV in ipairs(tbRandomAttr) do
    local tbAttrCfg = self:GetExtAttrCfg(tbV[1], tbZYItemCfg.nOrder, tbV[2])
    if tbAttrCfg then
      local tbMin = {}
      local tbMax = {}
      for i = 1, 3 do
        tbMin[i] = tbAttrCfg["nAttrValueMin" .. i]
        tbMax[i] = tbAttrCfg["nAttrValueMax" .. i]
      end
      local szAttrType, _ = FightSkill:GetMagicDescSplit(tbAttrCfg.szAttrType, tbV[3])
      tbRet[szAttrType] = {tbMin, tbMax}
    end
  end
  return tbRet
end

function ZhenYuan:GetAttrColor(tbAttrCfg, tbAttrValue)
  local tbColor = {
    "NewBlue",
    "NewPurple",
    "NewOrange"
  }
  if #tbColor ~= #tbAttrCfg.tbnColorList then
    return tbColor[1]
  end
  local nValue = tbAttrValue[tbAttrCfg.nCompareIndex]
  for index, value in ipairs(tbAttrCfg.tbnColorList) do
    if nValue <= tbAttrCfg.nReference * value / 100 then
      return tbColor[index]
    end
  end
end

function ZhenYuan:OnUseOrUnuse(pNpc, pEquip, bIsUseApply, nPlayerId)
  for _, tbValue in ipairs(self:GetEquipRandomAttr(pEquip)) do
    self:ApplyOrRemoveAnAttributeByIdentify(pNpc, tbValue.szAttrType, tbValue.tbAttrValue, bIsUseApply, nPlayerId)
  end
  local nSkillId = self.tbZhenYuanItemCfg[pEquip.dwTemplateId].nSkillId
  local nSkillLv = self.tbZhenYuanItemCfg[pEquip.dwTemplateId].nSkillLv
  if nSkillId and nSkillLv and nSkillId > 0 and nSkillLv > 0 then
    self:ApplyOrRemoveSkill(pNpc, nSkillId, nSkillLv, bIsUseApply)
  end
end

function ZhenYuan:ApplyOrRemoveSkill(pNpc, nSkillId, nSkillLevel, bIsApply)
  if bIsApply then
    pNpc.AddSkill(nSkillId, nSkillLevel)
  else
    pNpc.RemoveSkill(nSkillId)
  end
end

function ZhenYuan:ApplyOrRemoveAnAttributeByIdentify(pNpc, szIdentify, tbValue, bIsApply, nPlayerId)
  if bIsApply then
    pNpc.ApplyMagicAttrib(szIdentify, tbValue[1], tbValue[2], tbValue[3])
    if nPlayerId then
      AttributeHelp:LogAttrApply(nPlayerId, "ZhenYuan", szIdentify, tbValue[1], tbValue[2], tbValue[3])
    end
  else
    pNpc.RemoveMagicAttrib(szIdentify, tbValue[1], tbValue[2], tbValue[3])
    if nPlayerId then
      AttributeHelp:LogAttrRemove(nPlayerId, "ZhenYuan", szIdentify, tbValue[1], tbValue[2], tbValue[3])
    end
  end
end

function ZhenYuan:CheckEquipComposeCfg()
  for _, tbListInfo in pairs(self.tbZhenYuanCfg) do
    local nOldOrder = 1
    for _, tbInfo in pairs(tbListInfo) do
      if nOldOrder ~= tbInfo.nOrder then
        LogErr(string.format("load CheckEquipComposeCfg.tab failed, nPartnerId %d  nOrder %d error ", tbInfo.nPartnerId, tbInfo.nOrder))
        return false
      end
      nOldOrder = nOldOrder + 1
      for _, nTemplateId in ipairs(tbInfo.tbnItemId) do
        if not Item:IsExist(nTemplateId) then
          LogErr(string.format("load CheckEquipComposeCfg.tab failed, nTemplateId %d not exist", nTemplateId))
          return false
        end
        if not self.tbZhenYuanItemCfg[nTemplateId] then
          LogErr(string.format("load CheckEquipComposeCfg.tab failed, tbZhenYuanItemCfg not find nTemplateId %d", nTemplateId))
          return false
        end
      end
      local nTotal = 0
      for _, nWeight in ipairs(tbInfo.tbnWeight) do
        nTotal = nTotal + nWeight
      end
      if nTotal < Award.MaxRandValue then
        return false
      end
    end
  end
  return true
end

function ZhenYuan:CheckHoleEnhanceCfg()
  for _, tbListInfo in pairs(self.tbZhenYuanAllHoleCfg) do
    local nOldLv = 1
    for _, tbInfo in pairs(tbListInfo) do
      if nOldLv ~= tbInfo.nEnhanceLevel then
        LogErr(string.format("load CheckHoleEnhanceCfg.tab failed, nHole %d  nEnhanceLevel %d error ", tbInfo.nHole, tbInfo.nEnhanceLevel))
        return false
      end
      nOldLv = nOldLv + 1
    end
  end
  return true
end

function ZhenYuan:CheckZhenYuanSkillCfg()
  for _, tbInfo in pairs(self.tbZhenYuanItemCfg) do
    if tbInfo.nSkillId and not FightSkill.GetSkillUpInfo(tbInfo.nSkillId, tbInfo.nSkillLv) then
      LogErr(string.format("load CheckZhenYuanSkillCfg failed, nSkillId %d error ", tbInfo.nSkillId))
      return false
    end
  end
  return true
end

function ZhenYuan:onApplyRobotNpcAllEquip(pPlayerAsync, pNpc, tbEquipList, tbExtEquipList, tbSeries)
  for i = 1, 5 do
    local bRet = self:IsHuTiActiveByGroupIdAndNpc(tbExtEquipList, i)
    if bRet then
      local nLv = pPlayerAsync.GetZhenYuanHutiLevel(i)
      self:ApplyHTAttrib(pNpc, i, nLv)
    end
  end
  for i = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local pEquip = tbExtEquipList[i]
    if pEquip then
      local nLv = pPlayerAsync.GetZhenYuanHoleLevel(i - Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL + 1)
      self:ApplyHDZYAttrib(pNpc, i, nLv)
    end
  end
  local nCurStar = pPlayerAsync.GetZhenYuanHutiStar()
  local nSkilllv = 1
  if nCurStar + 1 > #Item.ZhenYuan.nHTSkillLevel then
    nSkilllv = Item.ZhenYuan.nHTSkillLevel[#Item.ZhenYuan.nHTSkillLevel]
  else
    nSkilllv = Item.ZhenYuan.nHTSkillLevel[nCurStar + 1]
  end
  pNpc.AddSkill(self.nHTSkillId, nSkilllv)
  for nSlot = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local pEquip = tbExtEquipList[nSlot]
    if pEquip then
      self:OnUseOrUnuse(pNpc, pEquip, true)
    end
  end
end

function ZhenYuan:CalcFightPower(pPlayer)
  local nTotalPower = 0
  for nEquipPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local pEquip = Item.ExtEquip:GetEquipObj(pPlayer, nEquipPos)
    if pEquip then
      nTotalPower = nTotalPower + self:GetEquipPower(pEquip, pPlayer, nEquipPos)
    end
  end
  for i = 1, 5 do
    if self:IsHuTiActiveByGroupId(pPlayer, i) then
      local nLevel = self:GetHuTiEnhanceLevel(pPlayer, i)
      local tbHuTiCfg = self:GetHTGroupCfg(i, nLevel + 1)
      if tbHuTiCfg then
        nTotalPower = nTotalPower + tbHuTiCfg.FightPower
      end
    end
  end
  return nTotalPower
end
