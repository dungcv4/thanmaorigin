local tbItem = Item:GetClass("ZhenyuanMaterialsItem")

function tbItem:GetUseSetting(nItemTemplateId, nItemId)
  if not nItemId or not KItem.GetItemObj(nItemId) then
    return
  end
  
  local function fnCompose()
    if Item.ZhenYuan:IsSystemOpen(me) then
      Partner:OpenOtherUI("UIZhenyuanEnhance", Item.ZhenYuan.PANEL_TYPE_NINGJU)
    else
      me.CenterMsg(i18n.EasyGet("\229\175\187\232\174\191\232\191\152\230\156\170\229\188\128\230\148\190"))
    end
  end
  
  return {
    szFirstName = i18n.EasyGet("\228\189\191\231\148\168"),
    fnFirst = fnCompose
  }
end
