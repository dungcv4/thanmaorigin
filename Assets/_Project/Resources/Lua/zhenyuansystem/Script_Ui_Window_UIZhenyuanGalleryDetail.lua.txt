local tbWnd = Ui:CreateWindow("UIZhenyuanGalleryDetail")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  goUIItemGrid = "node/imgBG/UIItemGrid",
  RegularValue = "node/imgBG/Content/RegularValue",
  RegularTemplate = "node/imgBG/Content/RegularValue/Template",
  SpecialValue = "node/imgBG/Content/SpecialValue",
  SpecialTemplate = "node/imgBG/Content/SpecialValue/Template",
  txtSpecial = "node/imgBG/Content/SpecialValue/Template/txtSpecial",
  AddTemplate = "node/imgBG/Content/AddValue/Viewport/Content/Add/Template",
  txtName = "node/imgBG/txtName",
  txtExtra = "node/imgBG/Content/AddValue/Title/txt"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self:InitAttrObj()
  self.tbItemGrid = Ui:NewExistItemGrid(self, tbControls.goUIItemGrid)
  self.tbExtraAttrScrollView = Ui:NewScrollView(self, tbControls.AddTemplate)
end

function tbWnd:OnOpen(tbData)
  self.tbData = tbData
  self.tbItemGrid:SetItemByTemplate(tbData.nTemplateId)
  self.tbItemGrid:SetClickEnable(false)
  self.pPanel:Label_SetText(tbControls.txtName, Item:GetName(tbData.nTemplateId, me))
  self:UpdateRegularAttr()
  self:UpdatePassiveSkill()
  self:UpdateExtraAttr()
end

function tbWnd:OnClose()
  self.tbData = nil
end

function tbWnd:InitAttrObj()
  for i = 1, 2 do
    local szKey = self.pPanel:CloneObject(tbControls.RegularTemplate, "Attr" .. i)
  end
  self.pPanel:SetActive(tbControls.RegularTemplate, false)
end

function tbWnd:UpdateRegularAttr()
  local tbAttrib = KItem.GetEquipBaseProp(self.tbData.nTemplateId).tbBaseAttrib
  local szAttribMsg1, szValue1 = FightSkill:GetMagicDescSplit(tbAttrib[1].szName, tbAttrib[1].tbValue)
  self.pPanel:Label_SetText(string.format("%s%s", tbControls.RegularValue, "/Attr1/txtValueName"), szAttribMsg1)
  self.pPanel:Label_SetText(string.format("%s%s", tbControls.RegularValue, "/Attr1/txtValue"), szValue1)
  local szAttribMsg2, szValue2 = FightSkill:GetMagicDescSplit(tbAttrib[2].szName, tbAttrib[2].tbValue)
  self.pPanel:Label_SetText(string.format("%s%s", tbControls.RegularValue, "/Attr2/txtValueName"), szAttribMsg2)
  self.pPanel:Label_SetText(string.format("%s%s", tbControls.RegularValue, "/Attr2/txtValue"), szValue2)
end

function tbWnd:UpdatePassiveSkill()
  local nSKillId = self.tbData.nSkillId
  if nSKillId and nSKillId > 0 then
    self.pPanel:SetActive(tbControls.SpecialValue, true)
    local szText = FightSkill:GetSkillMagicDesc(nSKillId, self.tbData.nSkillLv)
    self.pPanel:Label_SetText(tbControls.txtSpecial, szText)
  else
    self.pPanel:SetActive(tbControls.SpecialValue, false)
  end
end

function tbWnd:UpdateExtraAttr()
  local nExtraAttrNum = self.tbData.nExtAttrNum
  local nExtAttrGroupId = self.tbData.nExtAttrGroupId
  self.pPanel:Label_SetText(tbControls.txtExtra, i18n.EasyGet("*\233\154\143\230\156\186\232\142\183\229\190\151{0}\230\157\161\229\177\158\230\128\167", nExtraAttrNum))
  local tbExtAttrCfg = Item.ZhenYuan.tbZhenYuanExtAttriCfg[nExtAttrGroupId][self.tbData.nOrder]
  self.tbExtraAttrScrollView:Reset()
  for i, tbV in ipairs(tbExtAttrCfg) do
    self.tbExtraAttrScrollView:SetElement(i, function(tbElement)
      local pPanel = tbElement.pPanel
      local szAttribMsg, szValueMin = FightSkill:GetMagicDescSplit(tbV.szAttrType, {
        tbV.nAttrValueMin1,
        tbV.nAttrValueMin2,
        tbV.nAttrValueMin3
      })
      local _, szValueMax = FightSkill:GetMagicDescSplit(tbV.szAttrType, {
        tbV.nAttrValueMax1,
        tbV.nAttrValueMax2,
        tbV.nAttrValueMax3
      })
      pPanel:Label_SetText("txtValueName", szAttribMsg)
      pPanel:Label_SetText("txtValue", i18n.EasyGet("{0}~{1}", szValueMin, szValueMax))
    end)
  end
end
