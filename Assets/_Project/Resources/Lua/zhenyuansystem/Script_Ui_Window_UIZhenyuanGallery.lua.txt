local tbWnd = Ui:CreateWindow("UIZhenyuanGallery")

function tbWnd:NotifyEvents()
  return {}
end

local tbControls = {
  btnAll = "node/imgBG/btnList/btnAll",
  btnGold = "node/imgBG/btnList/btnGold",
  btnWood = "node/imgBG/btnList/btnWood",
  btnWater = "node/imgBG/btnList/btnWater",
  btnFire = "node/imgBG/btnList/btnFire",
  btnEarth = "node/imgBG/btnList/btnEarth",
  DropdownQuality = "node/imgBG/DropdownQuality",
  ZYScrollView = "node/imgBG/Content/ScrollView",
  Template = "node/imgBG/Content/ScrollView/Viewport/InfoContent/Template",
  btnClose = "node/imgBG/btnClose"
}
tbWnd.tbControls = tbControls
tbWnd.tbSeriesControls = {
  [Item.emSERIES_TYPE.SERIES_METAL] = "txtSeries/txtSeriesMetal",
  [Item.emSERIES_TYPE.SERIES_WOOD] = "txtSeries/txtSeriesWood",
  [Item.emSERIES_TYPE.SERIES_WATER] = "txtSeries/txtSeriesWater",
  [Item.emSERIES_TYPE.SERIES_FIRE] = "txtSeries/txtSeriesFire",
  [Item.emSERIES_TYPE.SERIES_EARTH] = "txtSeries/txtSeriesEarth"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    Ui:OpenWindow("UIZhenyuan")
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbSeriesGroup = Ui:NewButtonGroup(self, {
    [1] = tbControls.btnAll,
    [2] = tbControls.btnGold,
    [3] = tbControls.btnWood,
    [4] = tbControls.btnWater,
    [5] = tbControls.btnFire,
    [6] = tbControls.btnEarth
  }, true, function(self, nIndex)
    self.nSelectSeries = nIndex - 1
    self:UpdateZhenYuanSV()
  end)
  self.tbZhenYuanSV = Ui:NewSuperGridView(self, tbControls.Template, self.SetZhenYuanItem2Scroll, nil, self.OnZhenYuanClick)
  self:InitLevelDropDown()
  self.pPanel:Dropdown_BindEvent(tbControls.DropdownQuality, function(index)
    self.nDropIndex = index + 1
    self:UpdateZhenYuanSV()
  end)
end

function tbWnd:InitLevelDropDown()
  self.tbGalleryCfg = Item.ZhenYuan:GetGalleryCfg()
  self.nMaxGrade = Item.ZhenYuan:GetMaxOrder()
  local tbGradeOptions = {}
  for i = 1, self.nMaxGrade do
    table.insert(tbGradeOptions, i18n.EasyGet("{0}\233\152\182", i))
  end
  self.pPanel:Dropdown_ResetOptions(tbControls.DropdownQuality, tbGradeOptions)
end

function tbWnd:OnOpen()
  self.nSelectSeries = 0
  self.nDropIndex = Item.ZhenYuan:GetMaxOrder()
  self.tbSeriesGroup:Select(1)
  self.pPanel:Dropdown_SetValue(tbControls.DropdownQuality, self.nMaxGrade - 1)
end

function tbWnd:OnClose()
end

function tbWnd:UpdateZhenYuanSV()
  self.tbSVData = {}
  if 0 == self.nSelectSeries then
    for nSeries, tbV1 in ipairs(self.tbGalleryCfg) do
      Lib:MergeTable(self.tbSVData, tbV1[self.nDropIndex])
    end
  else
    Lib:MergeTable(self.tbSVData, self.tbGalleryCfg[self.nSelectSeries][self.nDropIndex])
  end
  self.tbZhenYuanSV:SetAllElement(self.tbSVData)
end

function tbWnd:OnZhenYuanClick(tbElement, tbData, nIndex)
  Ui:OpenWindow("UIZhenyuanGalleryDetail", tbData)
end

function tbWnd:SetZhenYuanItem2Scroll(tbElement, tbData, nIndex)
  local pPanel = tbElement.pPanel
  pPanel:Label_SetText("txtZhenYuanName", Item:GetName(tbData.nTemplateId, me))
  local itemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
  itemGrid:SetItemByTemplate(tbData.nTemplateId)
  itemGrid:SetClickEnable(true, function()
    self:OnZhenYuanClick(tbElement, tbData, nIndex)
  end)
end
