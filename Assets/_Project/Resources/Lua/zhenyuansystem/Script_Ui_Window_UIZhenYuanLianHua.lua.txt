    self:ClickSeriesXiangSheng(tbSeriesXiangSheng.ZHUIYING)
  end,
  btnGotoEquip = function(self)
    Ui:OpenWindow("UIZhenyuanEquip", tbSeriesMap[self.nSelectType].nPos)
  end,
  btnRecover = function(self)
    Ui:OpenWindow("UIItemBatchSellPanel", 5)
  end,
  btnEnhance = function(self)
    if not self:CheckMoney() then
      return
    end
    if not self.bIsAuto then
      if not tbSeriesMap[self.nSelectType].nPos then
        Item.ZhenYuan:HuTiEnhance(self.nSelectXiangSheng, self.nSelectCostIndex)
      elseif self.nSelectType ~= self.nSelectCostIndex then
        self:LianHuaPopUpWindow(tbSeriesMap[self.nSelectType].nPos, self.nSelectCostIndex)
      else
        Item.ZhenYuan:ZhenYuanEnhance(tbSeriesMap[self.nSelectType].nPos, self.nSelectCostIndex)
      end
      return
    end
    if self.nAutoLianHuaTimer then
      self:StopLianHua()
    else
      local nSelectType = self.nSelectType
      local nSelectXiangSheng = self.nSelectXiangSheng
      local nSelectCostIndex = self.nSelectCostIndex
      self.pPanel:Label_SetText(tbControls.btnEnhance .. "/Text", i18n.EasyGet("\229\129\156\230\173\162"))
      self.pPanel:SetActive(tbControls.btnStopAuto, true)
      self.nAutoLianHuaTimer = Timer:Register(math.floor(Env.GAME_FPS * 0.5), self.AutoLianHua, self, nSelectType, nSelectXiangSheng, nSelectCostIndex)
    end
  end
}

function tbWnd:OnCreate()
  self.tbEquipPos2ItemGrid = {}
  self.nSelectType = 1
  self.nSelectXiangSheng = 1
  self.nSelectCostIndex = 1
  self.nSuccessEffectTimer = nil
  self.nUpLevelEffectTimer = nil
  self.nUpStarEffectTimer = nil
  self.nAutoLianHuaTimer = nil
  self:NewSeriesAttrList()
  self:InitSelectCost()
  self:RegistRedPoint()
  self.pPanel:SetActive(tbControls.Baoji_Effect_02, false)
  self.pPanel:SetActive(tbControls.Baoji_Effect_05, false)
  self.pPanel:SetActive(tbControls.LevelUp_Effect, false)
  self.tbBaseScrollView = Ui:NewScrollView(self, tbControls.BaseAttrTemplate, nil)
  self.tbExtScrollView = Ui:NewScrollView(self, tbControls.ExtAttrTemplate, nil)
  self.tbSkillScrollView = Ui:NewScrollView(self, tbControls.SkillTemplate, nil)
  self.tbCostScrollView = Ui:NewScrollView(self, tbControls.CostViewport, nil)
  local obj = self.pPanel:GetObject(tbControls.imgModel)
  self.NpcModel = obj:GetComponent(Ui.UINpcModel)
  Ui:Toggle_UnBindEvent(self.pPanel, tbControls.togEnhanceAuto)
  self.pPanel:Toggle_SetChecked(tbControls.togEnhanceAuto, self.bIsAuto)
  Ui:Toggle_BindEvent(self.pPanel, tbControls.togEnhanceAuto, function(self, isOn)
    self.pPanel:Toggle_SetChecked(tbControls.togEnhanceAuto, isOn)
    self.bIsAuto = isOn
  end, self)
end

function tbWnd:OnOpen()
  self:UpdateEquipInfo()
  self.nSelectXiangSheng = 1
  self.nSelectCostIndex = 1
  self.bIsAuto = false
  self:RefreshSeriesAttr(1)
  self.pPanel:SetActive(tbControls.StarUpgrade, false)
  self.pPanel:SetActive(tbControls.Cost, true)
  self.pPanel:SetActive(tbControls.btnHuti, Item.ZhenYuan:IsHuTiActive(me))
  self.pPanel:SetActive(tbControls.Viewport, true)
  self:RefreshUpIcon()
end

function tbWnd:OnClose()
  self.pPanel:SetActive(tbControls.btnHuti, false)
  self.pPanel:SetActive(tbControls.Viewport, false)
  self.pPanel:SetActive(tbControls.imgModel, false)
  self.pPanel:SetActive(tbControls.EmptyPanel, false)
  self.pPanel:SetActive(tbControls.InfoPanel, false)
  self.pPanel:SetActive(tbControls.HutiPanel, false)
  self.pPanel:SetActive(tbControls.SeriesPanel, false)
  self.pPanel:SetActive(tbControls.Cost, false)
  self:HiddenModel()
  self.pPanel:SetActive(tbControls.Baoji_Effect_02, false)
  self.pPanel:SetActive(tbControls.Baoji_Effect_05, false)
  self.pPanel:SetActive(tbControls.LevelUp_Effect, false)
  if self.nSuccessEffectTimer then
    Timer:Close(self.nSuccessEffectTimer)
  end
  if self.nUpLevelEffectTimer then
    Timer:Close(self.nUpLevelEffectTimer)
  end
  if self.nUpStarEffectTimer then
    Timer:Close(self.nUpStarEffectTimer)
  end
  if self.nAutoLianHuaTimer then
    Timer:Close(self.nAutoLianHuaTimer)
  end
  self.nSuccessEffectTimer = nil
  self.nUpLevelEffectTimer = nil
  self.nUpStarEffectTimer = nil
  self.nAutoLianHuaTimer = nil
end

function tbWnd:RegistRedPoint()
  for nPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbZhenYuanRedPosKey[nPos], {
      Item.ZhenYuan.tbEquipPosRedPoint[nPos].nLianHuaRed
    })
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnHuti, {
    RedPointMgr.EVENT.ZhenYuanLianHuaHuTi
  })
end

function tbWnd:AutoLianHua(nSelectType, nSelectXiangSheng, nSelectCostIndex)
  if not tbSeriesMap[nSelectType].nPos then
    Item.ZhenYuan:HuTiEnhance(nSelectXiangSheng, nSelectCostIndex)
  else
    Item.ZhenYuan:ZhenYuanEnhance(tbSeriesMap[nSelectType].nPos, nSelectCostIndex)
  end
  return true
end

function tbWnd:LianHuaPopUpWindow(nPos, nCostIndex)
  local nLastTime = LocalData:GetRoleNumber("AutoMedicine_CloseZhenYuanLianHuaNoticeTime", 0)
  local bIsDiffDay = Lib:IsDiffDay(14400, nLastTime)
  
  local function fnOk()
    Item.ZhenYuan:ZhenYuanEnhance(nPos, nCostIndex)
  end
  
  if not bIsDiffDay then
    fnOk()
    return
  end
  local szNotice = i18n.EasyGet("\229\189\147\229\137\141\230\137\128\233\128\137\230\157\144\230\150\153\233\157\158\229\175\185\229\186\148\228\186\148\232\161\140\230\157\144\230\150\153\239\188\140\231\130\188\229\140\150\230\151\182\229\162\158\229\138\160\232\190\131\229\176\145\231\187\143\233\170\140\229\128\188\239\188\140\230\152\175\229\144\166\231\187\167\231\187\173\231\130\188\229\140\150")
  Ui:OpenWindow("UIMessageBox", szNotice, {
    {fnOk},
    {}
  }, {
    i18n.szOk,
    i18n.szCancel
  }, "ZhenYuan", nil, i18n.EasyGet("\228\187\138\230\151\165\228\184\141\229\134\141\230\143\144\233\134\146"))
end

function tbWnd:StopLianHua()
  if self.nAutoLianHuaTimer then
    Timer:Close(self.nAutoLianHuaTimer)
    self.nAutoLianHuaTimer = nil
    self.pPanel:SetActive(tbControls.btnStopAuto, false)
    self.pPanel:Label_SetText(tbControls.btnEnhance .. "/Text", i18n.EasyGet("\231\130\188\229\140\150"))
  end
end

function tbWnd:ShowModel()
  local tbFeature, tbEffect = me.GetNpcResInfo()
  self.pPanel:SetActive(tbControls.imgModel, true)
  self.nNpcModelDir = 0
  self.NpcModel:ShowModel(me.nSex, 1, tbFeature[Npc.NpcResPartsDef.npc_part_body], tbFeature[Npc.NpcResPartsDef.npc_part_weapon], tbFeature[Npc.NpcResPartsDef.npc_part_mantle], 0, tbFeature[Npc.NpcResPartsDef.npc_part_head], tbFeature[Npc.NpcResPartsDef.npc_part_wind], tbEffect[Npc.NpcResPartsDef.npc_part_weapon])
  self.NpcModel:SetAction(Npc.NpcAction.SitDown)
end

function tbWnd:ShowSuccessEffect(nPower)
  if not tbBaojiEffect[nPower] then
    return
  end
  if self.nUpLevelEffectTimer then
    Timer:Close(self.nUpLevelEffectTimer)
    self.pPanel:SetActive(tbControls.LevelUp_Effect, false)
  end
  self.pPanel:SetActive(tbControls.LevelUp_Effect, true)
  Sound:PlaySound("CastingWin", _)
  self.nUpLevelEffectTimer = Timer:Register(Env.GAME_FPS, function()
    self.pPanel:SetActive(tbControls.LevelUp_Effect, false)
    self.nUpLevelEffectTimer = nil
  end)
  if self.nSuccessEffectTimer then
    return
  end
  self.pPanel:SetActive(tbBaojiEffect[nPower], true)
  Sound:PlaySound("CastingWin", _)
  self.nSuccessEffectTimer = Timer:Register(Env.GAME_FPS, function()
    self.pPanel:SetActive(tbBaojiEffect[nPower], false)
    self.nSuccessEffectTimer = nil
  end)
end

function tbWnd:ShowStarEffect(nIndex)
  if nIndex > 9 then
    nIndex = nIndex % 10 + 1
  end
  local szPath = tbControls.Star .. "/Star" .. nIndex .. "/Stars_Effect"
  if self.nUpStarEffectTimer then
    Timer:Close(self.nUpStarEffectTimer)
    self.pPanel:SetActive(szPath, false)
  end
  self.pPanel:SetActive(szPath, true)
  Sound:PlaySound("CastingWin", _)
  self.nUpStarEffectTimer = Timer:Register(Env.GAME_FPS, function()
    self.pPanel:SetActive(szPath, false)
    self.nUpStarEffectTimer = nil
  end)
end

function tbWnd:RefreshSeriesAttr(nIndex)
  self.nSelectType = nIndex
  self:RefreshSelectbtn()
  self:RefreshSelectCost()
  self:RefreshCost()
  self:OnZhenYuanLianHuaInfo(nIndex)
  self:RefreshUpIcon()
  self:ShowStar(nIndex)
  local nPos = tbSeriesMap[nIndex].nPos
  local pEquip = Item.ExtEquip:GetEquipObj(me, nPos)
  if not pEquip then
    self.pPanel:SetActive(tbControls.InfoPanel, false)
    self.pPanel:SetActive(tbControls.EmptyPanel, true)
    return
  end
  self.pPanel:SetActive(tbControls.EmptyPanel, false)
  self.pPanel:SetActive(tbControls.InfoPanel, true)
  self.pPanel:SetActive(tbControls.HutiPanel, false)
  self.pPanel:SetActive(tbControls.SeriesPanel, true)
  local tbRet1 = Item.ZhenYuan:GetEquipBaseAttr(pEquip)
  local tbRet2 = Item.ZhenYuan:GetEquipRandomAttr(pEquip)
  local nIntensifyLv = Item.ZhenYuan:GetHoleEnhanLevel(me, nPos)
  local tbEnhanceCfg = Item.ZhenYuan:GetHoleEnhanceCfg(nPos, nIntensifyLv)
  local tbNextEnhanceCfg = Item.ZhenYuan:GetHoleEnhanceCfg(nPos, nIntensifyLv + 1)
  local tbLianHuaAttr = {}
  local tbNextLianHuaAttr = {}
  for i = 1, #tbRet1 do
    tbLianHuaAttr[tbRet1[i].szAttrType] = Lib:CopyTB(tbRet1[i].tbAttrValue)
    tbNextLianHuaAttr[tbRet1[i].szAttrType] = Lib:CopyTB(tbRet1[i].tbAttrValue)
  end
  if tbEnhanceCfg then
    for i = 1, tbEnhanceCfg.nAttrNum do
      local szAttrName = tbEnhanceCfg["szAttribType" .. i]
      local tbValue = tbEnhanceCfg["tbnAttribValue" .. i]
      local tbValue2 = tbLianHuaAttr[szAttrName]
      for index, value in ipairs(tbValue) do
        tbValue2[index] = tbValue2[index] + value
      end
      tbLianHuaAttr[szAttrName] = tbValue2
    end
  end
  if tbNextEnhanceCfg then
    for i = 1, tbNextEnhanceCfg.nAttrNum do
      local szAttrName = tbNextEnhanceCfg["szAttribType" .. i]
      local tbValue = tbNextEnhanceCfg["tbnAttribValue" .. i]
      local tbValue2 = tbNextLianHuaAttr[szAttrName]
      for index, value in ipairs(tbValue) do
        tbValue2[index] = tbValue2[index] + value
      end
      tbNextLianHuaAttr[szAttrName] = tbValue2
    end
  end
  self.tbBaseScrollView:Reset()
  self.tbExtScrollView:Reset()
  self.tbSkillScrollView:Reset()
  self.tbSkillScrollView:SetElement(1, function(tbElement)
    local tbItemCfg = Item.ZhenYuan.tbZhenYuanItemCfg[pEquip.dwTemplateId]
    if tbItemCfg and tbItemCfg.nSkillId and tbItemCfg.nSkillId > 0 then
      local szNowLevelDesc = FightSkill:GetSkillMagicDesc(tbItemCfg.nSkillId, tbItemCfg.nSkillLv)
      tbElement.pPanel:Label_SetText("txtSpecial", i18n.Parse(szNowLevelDesc))
    else
      tbElement.pPanel:Label_SetText("txtSpecial", i18n.EasyGet("\230\151\160"))
    end
    self.pPanel:ContentSizeFitter_Refresh(tbControls.Content)
  end)
  for i = 1, #tbRet1 do
    self.tbBaseScrollView:SetElement(i, function(tbElement)
      local szAttrType, szAttrName = FightSkill:GetMagicDescSplit(tbRet1[i].szAttrType, tbLianHuaAttr[tbRet1[i].szAttrType])
      local _, szNextAttrName = FightSkill:GetMagicDescSplit(tbRet1[i].szAttrType, tbNextLianHuaAttr[tbRet1[i].szAttrType])
      tbElement.pPanel:Label_SetText("txtValueName", i18n.Parse(szAttrType))
      tbElement.pPanel:Label_SetColorByName("txtValueName", "NewGreen")
      tbElement.pPanel:Label_SetText("txtValueBefore", i18n.Parse(szAttrName))
      tbElement.pPanel:Label_SetColorByName("txtValueBefore", "NewGreen")
      tbElement.pPanel:Label_SetText("txtValueAfter", i18n.Parse(szNextAttrName))
      tbElement.pPanel:Label_SetColorByName("txtValueAfter", "NewGreen")
      self.pPanel:ContentSizeFitter_Refresh(tbControls.Content)
    end)
  end
  for i = 1, #tbRet2 do
    self.tbExtScrollView:SetElement(i, function(tbElement)
      local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbRet2[i].szAttrType, tbRet2[i].tbAttrValue)
      tbElement.pPanel:Label_SetText("txtValueName", i18n.Parse(szAttrType))
      tbElement.pPanel:Label_SetColorByName("txtValueName", tbRet2[i].szColor)
      tbElement.pPanel:Label_SetText("txtValueBefore", i18n.Parse(szAttrValue))
      tbElement.pPanel:Label_SetColorByName("txtValueBefore", tbRet2[i].szColor)
      tbElement.pPanel:SetActive("txtValueAfter", false)
      tbElement.pPanel:SetActive("imgArrow", false)
      self.pPanel:ContentSizeFitter_Refresh(tbControls.Content)
    end)
  end
end

function tbWnd:ShowStar(nIndex)
  local nCurStar = 0
  if tbSeriesMap[nIndex].nPos then
    nCurStar = math.floor(Item.ZhenYuan:GetHoleEnhanLevel(me, tbSeriesMap[nIndex].nPos) / 10)
  else
    nCurStar = Item.ZhenYuan:GetHuTiStar(me)
  end
  local nStar2 = 0
  if nCurStar > 9 then
    nStar2 = nCurStar - 9
  end
  for i = 1, 9 do
    if i <= nCurStar then
      self.pPanel:SetActive(tbControls.Star .. "/Star" .. i, true)
      self.pPanel:Sprite_SetSprite(tbControls.Star .. "/Star" .. i, Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar1)
    else
      self.pPanel:SetActive(tbControls.Star .. "/Star" .. i, false)
    end
    if i <= nStar2 then
      self.pPanel:Sprite_SetSprite(tbControls.Star .. "/Star" .. i, Item.ZhenYuan.tbStarInfoImg.szImg_HuTiStar2)
    end
  end
end

function tbWnd:RefreshSeriesXiangShengAttr(nGroupId)
  local nLevel = Item.ZhenYuan:GetHuTiEnhanceLevel(me, nGroupId)
  local bIsAct = Item.ZhenYuan:CheckHuTiGroupActive(me, nGroupId)
  self.pPanel:Label_SetText(tbSeriesAttrLevelList[nGroupId], i18n.EasyGet("Lv." .. nLevel))
  local tbValue = Item.ZhenYuan.tbHuTiAttrCfg[nGroupId]
  local tbHTCfg = tbValue[nLevel + 1]
  if tbHTCfg then
    for i = 1, tbHTCfg.nAttrNum do
      local szAttrType, szAttrValue = FightSkill:GetMagicDescSplit(tbHTCfg["szAttribType" .. i], tbHTCfg["tbnAttribValue" .. i])
      self.pPanel:SetActive(tbSeriesAttrList[nGroupId], true)
      self.pPanel:Label_SetText(tbSeriesAttrType[nGroupId], i18n.Parse(szAttrType))
      self.pPanel:Label_SetColorByName(tbSeriesAttrType[nGroupId], bIsAct and "NewOrange" or "NewBlue")
      self.pPanel:Label_SetText(tbSeriesAttrValue[nGroupId], i18n.Parse(szAttrValue))
      self.pPanel:Label_SetColorByName(tbSeriesAttrValue[nGroupId], bIsAct and "NewOrange" or "NewBlue")
      local tbNextHTCfg = tbValue[nLevel + 2]
      if tbNextHTCfg and bIsAct then
        local _, szNextAttrValue = FightSkill:GetMagicDescSplit(tbNextHTCfg["szAttribType" .. i], tbNextHTCfg["tbnAttribValue" .. i])
        self.pPanel:SetActive(tbSeriesAttrAfterList[nGroupId], true)
        self.pPanel:Label_SetText(tbSeriesAttrAfterList[nGroupId], i18n.Parse(szNextAttrValue))
      else
        self.pPanel:SetActive(tbSeriesAttrAfterList[nGroupId], false)
      end
    end
    self.pPanel:Label_SetText(tbSeriesXiangShengName[nGroupId], i18n.Parse(tbHTCfg.szName))
  end
end

function tbWnd:UpdateEquipInfo()
  for nPos = Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL, Item.ExtEquip.CFG.SLOT.ZHENYUAN_EARTH do
    local pEquip = Item.ExtEquip:GetEquipObj(me, nPos)
    if pEquip then
      self.pPanel:SetActive(tbZhenYuanPosKey[nPos], true)
      local pItemGrid = self.tbEquipPos2ItemGrid[nPos]
      if not pItemGrid then
        pItemGrid = Ui:NewExistItemGrid(self, tbZhenYuanPosKey[nPos] .. "/UIItemGrid")
        self.tbEquipPos2ItemGrid[nPos] = pItemGrid
      end
      pItemGrid:SetItemByTemplate(pEquip.dwTemplateId, 1, me.nFaction, me.nFactionSect, {bDisableQualityEffect = true})
      pItemGrid:SetClickEnable(true, function()
        self.nSelectCostIndex = nPos - Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL + 1
        self:RefreshSeriesAttr(nPos - Item.ExtEquip.CFG.SLOT.ZHENYUAN_METAL + 1)
      end)
    else
      self.pPanel:SetActive(tbZhenYuanPosKey[nPos], false)
    end
  end
  return true
end

function tbWnd:OnHuTiLianHuaInfo(nSelectIndex)
  local nCurLevel = Item.ZhenYuan:GetHuTiEnhanceLevel(me, nSelectIndex)
  local nCurExp = Item.ZhenYuan:GetHuTiEnhanceExp(me, nSelectIndex)
  local tbValue = Item.ZhenYuan.tbHuTiAttrCfg[nSelectIndex]
  if not tbValue then
    return false
  end
  local tbHTCfg = tbValue[nCurLevel + 2]
  if not tbHTCfg then
    return false
  end
  local nNeedExp = tbHTCfg.nNeedExp
  local nCurStar = math.floor(nCurLevel / 10) + 1
  local tbHuTiCompensateExpCfg = Item.ZhenYuan:GetHuTiCompensateExpCfg()
  if tbHuTiCompensateExpCfg and tbHuTiCompensateExpCfg.CompensateExp[nCurStar] then
    nNeedExp = tbHuTiCompensateExpCfg.CompensateExp[nCurStar]
  end
  if nNeedExp == tbHTCfg.nNeedExp then
    self.pPanel:SetActive(tbControls.txtEXPBefore, false)
  else
    self.pPanel:SetActive(tbControls.txtEXPBefore, true)
    self.pPanel:Label_SetText(tbControls.txtEXPBefore, i18n.EasyGet("{0}", tbHTCfg.nNeedExp))
  end
  local nNeedCoin = tbHTCfg.nSilver
  self.pPanel:Label_SetText(tbControls.LianHuaName, i18n.EasyGet("\230\138\164\228\189\147\231\156\159\229\133\131"))
  self.pPanel:Label_SetText(tbControls.CurrentLevel, i18n.EasyGet("\231\173\137\231\186\167.{0}", nCurLevel))
  self.pPanel:Label_SetText(tbControls.SeriesSelect, i18n.Parse(tbHTCfg.szName))
  self.pPanel:Label_SetText(tbControls.TxtLevelExp, i18n.EasyGet("{0}/{1}", nCurExp, nNeedExp))
  self.pPanel:Sprite_SetFill(tbControls.ImgLevelProgress, nCurExp / nNeedExp)
  self.pPanel:Label_SetText(tbControls.TxtEnhanceCost, i18n.EasyGet("{0}", nNeedCoin))
end

function tbWnd:OnZhenYuanLianHuaInfo(nSelectIndex)
  local nPos = tbSeriesMap[nSelectIndex].nPos
  local pEquip = Item.ExtEquip:GetEquipObj(me, nPos)
  if not pEquip then
    return
  end
  local szItemName = Item:GetName(pEquip.dwTemplateId, me)
  local nCurLevel = Item.ZhenYuan:GetHoleEnhanLevel(me, nPos)
  local nCurExp = Item.ZhenYuan:GetHoleEnhanExp(me, nPos)
  local tbValue = Item.ZhenYuan:GetHoleEnhanceCfg(nPos, nCurLevel + 1)
  if not tbValue then
    return
  end
  local nNeedExp = tbValue.nNeedExp
  local nStar = math.floor(nCurLevel / 10) + 1
  local tbHunDunCompensateExpCfg = Item.ZhenYuan:GetHunDunCompensateExpCfg()
  if tbHunDunCompensateExpCfg and tbHunDunCompensateExpCfg.CompensateExp[nStar] then
    nNeedExp = tbHunDunCompensateExpCfg.CompensateExp[nStar]
  end
  self:HiddenModel()
  self.pPanel:SetActive(tbControls.Item, true)
  if not self.pItemGrid then
    self.pItemGrid = Ui:NewExistItemGrid(self, tbControls.UIItemGrid)
  end
  self.pItemGrid:SetItem(pEquip.dwId, 1)
  self.pItemGrid:SetClickEnable(true)
  if nNeedExp == tbValue.nNeedExp then
    self.pPanel:SetActive(tbControls.txtEXPBefore, false)
  else
    self.pPanel:SetActive(tbControls.txtEXPBefore, true)
    self.pPanel:Label_SetText(tbControls.txtEXPBefore, i18n.EasyGet("{0}", tbValue.nNeedExp))
  end
  local nNeedCoin = tbValue.nSilver or 0
  self.pPanel:Label_SetText(tbControls.LianHuaName, i18n.EasyGet("{0}", szItemName))
  self.pPanel:Label_SetText(tbControls.CurrentLevel, i18n.EasyGet("\231\173\137\231\186\167.{0}", nCurLevel))
  self.pPanel:Label_SetText(tbControls.SeriesSelect, tbSeriesMap[nSelectIndex].szSeriesName())
  self.pPanel:Label_SetText(tbControls.TxtLevelExp, i18n.EasyGet("{0}/{1}", nCurExp, nNeedExp))
  self.pPanel:Sprite_SetFill(tbControls.ImgLevelProgress, nCurExp / nNeedExp)
  self.pPanel:Label_SetText(tbControls.TxtEnhanceCost, i18n.EasyGet("{0}", nNeedCoin))
end

function tbWnd:InitSelectCost()
  for nIndex, szValue in ipairs(tbCostSelectList) do
    Ui:Toggle_UnBindEvent(self.pPanel, szValue)
    self.pPanel:Toggle_SetChecked(szValue, self.nSelectCostIndex == nIndex)
    Ui:Toggle_BindEvent(self.pPanel, szValue, function(self, isOn)
      if self.bRefreshSelectCost then
        return
      end
      if isOn then
        if tbCostSelectList[self.nSelectCostIndex] then
          self.pPanel:Toggle_SetChecked(tbCostSelectList[self.nSelectCostIndex], false)
        end
        self.nSelectCostIndex = nIndex
        self.pPanel:Toggle_SetChecked(szValue, true)
      else
        self.nSelectCostIndex = 0
      end
    end, self)
  end
end

function tbWnd:RefreshSelectCost()
  self.bRefreshSelectCost = true
  for nIndex, szValue in ipairs(tbCostSelectList) do
    if self.nSelectCostIndex == nIndex then
      self.pPanel:Toggle_SetChecked(szValue, true)
    elseif self.nSelectCostIndex ~= nIndex and self.pPanel:Toggle_GetChecked(szValue) then
      self.pPanel:Toggle_SetChecked(szValue, false)
    end
  end
  self.bRefreshSelectCost = false
end

function tbWnd:CheckMoney()
  local nNeedCoin = 0
  if tbSeriesMap[self.nSelectType].nPos then
    local nPos = tbSeriesMap[self.nSelectType].nPos
    local nCurLevel = Item.ZhenYuan:GetHoleEnhanLevel(me, nPos)
    local tbValue = Item.ZhenYuan:GetHoleEnhanceCfg(nPos, nCurLevel + 1)
    if not tbValue then
      return false
    end
    nNeedCoin = tbValue.nSilver or 0
  else
    local nCurLevel = Item.ZhenYuan:GetHuTiEnhanceLevel(me, self.nSelectXiangSheng)
    local tbValue = Item.ZhenYuan.tbHuTiAttrCfg[self.nSelectXiangSheng]
    if not tbValue then
      return false
    end
    local tbHTCfg = tbValue[nCurLevel + 2]
    if not tbHTCfg then
      return false
    end
    nNeedCoin = tbHTCfg.nSilver
  end
  local nMoneyNum = me.GetMoney("Coin")
  if nNeedCoin > nMoneyNum then
    GainCurrency:ReadyForJump("Coin")
    return false
  end
  return true
end

function tbWnd:RefreshCost()
  local nCostNum = 0
  if tbSeriesMap[self.nSelectType].nPos then
    local nLv = Item.ZhenYuan:GetHoleEnhanLevel(me, tbSeriesMap[self.nSelectType].nPos)
    local tbEnhanceCfg = Item.ZhenYuan:GetHoleEnhanceCfg(tbSeriesMap[self.nSelectType].nPos, nLv + 1)
    if not tbEnhanceCfg then
      return
    end
    nCostNum = tbEnhanceCfg.nMaterialsNum
  else
    local nLv = Item.ZhenYuan:GetHuTiEnhanceLevel(me, self.nSelectXiangSheng)
    local tbHutiCfg = Item.ZhenYuan:GetHTGroupCfg(self.nSelectXiangSheng, nLv + 2)
    if not tbHutiCfg then
      return
    end
    nCostNum = tbHutiCfg.nMaterialsNum
  end
  for nIndex, szValue in ipairs(tbCostItemList) do
    local tbCostCfg = Item.ZhenYuan.tbZhenYuanCostCfg[nIndex]
    if tbCostCfg then
      local nMoneyNum = me.GetMoney(tbCostCfg.szCurrencyType)
      self.pPanel:Label_SetText(szValue, i18n.EasyGet("{0}/{1}", nMoneyNum, nCostNum))
    end
  end
end

function tbWnd:RefreshUpIcon()
  if tbSeriesMap[self.nSelectType].nPos then
    for index, szValue in ipairs(tbCostItemUpIcon) do
      self.pPanel:SetActive(szValue, index == self.nSelectType)
    end
  else
    for _, szValue in ipairs(tbCostItemUpIcon) do
      self.pPanel:SetActive(szValue, true)
    end
  end
end
