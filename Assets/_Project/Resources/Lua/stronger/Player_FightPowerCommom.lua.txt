FightPower.tbKey = {
  Level = true,
  Skill = true,
  Attribute = true,
  Equip = true,
  EquipForge = true,
  EquipJingZhu = true,
  Stone = true,
  HonorLevel = true,
  KinSkill = true,
  QiHun = true
}
FightPower.tbFightType2SystemModule = FightPower.tbFightType2SystemModule or {}

function FightPower:GetEquipBaseFightPower(pEquip)
  return pEquip.nFightPower + FightPower:GetEquipForgePower(pEquip) + Item:GetClass("EquipForgeSeed"):GetRandomAttrFightPower(pEquip)
end

function FightPower:GetEquipForgePower(pEquip)
  local tbAttrList = Item:GetClass(Item.MAIN_EQUIP_CLASS_NAME):GetSeriesActiveAttr(pEquip)
  local nTotalFightPower = 0
  for nIndex, nAttr in pairs(tbAttrList) do
    if 0 ~= nAttr then
      local tbAttribInfo = FightSkill:GetExternAttrib(Item.EquipForge:UnpackAttr(nAttr))
      if tbAttribInfo then
        nTotalFightPower = nTotalFightPower + tbAttribInfo.nFightPower
      end
    end
  end
  return nTotalFightPower
end

function FightPower:GetEquipForgePowerByAttrList(tbAttrList)
  local nTotalFightPower = 0
  for i = 1, #tbAttrList do
    local tbAttribInfo = FightSkill:GetExternAttrib(table.unpack(tbAttrList[i]))
    if tbAttribInfo then
      nTotalFightPower = nTotalFightPower + tbAttribInfo.nFightPower
    end
  end
  return nTotalFightPower
end

function FightPower:RegisterFPSystemModule(tbFightPowerKey, tbFightPowerModeule)
  for _, szFightPowerKey in pairs(tbFightPowerKey) do
    if self.tbFightType2SystemModule[szFightPowerKey] then
      Log("FightPower:RegisterFPSystemModule repeat! szFPSystemModule", szFightPowerKey)
    else
      self.tbFightType2SystemModule[szFightPowerKey] = tbFightPowerModeule
    end
  end
end

function FightPower:UnRegisterFPSystemModule(tbFightPowerKey)
  for _, szFightPowerKey in pairs(tbFightPowerKey) do
    self.tbFightType2SystemModule[szFightPowerKey] = nil
  end
end
