local tbWnd = Ui:CreateWindow("UISavePower")
local tbControls = {
  Panel = "Panel",
  BtnLock = "Panel/UnlockSlider/lockSlideArea/btnlock",
  SliderUnlock = "Panel/UnlockSlider"
}
tbWnd.tbControls = tbControls
tbWnd.nEventHoldThreshold = 0.98
tbWnd.tbOnUISliderPointUp = {
  SliderUnlock = function(self, nValue)
    if nValue >= self.nEventHoldThreshold then
      Ui:LeaveSavePowerMode()
    else
      self:ResetSlider()
    end
  end
}

function tbWnd:ResetSlider()
  local uislider = self.pPanel:GetUISlider(self.SliderUnlock)
  if uislider then
    uislider:resetSlider()
  end
end

function tbWnd:OnCreate()
end

function tbWnd:OnOpen()
  self:ResetSlider()
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_MESSAGEBOX_OPEN)
end

function tbWnd:OnClose()
  EventNotify.OnNotify(EventNotify.emNOTIFY_UI_MESSAGEBOX_CLOSE)
end
