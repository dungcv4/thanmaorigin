Require("CommonScript/Shop/Shop.lua")

function Shop:LoadNpcShop()
  self.tbNpcShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/NpcShop.tab", "dddd", nil, {
    "nWareId",
    "nSort",
    "nNpcId",
    "nMapId"
  })
  for i = 1, #tbWares do
    local v = tbWares[i]
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.Npc then
      error(string.format("Load NpcShop.tab Error: not exist ware or shoptype error, [id=%d, shoptype=%s, error_shop_type=%s]", v.nWareId, Shop.ShopTypeStr.Npc, tbWareInfo and tbWareInfo.szShopType))
    else
      self.tbNpcShopWares[v.nNpcId] = self.tbNpcShopWares[v.nNpcId] or {}
      table.insert(self.tbNpcShopWares[v.nNpcId], v)
    end
  end
  self.tbNpcShopWares = SetReadOnly(self.tbNpcShopWares)
end

function Shop:IsExistShop(nNpcId)
  local tbWares = self.tbNpcShopWares[nNpcId]
  return nil ~= tbWares
end

function Shop:GetANpcShopWares(nNpcId, nMapId)
  local tbList = self.tbNpcShopWares[nNpcId]
  local tbRtn = {}
  if not tbList then
    return tbRtn
  end
  for _, v in pairs(tbList) do
    if not nMapId or v.nMapId <= 0 or v.nMapId == nMapId then
      local tbWare = Shop:GetAWare(v.nWareId)
      if tbWare then
        local isAvaliable = Shop:CheckWareAvaliable(tbWare, me)
        if true == isAvaliable then
          table.insert(tbRtn, v)
        end
      end
    end
  end
  if not Lib:IsEmptyTB(tbRtn) then
    table.sort(tbRtn, function(a, b)
      return a.nSort < b.nSort
    end)
  end
  return tbRtn
end

function Shop:FindNpcShopWare(nNpcId, nItemTemplateId)
  local tbList = self.tbNpcShopWares[nNpcId]
  for _, v in pairs(tbList) do
    local tbWare = Shop:GetAWare(v.nWareId)
    if tbWare and tbWare.nItemId == nItemTemplateId then
      return tbWare
    end
  end
  return nil
end

Shop:LoadNpcShop()
