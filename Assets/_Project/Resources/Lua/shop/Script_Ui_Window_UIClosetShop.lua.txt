local tbWnd = Ui:CreateWindow("UIClosetShop")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_CHANGE_MONEY,
      self.OnChangeMoney,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_PLAYER_DATA_END,
      self.OnChangeMoney,
      self
    },
    {
      EventNotify.emNOTIFY_SHOP_FASHION_UPDATE,
      self.UpdateFashionShop,
      self
    },
    {
      EventNotify.emNOTIFY_SHOP_FASHION_BUY_RESULT,
      self.FashionBuyResult,
      self
    },
    {
      EventNotify.emNOTIFY_CROSS_DAYS,
      self.CrossDays,
      self
    }
  }
end

local tbControls = {
  btnClose = "imgBG/btnClose",
  btnHelp = "imgBG/BG/btnHelp",
  btnALL = "imgBG/TabGroup/btnALL",
  btnSuit = "imgBG/TabGroup/btnSuit",
  btnSingle = "imgBG/TabGroup/btnSingle",
  btnPromote = "imgBG/TabGroup/btnPromote",
  AllPanel = "imgBG/AllPanel",
  SuitPanel = "imgBG/SuitPanel",
  SinglePanel = "imgBG/SinglePanel",
  UIMoneyPanel = "imgBG/UIMoneyPanel"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self.tbTabControl:UpdateSubPanel(self.tbTabControl.nPanelType, "OnClose")
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked(self.szHelpKey)
  end
}
tbWnd.szHelpKey = "ClosetShop"
local tbSubPanelName = {
  PromotePanel = 4,
  AllPanel = 1,
  SuitPanel = 2,
  SinglePanel = 3
}
tbWnd.PANEL_SETTING = {
  [tbSubPanelName.SuitPanel] = {
    CLASS = "UIClosetSuit",
    OBJ = tbControls.SuitPanel,
    BTN = tbControls.btnSuit,
    OpenFunc = function()
      return true
    end
  },
  [tbSubPanelName.SinglePanel] = {
    CLASS = "UIClosetSingle",
    OBJ = tbControls.SinglePanel,
    BTN = tbControls.btnSingle,
    OpenFunc = function()
      return true
    end
  },
  [tbSubPanelName.AllPanel] = {
    CLASS = "UIClosetAll",
    OBJ = tbControls.AllPanel,
    BTN = tbControls.btnALL,
    OpenFunc = function()
      return true
    end
  }
}

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
end

function tbWnd:OnOpen(nSelectType, bFix, ...)
  nSelectType = nSelectType or tbSubPanelName.AllPanel
  if not self.tbTabControl:OnOpen(nSelectType) then
    return
  end
  self.tbMoneyPanel:SetMoneyValue()
  if bFix then
    local tbParams = {
      ...
    }
    if 2 == #tbParams then
      self:RefreshSuitPanelSelectShop(tbParams[1], tbParams[2])
    elseif 4 == #tbParams then
      self:RefreshSinglePanelSelectShop(tbParams[1], tbParams[2], tbParams[3], tbParams[4])
    end
  end
  self.pPanel:SetActive(tbControls.btnPromote, false)
  local tbFashionShops = Shop:GetFashionShopGoods()
  if not tbFashionShops or Lib:IsEmptyTB(tbFashionShops) then
    Log("UIClosetShop OnOpen GetFashionShopGoods is nil")
    return
  end
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
  local tbUIShop = Ui:GetClass("UIShop")
  if tbUIShop and Ui:WindowVisible("UIShop") then
    tbUIShop:RefreshClosetShopEffectShow()
  end
end

function tbWnd:OnChangeMoney()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:GetSelectId()
  return self.tbTabControl.nPanelType
end

function tbWnd:SelectPanel(nSelectPanel)
  self.tbTabControl:SelectPanel(nSelectPanel)
end

function tbWnd:UpdateFashionShop()
  local nSelectType = self.tbTabControl.nPanelType or tbSubPanelName.AllPanel
  self.tbTabControl:UpdateSubPanel(nSelectType, "UpdateView")
end

function tbWnd:CrossDays()
  self:UpdateFashionShop()
end

function tbWnd:FashionBuyResult(bSuccess)
  self.tbTabControl:UpdateSubPanel(tbSubPanelName.SuitPanel, "FashionBuyResult", bSuccess)
  self.tbTabControl:UpdateSubPanel(tbSubPanelName.SinglePanel, "FashionBuyResult", bSuccess)
end

function tbWnd:RefreshSinglePanelSelectShop(nSelectPartId, nSelectLayer, nSelectShopId, nSelectSingleId)
  self.tbTabControl:UpdateSubPanel(tbSubPanelName.SinglePanel, "RefreshFixSelectShop", nSelectPartId, nSelectLayer, nSelectShopId, nSelectSingleId)
end

function tbWnd:RefreshSuitPanelSelectShop(nSelectShopId, nSelectSuitId)
  self.tbTabControl:UpdateSubPanel(tbSubPanelName.SuitPanel, "RefreshFixSelectShop", nSelectShopId, nSelectSuitId)
end
