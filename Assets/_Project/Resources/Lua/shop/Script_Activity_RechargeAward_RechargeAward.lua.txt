ActivityMgr.RechargeAward = ActivityMgr.RechargeAward or {}
local rechargeAward = ActivityMgr.RechargeAward

function rechargeAward:OnClickIcon()
  Ui:OpenWindow("UIRechargeAwards")
end

function rechargeAward:IsShow()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.RechargeRebate)
  return pInstance and pInstance:IsOpen() and pInstance:CanJoin(me.nLevel)
end

function rechargeAward:GetShowInfo()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.RechargeRebate)
  if not pInstance or not pInstance:IsOpen() then
    return {}
  end
  return pInstance:GetShowInfo()
end

function rechargeAward:GainAward(nID, nIndex)
  local pInstance = ActivityMgr:GetInstanceById(nID)
  if not pInstance then
    return
  end
  pInstance:GainAward(nIndex)
end
