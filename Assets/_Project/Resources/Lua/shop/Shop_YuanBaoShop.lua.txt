Require("CommonScript/Shop/Shop.lua")

function Shop:LoadYuanBaoShop()
  self.tbYuanBaoShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/YuanBaoShop.tab", "dd", nil, {"nWareId", "nSort"})
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.YuanBao then
      print("Load YuanBaoShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.YuanBao, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbYuanBaoShopWares, v)
    end
  end
end

Shop:LoadYuanBaoShop()
