Require("CommonScript/Shop/Shop.lua")

function Shop:LoadMantleShop()
  self.tbMantleShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/MantleShop.tab", "ddd", nil, {
    "nWareId",
    "nSort",
    "nNeedHonorLevel"
  })
  for _, tbInfo in pairs(tbWares) do
    local tbWareInfo = self.tbWares[tbInfo.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.HunShi then
      LogErr("Load MantleShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", tbInfo.nWareId, Shop.ShopTypeStr.Mantle, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbMantleShopWares, tbInfo)
    end
  end
end

function Shop:LoadSecretBookShop()
  self.tbSecretBookShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/SecretBookShop.tab", "dd", nil, {"nWareId", "nSort"})
  for _, tbInfo in pairs(tbWares) do
    local tbWareInfo = self.tbWares[tbInfo.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.HunShi then
      LogErr("Load SecretBookShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", tbInfo.nWareId, Shop.ShopTypeStr.Mantle, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbSecretBookShopWares, tbInfo)
    end
  end
end

function Shop:GetMantleWares()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbMantleShopWares) do
    if self:CheckWareAvaliable(self:GetAWare(tbInfo.nWareId), me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

function Shop:GetSecretBookWares()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbSecretBookShopWares) do
    if self:CheckWareAvaliable(self:GetAWare(tbInfo.nWareId), me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

function Shop:GetMantleWaresByItemTemplateId(pPlayer, nItemTemplateId)
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbMantleShopWares) do
    local tbWare = self:GetAWare(tbInfo.nWareId)
    if nItemTemplateId == tbWare.nItemId and self:CheckWareAvaliable(tbWare, pPlayer) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

function Shop:CanBuyMantleWare(pPlayer, nWareId)
  local nNeedHonorLevel = -1
  for _, tbInfo in pairs(self.tbMantleShopWares) do
    if tbInfo.nWareId == nWareId then
      nNeedHonorLevel = tbInfo.nNeedHonorLevel
      break
    end
  end
  if nNeedHonorLevel > pPlayer.nHonorLevel then
    return false, i18n.Get(Shop.Terms.HonorLevelTooLow)
  end
  return true
end

Shop:LoadMantleShop()
Shop:LoadSecretBookShop()
