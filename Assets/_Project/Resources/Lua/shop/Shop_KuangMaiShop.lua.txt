Require("CommonScript/Shop/Shop.lua")

function Shop:LoadKuangMaiShop()
  self.tbKuangMaiShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/KuangMai.tab", "dd", nil, {"nWareId", "nSort"})
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.KuangMai then
      print("Load KuangMaiShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.ShiTu, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbKuangMaiShopWares, v)
    end
  end
end

function Shop:GetKuangMaiShop()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbKuangMaiShopWares) do
    if self:CheckWareAvaliable(self:GetAWare(tbInfo.nWareId), me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

Shop:LoadKuangMaiShop()
