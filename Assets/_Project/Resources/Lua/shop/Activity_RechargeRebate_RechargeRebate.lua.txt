Require("CommonScript/Activity/ActivityMgrDef.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.RechargeRebate)
tbActivity.SaveID = {
  LastStartTime = 1,
  TotalAmount = 2,
  AwardRecord = 3
}
tbActivity.Protocol = {GainAward = 1, GainAwardResp = 2}

function tbActivity:LoadConfig()
  local szFilePath = "Setting/Activity/RechargeRebate/RechargeRebate.tab"
  local tbField = {
    "nConfigId",
    "szSaveKey",
    "szLogWay",
    "szShowTime",
    "szDesc",
    "nLevel",
    "szReminderMFrom",
    "szReminderMTitle",
    "szReminderMContent",
    "nReminderAheadDay"
  }
  self.ACTIVITY_CONFIG = Lib:LoadTabFileEx(szFilePath, "dssssdsssd", "nConfigId", tbField)
  szFilePath = "Setting/Activity/RechargeRebate/AwardLevelList.tab"
  tbField = {
    "nConfigId",
    "nIndex",
    "nAmount",
    "nAwardId",
    "nAwardKey"
  }
  local tbAwardLevelList = Lib:LoadTabFileEx(szFilePath, "ddddd", nil, tbField)
  for _, tbInfo in ipairs(tbAwardLevelList or {}) do
    assert(self.ACTIVITY_CONFIG[tbInfo.nConfigId], string.format("RechargeRebate:LoadConfig AwardLevelList nConfigId=%d not exist", tbInfo.nConfigId))
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAwardList = self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAwardList or {}
    self.ACTIVITY_CONFIG[tbInfo.nConfigId].tbAwardList[tbInfo.nIndex] = {
      nAmount = tbInfo.nAmount,
      nAwardId = tbInfo.nAwardId,
      nAwardKey = tbInfo.nAwardKey
    }
  end
end

function tbActivity:GenTruthTotalAmount(nTotalAmount)
  return math.floor(nTotalAmount / 100)
end

function tbActivity:CanJoin(nLevel)
  return nLevel >= self.tbConfig.nLevel
end
