Require("CommonScript/Shop/Shop.lua")

function Shop:LoadRemnantsShop()
  self.tbRemnantsShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/RemnantsShop.tab", "dd", nil, {"nWareId", "nSort"})
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.Remnants then
      print("Load RemnantsShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.ShiTu, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbRemnantsShopWares, v)
    end
  end
end

function Shop:GetRemnantsWares()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbRemnantsShopWares) do
    if self:CheckWareAvaliable(self:GetAWare(tbInfo.nWareId), me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

Shop:LoadRemnantsShop()
