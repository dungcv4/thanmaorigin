  Sdk:RecordCleverTapEvent("buy_vip_reward", tbArgs)

function Vip:GetNextLevelNeedMoney(nCurVipLevel)
  if nCurVipLevel >= #self.tbExpSetting then
    return 0
  end
  local nCurTotalExp = self:GetTotoalExp(me)
  local nNextTotalExp = self.tbExpSetting[nCurVipLevel + 1]
  local nNeedExp = nNextTotalExp - nCurTotalExp
  return math.floor(nNeedExp / Recharge.nMoney2ExpProportion)
end

function Vip:GetNextLevelNeedExp(nNextVipLevel)
  local nNextVipLevel = self.tbExpSetting[nNextVipLevel] or 0
  return nNextVipLevel
end

function Vip:OnLogin()
  me.UpdateVipLevel()
  LocalData:SetRoleNumber(Vip.szVipLevelGiftState, 0)
  for i = 0, self.nMaxLevel do
    self:OnCheckVipNameRedPoint(i)
  end
end

function Vip:GetVipClickState(nVipLevel)
  local nState = LocalData:GetRoleNumber(Vip.szVipLevelGiftState, 0)
  nState = nState >> nVipLevel
  nState = nState & 1
  return nState
end

function Vip:SetVipClickState(nVipLevel)
  local nClickState = LocalData:GetRoleNumber(Vip.szVipLevelGiftState, 0)
  nClickState = nClickState | 1 << nVipLevel
  LocalData:SetRoleNumber(Vip.szVipLevelGiftState, nClickState)
  Vip:OnCheckVipNameRedPoint(nVipLevel)
  Vip:OnCheckLevelBtnRedPoint(nVipLevel)
end

function Vip:OnCheckVipNameRedPoint(nVipLevel)
  local nClickState = self:GetVipClickState(nVipLevel)
  if 1 == nClickState then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.VipLevelGiftBagBegin + nVipLevel)
  elseif not Vip:CanBugLevelGiftBag(me, nVipLevel) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.VipLevelGiftBagBegin + nVipLevel)
  else
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.VipLevelGiftBagBegin + nVipLevel)
  end
end

function Vip:OnCheckLevelBtnRedPoint(nVipLevel)
  local nClickState = self:GetVipClickState(nVipLevel)
  if 1 == nClickState then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.VipLevelGiftBagCommon)
  elseif not Vip:CanBugLevelGiftBag(me, nVipLevel) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.VipLevelGiftBagCommon)
  else
    RedPointMgr:LightRedPoint(RedPointMgr.EVENT.VipLevelGiftBagCommon)
  end
end

function Vip:CanShowPioneerGift(pPlayer, nVipLevel)
  if not pPlayer or nVipLevel < 0 or nVipLevel > Vip.nMaxLevel then
    return false
  end
  if self.tbSetting[nVipLevel].bIsGivePioneerGB ~= "1" then
    return false
  end
  local nState = pPlayer.GetUserValue(self.SAVE_GROUP, self.KEY_LEVEL_GIFT_BAG_STATE)
  nState = nState & 1 << nVipLevel
  if 1 == nState then
    return false
  end
  local nPioneerGBDurationTime = self.tbSetting[nVipLevel].nPioneerGBDurationTime
  local nServerCreateTime = GetServerCreateTime()
  if GetTime() > nServerCreateTime + nPioneerGBDurationTime then
    return false, i18n.EasyGet("\229\133\136\233\148\139\231\164\188\229\140\133\230\180\187\229\138\168\229\183\178\231\187\147\230\157\159")
  end
  return true
end
