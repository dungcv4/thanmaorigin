function tbWnd:OnCreate()
  self.tbPartScrollView = Ui:NewScrollView(self, tbControls.leftTemplate)
  self.tbSuitGridView = Ui:NewSuperGridView(self, tbControls.suitTemplate, self.OnSetSuitElement)
  self.tbSingleGridView = Ui:NewSuperGridView(self, tbControls.singleTemplate, self.OnSetSingleElement)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
  self.tbSuitShops = Shop:GetFashionSuitShop()
  self.tbPartSetting = Fashion:GetFashionPartConfig()
  self.pNewRecommend = Ui:CreatePanel("UIClosetShopNew_Recommend", tbControls.Recommend, self)
end

function tbWnd:OnOpen(nCurShopType, tbShopParams, nPanelType)
  self:ButtonListShow()
  self:InitPreview()
  self:InitFashionCoupon()
  self:InitPanelType(nCurShopType, tbShopParams, nPanelType)
  self:SwitchShopTypeShow()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:OnClose()
  self.tbPartScrollView:Clear()
  self.nSelectShopIndex = nil
  self.nSelectPartIndex = nil
  self.nHorseRidingState = nil
  self.nSelectSuitId = nil
  self.tbSelectSingleId = {}
  if self.nScrollTimer then
    Timer:Close(self.nScrollTimer)
    self.nScrollTimer = nil
  end
  self.pNewRecommend:CloseTimer()
end

function tbWnd:InitPanelType(nCurShopType, tbParams, nPanelType)
  if nPanelType == self.tbPanelType.Recommend and Shop:HasNewFashion() then
    self:InitSelectData()
    self:SetActiveRecommend(true)
  else
    if nCurShopType == self.tbPanelType.Recommend then
      nCurShopType = nil
      tbParams = nil
    end
    self:SetActiveRecommend(false)
    self:InitSelectData(nCurShopType, tbParams)
  end
end

function tbWnd:InitSelectData(nCurShopType, tbParams)
  self.nCurShopType = nCurShopType or self.tbShopType.Suit
  if nCurShopType == self.tbShopType.Single and tbParams then
    local nSelectLeftIndex = self:GetPartIndexByPartLayer(tbParams.nPart, tbParams.nLayer)
    self.nSelectShopIndex = nil
    self.nSelectPartIndex = nSelectLeftIndex or 1
    self.nSelectSuitId = nil
    self.tbSelectSingleId = {
      [nSelectLeftIndex] = tbParams.nId
    }
  elseif nCurShopType == self.tbShopType.Suit and tbParams then
    local nSelectLeftIndex = self:GetShopIndexByShopId(tbParams.nShopId)
    self.nSelectPartIndex = 1
    self.nSelectShopIndex = nSelectLeftIndex
    self.nSelectSuitId = tbParams.nSuitId
    self.tbSelectSingleId = {}
  else
    self.nSelectPartIndex = 1
    self.nSelectShopIndex = nil
    self.nSelectSuitId = nil
    self.tbSelectSingleId = {}
  end
end

function tbWnd:GetShopIndexByShopId(nShopId)
  for nIndex, tbShop in ipairs(self.tbSuitShops) do
    if tbShop.nId == nShopId then
      return nIndex
    end
  end
  return 0
end

function tbWnd:GetPartIndexByPartLayer(nPart, nLayer)
  nLayer = nLayer or Npc.NpcPartLayerDef.npc_part_layer_base
  for nIndex, tbPart in ipairs(self.tbPartSetting) do
    if tbPart.nPartId == nPart and tbPart.nLayer == nLayer then
      return nIndex
    end
  end
  return 0
end

function tbWnd:ClearCloseInfoShow()
  self.pPanel:SetActive(tbControls.imgBtnsBg, false)
  self.pPanel:Label_SetText(tbControls.txtDesc, "")
  self.pPanel:Label_SetText(tbControls.txtName, "")
  for i = 1, self.nAttributeNum do
    self.pPanel:SetActive(string.format(tbControls.Attribute, i), false)
  end
end

function tbWnd:DelayClick(nIndex)
  if self.nScrollTimer then
    Timer:Close(self.nScrollTimer)
    self.nScrollTimer = nil
  end
  self.nScrollTimer = Timer:Register(self.nDelayClickFrame, function()
    if self.nCurShopType == self.tbShopType.Suit then
      self:OnClickSuitItem(nIndex)
    elseif self.nCurShopType == self.tbShopType.Single then
      self:OnClickSingleItem(nIndex)
    end
  end)
end

function tbWnd:ActiveAttribute(nTagetNum, ...)
  local tbAttr = {
    ...
  }
  for nNum = 1, self.nAttributeNum do
    if nTagetNum == nNum then
      self.pPanel:SetActive(string.format(tbControls.Attribute, nNum), true)
      for nAttrIndex = 1, nNum do
        local szAttr = string.format("txtAttribute%d", nAttrIndex)
        local szAttrPath = string.format(tbControls[szAttr], nNum)
        self.pPanel:Label_SetText(szAttrPath, tbAttr[nAttrIndex])
      end
    else
      self.pPanel:SetActive(string.format(tbControls.Attribute, nNum), false)
    end
  end
end

function tbWnd:SwitchShopTypeShow()
  local pSuitObj = self.pPanel:GetObject(tbControls.btnSuit)
  local pSuitTrans = pSuitObj:GetComponent(Ui.BGroupTrans)
  local pSingleObj = self.pPanel:GetObject(tbControls.btnSingle)
  local pSingleTrans = pSingleObj:GetComponent(Ui.BGroupTrans)
  if self.nCurShopType == self.tbShopType.Suit then
    if pSuitTrans then
      pSuitTrans:OnSelected()
    end
    if pSingleTrans then
      pSingleTrans:OnUnSelected()
    end
    self.pPanel:SetActive(tbControls.SuitList, true)
    self.pPanel:SetActive(tbControls.SingleList, false)
    self.pPanel:SetActive(tbControls.SuitSelcet, true)
    self.pPanel:SetActive(tbControls.SingleSelcet, false)
    self:RefreshSuitPart()
    self.tbSelectSingleId = {}
  elseif self.nCurShopType == self.tbShopType.Single then
    if pSuitTrans then
      pSuitTrans:OnUnSelected()
    end
    if pSingleTrans then
      pSingleTrans:OnSelected()
    end
    self.pPanel:SetActive(tbControls.SuitList, false)
    self.pPanel:SetActive(tbControls.SingleList, true)
    self.pPanel:SetActive(tbControls.SuitSelcet, false)
    self.pPanel:SetActive(tbControls.SingleSelcet, true)
    self:RefreshSinglePart()
    self.nSelectSuitId = nil
  end
end

function tbWnd:ButtonListShow()
  local pClothesObj = self.pPanel:GetObject(tbControls.btnMyClothes)

end

function tbWnd:ModelSwitchShow()
  local nPart = self:GetCurSelectPart()
  if not nPart then
    LogErr("UIClosetShopNew:ModelSwitchShow error. can not find part by GetCurSelectPart")
    return
  end
  if nPart == Npc.NpcResPartsDef.npc_part_horse then
    Fashion.bHorse = true
    self.pPanel:SetActive(tbControls.btnModelSwitchHorse, true)
    self.pPanel:SetActive(tbControls.btnModelSwitchPeople, false)
    if self.nHorseRidingState == self.tbModelState.HorseAndPeople then
      Fashion.bShieldHorse = false
      Fashion:Preview_RefreshPreview_AllPart()
      self.pPanel:SetActive(tbControls.stateHorse, false)
      self.pPanel:SetActive(tbControls.stateHorseAll, true)
    else
      if self.nHorseRidingState ~= self.tbModelState.OnlyHorse then
        self.nHorseRidingState = self.tbModelState.OnlyHorse
      end
      Fashion.bShieldHorse = true
      Fashion:Preview_RefreshPreview()
      self.pPanel:SetActive(tbControls.stateHorse, true)
      self.pPanel:SetActive(tbControls.stateHorseAll, false)
    end
  else
    Fashion.bHorse = false
    self.pPanel:SetActive(tbControls.btnModelSwitchHorse, false)
    self.pPanel:SetActive(tbControls.btnModelSwitchPeople, true)
    if self.nHorseRidingState == self.tbModelState.HorseAndPeople then
      Fashion.bShieldHorse = false
      Fashion:Preview_RefreshPreview_AllPart()
      self.pPanel:SetActive(tbControls.statePeople, false)
      self.pPanel:SetActive(tbControls.statePeopleAll, true)
      if not Fashion:CheckCanShowPart(Npc.NpcResPartsDef.npc_part_horse) then
        me.CenterMsg(i18n.EasyGet("\232\175\183\229\133\136\230\139\165\230\156\137\230\158\163\231\186\162\233\169\172"))
      end
    else
      if self.nHorseRidingState ~= self.tbModelState.OnlyPeople then
        self.nHorseRidingState = self.tbModelState.OnlyPeople
      end
      Fashion.bShieldHorse = true
      Fashion:Preview_RefreshPreview()
      self.pPanel:SetActive(tbControls.statePeople, true)
      self.pPanel:SetActive(tbControls.statePeopleAll, false)
    end
  end
end

function tbWnd:GetCurSelectPart()
  if self.nCurShopType == self.tbShopType.Single then
    local nPart = Fashion:GetFashionPartByIndex(self.nSelectPartIndex)
    if nPart then
      return nPart
    end
    LogErr("UIClosetShopNew:GetCurSelectPart error. can not find part in settting by", self.nSelectPartIndex)
  elseif self.nCurShopType == self.tbShopType.Suit then
    local tbShopSetting = self.tbSuitShops[self.nSelectShopIndex]
    if tbShopSetting then
      return tbShopSetting.nPart
    end
    LogErr("UIClosetShopNew:GetCurSelectPart error. can not find tbShopSetting by", self.nSelectShopIndex)
  end
end

function tbWnd:OnClickPart(nScrollIndex, nDataIndex, bIsClick)
  local tbIconSetting = {}
  if self.nCurShopType == self.tbShopType.Single then
    if nDataIndex <= 0 or nDataIndex > #self.tbPartSetting then
      return
    end
    self.nSelectPartIndex = nDataIndex
    local nPart, nLayer = Fashion:GetFashionPartByIndex(nDataIndex)
    if not nPart then
      return
    end
    self.tbSingleItems = Shop:GetFashionSinglesByPart(nPart, nLayer)
    if not self.tbSingleItems then
      LogErr("UIClosetShopNew:OnClickPart error. GetFashionSinglesByPart is nil by nPart:", nPart, "nLayer:", nLayer)
    end
    for _, tbPart in ipairs(self.tbPartSetting) do
      if Fashion:CheckCanShowPart(tbPart.nPartId) then
        table.insert(tbIconSetting, tbPart)
      end
    end
    self:RefreshSingleScrollView()
  elseif self.nCurShopType == self.tbShopType.Suit then
    if nDataIndex <= 0 or nDataIndex > #self.tbSuitShops then
      return
    end
    self.nSelectShopIndex = nDataIndex
    if bIsClick then
      self.nSelectSuitId = nil
    end
    self.tbSuitItems = Shop:GetFashionSuitsByShopId(nDataIndex)
    if not self.tbSuitItems then
      LogErr("UIClosetShopNew:OnClickPart error. GetFashionSinglesByPart is nil by nIndex:", nDataIndex)
    end
    for nIndex, tbShop in ipairs(self.tbSuitShops) do
      if Shop:CheckFashionSuitShopOpen(nIndex) and (not tbShop.nPart or Fashion:CheckCanShowPart(tbShop.nPart)) then
        table.insert(tbIconSetting, tbShop)
      end
    end
    self:RefreshSuitScrollView()
  end
  self.tbPartScrollView:SelectElement(nScrollIndex)
  if #tbIconSetting ~= self.tbPartScrollView:GetCount() then
    LogErr("UIClosetShopNew:OnClickPart error. tbIconSetting count is not equal to tbPartScrollView count", #tbIconSetting, self.tbPartScrollView:GetCount())
    return
  end
  for nIndex, tbSetting in ipairs(tbIconSetting) do
    local tbElement = self.tbPartScrollView:GetElement(nIndex)
    if tbElement then
      if nIndex == nScrollIndex then
        tbElement.pPanel:Sprite_SetSprite("imgIcon", tbSetting.szSelectedIcon)
      else
        tbElement.pPanel:Sprite_SetSprite("imgIcon", tbSetting.szNormalIcon)
      end
    end
  end
  self:ModelSwitchShow()
  self:SwitchNpcResDir()
end

function tbWnd:RefreshSuitPart()
  local nIndex = 0
  local nScrollToIndex = 0
  self.tbPartScrollView:Clear()
  for nShopIndex, tbShop in ipairs(self.tbSuitShops) do
    if Shop:CheckFashionSuitShopOpen(nShopIndex) and (not tbShop.nPart or Fashion:CheckCanShowPart(tbShop.nPart)) then
      nIndex = nIndex + 1
      self.tbPartScrollView:SetElement(nIndex, function(tbElement)
        tbElement.pPanel:Sprite_SetSprite("imgIcon", tbShop.szNormalIcon)
        tbElement.pPanel:Label_SetText("Text", i18n.Parse(tbShop.szName))
        Ui:Button_UnBindEvent(tbElement.pPanel, "")
        Ui:Button_BindEvent(tbElement.pPanel, "", self.OnClickPart, self, nIndex, nShopIndex, true)
        if not self.nSelectShopIndex or nShopIndex == self.nSelectShopIndex then
          self.nSelectShopIndex = nShopIndex
          nScrollToIndex = nIndex
        end
      end)
    end
  end
  if nScrollToIndex > 0 then
    self.tbPartScrollView:ScrollToIndex(nScrollToIndex)
    self:OnClickPart(nScrollToIndex, self.nSelectShopIndex, false)
  end
end

function tbWnd:RefreshSuitScrollView()
  if not self.tbSuitItems or Lib:IsEmptyTB(self.tbSuitItems) then
    self.pPanel:SetActive(tbControls.Right, false)
    self.pPanel:SetActive(tbControls.EmptyShop, true)
    self.pPanel:Label_SetText(tbControls.txtEmptyDesc, i18n.EasyGet("\230\154\130\230\151\160\229\143\175\229\135\186\229\148\174\231\154\132\229\164\150\232\167\130"))
    return
  else
    self.pPanel:SetActive(tbControls.Right, true)
    self.pPanel:SetActive(tbControls.EmptyShop, false)
  end
  self.tbSuitGridView:SetAllElement(self.tbSuitItems)
  local nSelectSuitId = self.nSelectSuitId
  if nSelectSuitId then
    for nIndex, tbSuit in ipairs(self.tbSuitItems) do
      if tbSuit.nSuitId == nSelectSuitId then
        self.nSelectSuitId = nil
        self.tbSuitGridView:MovePanelToItemIndex(nIndex)
        self:DelayClick(nIndex)
        break
      end
    end
  else
    self.tbSuitGridView:MovePanelToFirstItem()
    self:ClearCloseInfoShow()
  end
end

function tbWnd:OnSetSuitElement(tbElement, tbSuit, nIndex)
  local pPanel = tbElement.pPanel
  local tbHaveFashions = Fashion:GetHaveFashions()
  for nId, tbItem in ipairs(tbSuit.tbGroup) do
    local tbSetting = Fashion:GetFashionSetting(me, tbItem.nFashionID)
    if not tbSetting then
      LogErr("UIClosetShopNew OnSetSuitElement error. GetFashionSetting is nil by FashionId", tbItem.nFashionID)
      return
    end
    tbElement.pGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid" .. nId)
    Fashion:InitFashionGrid(tbElement.pGrid, tbSetting)
    if tbHaveFashions[tbItem.nFashionID] then
      pPanel:SetActive("imgHaveBg" .. nId, true)
    else
      pPanel:SetActive("imgHaveBg" .. nId, false)
    end
  end
  pPanel:SetActive("MarkNew", tbSuit.bShowNewTip)
  pPanel:SetActive("imgSelected", tbSuit.nSuitId == self.nSelectSuitId)
  if self.tbPriceShowType[tbSuit.nShowType] then
    pPanel:SetActive("Cost", false)
    pPanel:SetActive("MarkDiscounts", false)
    pPanel:Label_SetText("txtComingSoon", self.tbPriceShowType[tbSuit.nShowType]())
  else
    pPanel:SetActive("Cost", true)
    pPanel:Label_SetText("txtComingSoon", "")
    local szMoneyIcon = Currency:GetIcon(tbSuit.szMoneyType)
    pPanel:Sprite_SetSprite("Cost/imgMoney", szMoneyIcon)
    pPanel:SetActive("Cost/CostDiscounts", true)
    if tbSuit.nDiscountPrice > 0 then
      pPanel:SetActive("MarkDiscounts", true)

function tbWnd:OnClickSuitItem(nIndex)

function tbWnd:RefreshSinglePart()
  if not self.nSelectPartIndex then
    self.nSelectPartIndex = 1
  end
  local nIndex = 0
  local nScrollToIndex = 0
  self.tbPartScrollView:Clear()
  for nPartIndex, tbPart in ipairs(self.tbPartSetting) do
    if Fashion:CheckCanShowPart(tbPart.nPartId) then
      nIndex = nIndex + 1
      self.tbPartScrollView:SetElement(nIndex, function(tbElement)
        tbElement.pPanel:Sprite_SetSprite("imgIcon", tbPart.szNormalIcon)
        tbElement.pPanel:Label_SetText("Text", i18n.Parse(tbPart.szName))
        Ui:Button_UnBindEvent(tbElement.pPanel, "")
        Ui:Button_BindEvent(tbElement.pPanel, "", self.OnClickPart, self, nIndex, nPartIndex, true)
      end)
      if nPartIndex == self.nSelectPartIndex then
        nScrollToIndex = nIndex
      end
    end
  end
  if nScrollToIndex > 0 then
    self.tbPartScrollView:ScrollToIndex(nScrollToIndex)
    self:OnClickPart(nScrollToIndex, self.nSelectPartIndex, false)
  end
end

function tbWnd:ArrangeSingleItems()
  local nPart, nLayer = Fashion:GetFashionPartByIndex(self.nSelectPartIndex)
  if not nPart then
    LogErr("UIClosetShopNew ArrangeSingleItems error. get part is nil by partIndex:", self.nSelectPartIndex)
    return
  end
  local nSelectFashionId = self:GetSelectFashionIDByPart(nPart, Npc.NpcPartLayerDef.npc_part_layer_base)
  local nSelectEffectId = self:GetSelectFashionIDByPart(nPart, Npc.NpcPartLayerDef.npc_part_layer_effect)
  self.nSingleHightLight = #self.tbSingleItems
  if not nSelectFashionId or nSelectFashionId <= 0 then
    return self.tbSingleItems
  end
  local tbGrey = {}
  local tbHightLight = {}
  if nLayer == Npc.NpcPartLayerDef.npc_part_layer_effect then
    for _, tbItem in ipairs(self.tbSingleItems) do
      local tbSetting = Fashion:GetFashionSetting(me, tbItem.nFashionID)
      if tbSetting then
        if Lib:Is_include(nSelectFashionId, tbSetting.tbEffect2FashionIds) then
          table.insert(tbHightLight, tbItem)
        else
          table.insert(tbGrey, tbItem)
        end
      end
    end
  else
    if not nSelectEffectId or nSelectEffectId <= 0 then
      return self.tbSingleItems
    end
    for _, tbItem in ipairs(self.tbSingleItems) do
      local tbSetting = Fashion:GetFashionSetting(me, tbItem.nFashionID)
      if tbSetting then
        if Lib:Is_include(nSelectEffectId, tbSetting.tbFashion2EffectIds) then
          table.insert(tbHightLight, tbItem)
        else
          table.insert(tbGrey, tbItem)
        end
      end
    end
  end
  self.nSingleHightLight = #tbHightLight
  return Lib:MergeTable(tbHightLight, tbGrey)
end

function tbWnd:GetSelectFashionIDByPart(nPart, nLayer)
  local nFashionPartIndex = self:GetPartIndexByPartLayer(nPart, nLayer)
  if nFashionPartIndex > 0 then
    local nSelectId = self.tbSelectSingleId[nFashionPartIndex]
    if nSelectId then
      local tbFashionItem = Shop:GetFashionItem(nSelectId)
      return tbFashionItem.nFashionID
    end
  end
end

    self.pPanel:SetActive(tbControls.Right, true)
    self.pPanel:SetActive(tbControls.EmptyShop, false)
  end
  self.tbSingleItems = tbItems
  self.tbSingleGridView:SetAllElement(tbItems)

  if nSelectSingleId then
    for nIndex, tbItem in ipairs(self.tbSingleItems) do
      if tbItem.nId == nSelectSingleId then
        self.tbSelectSingleId[self.nSelectPartIndex] = nil

function tbWnd:PreviewWearFashion(nFashionId)
  if not self:CheckCanPreviewWearFashion(nFashionId) then
    return
  end
  local tbSetting = Fashion:GetFashionSetting(me, nFashionId)
  local nPreviewEffectId = self:GetPreviewEffectId(tbSetting.nPart)
  if nPreviewEffectId > 0 then
    if not Lib:Is_include(nPreviewEffectId, tbSetting.tbFashion2EffectIds) then
      local nPartIndex = self:GetPartIndexByPartLayer(tbSetting.nPart, Npc.NpcPartLayerDef.npc_part_layer_effect)
      self.tbSelectSingleId[nPartIndex] = nil
      Fashion:ClearPartEffectPreview(tbSetting.nPart)
      self:RefreshSingleScrollView()
      me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\232\175\149\231\169\191\229\164\150\232\167\130\228\184\142\233\162\132\232\167\136\231\137\185\230\149\136\228\184\141\229\140\185\233\133\141\239\188\140\229\183\178\228\184\186\230\130\168\229\143\150\230\182\136\231\137\185\230\149\136\233\162\132\232\167\136"))
    end
  elseif tbSetting.nPart == Npc.NpcResPartsDef.npc_part_weapon and (not tbSetting.nAddEnhanceEffect or tbSetting.nAddEnhanceEffect <= 0) then
    Fashion:ClearPartEffectPreview(tbSetting.nPart)
  end
  Fashion:Preview_TryOnAFashion(tbSetting.nPart, nFashionId)
end

function tbWnd:CheckCanPreviewWearFashion(nFashionId)
  local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
  if not tbSetting or Lib:IsEmptyTB(tbSetting) then
    return
  end
  if not bCanUse then
    return
  end
  if tbSetting.nLayer ~= Npc.NpcPartLayerDef.npc_part_layer_base then
    return
  end
  local tbPreviewList = Fashion:Preview_GetPreviewList()
  if not tbPreviewList or not tbPreviewList[tbSetting.nPart] then
    return
  end
  local nPreviewFashionId = self:GetPreviewFashionId(tbSetting.nPart)
  if nPreviewFashionId == nFashionId then
    return
  end
  return true
end

function tbWnd:PreviewWearEffect(nFashionId)
  local tbSetting, bCanUse = Fashion:GetFashionSetting(me, nFashionId)
  if not tbSetting or Lib:IsEmptyTB(tbSetting) then
    return
  end
  if not bCanUse then
    return
  end
  if tbSetting.nLayer ~= Npc.NpcPartLayerDef.npc_part_layer_effect then
    return
  end
  local nEffectFashionId = self:GetPreviewEffectId(tbSetting.nPart)
  if nEffectFashionId == nFashionId then
    return
  end
  local nPreviewFashionId = self:GetPreviewFashionId(tbSetting.nPart)
  if nPreviewFashionId and not Lib:Is_include(nPreviewFashionId, tbSetting.tbEffect2FashionIds) then
    local tbShopItems = Shop:GetFashionSinglesByPart(tbSetting.nPart)
    if not tbShopItems then
      LogErr("UIClosetShopNew:PreviewWearEffect error. GetFashionSinglesByPart is nil by ", tbSetting.nPart)
      return
    end
    for _, tbItem in ipairs(tbShopItems) do
      if Lib:Is_include(tbItem.nFashionID, tbSetting.tbEffect2FashionIds) then
        local nPartIndex = self:GetPartIndexByPartLayer(tbSetting.nPart)
        if nPartIndex > 0 then
          self.tbSelectSingleId[nPartIndex] = tbItem.nId
          if not self:CheckCanPreviewWearFashion(tbItem.nFashionID) then
            LogErr("UIClosetShopNew:PreviewWearEffect error. CheckCanPreviewWearFashion fail\239\188\129", tbItem.nFashionID, nFashionId)
            return
          end
          Fashion:Preview_TryOnAFashion(tbSetting.nPart, tbItem.nFashionID)
          self:RefreshSingleScrollView()
          me.CenterMsg(i18n.EasyGet("\229\189\147\229\137\141\232\175\149\231\169\191\231\137\185\230\149\136\228\184\142\233\162\132\232\167\136\229\164\150\232\167\130\228\184\141\229\140\185\233\133\141\239\188\140\229\183\178\228\184\186\230\130\168\233\135\141\230\150\176\229\140\185\233\133\141\229\164\150\232\167\130"))
          break
        end
        LogErr("UIClosetShopNew:PreviewWearEffect error. GetPartIndexByPartLayer error.", nPartIndex, tbSetting.nPart)
        break
      end
    end
  end
  nPreviewFashionId = self:GetPreviewFashionId(tbSetting.nPart)
  if not nPreviewFashionId or not Lib:Is_include(nPreviewFashionId, tbSetting.tbEffect2FashionIds) then
    LogErr("UIClosetShopNew:PreviewWearEffect error. No appearance matching the effects can be found", tbSetting.nPart, nFashionId)
    return
  end
  Fashion:PreviewTryOnEffect(tbSetting.nPart, nFashionId)
end

function tbWnd:GoToSuitBuy()
  if not self.nSelectPartIndex then
    return
  end
  local nSelectSingleId = self.tbSelectSingleId[self.nSelectPartIndex]
  if not nSelectSingleId then
    return
  end
  local tbSuit = Shop:GetFashionSuitBySingleItemId(nSelectSingleId)
  if not tbSuit or not Shop:IsFashionSuitCanShow(tbSuit.nSuitId) then
    return
  end
  if tbSuit.nShowType and tbSuit.nShowType > 0 then
    return
  end
  local nSuitShopIndex = 0
  for nIndex, tbShop in ipairs(self.tbSuitShops) do
    if Lib:Is_include(tbShop.nId, tbSuit.tbShopIds) then
      nSuitShopIndex = nIndex
    end
  end
  if nSuitShopIndex <= 0 then
    return
  end
  self.nCurShopType = self.tbShopType.Suit
  self.nSelectShopIndex = nSuitShopIndex
  self.nSelectSuitId = tbSuit.nSuitId
  self:SwitchShopTypeShow()
end

function tbWnd:OnClickSingleBuy()
  if self.nCurShopType ~= self.tbShopType.Single then
    return
  end
  if self.bIsBuying then
    return
  end
  if not self.nSelectPartIndex then
    return
  end
  local nSelectSingleId = self.tbSelectSingleId[self.nSelectPartIndex]
  if not nSelectSingleId then
    return
  end
  local tbItem = Shop:GetFashionItem(nSelectSingleId)
  if tbItem.nShowType and tbItem.nShowType > 0 then
    return
  end
  Ui:OpenWindow("UIClosetBuy", self.tbShopType.Single, nSelectSingleId)
end

function tbWnd:SetActiveRecommend(bActive)
  if bActive then
    if self.nPanelType ~= self.tbPanelType.Recommend then
      self.nPanelType = self.tbPanelType.Recommend
    end
  elseif self.nPanelType ~= self.tbPanelType.Shop then
    self.nPanelType = self.tbPanelType.Shop
  end
  if self.pNewRecommend then
    if bActive then
      self.pNewRecommend:UpdateView()
    else
      self.pNewRecommend:CloseTimer()
    end
    self.pPanel:SetActive(tbControls.Recommend, bActive)
    self.pPanel:SetActive(tbControls.leftTab, not bActive)
    self.pPanel:SetActive(tbControls.ShopPanel, not bActive)
  end
  local pShopObj = self.pPanel:GetObject(tbControls.btnClothesShop)
  local pShopTrans = pShopObj:GetComponent(Ui.BGroupTrans)
  if pShopTrans then
    if bActive then
      pShopTrans:OnUnSelected()
    else
      pShopTrans:OnSelected()
    end
  end
  local pRecommendObj = self.pPanel:GetObject(tbControls.btnNewRecommend)
  local pRecommendTrans = pRecommendObj:GetComponent(Ui.BGroupTrans)
  if pRecommendTrans then
    if bActive then
      pRecommendTrans:OnSelected()
    else
      pRecommendTrans:OnUnSelected()
    end
  end
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_CROSS_DAYS,
      self.CrossDays,
      self
    },
    {
      EventNotify.emNOTIFY_CHANGE_MONEY,
      self.OnChangeMoney,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_PLAYER_DATA_END,
      self.OnChangeMoney,
      self
    },
    {
      EventNotify.emNOTIFY_SHOP_FASHION_UPDATE,
      self.UpdateFashionShop,
      self
    },
    {
      EventNotify.emNOTIFY_SHOP_FASHION_BUY_RESULT,
      self.FashionBuyResult,
      self
    }
  }
end

function tbWnd:OnChangeMoney()
  self.tbMoneyPanel:SetMoneyValue()
end

function tbWnd:UpdateFashionShop()
  self:SwitchShopTypeShow()
end

function tbWnd:FashionBuyResult(bSuccess)
  self.bIsBuying = false
  if bSuccess then
    self:SwitchShopTypeShow()
  end
end

function tbWnd:CrossDays()
  self:SwitchShopTypeShow()
end
