function tbWnd:NotifyEvents()

    {

      EventNotify.emNOTIFY_SYNC_SHOP_WARE,
      self.OnSyncShopWare,
      self
    },
    {
      EventNotify.emNOTIFY_SYNC_ITEM,
      self.OnSyncItem,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.tbTabControl = Ui:NewTabControl(self, self.PANEL_SETTING)
  self.tbMoneyPanel = Ui:NewExistMoneyPanel(self, tbControls.UIMoneyPanel)
  self.tbWayScrollView = Ui:NewScrollView(self, tbControls.WayTemplate)
end

function tbWnd:OnOpen(szShopType, ...)
  szShopType = szShopType or Shop.ShopTypeStr.Repute
  if szShopType == Shop.ShopTypeStr.DrugShop then
    RemoteServer.OnShopRequest(Shop.Protocol.C2SCheckFamilyShop)
  end
  local tbParam = {
    szShopType,
    ...
  }
  self.nPanelType = self.ShopType2PanelIndex[szShopType]
  if not self.tbTabControl:OnOpen(self.nPanelType, table.unpack(tbParam)) then
    return
  end
  self:SwitchPanel(self.nPanelType, ...)
  self.pPanel:SetActive(tbControls.wndJumpTo, false)
  self.tbMoneyPanel:OnOpen()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.StoreShop)
  LocalData:SetRoleBoolean("StoreTip", true)
end

function tbWnd:OnClose()
  self.tbTabControl:OnClose()
end

function tbWnd:SwitchPanel(nPanelType, ...)
  if nPanelType == tbSubPanelIndex.PANEL_TYPE_DRUGSHOP then
    RemoteServer.OnShopRequest(Shop.Protocol.C2SCheckFamilyShop)
  end
  self.nPanelType = nPanelType
  if self.PANEL_SETTING[nPanelType].szCurrencyType then
    self:SetDynamicCurrencyPanel(self.PANEL_SETTING[nPanelType].szCurrencyType)
  else
    self.tbMoneyPanel:HideDynamicBtnPlus()
    self.tbMoneyPanel:SetDefaultCurrencyView()
    self.tbMoneyPanel:SetMoneyValue()
  end
  self.tbTabControl:UpdateSubPanel(-1, "OnOpen", self.PANEL_SETTING[self.nPanelType].szShopType, ...)
end

function tbWnd:SetDynamicCurrencyPanel(szCurrencyType)
  self.tbMoneyPanel:SetDynamicCurrencyType(szCurrencyType)
  if self.PANEL_SETTING[self.nPanelType].bShowCurrencyAddBtn then
    if self.GETTING_WAY[szCurrencyType] then
      self.tbMoneyPanel:SetDynamicBtnPlusCallBack(self.UpdateGettingWay, self, szCurrencyType)
    end
  else
    self.tbMoneyPanel:HideDynamicBtnPlus()
  end
end

function tbWnd:UpdateGettingWay(szCurrencyType)
  if not self.GETTING_WAY[szCurrencyType] then
    return
  end
  self.tbWayScrollView:Clear()
  for i, tbWayInfo in ipairs(self.GETTING_WAY[szCurrencyType]) do
    self.tbWayScrollView:SetElement(i, function(tbElement)
      tbElement.pPanel:Label_SetText("txtName", tbWayInfo.szTitle())
      tbElement.pPanel:Label_SetText("txtDesc", tbWayInfo.szDesc())
      Ui:Button_BindEvent(tbElement.pPanel, "", function()
        tbWayInfo.func()
        self.pPanel:SetActive(tbControls.wndJumpTo, false)
        self.tbWayScrollView:Clear()
      end)
    end)
  end
  self.pPanel:SetActive(tbControls.wndJumpTo, true)
end

function tbWnd:OnChangeMoney()
  if self.PANEL_SETTING[self.nPanelType].szCurrencyType then
    self.tbMoneyPanel:SetDynamicCurrencyValue()
  else
    self.tbMoneyPanel:SetMoneyValue()
  end
end

function tbWnd:OnResponseBuy(bSuccess, nItemId)
  if bSuccess then
    self.tbTabControl:UpdateSubPanel(-1, "UpdateShopItemList")
  end
end

function tbWnd:OnSyncShopWare()
  self.tbTabControl:UpdateSubPanel(-1, "UpdateShopItemList")
end

function tbWnd:OnSyncItem()
  if self.PANEL_SETTING[self.nPanelType].szCurrencyType then
    self.tbMoneyPanel:SetDynamicCurrencyValue()
  end
end
