Require("CommonScript/Shop/Shop.lua")

function Shop:LoadJinDingShop()
  self.tbJinDingShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/JinDingShop.tab", "ddd", nil, {
    "nWareId",
    "nSort",
    "nRecommend"
  })
  if not tbWares then
    return
  end
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.JinDingShop then
      print("Load JinDingShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.JinDingShop, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbJinDingShopWares, v)
    end
  end
end

function Shop:GetJinDingWares()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbJinDingShopWares) do
    if self:CheckWareAvaliable(self:GetAWare(tbInfo.nWareId), me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  if not Lib:IsEmptyTB(tbRetWares) then
    table.sort(tbRetWares, function(a, b)
      return a.nSort < b.nSort
    end)
  end
  return tbRetWares
end

Shop:LoadJinDingShop()
