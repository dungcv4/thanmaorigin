Require("CommonScript/Shop/Shop.lua")

function Shop:LoadTongShop()
  self.tbTongShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/TongShop.tab", "ddd", nil, {
    "nWareId",
    "nSort",
    "nDayLimit"
  })
  for _, tbInfo in pairs(tbWares) do
    local tbWareInfo = self.tbWares[tbInfo.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.Tong then
      LogErr("Load TongShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", tbInfo.nWareId, Shop.ShopTypeStr.Tong, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbTongShopWares, tbInfo)
    end
  end
end

function Shop:GetTongWareLimitNum(nWareId)
  for _, tbInfo in pairs(self.tbTongShopWares) do
    if tbInfo.nWareId == nWareId then
      return tbInfo.nDayLimit
    end
  end
end

function Shop:GetTongWares()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbTongShopWares) do
    if self:CheckWareAvaliable(self.tbWares[tbInfo.nWareId], me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

function Shop:CanBuyTongWare(pPlayer, tbWare)
  if MODULE_GAMESERVER then
    local tbMember = Tong:GetMemberData(pPlayer.dwID)
    if not tbMember or not tbMember:CheckAuthority(Tong.Def.Authority_UseBuildFund) then
      return false, i18n.Get(Tong.Cfg.Msg.NoAuthority)
    end
    local tbTongData = Tong:GetTongById(pPlayer.dwTongId)
    if not tbTongData then
      return false, i18n.EasyGet("\229\184\174\228\188\154\228\184\141\229\173\152\229\156\168")
    end
    local nLimitNum = self:GetTongWareLimitNum(tbWare.nId)
    if not nLimitNum then
      return false, i18n.EasyGet("\233\153\144\232\180\173\230\149\176\233\135\143\233\148\153\232\175\175")
    end
    local nHadBuyNum = tbTongData:GetTongShopLimit(tbWare.nId)
    if nLimitNum > 0 and nLimitNum <= nHadBuyNum then
      return false, i18n.Get(Shop.Terms.InsufficientRemainInventory)
    end
  end
  return true
end

Shop:LoadTongShop()
