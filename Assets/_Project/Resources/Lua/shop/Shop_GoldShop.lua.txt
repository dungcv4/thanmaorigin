Require("CommonScript/Shop/Shop.lua")

function Shop:LoadGoldShop()
  self.tbGoldShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/GoldShop.tab", "dd", nil, {"nWareId", "nSort"})
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.Gold then
      print("Load GoldShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.Gold, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbGoldShopWares, v)
    end
  end
end

Shop:LoadGoldShop()
