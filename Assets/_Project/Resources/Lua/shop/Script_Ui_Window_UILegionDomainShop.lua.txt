local tbWnd = Ui:CreateWindow("UILegionDomainShop")
tbWnd.tbControls = {
  btnClose = "node/imgBG/btnClose",
  uiMoneyPanel = "node/UIMoneyPanel",
  goGoodsTemplate = "node/imgBG/Content/ShopGoods/GoodsList/pnlMask/pnlContant/ElementTemplate",
  txtTip = "node/imgBG/Content/txtTips"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbWareScroll = Ui:NewScrollView(self, self.goGoodsTemplate, self.OnClickGoods, nil, true)
  self.pPanel:Label_SetText(self.txtTip, i18n.EasyGet("\230\175\143\229\156\186\232\183\168\230\156\141\233\162\134\229\156\159\230\136\152\229\136\183\230\150\176\233\162\134\229\143\150\230\172\161\230\149\176"))
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(self.uiMoneyPanel, false)
  if GlobalDomainBattle:IsInGame() then
    RemoteServer.RequestGlobalDomainBattle("LegionShopLimitReq")
  else
    GlobalDomainBattle.tbLegionShopLimit = nil
    self:UpdateShow()
  end
end

function tbWnd:UpdateShow()
  self.tbWareScroll:Reset()
  for nIndex, tbInfo in pairs(GlobalDomainBattle.SUPPORT_ITEM_LIST) do
    self.tbWareScroll:SetElement(nIndex, function(tbElement)
      local nItemId = tbInfo.nItemId
      local pPanel = tbElement.pPanel
      if not tbElement.tbItemGrid then
        tbElement.tbItemGrid = Ui:NewExistItemGrid(tbElement, "UIItemGrid")
        tbElement.tbItemGrid:SetClickEnable(true)
      end
      tbElement.tbItemGrid:SetItemByTemplate(nItemId)
      pPanel:Label_SetText("txtTitle", i18n.Parse(KItem.GetItemShowInfo(nItemId, me.nFaction, me.nFactionSect, me.nSex)))
      pPanel:Label_SetText("txtMoneyNum", i18n.EasyGet("\229\133\141\232\180\185\233\162\134\229\143\150"))
      pPanel:SetActive("imgZK", false)
      local nRemainCount = tbInfo.nCount
      if GlobalDomainBattle.tbLegionShopLimit and GlobalDomainBattle.tbLegionShopLimit[nIndex] then
        nRemainCount = GlobalDomainBattle.tbLegionShopLimit[nIndex].nCount
      end
      if 0 == nRemainCount then
        pPanel:SetActive("imgSoldOutBg", true)
        pPanel:SetActive("imgSoldOutMark", true)
      else
        pPanel:SetActive("imgSoldOutBg", false)
        pPanel:SetActive("imgSoldOutMark", false)
      end
      pPanel:SetActive("imgMoney", false)
      pPanel:SetActive("Limit", true)
      tbElement.nRemainCount = nRemainCount
      pPanel:Label_SetText("Limit/txtLimitNum", nRemainCount)
      pPanel:SetActive("imgHighlight", false)
      tbElement.nItemId = nItemId
    end)
  end
end

function tbWnd:OnResponseBuy(bSuccess, nItemId)
  if not Ui:WindowVisible(self.ClassName) then
    return
  end
  if not bSuccess then
    return
  end
  self:UpdateShow()
  Legion:UpdateBaseInfo()
end

function tbWnd:OnClickGoods(nIndex, tbElement)
  if not GlobalDomainBattle:IsInGame() then
    me.CenterMsg(i18n.EasyGet("\228\187\133\229\143\175\229\156\168\232\183\168\230\156\141\233\162\134\229\156\159\229\156\176\229\155\190\233\162\134\229\143\150"))
    return
  end
  local nItemId = tbElement.nItemId
  local szMsg = i18n.EasyGet("\230\152\175\229\144\166\233\162\134\229\143\150\232\175\165\233\129\147\229\133\183")
  me.MsgBox(szMsg, {
    {
      i18n.szOk,
      self.Receive,
      self,
      nItemId
    },
    {
      i18n.szCancel
    }
  })
end

function tbWnd:Receive(nItemId)
  RemoteServer.RequestGlobalDomainBattle("ReceiveSupportItemReq", nItemId)
end
