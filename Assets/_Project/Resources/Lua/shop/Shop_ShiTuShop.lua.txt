Require("CommonScript/Shop/Shop.lua")

function Shop:LoadShiTuShop()
  self.tbShiTuShopWares = {}
  local tbWares = Lib:LoadTabFileEx("Setting/Shop/ShiTu.tab", "dd", nil, {"nWareId", "nSort"})
  for _, v in pairs(tbWares) do
    local tbWareInfo = self.tbWares[v.nWareId]
    if not tbWareInfo or tbWareInfo.szShopType ~= Shop.ShopTypeStr.ShiTu then
      print("Load ShiTuShop.tab Error: not exist ware or shoptype error, [id, shoptype, error_shop_type] = ", v.nWareId, Shop.ShopTypeStr.ShiTu, tbWareInfo and tbWareInfo.szShopType)
    else
      table.insert(self.tbShiTuShopWares, v)
    end
  end
end

function Shop:GetShiTuWares()
  local tbRetWares = {}
  for _, tbInfo in pairs(self.tbShiTuShopWares) do
    if self:CheckWareAvaliable(self:GetAWare(tbInfo.nWareId), me) then
      table.insert(tbRetWares, tbInfo)
    end
  end
  return tbRetWares
end

Shop:LoadShiTuShop()
