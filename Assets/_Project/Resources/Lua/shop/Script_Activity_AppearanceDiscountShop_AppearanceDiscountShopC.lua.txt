function tbActivity:GetRedPointEventList()
  local tbEvents = {}
  if self.tbConfig.szRedPoint and RedPointMgr.EVENT[self.tbConfig.szRedPoint] then
    table.insert(tbEvents, RedPointMgr.EVENT[self.tbConfig.szRedPoint])
  end
  return tbEvents
end
