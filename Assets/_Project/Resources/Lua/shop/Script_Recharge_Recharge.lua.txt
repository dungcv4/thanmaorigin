function Recharge:RequestCreateRechargeOrder(nProductId)
  if not IsPayOpen() then
    me.CenterMsg(i18n.EasyGet("\229\133\133\229\128\188\230\154\130\228\184\141\229\188\128\230\148\190"))
    return
  end
  local tbProductSetting = self:GetProductSettingById(nProductId)
  if not tbProductSetting then
    me.CenterMsg(i18n.EasyGet("\229\143\130\230\149\176\229\188\130\229\184\184"))
    return
  end
  local bOnSale, szOnSaleMsg = Recharge:IsProductOnSale(nProductId)
  if not bOnSale then
    me.CenterMsg(szOnSaleMsg)
    return
  end
  local bInLimitCount, szLimitMsg = Recharge:CheckLimitCount(me, nProductId)
  if not bInLimitCount then
    me.CenterMsg(szLimitMsg)
    return
  end
  if Sdk:GetOsType() == Sdk.eOSType_Windows and Client.EnableWebRecharge == true then
    Log("RequestCreateRechargeOrder calling web", tbProductSetting.nProductId, tbProductSetting.szWebProductId, tbProductSetting.szProductName, tbProductSetting.szProductDesc)
    local tbOrderInfo = {
      productId = tbProductSetting.szWebProductId,
      productName = tbProductSetting.szProductName,
      productDesc = tbProductSetting.szProductDesc
    }
    Client.ChannelModule.CallXGMethod("WebPay", tbOrderInfo, "")
    return
  end
  Ui:OpenWindow("UILoadingTips", i18n.EasyGet("\229\133\133\229\128\188\232\175\183\230\177\130\228\184\173..."), 120, function()
    Log(i18n.EasyGet("\233\157\158\229\184\184\230\138\177\230\173\137\239\188\140\229\133\133\229\128\188\232\175\183\230\177\130\232\182\133\230\151\182"))
  end)
  local szProductUnit = Client.RechargeCfg.szUnit
  local szCurrencyName = Client.RechargeCfg.szCurrencyName
  local tbOrderInfo = {
    szChannelId = Client.ChannelModule._ChannelID,
    szCurrencyName = szCurrencyName,
    nProductId = tbProductSetting.nProductId,
    szChannelProductId = tbProductSetting.szChannelProductId,
    szProductName = tbProductSetting.szProductName,
    szProductDesc = tbProductSetting.szProductDesc,
    nProductQuantity = 1,
    szProductUnit = szProductUnit,
    nProductPrice = tbProductSetting.nPrice,
    nTotalAmount = tbProductSetting.nPrice,
    nPaidAmount = tbProductSetting.nPrice,
    nVipLevel = me.GetVipLevel(),
    szPartyName = Kin:GetKinName(),
    szCustomInfo = ""
  }
  RemoteServer.RequestCreateRechargeOrder(tbOrderInfo)
end

function Recharge:OnCreateOrderResponse(tbOrder)
  if not tbOrder then
    Ui:CloseWindow("UILoadingTips")
    return
  end
  if not IsPayOpen() then
    me.CenterMsg(i18n.EasyGet("\229\133\133\229\128\188\230\154\130\228\184\141\229\188\128\230\148\190"))
    return
  end
  Sdk:Pay(tbOrder)
end

end
