local tbWnd = Ui:CreateWindow("UIRechargeAwards")

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_RECHARGEAWARD_DATA_SYNC,
      self.UpdateActivityInfo
    }
  }
end

local tbControls = {
  btnClose = "node/imgBG/btnClose",
  btnHelp = "node/imgBG/txtTitle/btnHelp",
  txtActivtyTime = "node/imgBG/Content/ActivtyTime/txtActivtyTime",
  txtTime = "node/imgBG/Content/ActivtyTime/txtTime",
  txtActivtyDetails = "node/imgBG/Content/txtActivtyDetails",
  Template = "node/imgBG/Content/Detail/AwardsList/Viewport/Content/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("RechargeAwards")
  end
}

function tbWnd:OnCreate()
  self.tbAwardSV = Ui:NewSuperListView(self, tbControls.Template, self.SetAwardList, nil)
end

function tbWnd:OnOpen()
  self.pPanel:SetActive(tbControls.btnHelp, false)
  self.tbAwardSV:SetFuncSetItem(self.SetAwardList)
  self:UpdateActivityInfo()
end

function tbWnd:UpdateActivityInfo()
  local tbShowInfo = ActivityMgr.RechargeAward:GetShowInfo()
  self.nActivityId = tbShowInfo.nId
  self.pPanel:Label_SetText(tbControls.txtTime, tbShowInfo.szShowTime)
  self.pPanel:Label_SetText(tbControls.txtActivtyDetails, tbShowInfo.szDesc)
  self.tbAwardSV:SetAllElement(tbShowInfo.tbAwardList)
  self.tbAwardSV:MovePanelToFirstItem()
end

function tbWnd:GenTxtTip1(tbInfo)
  if tbInfo.nSuccess > 0 or tbInfo.nAmount <= tbInfo.nTotalAmount then
    return i18n.EasyGet("\229\183\178\232\190\190\229\136\176")
  end
  if tbInfo.bFirstNotEnough then
    return i18n.EasyGet("\229\134\141\229\133\133\229\128\188{0}\229\133\131", tbInfo.nAmount - tbInfo.nTotalAmount)
  end
  return i18n.EasyGet("\230\156\170\232\190\190\229\136\176")
end

function tbWnd:GenTxtTip2(tbInfo)
  if tbInfo.nSuccess > 0 or tbInfo.nAmount <= tbInfo.nTotalAmount then
    return string.format("(%d/%d)", tbInfo.nAmount, tbInfo.nAmount)
  end
  return string.format("(%d/%d)", tbInfo.nTotalAmount, tbInfo.nAmount)
end

function tbWnd:GenBtnTextAndStatus(tbInfo)
  if tbInfo.nSuccess > 0 then
    return i18n.EasyGet("\229\183\178\233\162\134\229\143\150"), false
  end
  if tbInfo.nTotalAmount >= tbInfo.nAmount then
    return i18n.EasyGet("\229\143\175\233\162\134\229\143\150"), true, "GainAward"
  end
  if tbInfo.bFirstNotEnough then
    return i18n.EasyGet("\229\133\133\229\128\188"), true, "JumpToRechargeGold"
  end
  return i18n.EasyGet("\230\156\170\232\190\190\229\136\176"), false
end

function tbWnd:GainAward(tbInfo)
  ActivityMgr.RechargeAward:GainAward(self.nActivityId, tbInfo.nID)
end

function tbWnd:JumpToRechargeGold()
  GainCurrency:DirectJump("Gold")
end

function tbWnd:ExecFuncWithFuncName(szFuncName, tbInfo)
  if not self[szFuncName] then
    return
  end
  self[szFuncName](self, tbInfo)
end

function tbWnd:SetAwardList(tbElement, tbInfo, nIndex)
  tbElement.pPanel:Label_SetText("RechargeTips/txtTip1", self:GenTxtTip1(tbInfo))
  tbElement.pPanel:Label_SetText("RechargeTips/txtTip2", self:GenTxtTip2(tbInfo))
  local szText, bEnable, szFuncName = self:GenBtnTextAndStatus(tbInfo)
  tbElement.pPanel:Label_SetText("btnRecharge/Text", szText)
  tbElement.pPanel:Button_SetEnable("btnRecharge", bEnable)
  if szFuncName then
    Ui:Button_BindEvent(tbElement.pPanel, "btnRecharge", self.ExecFuncWithFuncName, self, szFuncName, tbInfo)
  end
  if not tbElement.tbItemGrid then
    tbElement.tbItemGrid = Ui:NewItemGridScrollView(tbElement, "ItemList/Viewport/Content/ItemTemplate", 3, 1)
  else
    tbElement.tbItemGrid:Reset()
  end
  local tbItemList = {}
  for _, tbAwardInfo in ipairs(tbInfo.tbAward) do
    table.insert(tbItemList, tbAwardInfo)
  end
  tbElement.tbItemGrid:UpdateGenericItem(tbItemList, true)
  if not bEnable or "GainAward" == szFuncName then
  end
end
