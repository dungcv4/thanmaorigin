Require("CommonScript/Activity/ActivityMgr.lua")
local tbActivity = ActivityMgr:GetClass(ActivityMgr.Def.ClassNameList.ShopSale)

function tbActivity:GetRedPointEventId()
  return {
    RedPointMgr.EVENT.ShopSale
  }
end

function tbActivity:TryLightRedPoint()
  local pInstance = ActivityMgr:GetInstanceByClassName(ActivityMgr.Def.ClassNameList.ShopSale)
  if pInstance then
    local tbScheduleCfg = pInstance:GetScheduleCfg()
    if LocalData:GetRoleNumber("ShopSale", tbScheduleCfg.nStartTime) <= tbScheduleCfg.nStartTime then
      RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ShopSale)
      LocalData:SetRoleNumber("ShopSale", GetTime())
    end
  end
end

function tbActivity:OnClientClose()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, tbActivity)
end

function tbActivity:OnOpen()
  EventNotify:UnRegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, tbActivity)
  EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, tbActivity.TryLightRedPoint, tbActivity)
end
