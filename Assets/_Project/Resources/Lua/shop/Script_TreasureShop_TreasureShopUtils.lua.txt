function TreasureShop:LoadWareSetting()
  self.tbWareSetting = Lib:EasyLoadTabFile("Setting/TreasureShop/TreasureShopWare.tab", {"nWareId"})
  
  self.tbItemSetting = {}
  for nWareId, tbSetting in pairs(self.tbWareSetting) do
    self.tbItemSetting[tbSetting.nItemTid] = nWareId
  end
  local tbKindFlag = {}
  for _, tbSetting in pairs(self.tbWareSetting) do
    tbKindFlag[i18n.Parse(tbSetting.szKind)] = true
  end
  self.tbKindSetting = Lib:SetToList(tbKindFlag)
end

TreasureShop:LoadWareSetting()

function TreasureShop:GetAllWareSetting()
  return self.tbWareSetting
end

function TreasureShop:GetWareSettingByWareId(nWareId)
  return self.tbWareSetting[nWareId]
end

function TreasureShop:GetWareSettingByItemTid(nItemTid)
  return self.tbItemSetting[nItemTid] and self.tbWareSetting[self.tbItemSetting[nItemTid]] or nil
end
