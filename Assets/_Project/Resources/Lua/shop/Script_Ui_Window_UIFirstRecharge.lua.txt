  end,

    else

    end

function tbUi:OnCreate()
  self.tbDayScrollView = Ui:NewScrollView(self, tbControls.DayTemplate)
  self.tbItemScrollView = {}
end

function tbUi:OnOpen()
  if me.nLevel < Recharge.nFirstRechargeJoinLevel then
    return 1
  end
  self.tbDayScrollView:Clear()
  for i = 1, 3 do
    self.tbDayScrollView:SetElement(i, function(tbElement)
      self.tbItemScrollView[i] = Ui:NewScrollView(tbElement, "Template/ItemGroup/ElementTemplate")
    end)
  end
  RedPointMgr:DynamicRegisteRedPoint(self.pPanel, tbControls.btnGet2, {
    RedPointMgr.EVENT.FirstRecharge
  })
  RedPointMgr:SetRedPointPos(self.pPanel, tbControls.btnGet2, -20, -20, 0)
  self:UpdateDetails()
end

function tbUi:UpdateDetails()
  if RedPointMgr:CheckEvent(RedPointMgr.EVENT.FirstRecharge) then
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.FirstRecharge)
  end
  self:UpdateBtnGet2()
  self:UpdateAllItem()
  self:UpdateValue()
end

function tbUi:GetAwards(nDay)
  local nGetFirst = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_FIRST_RECHARGE)
  if 1 ~= nGetFirst then
    if not IsPayOpen() then
      me.CenterMsg(i18n.EasyGet("\229\133\133\229\128\188\230\156\170\229\188\128\230\148\190"))
      return
    end
    GainCurrency:DirectJump("Gold")
    return
  end
  local nAward = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_AWARD)
  if 0 ~= nAward & Recharge.tbFirstRechargeDayBit[nDay] then
    me.CenterMsg(i18n.EasyGet("\232\175\165\229\165\150\229\138\177\229\183\178\233\162\134\229\143\150\232\191\135"))
    return
  end
  RemoteServer.RequestFirstChargeAward(nDay)
end

function tbUi:UpdateBtnGet2()
  local nGetFirst = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_FIRST_RECHARGE)
  local bHasAwardCanGet = FirstRecharge:CheckHasCanGetAward()
  local bHasAllGet = FirstRecharge:CheckGetAlreadyByDay(1) and FirstRecharge:CheckGetAlreadyByDay(2) and FirstRecharge:CheckGetAlreadyByDay(3)
  self.pPanel:SetActive(tbControls.btnGet2, true)
  self.pPanel:Button_SetEnable(tbControls.btnGet2, true)
  self.pPanel:SetActive(tbControls.Received2, false)
  if 1 == nGetFirst then
    if bHasAllGet then
      self.pPanel:SetActive(tbControls.btnGet2, false)
      self.pPanel:Label_SetText(tbControls.TxtReceived, i18n.EasyGet("\233\166\150\229\133\133\229\165\150\229\138\177\229\133\168\233\131\168\233\162\134\229\143\150"))
      self.pPanel:SetActive(tbControls.Received2, true)
    elseif not bHasAwardCanGet then
      self.pPanel:SetActive(tbControls.btnGet2, false)
      self.pPanel:Label_SetText(tbControls.TxtReceived, i18n.EasyGet("\228\187\138\230\151\165\229\165\150\229\138\177\229\183\178\233\162\134\229\143\150"))
      self.pPanel:SetActive(tbControls.Received2, true)
    else
      self.pPanel:SetActive(tbControls.btnGet2, true)
      self.pPanel:Label_SetText(tbControls.btnGet2Text, i18n.EasyGet("\233\162\134\229\143\150\229\165\150\229\138\177"))
    end
  else
    self.pPanel:SetActive(tbControls.btnGet2, true)
    self.pPanel:Label_SetText(tbControls.btnGet2Text, i18n.EasyGet("\228\187\187\230\132\143\229\133\133\229\128\188"))
  end
end

function tbUi:UpdateAllItem()
  local nGetFirst = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_GET_FIRST_RECHARGE)
  local nTime = me.GetUserValue(Recharge.FIRST_RECHARGE_GROUP, Recharge.KEY_FIRST_RECHARGE_TIME)
  local nDayOffset = Lib:CalcDayOffset(14400, nTime)
  for nDay = 1, 3 do
    local tbDayElement = self.tbDayScrollView:GetElement(nDay)
    tbDayElement.pPanel:Label_SetText("Template/txtDay", self.tbDayMark[nDay]())
    tbDayElement.pPanel:Label_SetText("Template/txtSendGroup/txtnub", Recharge.tbFirstGold[nDay].nValue)
    local bGetAlready = FirstRecharge:CheckGetAlreadyByDay(nDay)
    tbDayElement.pPanel:SetActive(tbTemplateControls.imgComplete, bGetAlready)
    if 1 == nGetFirst and nDay <= nDayOffset + 1 then
      tbDayElement.pPanel:SetActive(tbTemplateControls.imgBgHL, not bGetAlready)
      if not FirstRecharge:CheckGetAlreadyByDay(nDay) and not RedPointMgr:CheckEvent(RedPointMgr.EVENT.FirstRecharge) then
        RedPointMgr:LightRedPoint(RedPointMgr.EVENT.FirstRecharge)
      end
    else
      tbDayElement.pPanel:SetActive(tbTemplateControls.imgBgHL, false)
    end
    local tbAwards = Award:GetAward(Recharge.nFirstRechargeAwardId, nDay)
    
    local function fnSetItem(tbItemElement, nIndex, nDay)
      local tbAwardItem = tbAwards[nIndex]
      local bIsHL = 1 == nDay and 1 == nIndex
      tbItemElement.pPanel:SetActive("btn_Effect", bIsHL)
      tbItemElement.itemGrid1 = tbItemElement.itemGrid1 or Ui:NewExistItemGrid(tbItemElement, "UIItemGrid")
      tbItemElement.itemGrid1:SetGenericItem(tbAwardItem)
      tbItemElement.itemGrid1.fnClick = tbItemElement.itemGrid1.DefaultClick
    end
    
    for nIndex = 1, 4 do
      self.tbItemScrollView[nDay]:SetElement(nIndex, function(tbItemElement)
        fnSetItem(tbItemElement, nIndex, nDay)
      end)
    end
  end
end

local function fnGetNum(nNum)
  local nTmp = nNum
  local nRlt = 0
  while nTmp > 0 do
    nRlt = nRlt + 1
    nTmp = math.floor(nTmp / 10)
  end
  return nRlt
end

function tbUi:UpdateValue()
  local nPrice = FirstRecharge.Cfg.nTotalValueText
  local nNum = fnGetNum(nPrice)
  for nIdx, szPath in ipairs(tbValueNumList) do
    self.pPanel:SetActive(szPath, false)
  end
  for nIndex = nNum - 1, 0, -1 do
    local nValNum = math.floor(nPrice % 10)
    self.pPanel:Sprite_SetSprite(tbValueNumList[nIndex], tbValueNumImagePath[nValNum], true)
    self.pPanel:SetActive(tbValueNumList[nIndex], true)
    nPrice = nPrice / 10
  end
end
