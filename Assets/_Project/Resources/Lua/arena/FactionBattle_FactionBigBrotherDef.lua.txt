FactionBigBrother.KEY_VOTE = 1
FactionBigBrother.KEY_START_TIME = 2
FactionBigBrother.KEY_MAX_VOTE = 3
FactionBigBrother.KEY_MAX_VOTE_MONTH = 4

function FactionBigBrother:CheckCommondVote(pPlayer)
  if self:RemainVote(pPlayer) <= 0 then
    return false, i18n.EasyGet("\230\130\168\229\183\178\231\187\143\230\178\161\230\156\137\230\138\149\231\165\168\230\172\161\230\149\176")
  end
  return true
end

function FactionBigBrother:RemainVote(pPlayer)
  pPlayer = pPlayer or me
  return pPlayer.GetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_VOTE)
end

function FactionBigBrother:GetVoteMaxCount(pPlayer)
  pPlayer = pPlayer or me
  return pPlayer.GetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_MAX_VOTE)
end

function FactionBigBrother:SetVoteMaxCount(pPlayer, nVoteCount)
  pPlayer = pPlayer or me
  if nVoteCount < 0 then
    nVoteCount = 0
  end
  local nOldVoteCount = pPlayer.GetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_MAX_VOTE)
  if nVoteCount <= nOldVoteCount then
    return
  end
  local nMonth = Lib:GetLocalMonth(GetTime()) + 1
  pPlayer.SetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_MAX_VOTE, nVoteCount)
  pPlayer.SetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_MAX_VOTE_MONTH, nMonth)
end

function FactionBigBrother:ClearVoteMaxCount(pPlayer)
  pPlayer.SetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_MAX_VOTE, 0)
  pPlayer.SetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_VOTE, 0)
end
