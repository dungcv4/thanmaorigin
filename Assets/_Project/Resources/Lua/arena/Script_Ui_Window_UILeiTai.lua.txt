local tbWnd = Ui:CreateWindow("UILeiTai")
local tbControls = {
  btnCreate = "node/imgBG/Content/btns/btnCreate",
  btnRefresh = "node/imgBG/Content/btns/btnRefresh",
  btnNote = "node/imgBG/Content/btns/btnNote",
  btnHelp = "node/imgBG/txtTitle/btnInfo",
  btnClose = "node/imgBG/btnClose",
  objTemplate = "node/imgBG/Content/Scroll View/Viewport/Content/Template",
  objEmpty = "node/imgBG/Content/empty"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCreate = function(self)
    Ui:OpenWindow("UILeiTaiCreate")
  end,
  btnRefresh = function(self)
    LeiTai:SyncAllRoomDataRequest(true)
  end,
  btnNote = function(self)
    LeiTai:SyncNoteListRequest()
  end,
  btnHelp = function(self)
    Ui:OnHelpClicked("LeiTaiHelp")
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.svRoomList = Ui:NewScrollView(self, self.objTemplate, nil, true, true)
end

function tbWnd:OnOpen()
  self:RefreshRoomList()
end

function tbWnd:OnClose()
end

function tbWnd:RefreshRoomList()
  self.svRoomList:Clear()
  if not LeiTai.tbAllRoom then
    self.pPanel:SetActive(self.objEmpty, true)
    return
  end
  self.pPanel:SetActive(self.objEmpty, 0 == Lib:CountTB(LeiTai.tbAllRoom))
  for i, tbRoom in ipairs(LeiTai.tbAllRoom) do
    self.svRoomList:SetElement(i, function(tbElement)
      local szElementPath = "node/imgBG/Content/Scroll View/Viewport/Content/Element" .. i
      local szType = tbRoom.nRoomType == LeiTai.tbRoomType.Single and i18n.EasyGet("\229\141\149\228\186\186\230\147\130\229\143\176") or i18n.EasyGet("\233\152\159\228\188\141\230\147\130\229\143\176")
      self.pPanel:Label_SetText(szElementPath .. "/txtLeiTaiType", szType)
      self.pPanel:Label_SetText(szElementPath .. "/txtMemberNumber", Lib:CountTB(tbRoom.tbHostId))
      self.pPanel:Label_SetText(szElementPath .. "/txtShouLeiNumber", tbRoom.nMultiWinCount)
      self.pPanel:Label_SetText(szElementPath .. "/txtPlayerName", tbRoom.tbHostCaptainInfo.szName)
      self.pPanel:Label_SetText(szElementPath .. "/PortraitObj/Level/Text", tbRoom.tbHostCaptainInfo.nLevel)
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/imgFactionIcon", Faction:GetIcon(tbRoom.tbHostCaptainInfo.nFaction))
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbRoom.tbHostCaptainInfo.nPortrait))
      PlayerPortraitFrame:SetPortraitFrame(self.pPanel, szElementPath .. "/PortraitObj/PortraitFrame", tbRoom.tbHostCaptainInfo.nPortraitFrame)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/PortraitObj", self.PopUpMenu, self, tbRoom.tbHostCaptainInfo, tbElement)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/btnChallenge", self.OnClickChallenge, self, tbRoom.nRoomId)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/btnGo", self.OnClickGo, self, tbRoom.nRoomId)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/btnLook", self.OnClickLook, self, tbRoom.nRoomId)
      if LeiTai.tbMyRoom ~= nil and LeiTai.tbMyRoom.nRoomId == tbRoom.nRoomId then
        self.pPanel:SetActive(szElementPath .. "/btnChallenge", false)
        self.pPanel:SetActive(szElementPath .. "/btnLook", false)
        self.pPanel:SetActive(szElementPath .. "/btnGo", true)
      elseif tbRoom.nRoomState == LeiTai.tbRoomState.Fighting then
        self.pPanel:SetActive(szElementPath .. "/btnChallenge", false)
        self.pPanel:SetActive(szElementPath .. "/btnLook", true)
        self.pPanel:SetActive(szElementPath .. "/btnGo", false)
      elseif 0 == Lib:CountTB(tbRoom.tbChallengerId) then
        self.pPanel:SetActive(szElementPath .. "/btnChallenge", true)
        self.pPanel:SetActive(szElementPath .. "/btnLook", false)
        self.pPanel:SetActive(szElementPath .. "/btnGo", false)
      else
        self.pPanel:SetActive(szElementPath .. "/btnChallenge", true)
        self.pPanel:SetActive(szElementPath .. "/btnLook", false)
        self.pPanel:SetActive(szElementPath .. "/btnGo", false)
      end
    end, false)
  end
end

function tbWnd:PopUpMenu(tbPlayerInfo, tbElement)
  if tbPlayerInfo.dwID == me.dwID then
    return
  end
  local tbData = {
    dwRoleId = tbPlayerInfo.dwID,
    szName = tbPlayerInfo.szName,
    nPortrait = tbPlayerInfo.nPortrait,
    nFaction = tbPlayerInfo.nFaction,
    nLevel = tbPlayerInfo.nLevel,
    dwKinId = tbPlayerInfo.dwKinId,
    szKinOrTongName = tbPlayerInfo.szKinOrTongName
  }
  Ui:OpenWindow("UIRightPopUpMenu", "leiTai", tbData, tbElement, 420, -200)
end

function tbWnd:OnClickChallenge(nRoomId)
  LeiTai:ApplyJoinRoomRequest(nRoomId)
end

function tbWnd:OnClickGo(nRoomId)
  if not LeiTai.tbMyRoom then
    return
  end
  if nRoomId ~= LeiTai.tbMyRoom.nRoomId then
    return
  end
  Ui:OpenWindow("UILeiTaiRoom")
end

function tbWnd:OnClickLook(nRoomId)
  LeiTai:LookRequest(nRoomId)
end
