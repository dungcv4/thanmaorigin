function MultiPvP:IsTodayCheckPreOpen()
  local nWeekDay = Lib:GetLocalWeekDay()
  local nDate = tonumber(os.date("%d", GetTime()))
  local bCurMonthOpen = self:IsCurMonthOpen(GetTime() - Lib:GetTodaySec() + Lib:ParseTodayTime(self.Cfg.OPEN_TIME))
  local bPreOpen = true
  if nDate < self.Cfg.BEGIN_DATE or nDate > self.Cfg.END_DATE then
    bPreOpen = false
  elseif not self.OPEN_WEEKDAY[nWeekDay] then
    bPreOpen = false
  end
  return bCurMonthOpen and bPreOpen
end

function MultiPvP:IsTodayCheckFinalOpen()
  local nDate = tonumber(os.date("%d", GetTime()))
  local bCurMonthOpen = self:IsCurMonthOpen(GetTime() - Lib:GetTodaySec() + Lib:ParseTodayTime(self.Cfg.OPEN_TIME))
  local bFinalOpen = nDate == self.Cfg.FINAL_DATE
  return bCurMonthOpen and bFinalOpen
end

function MultiPvP:IsCalendarShow(nActivityTime)
  if not self:IsCurMonthOpen(nActivityTime) then
    return false
  end
  local nActivityMonthDay = Lib:GetLocalMonthDay(nActivityTime)
  if nActivityMonthDay < self.Cfg.BEGIN_DATE or nActivityMonthDay > self.Cfg.END_DATE then
    return false
  end
  return true
end

function MultiPvP:IsCurMonthOpen(nActivityTime)
  local nOpenTime = Lib:CalcTimeFrameOpenTime(self.Cfg.OPEN_FRAME)
  if Lib:GetLocalMonth(nActivityTime) == Lib:GetLocalMonth(nOpenTime) and Lib:GetLocalMonthDay(nOpenTime) >= self.Cfg.OPEN_LIMIT_DAY then
    return false
  end
  return true
end

function MultiPvP:IsPreliminaryOpen()
  if 1 ~= GetTimeFrameState(MultiPvP.Cfg.OPEN_FRAME) then
    return false
  end
  local nTimestamp = GetTime() - Env.NEW_DAY_OFFSET_TIME
  local nDate = tonumber(os.date("%d", nTimestamp))
  if nDate < MultiPvP.Cfg.BEGIN_DATE or nDate > MultiPvP.Cfg.END_DATE then
    return false
  end
  return true
end

function MultiPvP:IsFinalOpen()
  if 1 ~= GetTimeFrameState(MultiPvP.Cfg.OPEN_FRAME) then
    return false
  end
  local nTimestamp = GetTime() - Env.NEW_DAY_OFFSET_TIME
  local nDate = tonumber(os.date("%d", nTimestamp))
  if nDate ~= MultiPvP.Cfg.FINAL_DATE then
    return false
  end
  return true
end

function MultiPvP:IsShowFinalMatchDate()
  if 1 ~= GetTimeFrameState(MultiPvP.Cfg.OPEN_FRAME) then
    return false
  end
  local nTimestamp = GetTime() - Env.NEW_DAY_OFFSET_TIME
  local nDate = Lib:GetLocalMonthDay(nTimestamp)
  if nDate >= MultiPvP.Cfg.BEGIN_DATE and nDate < MultiPvP.Cfg.FINAL_DATE - 1 then
    return false
  elseif nDate == MultiPvP.Cfg.FINAL_DATE - 1 and Lib:GetLocalDayHour() < 21 then
    return false
  end
  return true
end

function MultiPvP:IsGuessOpen()
  return self:IsShowFinalMatchDate()
end

function MultiPvP:GetCurBattleSetting()
  if MODULE_GAMESERVER then
    self.m_tbSetting = MultiPvP.BATTLE_SETTING[self:GetOtherData("nLastOpenType")]
  else
    self.m_tbSetting = MultiPvP.BATTLE_SETTING[self:GetCurSeasonType()]
  end
  return self.m_tbSetting
end

function MultiPvP:GetNextBattleSetting()
  local nType = self:GetOtherData("nLastOpenType")
  if not nType then
    nType = MultiPvP.SEASON_SET[1]
    return MultiPvP.BATTLE_SETTING[nType]
  end
  if not MultiPvP.BATTLE_SETTING[nType + 1] then
    nType = MultiPvP.SEASON_SET[1]
  else
    nType = MultiPvP.SEASON_SET[nType + 1]
  end
  return MultiPvP.BATTLE_SETTING[nType]
end

function MultiPvP:GetNextSeasonValue(szKey)
  local tbData = self:GetNextBattleSetting()
  return tbData[szKey]
end

function MultiPvP:GetSeasonSettingValue(szKey, bForceValue)
  local tbData = MultiPvP:GetCurBattleSetting()
  if not tbData then
    if bForceValue then
      return MultiPvP.BATTLE_SETTING[MultiPvP.TYPE.TYPE_3V3MULTI][szKey]
    end
    return
  end
  return tbData[szKey]
end

function MultiPvP:LoadAwardSetting()
  self.tbAwardSetting = {}
  self.tbAwardSetting[self.TYPE_AWARD_PRERANK] = Lib:LoadTabFileEx("Setting/MultiPvP/PreRankAward.tab", "ddsdd", "nId", {
    "nId",
    "nRank",
    "szDesc",
    "nAwardId",
    "nAwardKey"
  })
  self.tbAwardSetting[self.TYPE_AWARD_WINRATE] = Lib:LoadTabFileEx("Setting/MultiPvP/WinRateAward.tab", "ddsdd", "nId", {
    "nId",
    "nRate",
    "szDesc",
    "nAwardId",
    "nAwardKey"
  })
  self.tbAwardSetting[self.TYPE_AWARD_FINALRANK] = Lib:LoadTabFileEx("Setting/MultiPvP/FinalRankAward.tab", "dddsdd", "nId", {
    "nId",
    "nPhase",
    "nWinRate",
    "szDesc",
    "nAwardId",
    "nAwardKey"
  })
end

function MultiPvP:GetAwardSetting(nType)
  return self.tbAwardSetting[nType]
end

function MultiPvP:GetPreRankAward(nPlayerRank)
  local tbAward
  local tbAllAward = self:GetAwardSetting(self.TYPE_AWARD_PRERANK)
  if not tbAllAward then
    return
  end
  for _, tbData in ipairs(tbAllAward) do
    if nPlayerRank <= tbData.nRank then
      tbAward = {
        tbData.nAwardId,
        tbData.nAwardKey
      }
      break
    end
  end
  return tbAward
end

function MultiPvP:GetRankAwardRemoveFinal(nPlayerRank, nWinRate)
  if nPlayerRank <= self.PHASE_INIT then
    return
  end
  local tbAward
  local tbAllAward = self:GetAwardSetting(self.TYPE_AWARD_FINALRANK)
  if not tbAllAward then
    return
  end
  for _, tbData in ipairs(tbAllAward) do
    if nPlayerRank <= tbData.nPhase and nWinRate >= tbData.nWinRate then
      tbAward = {
        tbData.nAwardId,
        tbData.nAwardKey
      }
      break
    end
  end
  return tbAward
end

function MultiPvP:GetWinRateAward(nRate)
  local tbAward
  local tbAllAward = self:GetAwardSetting(self.TYPE_AWARD_WINRATE)
  if not tbAllAward then
    return
  end
  for _, tbData in ipairs(tbAllAward) do
    if nRate >= tbData.nRate then
      tbAward = {
        tbData.nAwardId,
        tbData.nAwardKey
      }
      break
    end
  end
  return tbAward
end
