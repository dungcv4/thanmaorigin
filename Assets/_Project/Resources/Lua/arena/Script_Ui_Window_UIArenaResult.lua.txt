local tbWnd = Ui:CreateWindow("UIArenaResult")
tbWnd.tbControls = {
  Victory = "imgBG/Victory",
  Defeat = "imgBG/Defeat",
  WinText1 = "imgBG/Victory/VictoryEffect/imgBG/Text"
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(bResult, nCloseTimeCount, szWinTxt)
  local nSoundId = 0
  if 1 == bResult then
    self.pPanel:SetActive(self.Victory, true)
    self.pPanel:SetActive(self.Defeat, false)
    Sound:PlaySound("ArenaWin", _)
    if szWinTxt then
      self.pPanel:Label_SetText(self.WinText1, szWinTxt)
    else
      self.pPanel:Label_SetText(self.WinText1, i18n.EasyGet("\232\131\156\229\136\169"))
    end
  else
    self.pPanel:SetActive(self.Victory, false)
    self.pPanel:SetActive(self.Defeat, true)
    Sound:PlaySound("ArenaFail", _)
  end
  nCloseTimeCount = nCloseTimeCount or 5
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * nCloseTimeCount, function()
    self:Close()
  end)
end

function tbWnd:OnClose()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
end
