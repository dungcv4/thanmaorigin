function Battle:InitRandomName()
  self.tbRandomName1 = Lib:LoadTabFileEx("Setting/RandomName/Xing.tab", "s", nil, {"Name"})
  self.tbRandomName2 = Lib:LoadTabFileEx("Setting/RandomName/Ming.tab", "s", nil, {"Name"})
  self.tbRandomName3 = Lib:LoadTabFileEx("Setting/RandomName/Female.tab", "s", nil, {"Name"})
  if not self.tbRandomName1 or not self.tbRandomName2 then
    Log("error Battle:Init , load RandomName ")
  end
end

if not Battle.tbRandomName1 then
  Battle:InitRandomName()
end

function Battle:GetRandomName(nSex, nFaction)
  local nPreIndx = MathRandom(#self.tbRandomName1)
  local tbRandomMing = self.tbRandomName2
  if nSex and 1 == nSex then
    tbRandomMing = self.tbRandomName3
  end
  local nSuffiIndx = MathRandom(#tbRandomMing)
  local szName = self.tbRandomName1[nPreIndx].Name .. tbRandomMing[nSuffiIndx].Name
  return szName
end

function Battle:InitTitleAndScore()
  local tbTempTitle = Lib:LoadTabFileEx("Setting/Battle/BattleTitle.tab", "ddd", nil, {
    "Score",
    "SongTitleId",
    "JinTitleId"
  })
  self.tbTitle = {}
  for i = 1, #tbTempTitle do
    self.tbTitle[i] = {
      nScore = tbTempTitle[i].Score,
      tbTitleId = {
        [1] = tbTempTitle[i].SongTitleId,
        [2] = tbTempTitle[i].JinTitleId
      }
    }
  end
  local tbTempScore = Lib:LoadTabFileEx("Setting/Battle/BattleKillScore.tab", "ddddd", nil, {
    "col_1",
    "col_2",
    "col_3",
    "col_4",
    "col_5"
  })
  self.tbKillScore = {}
  for i = 1, #tbTempScore do
    self.tbKillScore[i] = {
      tbTempScore[i].col_1,
      tbTempScore[i].col_2,
      tbTempScore[i].col_3,
      tbTempScore[i].col_4,
      tbTempScore[i].col_5
    }
  end
end

Battle:InitTitleAndScore()

function Battle:Score2TitleInfo(nScore)
  for i = #self.tbTitle, 1, -1 do
    if nScore > self.tbTitle[i].nScore then
      return i, self.tbTitle[i].tbTitleId
    end
  end
  return 1, self.tbTitle[1].tbTitleId
end

function Battle:RemoveBattleTitle(pPlayer)
  for i = 1, #self.tbTitle do
    pPlayer.DeleteTitle(self.tbTitle[i].tbTitleId[1], true)
    pPlayer.DeleteTitle(self.tbTitle[i].tbTitleId[2], true)
  end
end

function Battle:OnLoginCheck(pPlayer)
  if not Battle.tbBattleMapTemplateId[pPlayer.nMapTemplateId] then
    Battle:RemoveBattleTitle(pPlayer)
  end
end

Battle.tbClass = Battle.tbClass or {}

function Battle:CreateClass(szNewClass, szBaseClass)
  if szBaseClass and self.tbClass[szBaseClass] then
    self.tbClass[szNewClass] = Lib:NewClass(self.tbClass[szBaseClass])
  else
    self.tbClass[szNewClass] = {}
  end
  local tbClassInst = self.tbClass[szNewClass]
  return tbClassInst
end

function Battle:GetClass(szClassName, szBaseClassName)
  if self.tbClass[szClassName] then
    return self.tbClass[szClassName]
  else
    return Battle:CreateClass(szClassName, szBaseClassName)
  end
end

function Battle:GetFakePlayerShowId(nFaction, nFactionSect, nSex)
  local _, _, _, _, nWeaponResID = KItem.GetItemShowInfo(self.tbFakePlayerEquipTemplateId[Item.EQUIPPOS_WEAPON], nFaction, nFactionSect, nSex)
  local _, _, _, _, nArmorResID = KItem.GetItemShowInfo(self.tbFakePlayerEquipTemplateId[Item.EQUIPPOS_ARMOR], nFaction, nFactionSect, nSex)
  local _, _, _, _, nHelmResID = KItem.GetItemShowInfo(self.tbFakePlayerEquipTemplateId[Item.EQUIPPOS_HELM], nFaction, nFactionSect, nSex)
  local nMantleResID = 0
  return {
    nBodyRes = nArmorResID,
    nHeadRes = nHelmResID,
    nWeaponRes = nWeaponResID,
    nMantleRes = nMantleResID
  }
end

function Battle:LoadNpcSetting()
  self.tbNpcSetting = {}
  self.tbBackCampPos = {}
  self.tbAidBuffPos = {}
  local szScoreAwardFile = "Setting/Battle/ScoreAward.tab"
  self.tbScoreAward = Lib:LoadTabFileEx(szScoreAwardFile, "dssd", nil, {
    "nScore",
    "szTimeFrameStart",
    "szTimeFrameEnd",
    "nAwardKey"
  }, nil, nil, false)
  local nFirstTime, nSecondTime
  table.sort(self.tbScoreAward, function(tbFirst, tbSecond)
    if tbFirst.nScore ~= tbSecond.nScore then
      return tbFirst.nScore > tbSecond.nScore
    end
    nFirstTime = Lib:IsEmptyStr(tbFirst.szTimeFrameStart) and 0 or Lib:CalcTimeFrameOpenTime(tbFirst.szTimeFrameStart)
    nSecondTime = Lib:IsEmptyStr(tbSecond.szTimeFrameStart) and 0 or Lib:CalcTimeFrameOpenTime(tbSecond.szTimeFrameStart)
    return nFirstTime > nSecondTime
  end)
  self.tbScoreAward = SetReadOnly(self.tbScoreAward)
end

Battle:LoadNpcSetting()

function Battle:LoadRevivePosSetting()
  self.tbRevivePos = {}
  local szFile = Map:GetMapPosFile(self.nBattleMapTemplateId, "battle_revivePos.txt")
  local tbPosList = Lib:LoadTabFileEx(szFile, "ddd", nil, {
    "Camp",
    "PosX",
    "PosY"
  }, nil, nil, true)
  local tbRevivePos = {}
  for _, tbPos in ipairs(tbPosList) do
    if not tbRevivePos[tbPos.Camp] then
      tbRevivePos[tbPos.Camp] = {}

      tbPos.PosX,
      tbPos.PosY

  end
  self.tbRevivePos[self.nBattleMapTemplateId] = tbRevivePos
  local szFile = Map:GetMapPosFile(self.nBattleKFMapTemplateId, "battle_revivePos.txt")

  for _, tbPos in ipairs(tbPosList) do

    end

      tbPos.PosX,
      tbPos.PosY
    })
  end
  self.tbRevivePos[self.nBattleKFMapTemplateId] = tbKFRevivePos
