function FactionBattle:LoadEnterPoint()
  FactionBattle.Def.ENTER_POS = {}
  local tbEnter_Pos = Lib:LoadTabFileEx(FactionBattle.Def.ENTER_POSFILE, "dd", nil, {"PosX", "PosY"}, nil, nil, true)
  if not tbEnter_Pos then
    error("Cannot read enter position")
  else
    for _, pos in ipairs(tbEnter_Pos) do
      table.insert(FactionBattle.Def.ENTER_POS, {
        pos.PosX,
        pos.PosY
      })
    end
  end
end

function FactionBattle:LoadNPC()
  local tbNPCList = Lib:LoadTabFileEx(FactionBattle.Def.NPC_SETTING, "sdd", nil, {
    "Type",
    "PosX",
    "PosY"
  })
  for _, tbPos in ipairs(tbNPCList) do
    if tbPos.Type == "FLAG_POS" then
      FactionBattle.Def.FLAG_POS = {
        tbPos.PosX,
        tbPos.PosY
      }
    elseif tbPos.Type == "AWARD_POS" then
      FactionBattle.Def.AWARD_POS = {
        tbPos.PosX,
        tbPos.PosY
      }
    elseif tbPos.Type == "tbGOUHUO_POS" then
      FactionBattle.Def.tbGOUHUO_POS = FactionBattle.Def.tbGOUHUO_POS or {}
      table.insert(FactionBattle.Def.tbGOUHUO_POS, {
        tbPos.PosX,
        tbPos.PosY
      })
    elseif tbPos.Type == "FIREWORK_POS" then
      FactionBattle.Def.FIREWORK_POS = FactionBattle.Def.FIREWORK_POS or {}
      table.insert(FactionBattle.Def.FIREWORK_POS, {
        tbPos.PosX,
        tbPos.PosY
      })
    end
  end
  local szPath = "Setting/Map/menpaijingjichang/info/npc_info.txt"
  FactionBattle.Def.tbFireWorkNpc = Lib:LoadTabFileEx(szPath, "dddd", nil, {
    "NpcTemplateId",
    "NpcLevel",
    "XPos",
    "YPos"
  })
end

function FactionBattle:LoadMailSetting()
  FactionBattle.Def.tbMailSetting = {}
  local tbReadMailSetting = Lib:LoadTabFileEx(FactionBattle.Def.MAIL_SETTING, "dsss", "nFaction", {
    "nFaction",
    "szSender",
    "szTitle",
    "szText"
  })
  if not tbReadMailSetting then
    return
  end
  for _, tbMail in ipairs(tbReadMailSetting) do
    FactionBattle.Def.tbMailSetting[tbMail.nFaction] = {
      szSender = tbMail.szSender,
      szTitle = tbMail.szTitle,
      szText = tbMail.szText
    }
  end
end
