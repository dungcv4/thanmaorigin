function TongBattle:IsPreviewDay()
  if not self.IS_SYS_OPEN then
    return false
  end
  if self:GetTimeLocalDay(self.TimeCfg.nPreviewTime) ~= Lib:GetLocalDay(GetTime()) then
    return false
  end
  return true
end

  if not self.IS_SYS_OPEN then

function TongBattle:GetTimeLocalDay(nDateTime)
  local nTime = Lib:ParseDateTime(tostring(nDateTime))
  if not nTime then
    return false
  end
  return Lib:GetLocalDay(nTime)
end

function TongBattle:IsGameMap(nMapTid)
  for _, tbData in pairs(self.MAP_FIGHT) do
    if tbData.nMapId == nMapTid then
      return true
    end
  end
  return false
end

function TongBattle:GetStageGuessStartTime(nWarZone, nStage)
  if not self.FinalsStageCfg[nStage] then
    return
  end
  if not self.GuessCfg.Time[nStage] then
    return
  end
  return Lib:GetDate2Time(self.GuessCfg.Time[nStage].nStartTime)
end

function TongBattle:GetStageGuessEndTime(nWarZone, nStage)
  if not self.FinalsStageCfg[nStage] then
    return
  end
  local nMatchIndex = self.FinalsStageCfg[nStage].nStartMatch
  for nDate, tbData1 in pairs(self.tbFinalsSchedule) do
    for nTime, tbData2 in pairs(tbData1) do
      for _, tbData3 in ipairs(tbData2) do
        if tbData3.nWarZone == nWarZone then
          for _, nMatch in pairs(tbData3.tbMatch) do
            if nMatch == nMatchIndex then
              return Lib:GetDate2Time(nDate * 10000 + nTime) - 600
            end
          end
        end
      end
    end
  end
end

function TongBattle:GetStageByFinalsMatch(nFinalsMatch)
  for nStage, tbData in ipairs(self.FinalsStageCfg) do
    if nFinalsMatch >= tbData.nStartMatch and nFinalsMatch <= tbData.nEndMatch then
      return nStage
    end
  end
  return 0
end

function TongBattle:GetNowFinalsSatge(tbFinalsData)
  local nMatchIndex = 0
  for nFinalsMatch = 1, self.FINALS_MATCH_NUM do
    if tbFinalsData[nFinalsMatch] and -1 == tbFinalsData[nFinalsMatch].nWinner then
      nMatchIndex = nFinalsMatch
      break
    end
  end
  return self:GetStageByFinalsMatch(nMatchIndex)
end

function TongBattle:CheckGmWarch(pPlayer)
  local nGmCard = pPlayer.GetItemCountByTypeInBag(TongBattle.Cfg.nGmCardId)
  return nGmCard > 0
end
