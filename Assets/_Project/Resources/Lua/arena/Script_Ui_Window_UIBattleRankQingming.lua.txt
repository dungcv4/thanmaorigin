local tbWnd = Ui:CreateWindow("UIBattleRankQingming")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  RankTemplate = "node/imgBG/Content/imgList/Viewport/List/Template",
  txtTitle = "node/imgBG/txtTitle",
  txtRank = "node/imgBG/Content/myInfo/txtRank",
  txtName = "node/imgBG/Content/myInfo/txtName",
  txtTime = "node/imgBG/Content/myInfo/txtTime",
  txtScore = "node/imgBG/Content/myInfo/txtScore",
  txtRankTitle = "node/imgBG/Content/PlayerListTitle/txtRank",
  txtNameTitle = "node/imgBG/Content/PlayerListTitle/txtName",
  txtTimeTitle = "node/imgBG/Content/PlayerListTitle/txtTime",
  txtScoreTitle = "node/imgBG/Content/PlayerListTitle/txtScore"
}
tbWnd.tbControls = tbControls
local tbTitle = {
  tbControls.txtRankTitle,
  tbControls.txtNameTitle,
  tbControls.txtTimeTitle,
  tbControls.txtScoreTitle
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewLoopScrollView(self, tbControls.RankTemplate)
end

function tbWnd:OnOpen(tbData)
  self.pPanel:Label_SetText(tbControls.txtTitle, tbData.szPanelName)
  self.tbTitle = tbData.tbTitle
  self.tbRankData = tbData.tbRankData
  self.nMyRank = tbData.nMyRank
  self:SetTitle()
  self:SetRank()
end

function tbWnd:SetTitle()
  for nIndex, szPath in ipairs(tbTitle) do
    self.pPanel:Label_SetText(szPath, self.tbTitle[nIndex])
  end
end

function tbWnd:SetRank()
  self.tbScrollView:Clear()
  for nRank = 1, #self.tbRankData do
    self.tbScrollView:SetElement(nRank, function(tbLoopScroll, nInnerRank)
      local tbElement = tbLoopScroll:GetElement(nInnerRank)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtRank", self.tbRankData[nInnerRank].nParam1)
      pPanel:Label_SetText("txtName", self.tbRankData[nInnerRank].nParam2)
      pPanel:Label_SetText("txtTime", self.tbRankData[nInnerRank].nParam3)
      pPanel:Label_SetText("txtScore", self.tbRankData[nInnerRank].nParam4)
    end)
  end
  local tbMyPlayerData = self.tbRankData[self.nMyRank]
  self.pPanel:Label_SetText(tbControls.txtRank, tbMyPlayerData.nParam1)
  self.pPanel:Label_SetText(tbControls.txtName, tbMyPlayerData.nParam2)
  self.pPanel:Label_SetText(tbControls.txtTime, tbMyPlayerData.nParam3)
  self.pPanel:Label_SetText(tbControls.txtScore, tbMyPlayerData.nParam4)
end
