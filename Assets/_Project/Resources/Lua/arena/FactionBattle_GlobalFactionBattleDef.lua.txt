function GlobalFactionBattle:LoadEnterPoint()
  GlobalFactionBattle.Def.ENTER_POS = {}
  local tbEnter_Pos = Lib:LoadTabFileEx(GlobalFactionBattle.Def.ENTER_POSFILE, "dd", nil, {"PosX", "PosY"}, nil, nil, true)
  if not tbEnter_Pos then
    error("Cannot read enter position")
  else
    for _, pos in ipairs(tbEnter_Pos) do
      table.insert(GlobalFactionBattle.Def.ENTER_POS, {
        pos.PosX,
        pos.PosY
      })
    end
  end
end

function GlobalFactionBattle:LoadNPC()
  local tbNPCList = Lib:LoadTabFileEx(GlobalFactionBattle.Def.NPC_SETTING, "sdd", nil, {
    "Type",
    "PosX",
    "PosY"
  })
  for _, tbPos in ipairs(tbNPCList) do
    if tbPos.Type == "FLAG_POS" then
      GlobalFactionBattle.Def.FLAG_POS = {
        tbPos.PosX,
        tbPos.PosY
      }
    elseif tbPos.Type == "AWARD_POS" then
      GlobalFactionBattle.Def.AWARD_POS = {
        tbPos.PosX,
        tbPos.PosY
      }
    elseif tbPos.Type == "tbGOUHUO_POS" then
      GlobalFactionBattle.Def.tbGOUHUO_POS = GlobalFactionBattle.Def.tbGOUHUO_POS or {}
      table.insert(GlobalFactionBattle.Def.tbGOUHUO_POS, {
        tbPos.PosX,
        tbPos.PosY
      })
    elseif tbPos.Type == "FIREWORK_POS" then
      GlobalFactionBattle.Def.FIREWORK_POS = GlobalFactionBattle.Def.FIREWORK_POS or {}
      table.insert(GlobalFactionBattle.Def.FIREWORK_POS, {
        tbPos.PosX,
        tbPos.PosY
      })
    end
  end
  local szPath = "Setting/Map/menpaijingjichang/info/npc_info.txt"
  GlobalFactionBattle.Def.tbFireWorkNpc = Lib:LoadTabFileEx(szPath, "dddd", nil, {
    "NpcTemplateId",
    "NpcLevel",
    "XPos",
    "YPos"
  })
end

function GlobalFactionBattle:LoadMailSetting()
  GlobalFactionBattle.Def.tbMailSetting = {}
  local tbReadMailSetting = Lib:LoadTabFileEx(GlobalFactionBattle.Def.MAIL_SETTING, "dsss", "nFaction", {
    "nFaction",
    "szSender",
    "szTitle",
    "szText"
  })
  if not tbReadMailSetting then
    return
  end
  for _, tbMail in ipairs(tbReadMailSetting) do
    GlobalFactionBattle.Def.tbMailSetting[tbMail.nFaction] = {
      szSender = tbMail.szSender,
      szTitle = tbMail.szTitle,
      szText = tbMail.szText
    }
  end
end
