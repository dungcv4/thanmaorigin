local tbWnd = Ui:CreateWindow("UILeiTaiCreate")
local tbControls = {
  btnCreate = "imgBG/content/btnCreate",
  btnCancel = "imgBG/content/btnCancel",
  btnClose = "imgBG/btnClose",
  togSingle = "imgBG/content/tg/tgSingle",
  togTeam = "imgBG/content/tg/tgTeam"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCreate = function(self)
    self:CreateLeiTai()
    self:Close()
  end,
  btnCancel = function(self)
    self:Close()
  end,
  btnClose = function(self)
    self:Close()
  end
}
tbWnd.tbOnToggleClick = {
  togSingle = function(self, isOn)
    if isOn then
      self:SetRoomTypeToggle(LeiTai.tbRoomType.Single)
    end
  end,
  togTeam = function(self, isOn)
    if isOn then
      self:SetRoomTypeToggle(LeiTai.tbRoomType.Team)
    end
  end
}

function tbWnd:SetRoomTypeToggle(nRoomType)
  self.nRoomType = nRoomType
  self.pPanel:Toggle_SetChecked(tbControls.togSingle, self.nRoomType == LeiTai.tbRoomType.Single)
  self.pPanel:Toggle_SetChecked(tbControls.togTeam, self.nRoomType == LeiTai.tbRoomType.Team)
end

function tbWnd:OnOpen()
  self:SetRoomTypeToggle(LeiTai.tbRoomType.Single)
end

function tbWnd:CreateLeiTai()
  LeiTai:CreateRoomRequest(self.nRoomType)
end
