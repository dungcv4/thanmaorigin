local tbWnd = Ui:CreateWindow("UIBattleResult")
tbWnd.tbControls = {
  txtResult = "imgBG/Content/txtResult",
  txtResultValue = "imgBG/Content/txtResult/txtResultValue",
  btnQuickFinish = "imgBG/Content/btnQuickFinish",
  Victory = "imgBG/Victory",
  Defeat = "imgBG/Defeat"
}
tbWnd.tbOnClick = {
  btnQuickFinish = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
end

function tbWnd:OnOpen(bResult, nChangeScore, nCloseTimeCount)
  if 1 == bResult then
    self.pPanel:SetActive(self.Victory, true)
    self.pPanel:SetActive(self.Defeat, false)
  else
    self.pPanel:SetActive(self.Victory, false)
    self.pPanel:SetActive(self.Defeat, true)
  end
  if nChangeScore then
    self.pPanel:Label_SetText(self.txtResultValue, nChangeScore)
    self.pPanel:SetActive(self.txtResult, true)
    self.pPanel:SetActive(self.btnQuickFinish, false)
    if not nCloseTimeCount then
      nCloseTimeCount = 5
    end
  else
    self.pPanel:SetActive(self.txtResult, false)
    self.pPanel:SetActive(self.btnQuickFinish, false)
    nCloseTimeCount = nCloseTimeCount or 3
  end
  self.nCloseTimer = Timer:Register(Env.GAME_FPS * nCloseTimeCount, function()
    self:Close()
  end)
end

function tbWnd:OnClose()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
    self.nCloseTimer = nil
  end
end
