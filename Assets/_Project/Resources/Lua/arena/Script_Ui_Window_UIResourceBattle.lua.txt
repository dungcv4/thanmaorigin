local tbWnd = Ui:CreateClass("UIResourceBattle", "UIPanel")
local tbControls = {
  btnCheck = "btnCheck",
  Template = "imgList/List/Template",
  txtTitle = "DomainListTitle/txtTong",
  txtStar = "DomainListTitle/txtStar",
  txtCheck = "btnCheck/Text",
  txtInfo = "txtInfo2"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCheck = function(self)
    self.nWarZoneType = self.nWarZoneType % 2 + 1
    self:UpdateSV()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.pPanel:Label_SetText(self.txtCheck, i18n.EasyGet("\229\136\135\230\141\162\230\136\152\229\140\186"))
  self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\229\134\155\229\155\162"))
  self.pPanel:Label_SetText(self.txtStar, i18n.EasyGet("\231\167\175\229\136\134"))
end

function tbWnd:OnOpen()
  self.nWarZoneType = self.nWarZoneType or 1
  if not ResourceBattle.tbSeasonData then
    return
  end
  self:UpdateSV()
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSV()
  self.pPanel:Label_SetText(self.txtInfo, i18n.EasyGet("{0}\230\136\152\229\140\186\231\131\189\231\129\171\231\159\191\232\132\137\230\131\133\229\134\181\229\166\130\228\184\139", i18n.Get(Legion.Cfg.WarZone.tbName[self.nWarZoneType].szName)))
  local tbSVData = {}
  local tbTempData = {}
  if 1 == #ResourceBattle.tbSeasonData then
    self.pPanel:SetActive(self.btnCheck, false)
  end
  local tbData = ResourceBattle.tbSeasonData[self.nWarZoneType] or {}
  self.tbScrollView:Reset()
  
  local function fnShowItem(tbElement, nIdx, tbInfo)
    tbElement.pPanel:Label_SetText("txtRank", nIdx)
    tbElement.pPanel:Label_SetText("StarList/txtStar", tbInfo.nScore)
    tbElement.pPanel:Label_SetText("txtTong", tbInfo.szName)
    tbElement.pPanel:Label_SetText("txtMasterName", tbInfo.szMasterName)
    tbElement.pPanel:SetActive("StarList/StarTemplate", false)
  end
  
  for i = 1, #tbData do
    self.tbScrollView:SetElement(i, function(tbElement)
      fnShowItem(tbElement, i, tbData[i])
    end)
  end
end
