function ResourceBattleMining:GetPlayerPartnerListForSelect(tbPreferSerie)
  local tbQuality = {}
  local tbAllPartner = me.GetAllPartner()
  for nIndex, tbPartnerInfo in pairs(tbAllPartner) do
    local nTemplateId = tbPartnerInfo.nTemplateId
    local nSeries = tbPartnerInfo.nSeries
    local nQuality = Partner:GetPartnerQualityLevel(nTemplateId)
    tbQuality[nQuality] = tbQuality[nQuality] or {}
    table.insert(tbQuality[nQuality], {nTemplateId, nSeries})
  end
  local tbShow = {}
  for nQuality, v in pairs(tbQuality) do
    table.insert(tbShow, {nQuality, v})
    table.sort(v, function(a, b)
      local nSeriesA = a[2]
      local nSeriesB = b[2]
      if nSeriesA ~= nSeriesB then
        return nSeriesA < nSeriesB
      end
      return a[1] < b[1]
    end)
  end
  table.sort(tbShow, function(a, b)
    return a[1] < b[1]
  end)
  return tbShow
end

  EventNotify.OnNotify(EventNotify.emNOTIFY_SYNC_RESOURCE_BATTLE_MINING_INFO)
  self:CheckRedPoint()
end

function ResourceBattleMining:CheckRedPoint()
  if self.bIsCheckRedPoint then
    return
  end
  if not self:IsSystemOpen(me) then
    return
  end
  if not self:IsDoMiningTime() then
    return
  end
  if self:IsMinedToday(me) then
    return
  end
  self.bIsCheckRedPoint = true
  RedPointMgr:LightRedPoint(RedPointMgr.EVENT.ResourceBattleMining_CanMine)
end
