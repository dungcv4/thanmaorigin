function TongBattle:RequestApplyQualification(tbQualification)
  if type(tbQualification) ~= "table" then
    return
  end
  local tbTongBaseInfo = Tong:GetBaseInfo() or {}
  if tbTongBaseInfo.nMasterId ~= me.dwID then
    me.CenterMsg(i18n.EasyGet("\228\187\133\229\184\174\228\184\187\229\143\175\228\187\165\228\191\174\230\148\185\229\143\130\232\181\155\229\144\141\229\141\149"))
    return
  end
  if Lib:IsEmptyTB(tbQualification) then
    me.CenterMsg(i18n.EasyGet("\229\143\130\232\181\155\229\144\141\229\141\149\228\184\141\232\131\189\228\184\186\231\169\186"))
    return
  end
  if Lib:CountTB(tbQualification) > self.Cfg.nFinalMaxJoinPlayer then
    me.CenterMsg(i18n.EasyGet("\229\143\130\232\181\155\228\186\186\230\149\176\228\184\141\232\131\189\229\164\167\228\186\142{0}\228\186\186", self.Cfg.nFinalMaxJoinPlayer))
    return
  end
  RemoteServer.RequestTongBattle("ApplyQualification", tbQualification)
end

function TongBattle:OnSyncQualification(tbQualification, tbPlayerScore)
  self.tbQualification = tbQualification
  self.tbPlayerScore = tbPlayerScore
  local tbMemberList = Tong:GetMemberList() or {}
  local tbOnlineMembers = Tong:GetMemberState() or {}
  local tbTongBaseInfo = Tong:GetBaseInfo() or {}
  local tbUIRankDataList = {}
  for nRank, tbMember in ipairs(tbMemberList) do
    local szHonorName, szHonorColor = HonorLevel:GetHonorNameByStarLevel(tbMember.nHonorStarLevel, false)
    local szCareer = i18n.Get(Tong.Def.CareerName[tbMember.nCareer])
    if tbMember.nCareer == Tong.Def.Career_Leader and tbTongBaseInfo.nMasterId == tbMember.nMemberId then
      szCareer = szCareer .. "/" .. i18n.Get(Tong.Def.CareerName[Tong.Def.Career_Master])
    end
    local bOnLine = tbOnlineMembers[tbMember.nMemberId]
    local szOnLine = ""
    if bOnLine then
      szOnLine = i18n.Get(Tong.Cfg.Msg.Online)
    else
      szOnLine = string.format("<color=#352620>%s</color>", i18n.Get(Tong.Cfg.Msg.Offline))
    end
    table.insert(tbUIRankDataList, {
      [1] = {
        nPlayerId = tbMember.nMemberId,
        szPlayerName = tbMember.szName,
        tbTitle = {
          szHonorName = i18n.Parse(szHonorName),
          szHonorColor = szHonorColor
        },
        nVipLevel = tbMember.nVipLevel
      },
      [2] = szCareer,
      [3] = tbPlayerScore[tbMember.nMemberId] or 0,
      [4] = szOnLine,
      [5] = tbQualification[tbMember.nMemberId] or false
    })
  end
  table.sort(tbUIRankDataList, function(a, b)
    return a[3] > b[3]
  end)
  if not Ui:WindowVisible("UIGangRanking") then
    Ui:OpenWindow("UIGangRanking", {
      TongBattle.RequestApplyQualification,
      TongBattle
    }, nil, {
      [1] = i18n.EasyGet("\229\144\141\229\173\151"),
      [2] = i18n.EasyGet("\232\129\140\228\189\141"),
      [3] = i18n.EasyGet("\231\167\175\229\136\134"),
      [4] = i18n.EasyGet("\231\166\187\231\186\191"),
      [5] = nil
    }, tbUIRankDataList, TongBattle.Cfg.nFinalMaxJoinPlayer)
  end
  Ui:CloseWindow("UIActivity_Banghuizhengduosai")
end

function TongBattle:IsNeedFubenBtn()
  if self.tbPillarState then
    local tbBtnList = {}
    
    local function fnGoto(nIndex)
      if self.tbPillarState[nIndex] then
        if self.tbPillarState[nIndex].nTongID == me.dwTongId or TongBattle:CheckGmWarch(me) then
          RemoteServer.RequestTongBattle("Transmit", nIndex)
        else
          me.CenterMsg(i18n.EasyGet("\229\141\160\233\162\134\233\190\153\230\159\177\229\144\142\230\150\185\229\143\175\228\188\160\233\128\129"))
        end
      end
    end
    
    for nIndex, tbData in ipairs(self.tbPillarState) do
      if tbData.nTongID == me.dwTongId or TongBattle:CheckGmWarch(me) then
        table.insert(tbBtnList, {
          szName = i18n.Parse(tbData.szName),
          fnOnClick = function()
            fnGoto(nIndex)
          end
        })
      end
    end
    return true, nil, tbBtnList
  end
  return false
end
