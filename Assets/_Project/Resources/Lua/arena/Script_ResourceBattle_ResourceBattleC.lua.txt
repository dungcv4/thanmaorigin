function RaceMap:ShowPos()
  local tbTemp1 = Lib:CopyTB(self.tbFixPoint)
  local tbTemp2 = Lib:CopyTB(self.tbDynamicPoint)
  local tbPoint = Lib:MergeTable(tbTemp1, tbTemp2)
  Map:SyncDynMapPoint(ResourceBattle.Cfg.nRaceMapTemplate, tbPoint, GetTime() + self.nMiniMapTimeout)
end

function RaceMap:UpdateRaceMapStartUI(nTime, tbData)
  self.bInStart = true
  local nNow = GetTime()
  if nTime and tbData then
    self.nTime = nNow + nTime
    self.tbData = tbData
    self.tbData.nCarRefreshTime = nNow + tbData.nCarRefreshTime
  end
  local tbTime = {
    [1] = self.nTime - nNow,
    [2] = i18n.EasyGet("\231\172\172\228\184\128\233\152\182\230\174\181\228\186\137\229\164\186\229\137\169\228\189\153\239\188\154%s")
  }

  table.insert(tbContent, i18n.EasyGet("\232\181\132\230\186\144\229\174\136\229\141\171\229\137\169\228\189\153\239\188\154{0}", self.tbData.nBossNums))
  local szCarState = i18n.EasyGet("\231\137\169\232\181\132\232\189\166\239\188\154\229\183\178\229\136\183\230\150\176")
  if nNow < self.tbData.nCarRefreshTime then

  end

    szCarState = i18n.EasyGet("\231\137\169\232\181\132\232\189\166\239\188\154\229\183\178\231\187\147\230\157\159")

  table.insert(tbContent, i18n.EasyGet("\230\136\145\230\150\185\230\142\146\229\144\141\239\188\154{0}", self.tbData.nMyRank))

  FubenUIController:ShowDefaultInfo({tbTime}, tbContent)

function RaceMap:StartTimer()
  if 0 ~= self.nCheckTimer then
    Timer:Close(self.nCheckTimer)
  end
  self.nCheckTimer = Timer:Register(Env.GAME_FPS, self.OnTimer, self)
end

function RaceMap:OnTimer()
  self:CheckShowBelongPanel()
  if self.bInStart then
    self:UpdateRaceMapStartUI()
  end
  return true
end

function RaceMap:CheckShowBelongPanel()
  local tbNpcList = KNpc.GetNpcListInCurrentMap()
  local _, nMeX, nMeY = me.GetWorldPos()
  local bShow = false
  for _, pNpc in pairs(tbNpcList) do
    if pNpc.nTemplateId == ResourceBattle.Cfg.Car.nNpcId or pNpc.nTemplateId == ResourceBattle.Cfg.ResourcePoint.nNpcId then
      local _, nNpcX, nNpcY = pNpc.GetWorldPos()
      local nDiffX = math.abs(nNpcX - nMeX)
      local nDiffY = math.abs(nNpcY - nMeY)
      local nR = nDiffX * nDiffX + nDiffY * nDiffY
      if nR < ResourceBattle.Cfg.Car.nBelongRange * ResourceBattle.Cfg.Car.nBelongRange then
        if pNpc.nFightMode == Npc.FIGHT_MODE.emFightMode_None then
          if not Ui:WindowVisible("UIResourceResult") and self.tbBelongData[pNpc.nId] then
            Ui:OpenWindow("UIResourceResult", self.tbBelongData[pNpc.nId].nEndTime, self.tbBelongData[pNpc.nId].tbBelongData)
          elseif Ui:WindowVisible("UIResourceResult") then
            local tbWnd = Ui:GetWindow("UIResourceResult")
            tbWnd:Update(self.tbBelongData[pNpc.nId].nEndTime, self.tbBelongData[pNpc.nId].tbBelongData)
          end
          if Ui:WindowVisible("UIBloodBarBossTop") then
            BossHp:CloseBossHpView(pNpc.nId)
          end
          bShow = true
          break
        else
          if Ui:WindowVisible("UIResourceResult") then
            Ui:CloseWindow("UIResourceResult")
          end
          if not Ui:WindowVisible("UIBloodBarBossTop") then
            BossHp:OpenBossHpView(pNpc.nId)
          end
        end
      end
    end
  end
  if not bShow and Ui:WindowVisible("UIResourceResult") then
    Ui:CloseWindow("UIResourceResult")
  end
end

function RaceMap:OnLeave()
  if 0 ~= self.nCheckTimer then
    Timer:Close(self.nCheckTimer)
  end
  self.nCheckTimer = nil
  if Ui:WindowVisible("UIResourceResult") then
    Ui:CloseWindow("UIResourceResult")
  end
end

function RaceMap:OnGetBelongData(nNpcId, nEndTime, tbBelongData)
  nEndTime = nEndTime and nEndTime + GetTime()
  self.tbBelongData[nNpcId] = {nEndTime = nEndTime, tbBelongData = tbBelongData}
end

function RaceMap:IsNeedFuncScoreButton()
  return true, {
    self.TryGetMyRank,
    self
  }
end

function RaceMap:TryGetMyRank()
  local nNow = GetTime()
  if nNow < self.nLastRquestMyRankTime + 5 then
    self:ShowMyRank()
    return
  end
  self.nLastRquestMyRankTime = nNow
  RemoteServer.RequestZResourceBattle("OnC2ZGetMyRank")
end

function RaceMap:ShowMyRank()
  Ui:OpenWindow("UIResourceTips", self.tbMyRank)
end

function RaceMap:OnGetRank(tbMyRank)
  self.tbMyRank = tbMyRank
  self:ShowMyRank()
end

function RaceMap:OnLogout()
  self.tbBelongData = {}
  self.tbMyRank = {}
  self.nLastRquestMyRankTime = 0
  self.bInStart = nil
  self.nTime = nil
  self.tbData = nil
end
