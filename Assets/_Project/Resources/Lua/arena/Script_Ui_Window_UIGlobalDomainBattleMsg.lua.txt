local tbWnd = Ui:CreateClass("UIGlobalDomainBattleMsg", "UIPanel")
local tbControls = {
  btnCheck = "btnCheck",
  Template = "imgList/List/Template",
  txtTitle = "DomainListTitle/txtTong",
  txtCheck = "btnCheck/Text",
  txtInfo = "txtInfo2"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCheck = function(self)
    self.nWarZoneType = self.nWarZoneType % 2 + 1
    self:UpdateSV()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
  self.pPanel:Label_SetText(self.txtCheck, i18n.EasyGet("\229\136\135\230\141\162\230\136\152\229\140\186"))
  self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\229\134\155\229\155\162"))
end

function tbWnd:OnOpen()
  self.nWarZoneType = self.nWarZoneType or 1
  if not GlobalDomainBattle.tbRankList then
    RemoteServer.RequestGlobalDomainBattle("QueryBaseReq", GlobalDomainBattle.nBaseVersion)
  else
    self:UpdateSV()
  end
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSV()
  self.pPanel:Label_SetText(self.txtInfo, i18n.EasyGet("{0}\230\136\152\229\140\186\233\162\134\229\156\159\230\131\133\229\134\181\229\166\130\228\184\139", i18n.Get(GlobalDomainBattle.LEGION_SETTING.WARZONE_TYPE[self.nWarZoneType].szName)))
  local tbSVData = {}
  local tbTempData = {}
  local tbData = GlobalDomainBattle.tbRankList[self.nWarZoneType] or {}
  for nLegionId, tbV in pairs(tbData) do
    if nLegionId > 0 and tbV.nStarNum > 0 then
      if not tbTempData[nLegionId] then
        tbTempData[nLegionId] = {}
      end
      tbTempData[nLegionId].nStarNum = tbV.nStarNum or 0
      tbTempData[nLegionId].szLegionName = tbV.szLegionName
      tbTempData[nLegionId].szMasterName = tbV.szMasterName
      tbTempData[nLegionId].nRank = tbV.nRank
    end
  end
  for _, tbV in pairs(tbTempData) do
    table.insert(tbSVData, tbV)
  end
  table.sort(tbSVData, function(a, b)
    return a.nRank < b.nRank
  end)
  self.tbScrollView:Reset()
  
  local function fnShowItem(tbElement, nIdx, tbInfo)
    tbElement.pPanel:Label_SetText("txtRank", nIdx)
    tbElement.pPanel:Label_SetText("StarList/txtStar", tbInfo.nStarNum)
    tbElement.pPanel:Label_SetText("txtTong", tbInfo.szLegionName)
    tbElement.pPanel:Label_SetText("txtMasterName", tbInfo.szMasterName)
  end
  
  for i = 1, #tbSVData do
    self.tbScrollView:SetElement(i, function(tbElement)
      fnShowItem(tbElement, i, tbSVData[i])
    end)
  end
end
