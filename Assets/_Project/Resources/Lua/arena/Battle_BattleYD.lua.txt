BattleYD.szLimitTimeFrame = "OpenLevel69"
BattleYD.nLimitLevel = 60

function BattleYD:GetNextMonthOpenday()
  local tbNowTime = Lib:GetLocalDate()
  local nNextMonth = tbNowTime.month + 1
  local nNextYear = tbNowTime.year
  if nNextMonth > BattleYD.MONTH_PER_YEAR then
    nNextMonth = 1
    nNextYear = nNextYear + 1
  end
  local tbFirstDayParam = {
    year = nNextYear,
    month = nNextMonth,
    day = 1,
    hour = 4,
    min = 0,
    sec = 0
  }
  local nNextMonthFirstDay = os.time(tbFirstDayParam)
  local nDiffDay = 0
  if tonumber(os.date("%w", nNextMonthFirstDay)) <= BattleYD.nBattleYDTargetOpenDay then
    nDiffDay = BattleYD.nBattleYDTargetOpenDay - os.date("%w", nNextMonthFirstDay)
  else
    nDiffDay = BattleYD.DAY_PER_WEEK - os.date("%w", nNextMonthFirstDay) + BattleYD.nBattleYDTargetOpenDay
  end
  local tbFirstOpenDayParam = {
    year = nNextYear,
    month = nNextMonth,
    day = 1 + nDiffDay,
    hour = 4,
    min = 0,
    sec = 0
  }
  return Lib:GetTimeNum(os.time(tbFirstOpenDayParam))
end

function BattleSignUpYD:CheckPlayerQualification(pPlayer, nTime)
  if not pPlayer then
    return false
  end
  if 1 ~= GetTimeFrameState(BattleYD.szLimitTimeFrame) or pPlayer.nLevel < BattleYD.nLimitLevel then
    return false
  end
  local tbTime = nTime and os.date("*t", nTime) or Lib:GetLocalDate()
  local tbTodayParam = {
    year = tbTime.year,
    month = tbTime.month,
    day = tbTime.day,
    hour = tbTime.hour,
    min = tbTime.min,
    sec = tbTime.sec
  }
  local nCurTimeNum = Lib:GetTimeNum(os.time(tbTodayParam))
  local bHaveQualification = false
  for index, nKey in ipairs(Battle.tbQualificationKey) do
    local nRecordTime = pPlayer.GetUserValue(Battle.nSaveGroup, nKey)
    if nRecordTime == nCurTimeNum then
      bHaveQualification = true
    end
  end
  return bHaveQualification
end

function BattleSignUpYD:IsCalendarShow(nActivityTime)
  return BattleSignUpYD:CheckPlayerQualification(me, nActivityTime)
end
