Require("CommonScript/DomainBattle/DomainBattleCfg.lua")
DomainBattle.IS_SYS_OPEN = true
DomainBattle.GAME_STATE = {START = 1, CLOSE = 0}
DomainBattle.nGameState = DomainBattle.nGameState or DomainBattle.GAME_STATE.CLOSE
DomainBattle.tbClaimList = DomainBattle.tbClaimList or {}
DomainBattle.tbOccupyList = DomainBattle.tbOccupyList or {}
DomainBattle.tbLastOccupyList = DomainBattle.tbLastOccupyList or {}
DomainBattle.tbMonsterFlagList = DomainBattle.tbMonsterFlagList or {}
DomainBattle.nBaseVersion = DomainBattle.nBaseVersion or 0

function DomainBattle:LoadExternSetting()
  for _, tbSetting in pairs(self.MAP_FIGHT) do
    local tbPosList = Lib:LoadTabFileEx(tbSetting.POS_FILE, "sdd", nil, {
      "Type",
      "PosX",
      "PosY"
    })
    local tbGrpList = {
      FRONT = {}
    }
    for _, tbPos in ipairs(tbPosList) do
      local szFrontIdx = string.match(tbPos.Type, "FRONT_(%d)")
      if szFrontIdx then
        tbGrpList.FRONT[tonumber(szFrontIdx)] = {
          tbPos.PosX,
          tbPos.PosY
        }
      end
    end
    tbSetting.POS = tbGrpList
  end
end

DomainBattle:LoadExternSetting()
