local function FnSortByScore(tbA, tbB)
  return tbA.nScore > tbB.nScore
end

function TwoAreaBattle:ShowMyRankInTong()
  local tbData = {}
  for dwPlayerId, tbInfo in pairs(self.tbMyRank) do
    table.insert(tbData, tbInfo)
  end
  table.sort(tbData, FnSortByScore)
  for nIndex, tbInfo in ipairs(tbData) do
    if tbInfo.szName == me.szName then
      tbData.myIndex = nIndex
      tbData.myInfo = {}
      tbData.myInfo.szName = tbInfo.szName
      tbData.myInfo.nKill = tbInfo.nKill
      tbData.myInfo.nScore = tbInfo.nScore
      tbData.myInfo.nFaction = tbInfo.nFaction
    end
    Ui:OpenWindow("UITieFuChengResult", tbData)
  end
end

function RaceMap:TryGetMyRank()
  local nNow = GetTime()
  if nNow < TwoAreaBattle.nLastRquestMyRankTime + 5 then
    TwoAreaBattle:ShowMyRankInTong()
    return
  end
  TwoAreaBattle.nLastRquestMyRankTime = nNow
  RemoteServer.RequestZTwoAreaBattle("OnC2ZGetMyRank")
end

function RaceMap:UpdateRaceMapUI(nTime, tbRank, nMyScore, bEnd, bDouble)
  local tbTime = {
    [1] = nTime,
    [2] = bEnd and i18n.EasyGet("\230\180\187\229\138\168\231\187\147\230\157\159\239\188\140\232\175\183\231\166\187\229\188\128\230\156\172\229\156\176\229\155\190\239\188\154%s") or i18n.EasyGet("\230\180\187\229\138\168\229\128\146\232\174\161\230\151\182\239\188\154%s")
  }
  table.sort(tbRank, FnSortByScore)
  local tbContent = {}
  if bDouble then
    table.insert(tbContent, i18n.EasyGet("\229\189\147\229\137\141\233\152\182\230\174\181\229\143\175\232\142\183\229\190\151\229\143\140\229\128\141\231\137\169\232\181\132"))
  end
  local szTemp = i18n.EasyGet("\231\137\169\232\181\132\230\142\146\229\144\141")
  for nIndex, tbInfo in ipairs(tbRank) do
    szTemp = szTemp .. "\n" .. string.format("%d %s %d", nIndex, tbInfo.szTongName, tbInfo.nScore)
  end
  table.insert(tbContent, szTemp)
  local nNow = GetTime()
  if #self.tbFlagRefreshTime > 0 then
    table.insert(tbContent, i18n.EasyGet("\230\150\176\231\154\132\230\136\152\230\151\151\229\141\179\229\176\134\229\135\186\231\142\176:{0}\231\167\146", math.floor(self.tbFlagRefreshTime[1] - nNow)))
  end
  if #self.tbCarRefreshTime > 0 then
    table.insert(tbContent, i18n.EasyGet("\230\150\176\231\154\132\231\137\169\232\181\132\232\189\166\229\141\179\229\176\134\229\135\186\231\142\176:{0}\231\167\146", math.floor(self.tbCarRefreshTime[1] - nNow)))
  end
  table.insert(tbContent, i18n.EasyGet("\n\228\184\170\228\186\186\231\167\175\229\136\134\239\188\154{0}", nMyScore))
  FubenUIController:OpenDefaultFuBenUI("TwoAreaBattleHelp", {tbTime}, tbContent)
end

function RaceMap:OnGetScoreData(nTime, tbRank, nMyScore, bEnd, bDouble)
  self.nEndTime = GetTime() + nTime
  self.tbRank = tbRank
  self.nMyScore = nMyScore
  self.bDouble = bDouble
  self:UpdateRaceMapUI(nTime, tbRank, nMyScore, bEnd, bDouble)
end

function RaceMap:OnGetRefreshData(tbCarRefreshTime, tbFlagRefreshTime)
  table.sort(tbCarRefreshTime)
  table.sort(tbFlagRefreshTime)
  self.tbCarRefreshTime = tbCarRefreshTime
  self.tbFlagRefreshTime = tbFlagRefreshTime
  if not self.nRefreshTimer then
    self.nRefreshTimer = Timer:Register(Env.GAME_FPS, self.Refresh_Helper, self)
  end
end

function RaceMap:OnAddRefreshTime(szType, nTime)
  self[szType] = self[szType] or {}
  table.insert(self[szType], nTime)
  if not self.nRefreshTimer then
    self.nRefreshTimer = Timer:Register(Env.GAME_FPS, self.Refresh_Helper, self)
  end
end

function RaceMap:Refresh_Helper()
  local nNow = GetTime()
  for i = #self.tbCarRefreshTime, 1, -1 do
    if nNow > self.tbCarRefreshTime[i] then
      table.remove(self.tbCarRefreshTime, i)
    end
  end
  for i = #self.tbFlagRefreshTime, 1, -1 do
    if nNow > self.tbFlagRefreshTime[i] then
      table.remove(self.tbFlagRefreshTime, i)
    end
  end
  if #self.tbCarRefreshTime > 0 or #self.tbFlagRefreshTime > 0 then
    self:UpdateRaceMapUI(self.nEndTime - nNow, self.tbRank, self.nMyScore, self.bDouble)
    return true
  end
  self.nRefreshTimer = nil
  return false
end

function RaceMap:OnShowEndInfo(nTime, tbTongRank, tbMyRank, nMyScore)
  self.tbCarRefreshTime = {}
  self.tbCarRefreshTime = {}
  TwoAreaBattle.tbMyRank = tbMyRank
  self:OnGetScoreData(nTime, tbTongRank, nMyScore, true)
  BelongUI:EndCheckShow()
end

function RaceMap:OnGetFixPos(tbPoint)
  self.tbFixPoint = tbPoint
  self:ShowPos()
end

function RaceMap:OnGetDynamicPos(tbPoint)
  self.tbDynamicPoint = tbPoint
  self:ShowPos()
end

function RaceMap:ShowPos()
  local tbTemp1 = Lib:CopyTB(self.tbFixPoint)
  local tbTemp2 = Lib:CopyTB(self.tbDynamicPoint)
  local tbPoint = Lib:MergeTable(tbTemp1, tbTemp2)
  Map:SyncDynMapPoint(TwoAreaBattle.Cfg.nRaceMapTemplate, tbPoint, GetTime() + TwoAreaBattle.Cfg.nSyncCarPosTime + 1)
end

function RaceMap:StopAutoPath()
  AutoPath:StopGoto()
  if TeamMgr:IsFollowTeammate() then
    TeamMgr:StopFollowMember()
  end
end

function RaceMap:OnLogout()
  TwoAreaBattle.tbMyRank = {}
  TwoAreaBattle.nLastRquestMyRankTime = 0
  if self.nRefreshTimer then
    Timer:Close(self.nRefreshTimer)
    self.nRefreshTimer = nil
  end
end

EventNotify:UnRegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, RaceMap)
EventNotify:RegistNotify(EventNotify.emNOTIFY_RETURN_TO_LOGIN, RaceMap.OnLogout, RaceMap)
