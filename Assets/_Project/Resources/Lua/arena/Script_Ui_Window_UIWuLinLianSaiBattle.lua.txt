local tbWnd = Ui:CreateWindow("UIWuLinLianSaiBattle")
tbWnd.tbControls = {
  MyTemplate = "imgBG/BlueTeam/MyGroup/MyTemplate",
  EnemyTemplate = "imgBG/RedTeam/EnemyGroup/EnemyTemplate1",
  Victory = "imgBG/Victory",
  Defeat = "imgBG/Defeat",
  Draw = "imgBG/Draw",
  Ready = "imgBG/Ready",
  btnOk = "imgBG/btnOk",
  txtMyTeam = "imgBG/BlueTeam/Img_MyTeam/txtMyTeam",
  txtEnemyTeam = "imgBG/RedTeam/Img_EnemyTeam/txtEnemyTeam",
  txtEnemyOrder1 = "imgBG/RedTeam/MyEnemyTittle/Team1/TexMyNum",
  txtEnemyOrder2 = "imgBG/RedTeam/MyEnemyTittle/Team2/TexMyNum",
  txtEnemyOrder3 = "imgBG/RedTeam/MyEnemyTittle/Team3/TexMyNum"
}
tbWnd.tbOnClick = {
  btnOk = function(self)
    self:CloseView()
  end
}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_MULTIPVP_SYNCORDER,
      self.OnSyncOrder,
      self
    }
  }
end

function tbWnd:OnCreate()
  self.pSelfTeamScrollView = Ui:NewScrollView(self, self.MyTemplate, function(_, nIndex, tbElement)
  end, false, false)
  self.pEnemyTeamScrollView = Ui:NewScrollView(self, self.EnemyTemplate, function(_, nIndex, tbElement)
  end, false, false)
end

function tbWnd:OnOpen(tbTeamDataLeft, tbTeamDataRight, nState, nCloseTime, bCanChangeOrder)
  self.bCanChangeOrder = bCanChangeOrder
  self.nState = nState
  local bIsReady = nState == Ui.TeamBattleState.Ready
  local bIsWin = nState == Ui.TeamBattleState.Win
  local bIsLose = nState == Ui.TeamBattleState.Lose
  local bIsDraw = nState == Ui.TeamBattleState.Draw
  self.tbTeamDataLeftList = tbTeamDataLeft.tbTeamPlayerInfoDataList
  self.tbTeamDataRightList = tbTeamDataRight.tbTeamPlayerInfoDataList
  local nLeftCount = 0
  local nRightCount = 0
  for k, tbPlayerInfo in ipairs(tbTeamDataLeft.tbTeamPlayerInfoDataList) do
    if tbPlayerInfo.nPlayerId then
      nLeftCount = nLeftCount + 1
    end
  end
  for k, tbPlayerInfo in ipairs(tbTeamDataRight.tbTeamPlayerInfoDataList) do
    if tbPlayerInfo.nPlayerId then
      nRightCount = nRightCount + 1
    end
  end
  self.b1v1 = 1 == nLeftCount and 1 == nRightCount
  local Animator = self.pPanel:GetAnimator("Main")
  if bIsReady then
    Animator:Play("UIWuLinLianSaiBattle_InTo", 0, 0)
  else
    local nPoint1 = 0
    local nPoint2 = 0
    for _, tbPlayerInfo in ipairs(tbTeamDataLeft.tbTeamPlayerInfoDataList) do
      if tbPlayerInfo.bWin then
        nPoint1 = nPoint1 + 1
      end
    end
    for _, tbPlayerInfo in ipairs(tbTeamDataRight.tbTeamPlayerInfoDataList) do
      if tbPlayerInfo.bWin then
        nPoint2 = nPoint2 + 1
      end
    end
    local szPoint = string.format("%d:%d", nPoint1, nPoint2)
    if bIsWin then
      self.pPanel:Label_SetText(string.format("%s%s", self.Victory, "/VictoryEffect/imgBG/txtTime"), szPoint)
      Animator:Play("UIWuLinLianSaiBattle_VictoryEffect", 0, 0)
    elseif bIsLose then
      self.pPanel:Label_SetText(string.format("%s%s", self.Defeat, "/DefeatEffect/imgBG/txtTime"), szPoint)
      Animator:Play("UIWuLinLianSaiBattle_DefeatEffect", 0, 0)
    elseif bIsDraw then
      self.pPanel:Label_SetText(string.format("%s%s", self.Draw, "/DogfallEffect/imgBG/txtTime"), szPoint)
      Animator:Play("UIWuLinLianSaiBattle_DrawEffect", 0, 0)
    end
  end
  self.pPanel:SetActive(self.Ready, bIsReady)
  self.pPanel:SetActive(self.Victory, bIsWin)
  self.pPanel:SetActive(self.Defeat, bIsLose)
  self.pPanel:SetActive(self.Draw, bIsDraw)
  if bIsReady then
    self.pPanel:Label_SetText(self.txtEnemyOrder1, i18n.EasyGet("\230\156\170\231\159\165"))
    self.pPanel:Label_SetText(self.txtEnemyOrder2, i18n.EasyGet("\230\156\170\231\159\165"))
    self.pPanel:Label_SetText(self.txtEnemyOrder3, i18n.EasyGet("\230\156\170\231\159\165"))
  else
    self.pPanel:Label_SetText(self.txtEnemyOrder1, i18n.EasyGet("\231\172\172\228\184\128\228\189\141"))
    self.pPanel:Label_SetText(self.txtEnemyOrder2, i18n.EasyGet("\231\172\172\228\186\140\228\189\141"))
    self.pPanel:Label_SetText(self.txtEnemyOrder3, i18n.EasyGet("\231\172\172\228\184\137\228\189\141"))
  end
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
  end
  self.nCloseTime = nCloseTime
  if nCloseTime then
    if bIsReady then
      self.pPanel:Label_SetText(string.format("%s%s", self.Ready, "/txtTime"), i18n.EasyGet("\229\128\146\232\174\161\230\151\182\239\188\154{0}s", nCloseTime))
      self.pPanel:SetActive(self.btnOk, false)
    else
      self.pPanel:SetActive(self.btnOk, true)
      self.pPanel:Label_SetText(string.format("%s%s", self.btnOk, "/Text"), i18n.EasyGet("\229\133\179\233\151\173\239\188\136{0}s\239\188\137", nCloseTime))
    end
    self.nCloseTimer = Timer:Register(Env.GAME_FPS, function()
      return self:OnUpdateTimer()
    end)
  else
    self.pPanel:Label_SetText(string.format("%s%s", self.btnOk, "/Text"), i18n.EasyGet("\229\133\179\233\151\173"))
  end
  if tbTeamDataLeft.szTeamName and tbTeamDataLeft.szTeamName ~= "" then
    self.pPanel:SetActive(self.txtMyTeam, true)
    self.pPanel:Label_SetText(self.txtMyTeam, i18n.EasyGet("{0} \230\136\152\233\152\159", tbTeamDataLeft.szTeamName))
  else
    self.pPanel:SetActive(self.txtMyTeam, false)
  end
  if tbTeamDataRight.szTeamName and tbTeamDataRight.szTeamName ~= "" then
    self.pPanel:SetActive(self.txtEnemyTeam, true)
    self.pPanel:Label_SetText(self.txtEnemyTeam, i18n.EasyGet("{0} \230\136\152\233\152\159", tbTeamDataRight.szTeamName))
  else
    self.pPanel:SetActive(self.txtEnemyTeam, false)
  end
  self:InitSelfTeamScrollView(tbTeamDataLeft.tbTeamPlayerInfoDataList)
  self:InitEnemyTeamScrollView(tbTeamDataRight.tbTeamPlayerInfoDataList)
end

function tbWnd:OnDestroyed()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
  end
end

function tbWnd:OnUpdateTimer()
  self.nCloseTime = self.nCloseTime - 1
  if 0 == self.nCloseTime then
    self:CloseView()
    return false
  end
  self.pPanel:Label_SetText(string.format("%s%s", self.Ready, "/txtTime"), i18n.EasyGet("\229\128\146\232\174\161\230\151\182\239\188\154{0}s", self.nCloseTime))
  if self.nState ~= Ui.TeamBattleState.Ready and self.nState ~= Ui.TeamBattleState.ReadyNoTitle then
    self.pPanel:Label_SetText(string.format("%s%s", self.btnOk, "/Text"), i18n.EasyGet("\229\133\179\233\151\173\239\188\136{0}s\239\188\137", self.nCloseTime))
  end
  return true
end

function tbWnd:InitSelfTeamScrollView(tbTeamPlayerInfoDataList)
  self.pSelfTeamScrollView:Clear()
  for i, playerInfoData in ipairs(tbTeamPlayerInfoDataList) do
    self.pSelfTeamScrollView:SetElement(i, function(tbElement)
      self:SetPlayerInfo(tbElement, playerInfoData, i)
    end)
  end
end

function tbWnd:InitEnemyTeamScrollView(tbTeamPlayerInfoDataList)
  self.pEnemyTeamScrollView:Clear()
  for i, playerInfoData in ipairs(tbTeamPlayerInfoDataList) do
    self.pEnemyTeamScrollView:SetElement(i, function(tbElement)
      self:SetPlayerInfo(tbElement, playerInfoData)
    end)
  end
end

function tbWnd:SetPlayerInfo(tbElement, tbPlayerInfoData, nOrder)
  if not tbPlayerInfoData.nPlayerId then
    tbElement.pPanel:SetActive("Full", false)
    tbElement.pPanel:SetActive("Empty", true)
    tbElement.pPanel:SetActive("Img_Trophy", false)
    return
  end
  tbElement.pPanel:SetActive("Full", true)
  tbElement.pPanel:SetActive("Empty", false)
  tbElement.pPanel:SetActive("Img_Trophy", false)
  local tbPortraitObjPanel = Ui:NewExistPortraitObjPanel(tbElement, "Full/PortraitObj")
  tbPortraitObjPanel:SetRoleIcon(tbPlayerInfoData.nPortrait)
  tbPortraitObjPanel:SetPortraitFrame(tbPlayerInfoData.nPortraitFrame)
  tbPortraitObjPanel:SetFactionIcon(tbPlayerInfoData.nFaction)
  tbPortraitObjPanel:SetLevel(tbPlayerInfoData.nLevel)
  tbElement.pPanel:Label_SetText("Full/txtName", tbPlayerInfoData.szPlayerName)
  tbElement.pPanel:SetActive("Full/btnUp", false)
  tbElement.pPanel:SetActive("Full/btnDown", false)
  tbElement.pPanel:SetActive("Img_Trophy", tbPlayerInfoData.bWin == true)
  Ui:Button_BindEvent(tbElement.pPanel, "Full/btnUp", self.ChangeOrder, self, MultiPvP.UP, nOrder)
  Ui:Button_BindEvent(tbElement.pPanel, "Full/btnDown", self.ChangeOrder, self, MultiPvP.DOWN, nOrder)
  if self.nState == Ui.TeamBattleState.Ready or self.nState == Ui.TeamBattleState.ReadyNoTitle then
    tbElement.pPanel:SetActive("Full/Ready", true)
    tbElement.pPanel:SetActive("Full/Result", false)
    tbElement.pPanel:Label_SetText("Full/Ready/txtKin", i18n.EasyGet("\229\174\182\230\151\143 {0}", tbPlayerInfoData.szKinOrTongName))
    tbElement.pPanel:Label_SetText("Full/Ready/txtFightPower", i18n.EasyGet("\230\136\152\229\138\155 {0}", tbPlayerInfoData.nFight))
    if nOrder then
      tbElement.pPanel:SetActive("Full/btnUp", true)
      tbElement.pPanel:SetActive("Full/btnDown", true)
      tbElement.pPanel:Button_SetEnable("Full/btnUp", 1 ~= nOrder and not self.b1v1 and self.bCanChangeOrder ~= false)
      tbElement.pPanel:Button_SetEnable("Full/btnDown", 3 ~= nOrder and not self.b1v1 and self.bCanChangeOrder ~= false)
    end
  else
    tbElement.pPanel:SetActive("Full/Ready", false)
    tbElement.pPanel:SetActive("Full/Result", true)
    tbElement.pPanel:Label_SetText("Full/Result/txtKillCount", i18n.EasyGet("\229\135\187\232\180\165\230\149\176\239\188\154{0}", tbPlayerInfoData.nKill or 0))
    tbElement.pPanel:Label_SetText("Full/Result/txtDamage", i18n.EasyGet("\228\188\164\229\174\179\233\135\143\239\188\154{0}", tbPlayerInfoData.nDamage or 0))
  end
end

function tbWnd:ChangeOrder(nType, nOrder)
  MultiPvP:ApplyChangeOrder(nType, nOrder)
end

function tbWnd:OnSyncOrder(tbOrder)
  local tbTeamDataList = {}
  for nOrder, nPlayerId in ipairs(tbOrder) do
    if 0 == nPlayerId then
      tbTeamDataList[nOrder] = {}
    else
      for _, tbPlayerData in ipairs(self.tbTeamDataLeftList) do
        if tbPlayerData.nPlayerId == nPlayerId then
          tbTeamDataList[nOrder] = tbPlayerData
        end
      end
    end
  end
  self.tbTeamDataLeftList = tbTeamDataList
  self:InitSelfTeamScrollView(tbTeamDataList)
end

function tbWnd:CloseView()
  self:Close()
  Ui:DestroyWindow("UIWuLinLianSaiBattle")
end
