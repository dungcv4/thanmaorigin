local tbWnd = Ui:CreateWindow("UIFactionBattleKingServer")
local ITEM_GRID_COUNT = 4
tbWnd.tbControls = {
  btnClose = "imgBG/btnClose",
  NewKingTitle = "imgBG/Content/txtNewKingTitle",
  ScrollView = "imgBG/Content/ScrollView",
  ScrollViewTemplet = "imgBG/Content/ScrollView/List/TemplateDefault"
}
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

local function _GetInfo(info)
  if not info then
    return nil
  end
  local headIcon = PlayerPortrait:GetSprite(info.nPortrait)
  if info.nPlayerId <= 0 then
    headIcon = nil
  end
  return {
    faction = Faction:GetName(info.nFaction),
    kinName = info.szKinName,
    fightName = info.nFightPower,
    playerName = info.szName,
    spRoleHead = headIcon,
    spFaction = Faction:GetIcon(info.nFaction),
    level = info.nLevel,
    nSvrNameID = info.nSvrNameID
  }
end

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, self.ScrollViewTemplet)
end

function tbWnd:OnOpen(tbNewInfo)
  self.pPanel:SetActive(self.NewKingTitle, tbNewInfo and true or false)
  if not tbNewInfo then
    return
  end
  self.pPanel:Label_SetText(self.NewKingTitle, i18n.EasyGet("\232\183\168\230\156\141\233\151\168\230\180\190\231\171\158\230\138\128\229\134\160\229\134\155"))
  tbWnd:Update(tbNewInfo)
end

function tbWnd:Update(tbData)
  local tbOtherData = {}
  local tbMyFactionData = {}
  for i = 1, Faction.MAX_FACTION_COUNT do
    tbData[i] = tbData[i] or {
      nPlayerId = 0,
      nLevel = "",
      nFaction = i,
      nPortrait = 0,
      nHonorLevel = 0,
      nFightPower = "",
      szName = i18n.EasyGet("\232\153\154\228\189\141\228\187\165\229\190\133"),
      szKinName = ""
    }
    if i == me.nFaction then
      tbMyFactionData = Lib:CopyTB(tbData[i])
    elseif Faction:IsFactionOpen(tbData[i].nFaction) then
      table.insert(tbOtherData, Lib:CopyTB(tbData[i]))
    end
  end
  local myInfo = _GetInfo(tbMyFactionData)
  if myInfo then
    self.pPanel:SetActive("imgBG/Content/MyFaction", true)
    self.pPanel:Sprite_SetSprite("imgBG/Content/MyFaction/PortraitObj/imgFactionIcon", myInfo.spFaction)
    self.pPanel:Label_SetText("imgBG/Content/MyFaction/txtFightPower/txtFightPowerValue", myInfo.fightName)
    self.pPanel:Label_SetText("imgBG/Content/MyFaction/txtPlayerName", myInfo.playerName)
    self.pPanel:Label_SetText("imgBG/Content/MyFaction/PortraitObj/Level/Text", myInfo.level)
    self.pPanel:Label_SetText("imgBG/Content/MyFaction/txtServer/txtServerName", self:GetSvrName(myInfo.nSvrNameID))
    if myInfo.spRoleHead then
      self.pPanel:Sprite_SetSprite("imgBG/Content/MyFaction/PortraitObj/Portrait/Image", myInfo.spRoleHead)
      self.pPanel:SetActive("imgBG/Content/MyFaction/PortraitObj/Portrait/Image", true)
    else
      self.pPanel:SetActive("imgBG/Content/MyFaction/PortraitObj/Portrait/Image", false)
    end
  else
    self.pPanel:SetActive("imgBG/Content/MyFaction", false)
  end
  
  local function fnShowItem(pItem, nIdx)
    for i = 1, ITEM_GRID_COUNT do
      local nDataIdx = (nIdx - 1) * ITEM_GRID_COUNT + i
      local info = _GetInfo(tbOtherData[nDataIdx])
      local szItemKey = string.format("FactionOtherObj/FactionInfo%d", i)
      if info then
        pItem.pPanel:SetActive(szItemKey, true)
        pItem.pPanel:Label_SetText(szItemKey .. "/Faction/txtFaction", info.faction)
        pItem.pPanel:Sprite_SetSprite(szItemKey .. "/PortraitObj/imgFactionIcon", info.spFaction)
        pItem.pPanel:Label_SetText(szItemKey .. "/txtFightPowerValue", info.fightName)
        pItem.pPanel:Label_SetText(szItemKey .. "/txtPlayerName", info.playerName)
        pItem.pPanel:Label_SetText(szItemKey .. "/txtServerName", self:GetSvrName(info.nSvrNameID))
        pItem.pPanel:Label_SetText(szItemKey .. "/PortraitObj/Level/Text", info.level)
        if info.spRoleHead then
          pItem.pPanel:Sprite_SetSprite(szItemKey .. "/PortraitObj/Portrait/Image", info.spRoleHead)
          pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Portrait/Image", true)
        else
          pItem.pPanel:SetActive(szItemKey .. "/PortraitObj/Portrait/Image", false)
        end
      else
        pItem.pPanel:SetActive(szItemKey, false)
      end
    end
  end
  
  local nRow = math.ceil((Faction.MAX_FACTION_COUNT - 1) / ITEM_GRID_COUNT)
  self.tbScrollView:Reset()
  for i = 1, nRow do
    self.tbScrollView:SetElement(i, function(tbElement)
      fnShowItem(tbElement, i)
    end)
  end
end

function tbWnd:GetSvrName(nSvrNameID)
  if nSvrNameID then
    return i18n.EasyGet("[{0}\230\156\141]", nSvrNameID)
  else
    return i18n.EasyGet("\230\154\130\230\151\160")
  end
end
