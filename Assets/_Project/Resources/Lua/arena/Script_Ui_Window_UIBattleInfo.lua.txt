local tbWnd = Ui:CreateWindow("UIBattleInfo")
tbWnd.tbControls = {
  panelCampBattleScore = "imgBG/CampBattleScore",
  txtCamp1 = "imgBG/CampBattleScore/txtCamp1",
  txtCamp2 = "imgBG/CampBattleScore/txtCamp2",
  myMark1 = "imgBG/CampBattleScore/imgMyMark1",
  myMark2 = "imgBG/CampBattleScore/imgMyMark2",
  txtCampScore1 = "imgBG/CampBattleScore/scroeCell/txtCampScore1",
  txtCampScore2 = "imgBG/CampBattleScore/scroeCell2/txtCampScore2",
  panelFlagCell1 = "imgBG/CampBattleScore/scroeCell/flagCell",
  panelFlagCell2 = "imgBG/CampBattleScore/scroeCell2/flagCell2",
  txtFlagCount1 = "imgBG/CampBattleScore/scroeCell/flagCell/txtFlagCampScore1",
  txtFlagCount2 = "imgBG/CampBattleScore/scroeCell2/flagCell2/txtFlagCampScore2"
}
tbWnd.tbOnClick = {}

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_BATTLE_REPORT,
      self.OnSynData
    }
  }
end

function tbWnd:OnCreate()
  self.pPanel:SetActive(self.panelCampBattleScore, true)
end

function tbWnd:OnOpen()
end

function tbWnd:OnClose()
end

function tbWnd:SetInfo(nTeamIndex)
  self:SetTeamIndex(nTeamIndex)
  self.pPanel:SetActive(self.panelCampBattleScore, true)
  if Battle.tbBattleRank and Battle.tbTeamScore then
    self:OnSynData(Battle.tbBattleRank, Battle.nMyRank, Battle.tbTeamScore, nil)
  end
end

function tbWnd:SetTeamIndex(nTeamIndex)
  self.pPanel:SetActive(self.myMark1, 1 == nTeamIndex)
  self.pPanel:SetActive(self.myMark2, 1 ~= nTeamIndex)
end

function tbWnd:OnSynData(tbSelfData, nMyRank, tbTeamScore, nLeftTime, nTeamIndex, tbFlagCommitResult)
  self.pPanel:Label_SetText(self.txtCampScore1, tbTeamScore[1])
  self.pPanel:Label_SetText(self.txtCampScore2, tbTeamScore[2])
  if tbFlagCommitResult then
    self.pPanel:SetActive(self.panelFlagCell1, true)
    self.pPanel:SetActive(self.panelFlagCell2, true)
    self.pPanel:Label_SetText(self.txtFlagCount1, tbFlagCommitResult[1])
    self.pPanel:Label_SetText(self.txtFlagCount2, tbFlagCommitResult[2])
  else
    self.pPanel:SetActive(self.panelFlagCell1, false)
    self.pPanel:SetActive(self.panelFlagCell2, false)
  end
end
