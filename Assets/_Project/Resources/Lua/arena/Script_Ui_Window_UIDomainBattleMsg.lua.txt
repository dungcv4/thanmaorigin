local tbWnd = Ui:CreateClass("UIDomainBattleMsg", "UIPanel")
local tbControls = {
  btnCheck = "btnCheck",
  Template = "imgList/List/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnCheck = function(self)
    Ui:OpenWindow("UITong", "ACTIVITY")
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewScrollView(self, tbControls.Template)
end

function tbWnd:OnOpen()
  RemoteServer.RequestDomainBattle("GetOccupyListInfo")
end

function tbWnd:OnClose()
end

function tbWnd:UpdateSV(tbData)
  local tbSVData = {}
  local tbTempData = {}
  for _, tbV in ipairs(tbData) do
    if not tbTempData[tbV.nTongId] then
      tbTempData[tbV.nTongId] = {}
    end
    tbTempData[tbV.nTongId].nStar = tbTempData[tbV.nTongId].nStar or 0
    local nStar = DomainBattle.MAP_FIGHT[tbV.nMapTId].STAR_LV
    tbTempData[tbV.nTongId].nStar = tbTempData[tbV.nTongId].nStar + nStar
    tbTempData[tbV.nTongId].szTongName = tbV.szTongName
    tbTempData[tbV.nTongId].szMasterName = tbV.szMasterName
    tbTempData[tbV.nTongId].nRank = tbV.nRank
  end
  for _, tbV in pairs(tbTempData) do
    table.insert(tbSVData, tbV)
  end
  table.sort(tbSVData, function(a, b)
    return a.nRank < b.nRank
  end)
  self.tbScrollView:Reset()
  
  local function fnShowItem(tbElement, nIdx, tbInfo)
    tbElement.pPanel:Label_SetText("txtRank", nIdx)
    tbElement.pPanel:Label_SetText("StarList/txtStar", tbInfo.nStar)
    tbElement.pPanel:Label_SetText("txtTong", tbInfo.szTongName)
    tbElement.pPanel:Label_SetText("txtMasterName", tbInfo.szMasterName)
  end
  
  for i = 1, #tbSVData do
    self.tbScrollView:SetElement(i, function(tbElement)
      fnShowItem(tbElement, i, tbSVData[i])
    end)
  end
end
