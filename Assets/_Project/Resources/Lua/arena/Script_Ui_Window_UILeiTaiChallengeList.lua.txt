local tbWnd = Ui:CreateWindow("UILeiTaiChallengeList")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  objTemplate = "node/imgBG/Content/Scroll View/Viewport/Content/Template",
  objEmpty = "node/imgBG/Content/empty"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.challengeScrollView = Ui:NewScrollView(self, self.objTemplate, nil, true, true)
end

function tbWnd:OnOpen()
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.LeiTaiNewCandidate)
  self:RefreshChallengeList()
end

function tbWnd:OnClose()
end

function tbWnd:RefreshChallengeList()
  self.challengeScrollView:Clear()
  if not LeiTai.tbMyRoom or not LeiTai.tbMyRoom.tbCandidateMap then
    self.pPanel:SetActive(self.objEmpty, true)
    return
  end
  local tbCandidateIdList = {}
  for nPlayerId, tbInfo in pairs(LeiTai.tbMyRoom.tbCandidateMap) do
    table.insert(tbCandidateIdList, nPlayerId)
  end
  self.pPanel:SetActive(self.objEmpty, 0 == #tbCandidateIdList)
  for i, nPlayerId in ipairs(tbCandidateIdList) do
    self.challengeScrollView:SetElement(i, function(tbElement)
      local tbInfo = LeiTai.tbMyRoom.tbCandidateMap[nPlayerId]
      local szElementPath = "node/imgBG/Content/Scroll View/Viewport/Content/Element" .. i
      self.pPanel:Label_SetText(szElementPath .. "/txtMemberNumber", tbInfo.nMemberCount)
      self.pPanel:Label_SetText(szElementPath .. "/txtPlayerName", tbInfo.tbCandidateInfo.szName)
      self.pPanel:Label_SetText(szElementPath .. "/PortraitObj/Level/Text", tbInfo.tbCandidateInfo.nLevel)
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/imgFactionIcon", Faction:GetIcon(tbInfo.tbCandidateInfo.nFaction))
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbInfo.tbCandidateInfo.nPortrait))
      PlayerPortraitFrame:SetPortraitFrame(self.pPanel, szElementPath .. "/PortraitObj/PortraitFrame", tbInfo.tbCandidateInfo.nPortraitFrame)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/PortraitObj", self.PopUpMenu, self, tbInfo.tbCandidateInfo, tbElement)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/btnAccept", self.OnClickAccept, self, tbInfo.tbCandidateInfo.dwID)
    end, false)
  end
end

function tbWnd:PopUpMenu(tbPlayerInfo, tbElement)
  if tbPlayerInfo.dwID == me.dwID then
    return
  end
  local tbData = {
    dwRoleId = tbPlayerInfo.dwID,
    szName = tbPlayerInfo.szName,
    nPortrait = tbPlayerInfo.nPortrait,
    nFaction = tbPlayerInfo.nFaction,
    nLevel = tbPlayerInfo.nLevel,
    dwKinId = tbPlayerInfo.dwKinId,
    szKinOrTongName = tbPlayerInfo.szKinOrTongName
  }
  Ui:OpenWindow("UIRightPopUpMenu", "leiTai", tbData, tbElement, 420, -200)
end

function tbWnd:OnClickAccept(nPlayerId)
  LeiTai:AgreeJoinRoom(LeiTai.tbMyRoom.nRoomId, nPlayerId)
  self:Close()
end
