function TongBattle:LoadAwardFile()
  self.tbTongRankAward = Lib:EasyLoadTabFile("Setting/TongBattle/TongRankAward.tab")

  })
end

function TongBattle:LoadFinalsSchedule()
  self.tbFinalsSchedule = {}
  self.tbScheduleFinals = {}
  local tbFileData = Lib:EasyLoadTabFile("Setting/TongBattle/FinalsSchedule.tab")
  for _, tbInfo in pairs(tbFileData) do
    local tbMatch = {
      nWarZone = tbInfo.nWarZone,
      tbMatch = Lib:SplitStr2Num(tbInfo.szMatchS, ",")
    }
    self.tbFinalsSchedule[tbInfo.nDate] = self.tbFinalsSchedule[tbInfo.nDate] or {}
    self.tbFinalsSchedule[tbInfo.nDate][tbInfo.nTime] = self.tbFinalsSchedule[tbInfo.nDate][tbInfo.nTime] or {}
    table.insert(self.tbFinalsSchedule[tbInfo.nDate][tbInfo.nTime], tbMatch)
    self.tbScheduleFinals[tbInfo.nWarZone] = self.tbScheduleFinals[tbInfo.nWarZone] or {}
    for _, nMatchIndex in pairs(tbMatch.tbMatch) do
      self.tbScheduleFinals[tbInfo.nWarZone][nMatchIndex] = {
        nDate = tbInfo.nDate,
        nTime = tbInfo.nTime
      }
    end
  end
  self.tbFinalsSchedule = SetReadOnly(self.tbFinalsSchedule)
end

function TongBattle:GetSkillList(nHRSSubType)
  local tbSkillList = self.tbSkillList
  if not tbSkillList then
    tbSkillList = {}
    local szFilePath = "Setting/TongBattle/Skill.tab"
    local tbField = {
      "nNpcId",
      "nSkillId",
      "szDesc",
      "szIconAltlas",
      "nSortInUI",
      "CastType"
    }
    local tbFile = Lib:LoadTabFileEx(szFilePath, "ddssdd", nil, tbField, nil, nil, true)
    if not tbFile then
      LogErr("fail to load file:", szFilePath)
      return
    end
    for _, tbInfo in ipairs(tbFile) do
      tbSkillList[tbInfo.nNpcId] = tbSkillList[tbInfo.nNpcId] or {}
      table.insert(tbSkillList[tbInfo.nNpcId], {
        SkillId = tbInfo.nSkillId,
        SortInUI = tbInfo.nSortInUI,
        szDesc = tbInfo.szDesc,
        szIconAltlas = tbInfo.szIconAltlas,
        CastType = tbInfo.CastType
      })
    end
    for nNpcId, tbInfo in pairs(tbSkillList) do
      table.sort(tbInfo, function(a, b)
        return a.SortInUI < b.SortInUI
      end)
    end
    tbSkillList = SetReadOnly(tbSkillList)
    self.tbSkillList = tbSkillList
  end
  if nHRSSubType and tbSkillList then
    for nNpcId, tbData in pairs(self.SUPPLIES.SETTING) do
      if tbData.nHRSSubType == nHRSSubType then
        return tbSkillList[nNpcId]
      end
    end
  end
  return tbSkillList
end

function TongBattle:GetPlayerOneSkillInfoWithoutConfig(pPlayer, nSortInUI, nHRSSubType)
  local tbSkillList = self:GetSkillList(nHRSSubType)
  local tbInfo = tbSkillList[nSortInUI]
  if not tbInfo then
    LogErr("fail to get skill info by nSortInUI: ", nSortInUI, #tbSkillList)
    return
  end
  local nSkillId = tbInfo.SkillId
  local nSkillLevel = pPlayer.GetSkillLevel(nSkillId)
  local nSkillInfoLevel = nSkillLevel
  if not nSkillInfoLevel or nSkillInfoLevel <= 0 then
    nSkillInfoLevel = 1
  end
  local tbSkillInfo = FightSkill:GetSkillSetting(nSkillId, nSkillInfoLevel)
  local nMaxSkillLevel = 1
  local bMax = nSkillLevel >= nMaxSkillLevel
  local szCurMagicDesc, szNextMagicDesc, szClassName = FightSkill:FormatMagicDesc(nSkillId, nSkillLevel)
  local tbSkillResultInfo = {
    nId = nSkillId,
    nLevel = nSkillLevel,
    nMaxLevel = nMaxSkillLevel,
    bMax = bMax,
    szName = tbSkillInfo.SkillName or "",
    szDesc = tbSkillInfo.Desc or "",
    szProperty = tbSkillInfo.Property or "",
    nCD = tbSkillInfo.TimePerCast or 0,
    bPassive = tbSkillInfo.SkillType == FightSkill.SkillTypeDef.skill_type_passivity or tbSkillInfo.SkillType == FightSkill.SkillTypeDef.skill_type_inst_missile,
    nRadius = tbSkillInfo.AttackRadius or 0,
    nSort = tbInfo.SortInUI,
    szIconAltlas = tbInfo.szIconAltlas,
    szCurMagicDesc = szCurMagicDesc or "",
    szNextMagicDesc = szNextMagicDesc or "",
    szClassName = szClassName or "",
    bNotNextInfo = bMax
  }
  if tbSkillResultInfo.IsAura then
    tbSkillResultInfo.bPassive = true
  end
  return tbSkillResultInfo
end
