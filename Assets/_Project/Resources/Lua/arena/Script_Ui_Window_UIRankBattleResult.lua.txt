local tbWnd = Ui:CreateWindow("UIRankBattleResult")
tbWnd.tbControls = {
  btnOk = "imgBG/btnOk",
  VictoryEffect = "imgBG/Victory",
  FailureEffect = "imgBG/Defeat",
  MyGroup = "imgBG/MyGroup",
  EnemyGroup = "imgBG/EnemyGroup",
  MyPortrait = "imgBG/MyGroup/My1/PortraitObj/Portrait/Image",
  MyLevel = "imgBG/MyGroup/My1/PortraitObj/Level/Text",
  MyFaction = "imgBG/MyGroup/My1/PortraitObj/imgFactionIcon",
  MyDamage = "imgBG/MyGroup/My1/Group/Damage/Damage",
  MyDamageNum = "imgBG/MyGroup/My1/Group/DamageNum",
  MyName = "imgBG/MyGroup/My1/txtName",
  EnemyPortrait = "imgBG/EnemyGroup/Enemy1/PortraitObj/Portrait/Image",
  EnemyLevel = "imgBG/EnemyGroup/Enemy1/PortraitObj/Level/Text",
  EnemyFaction = "imgBG/EnemyGroup/Enemy1/PortraitObj/imgFactionIcon",
  EnemyDamage = "imgBG/EnemyGroup/Enemy1/Group/Damage/Damage",
  EnemyDamageNum = "imgBG/EnemyGroup/Enemy1/Group/DamageNum",
  EnemyName = "imgBG/EnemyGroup/Enemy1/txtName"
}

function tbWnd:CloseByDefault()
  self:Close()
end

tbWnd.tbOnClick = {
  btnOk = function(self)
    self:CloseByDefault()
  end
}

function tbWnd:OnOpen(nResult, varMe, varEnemy, tbDamageCounter, tbCallback, szBattleClassType)
  self.tbOKCallback = tbCallback
  if 1 == nResult then
    self.pPanel:SetActive(self.VictoryEffect, true)
    self.pPanel:SetActive(self.FailureEffect, false)
  else
    self.pPanel:SetActive(self.VictoryEffect, false)
    self.pPanel:SetActive(self.FailureEffect, true)
  end
  self.tbDamageCounter = tbDamageCounter
  self.nMaxDamage = 0
  for _, nDamage in pairs(tbDamageCounter) do
    if nDamage > self.nMaxDamage then
      self.nMaxDamage = nDamage
    end
  end
  if type(varMe) == "table" then
    self:SetNpcShow(varMe, 1)
  end
  self.pPanel:SetActive(self.EnemyGroup, true)
  if type(varEnemy) == "table" then
    self:SetNpcShow(varEnemy, 2)
  end
end

function tbWnd:SetNpcShow(tbEnemy, nCamp)
  local tbCtrls = {
    [1] = {
      Portrait = self.MyPortrait,
      Level = self.MyLevel,
      Faction = self.MyFaction,
      Damage = self.MyDamage,
      DamageNum = self.MyDamageNum,
      Name = self.MyName
    },
    [2] = {
      Portrait = self.EnemyPortrait,
      Level = self.EnemyLevel,
      Faction = self.EnemyFaction,
      Damage = self.EnemyDamage,
      DamageNum = self.EnemyDamageNum,
      Name = self.EnemyName
    }
  }
  self.pPanel:SetActive(self.MyDamageNum, true)
  self.pPanel:SetActive(self.EnemyDamageNum, true)
  local szName, portrait, nFaction, nLevel = table.unpack(tbEnemy[1])
  self.pPanel:Label_SetText(tbCtrls[nCamp].Name, szName)
  self.pPanel:Label_SetText(tbCtrls[nCamp].Level, tostring(nLevel))
  if nFaction then
    self.pPanel:SetActive(tbCtrls[nCamp].Faction, true)
    self.pPanel:Sprite_SetSprite(tbCtrls[nCamp].Faction, Faction:GetIcon(nFaction))
  else
    self.pPanel:SetActive(tbCtrls[nCamp].Faction, false)
  end
  if type(portrait) == "number" then
    self.pPanel:Sprite_SetSprite(tbCtrls[nCamp].Portrait, PlayerPortrait:GetIcon(portrait))
  elseif "table" == type(portrait) then
    local nFaceId = KNpc.GetNpcShowInfo(table.unpack(portrait))
    self.pPanel:Sprite_SetSprite(tbCtrls[nCamp].Portrait, Npc:GetFace(nFaceId))
  end
  self.pPanel:Sprite_SetFill(tbCtrls[nCamp].Damage, (self.tbDamageCounter[nCamp * 10 + 1] or 0) / self.nMaxDamage)
  self.pPanel:Label_SetText(tbCtrls[nCamp].DamageNum, self.tbDamageCounter[nCamp * 10 + 1] or 0)
end
