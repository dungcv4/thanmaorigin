LeiTai.FuBenMgrC = LeiTai.FuBenMgrC or {}
local Mgr = LeiTai.FuBenMgrC

function Mgr:RegisterFubenUIChecker()
  FubenUIController:RegisterChecker({
    LeiTai.Cfg.MAP_TEMPLATE_ID
  }, self)
end

function Mgr:IsNeedExitFubenButton()
  if 1 == me.GetUserValue(LeiTai.SAVE_GROUP, LeiTai.SAVE_WATCH) then
    return true
  end
  return false
end

function Mgr:OnFubenMapLoaded(pPlayer)
  FubenUIController:OpenDefaultFuBenUI()
end

function Mgr:OnLeave()
  FubenUIController:CloseDefaultFubenUI()
end

function Mgr:IsInFubenMap()
  return true
end

function Mgr:UpdateFuBenUIInfo(nFightEndTime, nLookerCount)
  local bIsOver = nFightEndTime < GetTime()
  if not bIsOver then
    if 1 == me.GetUserValue(LeiTai.SAVE_GROUP, LeiTai.SAVE_WATCH) then
      FubenUIController:ShowDefaultInfo({
        {
          nFightEndTime - GetTime(),
          i18n.EasyGet("\230\136\152\230\150\151\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154%s")
        }
      }, {
        i18n.EasyGet("\229\189\147\229\137\141\232\167\130\230\136\152\228\186\186\230\149\176\239\188\154{0}", nLookerCount)
      })
    else
      FubenUIController:ShowDefaultInfo({
        {
          nFightEndTime - GetTime(),
          i18n.EasyGet("\230\136\152\230\150\151\229\137\169\228\189\153\230\151\182\233\151\180\239\188\154%s")
        }
      }, {
        i18n.EasyGet("\232\131\156\229\136\169\230\157\161\228\187\182\239\188\154\229\135\187\232\180\165\230\149\140\230\150\185\229\133\168\233\131\168\231\155\174\230\160\135")
      })
    end
  end
end

Mgr:RegisterFubenUIChecker()
