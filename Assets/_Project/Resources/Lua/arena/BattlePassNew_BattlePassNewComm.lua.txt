function BattlePassNew:LoadSetting()
  self.tbBattlePassLvCfg = Lib:LoadTabFileEx("Setting/Activity/BattlePassNew/BattlePassExp.tab", "ddddddd", "nLevel", {
    "nLevel",
    "nExp",
    "nAwardKeyN",
    "nAwardKeyM",
    "nAwardKeyH",
    "nShow",
    "nHard"
  })
  self.tbBattlePassTypeCfg = Lib:LoadTabFileEx("Setting/Activity/BattlePassNew/BattlePassType.tab", "sssssddd", "szTaskType", {
    "szTaskType",
    "szTaskDesc",
    "szTaskName",
    "szTimeFrameEvent",
    "szTimeFrameEnd",
    "nStartSeason",
    "nEndSeason",
    "nIsClose"
  })
  self.tbBattlePassTaskCfg = Lib:EasyLoadTabFile("Setting/Activity/BattlePassNew/BattlePassTask.tab", {"nTaskId", "nTag"}, {
    "nTaskId",
    "nTag",
    "szTaskType",
    "nFinishCount",
    "nTaskClass",
    "nAddExp",
    "nCostGold",
    "nExtraParam",
    "szGuideFunc",
    "szGuideParam",
    "nNotExtraConcat"
  })
  self.tbBattlePassBuyAwardCfg = Lib:LoadTabFileEx("Setting/Activity/BattlePassNew/BattlePassBuyAward.tab", "ddd", "nProductId", {
    "nProductId",
    "nAwardId",
    "nAwardKey"
  })
  self.tbPreviewbBattlePassAward = {}
  for nLevel, battlePassLvCfgData in ipairs(self.tbBattlePassLvCfg) do
    if 1 == battlePassLvCfgData.nShow then
      table.insert(self.tbPreviewbBattlePassAward, battlePassLvCfgData)
    end
  end
  
  local function SortFunc(battlePassLvCfgDataA, battlePassLvCfgDataB)
    local nLvA = battlePassLvCfgDataA.nLevel
    local nLvB = battlePassLvCfgDataB.nLevel
    return nLvA <= nLvB
  end
  
  table.sort(self.tbPreviewbBattlePassAward, SortFunc)
  self.tbBattlePassLvNeedTotalExp = {}
  local nTotalExp = 0
  for nLevel, battlePassLvCfgData in ipairs(self.tbBattlePassLvCfg) do
    self.tbBattlePassLvNeedTotalExp[nLevel] = nTotalExp + battlePassLvCfgData.nExp
    nTotalExp = self.tbBattlePassLvNeedTotalExp[nLevel]
  end
  Validator:Register(self.CheckBuyAwardCfg, self)
end

function BattlePassNew:CheckBuyAwardCfg()
  for nProductId, _ in pairs(self.tbBattlePassBuyAwardCfg) do
    if not Recharge:GetProductSettingById(nProductId) then
      LogErr(string.format("load BattlePassBuyAward.tab failed, nProductId %d not found in RechargeProducts.tab", nProductId))
      return false
    end
  end
  return true
end

function BattlePassNew:IsOpen(pPlayer)
  if pPlayer.nLevel < self.nMinLevel then
    return
  end
end

function BattlePassNew:GetSeason()
  local nTimestamp = GetTime() - self.REF_TIME
  local nTimeNum = Lib:GetTimeNum(nTimestamp)
  return math.floor(nTimeNum / 100)
end

function BattlePassNew:GenInitTaskData()
  return {nCurCount = 0, bIsCompleted = false}
end

function BattlePassNew:CheckBattlePassTaskTypeIsValid(nTaskId, nTag, nServerOpenDay)
  local battlePassTaskCfgData = self:GetBattlePassTaskCfgData(nTaskId, nTag)
  if nil == battlePassTaskCfgData then
    return false
  end
  local szTaskType = battlePassTaskCfgData.szTaskType
  local battlePassTypeCfgData = self.tbBattlePassTypeCfg[szTaskType]
  if nil == battlePassTypeCfgData then
    return false
  end
  if 1 == battlePassTypeCfgData.nIsClose then
    return false
  end
  local nSeason = math.ceil(nServerOpenDay / BattlePassNew.RestCycle)
  if nSeason < battlePassTypeCfgData.nStartSeason or nSeason > battlePassTypeCfgData.nEndSeason then
    return false
  end
  return true
end

function BattlePassNew:GetTaskAndTagMapIndex(nTaskId, nTag)
  if self.tbBattlePassServerData ~= nil and nil ~= self.tbBattlePassServerData.tbTaskAndTagMap and nil ~= self.tbBattlePassServerData.tbTaskAndTagMap[nTaskId] and nil ~= self.tbBattlePassServerData.tbTaskAndTagMap[nTaskId][nTag] then
    return self.tbBattlePassServerData.tbTaskAndTagMap[nTaskId][nTag]
  else
    return nil
  end
end

function BattlePassNew:GetTaskMapIndex(nTaskId)
  if self.tbBattlePassServerData ~= nil and nil ~= self.tbBattlePassServerData.tbTaskMap and nil ~= self.tbBattlePassServerData.tbTaskMap[nTaskId] then
    return self.tbBattlePassServerData.tbTaskMap[nTaskId]
  else
    return nil
  end
end

function BattlePassNew:GetTaskIsCompleted(pPlayer, nTaskId, nTag)
  if nil == pPlayer then
    return nil
  end
  local nCompletedIndex = self:GetTaskAndTagMapIndex(nTaskId, nTag)
  if nil ~= nCompletedIndex then
    local nGroupIndex = nCompletedIndex / BattlePassNew.MaxBit + 1
    local nBitIndex = nCompletedIndex % BattlePassNew.MaxBit
    if 0 == nBitIndex then
      nBitIndex = BattlePassNew.MaxBit
      nGroupIndex = nGroupIndex - 1
    end
    local nTaskCompletedValue = pPlayer.GetUserValue(BattlePassNew.TaskCompletedGroupKey, nGroupIndex)
    local bIsCompleted = Lib:GetBit(nTaskCompletedValue, nBitIndex)
    return bIsCompleted
  else
    return false
  end
end

function BattlePassNew:GetTaskCount(pPlayer, nTaskId)
  if nil == pPlayer then
    return 0
  end
  local nCountIndex = self:GetTaskMapIndex(nTaskId)
  if nil ~= nCountIndex then
    local nCurCount = pPlayer.GetUserValue(BattlePassNew.TaskCountGroupKey, nCountIndex)
    return nCurCount
  else
    LogErr("[BattlePassNew] nCountIndex is nill")
    return 0
  end
end

function BattlePassNew:GetAwardIsGet(pPlayer, nLv, nAwardTag)
  if nil == pPlayer then
    return nil
  end
  local nAwardGroupKey = self:GetAwardGroupKey(nAwardTag)
  local nGroupIndex = nLv / BattlePassNew.MaxBit + 1
  local nBitIndex = nLv % BattlePassNew.MaxBit
  if 0 == nBitIndex then
    nBitIndex = BattlePassNew.MaxBit
    nGroupIndex = nGroupIndex - 1
  end
  local nAwardGetValue = pPlayer.GetUserValue(nAwardGroupKey, nGroupIndex)
  local bIsGet = Lib:GetBit(nAwardGetValue, nBitIndex)
  return bIsGet
end

function BattlePassNew:GetAwardGroupKey(nAwardTag)
  if nAwardTag == BattlePassNew.AwardTag.Normal then
    return BattlePassNew.NormalAwardGroupKey
  elseif nAwardTag == BattlePassNew.AwardTag.Elite then
    return BattlePassNew.EliteAwardGroupKey
  elseif nAwardTag == BattlePassNew.AwardTag.Legend then
    return BattlePassNew.LegendAwardGroupKey
  else
    return BattlePassNew.NormalAwardGroupKey
  end
end
