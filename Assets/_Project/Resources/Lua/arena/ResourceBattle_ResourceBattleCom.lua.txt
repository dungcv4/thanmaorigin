function ResourceBattle:CheckIsOpen(nActivityTime)
  if Legion:GetState() ~= Legion.Def.State.Play then
    return false
  end
  local tbTime = Lib:GetLocalDate(nActivityTime)
  if 7 ~= tbTime.wday then
    return false
  end
  local bFirstSaturday = tbTime.day <= 7
  if bFirstSaturday and 1 == tbTime.month % 2 then
    return false
  end
  return true
end

function ResourceBattle:CheckIsNotOpen(nActivityTime)
  return not self:CheckIsOpen(nActivityTime)
end

function ResourceBattle:IsGamePhase()
  return self.nGameState ~= self.Def.GAME_STATE.CLOSE
end
