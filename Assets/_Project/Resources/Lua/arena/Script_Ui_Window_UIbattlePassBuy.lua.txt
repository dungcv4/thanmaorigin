local tbWnd = Ui:CreateWindow("UIbattlePassBuy")
tbWnd.tbControls = {
  btnBuyElite = "content/battlePassElite/btnBuy",
  btnBuyLegend = "content/battlePassLegend/btnBuy",
  btnClose = "content/btnClose",
  EliteAwardItemTemplete = "content/battlePassElite/AwardList/Viewport/Content/Item",
  LegendAwardItemTemplete = "content/battlePassLegend/AwardList/Viewport/Content/Item"
}
tbWnd.tbOnClick = {
  btnBuyElite = function(self)
    BattlePass:BuyBattlePass(BattlePass.AwardTag.Elite)
    self.pPanel:Button_SetEnable(self.btnBuyElite, false)
    self.pPanel:Button_SetEnable(self.btnBuyLegend, false)
  end,
  btnBuyLegend = function(self)
    BattlePass:BuyBattlePass(BattlePass.AwardTag.Legend)
    self.pPanel:Button_SetEnable(self.btnBuyElite, false)
    self.pPanel:Button_SetEnable(self.btnBuyLegend, false)
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.pEliteAwardScorllView = Ui:NewScrollView(self, self.EliteAwardItemTemplete, function(_, nIndex, tbElement)
  end, false, false)
  self.pLegendAwardScorllView = Ui:NewScrollView(self, self.LegendAwardItemTemplete, function(_, nIndex, tbElement)
  end, false, false)
end

function tbWnd:OnOpen()
  self:UpdateBuyState()
  self:ShowAwardList()
end

function tbWnd:RefreshView()
  self:UpdateBuyState()
  self:ShowAwardList()
end

function tbWnd:UpdateBuyState()
  local nBattlePassPlayerType = BattlePass:GetBattlePassPlayerType()
  local bIsUnlockElite = nBattlePassPlayerType >= BattlePass.AwardTag.Elite
  local bIsUnlockLegend = nBattlePassPlayerType >= BattlePass.AwardTag.Legend
  if bIsUnlockElite then
    self.pPanel:Button_SetText(self.btnBuyElite, i18n.EasyGet("\229\183\178\232\180\173\228\185\176"), "Text")
    self.pPanel:Button_SetEnable(self.btnBuyElite, false)
  else
    local tbProductSetting = Recharge:GetProductSettingById(BattlePass.Recharge.Elite)
    local szShowPrice = Recharge:GetShowPrice(tbProductSetting)
    self.pPanel:Button_SetText(self.btnBuyElite, szShowPrice, "Text")
    self.pPanel:Button_SetEnable(self.btnBuyElite, true)
  end
  if bIsUnlockLegend then
    self.pPanel:Button_SetText(self.btnBuyLegend, i18n.EasyGet("\229\183\178\232\180\173\228\185\176"), "Text")
    self.pPanel:Button_SetEnable(self.btnBuyLegend, false)
  else
    local tbProductSetting
    if nBattlePassPlayerType == BattlePass.AwardTag.Normal then
      tbProductSetting = Recharge:GetProductSettingById(BattlePass.Recharge.Legend)
    elseif nBattlePassPlayerType == BattlePass.AwardTag.Elite then
      tbProductSetting = Recharge:GetProductSettingById(BattlePass.Recharge.PriceDifference)
    end
    local szShowPrice = Recharge:GetShowPrice(tbProductSetting)
    self.pPanel:Button_SetText(self.btnBuyLegend, szShowPrice, "Text")
    self.pPanel:Button_SetEnable(self.btnBuyLegend, true)
  end
end

function tbWnd:ShowAwardList()
  self:InitEliteAwardScorllView()
  self:InitLegendAwardScorllView()
end

function tbWnd:InitEliteAwardScorllView()
  local function awardFunc(tbAwardElement, awardData)
    local tbItemGrid = tbAwardElement.tbItemGrid
    
    tbItemGrid = tbItemGrid or Ui:NewExistItemGrid(tbAwardElement, "UIItemGrid")
    tbItemGrid:SetGenericItem(awardData)
    tbItemGrid:SetClickEnable(true)
  end
  
  self.pEliteAwardScorllView:Clear()
  local battlePassBuyAwardCfgData = BattlePass:GetBattlePassBuyAwardCfgData(BattlePass.Recharge.Elite)
  local tbAwards = Award:GetAward(battlePassBuyAwardCfgData.nAwardId, battlePassBuyAwardCfgData.nAwardKey, {})
  for i, awardData in pairs(tbAwards) do
    self.pEliteAwardScorllView:SetElement(i, function(tbAwardElement)
      awardFunc(tbAwardElement, awardData)
    end)
  end
end

function tbWnd:InitLegendAwardScorllView()
  local function awardFunc(tbAwardElement, awardData)
    local tbItemGrid = tbAwardElement.tbItemGrid
    
    tbItemGrid = tbItemGrid or Ui:NewExistItemGrid(tbAwardElement, "UIItemGrid")
    tbItemGrid:SetGenericItem(awardData)
    tbItemGrid:SetClickEnable(true)
  end
  
  self.pLegendAwardScorllView:Clear()
  local nBattlePassPlayerType = BattlePass:GetBattlePassPlayerType()
  local battlePassBuyAwardCfgData
  if nBattlePassPlayerType ~= BattlePass.AwardTag.Elite then
    battlePassBuyAwardCfgData = BattlePass:GetBattlePassBuyAwardCfgData(BattlePass.Recharge.Legend)
  else
    battlePassBuyAwardCfgData = BattlePass:GetBattlePassBuyAwardCfgData(BattlePass.Recharge.PriceDifference)
  end
  local tbAwards = Award:GetAward(battlePassBuyAwardCfgData.nAwardId, battlePassBuyAwardCfgData.nAwardKey, {})
  for i, awardData in pairs(tbAwards) do
    self.pLegendAwardScorllView:SetElement(i, function(tbAwardElement)
      awardFunc(tbAwardElement, awardData)
    end)
  end
end

function tbWnd:OnClose()
  if not Ui:WindowVisible("UIBattlePass") then
    Ui:OpenWindow("UIBattlePass")
  end
end
