function tbWnd:Update()
  tbWnd:hiddle()
  if FactionBattle:IsFactionBattleGlobal() then
    self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\232\183\168\230\156\141\233\151\168\230\180\190\231\171\158\230\138\128"))
  else
    self.pPanel:Label_SetText(self.txtTitle, i18n.EasyGet("\233\151\168\230\180\190\231\171\158\230\138\128"))
  end
  for i = 1, 31 do
    local ctrlName = string.format(szLabel_Name, i)
    local name, nWinCount = self:GetName(i)
    if tostring(name) == me.szName then
      local colorName = string.format("<color=#0A6F32>%s</color>", name)
      self.pPanel:Label_SetText(self[ctrlName], colorName)
    else
      self.pPanel:Label_SetText(self[ctrlName], name)
    end
    if 4 == nWinCount then
      for _, v in pairs(FactionBattle.tb16thPlayer) do
        if name == v.szName then
          local szHead = PlayerPortrait:GetIcon(v.nFace)
          if szHead then
            self.pPanel:Sprite_SetSprite(self.PortraitPath .. "/Portrait/Image", szHead)
            self.pPanel:SetActive(self.PortraitPath .. "/Portrait/Image", true)
            self.pPanel:SetActive(self.PortraitPath .. "/Portrait", true)
          end
          break
        end
      end
    end
  end
end

function tbWnd:GetName(idx)
  if idx > 0 and idx <= #idListAtIdx then
    local minWin = self:GetMinWin(idx)
    local idList = idListAtIdx[idx]
    for _, id in ipairs(idList) do
      local info = FactionBattle.tb16thPlayer[id]
      if info and minWin <= info.nWinCount then
        return info.szName, info.nWinCount
      end
    end
  end
  return "", 0
end

function tbWnd:GetMinWin(idx)
  if idx <= 16 then
    return 0
  elseif idx <= 24 then
    return 1
  elseif idx <= 28 then
    return 2
  elseif idx <= 30 then
    return 3
  else
    return 4
  end
  return 0
end

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_FACTION_TOP_CHANGE,
      self.Update
    }
  }
end

function tbWnd:hiddle()
  self.pPanel:SetActive(self.btnGuess, false)
  self.pPanel:SetActive(self.txtGuessFirst, false)
  self.pPanel:SetActive(self.txtGuessSecond, false)
  self.pPanel:SetActive(self.txtGuessTime, false)
end
