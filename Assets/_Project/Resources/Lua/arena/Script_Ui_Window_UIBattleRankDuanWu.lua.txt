local tbWnd = Ui:CreateWindow("UIBattleRankDuanWu")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  RankTemplate = "node/imgBG/Content/imgList/Viewport/List/Template",
  txtRank = "node/imgBG/Content/myInfo/txtRank",
  txtName = "node/imgBG/Content/myInfo/txtName",
  txtTime = "node/imgBG/Content/myInfo/txtTime",
  txtScore = "node/imgBG/Content/myInfo/txtScore"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbScrollView = Ui:NewLoopScrollView(self, tbControls.RankTemplate)
end

function tbWnd:OnOpen(tbMatchResult)
  self.tbScrollView:Clear()
  for nRank = 1, #tbMatchResult.tbMatchResultPlayerData do
    self.tbScrollView:SetElement(nRank, function(tbLoopScroll, nInnerRank)
      local tbElement = tbLoopScroll:GetElement(nInnerRank)
      local pPanel = tbElement.pPanel
      pPanel:Label_SetText("txtRank", nInnerRank)
      pPanel:Label_SetText("txtName", tbMatchResult.tbMatchResultPlayerData[nInnerRank].szName)
      local nFinishTime = tbMatchResult.tbMatchResultPlayerData[nInnerRank].nFinishTime
      if nil == nFinishTime then
        nFinishTime = DragonBoatRace.Cfg.nNotFinishRecordTime
      end
      local nHour, nMinute, nSecond = Lib:TransferSecond2NormalTime(nFinishTime)
      pPanel:Label_SetText("txtTime", string.format("%02d:%02d:%02d", nHour, nMinute, nSecond))
      pPanel:Label_SetText("txtScore", tbMatchResult.tbMatchResultPlayerData[nInnerRank].nScore)
    end)
  end
  local tbMyPlayerData = tbMatchResult.tbMatchResultPlayerData[tbMatchResult.nMyRank]
  self.pPanel:Label_SetText(tbControls.txtRank, tbMatchResult.nMyRank)
  self.pPanel:Label_SetText(tbControls.txtName, tbMyPlayerData.szName)
  local nFinishTime = tbMyPlayerData.nFinishTime
  if nil == nFinishTime then
    nFinishTime = DragonBoatRace.Cfg.nNotFinishRecordTime
  end
  local nHour, nMinute, nSecond = Lib:TransferSecond2NormalTime(nFinishTime)
  self.pPanel:Label_SetText(tbControls.txtTime, string.format("%02d:%02d:%02d", nHour, nMinute, nSecond))
  self.pPanel:Label_SetText(tbControls.txtScore, tbMyPlayerData.nScore)
end
