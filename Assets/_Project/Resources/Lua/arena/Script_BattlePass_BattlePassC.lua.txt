function BattlePass:GetRemainingTime()
  local nCreateRealTime = GetServerCreateTime() - 14400
  nCreateRealTime = nCreateRealTime - Lib:GetLocalDayTime(nCreateRealTime)
  local nSeason = self.battlePassPlayerData.nSeason
  local nFinishTime = nCreateRealTime + nSeason * BattlePass.RestCycle * 3600 * 24
  local nCurTime = GetTime() - 14400
  local nSurplusTime = nFinishTime - nCurTime
  return Lib:TimeDesc2(nSurplusTime)
end

end

function BattlePass:UpdateSingleBattlePassTaskData(nTaskId, nTag, nCount)

    return
  end

function BattlePass:GetHaveAwardMinLevel()
  if not self.battlePassPlayerData then
    return nil
  end
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceiveTable()
  for nLv, _table in ipairs(tbAwardReceiveTable) do
    if nLv > nCurLv then
      break
    end
    for nAwardTag, bAwardReceive in pairs(_table) do
      if false == bAwardReceive and nAwardTag <= nBattlePassPlayerType then
        return nLv
      end
    end
  end
  return nCurLv
end

function BattlePass:ReqCompleteTask(nTaskId, nTag)
  local battlePassTaskData = self:GetBattlePassTaskData(nTaskId, nTag)
  if nil == battlePassTaskData then
    return
  end
  local tbBattlePassTaskCfgData = self:GetBattlePassTaskCfgData(nTaskId, nTag)
  if battlePassTaskData.bIsCompleted == false and battlePassTaskData.nCurCount >= tbBattlePassTaskCfgData.nFinishCount then
    RemoteServer.OnBattlePassRequest(BattlePass.Protocol.C2SReqCompleteTask, nTaskId, nTag)
  end
end

function BattlePass:ReqQuickCompleteTask(nTaskId, nTag)
  local nHaveGold = me.GetMoney("Gold")
  local tbBattlePassTaskCfgData = self:GetBattlePassTaskCfgData(nTaskId, nTag)
  if nTag == BattlePass.TaskTag.Legend then
    me.CenterMsg(i18n.EasyGet("\230\173\164\228\187\187\229\138\161\230\151\160\230\179\149\229\191\171\233\128\159\229\174\140\230\136\144"))
    return
  end
  if nHaveGold >= tbBattlePassTaskCfgData.nCostGold then
    RemoteServer.OnBattlePassRequest(BattlePass.Protocol.C2SReqQuickCompleteTask, nTaskId, nTag)
  else
    me.CenterMsg(i18n.EasyGet("\230\130\168\231\154\132\233\135\145\229\184\129\228\184\141\232\182\179"))
  end
end

function BattlePass:ReqReceiveAward(nLv, nAwardTag)
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceive(nLv)
  local bIsHaveReceive = tbAwardReceiveTable[nAwardTag]
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  if nLv <= nCurLv and false == bIsHaveReceive and nAwardTag <= nBattlePassPlayerType then
    RemoteServer.OnBattlePassRequest(BattlePass.Protocol.C2SReqReceiveAward, nLv, nAwardTag)
  end
  if nAwardTag > nBattlePassPlayerType then
    me.CenterMsg(i18n.EasyGet("\230\156\170\232\167\163\233\148\129"))
  end
end

function BattlePass:ReqReceiveAllAward()
  local tbAllAwardList = {}
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceiveTable()
  for nLv, _table in ipairs(tbAwardReceiveTable) do
    if nLv > nCurLv then
      break
    end
    for nAwardTag, bAwardReceive in pairs(_table) do
      if false == bAwardReceive and nAwardTag <= nBattlePassPlayerType then
        local allAwardData = {}
        allAwardData.nBattlePassLv = nLv
        allAwardData.nAwardTag = nAwardTag
        table.insert(tbAllAwardList, allAwardData)
      end
    end
  end
  if #tbAllAwardList > 0 then
    RemoteServer.OnBattlePassRequest(BattlePass.Protocol.C2SReqReceiveAllAward)
  end
end

function BattlePass:ReqCompleteAllTask()
  local tbAllTaskList = {}
  for nTaskId, tbCfg in pairs(self.tbBattlePassTaskCfg) do
    for nTag, tbBattlePassTaskCfgData in pairs(tbCfg) do
      local battlePassTaskData = self:GetBattlePassTaskData(nTaskId, nTag)
      if battlePassTaskData and battlePassTaskData.nCurCount >= tbBattlePassTaskCfgData.nFinishCount and battlePassTaskData.bIsCompleted == false then
        local battlePassTaskInfoData = {}
        battlePassTaskInfoData.nTaskId = nTaskId
        battlePassTaskInfoData.nTag = nTag
        table.insert(tbAllTaskList, battlePassTaskInfoData)
      end
    end
  end
  if #tbAllTaskList > 0 then
    RemoteServer.OnBattlePassRequest(BattlePass.Protocol.C2SReqCompleteAllTask)
  end
end

function BattlePass:RefreshBattlePassAwardRedPoint()
  if BattlePassNew.bIsNewServer == true then
    return
  end
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceiveTable()
  if nil ~= tbAwardReceiveTable then
    for nLv, _table in ipairs(tbAwardReceiveTable) do
      if nLv > nCurLv then
        break
      end
      for nAwardTag, bAwardReceive in pairs(_table) do
        if false == bAwardReceive and nAwardTag <= nBattlePassPlayerType then
          RedPointMgr:LightRedPoint(RedPointMgr.EVENT.BattlePassAward)
          return
        end
      end
    end
  end
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.BattlePassAward)
end

function BattlePass:RefreshBattlePassTaskRedPoint()
  if BattlePassNew.bIsNewServer == true then
    return
  end
  local tbRedPointState = {}
  for i, v in pairs(BattlePass.TaskClass) do
    tbRedPointState[v] = false
  end
  for nTaskId, tbCfg in pairs(self.tbBattlePassTaskCfg) do
    for nTag, tbBattlePassTaskCfgData in pairs(tbCfg) do
      local battlePassTaskData = self:GetBattlePassTaskData(nTaskId, nTag)
      if battlePassTaskData and battlePassTaskData.nCurCount >= tbBattlePassTaskCfgData.nFinishCount and false == battlePassTaskData.bIsCompleted then
        tbRedPointState[tbBattlePassTaskCfgData.nTaskClass] = true
      end
    end
  end
  for i, v in pairs(tbRedPointState) do
    if true == v then
      RedPointMgr:LightRedPoint(BattlePass.RedPointMapping[i])
    else
      RedPointMgr:FadeRedPoint(BattlePass.RedPointMapping[i])
    end
  end
end

function BattlePass:GetBattlePassTaskClassList(nTaskClass)
  if 1 == BattlePass.bIsNeedUpdateBattlePassTask then
    self.tbBattlePassTask = {}
    for i, nTaskClass in ipairs(BattlePass.TaskClass) do
      self.tbBattlePassTask[nTaskClass] = {}
    end
    for nTaskId, v in pairs(self.tbBattlePassTaskCfg) do
      for nTag, battlePassTaskCfgData in pairs(v) do
        if self:CheckBattlePassTaskTypeIsValid(nTaskId, nTag) then
          local szTaskType = battlePassTaskCfgData.szTaskType
          local battlePassTypeCfgData = BattlePass:GetBattlePassTypeCfgData(szTaskType)
          local szTimeFrameEvent = battlePassTypeCfgData.szTimeFrameEvent
          if 1 == GetTimeFrameState(szTimeFrameEvent) then
            table.insert(self.tbBattlePassTask[battlePassTaskCfgData.nTaskClass], battlePassTaskCfgData)
          end
        end
      end
    end
    BattlePass.bIsNeedUpdateBattlePassTask = 0
  end
  return self.tbBattlePassTask[nTaskClass]
end
