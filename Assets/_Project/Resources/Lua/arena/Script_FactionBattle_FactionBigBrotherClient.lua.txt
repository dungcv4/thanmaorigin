FactionBigBrother.tbBigBrotherData = FactionBigBrother.tbBigBrotherData or {}

function FactionBigBrother:OnPlayerLogin()
  self:SynStartTimeData()
end

function FactionBigBrother:SynStartTimeData()
  RemoteServer.RequestFactionBigBrother("SynFactionBigBrotherStartTimeData")
end

function FactionBigBrother:SynData()
  RemoteServer.RequestFactionBigBrother("SynFactionBigBrotherData")
end

function FactionBigBrother:OnSynData(tbBigBrotherData)
  self.tbBigBrotherData = tbBigBrotherData
  for nFaction, tbFactionData in pairs(self.tbBigBrotherData.tbAll) do
    if tbFactionData and next(tbFactionData) then
      self.tbBigBrotherData.tbAll[nFaction] = self:SortBigBrother(self.tbBigBrotherData.tbAll[nFaction])
    end
  end
  self.tbBigBrotherData.nVoteNumLeft = me.GetUserValue(GetGroupKey("FactionBigBrother"), self.KEY_VOTE)
  local tbWnd = Ui:GetWindow("UIDaShiXiong")
  if tbWnd then
    tbWnd:UpdateDaShiXiongData()
  end
end

function FactionBigBrother:GetBigBrotherWinnerInfoRsp(tbWinnerData)
  if Ui:WindowVisible("UINewInformation") then
    Ui:GetWindow("UINewInformation"):UpdateDaShiXiongPanel(tbWinnerData)
  end
end

function FactionBigBrother:IsHadWinnerInfo(bIsHadWinnerInfo)
  self.bIsHadWinnerInfo = bIsHadWinnerInfo
end

function FactionBigBrother:IsBigBrotherStarting()
  if not self.tbBigBrotherData or type(self.tbBigBrotherData) ~= "table" then
    return false
  end
  local nStartTime = self.tbBigBrotherData and self.tbBigBrotherData.nStartTime or 0
  return 0 ~= nStartTime and nStartTime <= GetTime() and GetTime() - nStartTime <= 86400
end

function FactionBigBrother:CheckVote()
  if not self:IsBigBrotherStarting() then
    return false, i18n.EasyGet("\232\175\132\233\128\137\230\180\187\229\138\168\229\183\178\231\187\143\231\187\147\230\157\159\239\188\129")
  end
  return self:CheckCommondVote(me)
end

function FactionBigBrother:SynSwitch(nStartTime)
  self.tbBigBrotherData.nStartTime = nStartTime
  self:UpdateRedPoint()
end

function FactionBigBrother:UpdateRedPoint()
end

function FactionBigBrother:SortBigBrother(tbBigBrother)
  local function SortBySession(a, b)
    return a.nSession < b.nSession
  end
  
  table.sort(tbBigBrother, SortBySession)
  return tbBigBrother
end

function FactionBigBrother:ManageBigBrother(tbBigBrother)
  local tbResultBigBrother = {}
  for nFaction = 1, Faction.MAX_FACTION_COUNT do
    if tbBigBrother[nFaction] then
      table.insert(tbResultBigBrother, tbBigBrother[nFaction])
    end
  end
  return tbResultBigBrother
end

function FactionBigBrother:OpenDaShiXiongUI()
  if FactionBigBrother:IsBigBrotherStarting() then
    Ui:OpenWindow("UIDaShiXiong")
  end
end

EventNotify:RegistNotify(EventNotify.emNOTIFY_PLAYER_LOGIN, FactionBigBrother.OnPlayerLogin, FactionBigBrother)
