function MultiPvP:ProtocolProcessTeamInfo(tbData)
  if tbData then
    self.m_tbTeamInfo = tbData.tbTeamData
    self.m_tbBatleInfo = tbData.tbBatleInfo
    self.nSeasonType = tbData.tbBatleInfo.nSeasonType
    if self.nSeasonType ~= self:GetCurSeasonType() then
      MultiPvP:SetSeasonType(self.nSeasonType)
    end
    self.nLeftCount = tbData.tbBatleInfo.nLeftCount
    if self.m_tbTeamInfo and self.m_tbTeamInfo[MPTeam.TEAM_ID] == me.GetUserValue(MultiPvP.SAVE_GROUP, MultiPvP.SAVE_TEAMID) then
      self.m_tbMyTeam = self.m_tbTeamInfo
    end
  else
    self.m_tbTeamInfo = nil
  end
  EventNotify.OnNotify(EventNotify.emNOTIFY_MULTIPVP_SYNC_TEAM)
end

function MultiPvP:OpenGuessUI(nId)
  local tbDataList = {}
  for _, tbMatchData in pairs(self.tbMatchList[self.PHASE_INIT]) do
    local tbTeamData = self.tbTeamData[tbMatchData.nTeamId]
    if tbTeamData then
      local tbData = {}
      tbData.nId = tbTeamData.nTeamId
      tbData.szName = tbTeamData.szTeamName
      tbData.nFight = 0
      tbData.tbMember = {}
      for key, tbMemberData in pairs(tbTeamData.tbMember) do
        table.insert(tbData.tbMember, {
          nId = tbMemberData.nPlayerId,
          szPlayerName = tbMemberData.szPlayerName,
          nLevel = tbMemberData.nLevel,
          nFaction = tbMemberData.nFaction,
          nPortrait = tbMemberData.nPortrait,
          nFight = tbMemberData.nFight
        })
        tbData.nFight = tbData.nFight + tbMemberData.nFight
      end
      table.insert(tbDataList, tbData)
    end
  end
  local tbSetting = {
    szTitle = i18n.EasyGet("\230\173\166\230\158\151\232\129\148\232\181\155\231\171\158\231\140\156"),
    szTitle1 = i18n.EasyGet("\229\133\165\229\155\180\230\136\152\233\152\159"),
    szTitle2 = i18n.EasyGet("\230\136\152\233\152\159\230\136\144\229\145\152"),
    szTitle3 = i18n.EasyGet("\230\136\152\233\152\159\230\128\187\230\136\152\229\138\155")
  }
  local tbCallBack = {
    MultiPvP.SendProtocolApplyGuess,
    MultiPvP
  }
  Ui:OpenWindow("UITeamBattleGuessing", tbSetting, tbDataList, tbCallBack, nId)
end

function MultiPvP:__ChangeAutoFightMode(nMode)
  nMode = nMode or MultiPvP.AUTO_FIGHT_MODE.TotalEnd
  if nMode == MultiPvP.AUTO_FIGHT_MODE.TotalEnd then
    if TeamMgr:IsFollowTeammate() then
      TeamMgr:StopFollowMember()
    end
    AFK:Stop()
  elseif nMode == MultiPvP.AUTO_FIGHT_MODE.FollowMember then
    LogErr("[MultiPvP:__ChangeAutoFightMode] MultiPvP.AUTO_FIGHT_MODE.FollowMember is not implemented")
  elseif nMode == MultiPvP.AUTO_FIGHT_MODE.OnlyAFK then
    if TeamMgr:IsFollowTeammate() then
      TeamMgr:StopFollowMember()
    end
    AFK:Start()
  end
end
