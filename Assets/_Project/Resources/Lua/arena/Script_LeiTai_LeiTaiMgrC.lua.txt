function LeiTai:OnLogin()
  self.tbAllRoom = nil
  
  self.nMySide = nil
  self.nEndTime = nil
end

function LeiTai:SyncAllRoomDataRequest(bShowUI)
  RemoteServer.RequestLeiTai("C2S_SyncAllRoomData", bShowUI)
end

function LeiTai:SyncAllRoomDataRespond(tbAllRoom, bShowUI)
  self.tbAllRoom = tbAllRoom
  if Ui:IsWindowLogicOpen("UILeiTai") then
    local tbWnd = Ui:GetWindow("UILeiTai")
    tbWnd:RefreshRoomList()
  elseif bShowUI then
    Ui:OpenWindow("UILeiTai")
  end
end

function LeiTai:SyncSelfRoomData(tbMyRoom, nSide, bShowUI)
  self.tbMyRoom = tbMyRoom
  self.nMySide = nSide
  self.nEndTime = nil
  if nil == tbMyRoom then
    if Ui:IsWindowLogicOpen("UILeiTaiRoom") then
      Ui:CloseWindow("UILeiTaiRoom")
    end
    if Ui:IsWindowLogicOpen("UILeiTai") then
      local tbWnd = Ui:GetWindow("UILeiTai")
      tbWnd:RefreshRoomList()
    end
    RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.LeiTaiNewCandidate)
    return
  else
    self.nMyRoomCountDown = tbMyRoom.nCountDown
    if tbMyRoom.nCountDown then
      self.nEndTime = GetTime() + tbMyRoom.nCountDown
    end
  end
  if Ui:IsWindowLogicOpen("UILeiTaiRoom") then
    local tbWnd = Ui:GetWindow("UILeiTaiRoom")
    tbWnd:RefreshUI()
  elseif bShowUI then
    Ui:OpenWindow("UILeiTaiRoom")
  end
  if Ui:IsWindowLogicOpen("UILeiTai") then
    local tbWnd = Ui:GetWindow("UILeiTai")
    tbWnd:RefreshRoomList()
  end
end

function LeiTai:CheckAllReady()
  if 0 == Lib:CountTB(self.tbMyRoom.tbHostId) then
    return false
  end
  for nPlayerId, bReady in pairs(self.tbMyRoom.tbPlayerReadyState) do
    if not bReady then
      return false
    end
  end
  return true
end

function LeiTai:RemindReady()
  RemoteServer.RequestLeiTai("C2S_RemindReady", self.tbMyRoom.nRoomId)
end

function LeiTai:GoToNpcAndOpenUI(szUIName)
  TeamMgr:StopFollowMember()
  AFK:Stop()
  local nPosX, nPosY, nWalkNearLength = Npc:GetNpcPos(self.Cfg.NPC_TEMPLATE_ID, self.Cfg.NPC_MAP_ID, true)
  
  local function fnCallback()
    local function __timer_Callback()
      Ui:OpenWindow(szUIName)
    end
    
    Timer:Register(5, __timer_Callback)
  end
  
  AutoPath:GotoAndCallEx(self.Cfg.NPC_MAP_ID, nPosX, nPosY, fnCallback, nWalkNearLength, nil, self.Cfg.NPC_TEMPLATE_ID)
  me.CenterMsg(i18n.EasyGet("\230\173\163\229\156\168\229\137\141\229\190\128\230\175\148\230\173\166\230\147\130\229\143\176"))
end

function LeiTai:CloseAllWindow()
  if Ui:IsWindowLogicOpen("UILeiTai") then
    Ui:CloseWindow("UILeiTai")
  end
  if Ui:IsWindowLogicOpen("UILeiTaiRoom") then
    Ui:CloseWindow("UILeiTaiRoom")
  end
  if Ui:IsWindowLogicOpen("UILeiTaiChallengeList") then
    Ui:CloseWindow("UILeiTaiChallengeList")
  end
end

function LeiTai:SyncFuBenInfo(bPrepare, bWin, tbFuBenInfo, nCloseCountDown)
  if Ui:IsWindowLogicOpen("UITeamBattleInfo") then
    Ui:GetWindow("UITeamBattleInfo"):Close()
  end
  Ui:OpenWindow("UILeiTaiResult", bPrepare, bWin, tbFuBenInfo, nCloseCountDown)
end

function LeiTai:UpdateFuBenTopInfo(bOpen, tbData, nMyCamp)
  if not bOpen then
    if Ui:IsWindowLogicOpen("UITeamBattleInfo") then
      Ui:CloseWindow("UITeamBattleInfo")
    end
    return
  end
  if not Ui:IsWindowLogicOpen("UITeamBattleInfo") then
    Ui:OpenWindow("UITeamBattleInfo", tbData, nMyCamp)
  else
    Ui:GetWindow("UITeamBattleInfo"):RefreshView(tbData, nMyCamp)
  end
end

  end
end
