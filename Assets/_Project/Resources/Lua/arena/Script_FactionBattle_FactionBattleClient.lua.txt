function FactionBattle:Join()
  local function fnEnter()
    RemoteServer.FactionBattleTryJoin()
  end
  
  local bRet, szInfo = ActivityCheck:CheckCanEnterActivityEx(me, FactionBattle.CheckCanEnter, fnEnter, false, false, true, {
    FactionBattle
  })
  if not bRet then
    me.CenterMsg(szInfo)
  end
end

function FactionBattle:OnSyncLeftInfo(szType, tbParam)
  local tbCfg = tbTipsCfg[szType].tbTipsCfg
  if not tbCfg then
    LogErr("tbTipsCfg Have Not ", szType)
    return
  end
  local tbTimerParam, szContent
  for i = 1, #tbCfg do
    if tbCfg[i].type == ParamType.Timer then
      tbTimerParam = tbTimerParam or {}
      table.insert(tbTimerParam, {
        tbParam[i],
        tbCfg[i].szTitle()
      })
    elseif tbCfg[i].type == ParamType.None then
      szContent = szContent or ""
      szContent = string.format([[
%s%s

]], szContent, tbCfg[i].szTitle())
    elseif tbCfg[i].type == ParamType.String then
      szContent = szContent or ""
      local nState = tonumber(tbParam[i])
      local szState = self.STATE_TRANS[nState] and i18n.Get(self.STATE_TRANS[nState][2]) or ""
      szContent = string.format([[
%s%s

]], szContent, string.format(tbCfg[i].szTitle(), szState))
    else
      szContent = szContent or ""
      szContent = string.format([[
%s%s

]], szContent, string.format(tbCfg[i].szTitle(), tbParam[i]))
    end
  end
  FubenUIController:OpenDefaultFuBenUI("FactionBattleHelp", tbTimerParam, {szContent})
  FactionBattle:SetLeaveShow(not tbTipsCfg[szType].bNotShowLeave, not tbTipsCfg[szType].bNotShowLeaveTip)
  if "FactionBattlePrepare" == szType then
    self.tb16thPlayer = nil
    self.bIsGlobal = false
    self:SetReportShow(false)
  end
end

function FactionBattle:OnSyncDamage(nRestTime, tbDamage)
  local nSelfDmg = 0
  local nEnemyDmg = 0
  for nPlayerID, nDmgCount in pairs(tbDamage) do
    if nPlayerID == me.dwID then
      nSelfDmg = nDmgCount
    else
      nEnemyDmg = nDmgCount
    end
  end
  local tbParam = {
    nSelfDmg,
    nEnemyDmg,
    nRestTime
  }
  self:OnSyncLeftInfoFSM("FactionBattleElimination", tbParam, false)
end

function FactionBattle:CheckCanEnter(pPlayer)
  if not pPlayer then
    return false, i18n.EasyGet("\230\156\170\231\159\165\233\148\153\232\175\175\239\188\140\232\175\183\233\135\141\232\175\149")
  end
  if pPlayer.nLevel < FactionBattle.Cfg.MIN_LEVEL then
    return false, i18n.EasyGet("\231\173\137\231\186\167\228\184\141\232\182\179{0}\231\186\167\239\188\140\228\184\141\232\131\189\229\143\130\229\138\160\230\175\148\232\181\155\227\128\130", FactionBattle.Cfg.MIN_LEVEL)
  end
  if pPlayer.nFightMode ~= Npc.FIGHT_MODE.emFightMode_None then
    return false, i18n.EasyGet("\228\184\141\229\156\168\229\174\137\229\133\168\229\140\186\228\184\141\232\131\189\229\143\130\229\138\160\230\173\164\230\180\187\229\138\168")
  end
  return true, ""
end

end

FactionBattle.bShowReportButton = false

function FactionBattle:IsNeedFuncScoreButton()
  return self.bShowReportButton, {
    Ui.OpenWindow,
    Ui,
    "UIFactionBattle"
  }
end

function FactionBattle:IsNeedExitFubenButton()
  return self.bShowLeaveButton, {
    function()
      if self.bShowLeaveButtonTip then
        me.MsgBox(i18n.Get(FactionBattle.Cfg.MSG_ExitMap, FactionBattle.Cfg.MAX_ATTEND_PLAYER), {
          {
            i18n.szOk,
            function()
              FubenUIController:RequireQuit()
            end
          },
          {
            i18n.szCancel
          }
        })
      else
        FubenUIController:RequireQuit()
      end
    end
  }
end

function FactionBattle:IsShowTipWhenQuit()
  return false
end

function FactionBattle:IsFactionBattleGlobal()
  return self.bIsGlobal == true
end

function FactionBattle:OnSync16thInfo(tb16thPlayer, bIsGlobal)
  self.bIsGlobal = bIsGlobal
  self.tb16thPlayer = tb16thPlayer
  self:SetReportShow(true)
end

function FactionBattle:OnClose()
  self.tb16thPlayer = nil
  self.bIsGlobal = false
  self:SetReportShow(false)
end

function FactionBattle:SetReportShow(bCanShow)
  self.bShowReportButton = bCanShow
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_REPORT_BUTTON)
end

function FactionBattle:SetLeaveShow(bCanShow, bCanShowTip)
  self.bShowLeaveButton = bCanShow
  self.bShowLeaveButtonTip = bCanShowTip
  EventNotify.OnNotify(EventNotify.emNOTIFY_CHANGE_LEAVE_BUTTON)
end

function FactionBattle:OnSyncWinnerData(tbData, nReqType)
  if Lib:CountTB(tbData) > 0 then
    self.bIsHadWinnerInfo = true
  end
  if 1 == nReqType then
    Ui:OpenWindow("UIFactionBattleKing", tbData)
  elseif 2 == nReqType and Ui:WindowVisible("UINewInformation") then
    Ui:GetWindow("UINewInformation"):UpdateFactionBattlePanel(tbData)
  end
end
