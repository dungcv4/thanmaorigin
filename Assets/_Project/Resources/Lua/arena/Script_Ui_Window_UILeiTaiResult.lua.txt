local tbWnd = Ui:CreateWindow("UILeiTaiResult")
local tbControls = {
  btnOk = "imgBG/btnOk",
  objReady = "imgBG/Ready",
  objVictory = "imgBG/Victory",
  objDefeat = "imgBG/Defeat",
  objHostTemplate = "imgBG/ShouLei/Template",
  objChallengerTemplate = "imgBG/GongLei/Template"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnOk = function(self)
    self:OnClickBtnOk()
  end
}

function tbWnd:OnCreate()
  self.svHost = Ui:NewScrollView(self, self.objHostTemplate, nil, true, true)
  self.svChallenger = Ui:NewScrollView(self, self.objChallengerTemplate, nil, true, true)
end

function tbWnd:OnOpen(bPrepare, bWin, tbFuBenInfo, nCloseCountDown)
  self.bPrepare = bPrepare
  self.bWin = bWin
  self.tbFuBenInfo = tbFuBenInfo
  self:RefreshUI()
  self:StartCloseTimer(nCloseCountDown)
end

function tbWnd:StartCloseTimer(nCloseCountDown)
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
  end
  self.nCloseTimer = Timer:Register(nCloseCountDown * Env.GAME_FPS, function()
    self:Close()
  end)
end

function tbWnd:OnClose()
  if self.nCloseTimer then
    Timer:Close(self.nCloseTimer)
  end
  self.nCloseTimer = nil
  self.bPrepare = nil
  self.bWin = nil
  self.tbFuBenInfo = nil
end

function tbWnd:RefreshUI()
  self.pPanel:SetActive(self.objReady, self.bPrepare)
  self.pPanel:SetActive(self.objVictory, not self.bPrepare and self.bWin)
  self.pPanel:SetActive(self.objDefeat, not self.bPrepare and not self.bWin)
  self:RefreshHostSV()
  self:RefreshChallengerSV()
end

function tbWnd:RefreshHostSV()
  self.svHost:Clear()
  for i, nPlayerId in ipairs(self.tbFuBenInfo.tbHostId) do
    self.svHost:SetElement(i, function(tbElement)
      local szElementPath = "imgBG/ShouLei/Element" .. i
      self:SetPlayerElement(nPlayerId, szElementPath, tbElement)
    end, false)
  end
end

function tbWnd:RefreshChallengerSV()
  self.svChallenger:Clear()
  for i, nPlayerId in ipairs(self.tbFuBenInfo.tbChallengerId) do
    self.svChallenger:SetElement(i, function(tbElement)
      local szElementPath = "imgBG/GongLei/Element" .. i
      self:SetPlayerElement(nPlayerId, szElementPath, tbElement)
    end, false)
  end
end

function tbWnd:SetPlayerElement(nPlayerId, szElementPath, tbElement)
  local tbPlayerInfo = self.tbFuBenInfo.tbPlayerInfoList[nPlayerId]
  self.pPanel:Label_SetText(szElementPath .. "/txtName", tbPlayerInfo.szName)
  self.pPanel:Label_SetText(szElementPath .. "/PortraitObj/Level/Text", tbPlayerInfo.nLevel)
  self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/imgFactionIcon", Faction:GetIcon(tbPlayerInfo.nFaction))
  self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbPlayerInfo.nPortrait))
  PlayerPortraitFrame:SetPortraitFrame(self.pPanel, szElementPath .. "/PortraitObj/PortraitFrame", tbPlayerInfo.nPortraitFrame)
  self.pPanel:SetActive(szElementPath .. "/Ready", self.bPrepare)
  self.pPanel:SetActive(szElementPath .. "/Result", not self.bPrepare)
  if self.bPrepare then
    self.pPanel:Label_SetText(szElementPath .. "/Ready/txtFightPower", i18n.EasyGet("\230\136\152\229\138\155 {0}", tbPlayerInfo.nFightPower))
  else
    self.pPanel:Label_SetText(szElementPath .. "/Result/txtKillCount", i18n.EasyGet("\229\135\187\232\180\165 {0}", tbPlayerInfo.nKillCount))
    self.pPanel:Label_SetText(szElementPath .. "/Result/txtDamage", i18n.EasyGet("\228\188\164\229\174\179\233\135\143 {0}", tbPlayerInfo.nDamageCount))
  end
end

function tbWnd:OnClickBtnOk()
  if self.bPrepare then
    self:Close()
  else
    self:Close()
  end
end
