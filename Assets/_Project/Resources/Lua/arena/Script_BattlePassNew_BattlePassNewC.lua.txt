function BattlePassNew:GetRemainingTime()
  local nCreateRealTime = GetServerCreateTime() - 14400
  nCreateRealTime = nCreateRealTime - Lib:GetLocalDayTime(nCreateRealTime)
  local nSeason = self.battlePassPlayerData.nSeason
  local nFinishTime = nCreateRealTime + nSeason * BattlePassNew.RestCycle * 3600 * 24
  local nCurTime = GetTime() - 14400
  local nSurplusTime = nFinishTime - nCurTime
  return Lib:TimeDesc2(nSurplusTime)
end

function BattlePassNew:OnClinetLogin(nBeginDay)
  self.battlePassPlayerData = {}
  local nCurLv = me.GetUserValue(BattlePassNew.GroupKey, BattlePassNew.CurLv)
  local nCurExp = me.GetUserValue(BattlePassNew.GroupKey, BattlePassNew.CurExp)
  local nCanUseQuickCompleteCount = me.GetUserValue(BattlePassNew.GroupKey, BattlePassNew.CanUseQuickCompleteCount)
  local nBattlePassType = me.GetUserValue(BattlePassNew.GroupKey, BattlePassNew.BattlePassType)
  local nSeason = me.GetUserValue(BattlePassNew.GroupKey, BattlePassNew.Season)
  self.battlePassPlayerData.nCurLv = nCurLv
  self.battlePassPlayerData.nCurExp = nCurExp
  self.battlePassPlayerData.nCanUseQuickCompleteCount = nCanUseQuickCompleteCount
  self.battlePassPlayerData.nBattlePassType = nBattlePassType
  self.battlePassPlayerData.nSeason = nSeason
  self:SynBattlePassBeginDay(nBeginDay)
  self.battlePassPlayerData.tbAwardReceiveTable = {}
  for nLv, battlePassLvCfgData in ipairs(self.tbBattlePassLvCfg) do
    for i, nAwardTag in pairs(BattlePassNew.AwardTag) do
      if self.battlePassPlayerData.tbAwardReceiveTable[nLv] == nil then
        self.battlePassPlayerData.tbAwardReceiveTable[nLv] = {}
      end
      if self.battlePassPlayerData.tbAwardReceiveTable[nLv][nAwardTag] == nil then
        self.battlePassPlayerData.tbAwardReceiveTable[nLv][nAwardTag] = {}
      end
      local bIsAwardReceive = self:GetAwardIsGet(me, nLv, nAwardTag)
      self.battlePassPlayerData.tbAwardReceiveTable[nLv][nAwardTag] = bIsAwardReceive
    end
  end
end

function BattlePassNew:InitBattlePassTaskTable()
  self.battlePassPlayerData.battlePassTaskTable = {}
  for nTaskId, tb in pairs(self.tbBattlePassServerData.tbTaskAndTagMap) do
    for nTag, nIndex in pairs(tb) do
      if self.battlePassPlayerData.battlePassTaskTable[nTaskId] == nil then
        self.battlePassPlayerData.battlePassTaskTable[nTaskId] = {}
      end
      if self.battlePassPlayerData.battlePassTaskTable[nTaskId][nTag] == nil then
        self.battlePassPlayerData.battlePassTaskTable[nTaskId][nTag] = {}
      end
      local tbBattlePassTaskData = {}
      local nCurCount = self:GetTaskCount(me, nTaskId)
      local bIsCompleted = self:GetTaskIsCompleted(me, nTaskId, nTag)
      tbBattlePassTaskData.nCurCount = nCurCount
      tbBattlePassTaskData.bIsCompleted = bIsCompleted
      self.battlePassPlayerData.battlePassTaskTable[nTaskId][nTag] = tbBattlePassTaskData
    end
  end
end

function BattlePassNew:GetHaveAwardMinLevel()
  if not self.battlePassPlayerData then
    return nil
  end
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceiveTable()
  for nLv, _table in ipairs(tbAwardReceiveTable) do
    if nLv > nCurLv then
      break
    end
    for nAwardTag, bAwardReceive in pairs(_table) do
      if false == bAwardReceive and nAwardTag <= nBattlePassPlayerType then
        return nLv
      end
    end
  end
  return nCurLv
end

function BattlePassNew:CheckCanReceiveAllAward()
  local tbAllAwardList = {}
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceiveTable()
  for nLv, _table in ipairs(tbAwardReceiveTable) do
    if nLv > nCurLv then
      break
    end
    for nAwardTag, bAwardReceive in pairs(_table) do
      if false == bAwardReceive and nAwardTag <= nBattlePassPlayerType then
        local allAwardData = {}
        allAwardData.nBattlePassLv = nLv
        allAwardData.nAwardTag = nAwardTag
        table.insert(tbAllAwardList, allAwardData)
      end
    end
  end
  if #tbAllAwardList > 0 then
    return true
  else
    return false
  end
end

function BattlePassNew:ReqReceiveAllAward()
  if self:CheckCanReceiveAllAward() == true then
    RemoteServer.OnBattlePassNewRequest(BattlePassNew.Protocol.C2SReqReceiveAllAward)
  end
end

function BattlePassNew:RefreshBattlePassAwardRedPoint()
  local nBattlePassPlayerType = self:GetBattlePassPlayerType()
  local nCurLv = self:GetBattlePassPlayerLv()
  local tbAwardReceiveTable = self:GetLvAwardReceiveTable()
  if nil ~= tbAwardReceiveTable then
    for nLv, _table in ipairs(tbAwardReceiveTable) do
      if nLv > nCurLv then
        break
      end
      for nAwardTag, bAwardReceive in pairs(_table) do
        if false == bAwardReceive and nAwardTag <= nBattlePassPlayerType then
          RedPointMgr:LightRedPoint(RedPointMgr.EVENT.BattlePassAward)
          return
        end
      end
    end
  end
  RedPointMgr:FadeRedPoint(RedPointMgr.EVENT.BattlePassAward)
end

function BattlePassNew:GetBattlePassTaskClassList(nTaskClass)
  self.tbBattlePassTask = {}
  for i, nTaskClass in ipairs(BattlePassNew.TaskClass) do
    self.tbBattlePassTask[nTaskClass] = {}
  end
  local nServerOpenDay = Lib:GetServerOpenDay_4AM()
  local szTimeFrameEnd
  for nTaskId, v in pairs(self.tbBattlePassTaskCfg) do
    for nTag, battlePassTaskCfgData in pairs(v) do
      if self:CheckBattlePassTaskTypeIsValid(nTaskId, nTag, nServerOpenDay) and BattlePassNew:GetBattlePassTaskData(nTaskId, nTag) ~= nil then
        local szTaskType = battlePassTaskCfgData.szTaskType
        local battlePassTypeCfgData = BattlePassNew:GetBattlePassTypeCfgData(szTaskType)
        local szTimeFrameEvent = battlePassTypeCfgData.szTimeFrameEvent
        szTimeFrameEnd = battlePassTypeCfgData.szTimeFrameEnd
        if 1 == GetTimeFrameState(szTimeFrameEvent) and (Lib:IsEmptyStr(szTimeFrameEnd) or 0 == GetTimeFrameState(szTimeFrameEnd)) then
          local tbBattlePassTaskData = self:GetBattlePassTaskData(nTaskId, nTag)
          local nPreTag = nTag - 1
          local nNextTag = nTag + 1
          if nil == self:GetBattlePassTaskCfgData(nTaskId, nPreTag) then
            if nil ~= tbBattlePassTaskData and tbBattlePassTaskData.bIsCompleted == false then
              table.insert(self.tbBattlePassTask[battlePassTaskCfgData.nTaskClass], battlePassTaskCfgData)
              break
            end
          else
            local tbPreBattlePassTaskData = self:GetBattlePassTaskData(nTaskId, nPreTag)
            if nil ~= tbPreBattlePassTaskData and tbPreBattlePassTaskData.bIsCompleted == true and tbBattlePassTaskData.bIsCompleted == false then
              table.insert(self.tbBattlePassTask[battlePassTaskCfgData.nTaskClass], battlePassTaskCfgData)
              break
            end
          end
          if nil == self:GetBattlePassTaskCfgData(nTaskId, nNextTag) then
            table.insert(self.tbBattlePassTask[battlePassTaskCfgData.nTaskClass], battlePassTaskCfgData)
            break
          end
        end
      end
    end
  end
  return self.tbBattlePassTask[nTaskClass]
end
