local tbWnd = Ui:CreateWindow("UILeiTaiNote")
local tbControls = {
  btnClose = "node/imgBG/btnClose",
  objTemplate = "node/imgBG/Content/Scroll View/Viewport/Content/Template",
  txtTips = "node/imgBG/Content/txtTips",
  objEmpty = "node/imgBG/Content/empty"
}
tbWnd.tbControls = tbControls
tbWnd.tbOnClick = {
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.svNoteList = Ui:NewScrollView(self, self.objTemplate, nil, true, true)
end

function tbWnd:OnOpen()
  self:RefreshNoteList()
end

function tbWnd:OnClose()
end

function tbWnd:RefreshNoteList()
  self.pPanel:Label_SetText(self.txtTips, i18n.EasyGet("\230\173\164\229\164\132\232\174\176\229\189\149\230\156\172\230\156\141\230\175\148\230\173\166\230\147\130\229\143\176\232\131\156\232\180\159\231\187\147\230\158\156\239\188\140\228\187\133\228\191\157\229\173\152\230\156\128\232\191\145{0}\229\156\186\230\175\148\230\173\166\230\136\152\231\187\169", LeiTai.Cfg.MAX_NOTE_COUNT))
  self.pPanel:SetActive(self.objEmpty, 0 == Lib:CountTB(LeiTai.tbNoteList))
  self.svNoteList:Clear()
  for i, tbNote in ipairs(LeiTai.tbNoteList) do
    self.svNoteList:SetElement(i, function(tbElement)
      local szElementPath = "node/imgBG/Content/Scroll View/Viewport/Content/Element" .. i
      local szType = tbNote.nRoomType == LeiTai.tbRoomType.Single and i18n.EasyGet("\229\141\149\228\186\186\230\147\130\229\143\176") or i18n.EasyGet("\233\152\159\228\188\141\230\147\130\229\143\176")
      self.pPanel:Label_SetText(szElementPath .. "/txtLeiTaiType", szType)
      self.pPanel:Label_SetText(szElementPath .. "/txtPlayerName", tbNote.tbHostCaptainInfo.szName)
      self.pPanel:Label_SetText(szElementPath .. "/PortraitObj/Level/Text", tbNote.tbHostCaptainInfo.nLevel)
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/imgFactionIcon", Faction:GetIcon(tbNote.tbHostCaptainInfo.nFaction))
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj/Portrait/Image", PlayerPortrait:GetIcon(tbNote.tbHostCaptainInfo.nPortrait))
      PlayerPortraitFrame:SetPortraitFrame(self.pPanel, szElementPath .. "/PortraitObj/PortraitFrame", tbNote.tbHostCaptainInfo.nPortraitFrame)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/PortraitObj", self.PopUpMenu, self, tbNote.tbHostCaptainInfo, tbElement)
      self.pPanel:Label_SetText(szElementPath .. "/txtPlayerName2", tbNote.tbChallengerCaptainInfo.szName)
      self.pPanel:Label_SetText(szElementPath .. "/PortraitObj2/Level/Text", tbNote.tbChallengerCaptainInfo.nLevel)
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj2/imgFactionIcon", Faction:GetIcon(tbNote.tbChallengerCaptainInfo.nFaction))
      self.pPanel:Sprite_SetSprite(szElementPath .. "/PortraitObj2/Portrait/Image", PlayerPortrait:GetIcon(tbNote.tbChallengerCaptainInfo.nPortrait))
      PlayerPortraitFrame:SetPortraitFrame(self.pPanel, szElementPath .. "/PortraitObj2/PortraitFrame", tbNote.tbChallengerCaptainInfo.nPortraitFrame)
      Ui:Button_BindEvent(self.pPanel, szElementPath .. "/PortraitObj2", self.PopUpMenu, self, tbNote.tbChallengerCaptainInfo, tbElement)
      self.pPanel:Label_SetText(szElementPath .. "/txtTime1", Lib:GetTimeStr(tbNote.nTime))
      self.pPanel:Label_SetText(szElementPath .. "/txtTime1/txtTime2", Lib:GetTimeStr5(tbNote.nTime))
      local szResult = tbNote.bWin and i18n.EasyGet("<color=#54ab4d>\229\174\136\230\147\130\230\136\144\229\138\159</color>") or i18n.EasyGet("<color=#c50505>\229\174\136\230\147\130\229\164\177\232\180\165</color>")
      self.pPanel:Label_SetText(szElementPath .. "/txtPlayerResult", szResult)
    end, false)
  end
end

function tbWnd:PopUpMenu(tbPlayerInfo, tbElement)
  if tbPlayerInfo.dwID == me.dwID then
    return
  end
  local tbData = {
    dwRoleId = tbPlayerInfo.dwID,
    szName = tbPlayerInfo.szName,
    nPortrait = tbPlayerInfo.nPortrait,
    nFaction = tbPlayerInfo.nFaction,
    nLevel = tbPlayerInfo.nLevel,
    dwKinId = tbPlayerInfo.dwKinId,
    szKinOrTongName = tbPlayerInfo.szKinOrTongName
  }
  Ui:OpenWindow("UIRightPopUpMenu", "leiTai", tbData, tbElement, 420, -200)
end
