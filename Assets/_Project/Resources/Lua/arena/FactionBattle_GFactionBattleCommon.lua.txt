function FactionBattle:CheckQualification(pPlayer)
  if not pPlayer then
    return false
  end
  if 1 ~= GetTimeFrameState("OpenLevel69") then
    return false
  end
  local now = Lib:GetLocalDate()
  local curMonth = now.month
  local curYear = now.year
  local curDay = now.day
  local curMonthFirstDay = os.time({
    year = curYear,
    month = curMonth,
    day = 1
  })
  local curTime = os.time({
    year = curYear,
    month = curMonth,
    day = curDay
  })
  local curMonthFirsThursday = curMonthFirstDay + (11 - os.date("%w", curMonthFirstDay)) % 7 * 86400
  if Lib:GetTimeNum(curTime) ~= Lib:GetTimeNum(curMonthFirsThursday) then
    return false
  end
  local nQualityTime = pPlayer.GetUserValue(GetGroupKey("FactionBattle"), FactionBattle.Def.KUOFUQUALIFICATIONID)
  return nQualityTime == Lib:GetTimeNum(curTime)
end
