Require("CommonScript/GlobalDomainBattle/GlobalDomainBattleCfg.lua")
GlobalDomainBattle.IS_SYS_OPEN = true
GlobalDomainBattle.GAME_STATE = {
  CLOSE = 0,
  CLAIM = 1,
  PREPARE = 2,
  FIGHT = 3
}
GlobalDomainBattle.PROGRESS_TIME = 3
GlobalDomainBattle.tbClaimList = GlobalDomainBattle.tbClaimList or {}
GlobalDomainBattle.tbOccupyList = GlobalDomainBattle.tbOccupyList or {}
GlobalDomainBattle.tbLastOccupyList = GlobalDomainBattle.tbLastOccupyList or {}
GlobalDomainBattle.tbMonsterFlagList = GlobalDomainBattle.tbMonsterFlagList or {}
GlobalDomainBattle.nBaseVersion = GlobalDomainBattle.nBaseVersion or 0

function GlobalDomainBattle:LoadComExternSetting()
  for _, tbSetting in pairs(self.MAP_FIGHT) do
    local tbPosList = Lib:LoadTabFileEx(tbSetting.POS_FILE, "sdd", nil, {
      "Type",
      "PosX",
      "PosY"
    })
    local tbGrpList = {
      FRONT = {}
    }
    for _, tbPos in ipairs(tbPosList) do
      local szFrontIdx = string.match(tbPos.Type, "FRONT_(%d)")
      if szFrontIdx then
        tbGrpList.FRONT[tonumber(szFrontIdx)] = {
          tbPos.PosX,
          tbPos.PosY
        }
      end
    end
    tbSetting.POS = tbGrpList
  end
end

GlobalDomainBattle:LoadComExternSetting()
