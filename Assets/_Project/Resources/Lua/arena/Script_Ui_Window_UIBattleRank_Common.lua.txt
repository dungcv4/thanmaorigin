local tbWnd = Ui:CreateWindow("UIBattleRank_Common")
local tbControls = {
  txtTitle = "node/imgBG/txtTitle",
  btnClose = "node/imgBG/btnClose",
  PlayerListTitle = "node/imgBG/Content/PlayerListTitle",
  RankTemplate = "node/imgBG/Content/imgList/Viewport/List/Template",
  myInfo = "node/imgBG/Content/myInfo",
  btnLeave = "node/imgBG/Content/btnLeave"
}
tbWnd.tbControls = tbControls

function tbWnd:NotifyEvents()
  return {
    {
      EventNotify.emNOTIFY_SYNC_RANKBOARD_DATA,
      self._UpdateMyRankInfo
    }
  }
end

local MAX_COL_NUM = 7
local tbTitlePath = {
  [1] = "txtRank",
  [2] = "txtFaction",
  [3] = "txtName",
  [4] = "txtCamp",
  [5] = "txtScore",
  [6] = "txtKill",
  [7] = "txtCombo"
}
local tbAllRankPath = {
  [1] = "txtRank",
  [2] = {
    "txtFaction/imgFaction",
    "txtFaction"
  },
  [3] = "txtName",
  [4] = "txtCamp",
  [5] = "txtScore",
  [6] = "txtKill",
  [7] = "txtCombo"
}
local tbMyRankPath = {
  [1] = "txtRank",
  [2] = {"imgFaction", "txtFaction"},
  [3] = "txtName",
  [4] = "txtCamp",
  [5] = "txtScore",
  [6] = "txtKill",
  [7] = "txtCombo"
}
tbWnd.tbOnClick = {
  btnLeave = function(self)
    if not Lib:IsEmptyTB(self.tbBtnCallBack) then
      Lib:CallBack(self.tbBtnCallBack)
    end
  end,
  btnClose = function(self)
    self:Close()
  end
}

function tbWnd:OnCreate()
  self.tbRankView = Ui:NewLoopScrollView(self, tbControls.RankTemplate)
end

function tbWnd:OnOpen(szBtnText, tbBtnCallBack, tbTitleList, tbRankDataList, tbMyRankInfo)
  self.tbTitleList = tbTitleList or {}
  self.tbRankDataList = tbRankDataList or {}
  self.tbMyRankInfo = tbMyRankInfo or {}
  self.szBtnText = szBtnText or i18n.EasyGet("\231\166\187\229\188\128\230\136\152\229\156\186")
  self.tbBtnCallBack = tbBtnCallBack or {}
  Lib:CallBack({
    self._UpdateRankTitle,
    self
  })
  Lib:CallBack({
    self._UpdateAllRank,
    self
  })
  Lib:CallBack({
    self._UpdateMyRankInfo,
    self
  })
end

function tbWnd:_UpdateRankTitle()
  for i = 1, MAX_COL_NUM do
    local szTitle = self.tbTitleList[i]
    local szPath = string.format("%s/%s", tbControls.PlayerListTitle, tbTitlePath[i])
    if false == szTitle or "string" == type(szTitle) and Lib:IsEmptyStr(szTitle) then
      self.pPanel:SetActive(szPath, false)
    else
      self.pPanel:Label_SetText(szPath, szTitle)
      self.pPanel:SetActive(szPath, true)
    end
  end
end

function tbWnd:_UpdateAllRank()
  self.tbRankView:SetElement(#self.tbRankDataList, function(tbLoopScroll, nIndex)
    local tbElement = tbLoopScroll:GetElement(nIndex)
    self:_SetRankElement(tbElement, self.tbRankDataList[nIndex])
  end)
end

function tbWnd:_SetRankElement(tbElement, tbRankInfo)
  for i = 1, MAX_COL_NUM do
    local stContent = tbRankInfo[i]
    if 2 == i then
      if not stContent or Lib:IsEmptyTB(stContent or {}) then
        tbElement.pPanel:SetActive(tbAllRankPath[i][1], false)
        tbElement.pPanel:SetActive(tbAllRankPath[i][2], false)
      else
        tbElement.pPanel:Sprite_SetSprite(tbAllRankPath[i][1], stContent.szImg)
        tbElement.pPanel:Label_SetText(tbAllRankPath[i][2], stContent.szContent)
        tbElement.pPanel:SetActive(tbAllRankPath[i][1], true)
        tbElement.pPanel:SetActive(tbAllRankPath[i][2], true)
      end
    elseif not stContent then
      tbElement.pPanel:SetActive(tbAllRankPath[i], false)
      tbElement.pPanel:SetActive(tbAllRankPath[i], false)
    else
      tbElement.pPanel:Label_SetText(tbAllRankPath[i], stContent or "")
      tbElement.pPanel:SetActive(tbAllRankPath[i], true)
      tbElement.pPanel:SetActive(tbAllRankPath[i], true)
    end
  end
end

function tbWnd:_UpdateMyRankInfo()
  for i = 1, MAX_COL_NUM do
    if 2 == i then
      local szImgPath = string.format("%s/%s", tbControls.myInfo, tbMyRankPath[i][1])
      local szContentPath = string.format("%s/%s", tbControls.myInfo, tbMyRankPath[i][2])
      if type(self.tbMyRankInfo[i]) == "table" and not Lib:IsEmptyTB(self.tbMyRankInfo[i]) then
        self.pPanel:Sprite_SetSprite(szImgPath, self.tbMyRankInfo[i].szImg)
        self.pPanel:Label_SetText(szContentPath, self.tbMyRankInfo[i].szContent)
        self.pPanel:SetActive(szImgPath, true)
        self.pPanel:SetActive(szContentPath, true)
      else
        self.pPanel:SetActive(szImgPath, false)
        self.pPanel:SetActive(szContentPath, false)
      end
    else
      local szPath = ""
      szPath = string.format("%s/%s", tbControls.myInfo, tbMyRankPath[i])
      if self.tbMyRankInfo[i] then
        self.pPanel:SetActive(szPath, true)
        self.pPanel:Label_SetText(szPath, self.tbMyRankInfo[i])
      else
        self.pPanel:SetActive(szPath, false)
      end
    end
  end
  self.pPanel:Button_SetText(tbControls.btnLeave, self.szBtnText, "Text")
end
